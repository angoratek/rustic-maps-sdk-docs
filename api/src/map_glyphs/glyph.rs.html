1//! The Mapbox glyph PBF reader and its typed model.
2//!
3//! One fetched glyph range is the `glyphs` message of the Mapbox glyph
4//! protocol: a list of font stacks, each carrying its name, its range label
5//! (`"0-255"`, …) and its glyphs. Each glyph carries a Unicode codepoint,
6//! its ink-box metrics and an optional signed-distance-field bitmap whose
7//! border is
8//! [`GLYPH_SIGNED_DISTANCE_FIELD_BORDER_PIXELS`] pixels wide on every side. The reader reuses
9//! the protobuf wire reader of `map-vector-tiles`; every malformed path is a
10//! typed [`GlyphError`], never a panic.
11//!
12//! The parser accepts whichever codepoints a producer serves. Text shaping —
13//! bidirectional ordering, ligatures and complex-script cluster layout — is
14//! explicitly out of scope; the label pipeline handles Latin text first and
15//! nothing here guesses at a shaping algorithm.
16
17use crate::error::GlyphError;
18use map_vector_tiles::VectorTileError;
19use map_vector_tiles::wire::{
20    WIRE_TYPE_LENGTH_DELIMITED, WIRE_TYPE_VARINT, WireReader, zigzag_decode,
21};

23/// The size of the signed-distance-field border the glyph protocol puts
24/// around every glyph's ink box, in pixels.
25///
26/// The bitmap of a glyph with a `width` × `height` ink box is therefore
27/// `(width + 2 × GLYPH_SIGNED_DISTANCE_FIELD_BORDER_PIXELS)` ×
28/// `(height + 2 × GLYPH_SIGNED_DISTANCE_FIELD_BORDER_PIXELS)` pixels, one `R8Unorm` byte each.
29pub const GLYPH_SIGNED_DISTANCE_FIELD_BORDER_PIXELS: u32 = 3;
30
31/// The number of codepoints one glyph range covers. Range labels name a
32/// 256-codepoint block (`"0-255"`, `"256-511"`, …).
33pub const GLYPH_RANGE_CODEPOINT_COUNT: u32 = 256;
34
35/// The bit mask selecting the first codepoint of the 256-codepoint block a
36/// codepoint belongs to.
37const GLYPH_RANGE_FIRST_CODEPOINT_MASK: u32 = GLYPH_RANGE_CODEPOINT_COUNT - 1;
38
39/// One 256-codepoint block of glyphs, the unit a glyph URL requests with its
40/// `{range}` placeholder.
41#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
42pub struct GlyphRange {
43    /// The range's first codepoint (inclusive).
44    pub first_codepoint: u32,
45    /// The range's last codepoint (inclusive).
46    pub last_codepoint: u32,
47}
48
49impl GlyphRange {
50    /// Basic Latin plus Latin-1 Supplement (`U+0000`–`U+00FF`).
51    pub const BASIC_LATIN: Self = Self::new(0x0000, 0x00FF);
52    /// Latin Extended-A (`U+0100`–`U+017F`).
53    pub const LATIN_EXTENDED_A: Self = Self::new(0x0100, 0x017F);
54
55    /// Builds an ordered codepoint span. Callers supply a first codepoint at
56    /// or below the last; [`GlyphRange::containing`] is the checked builder
57    /// for an arbitrary codepoint.
58    pub const fn new(first_codepoint: u32, last_codepoint: u32) -> Self {
59        Self {
60            first_codepoint,
61            last_codepoint,
62        }
63    }
64
65    /// The 256-codepoint block that contains `codepoint`.
66    pub const fn containing(codepoint: u32) -> Self {
67        let first_codepoint = codepoint & !GLYPH_RANGE_FIRST_CODEPOINT_MASK;
68        Self::new(
69            first_codepoint,
70            first_codepoint | GLYPH_RANGE_FIRST_CODEPOINT_MASK,
71        )
72    }
73
74    /// Whether `codepoint` falls inside this range.
75    pub const fn contains(&self, codepoint: u32) -> bool {
76        codepoint >= self.first_codepoint && codepoint <= self.last_codepoint
77    }
78
79    /// Whether `codepoint` belongs to the Latin ranges the atlas packs first:
80    /// Basic Latin with Latin-1 Supplement, or Latin Extended-A.
81    pub const fn is_latin_codepoint(codepoint: u32) -> bool {
82        Self::BASIC_LATIN.contains(codepoint) || Self::LATIN_EXTENDED_A.contains(codepoint)
83    }
84
85    /// The range's URL segment (`"0-255"`, `"256-511"`, …), the value of the
86    /// glyph URL template's `{range}` placeholder.
87    pub fn url_segment(&self) -> String {
88        format!("{}-{}", self.first_codepoint, self.last_codepoint)
89    }
90}

92/// One decoded glyph: its codepoint, ink-box metrics and optional
93/// signed-distance-field bitmap.
94#[derive(Debug, Clone, PartialEq, Eq, Default)]
95pub struct Glyph {
96    /// The glyph's Unicode codepoint (the protocol's `id` field).
97    pub codepoint: u32,
98    /// The signed-distance-field bitmap, one byte per pixel, row-major. Empty
99    /// when the producer omits the bitmap (as it does for a space).
100    pub bitmap: Vec<u8>,
101    /// The ink-box width in pixels, excluding the signed-distance-field border.
102    pub width: u32,
103    /// The ink-box height in pixels, excluding the signed-distance-field border.
104    pub height: u32,
105    /// The horizontal bearing from the pen origin to the ink box, in pixels.
106    pub left: i32,
107    /// The vertical bearing from the baseline to the ink box, in pixels.
108    pub top: i32,
109    /// The horizontal advance to the next pen position, in pixels.
110    pub advance: u32,
111}
112
113impl Glyph {
114    /// The padded signed-distance-field bitmap width in pixels: the ink-box width plus the
115    /// border on both sides.
116    pub fn bitmap_width_pixels(&self) -> u64 {
117        u64::from(self.width) + 2 * u64::from(GLYPH_SIGNED_DISTANCE_FIELD_BORDER_PIXELS)
118    }
119
120    /// The padded signed-distance-field bitmap height in pixels: the ink-box height plus the
121    /// border on both sides.
122    pub fn bitmap_height_pixels(&self) -> u64 {
123        u64::from(self.height) + 2 * u64::from(GLYPH_SIGNED_DISTANCE_FIELD_BORDER_PIXELS)
124    }
125
126    /// The byte length a present signed-distance-field bitmap must have, or `None` when the
127    /// padded dimensions overflow a 64-bit byte count.
128    pub fn expected_bitmap_byte_count(&self) -> Option<u64> {
129        self.bitmap_width_pixels()
130            .checked_mul(self.bitmap_height_pixels())
131    }
132
133    /// The bytes this glyph retains: its bitmap bytes plus the inline glyph
134    /// record. Allocator slack is deliberately not counted.
135    pub fn retained_byte_count(&self) -> usize {
136        self.bitmap.len() + std::mem::size_of::<Self>()
137    }
138}

140/// One decoded font stack: the font family list the glyphs were rendered
141/// from, the range label the payload covers and the decoded glyphs.
142#[derive(Debug, Clone, PartialEq, Eq, Default)]
143pub struct GlyphFontStack {
144    /// The font stack's name (a comma-separated family list).
145    pub name: String,
146    /// The range label the payload declares (`"0-255"`, …).
147    pub range: String,
148    /// The decoded glyphs, in payload order.
149    pub glyphs: Vec<Glyph>,
150}
151
152impl GlyphFontStack {
153    /// Finds a glyph by codepoint.
154    pub fn glyph(&self, codepoint: u32) -> Option<&Glyph> {
155        self.glyphs
156            .iter()
157            .find(|glyph| glyph.codepoint == codepoint)
158    }
159
160    /// The bytes this font stack retains: its name and range text, its glyph
161    /// records and their bitmaps, plus the inline stack record. Allocator
162    /// slack is deliberately not counted.
163    pub fn retained_byte_count(&self) -> usize {
164        std::mem::size_of::<Self>()
165            + self.name.len()
166            + self.range.len()
167            + self.glyphs.len() * std::mem::size_of::<Glyph>()
168            + self
169                .glyphs
170                .iter()
171                .map(|glyph| glyph.bitmap.len())
172                .sum::<usize>()
173    }
174}

176/// Decodes one fetched glyph range (the `glyphs` protobuf message).
177///
178/// A range response carries exactly one font stack; a payload with zero or
179/// several reports [`GlyphError::UnexpectedFontStackCount`]. Unknown fields
180/// are skipped for forward compatibility, and every known field is checked
181/// for its wire type before it is read.
182pub fn decode_glyph_font_stack(bytes: &[u8]) -> Result<GlyphFontStack, GlyphError> {
183    let mut reader = WireReader::new(bytes);
184    let mut font_stacks = Vec::new();
185    while !reader.at_end() {
186        let field_key = reader.read_field_key("glyph response")?;
187        match (field_key.field_number, field_key.wire_type) {
188            (1, WIRE_TYPE_LENGTH_DELIMITED) => {
189                let font_stack_bytes = reader.read_length_delimited("glyph response font stack")?;
190                font_stacks.push(decode_font_stack(font_stack_bytes)?);
191            }
192            (1, wire_type) => {
193                return Err(unsupported_wire_type(1, wire_type, "glyph response"));
194            }
195            _ => reader.skip_field(field_key, "glyph response")?,
196        }
197    }
198    if font_stacks.len() != 1 {
199        return Err(GlyphError::UnexpectedFontStackCount {
200            found: font_stacks.len(),
201        });
202    }
203    Ok(font_stacks.remove(0))
204}

206/// Decodes one `fontstack` message.
207fn decode_font_stack(bytes: &[u8]) -> Result<GlyphFontStack, GlyphError> {
208    let mut reader = WireReader::new(bytes);
209    let mut name = None;
210    let mut range = None;
211    let mut glyphs = Vec::new();
212    while !reader.at_end() {
213        let field_key = reader.read_field_key("font stack")?;
214        match (field_key.field_number, field_key.wire_type) {
215            (1, WIRE_TYPE_LENGTH_DELIMITED) => {
216                let name_bytes = reader.read_length_delimited("font stack name")?;
217                name = Some(decode_utf8(name_bytes, "font stack name")?);
218            }
219            (2, WIRE_TYPE_LENGTH_DELIMITED) => {
220                let range_bytes = reader.read_length_delimited("font stack range")?;
221                range = Some(decode_utf8(range_bytes, "font stack range")?);
222            }
223            (3, WIRE_TYPE_LENGTH_DELIMITED) => {
224                let glyph_bytes = reader.read_length_delimited("font stack glyph")?;
225                glyphs.push(decode_glyph(glyph_bytes)?);
226            }
227            (1..=3, wire_type) => {
228                return Err(unsupported_wire_type(
229                    field_key.field_number,
230                    wire_type,
231                    "font stack",
232                ));
233            }
234            _ => reader.skip_field(field_key, "font stack")?,
235        }
236    }
237    Ok(GlyphFontStack {
238        name: name.ok_or(GlyphError::MissingFontStackName)?,
239        range: range.ok_or(GlyphError::MissingFontStackRange)?,
240        glyphs,
241    })
242}

244/// Decodes one `glyph` message and validates its bitmap length.
245fn decode_glyph(bytes: &[u8]) -> Result<Glyph, GlyphError> {
246    let mut reader = WireReader::new(bytes);
247    let mut codepoint = None;
248    let mut bitmap = Vec::new();
249    let mut bitmap_was_present = false;
250    let mut width = None;
251    let mut height = None;
252    let mut left = None;
253    let mut top = None;
254    let mut advance = None;
255    while !reader.at_end() {
256        let field_key = reader.read_field_key("glyph")?;
257        match (field_key.field_number, field_key.wire_type) {
258            (1, WIRE_TYPE_VARINT) => {
259                codepoint = Some(read_unsigned_field(&mut reader, "id")?);
260            }
261            (2, WIRE_TYPE_LENGTH_DELIMITED) => {
262                let bitmap_bytes = reader.read_length_delimited("glyph bitmap")?;
263                bitmap = bitmap_bytes.to_vec();
264                bitmap_was_present = true;
265            }
266            (3, WIRE_TYPE_VARINT) => {
267                width = Some(read_unsigned_field(&mut reader, "width")?);
268            }
269            (4, WIRE_TYPE_VARINT) => {
270                height = Some(read_unsigned_field(&mut reader, "height")?);
271            }
272            (5, WIRE_TYPE_VARINT) => {
273                left = Some(read_signed_field(&mut reader, "left")?);
274            }
275            (6, WIRE_TYPE_VARINT) => {
276                top = Some(read_signed_field(&mut reader, "top")?);
277            }
278            (7, WIRE_TYPE_VARINT) => {
279                advance = Some(read_unsigned_field(&mut reader, "advance")?);
280            }
281            (1..=7, wire_type) => {
282                return Err(unsupported_wire_type(
283                    field_key.field_number,
284                    wire_type,
285                    "glyph",
286                ));
287            }
288            _ => reader.skip_field(field_key, "glyph")?,
289        }
290    }
291    let glyph = Glyph {
292        codepoint: codepoint.ok_or(GlyphError::MissingGlyphField { field: "id" })?,
293        bitmap,
294        width: width.ok_or(GlyphError::MissingGlyphField { field: "width" })?,
295        height: height.ok_or(GlyphError::MissingGlyphField { field: "height" })?,
296        left: left.ok_or(GlyphError::MissingGlyphField { field: "left" })?,
297        top: top.ok_or(GlyphError::MissingGlyphField { field: "top" })?,
298        advance: advance.ok_or(GlyphError::MissingGlyphField { field: "advance" })?,
299    };
300    if bitmap_was_present {
301        let expected_byte_count = glyph.expected_bitmap_byte_count().ok_or(
302            GlyphError::GlyphBitmapDimensionsOverflow {
303                codepoint: glyph.codepoint,
304                width: glyph.width,
305                height: glyph.height,
306            },
307        )?;
308        let actual_byte_count = glyph.bitmap.len() as u64;
309        if expected_byte_count != actual_byte_count {
310            return Err(GlyphError::GlyphBitmapLengthMismatch {
311                codepoint: glyph.codepoint,
312                width: glyph.width,
313                height: glyph.height,
314                expected_byte_count,
315                actual_byte_count,
316            });
317        }
318    }
319    Ok(glyph)
320}

322/// Reads a varint field that must fit into a `u32`.
323fn read_unsigned_field(
324    reader: &mut WireReader<'_>,
325    field: &'static str,
326) -> Result<u32, GlyphError> {
327    let value = reader.read_varint(field)?;
328    u32::try_from(value).map_err(|_| GlyphError::UnsignedValueOutOfRange { field, value })
329}
330
331/// Reads a ZigZag-encoded `sint32` field.
332fn read_signed_field(reader: &mut WireReader<'_>, field: &'static str) -> Result<i32, GlyphError> {
333    let value = reader.read_varint(field)?;
334    let signed_value = zigzag_decode(value);
335    i32::try_from(signed_value).map_err(|_| GlyphError::SignedValueOutOfRange {
336        field,
337        value: signed_value,
338    })
339}
340
341/// Decodes a UTF-8 string field.
342fn decode_utf8(bytes: &[u8], context: &'static str) -> Result<String, GlyphError> {
343    std::str::from_utf8(bytes)
344        .map(str::to_owned)
345        .map_err(|_| GlyphError::MalformedProtobuf {
346            source: VectorTileError::InvalidUtf8 { context },
347        })
348}
349
350/// Builds the typed wire-format refusal for a known field with an unexpected
351/// wire type.
352fn unsupported_wire_type(field_number: u32, wire_type: u64, context: &'static str) -> GlyphError {
353    GlyphError::MalformedProtobuf {
354        source: VectorTileError::UnsupportedWireType {
355            field_number,
356            wire_type,
357            context,
358        },
359    }

366    /// Appends the base-128 varint encoding of `value`.
367    fn append_varint(value: u64, bytes: &mut Vec<u8>) {
368        let mut rest = value;
369        loop {
370            let byte = (rest & 0x7f) as u8;
371            rest >>= 7;
372            if rest == 0 {
373                bytes.push(byte);
374                break;
375            }
376            bytes.push(byte | 0x80);
377        }
378    }
379
380    /// Appends a field key.
381    fn append_field_key(field_number: u32, wire_type: u64, bytes: &mut Vec<u8>) {
382        append_varint((u64::from(field_number) << 3) | wire_type, bytes);
383    }
384
385    /// Appends a varint field.
386    fn append_varint_field(field_number: u32, value: u64, bytes: &mut Vec<u8>) {
387        append_field_key(field_number, WIRE_TYPE_VARINT, bytes);
388        append_varint(value, bytes);
389    }
390
391    /// Appends a length-delimited field.
392    fn append_length_delimited_field(field_number: u32, payload: &[u8], bytes: &mut Vec<u8>) {
393        append_field_key(field_number, WIRE_TYPE_LENGTH_DELIMITED, bytes);
394        append_varint(payload.len() as u64, bytes);
395        bytes.extend_from_slice(payload);
396    }
397
398    /// The ZigZag encoding of a signed value, the mirror of the decoder.
399    fn zigzag_encode(value: i64) -> u64 {
400        ((value as u64) << 1) ^ ((value >> 63) as u64)
401    }
402
403    /// The signed-distance-field byte count for an ink box.
404    fn signed_distance_field_byte_count(width: u32, height: u32) -> usize {
405        ((width as usize) + 2 * GLYPH_SIGNED_DISTANCE_FIELD_BORDER_PIXELS as usize)
406            * ((height as usize) + 2 * GLYPH_SIGNED_DISTANCE_FIELD_BORDER_PIXELS as usize)
407    }
408
409    /// Encodes one glyph, omitting any field whose closure returns `None`.
410    fn encode_glyph(
411        codepoint: u32,
412        width: u32,
413        height: u32,
414        left: i32,
415        top: i32,
416        advance: u32,
417        bitmap: Option<&[u8]>,
418    ) -> Vec<u8> {
419        let mut bytes = Vec::new();
420        append_varint_field(1, u64::from(codepoint), &mut bytes);
421        if let Some(bitmap_bytes) = bitmap {
422            append_length_delimited_field(2, bitmap_bytes, &mut bytes);
423        }
424        append_varint_field(3, u64::from(width), &mut bytes);
425        append_varint_field(4, u64::from(height), &mut bytes);
426        append_varint_field(5, zigzag_encode(i64::from(left)), &mut bytes);
427        append_varint_field(6, zigzag_encode(i64::from(top)), &mut bytes);
428        append_varint_field(7, u64::from(advance), &mut bytes);
429        bytes
430    }
431
432    /// Encodes one font stack.
433    fn encode_font_stack(name: &str, range: &str, glyphs: &[Vec<u8>]) -> Vec<u8> {
434        let mut bytes = Vec::new();
435        append_length_delimited_field(1, name.as_bytes(), &mut bytes);
436        append_length_delimited_field(2, range.as_bytes(), &mut bytes);
437        for glyph in glyphs {
438            append_length_delimited_field(3, glyph, &mut bytes);
439        }
440        bytes
441    }
442
443    /// Encodes one glyph response around the supplied font stacks.
444    fn encode_glyph_response(font_stacks: &[Vec<u8>]) -> Vec<u8> {
445        let mut bytes = Vec::new();
446        for font_stack in font_stacks {
447            append_length_delimited_field(1, font_stack, &mut bytes);
448        }
449        bytes
450    }

455        let height = 5;
456        let bitmap = vec![0x7f; signed_distance_field_byte_count(width, height)];
