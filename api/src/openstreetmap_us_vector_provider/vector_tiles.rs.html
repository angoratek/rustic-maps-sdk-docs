53/// The imagery-caching ceiling this provider installs: **30 calendar days**,
54/// the common maximum under imagery-provider terms (PRD §6.7, owner ruling
55/// 2026-10-28) and the same ceiling the MapQuest raster path inherits. The
56/// usage policy separately requires honoring `Cache-Control` response
57/// headers; a live probe (2026-10-29) observed
58/// `Cache-Control: public, max-age=86400` — an 86400-second per-response
59/// tightening is a refinement the decoder slice may take, while this 30-day
60/// value remains the SDK's terms ceiling, enforced by this crate's lockstep
61/// test against `map_cache::ResponseCategory::VectorTiles`.
62pub const OSMUS_VECTOR_TILE_CACHE_CEILING_SECONDS: u64 = 30 * 24 * 60 * 60;

69/// Serves the Sourdough tileset of the OpenStreetMap US tileservice.
70pub struct OpenStreetMapUsVectorTileProvider {
71    http_client: OpenStreetMapUsTileHttpClient,
72    rate_limiter: TokenBucketRateLimiter,
73}

75impl OpenStreetMapUsVectorTileProvider {
76    /// Builds the provider from the given configuration.
77    pub fn new(configuration: OpenStreetMapUsVectorTileConfiguration) -> Result<Self, SDKError> {
78        Ok(Self {
79            http_client: OpenStreetMapUsTileHttpClient::new(
80                configuration.base_url,
81                configuration.user_agent_text,
82                configuration.maximum_retry_attempts,
83            )?,
84            rate_limiter: TokenBucketRateLimiter::new(
85                configuration.token_bucket_capacity,
86                configuration.refill_tokens_per_second,
87                configuration.maximum_wait_seconds,
88            )?,
89        })
90    }
91
92    /// Builds the provider with the default configuration's pacing against
93    /// an explicit base URL (the test seam; production code uses
94    /// [`OpenStreetMapUsVectorTileProvider::new`] with the defaults).
95    #[cfg(test)]
96    pub(crate) fn with_default_pacing(base_url: String) -> Self {
97        Self::new(OpenStreetMapUsVectorTileConfiguration {
98            base_url,
99            ..OpenStreetMapUsVectorTileConfiguration::default()
100        })
101        .expect("the default configuration is valid")
102    }
103
104    /// Request path of one tile, per the documented URL template
105    /// `[SOURDOUGH_TILE_PATH_ROOT]/{zoom}/{x}/{y}[.mvt]`.
106    fn tile_request_path(&self, tile_x: u32, tile_y: u32, zoom_level: u8) -> String {
107        format!(
108            "{SOURDOUGH_TILE_PATH_ROOT}/{zoom_level}/{tile_x}/{tile_y}.{SOURDOUGH_TILE_FILE_EXTENSION}"
109        )
110    }
111}

113#[async_trait]
114impl VectorTileProvider for OpenStreetMapUsVectorTileProvider {
115    fn source_descriptor(&self) -> VectorTileSourceDescriptor {
116        VectorTileSourceDescriptor {
117            source_name: SOURDOUGH_TILESET_SOURCE_NAME.to_string(),
118            attribution_text: OSMUS_SOURDOUGH_ATTRIBUTION_TEXT.to_string(),
119            minimum_zoom_level: SOURDOUGH_MINIMUM_ZOOM_LEVEL,
120            maximum_zoom_level: SOURDOUGH_MAXIMUM_ZOOM_LEVEL,
121        }
122    }
123
124    async fn fetch_vector_tile(
125        &self,
126        tile_x: u32,
127        tile_y: u32,
128        zoom_level: u8,
129    ) -> Result<VectorTileData, SDKError> {
130        if zoom_level > SOURDOUGH_MAXIMUM_ZOOM_LEVEL {
131            return Err(SDKError::InvalidCoordinateInput(format!(
132                "zoom level {zoom_level} exceeds the Sourdough TileJSON maximum of \
133                 {SOURDOUGH_MAXIMUM_ZOOM_LEVEL}"
134            )));
135        }
136        let maximum_tile_index = (1u32 << zoom_level) - 1;
137        if tile_x > maximum_tile_index || tile_y > maximum_tile_index {
138            return Err(SDKError::InvalidCoordinateInput(format!(
139                "tile x={tile_x} y={tile_y} exceeds the {maximum_tile_index} maximum at zoom \
140                 {zoom_level}"
141            )));
142        }
143        self.rate_limiter.acquire().await?;
144        let wire_bytes = self
145            .http_client
146            .get_tile_bytes(&self.tile_request_path(tile_x, tile_y, zoom_level))
147            .await?;
148        VectorTileData::from_wire_bytes(tile_x, tile_y, zoom_level, wire_bytes)
149    }
150}

291    /// ASCII name the fixture's first layer carries (its `name` string
292    /// field, read out at fetch time 2026-10-29): the schema pin the other
293    /// decoded assertion compares against.
294    const FIRST_SAMPLE_LAYER_NAME: &[u8] = b"aeroways";

296    #[test]
297    fn committed_fixture_tile_carries_the_mvt_envelope() {
298        // Byte-level envelope sanity only: the MVT Tile protobuf leads with
299        // the repeated `layers` field, and the first layer leads with its
300        // `name` string field. Parsing beyond that is the decoder slice's
301        // job (M9) — this asserts the committed bytes look like an MVT
302        // tile, with no decoder involved.
303        let fixture_bytes = fixture_tile();
304        assert_eq!(fixture_bytes[0], MVT_LAYER_FIELD_PROTOBUF_KEY);
305        let (_layer_length, layer_length_varint_width) = read_varint(&fixture_bytes[1..]);
306        let layer_bytes = &fixture_bytes[1 + layer_length_varint_width..];
307        assert_eq!(layer_bytes[0], MVT_LAYER_NAME_FIELD_PROTOBUF_KEY);
308        let (name_length, name_varint_width) = read_varint(&layer_bytes[1..]);
309        let name_start = 1 + name_varint_width;
310        let layer_name = &layer_bytes[name_start..name_start + name_length as usize];
311        assert_eq!(layer_name, FIRST_SAMPLE_LAYER_NAME);
312    }

314    /// Reads one protobuf varint from the front of `bytes`, returning
315    /// `(value, bytes consumed)`.
316    fn read_varint(bytes: &[u8]) -> (u64, usize) {
317        let mut value: u64 = 0;
318        for (shift_offset, byte) in bytes.iter().take(10).enumerate() {
319            value |= u64::from(byte & 0x7f) << (7 * shift_offset);
320            if byte & 0x80 == 0 {
321                return (value, shift_offset + 1);
322            }
323        }
324        panic!("varint over ten bytes cannot appear in an MVT envelope");
325    }
