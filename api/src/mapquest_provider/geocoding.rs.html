12/// Geocoding against the MapQuest Geocoding API v1.
13pub struct MapQuestGeocodingProvider {
14    provider_client: Arc<MapQuestProviderClient>,
15}
16
17impl MapQuestGeocodingProvider {
18    /// Wraps the shared rate-limited client.
19    pub fn new(provider_client: Arc<MapQuestProviderClient>) -> Self {
20        Self { provider_client }
21    }
22}
23
24#[async_trait]
25impl GeocodingProvider for MapQuestGeocodingProvider {
26    async fn forward_geocode(&self, address_text: &str) -> Result<Vec<PlaceResult>, SDKError> {
27        if address_text.trim().is_empty() {
28            return Err(SDKError::InvalidCoordinateInput(
29                "address text must not be empty".to_string(),
30            ));
31        }
32        let query_pairs = vec![("location".to_string(), address_text.to_string())];
33        let payload = self
34            .provider_client
35            .get_json("geocoding/v1/address", &query_pairs)
36            .await?;
37        parse_geocoding_payload(&payload)
38    }
39
40    async fn reverse_geocode(
41        &self,
42        coordinate: &GeographicCoordinate,
43    ) -> Result<Vec<PlaceResult>, SDKError> {
44        let query_pairs = vec![(
45            "location".to_string(),
46            format!("{},{}", coordinate.latitude, coordinate.longitude),
47        )];
48        let payload = self
49            .provider_client
50            .get_json("geocoding/v1/reverse", &query_pairs)
51            .await?;
52        parse_geocoding_payload(&payload)
53    }
54}

56/// Parses `results[].locations[]` into vendor-neutral places, joining the
57/// populated admin-area fields into a display string and mapping the
58/// documented `geocodeQuality` vocabulary onto [`GeocodeQuality`].
59pub fn parse_geocoding_payload(payload: &Value) -> Result<Vec<PlaceResult>, SDKError> {
60    let result_list = payload
61        .get("results")
62        .and_then(Value::as_array)
63        .ok_or_else(|| {
64            SDKError::ResponseParsingFailure("geocoding payload lacks results".to_string())
65        })?;
66    let mut places = Vec::new();
67    for result in result_list {
68        let locations = result
69            .get("locations")
70            .and_then(Value::as_array)
71            .ok_or_else(|| {
72                SDKError::ResponseParsingFailure("geocoding result lacks locations".to_string())
73            })?;
74        for location in locations {
75            places.push(parse_location(location)?);
76        }
77    }
78    Ok(places)
79}
80
81fn parse_location(location: &Value) -> Result<PlaceResult, SDKError> {
82    let latitude = location
83        .pointer("/latLng/lat")
84        .and_then(Value::as_f64)
85        .ok_or_else(|| {
86            SDKError::ResponseParsingFailure("geocoding location lacks latLng".to_string())
87        })?;
88    let longitude = location
89        .pointer("/latLng/lng")
90        .and_then(Value::as_f64)
91        .ok_or_else(|| {
92            SDKError::ResponseParsingFailure("geocoding location lacks latLng".to_string())
93        })?;
94    let coordinate = GeographicCoordinate::from_degrees(latitude, longitude)?;
95
96    let street = location
97        .get("street")
98        .and_then(Value::as_str)
99        .unwrap_or_default();
100    let city = location
101        .get("adminArea5")
102        .and_then(Value::as_str)
103        .unwrap_or_default();
104    let state = location
105        .get("adminArea3")
106        .and_then(Value::as_str)
107        .unwrap_or_default();
108    let country = location
109        .get("adminArea1")
110        .and_then(Value::as_str)
111        .unwrap_or_default();
112    let postal_code = location
113        .get("postalCode")
114        .and_then(Value::as_str)
115        .unwrap_or_default();
116    let structured_address_text = [street, city, state, postal_code, country]
117        .iter()
118        .filter(|part| !part.is_empty())
119        .cloned()
120        .collect::<Vec<&str>>()
121        .join(", ");
122
123    let quality_text = location
124        .get("geocodeQuality")
125        .and_then(Value::as_str)
126        .unwrap_or_default();
127    let quality_code = location
128        .get("geocodeQualityCode")
129        .and_then(Value::as_str)
130        .unwrap_or_default();
131
132    Ok(PlaceResult {
133        structured_address_text,
134        coordinate,
135        geocode_quality: map_geocode_quality(quality_text, quality_code),
136    })
137}

139/// Maps the documented `geocodeQuality` vocabulary onto [`GeocodeQuality`],
140/// falling back to the precision letter of the five-character
141/// `geocodeQualityCode` when the vocabulary is unknown.
142pub fn map_geocode_quality(quality_text: &str, quality_code: &str) -> GeocodeQuality {
143    match quality_text {
144        "POINT" => GeocodeQuality::Point,
145        "ADDRESS" => GeocodeQuality::StreetAddress,
146        "INTERSECTION" => GeocodeQuality::Intersection,
147        "STREET" => GeocodeQuality::Street,
148        "ZIP" => GeocodeQuality::PostalCode,
149        "CITY" => GeocodeQuality::City,
150        "COUNTY" => GeocodeQuality::County,
151        "STATE" => GeocodeQuality::State,
152        "COUNTRY" => GeocodeQuality::Country,
153        // Neighborhood resolves below settlement level; City is the closest
154        // structural precision in the SDK model.
155        "NEIGHBORHOOD" => GeocodeQuality::City,
156        _ => map_quality_code_letter(quality_code.chars().next().unwrap_or_default()),
157    }
158}
159
160fn map_quality_code_letter(precision_letter: char) -> GeocodeQuality {
161    match precision_letter {
162        'P' | 'L' => GeocodeQuality::Point,
163        'A' => GeocodeQuality::StreetAddress,
164        'I' => GeocodeQuality::Intersection,
165        'S' => GeocodeQuality::Street,
166        'Z' => GeocodeQuality::PostalCode,
167        'C' => GeocodeQuality::City,
168        'D' => GeocodeQuality::County,
169        'T' => GeocodeQuality::State,
170        'R' => GeocodeQuality::Country,
171        _ => GeocodeQuality::Street,
172    }
173}

175#[cfg(test)]
176mod tests {
177    use super::{map_geocode_quality, parse_geocoding_payload};
178    use geospatial_types::place::GeocodeQuality;

180    fn fixture(name: &str) -> serde_json::Value {
181        let fixture_path = format!(
182            "{}/../../fixtures/mapquest/{name}",
183            env!("CARGO_MANIFEST_DIR")
184        );
185        serde_json::from_str(&std::fs::read_to_string(fixture_path).unwrap()).unwrap()
186    }
187
188    #[test]
189    fn parses_recorded_city_level_geocode() {
190        let payload = fixture("geocode_address_city.json");
191        let places = parse_geocoding_payload(&payload).unwrap();
192        assert!(!places.is_empty());
193        let first = &places[0];
194        assert_eq!(first.coordinate.latitude, 38.57944);
195        assert_eq!(first.coordinate.longitude, -121.49085);
196        assert_eq!(first.geocode_quality, GeocodeQuality::City);
197        assert!(first.structured_address_text.contains("Sacramento"));
198        assert!(first.structured_address_text.ends_with("US"));
199    }

201    #[test]
202    fn document_quality_vocabulary_maps_every_level() {
203        assert_eq!(map_geocode_quality("POINT", ""), GeocodeQuality::Point);
204        assert_eq!(
205            map_geocode_quality("ADDRESS", ""),
206            GeocodeQuality::StreetAddress
207        );
208        assert_eq!(
209            map_geocode_quality("INTERSECTION", ""),
210            GeocodeQuality::Intersection
211        );
212        assert_eq!(map_geocode_quality("STREET", ""), GeocodeQuality::Street);
213        assert_eq!(map_geocode_quality("ZIP", ""), GeocodeQuality::PostalCode);
214        assert_eq!(map_geocode_quality("CITY", ""), GeocodeQuality::City);
215        assert_eq!(map_geocode_quality("COUNTY", ""), GeocodeQuality::County);
216        assert_eq!(map_geocode_quality("STATE", ""), GeocodeQuality::State);
217        assert_eq!(map_geocode_quality("COUNTRY", ""), GeocodeQuality::Country);
218        assert_eq!(

223        assert_eq!(
