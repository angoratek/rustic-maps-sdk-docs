1//! Shared facade for the MapQuest endpoint implementations: every outbound
2//! call passes the rate limiter first, then transport, then the `info`
3//! envelope check.
4
5use crate::error_mapping::validate_info_envelope;
6use crate::http::MapQuestHttpClient;
7use geospatial_types::SDKError;
8use provider_transport::TokenBucketRateLimiter;
9use serde_json::Value;
10
11/// Rate-limited, envelope-validated wrapper around [`MapQuestHttpClient`]
12/// shared by all MapQuest provider implementations.
13pub struct MapQuestProviderClient {
14    http_client: MapQuestHttpClient,
15    rate_limiter: TokenBucketRateLimiter,
16}
17
18/// Envelopes are the status channel of Directions/Geocoding/Traffic/Tiles;
19/// the Search Ahead API carries none and only fails at transport level, so
20/// absence of the envelope is tolerated here.
21fn validate_info_envelope_if_present(payload: &Value) -> Result<(), SDKError> {
22    if payload.get("info").is_some() {
23        return validate_info_envelope(payload);
24    }
25    Ok(())
26}

46/// Bundled tuning for the shared client.
47pub struct MapQuestProviderConfiguration {
48    /// Base URL; defaults to the public MapQuest platform and is overridden
49    /// in tests.
50    pub base_url: String,
51    /// Consumer key issued by the MapQuest developer portal.
52    pub api_key: String,
53    /// Bucket capacity; bursts up to this many requests start immediately.
54    pub token_bucket_capacity: f64,
55    /// Continuous refill rate in tokens per second.
56    pub refill_tokens_per_second: f64,
57    /// Longest wait accepted before surfacing [`SDKError::RateLimitExhausted`].
58    pub maximum_wait_seconds: u64,
59    /// Retry budget for idempotent GETs.
60    pub maximum_retry_attempts: u32,
61}
62
63impl Default for MapQuestProviderConfiguration {
64    fn default() -> Self {
65        Self {
66            base_url: crate::http::MAPQUEST_BASE_URL.to_string(),
67            api_key: String::new(),
68            token_bucket_capacity: 30.0,
69            refill_tokens_per_second: 5.0,
70            maximum_wait_seconds: 10,
71            maximum_retry_attempts: 2,
72        }
73    }
74}

76impl MapQuestProviderClient {
77    /// Builds the shared client from the given configuration.
78    pub fn new(configuration: MapQuestProviderConfiguration) -> Result<Self, SDKError> {
79        Ok(Self {
80            http_client: MapQuestHttpClient::new(
81                configuration.base_url,
82                configuration.api_key,
83                configuration.maximum_retry_attempts,
84            )?,
85            rate_limiter: TokenBucketRateLimiter::new(
86                configuration.token_bucket_capacity,
87                configuration.refill_tokens_per_second,
88                configuration.maximum_wait_seconds,
89            )?,
90        })
91    }
92
93    /// Rate-limited GET returning a payload validated against the `info`
94    /// envelope when present (Search Ahead replies carry none).
95    pub async fn get_json(
96        &self,
97        path: &str,
98        query_pairs: &[(String, String)],
99        provider_contract: &str,
100    ) -> Result<Value, SDKError> {
101        self.rate_limiter.acquire().await?;
102        let payload = self
103            .http_client
104            .get_json(path, query_pairs, provider_contract)
105            .await?;
106        validate_info_envelope_if_present(&payload)?;
107        Ok(payload)
108    }
109
110    /// Rate-limited POST returning a payload validated against the `info`
111    /// envelope when present.
112    pub async fn post_json(
113        &self,
114        path: &str,
115        query_pairs: &[(String, String)],
116        request_body: &Value,
117        provider_contract: &str,
118    ) -> Result<Value, SDKError> {
119        self.rate_limiter.acquire().await?;
120        let payload = self
121            .http_client
122            .post_json(path, query_pairs, request_body, provider_contract)
123            .await?;
124        validate_info_envelope_if_present(&payload)?;
125        Ok(payload)
126    }
127
128    /// Rate-limited binary fetch (raster tiles have no `info` envelope).
129    pub async fn get_bytes(
130        &self,
131        path: &str,
132        provider_contract: &str,
133    ) -> Result<Vec<u8>, SDKError> {
134        self.rate_limiter.acquire().await?;
135        self.http_client.get_bytes(path, provider_contract).await
136    }
137}

153    fn client_with_base(base_url: String) -> MapQuestProviderClient {
154        MapQuestProviderClient::new(MapQuestProviderConfiguration {
155            base_url,
156            api_key: "test-key".to_string(),
157            ..MapQuestProviderConfiguration::default()
158        })
159        .unwrap()
160    }

162    /// An observer that keeps every observation it receives.
163    #[derive(Default)]
164    struct RecordingObserver {
165        observations: Mutex<Vec<ProviderRequestObservation>>,
166    }
167
168    impl ProviderRequestObserver for RecordingObserver {
169        fn on_provider_request(&self, observation: ProviderRequestObservation) {
170            self.observations
171                .lock()
172                .expect("the observation list is never poisoned")
173                .push(observation);
174        }
175    }
176
177    #[tokio::test]
178    async fn a_request_is_observed_with_its_contract_and_key_but_not_the_credential() {
179        let server = MockServer::start().await;
180        Mock::given(method("GET"))
181            .and(path("/geocoding/v1/address"))
182            .respond_with(ResponseTemplate::new(200).set_body_json(serde_json::json!({
183                "info": {"statuscode": 0, "messages": []},
184                "results": []
185            })))
186            .mount(&server)
187            .await;
188        let recording_observer = Arc::new(RecordingObserver::default());
189        install_provider_request_observer(Some(recording_observer.clone()));
190        let client = client_with_base(server.uri());
191        client
192            .get_json("geocoding/v1/address", &[], GEOCODING_CONTRACT_NAME)
193            .await
194            .expect("the mocked provider answers");
195
196        let observations = recording_observer
197            .observations
198            .lock()
199            .expect("the observation list is never poisoned")
200            .clone();
201        // The sink is process-wide, so tests running in parallel are observed
202        // too; the assertions therefore look for this call and check that no
203        // observation of any call leaks the credential.
204        assert!(
205            observations.iter().any(|observation| {
206                observation.provider_contract == GEOCODING_CONTRACT_NAME
207                    && observation.provider_key == MAPQUEST_PROVIDER_KEY
208                    && observation.status_class == "2xx"
209                    && observation.request_path.contains("/geocoding/v1/address")
210                    && observation.request_path.contains("key=<redacted>")
211            }),
212            "the successful geocoding request is observed with its contract and key"
213        );
214        assert!(observations.iter().all(|observation| {
215            !observation.request_path.contains("test-key")
216                && !observation.response_snippet.contains("test-key")
217        }));
218        install_provider_request_observer(None);
219    }

221    #[tokio::test]
222    async fn application_level_statuscodes_reject_before_returning() {
223        let server = MockServer::start().await;
224        Mock::given(method("GET"))
225            .and(path("/directions/v2/route"))
226            .respond_with(ResponseTemplate::new(200).set_body_json(serde_json::json!({
227                "info": {"statuscode": 402, "messages": ["Invalid Location."]},
228                "route": {}
229            })))
230            .mount(&server)
231            .await;
232        let client = client_with_base(server.uri());
233        let error = client
234            .get_json(

237                crate::client::ROUTING_CONTRACT_NAME,
238            )
239            .await
240            .unwrap_err();
241        assert!(matches!(error, SDKError::ProviderRequestRejected(_)));
242    }
