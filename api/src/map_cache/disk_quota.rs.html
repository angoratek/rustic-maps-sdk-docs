1//! Disk quota resolution: absolute byte caps and free-disk shares.
2//!
3//! [`CacheQuota`] resolves a byte budget from two optional constraints —
4//! an absolute maximum and a percentage share of currently free disk space —
5//! taking the smaller whenever both are set. Free space comes from a
6//! [`DiskSpaceProbe`] so tests can drive eviction through a fixed synthetic
7//! probe without touching the real filesystem.
8
9use geospatial_types::{CacheOperationFailureReason, SDKError};
10use std::path::Path;
11
12/// Source of the currently available disk space under the cache.
13pub trait DiskSpaceProbe: Send + Sync {
14    /// Free bytes on the filesystem holding `cache_file_path`.
15    fn available_disk_bytes(&self, cache_file_path: &Path) -> Result<u64, SDKError>;
16}
17
18/// Probe backed by the real filesystem (fs4 statfs family).
19pub struct RealDiskSpaceProbe;
20
21impl DiskSpaceProbe for RealDiskSpaceProbe {
22    fn available_disk_bytes(&self, cache_file_path: &Path) -> Result<u64, SDKError> {
23        fs4::available_space(cache_file_path).map_err(|_| {
24            SDKError::CacheOperationFailure(CacheOperationFailureReason::StoreUnavailable)
25        })
26    }
27}
28
29/// Synthetic probe with a constant available byte count, for tests.
30pub struct FixedDiskSpaceProbe {
31    fixed_available_bytes: u64,
32}
33
34impl FixedDiskSpaceProbe {
35    /// Pins the reported free space to the given byte count.
36    pub fn reporting(fixed_available_bytes: u64) -> Self {
37        Self {
38            fixed_available_bytes,
39        }
40    }
41}
42
43impl DiskSpaceProbe for FixedDiskSpaceProbe {
44    fn available_disk_bytes(&self, _cache_file_path: &Path) -> Result<u64, SDKError> {
45        Ok(self.fixed_available_bytes)
46    }
47}
48
49/// Optional constraints bounding the total cache size on disk.
50#[derive(Debug, Clone, Default)]
51pub struct CacheQuota {
52    /// Hard cap in bytes, independent of filesystem state.
53    pub absolute_maximum_bytes: Option<u64>,
54    /// Percentage (0..=100) of currently free disk space the cache may use.
55    pub free_disk_share_percentage: Option<f64>,
56}
57
58impl CacheQuota {
59    /// Resolves the effective byte budget: the smallest among the set
60    /// constraints, or "unbounded" when none is configured.
61    pub fn resolve_budget_bytes(
62        &self,
63        disk_space_probe: &dyn DiskSpaceProbe,
64        cache_file_path: &Path,
65    ) -> u64 {
66        let mut candidate_budgets: Vec<u64> = Vec::new();
67        if let Some(absolute_maximum_bytes) = self.absolute_maximum_bytes {
68            candidate_budgets.push(absolute_maximum_bytes);
69        }
70        if let Some(free_disk_share_percentage) = self.free_disk_share_percentage {
71            if let Ok(available_disk_bytes) = disk_space_probe.available_disk_bytes(cache_file_path)
72            {
73                candidate_budgets.push(share_cap(available_disk_bytes, free_disk_share_percentage));
74            }
75        }
76        candidate_budgets.iter().min().copied().unwrap_or(u64::MAX)
77    }
78
79    /// Rejects percentage values outside 0..=100.
80    pub fn validate(&self) -> Result<(), SDKError> {
81        if let Some(free_disk_share_percentage) = self.free_disk_share_percentage {
82            if !(0.0..=100.0).contains(&free_disk_share_percentage) {
83                return Err(SDKError::InvalidCoordinateInput(
84                    "cache free-disk share percentage must be within 0..=100".to_string(),
85                ));
86            }
87        }
88        Ok(())
89    }
90}
91
92fn share_cap(available_disk_bytes: u64, free_disk_share_percentage: f64) -> u64 {
93    let share_bytes = (u128::from(available_disk_bytes)
94        * (free_disk_share_percentage * 100.0).round() as u128)
95        / 10_000;
96    share_bytes.min(u64::MAX as u128) as u64
97}

99#[cfg(test)]
100mod tests {
101    use super::{CacheQuota, DiskSpaceProbe, FixedDiskSpaceProbe, RealDiskSpaceProbe};
102    use std::path::Path;
103
104    #[test]
105    fn absolute_cap_wins_over_small_share() {
106        let quota = CacheQuota {
107            absolute_maximum_bytes: Some(1_000_000),
108            free_disk_share_percentage: Some(50.0),
109        };
110        let probe = FixedDiskSpaceProbe::reporting(10_000_000);
111        assert_eq!(
112            quota.resolve_budget_bytes(&probe, Path::new("/tmp")),
113            1_000_000
114        );
115    }
116
117    #[test]
118    fn share_cap_wins_over_small_absolute() {
119        let quota = CacheQuota {
120            absolute_maximum_bytes: Some(1_000),
121            free_disk_share_percentage: Some(50.0),
122        };
123        let probe = FixedDiskSpaceProbe::reporting(10_000_000);
124        assert_eq!(quota.resolve_budget_bytes(&probe, Path::new("/tmp")), 1_000);
125    }
126
127    #[test]
128    fn share_resolves_from_reported_free_space() {
129        let quota = CacheQuota {
130            absolute_maximum_bytes: None,
131            free_disk_share_percentage: Some(25.0),
132        };
133        let probe = FixedDiskSpaceProbe::reporting(8_000_000);
134        assert_eq!(
135            quota.resolve_budget_bytes(&probe, Path::new("/tmp")),
136            2_000_000
137        );
138    }
139
140    #[test]
141    fn unbounded_quota_has_no_budget() {
142        let quota = CacheQuota::default();
143        let probe = FixedDiskSpaceProbe::reporting(64);
144        assert_eq!(
145            quota.resolve_budget_bytes(&probe, Path::new("/tmp")),
146            u64::MAX
147        );
148    }
149
150    #[test]
151    fn rejects_out_of_range_percentages() {
152        let quota = CacheQuota {
153            free_disk_share_percentage: Some(150.0),
154            ..CacheQuota::default()
155        };
156        assert!(quota.validate().is_err());
157        assert!(CacheQuota::default().validate().is_ok());
158    }
159
160    #[test]
161    fn real_probe_reads_a_real_filesystem() {
162        let probe = RealDiskSpaceProbe;
163        // Any real readable file path works; only a positive value matters.
164        let real_path = std::env::temp_dir().join("map-cache-quota-probe");
165        std::fs::write(&real_path, b"probe").unwrap();
166        let available = probe.available_disk_bytes(&real_path).unwrap();
167        assert!(available > 0);
168        std::fs::remove_file(&real_path).unwrap();
169    }
