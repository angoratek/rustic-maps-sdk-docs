1//! Response categories and their default freshness policies.
2//!
3//! Every cache entry is tagged with one [`ResponseCategory`]; the category
4//! drives the default time-to-live so that volatile traffic data expires
5//! minutes after capture while immutable tile imagery lives for weeks.
6//! Individual calls may pass a shorter or longer TTL — the cached provider
7//! wrappers in [`crate::cached_providers`] accept a per-instance override.
8
9use geospatial_types::SDKError;
10use serde::{Deserialize, Serialize};
11
12/// Kind of provider response a cache entry holds.
13#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Serialize, Deserialize)]
14#[serde(rename_all = "snake_case")]
15pub enum ResponseCategory {
16    /// Forward and reverse geocoding results.
17    Geocoding,
18    /// Calculated routes and their maneuvers.
19    Routing,
20    /// Origin-destination travel-time matrices.
21    DistanceMatrix,
22    /// Search Ahead predictions (autocomplete and full search).
23    SearchPrediction,
24    /// Traffic disruption reports.
25    TrafficIncidents,
26    /// Traffic flow congestion polylines.
27    TrafficFlow,
28    /// Fetched raster XYZ tiles.
29    RasterTiles,
30    /// Fetched xyz vector tiles (MVT protobuf; the M9 vector seam).
31    VectorTiles,
32}
33
34impl ResponseCategory {
35    /// Default time-to-live in seconds for this category. Values reflect
36    /// how quickly the underlying MapQuest data changes; anything more
37    /// aggressive than these defaults is wasteful, anything longer risks
38    /// serving plainly stale live data.
39    ///
40    /// Tiles are the exception by owner ruling (2026-10-28, PRD §6.7):
41    /// tile imagery caches heavily by default — **30 calendar days**, the
42    /// common maximum under imagery-provider terms — and an integrator
43    /// tuning it shorter or longer does so knowingly per provider (the
44    /// provider contract declares its own caching terms; see
45    /// `mapquest-provider`'s raster-tile ceiling constant). The cached
46    /// provider wrappers accept the per-instance override.
47    ///
48    /// Vector tiles (M9) inherit the same tile-imagery policy by owner
49    /// direction: 30 calendar days here, clamped against the provider's own
50    /// ceiling constant (see `openstreetmap-us-vector-provider`'s lockstep
51    /// test).
52    pub fn default_time_to_live_seconds(self) -> u64 {
53        match self {
54            ResponseCategory::Geocoding => 24 * 60 * 60,
55            ResponseCategory::Routing => 10 * 60,
56            ResponseCategory::DistanceMatrix => 10 * 60,
57            ResponseCategory::SearchPrediction => 5 * 60,
58            ResponseCategory::TrafficIncidents => 60,
59            ResponseCategory::TrafficFlow => 30,
60            ResponseCategory::RasterTiles => 30 * 24 * 60 * 60,
61            ResponseCategory::VectorTiles => 30 * 24 * 60 * 60,
62        }
63    }
64
65    /// Stable name persisted in the SQLite `category` column.
66    pub fn storage_name(self) -> &'static str {
67        match self {
68            ResponseCategory::Geocoding => "geocoding",
69            ResponseCategory::Routing => "routing",
70            ResponseCategory::DistanceMatrix => "distance_matrix",
71            ResponseCategory::SearchPrediction => "search_prediction",
72            ResponseCategory::TrafficIncidents => "traffic_incidents",
73            ResponseCategory::TrafficFlow => "traffic_flow",
74            ResponseCategory::RasterTiles => "raster_tiles",
75            ResponseCategory::VectorTiles => "vector_tiles",
76        }
77    }
78
79    /// Reverses [`ResponseCategory::storage_name`].
80    pub fn from_storage_name(storage_name: &str) -> Result<Self, SDKError> {
81        match storage_name {
82            "geocoding" => Ok(ResponseCategory::Geocoding),
83            "routing" => Ok(ResponseCategory::Routing),
84            "distance_matrix" => Ok(ResponseCategory::DistanceMatrix),
85            "search_prediction" => Ok(ResponseCategory::SearchPrediction),
86            "traffic_incidents" => Ok(ResponseCategory::TrafficIncidents),
87            "traffic_flow" => Ok(ResponseCategory::TrafficFlow),
88            "raster_tiles" => Ok(ResponseCategory::RasterTiles),
89            "vector_tiles" => Ok(ResponseCategory::VectorTiles),
90            other => {
91                let _ = other;
92                Err(SDKError::CacheOperationFailure(
93                    geospatial_types::CacheOperationFailureReason::RecordCorrupted,
94                ))
95            }
96        }
97    }
98}

100#[cfg(test)]
101mod tests {
102    use super::{ResponseCategory, SDKError};
103
104    #[test]
105    fn expiry_ages_follow_data_volatility() {
106        assert!(ResponseCategory::TrafficFlow.default_time_to_live_seconds() < 60);
107        assert!(
108            ResponseCategory::TrafficIncidents.default_time_to_live_seconds()
109                < ResponseCategory::Routing.default_time_to_live_seconds()
110        );
111        assert_eq!(
112            ResponseCategory::Geocoding.default_time_to_live_seconds(),
113            86_400
114        );
115        assert_eq!(
116            ResponseCategory::RasterTiles.default_time_to_live_seconds(),
117            30 * 24 * 60 * 60
118        );
119    }
120
121    #[test]
122    fn storage_names_round_trip() {
123        for category in [
124            ResponseCategory::Geocoding,
125            ResponseCategory::Routing,
126            ResponseCategory::DistanceMatrix,
127            ResponseCategory::SearchPrediction,
128            ResponseCategory::TrafficIncidents,
129            ResponseCategory::TrafficFlow,
130            ResponseCategory::RasterTiles,
131            ResponseCategory::VectorTiles,
132        ] {
133            assert_eq!(
134                ResponseCategory::from_storage_name(category.storage_name()).unwrap(),
135                category
136            );
137        }
138        assert!(matches!(
139            ResponseCategory::from_storage_name("unknown"),
140            Err(SDKError::CacheOperationFailure(
141                geospatial_types::CacheOperationFailureReason::RecordCorrupted
142            ))
143        ));
144    }
