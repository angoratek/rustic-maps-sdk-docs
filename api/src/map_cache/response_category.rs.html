1//! Response categories and their default freshness policies.
2//!
3//! Every cache entry is tagged with one [`ResponseCategory`]; the category
4//! drives the default time-to-live so that volatile traffic data expires
5//! minutes after capture while immutable tile imagery lives for weeks.
6//! Individual calls may pass a shorter or longer TTL — the cached provider
7//! wrappers in [`crate::cached_providers`] accept a per-instance override.
8
9use geospatial_types::SDKError;
10use serde::{Deserialize, Serialize};
11
12/// Kind of provider response a cache entry holds.
13#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Serialize, Deserialize)]
14#[serde(rename_all = "snake_case")]
15pub enum ResponseCategory {
16    /// Forward and reverse geocoding results.
17    Geocoding,
18    /// Calculated routes and their maneuvers.
19    Routing,
20    /// Origin-destination travel-time matrices.
21    DistanceMatrix,
22    /// Search Ahead predictions (autocomplete and full search).
23    SearchPrediction,
24    /// Traffic disruption reports.
25    TrafficIncidents,
26    /// Traffic flow congestion polylines.
27    TrafficFlow,
28    /// Fetched raster XYZ tiles.
29    RasterTiles,
30}
31
32impl ResponseCategory {
33    /// Default time-to-live in seconds for this category. Values reflect
34    /// how quickly the underlying MapQuest data changes; anything more
35    /// aggressive than these defaults is wasteful, anything longer risks
36    /// serving plainly stale live data.
37    ///
38    /// Tiles are the exception by owner ruling (2026-10-28, PRD §6.7):
39    /// tile imagery caches heavily by default — **30 calendar days**, the
40    /// common maximum under imagery-provider terms — and an integrator
41    /// tuning it shorter or longer does so knowingly per provider (the
42    /// provider contract declares its own caching terms; see
43    /// `mapquest-provider`'s raster-tile ceiling constant). The cached
44    /// provider wrappers accept the per-instance override.
45    pub fn default_time_to_live_seconds(self) -> u64 {
46        match self {
47            ResponseCategory::Geocoding => 24 * 60 * 60,
48            ResponseCategory::Routing => 10 * 60,
49            ResponseCategory::DistanceMatrix => 10 * 60,
50            ResponseCategory::SearchPrediction => 5 * 60,
51            ResponseCategory::TrafficIncidents => 60,
52            ResponseCategory::TrafficFlow => 30,
53            ResponseCategory::RasterTiles => 30 * 24 * 60 * 60,
54        }
55    }
56
57    /// Stable name persisted in the SQLite `category` column.
58    pub fn storage_name(self) -> &'static str {
59        match self {
60            ResponseCategory::Geocoding => "geocoding",
61            ResponseCategory::Routing => "routing",
62            ResponseCategory::DistanceMatrix => "distance_matrix",
63            ResponseCategory::SearchPrediction => "search_prediction",
64            ResponseCategory::TrafficIncidents => "traffic_incidents",
65            ResponseCategory::TrafficFlow => "traffic_flow",
66            ResponseCategory::RasterTiles => "raster_tiles",
67        }
68    }
69
70    /// Reverses [`ResponseCategory::storage_name`].
71    pub fn from_storage_name(storage_name: &str) -> Result<Self, SDKError> {
72        match storage_name {
73            "geocoding" => Ok(ResponseCategory::Geocoding),
74            "routing" => Ok(ResponseCategory::Routing),
75            "distance_matrix" => Ok(ResponseCategory::DistanceMatrix),
76            "search_prediction" => Ok(ResponseCategory::SearchPrediction),
77            "traffic_incidents" => Ok(ResponseCategory::TrafficIncidents),
78            "traffic_flow" => Ok(ResponseCategory::TrafficFlow),
79            "raster_tiles" => Ok(ResponseCategory::RasterTiles),
80            other => {
81                let _ = other;
82                Err(SDKError::CacheOperationFailure(
83                    geospatial_types::CacheOperationFailureReason::RecordCorrupted,
84                ))
85            }
86        }
87    }
88}

90#[cfg(test)]
91mod tests {
92    use super::{ResponseCategory, SDKError};
93
94    #[test]
95    fn expiry_ages_follow_data_volatility() {
96        assert!(ResponseCategory::TrafficFlow.default_time_to_live_seconds() < 60);
97        assert!(
98            ResponseCategory::TrafficIncidents.default_time_to_live_seconds()
99                < ResponseCategory::Routing.default_time_to_live_seconds()
100        );
101        assert_eq!(
102            ResponseCategory::Geocoding.default_time_to_live_seconds(),
103            86_400
104        );
105        assert_eq!(
106            ResponseCategory::RasterTiles.default_time_to_live_seconds(),
107            30 * 24 * 60 * 60
108        );
109    }
110
111    #[test]
112    fn storage_names_round_trip() {
113        for category in [
114            ResponseCategory::Geocoding,
115            ResponseCategory::Routing,
116            ResponseCategory::DistanceMatrix,
117            ResponseCategory::SearchPrediction,
118            ResponseCategory::TrafficIncidents,
119            ResponseCategory::TrafficFlow,
120            ResponseCategory::RasterTiles,
121        ] {
122            assert_eq!(
123                ResponseCategory::from_storage_name(category.storage_name()).unwrap(),
124                category
125            );
126        }
127        assert!(matches!(
128            ResponseCategory::from_storage_name("unknown"),
129            Err(SDKError::CacheOperationFailure(
130                geospatial_types::CacheOperationFailureReason::RecordCorrupted
131            ))
132        ));
133    }
