1//! The pacing a tile source declares for its outbound requests.
2//!
3//! Providers publish different request budgets: a source with a generous
4//! budget admits a viewport's tiles together, while one with a stricter
5//! published budget must pace them from the first request. A
6//! [`TileSourcePacing`] carries the two rates a source declares — the
7//! immediate requests admitted and the sustained rate — so a declaration can
8//! lower the bucket without changing every caller. Absent fields keep
9//! [`TileSourcePacing::default`], the rate the renderer's fetch window was
10//! sized for.
11//!
12//! Every rate is validated against the source's name before the provider is
13//! built, so a bad declaration is a typed [`SDKError`] naming the source
14//! rather than a limiter failure with no subject.
15
16use crate::rate_limit::{
17    MINIMUM_EXCLUSIVE_REFILL_TOKENS_PER_SECOND, MINIMUM_TOKEN_BUCKET_CAPACITY_TOKENS,
18};
19use geospatial_types::SDKError;
20
21/// Immediate requests a tile source admits before pacing starts. It matches
22/// the renderer's eight-fetch in-flight window, so a normal viewport's tiles
23/// all start together instead of queueing behind the bucket.
24pub const DEFAULT_TOKEN_BUCKET_CAPACITY_TOKENS: f64 = 8.0;
25
26/// Sustained request rate a tile source admits, in tokens per second. It
27/// matches the renderer's eight-fetch window: a viewport's worth of tiles
28/// lands within a second, while sustained demand still paces.
29pub const DEFAULT_REFILL_TOKENS_PER_SECOND: f64 = 8.0;
30
31/// The two token-bucket rates one tile source declares.
32///
33/// [`TileSourcePacing::default`] is the rate every source carries when it
34/// declares neither field. A source that publishes a stricter budget lowers
35/// either value from its own declaration; [`TileSourcePacing::validate`]
36/// refuses an unusable declaration and names the source in the detail.
37#[derive(Debug, Clone, Copy, PartialEq)]
38pub struct TileSourcePacing {
39    /// Immediate requests the bucket admits before pacing starts. Must be
40    /// finite and at least one token.
41    pub capacity_tokens: f64,
42    /// Sustained refill rate in tokens per second. Must be finite and
43    /// strictly positive.
44    pub refill_tokens_per_second: f64,
45}
46
47impl Default for TileSourcePacing {
48    fn default() -> Self {
49        Self {
50            capacity_tokens: DEFAULT_TOKEN_BUCKET_CAPACITY_TOKENS,
51            refill_tokens_per_second: DEFAULT_REFILL_TOKENS_PER_SECOND,
52        }
53    }
54}
55
56impl TileSourcePacing {
57    /// Builds one declaration from its two rates. The values are not checked
58    /// here; a source validates them with [`TileSourcePacing::validate`]
59    /// while it knows its own name for the diagnostic.
60    pub const fn new(capacity_tokens: f64, refill_tokens_per_second: f64) -> Self {
61        Self {
62            capacity_tokens,
63            refill_tokens_per_second,
64        }
65    }
66
67    /// Refuses a rate the shared limiter cannot enforce: a non-finite or
68    /// below-one burst capacity, or a non-finite or non-positive refill
69    /// rate. The returned [`SDKError::InvalidCoordinateInput`] names the
70    /// declaring source, because the same rates can be valid for one source
71    /// and unusable for the next.
72    pub fn validate(&self, source_name: &str) -> Result<(), SDKError> {
73        if !self.capacity_tokens.is_finite()
74            || self.capacity_tokens < MINIMUM_TOKEN_BUCKET_CAPACITY_TOKENS
75        {
76            return Err(SDKError::InvalidCoordinateInput(format!(
77                "the {source_name} tile source declares a token-bucket capacity of {} tokens, \
78                 which must be finite and at least {MINIMUM_TOKEN_BUCKET_CAPACITY_TOKENS}",
79                self.capacity_tokens
80            )));
81        }
82        if !self.refill_tokens_per_second.is_finite()
83            || self.refill_tokens_per_second <= MINIMUM_EXCLUSIVE_REFILL_TOKENS_PER_SECOND
84        {
85            return Err(SDKError::InvalidCoordinateInput(format!(
86                "the {source_name} tile source declares a refill rate of {} tokens per second, \
87                 which must be finite and strictly positive",
88                self.refill_tokens_per_second
89            )));
90        }
91        Ok(())
92    }
93}

95#[cfg(test)]
96mod tests {
97    use super::{
98        DEFAULT_REFILL_TOKENS_PER_SECOND, DEFAULT_TOKEN_BUCKET_CAPACITY_TOKENS, TileSourcePacing,
99    };
100    use geospatial_types::SDKError;
101
102    #[test]
103    fn defaults_match_the_renderer_fetch_window() {
104        let pacing = TileSourcePacing::default();
105        assert_eq!(pacing.capacity_tokens, DEFAULT_TOKEN_BUCKET_CAPACITY_TOKENS);
106        assert_eq!(
107            pacing.refill_tokens_per_second,
108            DEFAULT_REFILL_TOKENS_PER_SECOND
109        );
110        pacing.validate("default").expect("the defaults are usable");
111    }
112
113    #[test]
114    fn a_stricter_declaration_is_accepted() {
115        TileSourcePacing::new(1.0, 0.5)
116            .validate("strict")
117            .expect("a finite positive rate is usable");
118    }
119
120    #[test]
121    fn unusable_rates_are_refused_and_name_the_source() {
122        for pacing in [
123            TileSourcePacing::new(f64::NAN, 1.0),
124            TileSourcePacing::new(f64::INFINITY, 1.0),
125            TileSourcePacing::new(0.0, 1.0),
126            TileSourcePacing::new(-1.0, 1.0),
127            TileSourcePacing::new(0.5, 1.0),
128            TileSourcePacing::new(8.0, f64::NAN),
129            TileSourcePacing::new(8.0, f64::INFINITY),
130            TileSourcePacing::new(8.0, 0.0),
131            TileSourcePacing::new(8.0, -1.0),
132        ] {
133            match pacing.validate("strict-budget") {
134                Err(SDKError::InvalidCoordinateInput(detail)) => {
135                    assert!(
136                        detail.contains("strict-budget tile source"),
137                        "the refusal names the source: {detail}"
138                    );
139                }
140                other => panic!("unusable pacing {pacing:?} was not refused: {other:?}"),
141            }
142        }
143    }
