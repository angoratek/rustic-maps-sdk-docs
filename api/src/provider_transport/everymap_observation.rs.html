1//! Bridges the provider layer's request observer into this crate's
2//! diagnostics vocabulary.
3//!
4//! The SDK's provider diagnostics are built from one observation per
5//! completed provider call: the contract it served, the provider's registry
6//! key, the method, the redacted path, the status class, the duration and the
7//! response size. The provider layer reports the same facts through its own
8//! [`everymap_core::client::RequestObserver`], so an adapter lets a provider
9//! built on that layer keep feeding the existing diagnostics ring, the typed
10//! `ProviderRequestTrace` and the browser's event stream unchanged.
11//!
12//! The adapter pairs each request with its response, because the provider
13//! layer reports the two separately and the SDK's vocabulary wants one
14//! record. A response without a recorded request is dropped rather than
15//! reported with invented fields.
16
17use crate::{HttpMethod, deliver_provider_request_observation};
18use everymap_core::client::{RequestObservation, RequestObserver, ResponseObservation};
19use std::sync::{Mutex, PoisonError};
20use std::time::Duration;
21use web_time::Instant;
22
23/// One request the provider layer has started and not yet answered.
24struct InFlightRequest {
25    /// The method the SDK's vocabulary names.
26    http_method: HttpMethod,
27    /// The redacted request URL the provider layer handed over.
28    ///
29    /// The provider layer redacts credential values with its own marker; the
30    /// SDK re-renders the path from this URL, so the marker never reaches the
31    /// diagnostic line.
32    request_url: String,
33    /// When the transport started the attempt.
34    started_at: Instant,
35}
36
37/// Publishes one provider's request attempts into the SDK's diagnostics.
38pub struct EveryMapRequestObserver {
39    /// The contract the calls through this client serve.
40    provider_contract: String,
41    /// The provider's registry key.
42    provider_key: String,
43    /// The attempt awaiting its response.
44    in_flight: Mutex<Option<InFlightRequest>>,
45}
46
47impl EveryMapRequestObserver {
48    /// Builds the observer for one provider and one contract.
49    ///
50    /// # Parameters
51    /// - `provider_contract` — the contract name the SDK reports.
52    /// - `provider_key` — the provider's registry key.
53    pub fn new(provider_contract: &str, provider_key: &str) -> Self {
54        Self {
55            provider_contract: provider_contract.to_string(),
56            provider_key: provider_key.to_string(),
57            in_flight: Mutex::new(None),
58        }
59    }
60
61    /// Renders one completed attempt in the SDK's vocabulary.
62    ///
63    /// The provider layer reports no response body, so the byte count is zero
64    /// and the snippet empty; both are facts about what was observed rather
65    /// than invented values.
66    fn translated_observation(
67        &self,
68        in_flight: InFlightRequest,
69        status: Option<u16>,
70        elapsed_upstream: Duration,
71    ) -> crate::ProviderRequestObservation {
72        let outcome = status.map(|status_code| (status_code, [].as_slice()));
73        crate::provider_request_observation(
74            &self.provider_contract,
75            &self.provider_key,
76            in_flight.http_method,
77            &in_flight.request_url,
78            outcome,
79            duration_or_zero(in_flight.started_at.elapsed(), elapsed_upstream),
80        )
81    }
82}
83
84impl RequestObserver for EveryMapRequestObserver {
85    fn on_request(&self, observation: &RequestObservation<'_>) {
86        let mut in_flight = self
87            .in_flight
88            .lock()
89            .unwrap_or_else(PoisonError::into_inner);
90        *in_flight = Some(InFlightRequest {
91            http_method: sdk_http_method(observation.method),
92            request_url: observation.url.to_string(),
93            started_at: Instant::now(),
94        });
95    }
96
97    fn on_response(&self, observation: &ResponseObservation<'_>) {
98        let Some(in_flight) = self
99            .in_flight
100            .lock()
101            .unwrap_or_else(PoisonError::into_inner)
102            .take()
103        else {
104            return;
105        };
106        deliver_provider_request_observation(self.translated_observation(
107            in_flight,
108            observation.status,
109            observation.elapsed,
110        ));
111    }
112}

114/// The SDK method word for one provider-layer method.
115///
116/// The SDK's vocabulary models the two methods its providers use; a provider
117/// layer method outside that pair is reported as a POST, which is what the
118/// request carries a body for.
119fn sdk_http_method(method: everymap_core::http::Method) -> HttpMethod {
120    match method {
121        everymap_core::http::Method::Get => HttpMethod::Get,
122        _ => HttpMethod::Post,
123    }
124}
125
126/// The attempt duration, preferring the provider layer's own measurement.
127fn duration_or_zero(measured_here: Duration, measured_upstream: Duration) -> Duration {
128    if measured_upstream.is_zero() {
129        measured_here
130    } else {
131        measured_upstream
132    }
133}

135#[cfg(test)]
136mod tests {
137    use super::{EveryMapRequestObserver, InFlightRequest, duration_or_zero};
138    use crate::HttpMethod;
139    use everymap_core::client::{RequestObservation, RequestObserver};

143    /// Builds the observer with one attempt already in flight.
144    fn observer_with_attempt(request_url: &str, method: HttpMethod) -> EveryMapRequestObserver {
145        let observer = EveryMapRequestObserver::new("GeocodingProvider", "mapquest");
146        *observer
147            .in_flight
148            .lock()
149            .expect("the test observer must lock") = Some(InFlightRequest {
150            http_method: method,
151            request_url: request_url.to_string(),
152            started_at: Instant::now(),
153        });
154        observer
155    }
156
157    /// Takes the translated record for the attempt in flight.
158    fn translated(
159        observer: &EveryMapRequestObserver,
160        status: Option<u16>,
161    ) -> crate::ProviderRequestObservation {
162        let in_flight = observer
163            .in_flight
164            .lock()
165            .expect("the test observer must lock")
166            .take()
167            .expect("the fixture leaves one attempt in flight");
168        observer.translated_observation(in_flight, status, Duration::from_millis(12))
169    }
170
171    #[test]
172    fn one_provider_request_becomes_one_sdk_observation() {
173        let observer = observer_with_attempt(
174            "https://example.test/geocoding/v1/address?key=***&location=Sacramento",
175            HttpMethod::Get,
176        );
177        let observation = translated(&observer, Some(200));
178        assert_eq!(observation.provider_contract, "GeocodingProvider");
179        assert_eq!(observation.provider_key, "mapquest");
180        assert_eq!(observation.http_method, "GET");
181        assert!(observation.request_path.contains("/geocoding/v1/address"));
182        assert!(observation.request_path.contains("key=<redacted>"));
183        assert_eq!(observation.status_class, "2xx");
184        assert_eq!(observation.outcome(), "success");
185        assert_eq!(observation.response_bytes, 0);
186    }
187
188    #[test]
189    fn the_request_and_response_callbacks_pair_into_one_record() {
190        let observer = EveryMapRequestObserver::new("GeocodingProvider", "mapquest");
191        observer.on_request(&RequestObservation {
192            provider: "mapquest",
193            method: everymap_core::http::Method::Get,
194            url: "https://example.test/geocoding/v1/address?key=***",
195            attempt: 1,
196        });
197        let in_flight = observer
198            .in_flight
199            .lock()
200            .expect("the test observer must lock")
201            .take()
202            .expect("the request callback records the attempt");
203        assert_eq!(in_flight.http_method, HttpMethod::Get);
204        assert!(
205            observer
206                .in_flight
207                .lock()
208                .expect("the test observer must lock")
209                .is_none()
210        );
211    }

213    #[test]

219                .lock()
