1//! Request headers a browser must not set.
2//!
3//! The Fetch specification reserves a set of header names for the user agent;
4//! page code may not set them. Forwarding one anyway advertises the name in a
5//! cross-origin request's CORS preflight, and a host whose policy does not
6//! allow it answers the preflight with a failure — a tile host that refuses a
7//! `user-agent` header, for example, then serves no tiles at all.
8//!
9//! The predicate is portable so it is unit-tested on every host, while only
10//! the browser transport applies it. The native transport keeps the declared
11//! user agent: a server is free to send one.
12
13/// Header names the Fetch specification forbids page code to set, matched
14/// case-insensitively.
15pub const FORBIDDEN_BROWSER_REQUEST_HEADER_NAMES: [&str; 8] = [
16    "accept-encoding",
17    "connection",
18    "content-length",
19    "cookie",
20    "host",
21    "origin",
22    "referer",
23    "user-agent",
24];
25
26/// Header-name prefixes the Fetch specification reserves for the user agent.
27pub const FORBIDDEN_BROWSER_REQUEST_HEADER_PREFIXES: [&str; 2] = ["proxy-", "sec-"];
28
29/// Whether the browser transport must drop `header_name` before building the
30/// request.
31pub fn is_forbidden_browser_request_header(header_name: &str) -> bool {
32    let normalized_name = header_name.trim().to_ascii_lowercase();
33    FORBIDDEN_BROWSER_REQUEST_HEADER_NAMES.contains(&normalized_name.as_str())
34        || FORBIDDEN_BROWSER_REQUEST_HEADER_PREFIXES
35            .iter()
36            .any(|prefix| normalized_name.starts_with(prefix))
37}
38
39#[cfg(test)]
40mod tests {
41    use super::*;
42
43    #[test]
44    fn every_reserved_name_is_forbidden_regardless_of_case_and_space() {
45        for reserved_name in FORBIDDEN_BROWSER_REQUEST_HEADER_NAMES {
46            assert!(is_forbidden_browser_request_header(reserved_name));
47            assert!(is_forbidden_browser_request_header(
48                &reserved_name.to_uppercase()
49            ));
50            assert!(is_forbidden_browser_request_header(&format!(
51                "  {reserved_name}\t"
52            )));
53        }
54    }
55
56    #[test]
57    fn reserved_prefixes_cover_the_sec_and_proxy_families() {
58        for prefixed_name in ["Sec-Fetch-Site", "sec-ch-ua", "Proxy-Authorization"] {
59            assert!(is_forbidden_browser_request_header(prefixed_name));
60        }
61    }
62
63    #[test]
64    fn ordinary_request_headers_are_forwarded() {
65        for allowed_name in ["Authorization", "Content-Type", "Accept", "X-Request-Id"] {
66            assert!(!is_forbidden_browser_request_header(allowed_name));
67        }
68    }
