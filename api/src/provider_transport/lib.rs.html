1//! The provider HTTP transport seam (round 19, M13): one request/response
2//! vocabulary and one [`HttpTransport`] contract that provider crates use on
3//! every target. Native targets get the `reqwest` implementation; the
4//! browser gets a `fetch` implementation, so the same provider request,
5//! parsing and error-mapping code runs in both places.
6//!
7//! On top of the transport the crate carries the plumbing every provider
8//! shares: the [`rate_limit::TokenBucketRateLimiter`] that paces outbound
9//! requests, and [`retry::execute_idempotent_get`] with its transient-error
10//! classification and back-off schedule. Providers keep their own
11//! status-to-[`SDKError`] maps — the statuses are vendor vocabulary.
12//!
13//! The seam deliberately exchanges plain owned bytes and strings — no
14//! provider crate ever names a platform HTTP type.
15

36/// The first HTTP status code that signals failure rather than content.
37pub const HTTP_ERROR_STATUS_CODE_THRESHOLD: u16 = 400;

39/// The HTTP methods providers use (the two the provider contracts need).
40#[derive(Debug, Clone, Copy, PartialEq, Eq)]
41pub enum HttpMethod {
42    Get,
43    Post,
44}
45
46/// One outbound request, independent of any platform HTTP client.
47#[derive(Debug, Clone)]
48pub struct HttpRequest {
49    /// The method the request uses.
50    pub method: HttpMethod,
51    /// The fully built absolute URL, query string included.
52    pub url: String,
53    /// Additional request headers (`(name, value)` pairs).
54    pub headers: Vec<(String, String)>,
55    /// An optional request body (JSON bytes for POSTs).
56    pub body: Option<Vec<u8>>,
57}
58
59/// One inbound response: status, headers and the raw body bytes.
60#[derive(Debug, Clone)]
61pub struct HttpResponse {
62    /// The HTTP status code.
63    pub status_code: u16,
64    /// Response headers (`(name, value)` pairs); the browser transport omits
65    /// them (providers only read status and body).
66    pub headers: Vec<(String, String)>,
67    /// The raw response body.
68    pub body: Vec<u8>,
69}
70
71/// A transport-level failure (as opposed to an HTTP status the provider maps
72/// into its typed error vocabulary).
73#[derive(Debug, Clone, thiserror::Error)]
74pub enum TransportError {
75    /// The request could not complete (DNS, TLS, connection, fetch refusal).
76    #[error("network request failed: {0}")]
77    Network(String),
78    /// The request exceeded the transport's deadline.
79    #[error("request timed out: {0}")]
80    Timeout(String),
81}
82
83/// The transport contract every provider crate executes requests through.
84#[async_trait]
85pub trait HttpTransport: Send + Sync {
86    /// Executes one request, returning status, headers and body.
87    async fn execute(&self, request: HttpRequest) -> Result<HttpResponse, TransportError>;
88}

97/// A portable sleep used by provider retry back-off and the rate limiter:
98/// `tokio::time` natively, a browser timer on wasm. The browser timer's
99/// future is not `Send`, so the wait is bridged through a oneshot channel and
100/// the timer callback; the awaited half stays `Send`.
101pub async fn sleep(duration: Duration) {
102    #[cfg(not(target_arch = "wasm32"))]
103    tokio::time::sleep(duration).await;
104    #[cfg(target_arch = "wasm32")]
105    {
106        // Browser timers take millisecond integers; clamp the value to the
107        // timer's representable range.
108        const MILLISECONDS_MAXIMUM: u128 = u32::MAX as u128;
109        let milliseconds = duration.as_millis().min(MILLISECONDS_MAXIMUM) as u32;
110        let (wake_sender, wake_receiver) = futures::channel::oneshot::channel::<()>();
111        gloo_timers::callback::Timeout::new(milliseconds, move || {
112            let _ = wake_sender.send(());
113        })
114        .forget();
115        let _ = wake_receiver.await;
116    }
117}

119/// The target's default transport: the `reqwest` implementation natively,
120/// browser `fetch` on wasm.
121pub fn default_transport() -> Result<Arc<dyn HttpTransport>, SDKError> {
122    #[cfg(not(target_arch = "wasm32"))]
123    {
124        let transport = ReqwestTransport::new()
125            .map_err(|error| SDKError::NetworkRequestFailed(error.to_string()))?;
126        Ok(Arc::new(transport))
127    }
128    #[cfg(target_arch = "wasm32")]
129    {
130        Ok(Arc::new(BrowserFetchTransport::new()))
131    }
