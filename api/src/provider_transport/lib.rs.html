1//! The provider HTTP transport seam (round 19, M13): one request/response
2//! vocabulary and one [`HttpTransport`] contract that provider crates use on
3//! every target. Native targets get the `reqwest` implementation; the
4//! browser gets a `fetch` implementation, so the same provider request,
5//! parsing and error-mapping code runs in both places.
6//!
7//! The seam deliberately exchanges plain owned bytes and strings — no
8//! provider crate ever names a platform HTTP type.
9
10use async_trait::async_trait;
11use std::time::Duration;
12
13#[cfg(target_arch = "wasm32")]
14mod browser_transport;
15#[cfg(not(target_arch = "wasm32"))]
16mod native_transport;
17
18#[cfg(target_arch = "wasm32")]
19pub use browser_transport::BrowserFetchTransport;
20#[cfg(not(target_arch = "wasm32"))]
21pub use native_transport::ReqwestTransport;
22
23/// The HTTP methods providers use (the two the provider contracts need).
24#[derive(Debug, Clone, Copy, PartialEq, Eq)]
25pub enum HttpMethod {
26    Get,
27    Post,
28}
29
30/// One outbound request, independent of any platform HTTP client.
31#[derive(Debug, Clone)]
32pub struct HttpRequest {
33    /// The method the request uses.
34    pub method: HttpMethod,
35    /// The fully built absolute URL, query string included.
36    pub url: String,
37    /// Additional request headers (`(name, value)` pairs).
38    pub headers: Vec<(String, String)>,
39    /// An optional request body (JSON bytes for POSTs).
40    pub body: Option<Vec<u8>>,
41}
42
43/// One inbound response: status, headers and the raw body bytes.
44#[derive(Debug, Clone)]
45pub struct HttpResponse {
46    /// The HTTP status code.
47    pub status_code: u16,
48    /// Response headers (`(name, value)` pairs); the browser transport omits
49    /// them (providers only read status and body).
50    pub headers: Vec<(String, String)>,
51    /// The raw response body.
52    pub body: Vec<u8>,
53}
54
55/// A transport-level failure (as opposed to an HTTP status the provider maps
56/// into its typed error vocabulary).
57#[derive(Debug, Clone, thiserror::Error)]
58pub enum TransportError {
59    /// The request could not complete (DNS, TLS, connection, fetch refusal).
60    #[error("network request failed: {0}")]
61    Network(String),
62    /// The request exceeded the transport's deadline.
63    #[error("request timed out: {0}")]
64    Timeout(String),
65}
66
67/// The transport contract every provider crate executes requests through.
68#[async_trait]
69pub trait HttpTransport: Send + Sync {
70    /// Executes one request, returning status, headers and body.
71    async fn execute(&self, request: HttpRequest) -> Result<HttpResponse, TransportError>;
72}
73
74/// The connect deadline the native transport applies.
75#[cfg(not(target_arch = "wasm32"))]
76pub const CONNECT_TIMEOUT: Duration = Duration::from_secs(10);
77/// The whole-request deadline the native transport applies.
78#[cfg(not(target_arch = "wasm32"))]
79pub const REQUEST_TIMEOUT: Duration = Duration::from_secs(30);

81/// A portable sleep used by provider retry back-off and the rate limiter:
82/// `tokio::time` natively, a browser timer on wasm. The browser timer's
83/// future is not `Send`, so the wait is bridged through a oneshot channel and
84/// the timer callback; the awaited half stays `Send`.
85pub async fn sleep(duration: Duration) {
86    #[cfg(not(target_arch = "wasm32"))]
87    tokio::time::sleep(duration).await;
88    #[cfg(target_arch = "wasm32")]
89    {
90        // Browser timers take millisecond integers; clamp the value to the
91        // timer's representable range.
92        const MILLISECONDS_MAXIMUM: u128 = u32::MAX as u128;
93        let milliseconds = duration.as_millis().min(MILLISECONDS_MAXIMUM) as u32;
94        let (wake_sender, wake_receiver) = futures::channel::oneshot::channel::<()>();
95        gloo_timers::callback::Timeout::new(milliseconds, move || {
96            let _ = wake_sender.send(());
97        })
98        .forget();
99        let _ = wake_receiver.await;
100    }
101}
