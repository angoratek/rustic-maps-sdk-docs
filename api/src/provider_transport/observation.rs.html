1//! Provider-request observation: the record a console shows for one provider
2//! call, and the sink the SDK installs to receive it.
3//!
4//! The observation is built here rather than in a provider crate so the
5//! redaction rule has one home: a request path keeps its structure, query
6//! values are dropped except the documented non-secret parameters, and the
7//! response snippet is capped at [`RESPONSE_SNIPPET_MAXIMUM_BYTES`] on a
8//! character boundary. A credential reaches this module as part of the request
9//! and never leaves it — the tests feed one through a request and assert the
10//! observation carries neither the value nor the parameter pair.
11//!
12//! The sink is process-wide because the provider crates are stateless with
13//! respect to the SDK: the native binding and the browser façade each install
14//! one observer at startup, and a provider crate used on its own (a unit test,
15//! a stand-alone integration) runs with none installed and pays no cost.
16

24/// Maximum length of the response snippet in an observation, in bytes,
25/// counted on character boundaries so a multi-byte glyph is never split.
26pub const RESPONSE_SNIPPET_MAXIMUM_BYTES: usize = 512;
27
28/// Query parameters whose values may appear in an observation because they
29/// describe the request shape rather than carry data or a secret. Every other
30/// parameter's value is dropped; its name is not.
31pub const OBSERVABLE_QUERY_PARAMETER_NAMES: [&str; 4] =
32    ["format", "overview", "geometries", "units"];
33
34/// The status class an observation reports when no HTTP response arrived.
35pub const TRANSPORT_FAILURE_STATUS_CLASS: &str = "transport-failure";
36
37/// The outcome an observation reports for a `2xx` response.
38pub const OUTCOME_SUCCESS: &str = "success";
39/// The outcome an observation reports for a `3xx` response.
40pub const OUTCOME_REDIRECT: &str = "redirect";
41/// The outcome an observation reports for a `4xx` response.
42pub const OUTCOME_CLIENT_ERROR: &str = "client-error";
43/// The outcome an observation reports for a `5xx` response.
44pub const OUTCOME_SERVER_ERROR: &str = "server-error";
45/// The outcome an observation reports when no HTTP response arrived.
46pub const OUTCOME_TRANSPORT_FAILURE: &str = "transport-failure";
47/// The outcome an observation reports for a status class it does not
48/// recognize.
49pub const OUTCOME_UNCLASSIFIED: &str = "unclassified";

51/// The `tracing` target every provider-request diagnostic event carries: the
52/// diagnostics sink keys its ring by this target, and the event's module path
53/// is the transport crate's own.
54pub const PROVIDER_REQUEST_DIAGNOSTIC_TARGET: &str = "provider_transport";

56/// One provider call, already redacted.
57#[derive(Debug, Clone, PartialEq, Eq)]
58pub struct ProviderRequestObservation {
59    /// The provider contract the call served, for example `RoutingProvider`.
60    pub provider_contract: String,
61    /// The registry key of the serving provider.
62    pub provider_key: String,
63    /// The HTTP method as a two-to-four-letter word (`GET`, `POST`).
64    pub http_method: String,
65    /// The redacted request path, query names included and values removed.
66    pub request_path: String,
67    /// `2xx`, `3xx`, `4xx`, `5xx` or [`TRANSPORT_FAILURE_STATUS_CLASS`].
68    pub status_class: String,
69    /// Wall-clock duration of the attempt, in milliseconds.
70    pub duration_milliseconds: u64,
71    /// Response payload size in bytes; zero when no response arrived.
72    pub response_bytes: u64,
73    /// The response prefix, capped at [`RESPONSE_SNIPPET_MAXIMUM_BYTES`].
74    pub response_snippet: String,
75}
76
77impl ProviderRequestObservation {
78    /// The outcome word for this observation, derived from its status class:
79    /// [`OUTCOME_SUCCESS`], [`OUTCOME_REDIRECT`], [`OUTCOME_CLIENT_ERROR`],
80    /// [`OUTCOME_SERVER_ERROR`], [`OUTCOME_TRANSPORT_FAILURE`] or
81    /// [`OUTCOME_UNCLASSIFIED`].
82    pub fn outcome(&self) -> &'static str {
83        outcome_text(&self.status_class)
84    }
85}

87/// Receives one observation per completed provider call.
88pub trait ProviderRequestObserver: Send + Sync {
89    /// Records one provider call. Called synchronously on the request's
90    /// task, so an implementation must not block.
91    fn on_provider_request(&self, observation: ProviderRequestObservation);
92}

94/// A transport decorator that reports every attempt to the installed
95/// observer before returning its result.
96///
97/// A provider client wraps its transport once at construction, so every call
98/// through that client is observed with the same contract and key without
99/// any request-building site changing. Retries are observed per attempt,
100/// which is what a reader debugging a failure needs to see.
101pub struct ObservedTransport {
102    /// The transport that performs the request.
103    inner: Arc<dyn crate::HttpTransport>,
104    /// The contract the calls through this transport serve.
105    provider_contract: String,
106    /// The registry key of the provider that owns this transport.
107    provider_key: String,
108}
109
110impl ObservedTransport {
111    /// Wraps a transport with the provider identity its calls serve.
112    pub fn new(
113        inner: Arc<dyn crate::HttpTransport>,
114        provider_contract: &str,
115        provider_key: &str,
116    ) -> Self {
117        Self {
118            inner,
119            provider_contract: provider_contract.to_string(),
120            provider_key: provider_key.to_string(),
121        }
122    }
123}

125#[async_trait::async_trait]
126impl crate::HttpTransport for ObservedTransport {
127    async fn execute(
128        &self,
129        request: crate::HttpRequest,
130    ) -> Result<crate::HttpResponse, crate::TransportError> {
131        let http_method = request.method;
132        let request_url = request.url.clone();
133        let started_at = web_time::Instant::now();
134        let outcome = self.inner.execute(request).await;
135        let duration = started_at.elapsed();
136        let observed_outcome = outcome
137            .as_ref()
138            .ok()
139            .map(|response| (response.status_code, response.body.as_slice()));
140        observe_provider_request(
141            &self.provider_contract,
142            &self.provider_key,
143            http_method,
144            &request_url,
145            observed_outcome,
146            duration,
147        );
148        outcome
149    }
150}

152/// The installed observer, if any.
153static INSTALLED_OBSERVER: OnceLock<Mutex<Option<Arc<dyn ProviderRequestObserver>>>> =
154    OnceLock::new();
155
156/// Installs the observer that receives every provider call made afterwards.
157///
158/// Replacing an installed observer is allowed: a binding constructs its SDK
159/// more than once in the same process, and the last installed observer is the
160/// one that matters. Passing no observer (`None`) stops delivery.
161pub fn install_provider_request_observer(observer: Option<Arc<dyn ProviderRequestObserver>>) {
162    let installed_observer = INSTALLED_OBSERVER.get_or_init(|| Mutex::new(None));
163    *installed_observer
164        .lock()
165        .unwrap_or_else(std::sync::PoisonError::into_inner) = observer;
166}

168/// A [`ProviderRequestObserver`] that publishes every observation as one
169/// structured `tracing` event under [`PROVIDER_REQUEST_DIAGNOSTIC_TARGET`].
170///
171/// The event carries the provider key, the contract, the HTTP method, the
172/// redacted request path, the status class, the duration, the response byte
173/// count and the derived outcome. Because it is the observation the transport
174/// already redacted, a credential cannot reach the event; the SDK's
175/// diagnostics sink records the event like any other SDK log line, so a
176/// request observation is visible through the existing diagnostics
177/// subscription and flush seams.
178///
179/// A provider crate used on its own never installs this bridge and pays no
180/// logging cost; each SDK seam installs it once at construction through
181/// [`install_tracing_observation_bridge`].
182#[derive(Debug, Default)]
183pub struct TracingObservationBridge;
184
185impl ProviderRequestObserver for TracingObservationBridge {
186    fn on_provider_request(&self, observation: ProviderRequestObservation) {
187        let outcome = observation.outcome();
188        tracing::info!(
189            target: PROVIDER_REQUEST_DIAGNOSTIC_TARGET,
190            provider_key = observation.provider_key.as_str(),
191            provider_contract = observation.provider_contract.as_str(),
192            http_method = observation.http_method.as_str(),
193            request_path = observation.request_path.as_str(),
194            status_class = observation.status_class.as_str(),
195            outcome,
196            duration_milliseconds = observation.duration_milliseconds,
197            response_bytes = observation.response_bytes,
198            "provider request completed"
199        );
200    }
201}

203/// Installs [`TracingObservationBridge`] as the process's provider-request
204/// observer, replacing any observer installed earlier (the slot's documented
205/// last-one-wins rule).
206pub fn install_tracing_observation_bridge() {
207    install_provider_request_observer(Some(Arc::new(TracingObservationBridge)));
208}

210/// Builds one redacted observation for a completed attempt and hands it to
211/// the installed observer. Returns immediately when none is installed.
212///
213/// # Parameters
214/// - `provider_contract` — the contract the call served.
215/// - `provider_key` — the serving provider's registry key.
216/// - `http_method` — the request's method.
217/// - `request_url` — the fully built request URL, before redaction.
218/// - `outcome` — the status code and body, when a response arrived.
219/// - `duration` — how long the attempt took.
220pub fn observe_provider_request(
221    provider_contract: &str,
222    provider_key: &str,
223    http_method: HttpMethod,
224    request_url: &str,
225    outcome: Option<(u16, &[u8])>,
226    duration: Duration,
227) {
228    deliver_provider_request_observation(provider_request_observation(
229        provider_contract,
230        provider_key,
231        http_method,
232        request_url,
233        outcome,
234        duration,
235    ));
236}

238/// Builds one [`ProviderRequestObservation`] without delivering it.
239///
240/// A caller that reports through a different transport — the provider layer
241/// has its own observer — can build the record here and hand it to
242/// [`deliver_provider_request_observation`], so both transports publish the
243/// same vocabulary.
244pub fn provider_request_observation(
245    provider_contract: &str,
246    provider_key: &str,
247    http_method: HttpMethod,
248    request_url: &str,
249    outcome: Option<(u16, &[u8])>,
250    duration: Duration,
251) -> ProviderRequestObservation {
252    let (status_class, response_bytes, response_snippet) = match outcome {
253        Some((status_code, body)) => (
254            status_class_text(status_code),
255            body.len() as u64,
256            response_snippet_text(body),
257        ),
258        None => (TRANSPORT_FAILURE_STATUS_CLASS.to_string(), 0, String::new()),
259    };
260    ProviderRequestObservation {
261        provider_contract: provider_contract.to_string(),
262        provider_key: provider_key.to_string(),
263        http_method: http_method_text(http_method).to_string(),
264        request_path: redacted_request_path(request_url),
265        status_class,
266        duration_milliseconds: duration.as_millis().min(u64::MAX as u128) as u64,
267        response_bytes,
268        response_snippet,
269    }
270}
271
272/// Hands one observation to the installed observer, if any.
273pub fn deliver_provider_request_observation(observation: ProviderRequestObservation) {
274    if let Some(observer) = installed_observer() {
275        observer.on_provider_request(observation);
276    }
277}

279/// Reduces a request URL to what an observation may show: the path, plus the
280/// names of its query parameters with values removed except the parameters
281/// whose value describes the request shape.
282pub fn redacted_request_path(request_url: &str) -> String {
283    let Ok(parsed_url) = Url::parse(request_url) else {
284        // An unparseable URL never reaches a provider in practice; refuse to
285        // echo it verbatim, because it may carry a credential in any shape.
286        return String::from("<unparseable request url>");
287    };
288    let mut redacted_path = parsed_url.path().to_string();
289    let mut redacted_parameters: Vec<String> = parsed_url
290        .query_pairs()
291        .map(|(parameter_name, parameter_value)| {
292            if OBSERVABLE_QUERY_PARAMETER_NAMES.contains(&parameter_name.as_ref()) {
293                format!("{parameter_name}={parameter_value}")
294            } else {
295                format!("{parameter_name}=<redacted>")
296            }
297        })
298        .collect();
299    redacted_parameters.sort();
300    if !redacted_parameters.is_empty() {
301        redacted_path.push('?');
302        redacted_path.push_str(&redacted_parameters.join("&"));
303    }
304    redacted_path
305}

307/// Caps a response body at [`RESPONSE_SNIPPET_MAXIMUM_BYTES`] on a character
308/// boundary, so a multi-byte glyph is never split.
309pub fn response_snippet_text(response_body: &[u8]) -> String {
310    let text = String::from_utf8_lossy(response_body);
311    if text.len() <= RESPONSE_SNIPPET_MAXIMUM_BYTES {
312        return text.into_owned();
313    }
314    let mut snippet_end = RESPONSE_SNIPPET_MAXIMUM_BYTES;
315    while !text.is_char_boundary(snippet_end) {
316        snippet_end -= 1;
317    }
318    text[..snippet_end].to_string()
319}

321/// The status class an observation reports for one HTTP status code.
322fn status_class_text(status_code: u16) -> String {
323    let status_class_prefix = status_code / 100;
324    format!("{status_class_prefix}xx")
325}
326
327/// The outcome word an observation reports for one status class.
328fn outcome_text(status_class: &str) -> &'static str {
329    match status_class {
330        "2xx" => OUTCOME_SUCCESS,
331        "3xx" => OUTCOME_REDIRECT,
332        "4xx" => OUTCOME_CLIENT_ERROR,
333        "5xx" => OUTCOME_SERVER_ERROR,
334        TRANSPORT_FAILURE_STATUS_CLASS => OUTCOME_TRANSPORT_FAILURE,
335        _ => OUTCOME_UNCLASSIFIED,
336    }
337}
338
339/// The word an observation reports for one HTTP method.
340fn http_method_text(http_method: HttpMethod) -> &'static str {
341    match http_method {
342        HttpMethod::Get => "GET",
343        HttpMethod::Post => "POST",
344    }
345}

347/// The installed observer, cloned out of the process-wide slot so delivery
348/// never holds the lock across the observer's own work.
349fn installed_observer() -> Option<Arc<dyn ProviderRequestObserver>> {
350    INSTALLED_OBSERVER
351        .get()?
352        .lock()
353        .unwrap_or_else(std::sync::PoisonError::into_inner)
354        .clone()
355}

361    /// The installed observer is process-wide, so the tests that install one
362    /// take this lock: two of them running at once would see each other's
363    /// observations (a parallel `cargo test` run did exactly that).
364    ///
365    /// The lock is async-aware because some of these tests await the transport
366    /// while they hold it; a standard-library guard must never be held across

373    }
