1//! The provider-failure diagnostic seam (round 12, addition-only; the owner
2//! ruling of 2026-10-29 on unclassified foreign provider failures): the
3//! bounded per-`MapSdk` ring [`ProviderFailureDiagnosticsRing`] recording
4//! every failed foreign-callback provider call, read back through
5//! `MapSdk::latest_provider_failure_diagnostics`.
6//!
7//! The write sites are the [`crate::foreign_providers`] adapters — the only
8//! boundary that sees both what the foreign implementation threw and the
9//! contract/method it occurred on. When the failure was the round-11
10//! unclassified-callback conversion's subject, the adapter classifies it
11//! (through the crate's `SdkError::unclassified_foreign_callback_reason`)
12//! and records the callback machinery's verbatim reason; otherwise the
13//! failure was a typed `SdkError` round trip and the entry carries its
14//! verbatim display form. Nothing is parsed or invented beyond that text —
15//! see the UDL seam section for the honest limitation.
16//!
17//! The ring is per `MapSdk` (one registration passes the owning object's
18//! ring into the adapters, mirroring how the adapters wrap exactly the
19//! callbacks registered on it), bounded so a systematically failing foreign
20//! provider cannot grow the process's memory unboundedly, and read-only from
21//! the foreign side: the readback neither records nor clears.
22
23use std::collections::VecDeque;
24use std::sync::{Arc, Mutex};
25
26use crate::records::{ProviderFailureClassification, ProviderFailureDiagnostic};
27
28/// Capacity of a [`ProviderFailureDiagnosticsRing`], in entries (bounded so
29/// a systematically failing foreign provider cannot grow the process's
30/// memory unboundedly; the newest failures are what integrators classify).
31pub(crate) const PROVIDER_FAILURE_DIAGNOSTIC_RING_CAPACITY: usize = 32;
32
33/// The bounded ring of provider-failure diagnostics of one `MapSdk`: every
34/// failed foreign-callback provider call registered on the object lands
35/// here, and `MapSdk::latest_provider_failure_diagnostics` reads it back.
36#[derive(Debug)]
37pub struct ProviderFailureDiagnosticsRing {
38    ring: Mutex<VecDeque<ProviderFailureDiagnostic>>,
39}
40
41impl ProviderFailureDiagnosticsRing {
42    /// Builds the empty ring.
43    pub(crate) fn new() -> Self {
44        Self {
45            ring: Mutex::new(VecDeque::with_capacity(
46                PROVIDER_FAILURE_DIAGNOSTIC_RING_CAPACITY,
47            )),
48        }
49    }
50
51    /// Records one failed foreign-callback provider call (ring retention: at
52    /// [`PROVIDER_FAILURE_DIAGNOSTIC_RING_CAPACITY`] the oldest entry
53    /// drops).
54    pub(crate) fn record(
55        &self,
56        provider_contract_name: &str,
57        provider_method_name: &str,
58        failure_classification: ProviderFailureClassification,
59        failure_detail: String,
60    ) {
61        let diagnostic = ProviderFailureDiagnostic {
62            timestamp_milliseconds_since_epoch: crate::diagnostics::unix_epoch_milliseconds(),
63            failure_classification,
64            provider_contract_name: provider_contract_name.to_string(),
65            provider_method_name: provider_method_name.to_string(),
66            failure_detail,
67        };
68        let mut ring = crate::panic_boundary::state_lock_guard(
69            &self.ring,
70            "the provider-failure diagnostics ring",
71        );
72        if ring.len() >= PROVIDER_FAILURE_DIAGNOSTIC_RING_CAPACITY {
73            ring.pop_front();
74        }
75        ring.push_back(diagnostic);
76    }
77
78    /// Reads the retained diagnostics, oldest first; the readback is
79    /// non-destructive (a later read returns the same entries, plus any
80    /// recorded since).
81    pub(crate) fn latest(&self) -> Vec<ProviderFailureDiagnostic> {
82        let ring = crate::panic_boundary::state_lock_guard(
83            &self.ring,
84            "the provider-failure diagnostics ring",
85        );
86        ring.iter().cloned().collect()
87    }
88}
89
90impl Default for ProviderFailureDiagnosticsRing {
91    /// Mirrors the inline `ProviderFailureDiagnosticsRing::new` constructor
92    /// (the clippy `new_without_default` convention).
93    fn default() -> Self {
94        Self::new()
95    }
96}
97
98/// Convenience alias for the shared-ring handle the registration paths pass
99/// into the foreign-callback adapters.
100pub(crate) type SharedProviderFailureDiagnostics = Arc<ProviderFailureDiagnosticsRing>;

102#[cfg(test)]
103mod tests {
104    use super::*;
105
106    /// A failure detail the recording tests use.
107    const TEST_FAILURE_DETAIL: &str = "the foreign geocoder rejected the request";
108
109    /// Records `count` entries of the given classification into `ring`,
110    /// each with its index as the detail suffix.
111    fn record_test_entries(
112        ring: &ProviderFailureDiagnosticsRing,
113        count: usize,
114        classification: ProviderFailureClassification,
115    ) {
116        for index in 0..count {
117            ring.record(
118                "GeocodingProvider",
119                "forward_geocode",
120                classification,
121                format!("{TEST_FAILURE_DETAIL} {index}"),
122            );
123        }
124    }
125
126    /// The ring reads back oldest first and is non-destructive.
127    #[test]
128    fn ring_reads_back_oldest_first_and_is_non_destructive() {
129        let ring = ProviderFailureDiagnosticsRing::new();
130        record_test_entries(&ring, 3, ProviderFailureClassification::MappedContractError);
131        let first_read = ring.latest();
132        let read_details: Vec<&str> = first_read
133            .iter()
134            .map(|diagnostic| diagnostic.failure_detail.as_str())
135            .collect();
136        assert_eq!(
137            read_details,
138            [
139                "the foreign geocoder rejected the request 0",
140                "the foreign geocoder rejected the request 1",
141                "the foreign geocoder rejected the request 2",
142            ],
143            "entries read back in recording order"
144        );
145        let second_read = ring.latest();
146        assert_eq!(
147            second_read, first_read,
148            "the readback never clears the ring"
149        );
150    }
151
152    /// The ring rotates at its named capacity: at capacity the oldest entry
153    /// drops, so exactly the newest [`PROVIDER_FAILURE_DIAGNOSTIC_RING_CAPACITY`]
154    /// entries are retained.
155    #[test]
156    fn ring_rotates_at_capacity_keeping_the_newest_entries() {
157        let ring = ProviderFailureDiagnosticsRing::new();
158        let over_capacity_count = PROVIDER_FAILURE_DIAGNOSTIC_RING_CAPACITY + 1;
159        record_test_entries(
160            &ring,
161            over_capacity_count,
162            ProviderFailureClassification::UnclassifiedCallbackFailure,
163        );
164        let retained = ring.latest();
165        assert_eq!(retained.len(), PROVIDER_FAILURE_DIAGNOSTIC_RING_CAPACITY);
166        assert_eq!(
167            retained[0].failure_detail,
168            format!("{TEST_FAILURE_DETAIL} 1"),
169            "the oldest entry (index 0) dropped at capacity"
170        );
171        assert_eq!(
172            retained
173                .last()
174                .expect("the ring holds capacity entries")
175                .failure_detail,
176            format!(
177                "{TEST_FAILURE_DETAIL} {over_capacity_count_minus_one}",
178                over_capacity_count_minus_one = over_capacity_count - 1
179            ),
180            "the newest entry is retained"
181        );
182        for diagnostic in &retained {
183            assert_eq!(
184                diagnostic.failure_classification,
185                ProviderFailureClassification::UnclassifiedCallbackFailure,
186            );
187        }
188    }
189
190    /// The recorded fields carry the contract and method names verbatim
191    /// alongside the classification.
192    #[test]
193    fn recorded_entries_carry_contract_and_method_names() {
194        let ring = ProviderFailureDiagnosticsRing::new();
195        ring.record(
196            "RasterTileProvider",
197            "fetch_raster_tile",
198            ProviderFailureClassification::MappedContractError,
199            TEST_FAILURE_DETAIL.to_string(),
200        );
201        let diagnostic = ring.latest().pop().expect("one recorded entry");
202        assert_eq!(diagnostic.provider_contract_name, "RasterTileProvider");
203        assert_eq!(diagnostic.provider_method_name, "fetch_raster_tile");
204        assert_eq!(diagnostic.failure_detail, TEST_FAILURE_DETAIL);
205    }
206}
