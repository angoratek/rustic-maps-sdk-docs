1//! The M17 engine facades (addition-only contract slice): typed capability
2//! objects constructed over an existing [`MapSdk`].
3//!
4//! Each engine holds the very same [`MapSdk`] the application already owns —
5//! one provider registry, one tile-cache installation, one diagnostics ring
6//! set — and delegates every member to the matching `MapSdk` method
7//! verbatim. The engines add no validation, no mapping and no error
8//! rewriting of their own: a typed failure is the SDK's typed failure with
9//! the same variant and the same detail message, so an integrator moving
10//! from `map_sdk.search(...)` to `search_engine.search(...)` observes no
11//! behavioural change. Their point is vocabulary: the same engine names
12//! reach Kotlin, Swift, React Native and the wasm mirror through one UDL
13//! contract, and an application holds one `MapSdk` plus one engine per
14//! capability instead of a single object with every call on it.
15//!
16//! The async members await the SDK's runtime-independent futures directly,
17//! so the exported futures stay `Send` (the provider reactor work already
18//! hands back through [`crate::runtime::drive`]).
19

30/// Search and geocoding facade over a [`MapSdk`]'s registered provider
31/// contracts.
32pub struct SearchEngine {
33    /// The SDK whose provider registry serves this engine's calls.
34    map_sdk: Arc<MapSdk>,
35}
36
37impl SearchEngine {
38    /// Builds the engine over the SDK whose provider registry serves its
39    /// calls.
40    pub fn new(map_sdk: Arc<MapSdk>) -> Self {
41        Self { map_sdk }
42    }
43
44    /// Returns full search results for a settled query; the boundary
45    /// restricts the search when given. Delegates to [`MapSdk::search`].
46    pub async fn search(
47        &self,
48        query_text: String,
49        search_boundary: Option<BoundingBox>,
50    ) -> Result<Vec<Place>, SdkError> {
51        self.map_sdk.search(query_text, search_boundary).await
52    }
53
54    /// Returns fast completion candidates while the user types; the
55    /// boundary restricts the search when given. Delegates to
56    /// [`MapSdk::suggest`].
57    pub async fn suggest(
58        &self,
59        query_text: String,
60        search_boundary: Option<BoundingBox>,
61    ) -> Result<Vec<Place>, SdkError> {
62        self.map_sdk.suggest(query_text, search_boundary).await
63    }
64
65    /// Resolves a coordinate into candidate places describing that
66    /// position. Delegates to [`MapSdk::reverse_geocode`].
67    pub async fn reverse_geocode(
68        &self,
69        coordinate: GeographicCoordinate,
70    ) -> Result<Vec<Place>, SdkError> {
71        self.map_sdk.reverse_geocode(coordinate).await
72    }
73}

140/// Diagnostics facade over a [`MapSdk`]'s structured log-event stream and
141/// bounded ring buffers.
142pub struct Diagnostics {
143    /// The SDK whose diagnostics this facade observes.
144    map_sdk: Arc<MapSdk>,
145}
146
147impl Diagnostics {
148    /// Builds the facade over the SDK whose diagnostics it observes.
149    pub fn new(map_sdk: Arc<MapSdk>) -> Self {
150        Self { map_sdk }
151    }
152
153    /// Subscribes the foreign listener to the SDK's structured internal log
154    /// events at `minimum_severity` or above. Delegates to
155    /// [`MapSdk::subscribe_to_diagnostic_events`].
156    pub fn add_listener(
157        &self,
158        listener: Arc<dyn SdkDiagnosticEventListener>,
159        minimum_severity: SdkLogSeverity,
160    ) {
161        self.map_sdk
162            .subscribe_to_diagnostic_events(listener, minimum_severity);
163    }
164
165    /// Drains up to `maximum_count` of the most recent ring events at
166    /// `minimum_severity` or above through the listener, oldest first, and
167    /// answers how many were delivered. Delegates to
168    /// [`MapSdk::flush_recent_diagnostic_events`].
169    pub fn flush_recent(
170        &self,
171        listener: Arc<dyn SdkDiagnosticEventListener>,
172        minimum_severity: SdkLogSeverity,
173        maximum_count: u32,
174    ) -> u32 {
175        self.map_sdk
176            .flush_recent_diagnostic_events(listener, minimum_severity, maximum_count)
177    }
178}

350        match offline_manager.tile_cache_terms() {

361        }

364        let manager_with_cache = OfflineManager::new(Arc::clone(&sdk_with_cache.0));
365        match manager_with_cache.set_tile_time_to_live(0) {
366            Err(SdkError::InvalidCoordinateInput { detail }) => {
367                assert!(detail.contains("must be positive"), "{detail}");

370        }
371        std::fs::remove_dir_all(&sdk_with_cache.1).expect("the test cache directory cleans up");
372    }

399    /// Registers the MapQuest providers with a freshly named tile cache and
400    /// answers the SDK with the cache directory the test cleans up.
401    fn registered_with_labelled_tile_cache(test_label: &str) -> (Arc<MapSdk>, std::path::PathBuf) {
402        let map_sdk = MapSdk::new();
403        let cache_directory = std::env::temp_dir().join(format!(
404            "map-native-core-{test_label}-{}",
405            std::process::id()
406        ));
407        map_sdk
408            .set_provider_credential_with_tile_cache(
409                MAPQUEST_TEST_CONSUMER_KEY.to_string(),
410                cache_directory
411                    .join("tiles.sqlite")
412                    .to_string_lossy()
413                    .to_string(),
414                TEST_TILE_CACHE_BUDGET_BYTES,
415            )
416            .expect("an empty path in an existing directory opens the cache");
417        (Arc::new(map_sdk), cache_directory)
418    }

420    /// A diagnostics listener recording every delivered event and counting
421    /// its deliveries.
422    #[derive(Default)]
423    struct RecordingDiagnosticEventListener {
424        delivered_events: Mutex<Vec<LogEvent>>,
425        delivery_count: AtomicU32,
426    }
427
428    impl SdkDiagnosticEventListener for RecordingDiagnosticEventListener {
429        fn on_diagnostic_event(&self, event: LogEvent) {
430            self.delivered_events.lock().unwrap().push(event);
431            self.delivery_count.fetch_add(1, Ordering::SeqCst);
432        }
433    }

435    #[test]
436    fn diagnostics_delegates_the_listener_and_the_bounded_flush() {
437        let diagnostics = Diagnostics::new(Arc::new(MapSdk::new()));
438        let listener = Arc::new(RecordingDiagnosticEventListener::default());
439        diagnostics.add_listener(
440            Arc::clone(&listener) as Arc<dyn SdkDiagnosticEventListener>,
441            SdkLogSeverity::Warn,
442        );
443        let delivered_count = diagnostics.flush_recent(
444            Arc::clone(&listener) as Arc<dyn SdkDiagnosticEventListener>,
445            SdkLogSeverity::Trace,
446            u32::MAX,
447        );
448        assert_eq!(
449            listener.delivery_count.load(Ordering::SeqCst),
450            delivered_count,
451            "the flush answers exactly the events it delivered"
452        );
453        assert_eq!(
454            listener.delivered_events.lock().unwrap().len() as u32,
455            delivered_count
456        );
457    }
