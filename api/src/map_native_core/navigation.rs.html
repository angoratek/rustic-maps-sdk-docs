1//! The navigation wiring surface (M6): the `NavigationStateEventListener`
2//! foreign-callback contract foreign code implements to observe the
3//! navigation state, the [`MapNavigationController`] driving a native
4//! `map-navigation` session behind it, and the policy records configuring
5//! the controller's step-advance and deviation strategies.
6//!
7//! The controller owns a [`NavigationStateBroadcaster`] and the single active
8//! [`NavigationSession`], and feeds the broadcaster's dispatch calls from the
9//! exported methods, so the per-topic subscription semantics, the delivery
10//! order and the outcome codes are exactly the native broadcaster's (see
11//! `map-navigation`'s `observer` module documentation).
12//!
13//! # What the FFI seam changes about the re-entrancy rules
14//!
15//! The broadcaster's own rules hold unchanged across the seam: the
16//! controller's active-trip lock is taken only to clone state snapshots and
17//! is released before the first dispatch, so the lock is never held across a
18//! listener callback and a listener may freely call back into the controller
19//! — including
20//! [`MapNavigationController::current_trip_state`] and subscription
21//! management. Two properties are worth naming at this seam:
22//!
23//! - A foreign listener calling back into a controller-providing method while
24//!   a dispatch from that controller is still in flight hits the
25//!   broadcaster's single-in-flight guard and its nested dispatch is
26//!   suppressed exactly as a Rust-side re-entrant dispatch would be; the
27//!   state change behind the suppressed dispatch has already been applied,
28//!   only its broadcast is skipped. The synchronous single-threaded callback
29//!   ordering of the native seam carries over: uniffi runs these sync
30//!   callbacks on the thread driving the controller call, so no cross-thread
31//!   queueing is introduced.
32//! - A panicking listener propagates out of the dispatch on the native seam.
33//!   The panic policy across the platform runtimes is the packaging slices'
34//!   concern; nothing is caught here.
35//!
36//! # Scope
37//!
38//! Trip progress and instructions are read-only for foreign code; the
39//! programmatic step advance is deliberately absent — the reducer has no such
40//! event yet, and the `Manual` step-advance policy consequently cannot be
41//! driven across this surface.
42

61/// State listener contract implemented by foreign code; register its
62/// subscriptions on [`MapNavigationController`] through the
63/// `subscribe_to_*` methods.
64///
65/// The callbacks exchange the binding records of the navigation state and
66/// have no error channel: a listener cannot fail into the broadcaster (the
67/// native dispatch rules). Each method of one registered subscription is
68/// called only for the topics the subscription was created for.
69pub trait NavigationStateEventListener: Send + Sync {
70    /// Receives the composed instruction payload of every dispatched state
71    /// change (the instructions topic).
72    fn on_navigation_instruction(&self, instruction: NavigationInstruction);
73
74    /// Receives a deviation-status stream update (the deviation topic):
75    /// off-route on entry and on every recorded measurement update while the
76    /// deviation persists, on-route when it clears.
77    fn on_deviation_status_change(&self, deviation_status_change: DeviationStatusChange);
78
79    /// Receives one phase transition per dispatched trip-state transition
80    /// (the phase topic).
81    fn on_navigation_phase_change(&self, navigation_phase_change: NavigationPhaseChange);
82
83    /// Receives one speech-trigger line the controller's pacing policy judged
84    /// newsworthy (round 10, the speech topic). Subscribers of the
85    /// `subscribe_to_speech_instructions` topic receive exactly these
86    /// deliveries; the composed speech text is carried verbatim and the
87    /// platform owns playback.
88    fn on_speech_instruction(&self, speech_instruction: SpeechInstruction);
89
90    /// Internally required by the uniffi foreign-handle machinery (the
91    /// foreign-handle probe of the uniffi 0.32 trait-object lowering);
92    /// Rust-side implementations keep the default, which reports a
93    /// Rust-implemented object.
94    #[doc(hidden)]
95    fn uniffi_foreign_handle(&self) -> Option<::uniffi::Handle> {
96        Option::None
97    }
98}

153/// Handle coupled to exactly one `(topic, listener)` subscription of the
154/// controller's broadcaster. Cancelling is explicit and idempotent; dropping
155/// the handle does not unsubscribe.
156pub struct NavigationTopicSubscription {
157    subscription: TopicSubscription,
158}
159
160impl NavigationTopicSubscription {
161    /// Wraps a native subscription handle.
162    pub(crate) fn new(subscription: TopicSubscription) -> Self {
163        Self { subscription }
164    }
165
166    /// Cancels this subscription: the listener stops receiving the
167    /// subscription's topic from the next dispatch on. Idempotent.
168    pub fn cancel(&self) {
169        self.subscription.cancel();
170    }
171}

173/// The one active trip: the native session plus the unit system its
174/// instruction payloads are composed in.
175struct ActiveTrip {
176    session: map_navigation::NavigationSession,
177    unit_system: UnitSystem,
178}

180/// The exported navigation controller: drives the native reducer's session,
181/// broadcasts every state change to the registered foreign listeners, and
182/// answers read-only state and instruction queries.
183///
184/// The active-trip lock is never held across a broadcaster dispatch (see the
185/// module-head notes on the re-entrancy rules across the seam).
186pub struct MapNavigationController {
187    active_trip: Mutex<Option<ActiveTrip>>,
188    broadcaster: NavigationStateBroadcaster,
189    /// The active trip's speech-trigger policy memory (`None` without a
190    /// trip). The lock is held only over the evaluation itself — never
191    /// across a broadcaster dispatch or a listener callback.
192    speech_planner: Mutex<Option<SpeechInstructionPlanner>>,
193}
194
195impl MapNavigationController {
196    /// Builds the controller with no active trip and no subscriptions.
197    pub fn new() -> Self {
198        Self {
199            active_trip: Mutex::new(None),
200            broadcaster: NavigationStateBroadcaster::new(),
201            speech_planner: Mutex::new(None),
202        }
203    }
204}

207    /// Builds the empty controller, exactly like [`MapNavigationController::new`].

248    /// Subscribes the listener to the speech-instruction trigger topic, in
249    /// registration order (round 10, OQ5).
250    pub fn subscribe_to_speech_instructions(
251        &self,
252        listener: Arc<dyn NavigationStateEventListener>,
253    ) -> Arc<NavigationTopicSubscription> {
254        self.subscribe(listener, |broadcaster, listener| {
255            broadcaster.subscribe_to_speech_utterances(listener)
256        })
257    }

259    /// Evaluates the active trip's speech-trigger policy over one dispatch's
260    /// state transition and delivers the utterances to the speech topic. A
261    /// suppressed state-change dispatch skips the evaluation — the policy
262    /// memory keeps its state, so the next delivered dispatch sees the
263    /// transition whole. The planner lock is held only over the evaluation
264    /// itself, never across a dispatch or listener callback.
265    fn dispatch_speech_policies(
266        &self,
267        previous_state: &map_navigation::TripNavigationState,
268        current_state: &map_navigation::TripNavigationState,
269        unit_system: UnitSystem,
270        dispatch_outcome: NavigationDispatchOutcome,
271    ) {
272        if dispatch_outcome != NavigationDispatchOutcome::Delivered {
273            return;
274        }
275        let mut speech_planner = self.speech_planner.lock().unwrap();
276        let Some(planner) = speech_planner.as_mut() else {
277            return;
278        };
279        let utterances = planner.evaluate_state_transition(
280            Some(previous_state),
281            current_state,
282            unit_system,
283            Instant::now(),
284        );
285        drop(speech_planner);
286        for utterance in &utterances {
287            self.broadcaster.dispatch_speech_utterance(utterance);
288        }
289    }

291    /// Registers the wrapped foreign listener on one topic list.
292    fn subscribe(
293        &self,
294        listener: Arc<dyn NavigationStateEventListener>,
295        topic_registration: impl FnOnce(
296            &NavigationStateBroadcaster,
297            Arc<dyn map_navigation::NavigationStateChangeListener>,
298        ) -> TopicSubscription,
299    ) -> Arc<NavigationTopicSubscription> {
300        let native_listener: Arc<dyn map_navigation::NavigationStateChangeListener> =
301            Arc::new(ForeignNavigationStateListener::new(listener));
302        Arc::new(NavigationTopicSubscription::new(topic_registration(
303            &self.broadcaster,
304            native_listener,
305        )))
306    }

308    /// Begins navigating `route` with the given policies, replacing any
309    /// active trip. The initial instruction payload is dispatched; no
310    /// deviation or phase transitions exist yet.
311    ///
312    /// A route without traversable steps represents an already-arrived trip
313    /// (the reducer's own semantics), not a rejection.
314    pub fn start_trip(
315        &self,
316        route: RouteData,
317        unit_system: crate::records::UnitSystem,
318        step_advance_policy: StepAdvancePolicy,
319        deviation_policy: DeviationPolicy,
320    ) -> Result<(), SdkError> {
321        let step_advance_condition = build_step_advance_condition(&step_advance_policy)?;
322        let deviation_detector = build_deviation_detector(&deviation_policy)?;
323        let native_route = route.into_native_route_data()?;
324        let native_unit_system = unit_system.to_native_unit_system();
325        let session = reduce_navigation_event(
326            // A fresh trip replaces any active session outright (the reducer
327            // reads no previous session for a StartTrip event).
328            None,
329            map_navigation::NavigationEvent::StartTrip {
330                route: native_route,
331                step_advance_condition,
332                deviation_detector,
333            },
334        )
335        .expect("a StartTrip event always produces a session");
336        let dispatched_state = session.state.clone();
337        *self.active_trip.lock().unwrap() = Some(ActiveTrip {
338            session,
339            unit_system: native_unit_system,
340        });
341        // The speech policy restarts with the trip: nothing is announced at
342        // trip start (the initial instruction payload is the dispatched
343        // instructions topic's own line, and the platforms speak it).
344        *self.speech_planner.lock().unwrap() = Some(SpeechInstructionPlanner::new());
345        self.broadcaster
346            .dispatch_trip_started(&dispatched_state, native_unit_system);
347        Ok(())
348    }

350    /// Delivers one position fix to the active trip; without one the fix is
351    /// dropped (the reducer's own semantics for event-before-trip).
352    /// The fix is judged for deviation, snapped and evaluated by the active
353    /// step-advance condition exactly as the native reducer documents, and
354    /// the resulting transition is broadcast. A broadcast suppressed by the
355    /// broadcaster's re-entrancy guard is not reported here; the state change
356    /// behind it has already been applied.
357    pub fn provide_location_update(&self, location_update: NavigationLocationUpdate) {
358        let native_update = location_update.into_native_location_update();
359        let (previous_state, current_state, unit_system) = {
360            // The lock guards only the session swap; it is released before the
361            // dispatch below, so a listener may call back into the controller.
362            let mut active_trip = self.active_trip.lock().unwrap();
363            let Some(trip) = active_trip.as_mut() else {
364                return;
365            };
366            let previous_state = trip.session.state.clone();
367            let session = reduce_navigation_event(
368                Some(trip.session.clone()),
369                map_navigation::NavigationEvent::LocationUpdated(native_update),
370            )
371            .expect("an active trip's location update always produces a session");
372            trip.session = session;
373            (previous_state, trip.session.state.clone(), trip.unit_system)
374        };
375        let dispatch_outcome =
376            self.broadcaster
377                .dispatch_location_update(&previous_state, &current_state, unit_system);
378        self.dispatch_speech_policies(
379            &previous_state,
380            &current_state,
381            unit_system,
382            dispatch_outcome,
383        );
384    }

386    /// Replaces the route of the active trip (a reroute). The replacement
387    /// route must be anchored at the trip's current snapped coordinate (the
388    /// reducer's anchoring contract; no trimming happens here). The step
389    /// advance strategy for the new route is configured by
390    /// `step_advance_policy`; the active deviation policy persists.
391    ///
392    /// A reroute offered to an arrived trip is ignored by the reducer and
393    /// produces no phase transition; without an active trip the event is
394    /// dropped.
395    pub fn provide_reroute(
396        &self,
397        route: RouteData,
398        step_advance_policy: StepAdvancePolicy,
399    ) -> Result<(), SdkError> {
400        let step_advance_condition = build_step_advance_condition(&step_advance_policy)?;
401        let native_route = route.into_native_route_data()?;
402        let (previous_state, current_state, unit_system, reroute_applied) = {
403            let mut active_trip = self.active_trip.lock().unwrap();
404            let Some(trip) = active_trip.as_mut() else {
405                return Ok(());
406            };
407            let previous_state = trip.session.state.clone();
408            let session = reduce_navigation_event(
409                Some(trip.session.clone()),
410                map_navigation::NavigationEvent::RerouteProvided {
411                    route: native_route,
412                    step_advance_condition,
413                },
414            )
415            .expect("an active trip's reroute always produces a session");
416            let reroute_applied = previous_state != session.state;
417            trip.session = session;
418            (
419                previous_state,
420                trip.session.state.clone(),
421                trip.unit_system,
422                reroute_applied,
423            )
424        };
425        let _ =
426            self.broadcaster
427                .dispatch_reroute_applied(&previous_state, &current_state, unit_system);
428        if reroute_applied {
429            // The route (and its step ladder) is new: the speech policy
430            // restarts.
431            *self.speech_planner.lock().unwrap() = Some(SpeechInstructionPlanner::new());
432        }
433        Ok(())
434    }

436    /// The plain state of the active trip, or `None` without one.
437    pub fn current_trip_state(&self) -> Option<crate::records::TripNavigationState> {
438        self.active_trip.lock().unwrap().as_ref().map(|trip| {
439            crate::records::TripNavigationState::from_native_trip_navigation_state(
440                &trip.session.state,
441            )
442        })
443    }
444
445    /// The composed turn-by-turn instruction payload of the current trip
446    /// state, or `None` without an active trip. Emission is pure in the
447    /// native core, so repeated reads of an unchanged state return the same
448    /// payload.
449    pub fn current_instructions(&self) -> Option<NavigationInstruction> {
450        let active_trip = self.active_trip.lock().unwrap();
451        let trip = active_trip.as_ref()?;
452        Some(NavigationInstruction::from_native_navigation_instruction(
453            &compose_navigation_instructions(&trip.session.state, trip.unit_system),
454        ))
455    }

458/// Builds the native step-advance condition behind a policy record, refusing
459/// non-finite or negative threshold values.
460pub(crate) fn build_step_advance_condition(
461    step_advance_policy: &StepAdvancePolicy,
462) -> Result<Arc<dyn StepAdvanceCondition>, SdkError> {
463    match step_advance_policy.policy_kind {
464        StepAdvancePolicyKind::Manual => Ok(Arc::new(ManualStepAdvanceCondition)),
465        StepAdvancePolicyKind::DistanceToEndOfStep => Ok(Arc::new(DistanceToEndOfStepCondition {
466            advance_threshold_meters: validated_policy_threshold(
467                step_advance_policy.advance_threshold_meters,
468                "step-advance distance threshold",
469            )?,
470            minimum_horizontal_accuracy_meters: validated_policy_threshold(
471                step_advance_policy.minimum_horizontal_accuracy_meters,
472                "step-advance minimum horizontal accuracy",
473            )?,
474        })),
475        StepAdvancePolicyKind::DistanceFromStep => Ok(Arc::new(DistanceFromStepCondition {
476            advance_threshold_meters: validated_policy_threshold(
477                step_advance_policy.advance_threshold_meters,
478                "step-advance distance threshold",
479            )?,
480            minimum_horizontal_accuracy_meters: validated_policy_threshold(
481                step_advance_policy.minimum_horizontal_accuracy_meters,
482                "step-advance minimum horizontal accuracy",
483            )?,
484        })),
485    }
486}

488/// Builds the native deviation detector behind a policy record, refusing a
489/// non-finite or negative threshold.
490pub(crate) fn build_deviation_detector(
491    deviation_policy: &DeviationPolicy,
492) -> Result<Arc<dyn DeviationDetector>, SdkError> {
493    match deviation_policy.policy_kind {
494        DeviationPolicyKind::DistanceThreshold => Ok(Arc::new(DistanceDeviationDetector {
495            deviation_threshold_meters: validated_policy_threshold(
496                deviation_policy.deviation_threshold_meters,
497                "deviation distance threshold",
498            )?,
499        })),
500    }
501}

503/// Validates one policy threshold: policies are configured by foreign code,
504/// where a NaN or negative threshold would silently disable or permanently
505/// flag the strategy, so the controller boundary refuses them instead.
506fn validated_policy_threshold(
507    policy_threshold_meters: f64,
508    threshold_description: &str,
509) -> Result<f64, SdkError> {
510    if !policy_threshold_meters.is_finite() || policy_threshold_meters.is_sign_negative() {
511        return Err(SdkError::InvalidNavigationInput {
512            detail: format!("{threshold_description} must be finite and non-negative"),
513        });
514    }
515    Ok(policy_threshold_meters)
516}
