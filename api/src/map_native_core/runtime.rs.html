1//! The crate-owned async runtime: the executor seam of the exported
2//! `[Async]` surface.
3//!
4//! # The driving problem
5//!
6//! UniFFI's foreign-driven async model lets the foreign executor (Swift's
7//! main actor, Kotlin's main dispatcher) poll the exported future directly.
8//! That model only holds for futures that never touch a process-wide
9//! executor's internals: the MapQuest provider futures run through
10//! `reqwest` and `tokio::time` retries, whose first poll requires a live
11//! tokio IO and time reactor ("there is no reactor running, must be called
12//! from the context of a Tokio 1.x runtime"). The earlier design note in
13//! `Cargo.toml` — that the FFI side drives futures *without any Rust
14//! executor* — proved false on every real native device: the first
15//! geocode, route or `MapSurfaceView::draw` call panicked there.
16//!
17//! # The handback
18//!
19//! Every exported async method therefore hands its runtime-dependent work
20//! to [`drive`]: the work runs on a crate-owned multi-thread tokio runtime
21//! (initialised lazily on the first exported async call, with the IO and
22//! time drivers enabled), and the returned future — the one the foreign
23//! executor polls — only awaits a completion channel, which is
24//! runtime-independent by construction. A panicking task resurfaces as the
25//! same panic on the poller (the FFI scaffolding catches it into the
26//! honest failure report), so the panic semantics of the contract are
27//! preserved end to end.
28
29use std::future::Future;
30use std::sync::OnceLock;
31
32use tokio::runtime::{Builder, Runtime};
33
34/// The lazily initialised crate-owned runtime backing every exported async
35/// call.
36static RUNTIME: OnceLock<Runtime> = OnceLock::new();
37
38/// Builds the crate-owned runtime: the multi-thread scheduler with every
39/// driver enabled, because the provider and draw work relies on the tokio
40/// IO and time reactors (`reqwest` connections, retry back-offs).
41fn runtime() -> &'static Runtime {
42    RUNTIME.get_or_init(|| {
43        Builder::new_multi_thread()
44            .enable_all()
45            .thread_name("map-native-core-driver")
46            .build()
47            .expect("the crate-owned multi-thread tokio runtime is buildable")
48    })
49}
50
51/// Hands a work future to the crate-owned runtime and returns the
52/// runtime-independent future the exported method awaits instead: whatever
53/// executor polls the returned future never touches the tokio context the
54/// work internally needs.
55///
56/// The work must be fully owned (`'static`) and `Send`, which the exported
57/// methods satisfy by moving cloned provider handles, records and session
58/// state into the closure rather than borrowing the exported object.
59pub(crate) fn drive<WorkOutput>(
60    work: impl Future<Output = WorkOutput> + Send + 'static,
61) -> impl Future<Output = WorkOutput> + Send
62where
63    WorkOutput: Send + 'static,
64{
65    let completion = runtime().spawn(work);
66    async move {
67        match completion.await {
68            Ok(work_output) => work_output,
69            // The worker task panicked (aborted tasks do not happen here:
70            // nothing aborts the spawned work). Resurface the panic on the
71            // poller, where the FFI scaffolding's panic boundary converts
72            // it into the same honest failure the contract reports.
73            Err(panic_join_error) => {
74                if panic_join_error.is_panic() {
75                    std::panic::resume_unwind(panic_join_error.into_panic());
76                } else {
77                    unreachable!("the spawned driver task is never aborted");
78                }
79            }
80        }
81    }
82}
83
84/// Runs a work future on the crate-owned runtime fire-and-forget: the frame
85/// path's detached tile-warm tasks (spec round 10) run without any exported
86/// future awaiting them, so the `JoinHandle` is dropped immediately. A
87/// panicking work task is caught at the runtime boundary and never surfaces
88/// to the caller — best-effort semantics by design (the warm fetches retry
89/// on a later pass), which is the difference from [`drive`], whose panics
90/// resurface on the poller.
91pub(crate) fn spawn_detached<WorkOutput>(work: impl Future<Output = WorkOutput> + Send + 'static)
92where
93    WorkOutput: Send + 'static,
94{
95    drop(runtime().spawn(work));
96}

98#[cfg(test)]
99mod tests {
100    use super::drive;
101    use std::time::Duration;
102    use tokio::runtime::Builder;
103
104    /// A short sleep proving the time driver; long enough to require a
105    /// real reactor, too short to matter.
106    const HANDBACK_PROBE_SLEEP: Duration = Duration::from_millis(1);
107
108    #[test]
109    fn runtime_dependent_work_completes_when_driven_outside_a_tokio_context() {
110        // A current-thread runtime built *without* `enable_*`: it has no
111        // reactor, exactly like a foreign executor's polling thread.
112        let bare_executor = Builder::new_current_thread().build().unwrap();
113        let driven_result = bare_executor.block_on(async {
114            drive(async {
115                tokio::time::sleep(HANDBACK_PROBE_SLEEP).await;
116                42usize
117            })
118            .await
119        });
120        assert_eq!(driven_result, 42);
121    }
122
123    #[test]
124    fn a_panicking_task_surfaces_the_panic_on_the_poller() {
125        // A foreign executor cannot catch a Rust panic through its own
126        // machinery, so the panic surfaces on the poller's thread
127        // (uniffi's panic boundary converts it there); the bare
128        // current-thread runtime mimics that polling context.
129        let joined = std::thread::spawn(|| {
130            let bare_executor = Builder::new_current_thread().build().unwrap();
131            bare_executor.block_on(drive(async { panic!("driver work panicked") }))
132        })
133        .join();
134        let payload = joined.expect_err("the surfacing panic must reach the poller");
135        let panic_message = *payload.downcast::<&str>().expect("the str panic payload");
136        assert_eq!(panic_message, "driver work panicked");
137    }
138}
