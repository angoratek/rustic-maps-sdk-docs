1//! The binding-surface records frozen in the UDL contract
2//! (`src/map_native_core.udl`).
3//!
4//! Records are plain value shapes on the FFI, so the mirror structs carry
5//! exactly the documented fields; the projection core's own types
6//! (`map-rendering` `TileKey`/two-component arrays, `geospatial-types`
7//! coordinates) convert through the explicit constructors here, because
8//! FFI traits cannot be implemented for foreign-crate types. The async
9//! provider surface (M4 slice 3) adds the mirror enums and records of the
10//! `geospatial-types` provider records under the same names.

16/// A position on the WGS 84 ellipsoid in decimal degrees.
17#[derive(Debug, Clone, Copy, PartialEq)]
18pub struct GeographicCoordinate {
19    /// Latitude in decimal degrees, valid within `[-90, 90]`.
20    pub latitude: f64,
21    /// Longitude in decimal degrees, valid within `[-180, 180]`.
22    pub longitude: f64,
23}
24
25impl GeographicCoordinate {
26    /// Builds the mirror record from a validated native coordinate.
27    pub(crate) fn from_native_coordinate(
28        native_coordinate: &geospatial_types::GeographicCoordinate,
29    ) -> Self {
30        Self {
31            latitude: native_coordinate.latitude,
32            longitude: native_coordinate.longitude,
33        }
34    }
35
36    /// Builds the native coordinate from the mirror record. The mirror never
37    /// carries canonical-degree guarantees (foreign implementations may hand
38    /// any value across), so this is a plain field copy without validation;
39    /// the call paths that need canonical degrees validate on their own
40    /// boundary.
41    pub(crate) fn to_native_coordinate(self) -> geospatial_types::GeographicCoordinate {
42        geospatial_types::GeographicCoordinate {
43            latitude: self.latitude,
44            longitude: self.longitude,
45        }
46    }
47}

546/// One maneuver within a route leg. The native maneuver classification
547/// crosses as a stable snake-case text label (see
548/// [`maneuver_type_text`]) and the native lane guidance records are not
549/// part of this surface slice.
550#[derive(Debug, Clone, PartialEq)]
551pub struct RouteStep {
552    /// Distance of this step in meters.
553    pub distance_meters: f64,
554    /// Travel time of this step in seconds.
555    pub duration_seconds: f64,
556    /// Maneuver classification as a stable snake-case text label; the
557    /// turn direction and the one-based roundabout exit index are carried
558    /// in the label (`turn-left`, `roundabout-2`).
559    pub maneuver_type: String,
560    /// Human-readable instruction in the display language.
561    pub instruction_display_text: String,
562    /// Instruction tuned for text-to-speech engines.
563    pub instruction_speech_text: String,
564    /// Step geometry from the previous step end to this maneuver point.
565    pub shape: Vec<GeographicCoordinate>,
566}
567
568/// Stable snake-case text labels of the native maneuver classification;
569/// turn directions and the roundabout exit index are embedded in the label,
570/// because the native classification carries payloads UDL enums cannot.
571///
572/// The spelling is owned by [`geospatial_types::ManeuverType::stable_label`]
573/// (the same single spelling the serde serialization of route records emits)
574/// — this converter delegates so the FFI surface and the serialized route
575/// records cannot drift apart.
576pub(crate) fn maneuver_type_text(native_maneuver_type: &geospatial_types::ManeuverType) -> String {
577    native_maneuver_type.stable_label()
578}
579
580/// Builds the native maneuver classification back from one of the stable
581/// snake-case labels [`maneuver_type_text`] emitted; the inverse of that
582/// labeling, needed because route records cross the FFI as labels. Unknown
583/// labels are refused at the caller's boundary (never guessed).
584pub(crate) fn native_maneuver_type_from_text(
585    maneuver_type_text: &str,
586) -> Result<geospatial_types::ManeuverType, SdkError> {
587    geospatial_types::ManeuverType::from_stable_label(maneuver_type_text).ok_or_else(|| {
588        SdkError::InvalidNavigationInput {
589            detail: format!("unrecognized step maneuver label `{maneuver_type_text}`"),
590        }
591    })
592}

594impl RouteStep {
595    /// Builds the mirror record from a native route step.
596    pub(crate) fn from_native_route_step(native_route_step: &geospatial_types::RouteStep) -> Self {
597        Self {
598            distance_meters: native_route_step.distance_meters,
599            duration_seconds: native_route_step.duration_seconds,
600            maneuver_type: maneuver_type_text(&native_route_step.maneuver_type),
601            instruction_display_text: native_route_step.instruction_display_text.clone(),
602            instruction_speech_text: native_route_step.instruction_speech_text.clone(),
603            shape: native_route_step
604                .shape
605                .iter()
606                .map(GeographicCoordinate::from_native_coordinate)
607                .collect(),
608        }
609    }
610
611    /// Builds the native route step from the mirror record, refusing a
612    /// maneuver label the converter cannot map. Lane guidance is not part of
613    /// this surface (see [`Self::from_native_route_step`]), so steps rebuilt
614    /// across the FFI arrive without it — mirroring the forward direction.
615    pub(crate) fn into_native_route_step(self) -> Result<geospatial_types::RouteStep, SdkError> {
616        Ok(geospatial_types::RouteStep {
617            distance_meters: self.distance_meters,
618            duration_seconds: self.duration_seconds,
619            maneuver_type: native_maneuver_type_from_text(&self.maneuver_type)?,
620            instruction_display_text: self.instruction_display_text,
621            instruction_speech_text: self.instruction_speech_text,
622            shape: self
623                .shape
624                .into_iter()
625                .map(|shape_coordinate| shape_coordinate.to_native_coordinate())
626                .collect(),
627            lane_guidances: Vec::new(),
628        })
629    }
