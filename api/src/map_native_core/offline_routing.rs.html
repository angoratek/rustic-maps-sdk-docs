1//! The offline routing graph seam of the native contract: archive bytes in,
2//! a route out, over the same `map-core` engine the wasm surface uses.
3//!
4//! The contract member exists because the offline routing capability must be
5//! reachable from Kotlin and Swift, not only from the browser free function:
6//! a compiled `RMOG` archive arrives as a byte buffer (a `ByteArray` on
7//! Kotlin, `Data` on Swift), exactly the in-memory shape the browser already
8//! carries, because every target shares one implementation. The provider
9//! registry is not involved: the archive is self-contained and the route is
10//! computed over the bytes alone.
11
12use crate::error::SdkError;
13use crate::projection::validated_coordinate;
14use crate::records::{GeographicCoordinate, Route};
15
16/// Routes between two coordinates over an offline routing graph loaded from
17/// in-memory archive bytes: the native counterpart of the wasm
18/// `offlineRouteFromArchiveBytes` export, and the contract's
19/// `offline_route_from_archive_bytes` free helper.
20///
21/// Both surfaces delegate to
22/// [`map_core::offline_routing::calculate_offline_route_from_archive_bytes`],
23/// so the answer cannot drift between native and browser. The request uses
24/// the graph profile's default options (automobile, fastest, metric); the
25/// archive is validated in memory before routing. Failures are typed:
26/// [`SdkError::OfflineRoutingFailure`] for an archive that fails validation,
27/// [`SdkError::InvalidCoordinateInput`] for a coordinate beyond the graph's
28/// snap tolerance, and [`SdkError::ProviderRequestRejected`] for a pair the
29/// graph cannot connect.
30pub async fn offline_route_from_archive_bytes(
31    archive_bytes: Vec<u8>,
32    start: GeographicCoordinate,
33    end: GeographicCoordinate,
34) -> Result<Route, SdkError> {
35    let native_start = validated_coordinate(start)?;
36    let native_end = validated_coordinate(end)?;
37    let native_route = map_core::offline_routing::calculate_offline_route_from_archive_bytes(
38        &archive_bytes,
39        &native_start,
40        &native_end,
41    )
42    .await
43    .map_err(SdkError::from)?;
44    Ok(Route::from_native_route_data(&native_route))
45}
