1//! The error model of the native binding surface: a fielded mirror of the
2//! `geospatial-types` `SDKError` variants the frozen calls can actually
3//! produce, carrying their verbatim detail messages across the FFI.
4
5use geospatial_types::SDKError;
6
7/// Failures thrown across the native binding surface of map-native-core.
8///
9/// The camera/projection surface produces [`SdkError::InvalidCoordinateInput`]
10/// only; the async provider surface additionally produces the transport,
11/// parsing, application-level, rate-limit and not-configured variants below;
12/// the navigation surface (M6) produces [`SdkError::InvalidNavigationInput`]
13/// for route records and navigation policies failing native validation.
14/// The variants mirror the `SDKError` subset returned by the frozen calls
15/// (see `src/map_native_core.udl`); later slices extend this enum as their
16/// surfaces start returning further variants.
17#[derive(Debug, Clone, PartialEq, Eq)]
18pub enum SdkError {
19    /// A coordinate, viewport extent or box corner failed validation; the
20    /// detail carries the verbatim message of the failing core validator.
21    InvalidCoordinateInput {
22        /// Verbatim message of the failing validator.
23        detail: String,
24    },
25    /// The network transport failed (connection, timeout, or DNS); the
26    /// detail carries the provider's verbatim message.
27    NetworkRequestFailed {
28        /// Verbatim transport-failure message.
29        detail: String,
30    },
31    /// The provider responded with a payload the parser could not use; the
32    /// detail carries the verbatim parsing failure.
33    ResponseParsingFailure {
34        /// Verbatim parsing-failure message.
35        detail: String,
36    },
37    /// The provider returned a documented application-level failure status;
38    /// the detail carries its verbatim message.
39    ProviderRequestRejected {
40        /// Verbatim application-level rejection message.
41        detail: String,
42    },
43    /// The token-bucket rate limiter is exhausted; the caller retries after
44    /// the next refill.
45    RateLimitExhausted,
46    /// No provider implementation is registered for the requested contract.
47    ProviderNotConfigured {
48        /// Name of the unregistered contract, e.g. `GeocodingProvider`.
49        contract_name: String,
50    },
51    /// A navigation input failed native-boundary validation; either a policy
52    /// carried an unusable threshold (non-finite or negative) or a route
53    /// record carried a step maneuver label the converter cannot map onto the
54    /// native maneuver classification. Unlike the camera surface this variant
55    /// is not a mirror of a core `SDKError` variant: navigation policies and
56    /// route records enter the SDK at this boundary and have no core-side
57    /// validator of their own yet.
58    InvalidNavigationInput {
59        /// Verbatim description of the refused navigation input.
60        detail: String,
61    },
62    /// An overlay descriptor did not describe a usable overlay (wrong kind
63    /// payload, an out-of-range color channel, or a non-positive or
64    /// non-finite width or radius) or a surface input was structurally
65    /// unusable (a null or ambiguous platform surface pointer, or a
66    /// non-positive viewport extent). Unlike the camera surface this variant
67    /// is not a mirror of a core `SDKError` variant: overlay descriptors and
68    /// surface handles enter the SDK at the surface-view boundary and have
69    /// no core-side validator of their own.
70    InvalidRenderInput {
71        /// Verbatim description of the refused surface input.
72        detail: String,
73    },
74    /// The live rendering could not proceed on the platform surface: the
75    /// platform handle or surface was unusable, the graphics context could
76    /// not be acquired, the surface format could not be negotiated or
77    /// configured, the frame texture could not be acquired even after the
78    /// stale-surface retry, the draw list would overflow, or the frame's
79    /// resources were busy in another draw. Unlike the camera surface this
80    /// variant is not a mirror of a core `SDKError` variant: it carries the
81    /// surface renderer's own failures across the binding surface.
82    RenderingSurfaceUnavailable {
83        /// Verbatim description of the unavailable rendering surface.
84        detail: String,
85    },
86}

88impl std::fmt::Display for SdkError {
89    fn fmt(&self, formatter: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
90        match self {
91            Self::InvalidCoordinateInput { detail } => {
92                write!(formatter, "invalid coordinate input: {detail}")
93            }
94            Self::NetworkRequestFailed { detail } => {
95                write!(formatter, "network request failed: {detail}")
96            }
97            Self::ResponseParsingFailure { detail } => {
98                write!(formatter, "response parsing failure: {detail}")
99            }
100            Self::ProviderRequestRejected { detail } => {
101                write!(formatter, "provider rejected request: {detail}")
102            }
103            Self::RateLimitExhausted => write!(formatter, "rate limit exhausted"),
104            Self::ProviderNotConfigured { contract_name } => {
105                write!(formatter, "no provider configured for {contract_name}")
106            }
107            Self::InvalidNavigationInput { detail } => {
108                write!(formatter, "invalid navigation input: {detail}")
109            }
110            Self::InvalidRenderInput { detail } => {
111                write!(formatter, "invalid render input: {detail}")
112            }
113            Self::RenderingSurfaceUnavailable { detail } => {
114                write!(formatter, "rendering surface unavailable: {detail}")
115            }
116        }
117    }
118}

122impl From<SDKError> for SdkError {
123    /// Maps a core `SDKError` onto the binding error model, preserving every
124    /// variant the frozen calls can produce one-to-one with its verbatim
125    /// detail. The only unreachable variants are the cache and offline
126    /// routing ones — no frozen call goes through the cache or the offline
127    /// graph yet — and those surface under
128    /// [`SdkError::InvalidCoordinateInput`] with their native display
129    /// message so an unexpected failure stays diagnosable and the
130    /// conversion never panics.
131    fn from(sdk_error: SDKError) -> Self {
132        match sdk_error {
133            SDKError::InvalidCoordinateInput(detail) => Self::InvalidCoordinateInput { detail },
134            SDKError::NetworkRequestFailed(detail) => Self::NetworkRequestFailed { detail },
135            SDKError::ResponseParsingFailure(detail) => Self::ResponseParsingFailure { detail },
136            SDKError::ProviderRequestRejected(detail) => Self::ProviderRequestRejected { detail },
137            SDKError::RateLimitExhausted => Self::RateLimitExhausted,
138            SDKError::ProviderNotConfigured(contract_name) => {
139                Self::ProviderNotConfigured { contract_name }
140            }
141            // Unreachable for the frozen calls (see the impl documentation);
142            // the native message keeps the unexpected failure diagnosable.
143            other => Self::InvalidCoordinateInput {
144                detail: other.to_string(),
145            },
146        }
147    }
148}

150impl From<SdkError> for SDKError {
151    /// Maps the binding error back onto the core error model, exact for the
152    /// six `SDKError`-mirroring variants: the forward
153    /// `From<SDKError> for SdkError` conversion and this one round-trip them
154    /// without information loss. This is the direction the foreign-callback
155    /// provider adapters need — a foreign implementation throws `SdkError`
156    /// across the FFI and the adapter re-raises it as the core error of the
157    /// implemented provider contract. The navigation-only
158    /// [`SdkError::InvalidNavigationInput`] has no core equivalent (nothing
159    /// re-raises a navigation-input failure into a provider contract), so it
160    /// carries its detail under `InvalidCoordinateInput` to keep the
161    /// conversion total and the failure diagnosable.
162    fn from(sdk_error: SdkError) -> Self {
163        match sdk_error {
164            SdkError::InvalidCoordinateInput { detail } => Self::InvalidCoordinateInput(detail),
165            SdkError::NetworkRequestFailed { detail } => Self::NetworkRequestFailed(detail),
166            SdkError::ResponseParsingFailure { detail } => Self::ResponseParsingFailure(detail),
167            SdkError::ProviderRequestRejected { detail } => Self::ProviderRequestRejected(detail),
168            SdkError::RateLimitExhausted => Self::RateLimitExhausted,
169            SdkError::ProviderNotConfigured { contract_name } => {
170                Self::ProviderNotConfigured(contract_name)
171            }
172            // Navigation-only variant with no core equivalent (see the impl
173            // documentation); the detail keeps the failure diagnosable.
174            SdkError::InvalidNavigationInput { detail } => Self::InvalidCoordinateInput(detail),
175            // Render-only variants with no core equivalent (see the impl
176            // documentation); the details keep the failures diagnosable, the
177            // same folding the navigation-only variant uses.
178            SdkError::InvalidRenderInput { detail } => Self::InvalidCoordinateInput(detail),
179            SdkError::RenderingSurfaceUnavailable { detail } => {
180                Self::InvalidCoordinateInput(detail)
181            }
182        }
183    }
184}

186#[cfg(test)]
187mod tests {
188    use super::{SDKError, SdkError};
189
190    /// Display text `SDKError` uses for the rate-limit variant.
191    const RATE_LIMIT_DISPLAY_TEXT: &str = "rate limit exhausted";
192
193    #[test]
194    fn invalid_coordinate_input_maps_onto_the_binding_variant_with_verbatim_detail() {
195        let sdk_error =
196            SDKError::InvalidCoordinateInput("latitude 91 outside [-90, 90]".to_string());
197        let mapped: SdkError = sdk_error.into();
198        assert_eq!(
199            mapped,
200            SdkError::InvalidCoordinateInput {
201                detail: "latitude 91 outside [-90, 90]".to_string(),
202            }
203        );
204    }
205
206    #[test]
207    fn provider_failures_map_onto_their_binding_variants_one_to_one() {
208        let network_mapped: SdkError = SDKError::NetworkRequestFailed("offline".to_string()).into();
209        match network_mapped {
210            SdkError::NetworkRequestFailed { detail } => {
211                assert_eq!(detail, "offline");
212            }
213            other => panic!("network failure maps to the wrong variant: {other:?}"),
214        }
215
216        let parsing_mapped: SdkError =
217            SDKError::ResponseParsingFailure("bad json".to_string()).into();
218        assert!(matches!(
219            parsing_mapped,
220            SdkError::ResponseParsingFailure { .. }
221        ));
222
223        let rejection_mapped: SdkError =
224            SDKError::ProviderRequestRejected("quota exceeded".to_string()).into();
225        assert!(matches!(
226            rejection_mapped,
227            SdkError::ProviderRequestRejected { .. }
228        ));
229
230        let rate_limited_mapped: SdkError = SDKError::RateLimitExhausted.into();
231        assert_eq!(rate_limited_mapped, SdkError::RateLimitExhausted);
232        assert_eq!(rate_limited_mapped.to_string(), RATE_LIMIT_DISPLAY_TEXT);
233
234        let not_configured_mapped: SdkError =
235            SDKError::ProviderNotConfigured("GeocodingProvider".to_string()).into();
236        assert_eq!(
237            not_configured_mapped,
238            SdkError::ProviderNotConfigured {
239                contract_name: "GeocodingProvider".to_string()
240            }
241        );
242        assert_eq!(
243            not_configured_mapped.to_string(),
244            "no provider configured for GeocodingProvider"
245        );
246    }
247
248    #[test]
249    fn unreachable_cache_and_offline_variants_keep_their_native_message() {
250        let mapped: SdkError = SDKError::OfflineRoutingFailure(
251            geospatial_types::OfflineGraphInvalidReason::ArchiveMalformed,
252        )
253        .into();
254        assert_eq!(
255            mapped.to_string(),
256            "invalid coordinate input: offline graph invalid: ArchiveMalformed"
257        );
258    }

260    #[test]
261    fn navigation_only_input_variant_keeps_its_detail_under_the_core_coordinate_variant() {
262        let navigation_detail = "step-advance threshold must be finite and non-negative";
263        let mapped: SDKError = SdkError::InvalidNavigationInput {
264            detail: navigation_detail.to_string(),
265        }
266        .into();
267        assert_eq!(
268            mapped,
269            SDKError::InvalidCoordinateInput(navigation_detail.to_string()),
270            "the navigation-only variant has no core equivalent and carries its detail"
271        );
272    }

274    #[test]
275    fn render_only_variants_keep_their_details_under_the_core_coordinate_variant() {
276        let render_detail = "overlay 0 exceeds the color channel range";
277        let mapped: SDKError = SdkError::InvalidRenderInput {
278            detail: render_detail.to_string(),

299    #[test]
300    fn binding_variants_round_trip_through_the_core_error_model() {
301        for sdk_error in [
302            SdkError::InvalidCoordinateInput {
303                detail: "latitude 91 outside [-90, 90]".to_string(),
304            },
305            SdkError::NetworkRequestFailed {
306                detail: "offline".to_string(),
307            },
308            SdkError::ResponseParsingFailure {
309                detail: "bad json".to_string(),
310            },
311            SdkError::ProviderRequestRejected {
312                detail: "quota exceeded".to_string(),
313            },
314            SdkError::RateLimitExhausted,
315            SdkError::ProviderNotConfigured {
316                contract_name: "GeocodingProvider".to_string(),
317            },
318        ] {
319            let native_sdk_error: SDKError = sdk_error.clone().into();
320            let round_tripped: SdkError = native_sdk_error.into();
321            assert_eq!(
322                round_tripped, sdk_error,
323                "variant {sdk_error:?} must round-trip"
324            );
325        }
