1//! The UniFFI-facing anchored camera, wrapping `map-rendering`'s camera; the
2//! validation the core leaves to callers happens on every binding call, so
3//! the frozen surface can only ever receive validated camera state.
4
5use map_rendering::MapCamera as NativeMapCamera;
6
7use crate::error::SdkError;
8use crate::projection::{validated_bounding_box, validated_coordinate, validated_viewport_pixels};
9use crate::records::{
10    BoundingBox, EdgeInsetsPixels, GeographicCoordinate, ScreenPoint, TileKeyRecord, ViewportPixels,
11};
12
13/// Anchored map camera (geographic center, zoom, bearing, pitch) mirroring the
14/// native [`NativeMapCamera`]. The ground point under the center always sits
15/// at the viewport center.
16#[derive(Debug, Clone, Copy)]
17pub struct MapCamera {
18    native_camera: NativeMapCamera,
19}
20
21impl MapCamera {
22    /// Builds a camera, rejecting an invalid center: latitude/longitude
23    /// outside their canonical ranges fail with
24    /// [`SdkError::InvalidCoordinateInput`]. Zoom and pitch are clamped and
25    /// the bearing wraps into `[0, 360)`, exactly like the native
26    /// constructor.
27    pub fn new(
28        center: GeographicCoordinate,
29        zoom_level: f64,
30        bearing_degrees: f64,
31        pitch_degrees: f64,
32    ) -> Result<Self, SdkError> {
33        let center = validated_coordinate(center)?;
34        Ok(Self {
35            native_camera: NativeMapCamera::new(center, zoom_level, bearing_degrees, pitch_degrees),
36        })
37    }
38
39    /// Builds the camera that frames a bounding box inside the viewport's
40    /// inset region: top-down and north-up by construction (pitch and bearing
41    /// zero), zoomed to the largest level keeping the whole box inside the
42    /// inset area and clamped to the caller's `maximum_zoom_level`; the center
43    /// is anchored on the inset region's center. Delegates to the native
44    /// `fit_bounds`; coordinate/viewport validation failures and the native
45    /// fit failures (insets, maximum zoom, box area) all surface under
46    /// [`SdkError::InvalidCoordinateInput`] with the verbatim detail.
47    pub fn fit_bounds(
48        bounding_box: BoundingBox,
49        viewport: ViewportPixels,
50        edge_insets: EdgeInsetsPixels,
51        maximum_zoom_level: f64,
52    ) -> Result<Self, SdkError> {
53        let bounding_box = validated_bounding_box(bounding_box)?;
54        let viewport = validated_viewport_pixels(viewport)?;
55        map_rendering::fit_bounds(
56            &bounding_box,
57            &viewport,
58            edge_insets.to_native_edge_insets(),
59            maximum_zoom_level,
60        )
61        .map(|native_camera| Self { native_camera })
62        .map_err(|message| SdkError::InvalidCoordinateInput { detail: message })
63    }
64
65    /// Projects a coordinate to viewport pixels (origin top-left, y down).
66    /// Returns `None` when the point lies behind the camera.
67    pub fn project_to_screen(
68        &self,
69        coordinate: GeographicCoordinate,
70        viewport: ViewportPixels,
71        field_of_view_degrees: f64,
72    ) -> Result<Option<ScreenPoint>, SdkError> {
73        let coordinate = validated_coordinate(coordinate)?;
74        let viewport = validated_viewport_pixels(viewport)?;
75        Ok(self
76            .native_camera
77            .project_to_screen(&coordinate, &viewport, field_of_view_degrees)
78            .map(ScreenPoint::from_screen_pixels))
79    }
80
81    /// Unprojects a viewport pixel back onto the ground plane. Returns `None`
82    /// when the ray misses the ground (typically below the horizon at pitch).
83    pub fn screen_to_ground_coordinate(
84        &self,
85        screen_point: ScreenPoint,
86        viewport: ViewportPixels,
87        field_of_view_degrees: f64,
88    ) -> Result<Option<GeographicCoordinate>, SdkError> {
89        let viewport = validated_viewport_pixels(viewport)?;
90        Ok(self
91            .native_camera
92            .screen_to_ground_coordinate(
93                [screen_point.x, screen_point.y],
94                &viewport,
95                field_of_view_degrees,
96            )
97            .map(|native_coordinate| {
98                GeographicCoordinate::from_native_coordinate(&native_coordinate)
99            }))
100    }
101
102    /// Ground meters represented by one screen pixel at the camera's center
103    /// (round 10, the scale bars' helper). The viewport and field-of-view
104    /// arguments mirror the projection methods' shape — call sites pass the
105    /// same camera context — and the viewport is validated here like
106    /// everywhere else; the honest flat-centered approximation itself is
107    /// independent of both (see the contract's member documentation).
108    pub fn ground_meters_per_screen_pixel(
109        &self,
110        viewport: ViewportPixels,
111        field_of_view_degrees: f64,
112    ) -> Result<f64, SdkError> {
113        validated_viewport_pixels(viewport)?;
114        let _ = field_of_view_degrees;
115        Ok(self.native_camera.ground_meters_per_screen_pixel())
116    }
117
118    /// Every tile at `tile_zoom_level` whose area intersects the visible
119    /// ground, sorted row-major from the north, mirroring the native
120    /// `visible_tile_keys`.
121    pub fn visible_tile_keys(
122        &self,
123        viewport: ViewportPixels,
124        field_of_view_degrees: f64,
125        tile_zoom_level: u8,
126    ) -> Result<Vec<TileKeyRecord>, SdkError> {
127        let viewport = validated_viewport_pixels(viewport)?;
128        Ok(self
129            .native_camera
130            .visible_tile_keys(&viewport, field_of_view_degrees, tile_zoom_level)
131            .iter()
132            .map(|tile_key| TileKeyRecord::from_native_tile_key(*tile_key))
133            .collect())
134    }
135
136    /// Nearest whole zoom level a tile fetch should use for this camera,
137    /// mirroring the native `tile_zoom_level`.
138    pub fn tile_zoom_level(&self) -> u8 {
139        self.native_camera.tile_zoom_level()
140    }
141
142    /// The camera's geographic center, verbatim as the camera was built.
143    pub fn center(&self) -> GeographicCoordinate {
144        GeographicCoordinate::from_native_coordinate(&self.native_camera.center)
145    }
146
147    /// The camera's zoom level, verbatim as the camera was built (clamped
148    /// values included: `fit_bounds`' readback path reads the actually
149    /// fitting level here).
150    pub fn zoom_level(&self) -> f64 {
151        self.native_camera.zoom_level
152    }
153
154    /// The camera's bearing in degrees within `[0, 360)`.
155    pub fn bearing_degrees(&self) -> f64 {
156        self.native_camera.bearing_degrees
157    }
158
159    /// The camera's pitch in degrees.
160    pub fn pitch_degrees(&self) -> f64 {
161        self.native_camera.pitch_degrees
162    }
163
164    /// Borrows the wrapped native camera for the surface seam: the surface
165    /// view's `draw` passes the validated pose to the map-rendering
166    /// renderer, which assembles its quads from the native camera directly.
167    /// The wrapper stays the validation boundary for every binding call.
168    pub(crate) fn native_camera(&self) -> &NativeMapCamera {
169        &self.native_camera
170    }
171}

267    #[test]
268    fn fit_bounds_frames_the_box_inside_the_inset_region() {
269        let zurich_box = bounded_box(47.34, 8.50, 47.42, 8.60);
270        let camera =
271            MapCamera::fit_bounds(zurich_box, viewport(), symmetric_edge_insets(), f64::MAX)
272                .expect("box, viewport and insets are all valid");
273        let insets = symmetric_edge_insets();
274        let mut touches_left = false;
275        let mut touches_right = false;
276        let mut touches_top = false;
277        let mut touches_bottom = false;
278        for box_corner in [zurich_box.south_west, zurich_box.north_east] {
279            let corner_screen = camera
280                .project_to_screen(box_corner, viewport(), 60.0)
281                .expect("viewport is valid")
282                .expect("a fitted top-down box corner is always in front of the camera");
283            assert!(
284                corner_screen.x >= insets.left - FIT_BOUNDS_CORNER_TOLERANCE_PIXELS,
285                "corner {box_corner:?} at x {} pierces the left inset",
286                corner_screen.x
287            );
288            assert!(
289                corner_screen.x
290                    <= VIEWPORT_WIDTH_PIXELS - insets.right + FIT_BOUNDS_CORNER_TOLERANCE_PIXELS,
291                "corner {box_corner:?} at x {} pierces the right inset",
292                corner_screen.x
293            );
294            assert!(
295                corner_screen.y >= insets.top - FIT_BOUNDS_CORNER_TOLERANCE_PIXELS,
296                "corner {box_corner:?} at y {} pierces the top inset",
297                corner_screen.y
298            );
299            assert!(
300                corner_screen.y
301                    <= VIEWPORT_HEIGHT_PIXELS - insets.bottom + FIT_BOUNDS_CORNER_TOLERANCE_PIXELS,
302                "corner {box_corner:?} at y {} pierces the bottom inset",
303                corner_screen.y
304            );
305            touches_left |= corner_screen.x <= insets.left + FIT_BOUNDS_CORNER_TOLERANCE_PIXELS;
306            touches_right |= corner_screen.x
307                >= VIEWPORT_WIDTH_PIXELS - insets.right - FIT_BOUNDS_CORNER_TOLERANCE_PIXELS;
308            touches_top |= corner_screen.y <= insets.top + FIT_BOUNDS_CORNER_TOLERANCE_PIXELS;
309            touches_bottom |= corner_screen.y
310                >= VIEWPORT_HEIGHT_PIXELS - insets.bottom - FIT_BOUNDS_CORNER_TOLERANCE_PIXELS;
311        }
312        // The binding axis fits exactly between its insets; the slack axis
313        // sits strictly inside.
314        assert!((touches_left && touches_right) || (touches_top && touches_bottom));
315    }
