1//! Course filtering (round 11, Ferrostar B7's mechanism with our own
2//! expressions): how a position fix's reported course becomes the trip's
3//! reported course.
4//!
5//! A coarse GNSS fix's course over ground is often the least trustworthy
6//! field the fix carries — a momentary standstill or an urban canyon reads as
7//! an arbitrary bearing, while the trip's own route line is a much stronger
8//! statement about the real direction of travel. The reference
9//! (`stadiamaps/ferrostar`) ships this as a controller-level configuration
10//! enum, `CourseFiltering::{Raw, SnapToRoute}`, decided once per trip:
11//!
12//! - [`CourseFiltering::Raw`] reports the fix's course verbatim — the
13//!   tunnel or canyon exit hatch where a garbage route projection must not
14//!   override a correct physical course.
15//! - [`CourseFiltering::SnapToRoute`] projects the course onto the route
16//!   line: while the fix is *not* judged fully off route, the reported course
17//!   is the bearing of the route line at the user's snapped position — every
18//!   materially disagreeing raw course is replaced in one step, and the
19//!   course stops jittering against the very line the position was already
20//!   snapped onto. Without a route line to trust (the fix did not snap, or
21//!   the projected segment carries no direction), the fix's own course
22//!   remains the fall-back, and an unreported course retains the trip's
23//!   previous course (the same degradation continuity the snapped position
24//!   keeps).
25//!
26//! The filtering is decided per trip like the step-advance and deviation
27//! policies (the `StartTrip` event carries it), and the reducer stamps every
28//! fix's effective course into
29//! [`TripNavigationState::course_heading_degrees`] during the same position
30//! snap pass — so consumers (a course wedge, a follow camera) read one
31//! course field instead of re-filtering raw courses per platform. The frozen
32//! FFI contract does not carry the field or the policy this round; this stays
33//! an internal core seam, and any later exposure is addition-only UDL.
34
35/// How the trip reports its course. The reference's
36/// `CourseFiltering::{Raw, SnapToRoute}` pairing.
37#[derive(Debug, Clone, Copy, Default, PartialEq)]
38pub enum CourseFiltering {
39    /// The fix's reported course travels with the fix unchanged: the trust in
40    /// the physical course overrides what the route line says (the raw-course
41    /// escape hatch for tunnels and canyons).
42    #[default]
43    Raw,
44    /// The reported course is snapped onto the route line's direction at the
45    /// user's projected position (the module documentation's semantics).
46    SnapToRoute,
47}
48
49/// Computes the effective course of one processed fix under the trip's
50/// course filtering.
51///
52/// `raw_course_heading_degrees` is the fix's own reported course;
53/// `snapped_bearing_degrees` the route line's bearing at the fix's snapped
54/// position (per [`PolylineSnap::bearing_degrees`]); `previous_course_heading_degrees`
55/// the effective course currently on the trip's state, retained when nothing
56/// newer can be reported (the snapped-position retention rule). `None` on
57/// either source means "not reported here".
58pub fn effective_course_heading_degrees(
59    course_filtering: CourseFiltering,
60    raw_course_heading_degrees: Option<f64>,
61    snapped_bearing_degrees: Option<f64>,
62    previous_course_heading_degrees: Option<f64>,
63) -> Option<f64> {
64    match course_filtering {
65        CourseFiltering::Raw => raw_course_heading_degrees,
66        CourseFiltering::SnapToRoute => snapped_bearing_degrees
67            .or(raw_course_heading_degrees)
68            .or(previous_course_heading_degrees),
69    }
70}

72#[cfg(test)]
73mod tests {
74    use super::{CourseFiltering, effective_course_heading_degrees};
75
76    /// The raw course under test: a fix reporting some physically plausible
77    /// but route-irrelevant bearing.
78    const RAW_COURSE_DEGREES: f64 = 200.0;
79    /// The route line's bearing at the snapped position under test (due
80    /// eastward, the synthetic walker routes' direction).
81    const ROUTE_BEARING_DEGREES: f64 = 90.0;
82    /// The previous effective course for the retention arms of the fallback
83    /// chain.
84    const PREVIOUS_COURSE_DEGREES: f64 = 42.0;
85    /// Tolerance for course measurements compared by value.
86    const BEARING_TOLERANCE_DEGREES: f64 = 1e-6;
87
88    fn assert_close(actual: Option<f64>, expected: f64, explanation: &str) {
89        let actual = actual.expect("a course must be reported in this case");
90        assert!(
91            (actual - expected).abs() < BEARING_TOLERANCE_DEGREES,
92            "{explanation}: expected {expected}, got {actual}"
93        );
94    }
95
96    #[test]
97    fn raw_filtering_reports_the_fixs_own_course_verbatim() {
98        let course = effective_course_heading_degrees(
99            CourseFiltering::Raw,
100            Some(RAW_COURSE_DEGREES),
101            Some(ROUTE_BEARING_DEGREES),
102            Some(PREVIOUS_COURSE_DEGREES),
103        );
104        assert_close(
105            course,
106            RAW_COURSE_DEGREES,
107            "Raw must never rewrite the reported course",
108        );
109    }
110
111    #[test]
112    fn raw_filtering_reports_nothing_without_a_report() {
113        assert_eq!(
114            effective_course_heading_degrees(
115                CourseFiltering::Raw,
116                None,
117                Some(ROUTE_BEARING_DEGREES),
118                Some(PREVIOUS_COURSE_DEGREES)
119            ),
120            None,
121            "Raw without a reported course reports none"
122        );
123    }
124
125    #[test]
126    fn snap_to_route_reports_the_route_bearing_while_it_exists() {
127        let course = effective_course_heading_degrees(
128            CourseFiltering::SnapToRoute,
129            Some(RAW_COURSE_DEGREES),
130            Some(ROUTE_BEARING_DEGREES),
131            Some(PREVIOUS_COURSE_DEGREES),
132        );
133        assert_close(
134            course,
135            ROUTE_BEARING_DEGREES,
136            "on route, the snapped route bearing replaces the raw course",
137        );
138    }
139
140    #[test]
141    fn snap_to_route_falls_back_to_the_fix_without_a_route_bearing() {
142        // The fix did not snap onto a directed line (off route, or a segment
143        // without a representable direction): the fix's own course remains.
144        let course = effective_course_heading_degrees(
145            CourseFiltering::SnapToRoute,
146            Some(RAW_COURSE_DEGREES),
147            None,
148            Some(PREVIOUS_COURSE_DEGREES),
149        );
150        assert_close(
151            course,
152            RAW_COURSE_DEGREES,
153            "off the route line, the raw course is the fall-back",
154        );
155    }
156
157    #[test]
158    fn snap_to_route_falls_back_to_the_previous_course_without_any_report() {
159        // Neither a route bearing nor a fresh fix course: the trip's last
160        // effective course is retained (the position snap's retention rule).
161        assert_eq!(
162            effective_course_heading_degrees(
163                CourseFiltering::SnapToRoute,
164                None,
165                None,
166                Some(PREVIOUS_COURSE_DEGREES)
167            ),
168            Some(PREVIOUS_COURSE_DEGREES),
169            "an unreported fix retains the trip's last course"
170        );
171        assert_eq!(
172            effective_course_heading_degrees(CourseFiltering::SnapToRoute, None, None, None),
173            None,
174            "and an unreported fix on a courseless trip stays courseless"
175        );
176    }
177
178    #[test]
179    fn the_default_filtering_is_raw() {
180        assert_eq!(
181            CourseFiltering::default(),
182            CourseFiltering::Raw,
183            "the tripless default must keep the raw-course behavior"
184        );
185    }
