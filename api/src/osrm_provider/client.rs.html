1//! Configuration and the HTTP facade shared by OSRM endpoint implementations.
2//!
3//! Unlike a credentialed commercial platform, OSRM is addressed by its
4//! deployment URL and needs no key, so this client carries no credential
5//! injection and no rate limiter: it holds one base URL and one user agent,
6//! performs idempotent GETs through [`provider_transport`], and hands each
7//! response to [`crate::error_mapping::interpret_json_response`].
8
9use crate::error_mapping::interpret_json_response;
10use geospatial_types::SDKError;
11use provider_transport::{
12    HttpMethod, HttpRequest, HttpTransport, ObservedTransport, RetryPolicy, default_transport,
13    execute_idempotent_get,
14};
15use std::sync::Arc;
16use url::Url;
17
18/// Base URL of a locally hosted OSRM server used when no configuration is
19/// provided. A local OSRM installation serves the HTTP API on port 5000.
20pub const DEFAULT_OSRM_BASE_URL: &str = "http://localhost:5000";
21
22/// The provider contract every call through this client serves.
23pub const ROUTING_CONTRACT_NAME: &str = "RoutingProvider";
24
25/// The registry key of this provider, used for request observation only.
26pub const OSRM_PROVIDER_KEY: &str = "osrm";
27
28/// User agent sent when the configuration carries none.
29pub const DEFAULT_OSRM_USER_AGENT: &str =
30    concat!("rustic-maps-osrm-provider/", env!("CARGO_PKG_VERSION"));
31
32/// Retries after the first attempt for one idempotent route GET. Retries
33/// apply only to transient transport and server failures.
34pub const DEFAULT_MAXIMUM_RETRY_ATTEMPTS: u32 = 2;
35
36/// Connection settings for one OSRM deployment.
37///
38/// OSRM is keyless, so the deployment URL is the only address information the
39/// service needs. `base_url` may name any host serving the OSRM HTTP API
40/// (`http://host:5000`, `https://routing.example.com`, and so on) and must be
41/// an absolute URL without the route service path.
42///
43/// `user_agent` identifies the SDK to the deployment's operator; an empty
44/// value suppresses the header entirely.
45#[derive(Debug, Clone, PartialEq, Eq)]
46pub struct OsrmProviderConfiguration {
47    /// Root URL of the OSRM deployment.
48    pub base_url: String,
49    /// `User-Agent` header value sent with every request.
50    pub user_agent: String,
51}
52
53impl Default for OsrmProviderConfiguration {
54    fn default() -> Self {
55        Self {
56            base_url: DEFAULT_OSRM_BASE_URL.to_string(),
57            user_agent: DEFAULT_OSRM_USER_AGENT.to_string(),
58        }
59    }
60}

62/// GET-only JSON client against one OSRM base URL.
63///
64/// The client validates the base URL once, then builds each request as
65/// `{base_url}/{path}` with the caller's query pairs. The route service has no
66/// request body, so only idempotent GETs are needed.
67pub struct OsrmHttpClient {
68    base_url: String,
69    user_agent: String,
70    maximum_retry_attempts: u32,
71    transport: Arc<dyn HttpTransport>,
72}
73
74impl OsrmHttpClient {
75    /// Builds a client around the target's default transport.
76    ///
77    /// # Parameters
78    /// - `configuration` — deployment base URL and user agent.
79    ///
80    /// # Returns
81    /// A client ready to issue requests.
82    ///
83    /// # Errors
84    /// Returns [`SDKError::NetworkRequestFailed`] when `base_url` is not an
85    /// absolute http or https URL or when the native HTTP transport cannot be
86    /// built.
87    pub fn new(configuration: OsrmProviderConfiguration) -> Result<Self, SDKError> {
88        Self::with_transport(
89            configuration,
90            DEFAULT_MAXIMUM_RETRY_ATTEMPTS,
91            Arc::new(ObservedTransport::new(
92                default_transport()?,
93                ROUTING_CONTRACT_NAME,
94                OSRM_PROVIDER_KEY,
95            )),
96        )
97    }
98
99    /// Builds a client around an explicit transport and retry budget (the
100    /// registration path and tests use this seam; request building,
101    /// interpretation and retry behavior are identical whichever transport is
102    /// supplied).
103    ///
104    /// # Parameters
105    /// - `configuration` — deployment base URL and user agent.
106    /// - `maximum_retry_attempts` — retries after the first attempt for a
107    ///   transient failure.
108    /// - `transport` — the platform transport that executes requests.
109    ///
110    /// # Errors
111    /// Returns [`SDKError::NetworkRequestFailed`] when `base_url` is not an
112    /// absolute http or https URL.
113    pub fn with_transport(
114        configuration: OsrmProviderConfiguration,
115        maximum_retry_attempts: u32,
116        transport: Arc<dyn HttpTransport>,
117    ) -> Result<Self, SDKError> {
118        validate_base_url(&configuration.base_url)?;
119        Ok(Self {
120            base_url: configuration.base_url,
121            user_agent: configuration.user_agent,
122            maximum_retry_attempts,
123            transport,
124        })
125    }
126
127    /// Performs a bounded-retry GET against `{base_url}/{path}` with the given
128    /// query pairs.
129    ///
130    /// # Parameters
131    /// - `path` — service path below the base URL, without a leading slash.
132    /// - `query_pairs` — query parameters appended in order.
133    ///
134    /// # Returns
135    /// The decoded JSON payload. An OSRM payload whose `code` is not `Ok` is
136    /// rejected before it reaches the caller.
137    ///
138    /// # Errors
139    /// - [`SDKError::NetworkRequestFailed`] for a transport failure or a
140    ///   non-success HTTP status that does not carry an OSRM error envelope.
141    /// - [`SDKError::ResponseParsingFailure`] when a successful response body
142    ///   is not JSON.
143    /// - [`SDKError::ProviderRequestRejected`] when OSRM answers with the
144    ///   provider's own `code`/`message` error envelope.
145    pub async fn get_json(
146        &self,
147        path: &str,
148        query_pairs: &[(String, String)],
149    ) -> Result<serde_json::Value, SDKError> {
150        let request = self.build_request(path, query_pairs)?;
151        execute_idempotent_get(
152            self.transport.as_ref(),
153            &request,
154            RetryPolicy::new(self.maximum_retry_attempts),
155            interpret_json_response,
156        )
157        .await
158    }
159
160    /// Builds one GET request against `{base_url}/{path}` with the configured
161    /// query pairs and user agent.
162    fn build_request(
163        &self,
164        path: &str,
165        query_pairs: &[(String, String)],
166    ) -> Result<HttpRequest, SDKError> {
167        let endpoint = format!(
168            "{}/{}",
169            self.base_url.trim_end_matches('/'),
170            path.trim_start_matches('/')
171        );
172        let mut url = Url::parse(&endpoint).map_err(|error| {
173            SDKError::NetworkRequestFailed(format!("invalid OSRM request URL {endpoint}: {error}"))
174        })?;
175        {
176            let mut query_pairs_mut = url.query_pairs_mut();
177            for (name, value) in query_pairs {
178                query_pairs_mut.append_pair(name, value);
179            }
180        }
181        let headers = if self.user_agent.is_empty() {
182            Vec::new()
183        } else {
184            vec![("user-agent".to_string(), self.user_agent.clone())]
185        };
186        Ok(HttpRequest {
187            method: HttpMethod::Get,
188            url: url.to_string(),
189            headers,
190            body: None,
191        })
192    }
193}
194
195/// Validates that a configuration names an absolute HTTP base URL.
196fn validate_base_url(base_url: &str) -> Result<(), SDKError> {
197    let parsed = Url::parse(base_url).map_err(|error| {
198        SDKError::NetworkRequestFailed(format!("invalid OSRM base URL {base_url}: {error}"))
199    })?;
200    if !matches!(parsed.scheme(), "http" | "https") || parsed.host_str().is_none() {
201        return Err(SDKError::NetworkRequestFailed(format!(
202            "invalid OSRM base URL {base_url}: expected an absolute http or https URL"
203        )));
204    }
205    Ok(())
206}

208#[cfg(test)]
209mod tests {
210    use super::{OsrmHttpClient, OsrmProviderConfiguration};
211    use geospatial_types::SDKError;
212
213    #[test]
214    fn relative_base_url_is_rejected_at_construction() {
215        let outcome = OsrmHttpClient::new(OsrmProviderConfiguration {
216            base_url: "localhost:5000".to_string(),
217            user_agent: "test-agent".to_string(),
218        });
219        match outcome {
220            Err(SDKError::NetworkRequestFailed(detail)) => {
221                assert!(detail.contains("localhost:5000"), "detail was {detail}");
222            }
223            Err(other) => panic!("expected a network failure, got {other:?}"),
224            Ok(_) => panic!("a relative base URL must not build a client"),
225        }
226    }
227
228    #[test]
229    fn default_configuration_targets_the_local_osrm_port() {
230        let configuration = OsrmProviderConfiguration::default();
231        assert_eq!(configuration.base_url, "http://localhost:5000");
232        assert!(!configuration.user_agent.is_empty());
233    }
