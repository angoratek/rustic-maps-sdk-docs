1//! OSRM route service implementation of
2//! [`geospatial_traits::RoutingProvider`].
3//!
4//! # Request shape
5//!
6//! One request is
7//! `GET {base_url}/route/v1/{profile}/{longitude},{latitude};…` with
8//! `overview=full`, `geometries=geojson`, `steps=true` and, when the caller
9//! asks to avoid features, `exclude` set to the requested OSRM classes.
10//!
11//! # Option mapping
12//!
13//! - [`TransportationMode::Automobile`] selects the `driving` profile,
14//!   [`TransportationMode::Bicycle`] the `cycling` profile and
15//!   [`TransportationMode::Pedestrian`] the `walking` profile.
16//! - [`AvoidOptions::toll_roads`] becomes `toll`,
17//!   [`AvoidOptions::limited_access_highways`] becomes `motorway`,
18//!   [`AvoidOptions::ferries`] becomes `ferry` and
19//!   [`AvoidOptions::unpaved_roads`] becomes `unpaved`, joined in one
20//!   `exclude` value.
21//! - [`geospatial_types::route::UnitSystem`] does not change the request: the
22//!   route contract is always meters and seconds, which is also what OSRM
23//!   returns.
24//!
25//! # Options OSRM cannot express
26//!
27//! These are documented rather than silently approximated:
28//!
29//! - [`geospatial_types::route::RouteStrategy`] has no route-service
30//!   parameter. The deployed profile's cost model fixes the ranking, so
31//!   `Fastest` and `Shortest` produce the same request.
32//! - [`RouteOptions::departure_timestamp_seconds`] has no route-service
33//!   parameter. The service routes for "now", so a requested departure instant
34//!   is not honored.
35//! - `exclude` classes must be declared by the deployed profile. Requesting a
36//!   class the profile does not declare makes OSRM reject the request, which
37//!   surfaces as [`SDKError::ProviderRequestRejected`].
38//! - OSRM returns no human-readable instruction, so this provider authors
39//!   display and speech text from the maneuver and road name, and it does not
40//!   map OSRM intersection lane records.
41//! - OSRM has no ferry maneuver type. A step whose `mode` is `ferry` is
42//!   classified as [`ManeuverType::Ferry`] before the maneuver-type table is
43//!   consulted, so a ferry crossing keeps its own instruction instead of
44//!   degrading to `Continue`.
45

59/// OSRM profile segment used for automobile routing.
60const DRIVING_PROFILE: &str = "driving";
61/// OSRM profile segment used for bicycle routing.
62const CYCLING_PROFILE: &str = "cycling";
63/// OSRM profile segment used for pedestrian routing.
64const WALKING_PROFILE: &str = "walking";

70const OVERVIEW_FULL: &str = "full";
71/// Query value requesting GeoJSON geometries.

74const STEPS_TRUE: &str = "true";

86const FERRY_EXCLUDE_CLASS: &str = "ferry";

99/// Routing through the OSRM HTTP route service.
100pub struct OsrmRoutingProvider {
101    http_client: OsrmHttpClient,
102}

104impl OsrmRoutingProvider {
105    /// Builds the provider for one OSRM deployment.
106    ///
107    /// # Parameters
108    /// - `configuration` — deployment base URL and user agent.
109    ///
110    /// # Returns
111    /// A provider ready to calculate routes.
112    ///
113    /// # Errors
114    /// Returns [`SDKError::NetworkRequestFailed`] when the base URL is not an
115    /// absolute http or https URL or the native HTTP transport cannot be
116    /// built.
117    pub fn new(configuration: OsrmProviderConfiguration) -> Result<Self, SDKError> {
118        Ok(Self {
119            http_client: OsrmHttpClient::new(configuration)?,
120        })
121    }

124#[async_trait]
125impl RoutingProvider for OsrmRoutingProvider {
126    /// OSRM's route service selects a profile by mode and has no
127    /// shortest-path switch, so the strategy choice cannot be honoured.
128    fn route_capabilities(&self) -> RouteCapabilities {
129        RouteCapabilities {
130            supported_transportation_modes: vec![
131                TransportationMode::Automobile,
132                TransportationMode::Bicycle,
133                TransportationMode::Pedestrian,
134            ],
135            supports_shortest_route_strategy: false,
136        }
137    }
138
139    async fn calculate_route(
140        &self,
141        waypoints: &[GeographicCoordinate],
142        route_options: &RouteOptions,
143    ) -> Result<RouteData, SDKError> {
144        if waypoints.len() < MINIMUM_ROUTE_WAYPOINT_COUNT {
145            return Err(SDKError::InvalidCoordinateInput(format!(
146                "a route needs at least {MINIMUM_ROUTE_WAYPOINT_COUNT} waypoints; got {}",
147                waypoints.len()
148            )));
149        }
150        let request_path = route_request_path(route_options.transportation_mode, waypoints);
151        let query_pairs = route_query_pairs(route_options);
152        let payload = self
153            .http_client
154            .get_json(&request_path, &query_pairs)
155            .await?;
156        parse_route_payload(&payload)
157    }
158}

160/// Builds the route-service path, including the profile and the ordered
161/// `longitude,latitude` waypoint list.
162fn route_request_path(
163    transportation_mode: TransportationMode,
164    waypoints: &[GeographicCoordinate],
165) -> String {
166    let coordinates = waypoints
167        .iter()
168        .map(|waypoint| {
169            format!(
170                "{}{LONGITUDE_LATITUDE_SEPARATOR}{}",
171                waypoint.longitude, waypoint.latitude
172            )
173        })
174        .collect::<Vec<String>>()
175        .join(COORDINATE_SEPARATOR);
176    format!(
177        "{ROUTE_SERVICE_PATH_PREFIX}/{}/{}",
178        profile_path_segment(transportation_mode),
179        coordinates
180    )
181}

183/// Maps the contract's transportation mode onto OSRM's profile path segment.
184fn profile_path_segment(transportation_mode: TransportationMode) -> &'static str {
185    match transportation_mode {
186        TransportationMode::Automobile => DRIVING_PROFILE,
187        TransportationMode::Bicycle => CYCLING_PROFILE,
188        TransportationMode::Pedestrian => WALKING_PROFILE,
189    }
190}

192/// Builds the route-service query pairs, including avoidance classes when the
193/// caller asked for any.
194fn route_query_pairs(route_options: &RouteOptions) -> Vec<(String, String)> {
195    let mut query_pairs = vec![
196        ("overview".to_string(), OVERVIEW_FULL.to_string()),
197        ("geometries".to_string(), GEOMETRIES_GEOJSON.to_string()),
198        ("steps".to_string(), STEPS_TRUE.to_string()),
199    ];
200    let exclude_classes = exclude_classes_text(&route_options.avoid_options);
201    if !exclude_classes.is_empty() {
202        query_pairs.push(("exclude".to_string(), exclude_classes));
203    }
204    query_pairs
205}
206
207/// Joins the requested avoidance classes into one OSRM `exclude` value.
208fn exclude_classes_text(avoid_options: &AvoidOptions) -> String {
209    let mut classes = Vec::new();
210    if avoid_options.toll_roads {
211        classes.push(TOLL_EXCLUDE_CLASS);
212    }
213    if avoid_options.limited_access_highways {
214        classes.push(MOTORWAY_EXCLUDE_CLASS);
215    }
216    if avoid_options.ferries {
217        classes.push(FERRY_EXCLUDE_CLASS);
218    }
219    if avoid_options.unpaved_roads {
220        classes.push(UNPAVED_EXCLUDE_CLASS);
221    }
222    classes.join(EXCLUDE_CLASS_SEPARATOR)
223}

225/// Parses an OSRM route payload into a [`RouteData`].
226///
227/// # Parameters
228/// - `payload` — a decoded OSRM route response whose `code` is `Ok`.
229///
230/// # Returns
231/// The first route's total distance and duration, the bounding box derived
232/// from its full geometry, and one leg per consecutive waypoint pair.
233///
234/// # Errors
235/// Returns [`SDKError::ResponseParsingFailure`] when the payload lacks a
236/// route, its numeric totals, its GeoJSON geometry, its legs or a leg's steps,
237/// or when a coordinate is outside the valid ranges.
238pub fn parse_route_payload(payload: &Value) -> Result<RouteData, SDKError> {
239    let routes = payload
240        .get("routes")
241        .and_then(Value::as_array)
242        .ok_or_else(|| parsing_failure("OSRM route payload carries no routes array"))?;
243    let route = routes
244        .first()
245        .ok_or_else(|| parsing_failure("OSRM route payload carries no routes"))?;
246
247    let distance_meters = required_number(route, "distance", "route")?;
248    let duration_seconds = required_number(route, "duration", "route")?;
249
250    let geometry = route
251        .get("geometry")
252        .ok_or_else(|| parsing_failure("OSRM route payload carries no geometry"))?;
253    let route_shape = parse_geojson_line_string(geometry, "route geometry")?;
254    let bounding_box = bounding_box_from_coordinates(&route_shape)?;
255
256    let leg_values = route
257        .get("legs")
258        .and_then(Value::as_array)
259        .ok_or_else(|| parsing_failure("OSRM route payload carries no legs array"))?;
260    if leg_values.is_empty() {
261        return Err(parsing_failure("OSRM route payload carries no legs"));
262    }
263    let legs = leg_values
264        .iter()
265        .map(parse_route_leg)
266        .collect::<Result<Vec<RouteLeg>, SDKError>>()?;
267
268    Ok(RouteData {
269        distance_meters,
270        duration_seconds,
271        bounding_box,
272        legs,
273    })
274}

276/// Parses one leg: totals plus its ordered steps.
277fn parse_route_leg(leg: &Value) -> Result<RouteLeg, SDKError> {
278    let distance_meters = required_number(leg, "distance", "route leg")?;
279    let duration_seconds = required_number(leg, "duration", "route leg")?;
280    let step_values = leg
281        .get("steps")
282        .and_then(Value::as_array)
283        .ok_or_else(|| parsing_failure("OSRM route leg carries no steps array"))?;
284    let steps = step_values
285        .iter()
286        .map(parse_route_step)
287        .collect::<Result<Vec<RouteStep>, SDKError>>()?;
288    Ok(RouteLeg {
289        distance_meters,
290        duration_seconds,
291        steps,
292    })
293}

295/// Parses one step: totals, maneuver classification, authored instruction
296/// text and the step's GeoJSON shape.
297fn parse_route_step(step: &Value) -> Result<RouteStep, SDKError> {
298    let distance_meters = required_number(step, "distance", "route step")?;
299    let duration_seconds = required_number(step, "duration", "route step")?;
300    let maneuver = step
301        .get("maneuver")
302        .ok_or_else(|| parsing_failure("OSRM route step carries no maneuver"))?;
303    let maneuver_type_text = maneuver
304        .get("type")
305        .and_then(Value::as_str)
306        .ok_or_else(|| parsing_failure("OSRM route step maneuver carries no type"))?;
307    let modifier = maneuver.get("modifier").and_then(Value::as_str);
308    let exit_index = maneuver
309        .get("exit")
310        .and_then(Value::as_u64)
311        .map(|exit| exit.min(u64::from(u8::MAX)) as u8);
312    // OSRM marks a ferry crossing on the step's `mode`, not on the maneuver
313    // type, whose vocabulary has no ferry entry. The step `mode` takes
314    // precedence so an entering, crossing or leaving ferry step is classified
315    // as a ferry maneuver.
316    let step_mode = step.get("mode").and_then(Value::as_str);
317    let maneuver_type = if step_mode == Some(FERRY_STEP_MODE) {
318        ManeuverType::Ferry
319    } else {
320        map_osrm_maneuver(maneuver_type_text, modifier, exit_index)
321    };
322
323    let road_name = step.get("name").and_then(Value::as_str).unwrap_or_default();
324    let instruction_display_text = instruction_text(&maneuver_type, road_name);
325    let shape = match step.get("geometry") {
326        Some(geometry) => parse_geojson_line_string(geometry, "step geometry")?,
327        None => Vec::new(),
328    };
329
330    Ok(RouteStep {
331        distance_meters,
332        duration_seconds,
333        maneuver_type,
334        instruction_display_text: instruction_display_text.clone(),
335        instruction_speech_text: instruction_display_text,
336        shape,
337        // OSRM carries lane data on intersections in a provider-specific
338        // vocabulary; it is not mapped into the contract's lane guidance.
339        lane_guidances: Vec::new(),
340    })
341}

351/// Parses a GeoJSON `LineString` geometry into coordinates.
352///
353/// GeoJSON positions are `[longitude, latitude]`; the contract's coordinate
354/// exposes the same pair in named fields.
355fn parse_geojson_line_string(
356    geometry: &Value,
357    context: &str,
358) -> Result<Vec<GeographicCoordinate>, SDKError> {
359    if let Some(geometry_type) = geometry.get("type").and_then(Value::as_str) {
360        if geometry_type != "LineString" {
361            return Err(parsing_failure(format!(
362                "OSRM {context} is {geometry_type}, not LineString"
363            )));
364        }
365    }
366    let coordinate_values = geometry
367        .get("coordinates")
368        .and_then(Value::as_array)
369        .ok_or_else(|| parsing_failure(format!("OSRM {context} carries no coordinates array")))?;
370    let mut coordinates = Vec::with_capacity(coordinate_values.len());
371    for coordinate_value in coordinate_values {
372        let position = coordinate_value.as_array().ok_or_else(|| {
373            parsing_failure(format!(
374                "OSRM {context} has a position that is not an array"
375            ))
376        })?;
377        if position.len() < 2 {
378            return Err(parsing_failure(format!(
379                "OSRM {context} has a position with fewer than two elements"
380            )));
381        }
382        let longitude = position[0].as_f64().ok_or_else(|| {
383            parsing_failure(format!("OSRM {context} has a non-numeric longitude"))
384        })?;
385        let latitude = position[1]
386            .as_f64()
387            .ok_or_else(|| parsing_failure(format!("OSRM {context} has a non-numeric latitude")))?;
388        coordinates.push(
389            GeographicCoordinate::from_degrees(latitude, longitude).map_err(|error| {
390                parsing_failure(format!("OSRM {context} has an invalid coordinate: {error}"))
391            })?,
392        );
393    }
394    Ok(coordinates)
395}

397/// Derives the route bounding box from its full geometry.
398fn bounding_box_from_coordinates(
399    coordinates: &[GeographicCoordinate],
400) -> Result<BoundingBox, SDKError> {
401    let first = coordinates
402        .first()
403        .ok_or_else(|| parsing_failure("OSRM route geometry is empty"))?;
404    let mut minimum_latitude = first.latitude;
405    let mut maximum_latitude = first.latitude;
406    let mut minimum_longitude = first.longitude;
407    let mut maximum_longitude = first.longitude;
408    for coordinate in coordinates {
409        minimum_latitude = minimum_latitude.min(coordinate.latitude);
410        maximum_latitude = maximum_latitude.max(coordinate.latitude);
411        minimum_longitude = minimum_longitude.min(coordinate.longitude);
412        maximum_longitude = maximum_longitude.max(coordinate.longitude);
413    }
414    BoundingBox::from_corners(
415        GeographicCoordinate {
416            latitude: minimum_latitude,
417            longitude: minimum_longitude,
418        },
419        GeographicCoordinate {
420            latitude: maximum_latitude,
421            longitude: maximum_longitude,
422        },
423    )
424    .map_err(|error| {
425        parsing_failure(format!(
426            "OSRM route geometry bounding box is invalid: {error}"
427        ))
428    })
429}

431/// Maps OSRM's documented maneuver vocabulary onto [`ManeuverType`].
432///
433/// | OSRM `maneuver.type` | Contract maneuver |
434/// |---|---|
435/// | `depart` | [`ManeuverType::Departure`] |
436/// | `arrive` | [`ManeuverType::Arrival`] |
437/// | `turn`, `end of road` | [`ManeuverType::Turn`] from `modifier`; [`ManeuverType::Continue`] when the modifier is missing or unknown |
438/// | `continue` | [`ManeuverType::Turn`] from `modifier`; `Turn(Straight)` when the modifier is missing or unknown |
439/// | `new name`, `use lane`, `notification` | [`ManeuverType::Continue`] |
440/// | `merge` | [`ManeuverType::Merge`] |
441/// | `on ramp` | [`ManeuverType::OnRamp`] |
442/// | `off ramp` | [`ManeuverType::OffRamp`] |
443/// | `fork` | [`ManeuverType::Fork`] |
444/// | `roundabout`, `rotary`, `roundabout turn`, `exit roundabout`, `exit rotary` | [`ManeuverType::Roundabout`] with the one-based `exit` index |
445/// | anything else | [`ManeuverType::Continue`] |
446///
447/// Ferry crossings are not part of this table: OSRM marks them on the step's
448/// `mode` field, and a step whose mode is `ferry` maps to
449/// [`ManeuverType::Ferry`] before this function is consulted.
450///
451/// The modifier vocabulary is `uturn` (or `u-turn`), `sharp left`, `left`,
452/// `slight left`, `straight`, `slight right`, `right` and `sharp right`. A
453/// `turn` or `end of road` maneuver with no recognized modifier maps to
454/// [`ManeuverType::Continue`] rather than guessing a turn direction. A
455/// roundabout without a positive `exit` index maps to exit `1`, keeping every
456/// maneuver's stable label valid.
457///
458/// # Parameters
459/// - `maneuver_type` — OSRM's `maneuver.type` string.
460/// - `modifier` — OSRM's optional `maneuver.modifier` string.
461/// - `exit_index` — OSRM's optional one-based roundabout `exit` index.
462///
463/// # Returns
464/// The contract maneuver for the OSRM classification. Unknown types use the
465/// documented [`ManeuverType::Continue`] fallback.
466pub fn map_osrm_maneuver(
467    maneuver_type: &str,
468    modifier: Option<&str>,
469    exit_index: Option<u8>,
470) -> ManeuverType {
471    match maneuver_type {
472        "depart" => ManeuverType::Departure,
473        "arrive" => ManeuverType::Arrival,
474        "turn" | "end of road" => turn_from_modifier(modifier).unwrap_or(ManeuverType::Continue),
475        "continue" => {
476            turn_from_modifier(modifier).unwrap_or(ManeuverType::Turn(TurnDirection::Straight))
477        }
478        "new name" | "use lane" | "notification" => ManeuverType::Continue,
479        "merge" => ManeuverType::Merge,
480        "on ramp" => ManeuverType::OnRamp,
481        "off ramp" => ManeuverType::OffRamp,
482        "fork" => ManeuverType::Fork,
483        "roundabout" | "rotary" | "roundabout turn" | "exit roundabout" | "exit rotary" => {
484            ManeuverType::Roundabout(
485                exit_index
486                    .filter(|index| *index >= DEFAULT_ROUNDABOUT_EXIT_INDEX)
487                    .unwrap_or(DEFAULT_ROUNDABOUT_EXIT_INDEX),
488            )
489        }
490        _ => ManeuverType::Continue,
491    }
492}

494/// Builds the turn maneuver a recognized OSRM modifier names; `None` for an
495/// absent or unrecognized modifier.
496fn turn_from_modifier(modifier: Option<&str>) -> Option<ManeuverType> {
497    let direction = match modifier? {
498        "straight" => TurnDirection::Straight,
499        "slight left" => TurnDirection::SlightLeft,
500        "left" => TurnDirection::Left,
501        "sharp left" => TurnDirection::SharpLeft,
502        "slight right" => TurnDirection::SlightRight,
503        "right" => TurnDirection::Right,
504        "sharp right" => TurnDirection::SharpRight,
505        "uturn" | "u-turn" => TurnDirection::UTurn,
506        _ => return None,
507    };
508    Some(ManeuverType::Turn(direction))
509}

511/// Authors the display instruction for one maneuver.
512///
513/// OSRM returns maneuver classifications rather than prose, so the text is
514/// generated from the maneuver and the road name OSRM reports for the step.
515/// The speech string carries the same text; there is no separate spoken form
516/// in the provider payload.
517fn instruction_text(maneuver_type: &ManeuverType, road_name: &str) -> String {
518    let onto = if road_name.is_empty() {
519        String::new()
520    } else {
521        format!(" onto {road_name}")
522    };
523    match maneuver_type {
524        ManeuverType::Departure => {
525            if road_name.is_empty() {
526                "Start navigation".to_string()
527            } else {
528                format!("Start on {road_name}")
529            }
530        }
531        ManeuverType::Arrival => "Arrive at your destination".to_string(),
532        ManeuverType::Continue => format!("Continue{onto}"),
533        ManeuverType::Turn(direction) => {
534            let action = match direction {
535                TurnDirection::Straight => "Continue straight",
536                TurnDirection::SlightLeft => "Turn slight left",
537                TurnDirection::Left => "Turn left",
538                TurnDirection::SharpLeft => "Turn sharp left",
539                TurnDirection::SlightRight => "Turn slight right",
540                TurnDirection::Right => "Turn right",
541                TurnDirection::SharpRight => "Turn sharp right",
542                TurnDirection::UTurn => "Make a U-turn",
543                TurnDirection::LeavingRoad => "Leave the current road",
544            };
545            format!("{action}{onto}")
546        }
547        ManeuverType::Merge => format!("Merge{onto}"),
548        ManeuverType::OnRamp => format!("Take the ramp{onto}"),
549        ManeuverType::OffRamp => format!("Take the exit{onto}"),
550        ManeuverType::Fork => format!("Take the fork{onto}"),
551        ManeuverType::Roundabout(exit_index) => {
552            format!("Take exit {exit_index} at the roundabout")
553        }
554        ManeuverType::Ferry => "Take the ferry".to_string(),
555    }
556}

619    /// own waypoints in OSRM's `longitude,latitude` order.
