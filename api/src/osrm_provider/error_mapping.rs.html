1//! OSRM response interpretation: HTTP status and provider error envelopes
2//! become typed [`SDKError`] values.
3//!
4//! OSRM reports application failures in a JSON envelope:
5//!
6//! ```json
7//! {"code": "NoRoute", "message": "Impossible route between points"}
8//! ```
9//!
10//! The envelope is checked before the HTTP status, so the provider's own
11//! message survives even when the service returns it with a 4xx status. A
12//! non-success status without such an envelope is a transport-level failure.
13
14use geospatial_types::SDKError;
15use provider_transport::{HTTP_ERROR_STATUS_CODE_THRESHOLD, HttpResponse};
16use serde_json::Value;
17
18/// The `code` value of a successful OSRM response.
19pub const OSRM_SUCCESS_CODE: &str = "Ok";
20
21/// Interprets one OSRM JSON response.
22///
23/// # Parameters
24/// - `response` — status, headers and raw body returned by the transport.
25///
26/// # Returns
27/// The decoded JSON payload when the response is successful and its `code` is
28/// either absent or [`OSRM_SUCCESS_CODE`].
29///
30/// # Errors
31/// - [`SDKError::ProviderRequestRejected`] when the body carries a `code`
32///   other than [`OSRM_SUCCESS_CODE`]; the detail is OSRM's `message` verbatim
33///   (or the code when no message is present).
34/// - [`SDKError::ResponseParsingFailure`] when a successful response body is
35///   not valid JSON.
36/// - [`SDKError::NetworkRequestFailed`] when the status signals failure and
37///   the body is not an OSRM error envelope.
38pub fn interpret_json_response(response: HttpResponse) -> Result<Value, SDKError> {
39    let successful_status = response.status_code < HTTP_ERROR_STATUS_CODE_THRESHOLD;
40    let body_text = String::from_utf8_lossy(&response.body);
41    match serde_json::from_slice::<Value>(&response.body) {
42        Ok(payload) => {
43            if let Some(rejection) = provider_rejection(&payload) {
44                return Err(rejection);
45            }
46            if successful_status {
47                Ok(payload)
48            } else {
49                Err(network_failure(response.status_code, &body_text))
50            }
51        }
52        Err(parse_error) if successful_status => Err(SDKError::ResponseParsingFailure(format!(
53            "OSRM response body is not valid JSON: {parse_error}"
54        ))),
55        Err(_) => Err(network_failure(response.status_code, &body_text)),
56    }
57}
58
59/// Returns the provider rejection an OSRM `code` names, or `None` for a
60/// successful or code-less payload.
61fn provider_rejection(payload: &Value) -> Option<SDKError> {
62    let code = payload.get("code").and_then(Value::as_str)?;
63    if code == OSRM_SUCCESS_CODE {
64        return None;
65    }
66    let message = payload
67        .get("message")
68        .and_then(Value::as_str)
69        .unwrap_or(code);
70    Some(SDKError::ProviderRequestRejected(message.to_string()))
71}
72
73/// Builds the transport-level failure for an unsuccessful status without a
74/// provider envelope.
75fn network_failure(status_code: u16, body_text: &str) -> SDKError {
76    SDKError::NetworkRequestFailed(format!(
77        "OSRM responded with HTTP status {status_code}: {body_text}"
78    ))
79}

81#[cfg(test)]
82mod tests {
83    use super::{OSRM_SUCCESS_CODE, interpret_json_response};
84    use geospatial_types::SDKError;
85    use provider_transport::HttpResponse;
86
87    fn response(status_code: u16, body: &str) -> HttpResponse {
88        HttpResponse {
89            status_code,
90            headers: Vec::new(),
91            body: body.as_bytes().to_vec(),
92        }
93    }
94
95    #[test]
96    fn success_envelope_yields_the_payload() {
97        let payload = interpret_json_response(response(
98            200,
99            &format!(r#"{{"code":"{OSRM_SUCCESS_CODE}","routes":[]}}"#),
100        ))
101        .unwrap();
102        assert_eq!(payload["code"], OSRM_SUCCESS_CODE);
103    }
104
105    #[test]
106    fn provider_rejection_keeps_the_message_verbatim() {
107        let error = interpret_json_response(response(
108            400,
109            r#"{"code":"NoRoute","message":"Impossible route between points"}"#,
110        ))
111        .unwrap_err();
112        match error {
113            SDKError::ProviderRequestRejected(message) => {
114                assert_eq!(message, "Impossible route between points");
115            }
116            other => panic!("expected a provider rejection, got {other:?}"),
117        }
118    }
119
120    #[test]
121    fn server_failure_without_an_envelope_is_a_network_failure() {
122        let error = interpret_json_response(response(500, "upstream failure")).unwrap_err();
123        assert!(matches!(error, SDKError::NetworkRequestFailed(_)));
124    }
125
126    #[test]
127    fn malformed_successful_body_is_a_parsing_failure() {
128        let error = interpret_json_response(response(200, "not json")).unwrap_err();
129        assert!(matches!(error, SDKError::ResponseParsingFailure(_)));
130    }
131}
