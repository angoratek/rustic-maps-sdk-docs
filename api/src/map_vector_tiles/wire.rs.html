1//! The protobuf wire-format reader the tile decoder is built on.
2//!
3//! A single pass, zero-copy, borrow-only reader: it hands out borrowed
4//! slices for length-delimited fields and never allocates in the
5//! hot path of the decode. Only the wire primitives the MVT schema uses
6//! are implemented — varints, fixed-width scalars and length-delimited
7//! bytes — with every malformed input surfacing a typed
8//! [`VectorTileError`] instead of a panic.
9
10use crate::error::VectorTileError;
11
12/// Wire type 0: the base-128 varint.
13pub(crate) const WIRE_TYPE_VARINT: u64 = 0;
14/// Wire type 1: a 64-bit little-endian value.
15pub(crate) const WIRE_TYPE_FIXED64: u64 = 1;
16/// Wire type 2: a length-prefixed byte string.
17pub(crate) const WIRE_TYPE_LENGTH_DELIMITED: u64 = 2;
18/// Wire type 5: a 32-bit little-endian value.
19pub(crate) const WIRE_TYPE_FIXED32: u64 = 5;
20
21/// The most bytes a 64-bit varint occupies (7 bits per byte).
22pub(crate) const MAX_VARINT_LENGTH_BYTES: u32 = 10;
23
24/// Number of payload bits each varint byte carries.
25const VARINT_BITS_PER_BYTE: u32 = 7;
26
27/// A decoded protobuf field key: the field number and its wire type.
28#[derive(Debug, Clone, Copy, PartialEq, Eq)]
29pub(crate) struct WireFieldKey {
30    /// The field's number in the message.
31    pub field_number: u32,
32    /// The wire type the field's value travels with.
33    pub wire_type: u64,
34}
35
36/// The borrow-only protobuf wire-format cursor.
37pub(crate) struct WireReader<'bytes> {
38    bytes: &'bytes [u8],
39    position: usize,
40}
41
42impl<'bytes> WireReader<'bytes> {
43    /// Starts a read pass over a complete encoded message's bytes.
44    pub fn new(bytes: &'bytes [u8]) -> Self {
45        Self { bytes, position: 0 }
46    }
47
48    /// Whether every byte of the message has been consumed.
49    pub fn at_end(&self) -> bool {
50        self.position >= self.bytes.len()
51    }
52
53    /// Reads the next byte, or reports truncation.
54    fn next_byte(&mut self, context: &'static str) -> Result<u8, VectorTileError> {
55        let byte = self
56            .bytes
57            .get(self.position)
58            .copied()
59            .ok_or(VectorTileError::TruncatedBuffer { context })?;
60        self.position += 1;
61        Ok(byte)
62    }
63
64    /// Reads a base-128 varint, rejecting byte counts past
65    /// [`MAX_VARINT_LENGTH_BYTES`] and payloads that overflow 64 bits.
66    pub fn read_varint(&mut self, context: &'static str) -> Result<u64, VectorTileError> {
67        let mut value: u64 = 0;
68        for byte_index in 0..MAX_VARINT_LENGTH_BYTES {
69            let byte = self.next_byte(context)?;
70            let payload = u64::from(byte & 0b0111_1111);
71            let shift = byte_index * VARINT_BITS_PER_BYTE;
72            let is_terminal_byte = byte & 0b1000_0000 == 0;
73            if shift >= 64 || (shift == 63 && payload > 1) {
74                return Err(VectorTileError::VarintValueExceeds64Bits { context });
75            }
76            value |= payload << shift;
77            if is_terminal_byte {
78                return Ok(value);
79            }
80        }
81        Err(VectorTileError::VarintByteCountExceeded {
82            maximum_byte_count: MAX_VARINT_LENGTH_BYTES,
83            context,
84        })
85    }
86
87    /// Reads a field key, splitting it into field number and wire type and
88    /// refusing zero field numbers and unknown wire types.
89    pub fn read_field_key(
90        &mut self,
91        context: &'static str,
92    ) -> Result<WireFieldKey, VectorTileError> {
93        let key = self.read_varint(context)?;
94        let wire_type = key & 0b0111;
95        let field_number = key >> 3;
96        if field_number == 0 {
97            return Err(VectorTileError::InvalidWireField {
98                field_number: 0,
99                wire_type,
100                context,
101            });
102        }
103        if field_number > u32::MAX as u64 {
104            return Err(VectorTileError::InvalidWireField {
105                field_number: field_number as u32,
106                wire_type,
107                context,
108            });
109        }
110        if wire_type != WIRE_TYPE_VARINT
111            && wire_type != WIRE_TYPE_FIXED64
112            && wire_type != WIRE_TYPE_LENGTH_DELIMITED
113            && wire_type != WIRE_TYPE_FIXED32
114        {
115            return Err(VectorTileError::InvalidWireField {
116                field_number: field_number as u32,
117                wire_type,
118                context,
119            });
120        }
121        Ok(WireFieldKey {
122            field_number: field_number as u32,
123            wire_type,
124        })
125    }
126
127    /// Reads a length-prefixed byte string, returning the borrowed slice.
128    pub fn read_length_delimited(
129        &mut self,
130        context: &'static str,
131    ) -> Result<&'bytes [u8], VectorTileError> {
132        let claimed_length = self.read_varint(context)?;
133        let remaining = &self.bytes[self.position..];
134        let length = claimed_length as usize;
135        if length > remaining.len() {
136            return Err(VectorTileError::LengthDelimitedOverrun {
137                claimed_length,
138                context,
139            });
140        }
141        let bytes = &remaining[..length];
142        self.position += length;
143        Ok(bytes)
144    }
145
146    /// Reads a 32-bit little-endian value, as `[u8; 4]` ready for integer
147    /// or floating point interpretation.
148    pub fn read_fixed32(&mut self, context: &'static str) -> Result<[u8; 4], VectorTileError> {
149        let remaining = &self.bytes[self.position..];
150        let bytes: [u8; 4] = remaining
151            .get(0..4)
152            .and_then(|slice| slice.try_into().ok())
153            .ok_or(VectorTileError::TruncatedBuffer { context })?;
154        self.position += 4;
155        Ok(bytes)
156    }
157
158    /// Reads a 64-bit little-endian value, as `[u8; 8]` ready for integer
159    /// or floating point interpretation.
160    pub fn read_fixed64(&mut self, context: &'static str) -> Result<[u8; 8], VectorTileError> {
161        let remaining = &self.bytes[self.position..];
162        let bytes: [u8; 8] = remaining
163            .get(0..8)
164            .and_then(|slice| slice.try_into().ok())
165            .ok_or(VectorTileError::TruncatedBuffer { context })?;
166        self.position += 8;
167        Ok(bytes)
168    }
169
170    /// Skips an unknown field's value on the forward-compatibility path
171    /// (proto extensions): varints and fixed scalars are consumed, byte
172    /// strings are jumped over, and proto group markers are refused as
173    /// invalid keys.
174    pub fn skip_field(
175        &mut self,
176        field_key: WireFieldKey,
177        context: &'static str,
178    ) -> Result<(), VectorTileError> {
179        match field_key.wire_type {
180            WIRE_TYPE_VARINT => {
181                self.read_varint(context)?;
182            }
183            WIRE_TYPE_FIXED64 => {
184                self.read_fixed64(context)?;
185            }
186            WIRE_TYPE_LENGTH_DELIMITED => {
187                self.read_length_delimited(context)?;
188            }
189            WIRE_TYPE_FIXED32 => {
190                self.read_fixed32(context)?;
191            }
192            _ => {
193                return Err(VectorTileError::InvalidWireField {
194                    field_number: field_key.field_number,
195                    wire_type: field_key.wire_type,
196                    context,
197                });
198            }
199        }
200        Ok(())
201    }
202}

204/// Decodes the ZigZag transform the specification applies to `ParameterInteger`
205/// deltas and `sint` attribute values: `value = (ParameterInteger >> 1) ^
206/// -(ParameterInteger & 1)`.
207pub(crate) fn zigzag_decode(parameter_integer: u64) -> i64 {
208    let magnitude = (parameter_integer >> 1) as i64;
209    let sign_bit = (parameter_integer & 1) as i64;
210    magnitude ^ -sign_bit
211}

221    /// Appends the base-128 varint encoding of `value` to `bytes`, the
222    /// mirror of the decoder's reader.
223    fn append_varint(value: u64, bytes: &mut Vec<u8>) {
224        let mut rest = value;
225        loop {
226            let byte = (rest & 0x7f) as u8;
227            rest >>= 7;
228            if rest == 0 {
229                bytes.push(byte);
230                break;
231            }
232            bytes.push(byte | 0x80);
233        }
234    }

236    /// Builds a one-field-key buffer for key-reading tests.
237    fn field_key_bytes(key: u64) -> [u8; 16] {
238        let mut bytes = Vec::new();
239        append_varint(key, &mut bytes);
240        bytes.extend_from_slice(&[0x00; 15]);
241        bytes.try_into().unwrap()
242    }

369    fn skip_field_consumes_each_supported_wire_type() {
370        let mut bytes = Vec::new();
371        // field 3, fixed32: four payload bytes.
372        append_varint((3 << 3) | WIRE_TYPE_FIXED32, &mut bytes);
373        bytes.extend_from_slice(&[0x01, 0x02, 0x03, 0x04]);
374        // field 3, fixed64: eight payload bytes.
375        append_varint((3 << 3) | WIRE_TYPE_FIXED64, &mut bytes);
376        bytes.extend_from_slice(&[0x01, 0x02, 0x03, 0x04, 0x05, 0x06, 0x07, 0x08]);
377        // field 3, length-delimited: two payload bytes.
378        append_varint((3 << 3) | WIRE_TYPE_LENGTH_DELIMITED, &mut bytes);
379        bytes.push(0x02);
380        bytes.extend_from_slice(&[0xaa, 0xbb]);
381        // field 3, varint: one payload byte.
382        append_varint((3 << 3) | WIRE_TYPE_VARINT, &mut bytes);
383        bytes.push(0x01);
384        let mut reader = WireReader::new(&bytes);
385        for wire_type in [
386            WIRE_TYPE_FIXED32,
387            WIRE_TYPE_FIXED64,
388            WIRE_TYPE_LENGTH_DELIMITED,
389            WIRE_TYPE_VARINT,
390        ] {
391            let field_key = reader.read_field_key("test").unwrap();
392            assert_eq!(field_key.field_number, 3);
393            assert_eq!(field_key.wire_type, wire_type);
394            reader.skip_field(field_key, "test").unwrap();
395        }
396        assert!(reader.at_end());
397    }

399    #[test]
400    fn zigzag_decoding_follows_the_spec_formula() {
401        // The specification's worked transform pairs.
402        assert_eq!(zigzag_decode(0), 0);
403        assert_eq!(zigzag_decode(1), -1);
404        assert_eq!(zigzag_decode(2), 1);
405        assert_eq!(zigzag_decode(3), -2);
406        // The point example's parameters: 50 -> +25, 34 -> +17.
407        assert_eq!(zigzag_decode(50), 25);
408        assert_eq!(zigzag_decode(34), 17);
409        // The extremes decode to the mirrored signed extremes: even
410        // parameter integers are magnitudes, odd ones negate them.
411        assert_eq!(zigzag_decode(u64::MAX), i64::MIN);
412        assert_eq!(zigzag_decode(u64::MAX - 1), i64::MAX);
413        assert_eq!(zigzag_decode(u64::MAX - 2), i64::MIN + 1);
414    }
