1//! GPU-free overlay assembly: route polylines, filled polygons, point
2//! markers, and traffic-flow lines projected into clip-space solid-color
3//! triangle lists drawn over the tile quads.
4//!
5//! Overlays are the second draw layer of a captured frame: the same camera
6//! pose and viewport that assemble the tile quads
7//! ([`crate::renderer::Renderer::frame_quads`]) also project overlay shapes
8//! into clip-space triangles via [`overlay_triangle_vertices`], which the
9//! headless renderer draws through a separate solid-color pipeline in the
10//! same render pass after the tiles (see the crate-level seam notes in the
11//! browser-loop documentation: the wasm canvas mirrors the vertex contract
12//! verbatim, so the assembly here is deliberately `wgpu`-free and emits one
13//! flat, non-indexed triangle list).
14//!
15//! Design decisions recorded for this slice:
16//!
17//! * **World copies / antimeridian.** Overlay geometry wraps around the
18//!   camera exactly like tile spans do (see
19//!   [`crate::renderer::FrameProjection`]): the first vertex of a shape sits
20//!   in the camera's world copy via the shortest wrapped delta, every
21//!   following vertex unwraps continuously from it, and each resulting
22//!   segment or triangle is split against the camera-centered world-copy
23//!   windows ([`WORLD_COPY_WINDOW_OFFSETS`], reusing the tile spans' window
24//!   half-width). A shape wider than the covered window range clips to
25//!   nothing beyond it; a single unwrapping step never spans more than half a
26//!   world, so three windows cover every reachable fragment.
27//! * **Frustum clipping.** Assembly relies on the GPU's clip-space frustum
28//!   clip, the same mechanism the tile pipeline relies on for near-plane
29//!   straddling spans; no extra CPU cull runs. Screen-space-thickened shapes
30//!   (route lines, markers, traffic) additionally drop the segments and
31//!   markers with a vertex at or behind the camera because the screen-space
32//!   width offset needs a positive homogeneous weight; filled polygons emit
33//!   genuine clip-space corners and keep near-plane straddling triangles for
34//!   the GPU to clip exactly.
35//! * **Line joins.** A polyline draws as butt-capped segment quads; there are
36//!   deliberately no miter or round joins, so interior joints of a wide line
37//!   show a notch on the outside of sharp angles. Closing that gap needs a
38//!   per-vertex screen-space fan pass, deferred to a later slice together
39//!   with sprite/atlas marker rendering.
40//! * **Polygon holes.** A filled polygon carries one outer ring and any number
41//!   of hole rings. Holes join the outer boundary through the standard
42//!   bridge-splicing step before ear clipping ([`FilledPolygonOverlay`]
43//!   documents the algorithm), so the emitted triangle stream covers exactly
44//!   the ring set's even-odd interior: the overlay pipeline paints a flat
45//!   triangle list with alpha blending and has no stencil test, so a hole must
46//!   leave its region out of that list rather than erase it after the fact.
47//! * **Markers.** A marker is a regular-octagon triangle fan centered on the
48//!   projected anchor point (center anchoring, not bottom pinning): the
49//!   projected coordinate is the fan's center, so a marker's center sits
50//!   exactly at its geographic position. Sprites and label callouts are
51//!   explicitly not this slice.

191/// A filled polygon: one outer ring plus any hole rings, bridge-joined and
192/// ear-clip triangulated in one pass and painted solid
193/// [`FilledPolygonOverlay::rgba_color`] (through the overlay pipeline's alpha
194/// blending, so translucent fills keep the base tiles readable).
195///
196/// Holes stay unfilled by construction: the assembly emits a flat,
197/// non-overlapping triangle list whose triangles cover only the outer ring's
198/// interior outside every hole. The overlay pipeline has no stencil test and
199/// no erase operation, so subtracting the holes from the emitted triangle
200/// stream is the only representation its solid-color fragment stage supports.
201/// The bridge splice (`ear_clip_triangulation_with_holes`) joins each hole
202/// to the outer boundary through a zero-width slit; the slit's duplicated
203/// vertices are collinear, never paint a fragment of their own, and keep the
204/// combined boundary a single simple ring the ear clipper can walk.
205#[derive(Debug, Clone, PartialEq)]
206pub struct FilledPolygonOverlay {
207    /// Outer ring in traversal order, without a repeated closing vertex (a
208    /// repeated one is tolerated and dropped). Self-intersecting or
209    /// near-collinear rings are degenerate and may fill nothing.
210    pub ring_coordinates: Vec<GeographicCoordinate>,
211    /// Hole rings subtracted from the fill, each in traversal order and
212    /// without a repeated closing vertex. The rings' winding is normalized
213    /// before triangulation, so decoded order does not matter. A hole that
214    /// falls outside the outer ring is not contained by it and may paint
215    /// outside the fill; producers emit contained holes.
216    pub hole_ring_coordinates: Vec<Vec<GeographicCoordinate>>,
217    /// Fill color; see [`DEFAULT_FILLED_POLYGON_RGBA`] for the default.
218    pub rgba_color: [f32; 4],
219}
220
221impl FilledPolygonOverlay {
222    /// Builds a filled polygon without holes and the documented default style.

224        Self {
225            ring_coordinates,
226            hole_ring_coordinates: Vec::new(),

343/// Assembles every overlay of a frame into one flat clip-space triangle list
344/// drawn over the tile quads in the same captured frame.
345///
346/// The shapes project through the same camera pose as the tile quads, wrap
347/// around the camera's world copy like the tile spans do (see the module
348/// notes), and leave frustum clipping to the GPU's clip-space clip. An empty
349/// `overlays` slice produces no vertices, and a shape whose geometry turns
350/// out degenerate (behind the camera for screen-thickened shapes, or
351/// numerically unusable anywhere) skips only that shape's own triangles.
352pub fn overlay_triangle_vertices(
353    camera: &MapCamera,
354    viewport: &ViewportPixels,
355    field_of_view_degrees: f64,
356    overlays: &[MapOverlay],
357) -> Vec<ClipSpaceOverlayVertex> {
358    let frame_projection =
359        OverlayFrameProjection::for_frame(camera, viewport, field_of_view_degrees);
360    let mut vertices = Vec::new();
361    for overlay in overlays {
362        match overlay {
363            MapOverlay::RoutePolyline(route_polyline) => {
364                thickened_polyline_triangles(
365                    &frame_projection,
366                    viewport,
367                    &route_polyline.coordinates,
368                    route_polyline.rgba_color,
369                    route_polyline.width_pixels,
370                    &mut vertices,
371                );
372            }
373            MapOverlay::FilledPolygon(filled_polygon) => {
374                filled_polygon_triangles(
375                    &frame_projection,
376                    &filled_polygon.ring_coordinates,
377                    &filled_polygon.hole_ring_coordinates,
378                    filled_polygon.rgba_color,
379                    &mut vertices,
380                );
381            }
382            MapOverlay::PointMarkers(point_markers) => {
383                for position in &point_markers.positions {
384                    point_marker_triangles(
385                        &frame_projection,
386                        viewport,
387                        position,
388                        point_markers.rgba_color,
389                        point_markers.radius_pixels,
390                        &mut vertices,
391                    );
392                }
393            }
394            MapOverlay::TrafficFlow(traffic_flow) => {
395                for traffic_flow_data in &traffic_flow.flows {
396                    thickened_polyline_triangles(
397                        &frame_projection,
398                        viewport,
399                        &traffic_flow_data.shape_coordinates,
400                        traffic_congestion_rgba(traffic_flow_data.congestion_level),
401                        traffic_flow.width_pixels,
402                        &mut vertices,
403                    );
404                }
405            }
406        }
407    }
408    vertices
409}

411/// Per-frame projection context for overlays: the view-projection matrix, the
412/// world pixel scale, and the camera's normalized-world x, mirroring the tile
413/// renderer's frame projection (see `crate::renderer::FrameProjection`) so
414/// both layers resolve the same world-position convention per frame. Shared
415/// (crate-visible) with the location-indicator layer, which projects through
416/// the same conventions.
417pub(crate) struct OverlayFrameProjection {
418    view_projection_matrix: DMat4,
419    zoom_scale_pixels: f64,
420    center_normalized_x: f64,
421}
422
423impl OverlayFrameProjection {
424    /// Builds the projection context for one frame.
425    pub(crate) fn for_frame(
426        camera: &MapCamera,
427        viewport: &ViewportPixels,
428        field_of_view_degrees: f64,
429    ) -> Self {
430        Self {
431            view_projection_matrix: camera.view_projection_matrix(viewport, field_of_view_degrees),
432            zoom_scale_pixels: zoom_scale_pixels(camera.zoom_level),
433            center_normalized_x: coordinate_to_normalized_world(&camera.center)[0],
434        }
435    }
436
437    /// The wrapped normalized-world x of a raw normalized-world x: the
438    /// position in `(-0.5, 0.5]` around the camera's world copy, shared with
439    /// the tile spans' wrapping rule.
440    pub(crate) fn wrapped_x(&self, normalized_x: f64) -> f64 {
441        self.center_normalized_x + normalized_shortest_delta(self.center_normalized_x, normalized_x)
442    }
443
444    /// Wraps a shape's coordinates into one continuous normalized-world chain:
445    /// the first vertex sits in the camera's world copy and every following
446    /// vertex unwraps from its predecessor by the shortest wrapped delta, so
447    /// chains crossing the antimeridian continue past the copy seam instead
448    /// of tearing back across the viewport (the same wrap-as-one-unit rule
449    /// the tile spans follow).
450    fn wrapped_continuous_chain(&self, coordinates: &[GeographicCoordinate]) -> Vec<[f64; 2]> {
451        self.wrapped_continuous_chain_near(self.center_normalized_x, coordinates)
452    }
453
454    /// [`Self::wrapped_continuous_chain`] anchored on a caller-supplied
455    /// normalized-world x rather than the camera center. A hole ring uses its
456    /// outer ring's first wrapped vertex, so both rings land in the same world
457    /// copy even when the polygon straddles the antimeridian and the camera
458    /// center alone would resolve the hole to the neighboring copy.
459    fn wrapped_continuous_chain_near(
460        &self,
461        reference_x: f64,
462        coordinates: &[GeographicCoordinate],
463    ) -> Vec<[f64; 2]> {
464        let mut points = Vec::with_capacity(coordinates.len());
465        let mut previous_x: Option<f64> = None;
466        for coordinate in coordinates {
467            let normalized_world = coordinate_to_normalized_world(coordinate);
468            let wrapped_x = match previous_x {
469                None => reference_x + normalized_shortest_delta(reference_x, normalized_world[0]),
470                Some(previous_x_value) => {
471                    previous_x_value
472                        + normalized_shortest_delta(previous_x_value, normalized_world[0])
473                }
474            };
475            points.push([wrapped_x, normalized_world[1]]);
476            previous_x = Some(wrapped_x);
477        }
478        points
479    }
480
481    /// The camera-space clip position of a ground point of one world-copy
482    /// window. Camera world space is y-north pixel space (see
483    /// `MapCamera::ground_position_vector`); the window's copy offset is
484    /// subtracted from the wrapped x so the fragment continues the camera's
485    /// own copy past the window edge, exactly as the tile fragments do.
486    pub(crate) fn ground_clip_position(&self, window_offset: f64, x: f64, y: f64) -> DVec4 {
487        let x_pixels = (x - window_offset) * self.zoom_scale_pixels;
488        let y_pixels = -y * self.zoom_scale_pixels;
489        self.view_projection_matrix * DVec4::new(x_pixels, y_pixels, 0.0, 1.0)
490    }
491
492    /// The wrapped normalized-world x-range one world-copy window spans:
493    /// `[window_start_x, window_end_x)` around the camera's world copy,
494    /// shared by the segment fragments' split rule and the location
495    /// indicator's per-window anchor containment.
496    pub(crate) fn window_x_range(&self, window_offset: f64) -> [f64; 2] {
497        let window_start_x =
498            self.center_normalized_x - CAMERA_WORLD_WINDOW_HALF_WIDTH + window_offset;
499        [
500            window_start_x,
501            window_start_x + 2.0 * CAMERA_WORLD_WINDOW_HALF_WIDTH,
502        ]
503    }
504}

506/// Divides a clip position to viewport pixel coordinates (origin top-left, y
507/// down), matching [`MapCamera::project_to_screen`]. Sound only for a
508/// positive homogeneous weight; callers check that first where screen-space
509/// math follows.
510pub(crate) fn clip_position_to_screen_pixels(
511    clip_position: DVec4,
512    viewport: &ViewportPixels,
513) -> [f64; 2] {
514    let normalized_device_x = clip_position.x / clip_position.w;
515    let normalized_device_y = clip_position.y / clip_position.w;
516    [
517        (normalized_device_x + 1.0) * 0.5 * viewport.width_pixels,
518        (1.0 - (normalized_device_y + 1.0) * 0.5) * viewport.height_pixels,
519    ]
520}

522/// Re-expands a screen-pixel offset of one clip-space reference point into a
523/// clip position carrying the reference's depth and homogeneous weight. The
524/// screen offset is applied in normalized-device space around the reference's
525/// divided position, so per-vertex interpolation across a thickened quad ends
526/// at exactly the offset screen corners; the quad's interior follows the
527/// same perspective weighting as the underlying ground line.
528pub(crate) fn screen_pixels_to_clip_position(
529    screen_pixels: [f64; 2],
530    reference_clip_position: DVec4,
531    viewport: &ViewportPixels,
532) -> [f64; 4] {
533    let normalized_device_x = 2.0 * screen_pixels[0] / viewport.width_pixels - 1.0;
534    let normalized_device_y = 1.0 - 2.0 * screen_pixels[1] / viewport.height_pixels;
535    [
536        normalized_device_x * reference_clip_position.w,
537        normalized_device_y * reference_clip_position.w,
538        reference_clip_position.z,
539        reference_clip_position.w,
540    ]
541}

544/// non-finite: an f32-overflowing corner would poison the whole draw with
545/// undefined geometry, while skipping keeps the rest of the overlay healthy.
546pub(crate) fn push_triangle(
547    vertices: &mut Vec<ClipSpaceOverlayVertex>,
548    corners: &[[f64; 4]; 3],
549    rgba_color: [f32; 4],
550) {
551    if corners
552        .iter()
553        .any(|corner| corner.iter().any(|component| !component.is_finite()))
554    {
555        return;
556    }
557    for corner in corners {
558        vertices.push(ClipSpaceOverlayVertex {
559            clip_position: [
560                corner[0] as f32,
561                corner[1] as f32,
562                corner[2] as f32,
563                corner[3] as f32,
564            ],
565            rgba_color,
566        });
567    }

570/// One piece of a shape segment after splitting against a world-copy window,
571/// with its continuous normalized-world endpoints and the window's copy
572/// offset for [`OverlayFrameProjection::ground_clip_position`].
573#[derive(Debug, Clone, Copy, PartialEq)]
574struct SegmentWindowFragment {
575    /// Continuous normalized-world start point of the fragment.
576    start: [f64; 2],
577    /// Continuous normalized-world end point of the fragment.
578    end: [f64; 2],
579    /// Whole-world copy offset of the window the fragment lies in.
580    window_offset: f64,
581}

583/// Splits a shape segment against the camera-centered world-copy windows.
584///
585/// Each window keeps the segment's intersection with its x-range, matching
586/// [`CAMERA_WORLD_WINDOW_HALF_WIDTH`] around the camera per offset. A
587/// vertical segment (zero x-span) joins the one window containing its x,
588/// half-open on the east so a boundary-exact segment cannot draw twice; a
589/// segment touching a window only at its edge produces no fragment, mirroring
590/// the tile spans' empty-fragment rule.
591fn segment_window_fragments(
592    frame_projection: &OverlayFrameProjection,
593    segment_start: [f64; 2],
594    segment_end: [f64; 2],
595) -> Vec<SegmentWindowFragment> {
596    let span_x = segment_end[0] - segment_start[0];
597    let mut fragments = Vec::new();
598    for window_offset in WORLD_COPY_WINDOW_OFFSETS {
599        let window_offset_f64 = f64::from(window_offset);
600        let [window_start_x, window_end_x] = frame_projection.window_x_range(window_offset_f64);
601        let (start_fraction, end_fraction) = if span_x == 0.0 {
602            let zero_span_inside_window =
603                segment_start[0] >= window_start_x && segment_start[0] < window_end_x;
604            if !zero_span_inside_window {
605                continue;
606            }
607            (0.0, 1.0)
608        } else {
609            let clamped_start_fraction =
610                ((window_start_x - segment_start[0]) / span_x).clamp(0.0, 1.0);
611            let clamped_end_fraction = ((window_end_x - segment_start[0]) / span_x).clamp(0.0, 1.0);
612            if clamped_start_fraction == clamped_end_fraction {
613                continue;
614            }
615            (clamped_start_fraction, clamped_end_fraction)
616        };
617        let fragment_start = [
618            segment_start[0] + span_x * start_fraction,
619            segment_start[1] + (segment_end[1] - segment_start[1]) * start_fraction,
620        ];
621        let fragment_end = [
622            segment_start[0] + span_x * end_fraction,
623            segment_start[1] + (segment_end[1] - segment_start[1]) * end_fraction,
624        ];
625        fragments.push(SegmentWindowFragment {
626            start: fragment_start,
627            end: fragment_end,
628            window_offset: window_offset_f64,
629        });
630    }
631    fragments
632}

667/// Assembles one thickened segment fragment as a screen-space quad: the
668/// perpendicular offset of half the line width, applied in screen pixels
669/// around each endpoint's divided position, so a line keeps its pixel width
670/// regardless of zoom and pitch.
671///
672/// A fragment with a vertex at or behind the camera draws nothing — the
673/// screen-space offset needs a positive homogeneous weight — which is the
674/// documented limitation next to the filled polygons' GPU-clipped near-plane
675/// straddling (see the module notes).
676fn thickened_segment_triangles(
677    frame_projection: &OverlayFrameProjection,
678    viewport: &ViewportPixels,
679    fragment: &SegmentWindowFragment,
680    rgba_color: [f32; 4],
681    width_pixels: f64,
682    vertices: &mut Vec<ClipSpaceOverlayVertex>,
683) {
684    let start_clip_position = frame_projection.ground_clip_position(
685        fragment.window_offset,
686        fragment.start[0],
687        fragment.start[1],
688    );
689    let end_clip_position = frame_projection.ground_clip_position(
690        fragment.window_offset,
691        fragment.end[0],
692        fragment.end[1],
693    );
694    if start_clip_position.w <= 0.0 || end_clip_position.w <= 0.0 {
695        return;
696    }
697    let start_screen = clip_position_to_screen_pixels(start_clip_position, viewport);
698    let end_screen = clip_position_to_screen_pixels(end_clip_position, viewport);
699    let direction = [
700        end_screen[0] - start_screen[0],
701        end_screen[1] - start_screen[1],
702    ];
703    let screen_length = direction[0].hypot(direction[1]);
704    if screen_length < MINIMUM_VISIBLE_SEGMENT_LENGTH_PIXELS {
705        return;
706    }
707    let perpendicular_unit = [-direction[1] / screen_length, direction[0] / screen_length];
708    let half_width_pixels = width_pixels * 0.5;
709    let start_offset = [
710        start_screen[0] + perpendicular_unit[0] * half_width_pixels,
711        start_screen[1] + perpendicular_unit[1] * half_width_pixels,
712    ];
713    let start_opposite = [
714        start_screen[0] - perpendicular_unit[0] * half_width_pixels,
715        start_screen[1] - perpendicular_unit[1] * half_width_pixels,
716    ];
717    let end_offset = [
718        end_screen[0] + perpendicular_unit[0] * half_width_pixels,
719        end_screen[1] + perpendicular_unit[1] * half_width_pixels,
720    ];
721    let end_opposite = [
722        end_screen[0] - perpendicular_unit[0] * half_width_pixels,
723        end_screen[1] - perpendicular_unit[1] * half_width_pixels,
724    ];
725    // Corner order mirrors the tile quads: start-offset, start-opposite,
726    // end-offset, end-opposite, split by the quads' `0,1,2 / 2,1,3` triangle
727    // pattern, each corner re-expanded around its own endpoint's depth.
728    let corner_clip_positions = [
729        (start_offset, start_clip_position),
730        (start_opposite, start_clip_position),
731        (end_offset, end_clip_position),
732        (end_opposite, end_clip_position),
733    ]
734    .map(|(corner_screen_pixels, reference_clip_position)| {
735        screen_pixels_to_clip_position(corner_screen_pixels, reference_clip_position, viewport)
736    });
737    push_triangle(
738        vertices,
739        &[
740            corner_clip_positions[0],
741            corner_clip_positions[1],
742            corner_clip_positions[2],
743        ],
744        rgba_color,
745    );
746    push_triangle(
747        vertices,
748        &[
749            corner_clip_positions[2],
750            corner_clip_positions[1],
751            corner_clip_positions[3],
752        ],
753        rgba_color,
754    );
755}

757/// Assembles one point marker: a regular-octagon triangle fan centered on the
758/// projected anchor (center anchoring; see the module notes), built around a
759/// single clip-space reference so the whole fan stays screen-facing.
760fn point_marker_triangles(
761    frame_projection: &OverlayFrameProjection,
762    viewport: &ViewportPixels,
763    position: &GeographicCoordinate,
764    rgba_color: [f32; 4],
765    radius_pixels: f64,
766    vertices: &mut Vec<ClipSpaceOverlayVertex>,
767) {

769    let center_clip_position = frame_projection.ground_clip_position(
770        0.0,
771        frame_projection.wrapped_x(normalized_world[0]),
772        normalized_world[1],
773    );
774    if center_clip_position.w <= 0.0 {
775        return;

807/// Assembles a filled polygon: the ring set (outer boundary plus holes)
808/// triangulates in continuous normalized-world space, every triangle splits
809/// against the world-copy windows (convex clipping keeps the pieces convex),
810/// and each clipped piece fans into triangles projected by their genuine
811/// clip-space corners. Hole rings anchor to the outer ring's world copy (see
812/// [`OverlayFrameProjection::wrapped_continuous_chain_near`]) so a polygon
813/// straddling the antimeridian keeps its holes with its exterior. Shared
814/// (crate-visible) with the location-indicator layer, whose ground-projected
815/// accuracy ring reuses the exact fill machinery (without holes).
816pub(crate) fn filled_polygon_triangles(
817    frame_projection: &OverlayFrameProjection,
818    ring_coordinates: &[GeographicCoordinate],
819    hole_ring_coordinates: &[Vec<GeographicCoordinate>],
820    rgba_color: [f32; 4],
821    vertices: &mut Vec<ClipSpaceOverlayVertex>,
822) {
823    let wrapped_ring = frame_projection.wrapped_continuous_chain(ring_coordinates);
824    let Some(reference_x) = wrapped_ring.first().map(|vertex| vertex[0]) else {
825        return;
826    };
827    let wrapped_holes: Vec<Vec<[f64; 2]>> = hole_ring_coordinates
828        .iter()
829        .map(|hole_ring| frame_projection.wrapped_continuous_chain_near(reference_x, hole_ring))
830        .collect();
831    for triangle in ear_clip_triangulation_with_holes(&wrapped_ring, &wrapped_holes) {
832        for window_offset in WORLD_COPY_WINDOW_OFFSETS {
833            let window_offset_f64 = f64::from(window_offset);
834            let [window_start_x, window_end_x] = frame_projection.window_x_range(window_offset_f64);
835            let mut clipped_ring = clip_polygon_to_x_half_plane(&triangle, window_start_x, true);
836            clipped_ring = clip_polygon_to_x_half_plane(&clipped_ring, window_end_x, false);
837            if clipped_ring.len() < 3 {
838                continue;
839            }
840            // Clipped corners project by their genuine clip-space positions,
841            // so a triangle straddling the near plane stays for the GPU to
842            // clip exactly (see the module notes); only an f32-overflowing
843            // corner drops via `push_triangle`.
844            for fan_index in 1..clipped_ring.len() - 1 {
845                let corner_clip_positions = [
846                    clipped_ring[0],
847                    clipped_ring[fan_index],
848                    clipped_ring[fan_index + 1],
849                ]
850                .map(|corner| {
851                    let corner_clip_position = frame_projection.ground_clip_position(
852                        window_offset_f64,
853                        corner[0],
854                        corner[1],
855                    );
856                    [
857                        corner_clip_position.x,
858                        corner_clip_position.y,
859                        corner_clip_position.z,
860                        corner_clip_position.w,
861                    ]
862                });
863                push_triangle(vertices, &corner_clip_positions, rgba_color);
864            }
865        }
866    }
867}

869/// Clips a convex polygon to one vertical half-plane (Sutherland–Hodgman):
870/// `keep_greater` selects `x >= boundary_x`, otherwise `x <= boundary_x`.
871///
872/// Every result stays convex (a triangle or the clipped triangle), so callers
873/// fan-triangulate it safely. A degenerate one-component x-span (all shared
874/// boundary) contributes vertices exactly on the boundary; the caller treats
875/// fewer than three remaining vertices as empty.
876fn clip_polygon_to_x_half_plane(
877    polygon: &[[f64; 2]],
878    boundary_x: f64,
879    keep_greater: bool,
880) -> Vec<[f64; 2]> {
881    let mut clipped_polygon = Vec::with_capacity(polygon.len() + 1);
882    for edge_index in 0..polygon.len() {
883        let edge_start = polygon[edge_index];
884        let edge_end = polygon[(edge_index + 1) % polygon.len()];
885        let start_inside = if keep_greater {
886            edge_start[0] >= boundary_x
887        } else {
888            edge_start[0] <= boundary_x
889        };
890        let end_inside = if keep_greater {
891            edge_end[0] >= boundary_x
892        } else {
893            edge_end[0] <= boundary_x
894        };
895        if start_inside {
896            clipped_polygon.push(edge_start);
897        }
898        if start_inside != end_inside {
899            let crossing_fraction = (boundary_x - edge_start[0]) / (edge_end[0] - edge_start[0]);
900            // A crossing at the edge's own endpoint is a vertex on the
901            // boundary; it already landed through the inside-start pushes of
902            // the neighboring edges, and pushing it again would degenerate
903            // the clipped ring into a zero-area fan triangle.
904            if crossing_fraction > 0.0 && crossing_fraction < 1.0 {
905                clipped_polygon.push([
906                    boundary_x,
907                    edge_start[1] + (edge_end[1] - edge_start[1]) * crossing_fraction,
908                ]);
909            }
910        }
911    }
912    clipped_polygon
913}

915/// Doubled signed area of a ring (the fan-formula sum, without the constant
916/// factor): positive for one winding, negative for the other, zero for a
917/// collinear ring.
918fn ring_doubled_signed_area(ring: &[[f64; 2]]) -> f64 {
919    ring.iter()
920        .enumerate()
921        .map(|(index, vertex)| {
922            let next = ring[(index + 1) % ring.len()];
923            vertex[0] * next[1] - next[0] * vertex[1]
924        })
925        .sum()
926}

928/// Twice the signed area of one triangle (a × b cross-product form): positive
929/// for counterclockwise winding, negative for clockwise.
930fn triangle_doubled_signed_area(
931    first_vertex: [f64; 2],
932    second_vertex: [f64; 2],
933    third_vertex: [f64; 2],
934) -> f64 {
935    (second_vertex[0] - first_vertex[0]) * (third_vertex[1] - first_vertex[1])
936        - (second_vertex[1] - first_vertex[1]) * (third_vertex[0] - first_vertex[0])
937}

939/// Whether a point lies inside a triangle or on its boundary, by winding-sign
940/// tests against the triangle's (consistently wound) edges.
941fn point_lies_within_triangle(
942    point: [f64; 2],
943    first_vertex: [f64; 2],
944    second_vertex: [f64; 2],
945    third_vertex: [f64; 2],
946) -> bool {
947    triangle_doubled_signed_area(point, first_vertex, second_vertex)
948        >= -EAR_CLIPPING_BOUNDARY_TOLERANCE
949        && triangle_doubled_signed_area(point, second_vertex, third_vertex)
950            >= -EAR_CLIPPING_BOUNDARY_TOLERANCE
951        && triangle_doubled_signed_area(point, third_vertex, first_vertex)
952            >= -EAR_CLIPPING_BOUNDARY_TOLERANCE
953}

955/// Triangulates a simple polygon ring by ear clipping, over the ring's own
956/// vertices in O(n²) — no external geometry dependency, consistent with the
957/// crate owning its minimal math.
958///
959/// The ring normalizes to one winding (positive doubled signed area) first,
960/// so input order does not matter. Degenerate rings (fewer than three
961/// vertices, near-zero area, or a remainder with no clipgable ear — fully
962/// collinear chains) contribute fewer triangles or none; a numerically stuck
963/// remainder stops the loop instead of spinning, matching the documented
964/// "degenerate rings may fill nothing" contract.
965fn ear_clip_triangulation(ring: &[[f64; 2]]) -> Vec<[[f64; 2]; 3]> {
966    let mut normalized_ring = ring.to_vec();
967    if normalized_ring.len() >= 2 && normalized_ring.first() == normalized_ring.last() {
968        normalized_ring.pop();
969    }
970    if normalized_ring.len() < 3 {
971        return Vec::new();
972    }
973    let doubled_signed_area = ring_doubled_signed_area(&normalized_ring);
974    if doubled_signed_area.abs() < POLYGON_AREA_DEGENERACY_TOLERANCE {
975        return Vec::new();
976    }
977    if doubled_signed_area < 0.0 {
978        normalized_ring.reverse();
979    }
980    let mut remaining_indices: Vec<usize> = (0..normalized_ring.len()).collect();
981    let mut triangles = Vec::with_capacity(normalized_ring.len() - 2);
982    while remaining_indices.len() > 3 {
983        let ear_position = (0..remaining_indices.len())
984            .find(|&position| is_ring_ear(&normalized_ring, &remaining_indices, position));
985        let Some(ear_position) = ear_position else {
986            break;
987        };
988        let corner_indices = ear_corner_indices(&remaining_indices, ear_position);
989        triangles.push([
990            normalized_ring[corner_indices[0]],
991            normalized_ring[corner_indices[1]],
992            normalized_ring[corner_indices[2]],
993        ]);
994        remaining_indices.remove(ear_position);
995    }
996    if remaining_indices.len() == 3 {
997        triangles.push([
998            normalized_ring[remaining_indices[0]],
999            normalized_ring[remaining_indices[1]],
1000            normalized_ring[remaining_indices[2]],
1001        ]);
1002    }
1003    triangles
1004}

1006/// Triangulates a ring set with holes: the outer boundary plus every hole,
1007/// bridge-joined into one simple ring and ear-clip triangulated in one pass.
1008///
1009/// The hole handling follows the standard bridge-splicing construction (the
1010/// `earcut` algorithm's `eliminateHoles` step) with the crate's own vector
1011/// math: each hole joins the boundary through a zero-width slit, so the
1012/// emitted triangles cover the ring set's interior minus the holes and never
1013/// the holes themselves. Winding normalizes first (outer positive, holes
1014/// negative), so the decoded ring order does not matter.
1015///
1016/// A degenerate outer ring fills nothing, and a hole ring that cannot be
1017/// bridged (fewer than three vertices, zero area, or no visible bridge) is
1018/// skipped rather than approximated; a numerically stuck ear-clipping
1019/// remainder stops early and leaves that remainder unfilled, matching the
1020/// documented "degenerate rings may fill nothing" contract.
1021fn ear_clip_triangulation_with_holes(
1022    outer_ring: &[[f64; 2]],
1023    hole_rings: &[Vec<[f64; 2]>],
1024) -> Vec<[[f64; 2]; 3]> {
1025    if hole_rings.is_empty() {
1026        return ear_clip_triangulation(outer_ring);
1027    }
1028    let normalized_outer = normalized_ring(outer_ring, true);
1029    if normalized_outer.len() < 3 {
1030        return Vec::new();
1031    }
1032    let mut nodes = Vec::new();
1033    let Some(ring_head) = ring_nodes(&normalized_outer, &mut nodes) else {
1034        return Vec::new();
1035    };
1036    let mut holes: Vec<Vec<[f64; 2]>> = hole_rings
1037        .iter()
1038        .map(|hole_ring| normalized_ring(hole_ring, false))
1039        .filter(|hole_ring| hole_ring.len() >= 3)
1040        .collect();
1041    // Left-to-right hole order, matching the bridge search's proven order:
1042    // each hole's westward ray then meets only edges already part of the
1043    // combined boundary.
1044    holes.sort_by(|left, right| {
1045        ring_leftmost_point(left)[0]
1046            .partial_cmp(&ring_leftmost_point(right)[0])
1047            .unwrap_or(std::cmp::Ordering::Equal)
1048    });
1049    for hole_ring in &holes {
1050        let Some(hole_head) = ring_nodes(hole_ring, &mut nodes) else {
1051            continue;
1052        };
1053        // The bridge search validates a bridge from the hole's leftmost
1054        // vertex; the splice must start the hole at that same vertex, or the
1055        // cycle would gain an unvalidated edge across the hole.
1056        let hole_leftmost = hole_leftmost_node(hole_head, &nodes);
1057        let Some(bridge_index) = find_hole_bridge(hole_leftmost, ring_head, &nodes) else {
1058            continue;
1059        };
1060        splice_hole_ring(bridge_index, hole_leftmost, &mut nodes);
1061    }
1062    ear_clip_nodes(&mut nodes, ring_head)
1063}

1065/// One node of the mutable circular ring list the hole-aware triangulator
1066/// clips ears from. Nodes are never freed: removal relinks the neighbors and
1067/// leaves the node unreachable, so arena indices stay valid across removals.
1068struct EarClipNode {
1069    /// The node's position in normalized-world space.
1070    vertex: [f64; 2],
1071    /// Arena index of the previous ring node.
1072    previous_index: usize,
1073    /// Arena index of the next ring node.
1074    next_index: usize,
1075}

1077/// Builds a circular node ring from a ring's vertices, dropping the repeated
1078/// closing vertex and consecutive duplicates. Returns the first node's arena
1079/// index, or `None` when fewer than three distinct vertices remain.
1080fn ring_nodes(points: &[[f64; 2]], nodes: &mut Vec<EarClipNode>) -> Option<usize> {
1081    let mut cleaned: Vec<[f64; 2]> = Vec::with_capacity(points.len());
1082    for &point in points {
1083        if cleaned.last() != Some(&point) {
1084            cleaned.push(point);
1085        }
1086    }
1087    if cleaned.len() >= 2 && cleaned.first() == cleaned.last() {
1088        cleaned.pop();
1089    }
1090    if cleaned.len() < 3 {
1091        return None;
1092    }
1093    let first_index = nodes.len();
1094    for &vertex in &cleaned {
1095        nodes.push(EarClipNode {
1096            vertex,
1097            previous_index: 0,
1098            next_index: 0,
1099        });
1100    }
1101    let vertex_count = cleaned.len();
1102    for position in 0..vertex_count {
1103        let node_index = first_index + position;
1104        nodes[node_index].previous_index =
1105            first_index + (position + vertex_count - 1) % vertex_count;
1106        nodes[node_index].next_index = first_index + (position + 1) % vertex_count;
1107    }
1108    Some(first_index)
1109}

1111/// Normalizes a ring to one winding: drops the repeated closing vertex and
1112/// consecutive duplicates, reverses the ring when its signed area has the
1113/// wrong sign, and returns an empty ring for fewer than three vertices or
1114/// near-zero area.
1115fn normalized_ring(ring: &[[f64; 2]], positive_winding: bool) -> Vec<[f64; 2]> {
1116    let mut normalized: Vec<[f64; 2]> = Vec::with_capacity(ring.len());
1117    for &point in ring {
1118        if normalized.last() != Some(&point) {
1119            normalized.push(point);
1120        }
1121    }
1122    if normalized.len() >= 2 && normalized.first() == normalized.last() {
1123        normalized.pop();
1124    }
1125    if normalized.len() < 3 {
1126        return Vec::new();
1127    }
1128    let doubled_signed_area = ring_doubled_signed_area(&normalized);
1129    if doubled_signed_area.abs() < POLYGON_AREA_DEGENERACY_TOLERANCE {
1130        return Vec::new();
1131    }
1132    if (doubled_signed_area > 0.0) != positive_winding {
1133        normalized.reverse();
1134    }
1135    normalized
1136}

1151/// Splices a hole ring into the outer ring at a bridge vertex: the outer
1152/// boundary turns into the hole, walks the hole ring, crosses back over the
1153/// zero-width bridge, and continues with the outer ring. `hole_head` must be
1154/// a node of the hole ring; the hole enters the cycle after `bridge_index`.
1155/// Returns the hole-head copy, the second cycle's head when the same wiring
1156/// splits a self-touching ring at a diagonal.
1157fn splice_hole_ring(bridge_index: usize, hole_head: usize, nodes: &mut Vec<EarClipNode>) -> usize {
1158    let bridge_copy_index = nodes.len();
1159    nodes.push(EarClipNode {
1160        vertex: nodes[bridge_index].vertex,
1161        previous_index: 0,
1162        next_index: 0,
1163    });
1164    let hole_head_copy_index = nodes.len();
1165    nodes.push(EarClipNode {
1166        vertex: nodes[hole_head].vertex,
1167        previous_index: 0,
1168        next_index: 0,
1169    });
1170    let bridge_next_index = nodes[bridge_index].next_index;
1171    let hole_last_index = nodes[hole_head].previous_index;
1172    // bridge -> hole head ... hole last -> hole head copy -> bridge copy ->
1173    // (original bridge next), the standard splitPolygon wiring.
1174    nodes[bridge_index].next_index = hole_head;
1175    nodes[hole_head].previous_index = bridge_index;
1176    nodes[hole_last_index].next_index = hole_head_copy_index;
1177    nodes[hole_head_copy_index].previous_index = hole_last_index;
1178    nodes[hole_head_copy_index].next_index = bridge_copy_index;
1179    nodes[bridge_copy_index].previous_index = hole_head_copy_index;
1180    nodes[bridge_copy_index].next_index = bridge_next_index;
1181    nodes[bridge_next_index].previous_index = bridge_copy_index;
1182    hole_head_copy_index
1183}

1199/// The outer ring node a hole ring should bridge to, or `None` when the hole
1200/// does not overlap the ring at all.
1201///
1202/// This is the `earcut` hole-bridge search: cast a ray westward from the
1203/// hole's leftmost vertex, take the nearest boundary edge it crosses, then
1204/// refine over the vertices of that crossing's wedge to the one with the
1205/// minimum angle that is locally inside the boundary. The refinement is what
1206/// keeps the bridge segment from crossing the ring on concave boundaries.
1207fn find_hole_bridge(
1208    hole_leftmost: usize,
1209    ring_head: usize,
1210    nodes: &[EarClipNode],
1211) -> Option<usize> {
1212    let [hole_x, hole_y] = nodes[hole_leftmost].vertex;
1213
1214    // First pass: the nearest boundary edge to the west of the hole's
1215    // leftmost vertex. The candidate is that edge's endpoint with the lesser
1216    // x, the side the hole approaches from.
1217    let mut nearest_crossing_x = f64::NEG_INFINITY;
1218    let mut bridge_index: Option<usize> = None;
1219    let mut ring_node = ring_head;
1220    loop {
1221        let ring_next = nodes[ring_node].next_index;
1222        let ring_vertex = nodes[ring_node].vertex;
1223        let ring_next_vertex = nodes[ring_next].vertex;
1224        if hole_y <= ring_vertex[1]
1225            && hole_y >= ring_next_vertex[1]
1226            && ring_next_vertex[1] != ring_vertex[1]
1227        {
1228            let crossing_x = ring_vertex[0]
1229                + (hole_y - ring_vertex[1]) * (ring_next_vertex[0] - ring_vertex[0])
1230                    / (ring_next_vertex[1] - ring_vertex[1]);
1231            if crossing_x <= hole_x && crossing_x > nearest_crossing_x {
1232                nearest_crossing_x = crossing_x;
1233                bridge_index = Some(if ring_vertex[0] < ring_next_vertex[0] {
1234                    ring_node
1235                } else {
1236                    ring_next
1237                });
1238                if crossing_x == hole_x {
1239                    return bridge_index;
1240                }
1241            }
1242        }
1243        ring_node = ring_next;
1244        if ring_node == ring_head {
1245            break;
1246        }
1247    }
1248    let initial_bridge_index = bridge_index?;
1249    let initial_bridge_vertex = nodes[initial_bridge_index].vertex;
1250    let stop_index = initial_bridge_index;
1251    let mut minimum_tangent = f64::INFINITY;
1252    let mut ring_node = initial_bridge_index;
1253    loop {
1254        let ring_vertex = nodes[ring_node].vertex;
1255        if hole_x >= ring_vertex[0]
1256            && ring_vertex[0] >= initial_bridge_vertex[0]
1257            && hole_x != ring_vertex[0]
1258            && point_lies_within_earcut_triangle(
1259                [
1260                    if hole_y < initial_bridge_vertex[1] {
1261                        hole_x
1262                    } else {
1263                        nearest_crossing_x
1264                    },
1265                    hole_y,
1266                ],
1267                initial_bridge_vertex,
1268                [
1269                    if hole_y < initial_bridge_vertex[1] {
1270                        nearest_crossing_x
1271                    } else {
1272                        hole_x
1273                    },
1274                    hole_y,
1275                ],
1276                ring_vertex,
1277            )
1278        {
1279            let tangent = (hole_y - ring_vertex[1]).abs() / (hole_x - ring_vertex[0]);
1280            if node_locally_inside(nodes, ring_node, [hole_x, hole_y]) && tangent < minimum_tangent
1281            {
1282                minimum_tangent = tangent;
1283                bridge_index = Some(ring_node);
1284            }
1285        }
1286        ring_node = nodes[ring_node].next_index;
1287        if ring_node == stop_index {
1288            break;
1289        }
1290    }
1291    bridge_index
1292}

1294/// Earcut's point-in-triangle test, written in the winding convention the
1295/// bridge search assumes (outer rings positive).
1296fn point_lies_within_earcut_triangle(
1297    point: [f64; 2],
1298    first: [f64; 2],
1299    second: [f64; 2],
1300    third: [f64; 2],
1301) -> bool {
1302    (third[0] - point[0]) * (first[1] - point[1]) >= (first[0] - point[0]) * (third[1] - point[1])
1303        && (first[0] - point[0]) * (second[1] - point[1])
1304            >= (second[0] - point[0]) * (first[1] - point[1])
1305        && (second[0] - point[0]) * (third[1] - point[1])
1306            >= (third[0] - point[0]) * (second[1] - point[1])
1307}

1309/// Whether the boundary is locally inside at a ring node for a candidate
1310/// bridge endpoint point: for a convex vertex the point must sit between the
1311/// node's two boundary edges, for a reflex vertex outside both.
1312fn node_locally_inside(nodes: &[EarClipNode], node_index: usize, point: [f64; 2]) -> bool {
1313    let previous_vertex = nodes[nodes[node_index].previous_index].vertex;
1314    let vertex = nodes[node_index].vertex;
1315    let next_vertex = nodes[nodes[node_index].next_index].vertex;
1316    if triangle_doubled_signed_area(previous_vertex, vertex, next_vertex) > 0.0 {
1317        triangle_doubled_signed_area(vertex, point, next_vertex) <= 0.0
1318            && triangle_doubled_signed_area(vertex, previous_vertex, point) <= 0.0
1319    } else {
1320        triangle_doubled_signed_area(vertex, point, previous_vertex) > 0.0
1321            || triangle_doubled_signed_area(vertex, next_vertex, point) > 0.0
1322    }
1323}

1330/// Whether the node at `ear_index` is a valid ear: convex, and with no
1331/// reflex boundary vertex inside or on its triangle. Duplicate bridge
1332/// vertices are collinear slits, so the reflex-only containment filter keeps
1333/// them from blocking every ear next to a splice.
1334fn is_ear_node(nodes: &[EarClipNode], ear_index: usize) -> bool {
1335    let previous_index = nodes[ear_index].previous_index;
1336    let next_index = nodes[ear_index].next_index;
1337    let previous_vertex = nodes[previous_index].vertex;
1338    let vertex = nodes[ear_index].vertex;
1339    let next_vertex = nodes[next_index].vertex;
1340    if triangle_doubled_signed_area(previous_vertex, vertex, next_vertex) <= 0.0 {
1341        return false;
1342    }
1343    let minimum_x = previous_vertex[0].min(vertex[0]).min(next_vertex[0]);
1344    let maximum_x = previous_vertex[0].max(vertex[0]).max(next_vertex[0]);
1345    let minimum_y = previous_vertex[1].min(vertex[1]).min(next_vertex[1]);
1346    let maximum_y = previous_vertex[1].max(vertex[1]).max(next_vertex[1]);
1347    let mut point_index = nodes[next_index].next_index;
1348    while point_index != previous_index {
1349        let point = nodes[point_index].vertex;
1350        if point[0] >= minimum_x
1351            && point[0] <= maximum_x
1352            && point[1] >= minimum_y
1353            && point[1] <= maximum_y
1354            && point_lies_within_earcut_triangle(point, previous_vertex, vertex, next_vertex)
1355        {
1356            let point_previous_vertex = nodes[nodes[point_index].previous_index].vertex;
1357            let point_next_vertex = nodes[nodes[point_index].next_index].vertex;
1358            if triangle_doubled_signed_area(point_previous_vertex, point, point_next_vertex) <= 0.0
1359            {
1360                return false;
1361            }
1362        }
1363        point_index = nodes[point_index].next_index;
1364    }
1365    true
1366}

1368/// Unlinks a node from the circular ring list, leaving it unreachable.
1369fn remove_ear_node(nodes: &mut [EarClipNode], node_index: usize) {
1370    let previous_index = nodes[node_index].previous_index;
1371    let next_index = nodes[node_index].next_index;
1372    nodes[previous_index].next_index = next_index;
1373    nodes[next_index].previous_index = previous_index;

1376/// Ear-clip loops over the assembled node ring, emitting each ear's triangle
1377/// and relinking around it. The traversal advances past the removed ear like
1378/// `earcut`, so well-formed rings stay near linear. Stalls recover in the
1379/// `earcut` order: first drop duplicate and collinear nodes, then cure local
1380/// self-intersections, then split the ring at a valid diagonal and clip both
1381/// halves. A ring that still has no ear stops instead of spinning, leaving
1382/// that remainder unfilled.
1383fn ear_clip_nodes(nodes: &mut Vec<EarClipNode>, ring_head: usize) -> Vec<[[f64; 2]; 3]> {
1384    let mut triangles = Vec::new();
1385    let mut ear_index = ring_head;
1386    let mut stop_index = ring_head;
1387    let mut filtered_once = false;
1388    let mut cured_once = false;
1389    while nodes[ear_index].previous_index != nodes[ear_index].next_index {
1390        let previous_index = nodes[ear_index].previous_index;
1391        let next_index = nodes[ear_index].next_index;
1392        if is_ear_node(nodes, ear_index) {
1393            triangles.push([
1394                nodes[previous_index].vertex,
1395                nodes[ear_index].vertex,
1396                nodes[next_index].vertex,
1397            ]);
1398            remove_ear_node(nodes, ear_index);
1399            ear_index = nodes[next_index].next_index;
1400            stop_index = ear_index;
1401            continue;
1402        }
1403        ear_index = next_index;
1404        if ear_index == stop_index {
1405            if !filtered_once {
1406                filtered_once = true;
1407                if let Some(filtered_head) = filter_ring_nodes(nodes, ear_index) {
1408                    ear_index = filtered_head;
1409                    stop_index = filtered_head;
1410                    continue;
1411                }
1412            }
1413            if !cured_once {
1414                cured_once = true;
1415                let cured_head = cure_local_intersections(nodes, ear_index, &mut triangles);
1416                ear_index = cured_head;
1417                stop_index = cured_head;
1418                continue;
1419            }
1420            if let Some((first_head, second_head)) = split_ring_at_valid_diagonal(nodes, ear_index)
1421            {
1422                triangles.extend(ear_clip_nodes(nodes, first_head));
1423                triangles.extend(ear_clip_nodes(nodes, second_head));
1424                return triangles;
1425            }
1426            break;
1427        }
1428    }
1429    triangles
1430}

1432/// The ear clipper's last resort (`earcut`'s `splitEarcut`): find any pair of
1433/// non-adjacent ring nodes joined by a valid diagonal, split the ring there
1434/// into two smaller cycles, and return both cycle heads. Splitting strictly
1435/// shrinks each piece, so the recursive call always terminates.
1436fn split_ring_at_valid_diagonal(
1437    nodes: &mut Vec<EarClipNode>,
1438    start: usize,
1439) -> Option<(usize, usize)> {
1440    let mut first = start;
1441    loop {
1442        let mut second = nodes[nodes[first].next_index].next_index;
1443        while second != nodes[first].previous_index {
1444            if is_valid_diagonal(nodes, first, second) {
1445                let second_head = splice_hole_ring(first, second, nodes);
1446                return Some((first, second_head));
1447            }
1448            second = nodes[second].next_index;
1449        }
1450        first = nodes[first].next_index;
1451        if first == start {
1452            break;
1453        }
1454    }
1455    None
1456}

1458/// Whether the diagonal between two ring nodes is safe to split at: the nodes
1459/// are non-adjacent, both ends are locally inside, the diagonal's midpoint is
1460/// inside the ring, it crosses no boundary edge, and it does not create
1461/// opposite-facing sectors (`earcut`'s `isValidDiagonal`).
1462fn is_valid_diagonal(nodes: &[EarClipNode], first: usize, second: usize) -> bool {
1463    let first_previous = nodes[first].previous_index;
1464    let first_next = nodes[first].next_index;
1465    let second_previous = nodes[second].previous_index;
1466    let second_next = nodes[second].next_index;
1467    first_next != second
1468        && first_previous != second
1469        && !diagonal_intersects_ring(nodes, first, second)
1470        && node_locally_inside_node(nodes, first, second)
1471        && node_locally_inside_node(nodes, second, first)
1472        && diagonal_middle_inside(nodes, first, second)
1473        && (triangle_doubled_signed_area(
1474            nodes[first_previous].vertex,
1475            nodes[first].vertex,
1476            nodes[second_previous].vertex,
1477        ) != 0.0
1478            || triangle_doubled_signed_area(
1479                nodes[first].vertex,
1480                nodes[second_previous].vertex,
1481                nodes[second].vertex,
1482            ) != 0.0
1483            || (nodes[first].vertex == nodes[second].vertex
1484                && triangle_doubled_signed_area(
1485                    nodes[first_previous].vertex,
1486                    nodes[first].vertex,
1487                    nodes[first_next].vertex,
1488                ) > 0.0
1489                && triangle_doubled_signed_area(
1490                    nodes[second_previous].vertex,
1491                    nodes[second].vertex,
1492                    nodes[second_next].vertex,
1493                ) > 0.0))
1494}

1496/// Whether a candidate split diagonal crosses any ring edge (`earcut`'s
1497/// `intersectsPolygon`).
1498fn diagonal_intersects_ring(nodes: &[EarClipNode], first: usize, second: usize) -> bool {
1499    let mut node = first;
1500    loop {
1501        let next = nodes[node].next_index;
1502        if node != first
1503            && next != first
1504            && node != second
1505            && next != second
1506            && segments_intersect(
1507                nodes[node].vertex,
1508                nodes[next].vertex,
1509                nodes[first].vertex,
1510                nodes[second].vertex,
1511            )
1512        {
1513            return true;
1514        }
1515        node = next;
1516        if node == first {
1517            break;
1518        }
1519    }
1520    false
1521}

1523/// Whether a candidate split diagonal's midpoint lies inside the ring
1524/// (even-odd ray cast; `earcut`'s `middleInside`).
1525fn diagonal_middle_inside(nodes: &[EarClipNode], first: usize, second: usize) -> bool {
1526    let midpoint = [
1527        (nodes[first].vertex[0] + nodes[second].vertex[0]) * 0.5,
1528        (nodes[first].vertex[1] + nodes[second].vertex[1]) * 0.5,
1529    ];
1530    let mut inside = false;
1531    let mut node = first;
1532    loop {
1533        let next = nodes[node].next_index;
1534        let current_vertex = nodes[node].vertex;
1535        let next_vertex = nodes[next].vertex;
1536        if (current_vertex[1] > midpoint[1]) != (next_vertex[1] > midpoint[1])
1537            && next_vertex[1] != current_vertex[1]
1538        {
1539            let boundary_x = (next_vertex[0] - current_vertex[0])
1540                * (midpoint[1] - current_vertex[1])
1541                / (next_vertex[1] - current_vertex[1])
1542                + current_vertex[0];
1543            if midpoint[0] < boundary_x {
1544                inside = !inside;
1545            }
1546        }
1547        node = next;
1548        if node == first {
1549            break;
1550        }
1551    }
1552    inside
1553}

1555/// The ear clipper's stall recovery: removes ring nodes that duplicate their
1556/// next node or are collinear with both neighbors, so the retried scan walks
1557/// a simpler boundary. Returns the head to resume from, or `None` when
1558/// nothing was removable.
1559fn filter_ring_nodes(nodes: &mut [EarClipNode], ring_head: usize) -> Option<usize> {
1560    let mut removed_any = false;
1561    let mut head = ring_head;
1562    loop {
1563        // Snapshot the current cycle so removal never invalidates the walk.
1564        let mut cycle_nodes = Vec::new();
1565        let mut node = head;
1566        loop {
1567            cycle_nodes.push(node);
1568            node = nodes[node].next_index;
1569            if node == head {
1570                break;
1571            }
1572        }
1573        if cycle_nodes.len() <= 3 {
1574            break;
1575        }
1576        let mut removable_node = None;
1577        for &candidate in &cycle_nodes {
1578            let previous_vertex = nodes[nodes[candidate].previous_index].vertex;
1579            let vertex = nodes[candidate].vertex;
1580            let next_vertex = nodes[nodes[candidate].next_index].vertex;
1581            let is_duplicate = vertex == next_vertex;
1582            let longer_edge_squared = (vertex[0] - previous_vertex[0])
1583                .hypot(vertex[1] - previous_vertex[1])
1584                .max((next_vertex[0] - vertex[0]).hypot(next_vertex[1] - vertex[1]))
1585                .powi(2);
1586            let is_collinear = triangle_doubled_signed_area(previous_vertex, vertex, next_vertex)
1587                .abs()
1588                <= EAR_CLIPPING_COLLINEAR_RELATIVE_TOLERANCE * longer_edge_squared;
1589            if is_duplicate || is_collinear {
1590                removable_node = Some(candidate);
1591                break;
1592            }
1593        }
1594        let Some(candidate) = removable_node else {
1595            break;
1596        };
1597        remove_ear_node(nodes, candidate);
1598        removed_any = true;
1599        if candidate == head {
1600            head = nodes[candidate].next_index;
1601        }
1602    }
1603    removed_any.then_some(head)
1604}

1606/// The ear clipper's second stall recovery (`earcut`'s
1607/// `cureLocalIntersections`): when two adjacent boundary edges cross, the
1608/// small triangle they form is emitted directly and the two crossing nodes
1609/// are removed, untangling a locally self-intersecting boundary (common in
1610/// real tile data where a hole touches its exterior at a vertex). Returns the
1611/// head to resume clipping from, after a final collinearity filter.
1612fn cure_local_intersections(
1613    nodes: &mut [EarClipNode],
1614    start: usize,
1615    triangles: &mut Vec<[[f64; 2]; 3]>,
1616) -> usize {
1617    let mut node = start;
1618    let mut head = start;
1619    loop {
1620        let previous_index = nodes[node].previous_index;
1621        let next_index = nodes[node].next_index;
1622        let after_next_index = nodes[next_index].next_index;
1623        if nodes[previous_index].vertex != nodes[after_next_index].vertex
1624            && segments_intersect(
1625                nodes[previous_index].vertex,
1626                nodes[node].vertex,
1627                nodes[next_index].vertex,
1628                nodes[after_next_index].vertex,
1629            )
1630            && node_locally_inside_node(nodes, previous_index, after_next_index)
1631            && node_locally_inside_node(nodes, after_next_index, previous_index)
1632        {
1633            triangles.push([
1634                nodes[previous_index].vertex,
1635                nodes[node].vertex,
1636                nodes[next_index].vertex,
1637            ]);
1638            remove_ear_node(nodes, node);
1639            remove_ear_node(nodes, next_index);
1640            node = after_next_index;
1641            head = after_next_index;
1642        }
1643        node = nodes[node].next_index;
1644        if node == head {
1645            break;
1646        }
1647    }
1648    filter_ring_nodes(nodes, node).unwrap_or(node)
1649}

1672/// Whether two closed segments intersect, including touching and collinear
1673/// overlaps (the `earcut` `intersects` test the local-intersection cure

1675fn segments_intersect(
1676    first_start: [f64; 2],
1677    first_end: [f64; 2],
1678    second_start: [f64; 2],
1679    second_end: [f64; 2],
1680) -> bool {
1681    let first_orientation = sign_of(triangle_doubled_signed_area(
1682        first_start,
1683        first_end,
1684        second_start,
1685    ));
1686    let second_orientation = sign_of(triangle_doubled_signed_area(
1687        first_start,
1688        first_end,
1689        second_end,
1690    ));
1691    let third_orientation = sign_of(triangle_doubled_signed_area(
1692        second_start,
1693        second_end,
1694        first_start,
1695    ));
1696    let fourth_orientation = sign_of(triangle_doubled_signed_area(
1697        second_start,
1698        second_end,
1699        first_end,
1700    ));
1701    if first_orientation != second_orientation && third_orientation != fourth_orientation {
1702        return true;
1703    }
1704    (first_orientation == 0 && point_lies_on_segment(second_start, first_start, first_end))
1705        || (second_orientation == 0 && point_lies_on_segment(second_end, first_start, first_end))
1706        || (third_orientation == 0 && point_lies_on_segment(first_start, second_start, second_end))
1707        || (fourth_orientation == 0 && point_lies_on_segment(first_end, second_start, second_end))
1708}

1710/// The previous, middle, and next remaining indices around a candidate ear
1711/// position, wrapping the current ring order.
1712fn ear_corner_indices(remaining_indices: &[usize], position: usize) -> [usize; 3] {
1713    [
1714        remaining_indices[(position + remaining_indices.len() - 1) % remaining_indices.len()],
1715        remaining_indices[position],
1716        remaining_indices[(position + 1) % remaining_indices.len()],
1717    ]
1718}

1720/// Whether the vertex at `position` of the remaining ring is a valid ear:
1721/// convex (positive triangle area in the normalized winding) and with no
1722/// other remaining vertex inside or on the candidate triangle.
1723fn is_ring_ear(ring: &[[f64; 2]], remaining_indices: &[usize], position: usize) -> bool {
1724    let corner_indices = ear_corner_indices(remaining_indices, position);
1725    let [previous_vertex, middle_vertex, next_vertex] = [
1726        ring[corner_indices[0]],
1727        ring[corner_indices[1]],
1728        ring[corner_indices[2]],
1729    ];
1730    if triangle_doubled_signed_area(previous_vertex, middle_vertex, next_vertex) <= 0.0 {
1731        return false;
1732    }
1733    remaining_indices
1734        .iter()
1735        .filter(|&&vertex_index| !corner_indices.contains(&vertex_index))
1736        .all(|&vertex_index| {
1737            !point_lies_within_triangle(
1738                ring[vertex_index],
1739                previous_vertex,
1740                middle_vertex,
1741                next_vertex,
1742            )
1743        })
1744}

2109            (end[0] - start[0]) * (point[1] - start[1])
2110                - (end[1] - start[1]) * (point[0] - start[0])

2113            edge_side(triangle[0], triangle[1]),
2114            edge_side(triangle[1], triangle[2]),
2115            edge_side(triangle[2], triangle[0]),
2116        ];
2117        !(sides.iter().any(|side| *side < 0.0) && sides.iter().any(|side| *side > 0.0))
2118    }
