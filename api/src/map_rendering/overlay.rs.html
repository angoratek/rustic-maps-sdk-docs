1//! GPU-free overlay assembly: route polylines, filled polygons, point
2//! markers, and traffic-flow lines projected into clip-space solid-color
3//! triangle lists drawn over the tile quads.
4//!
5//! Overlays are the second draw layer of a captured frame: the same camera
6//! pose and viewport that assemble the tile quads
7//! ([`crate::renderer::Renderer::frame_quads`]) also project overlay shapes
8//! into clip-space triangles via [`overlay_triangle_vertices`], which the
9//! headless renderer draws through a separate solid-color pipeline in the
10//! same render pass after the tiles (see the crate-level seam notes in the
11//! browser-loop documentation: the wasm canvas mirrors the vertex contract
12//! verbatim, so the assembly here is deliberately `wgpu`-free and emits one
13//! flat, non-indexed triangle list).
14//!
15//! Design decisions recorded for this slice:
16//!
17//! * **World copies / antimeridian.** Overlay geometry wraps around the
18//!   camera exactly like tile spans do (see
19//!   [`crate::renderer::FrameProjection`]): the first vertex of a shape sits
20//!   in the camera's world copy via the shortest wrapped delta, every
21//!   following vertex unwraps continuously from it, and each resulting
22//!   segment or triangle is split against the camera-centered world-copy
23//!   windows ([`WORLD_COPY_WINDOW_OFFSETS`], reusing the tile spans' window
24//!   half-width). A shape wider than the covered window range clips to
25//!   nothing beyond it; a single unwrapping step never spans more than half a
26//!   world, so three windows cover every reachable fragment.
27//! * **Frustum clipping.** Assembly relies on the GPU's clip-space frustum
28//!   clip, the same mechanism the tile pipeline relies on for near-plane
29//!   straddling spans; no extra CPU cull runs. Screen-space-thickened shapes
30//!   (route lines, markers, traffic) additionally drop the segments and
31//!   markers with a vertex at or behind the camera because the screen-space
32//!   width offset needs a positive homogeneous weight; filled polygons emit
33//!   genuine clip-space corners and keep near-plane straddling triangles for
34//!   the GPU to clip exactly.
35//! * **Line joins.** A polyline draws as butt-capped segment quads plus one
36//!   screen-space square (the line's width) per interior vertex, oriented
37//!   along the bend's bisector. The square fills the wedge the butt caps
38//!   leave outside a bend, which otherwise reads as a gap between styled road
39//!   segments; it never grows a miter spike at sharp angles. Endpoints keep
40//!   their butt caps (a per-feature end has no neighbouring segment to join),
41//!   and sharp joins can still show a sliver where the wedge exceeds the
42//!   square — a full round-join fan is the documented later refinement.
43//! * **Polygon holes.** A filled polygon carries one outer ring and any number
44//!   of hole rings. Holes join the outer boundary through the standard
45//!   bridge-splicing step before ear clipping ([`FilledPolygonOverlay`]
46//!   documents the algorithm), so the emitted triangle stream covers exactly
47//!   the ring set's even-odd interior: the overlay pipeline paints a flat
48//!   triangle list with alpha blending and has no stencil test, so a hole must
49//!   leave its region out of that list rather than erase it after the fact.
50//! * **Markers.** A marker is a regular-octagon triangle fan centered on the
51//!   projected anchor point (center anchoring, not bottom pinning): the
52//!   projected coordinate is the fan's center, so a marker's center sits
53//!   exactly at its geographic position. Sprites and label callouts are
54//!   explicitly not this slice.
55//! * **Pins.** A pin is a teardrop anchored the other way: its tip sits on the
56//!   projected coordinate and the circular head above it carries a hollow
57//!   center. The teardrop is star-shaped around the head center, so the
58//!   outline is one radial function of the angle and the hollow center is the
59//!   annulus between that outline and an inner circle — again a plain triangle
60//!   list, no erase pass.

213    /// Flattens a calculated route's per-step shape polylines into one

223            .collect();
224        Self::new(coordinates)
225    }

228/// A filled polygon: one outer ring plus any hole rings, bridge-joined and
229/// ear-clip triangulated in one pass and painted solid
230/// [`FilledPolygonOverlay::rgba_color`] (through the overlay pipeline's alpha
231/// blending, so translucent fills keep the base tiles readable).
232///
233/// Holes stay unfilled by construction: the assembly emits a flat,
234/// non-overlapping triangle list whose triangles cover only the outer ring's
235/// interior outside every hole. The overlay pipeline has no stencil test and
236/// no erase operation, so subtracting the holes from the emitted triangle
237/// stream is the only representation its solid-color fragment stage supports.
238/// The bridge splice (`ear_clip_triangulation_with_holes`) joins each hole
239/// to the outer boundary through a zero-width slit; the slit's duplicated
240/// vertices are collinear, never paint a fragment of their own, and keep the
241/// combined boundary a single simple ring the ear clipper can walk.
242#[derive(Debug, Clone, PartialEq)]
243pub struct FilledPolygonOverlay {
244    /// Outer ring in traversal order, without a repeated closing vertex (a
245    /// repeated one is tolerated and dropped). Self-intersecting or
246    /// near-collinear rings are degenerate and may fill nothing.
247    pub ring_coordinates: Vec<GeographicCoordinate>,
248    /// Hole rings subtracted from the fill, each in traversal order and
249    /// without a repeated closing vertex. The rings' winding is normalized
250    /// before triangulation, so decoded order does not matter. A hole that
251    /// falls outside the outer ring is not contained by it and may paint
252    /// outside the fill; producers emit contained holes.
253    pub hole_ring_coordinates: Vec<Vec<GeographicCoordinate>>,
254    /// Fill color; see [`DEFAULT_FILLED_POLYGON_RGBA`] for the default.
255    pub rgba_color: [f32; 4],
256}
257

259    /// Builds a filled polygon without holes and the documented default style.

421/// Assembles every overlay of a frame into one flat clip-space triangle list
422/// drawn over the tile quads in the same captured frame.
423///
424/// The shapes project through the same camera pose as the tile quads, wrap
425/// around the camera's world copy like the tile spans do (see the module
426/// notes), and leave frustum clipping to the GPU's clip-space clip. An empty
427/// `overlays` slice produces no vertices, and a shape whose geometry turns
428/// out degenerate (behind the camera for screen-thickened shapes, or
429/// numerically unusable anywhere) skips only that shape's own triangles.
430pub fn overlay_triangle_vertices(
431    camera: &MapCamera,
432    viewport: &ViewportPixels,
433    field_of_view_degrees: f64,
434    overlays: &[MapOverlay],
435) -> Vec<ClipSpaceOverlayVertex> {
436    let frame_projection =
437        OverlayFrameProjection::for_frame(camera, viewport, field_of_view_degrees);
438    let mut vertices = Vec::new();
439    for overlay in overlays {
440        match overlay {
441            MapOverlay::RoutePolyline(route_polyline) => {
442                thickened_polyline_triangles(
443                    &frame_projection,
444                    viewport,
445                    &route_polyline.coordinates,
446                    route_polyline.rgba_color,
447                    route_polyline.width_pixels,
448                    &mut vertices,
449                );
450            }
451            MapOverlay::FilledPolygon(filled_polygon) => {
452                filled_polygon_triangles(
453                    &frame_projection,
454                    &filled_polygon.ring_coordinates,
455                    &filled_polygon.hole_ring_coordinates,
456                    filled_polygon.rgba_color,
457                    &mut vertices,
458                );
459            }
460            MapOverlay::PointMarkers(point_markers) => {
461                for position in &point_markers.positions {
462                    point_marker_triangles(
463                        &frame_projection,
464                        viewport,
465                        position,
466                        point_markers.rgba_color,
467                        point_markers.radius_pixels,
468                        &mut vertices,
469                    );
470                }
471            }
472            MapOverlay::PointPins(point_pins) => {
473                for position in &point_pins.positions {
474                    point_pin_triangles(
475                        &frame_projection,
476                        viewport,
477                        position,
478                        point_pins.rgba_color,
479                        point_pins.radius_pixels,
480                        &mut vertices,
481                    );
482                }
483            }
484            MapOverlay::TrafficFlow(traffic_flow) => {
485                for traffic_flow_data in &traffic_flow.flows {
486                    thickened_polyline_triangles(
487                        &frame_projection,
488                        viewport,
489                        &traffic_flow_data.shape_coordinates,
490                        traffic_congestion_rgba(traffic_flow_data.congestion_level),
491                        traffic_flow.width_pixels,
492                        &mut vertices,
493                    );
494                }
495            }
496        }
497    }
498    vertices
499}

502/// world pixel scale, and the camera's normalized-world x, mirroring the tile
503/// renderer's frame projection (see `crate::renderer::FrameProjection`) so
504/// both layers resolve the same world-position convention per frame. Shared
505/// (crate-visible) with the location-indicator layer, which projects through
506/// the same conventions.
507pub(crate) struct OverlayFrameProjection {
508    view_projection_matrix: DMat4,
509    zoom_scale_pixels: f64,
510    center_normalized_x: f64,
511}

513impl OverlayFrameProjection {
514    /// Builds the projection context for one frame.
515    pub(crate) fn for_frame(
516        camera: &MapCamera,
517        viewport: &ViewportPixels,
518        field_of_view_degrees: f64,
519    ) -> Self {
520        Self {
521            view_projection_matrix: camera.view_projection_matrix(viewport, field_of_view_degrees),
522            zoom_scale_pixels: zoom_scale_pixels(camera.zoom_level),
523            center_normalized_x: coordinate_to_normalized_world(&camera.center)[0],
524        }
525    }
526
527    /// The wrapped normalized-world x of a raw normalized-world x: the
528    /// position in `(-0.5, 0.5]` around the camera's world copy, shared with
529    /// the tile spans' wrapping rule.
530    pub(crate) fn wrapped_x(&self, normalized_x: f64) -> f64 {
531        self.center_normalized_x + normalized_shortest_delta(self.center_normalized_x, normalized_x)
532    }
533
534    /// Wraps a shape's coordinates into one continuous normalized-world chain:
535    /// the first vertex sits in the camera's world copy and every following
536    /// vertex unwraps from its predecessor by the shortest wrapped delta, so
537    /// chains crossing the antimeridian continue past the copy seam instead
538    /// of tearing back across the viewport (the same wrap-as-one-unit rule
539    /// the tile spans follow).
540    fn wrapped_continuous_chain(&self, coordinates: &[GeographicCoordinate]) -> Vec<[f64; 2]> {
541        self.wrapped_continuous_chain_near(self.center_normalized_x, coordinates)
542    }
543
544    /// [`Self::wrapped_continuous_chain`] anchored on a caller-supplied
545    /// normalized-world x rather than the camera center. A hole ring uses its
546    /// outer ring's first wrapped vertex, so both rings land in the same world
547    /// copy even when the polygon straddles the antimeridian and the camera
548    /// center alone would resolve the hole to the neighboring copy.
549    fn wrapped_continuous_chain_near(
550        &self,
551        reference_x: f64,
552        coordinates: &[GeographicCoordinate],
553    ) -> Vec<[f64; 2]> {
554        let mut points = Vec::with_capacity(coordinates.len());
555        let mut previous_x: Option<f64> = None;
556        for coordinate in coordinates {
557            let normalized_world = coordinate_to_normalized_world(coordinate);
558            let wrapped_x = match previous_x {
559                None => reference_x + normalized_shortest_delta(reference_x, normalized_world[0]),
560                Some(previous_x_value) => {
561                    previous_x_value
562                        + normalized_shortest_delta(previous_x_value, normalized_world[0])
563                }
564            };
565            points.push([wrapped_x, normalized_world[1]]);
566            previous_x = Some(wrapped_x);
567        }
568        points
569    }
570
571    /// The camera-space clip position of a ground point of one world-copy
572    /// window. Camera world space is y-north pixel space (see
573    /// `MapCamera::ground_position_vector`); the window's copy offset is
574    /// subtracted from the wrapped x so the fragment continues the camera's
575    /// own copy past the window edge, exactly as the tile fragments do.
576    pub(crate) fn ground_clip_position(&self, window_offset: f64, x: f64, y: f64) -> DVec4 {
577        let x_pixels = (x - window_offset) * self.zoom_scale_pixels;
578        let y_pixels = -y * self.zoom_scale_pixels;
579        self.view_projection_matrix * DVec4::new(x_pixels, y_pixels, 0.0, 1.0)
580    }
581
582    /// The wrapped normalized-world x-range one world-copy window spans:
583    /// `[window_start_x, window_end_x)` around the camera's world copy,
584    /// shared by the segment fragments' split rule and the location
585    /// indicator's per-window anchor containment.
586    pub(crate) fn window_x_range(&self, window_offset: f64) -> [f64; 2] {
587        let window_start_x =
588            self.center_normalized_x - CAMERA_WORLD_WINDOW_HALF_WIDTH + window_offset;
589        [
590            window_start_x,
591            window_start_x + 2.0 * CAMERA_WORLD_WINDOW_HALF_WIDTH,
592        ]
593    }
594}

596/// Divides a clip position to viewport pixel coordinates (origin top-left, y
597/// down), matching [`MapCamera::project_to_screen`]. Sound only for a
598/// positive homogeneous weight; callers check that first where screen-space
599/// math follows.
600pub(crate) fn clip_position_to_screen_pixels(
601    clip_position: DVec4,
602    viewport: &ViewportPixels,
603) -> [f64; 2] {
604    let normalized_device_x = clip_position.x / clip_position.w;
605    let normalized_device_y = clip_position.y / clip_position.w;
606    [
607        (normalized_device_x + 1.0) * 0.5 * viewport.width_pixels,
608        (1.0 - (normalized_device_y + 1.0) * 0.5) * viewport.height_pixels,
609    ]
610}

612/// Re-expands a screen-pixel offset of one clip-space reference point into a
613/// clip position carrying the reference's depth and homogeneous weight. The
614/// screen offset is applied in normalized-device space around the reference's
615/// divided position, so per-vertex interpolation across a thickened quad ends
616/// at exactly the offset screen corners; the quad's interior follows the
617/// same perspective weighting as the underlying ground line.
618pub(crate) fn screen_pixels_to_clip_position(
619    screen_pixels: [f64; 2],
620    reference_clip_position: DVec4,
621    viewport: &ViewportPixels,
622) -> [f64; 4] {
623    let normalized_device_x = 2.0 * screen_pixels[0] / viewport.width_pixels - 1.0;
624    let normalized_device_y = 1.0 - 2.0 * screen_pixels[1] / viewport.height_pixels;
625    [
626        normalized_device_x * reference_clip_position.w,
627        normalized_device_y * reference_clip_position.w,
628        reference_clip_position.z,
629        reference_clip_position.w,
630    ]
631}

633/// Pushes one solid-color triangle, skipping it entirely when any corner is
634/// non-finite: an f32-overflowing corner would poison the whole draw with
635/// undefined geometry, while skipping keeps the rest of the overlay healthy.
636pub(crate) fn push_triangle(
637    vertices: &mut Vec<ClipSpaceOverlayVertex>,
638    corners: &[[f64; 4]; 3],
639    rgba_color: [f32; 4],
640) {
641    if corners
642        .iter()
643        .any(|corner| corner.iter().any(|component| !component.is_finite()))
644    {
645        return;
646    }
647    for corner in corners {
648        vertices.push(ClipSpaceOverlayVertex {
649            clip_position: [
650                corner[0] as f32,
651                corner[1] as f32,
652                corner[2] as f32,
653                corner[3] as f32,
654            ],
655            rgba_color,
656        });
657    }
658}

660/// One piece of a shape segment after splitting against a world-copy window,
661/// with its continuous normalized-world endpoints and the window's copy
662/// offset for [`OverlayFrameProjection::ground_clip_position`].
663#[derive(Debug, Clone, Copy, PartialEq)]
664struct SegmentWindowFragment {
665    /// Continuous normalized-world start point of the fragment.
666    start: [f64; 2],
667    /// Continuous normalized-world end point of the fragment.
668    end: [f64; 2],
669    /// Whole-world copy offset of the window the fragment lies in.
670    window_offset: f64,
671}

673/// Splits a shape segment against the camera-centered world-copy windows.
674///
675/// Each window keeps the segment's intersection with its x-range, matching
676/// [`CAMERA_WORLD_WINDOW_HALF_WIDTH`] around the camera per offset. A
677/// vertical segment (zero x-span) joins the one window containing its x,
678/// half-open on the east so a boundary-exact segment cannot draw twice; a
679/// segment touching a window only at its edge produces no fragment, mirroring
680/// the tile spans' empty-fragment rule.
681fn segment_window_fragments(
682    frame_projection: &OverlayFrameProjection,
683    segment_start: [f64; 2],
684    segment_end: [f64; 2],
685) -> Vec<SegmentWindowFragment> {
686    let span_x = segment_end[0] - segment_start[0];
687    let mut fragments = Vec::new();
688    for window_offset in WORLD_COPY_WINDOW_OFFSETS {
689        let window_offset_f64 = f64::from(window_offset);
690        let [window_start_x, window_end_x] = frame_projection.window_x_range(window_offset_f64);
691        let (start_fraction, end_fraction) = if span_x == 0.0 {
692            let zero_span_inside_window =
693                segment_start[0] >= window_start_x && segment_start[0] < window_end_x;
694            if !zero_span_inside_window {
695                continue;
696            }
697            (0.0, 1.0)
698        } else {
699            let clamped_start_fraction =
700                ((window_start_x - segment_start[0]) / span_x).clamp(0.0, 1.0);
701            let clamped_end_fraction = ((window_end_x - segment_start[0]) / span_x).clamp(0.0, 1.0);
702            if clamped_start_fraction == clamped_end_fraction {
703                continue;
704            }
705            (clamped_start_fraction, clamped_end_fraction)
706        };
707        let fragment_start = [
708            segment_start[0] + span_x * start_fraction,
709            segment_start[1] + (segment_end[1] - segment_start[1]) * start_fraction,
710        ];
711        let fragment_end = [
712            segment_start[0] + span_x * end_fraction,
713            segment_start[1] + (segment_end[1] - segment_start[1]) * end_fraction,
714        ];
715        fragments.push(SegmentWindowFragment {
716            start: fragment_start,
717            end: fragment_end,
718            window_offset: window_offset_f64,
719        });
720    }
721    fragments
722}

724/// Assembles a thick polyline: consecutive coordinate pairs thicken to
725/// per-segment screen-space quads split against the world-copy windows, and
726/// every interior vertex fills its joint with a screen-space square of the
727/// line's width.
728///
729/// The join fill is what keeps a multi-segment line continuous: butt-capped
730/// quads meet at the shared vertex but leave a wedge on a bend's outer side,
731/// which reads as a gap between segments — the most visible case is a styled
732/// road network, whose per-way geometry bends at nearly every vertex.
733/// Endpoints keep their butt caps (a per-feature end has no neighbouring
734/// segment to join), and the square join never grows a miter spike at sharp
735/// angles the way a mitered join would.
736fn thickened_polyline_triangles(
737    frame_projection: &OverlayFrameProjection,
738    viewport: &ViewportPixels,
739    coordinates: &[GeographicCoordinate],
740    rgba_color: [f32; 4],
741    width_pixels: f64,
742    vertices: &mut Vec<ClipSpaceOverlayVertex>,
743) {
744    let wrapped_points = frame_projection.wrapped_continuous_chain(coordinates);
745    for segment_index in 1..wrapped_points.len() {
746        for fragment in segment_window_fragments(
747            frame_projection,
748            wrapped_points[segment_index - 1],
749            wrapped_points[segment_index],
750        ) {
751            thickened_segment_triangles(
752                frame_projection,
753                viewport,
754                &fragment,
755                rgba_color,
756                width_pixels,
757                vertices,
758            );
759        }
760    }
761    for vertex_index in 1..wrapped_points.len().saturating_sub(1) {
762        polyline_join_triangles(
763            frame_projection,
764            viewport,
765            PolylineBend {
766                previous: wrapped_points[vertex_index - 1],
767                vertex: wrapped_points[vertex_index],
768                next: wrapped_points[vertex_index + 1],
769            },
770            rgba_color,
771            width_pixels,
772            vertices,
773        );
774    }
775}

777/// One interior polyline vertex and its two neighbours, in world-copy ground
778/// coordinates; the join pass fills the wedge the segment quads leave there.
779struct PolylineBend {
780    previous: [f64; 2],
781    vertex: [f64; 2],
782    next: [f64; 2],
783}

785/// Fills one interior polyline vertex's joint with a screen-space square of
786/// the line's own width, oriented along the bend's bisector. The square
787/// covers the wedge the two butt-capped segment quads leave outside a bend;
788/// see [`thickened_polyline_triangles`] for why the join exists.
789fn polyline_join_triangles(
790    frame_projection: &OverlayFrameProjection,
791    viewport: &ViewportPixels,
792    bend: PolylineBend,
793    rgba_color: [f32; 4],
794    width_pixels: f64,
795    vertices: &mut Vec<ClipSpaceOverlayVertex>,
796) {
797    // One join lives in the world-copy window that contains it; a vertex on a
798    // window's east half-open edge belongs to the following copy, matching
799    // the segment split's rule.
800    let window_offset = WORLD_COPY_WINDOW_OFFSETS.iter().copied().find(|offset| {
801        let [window_start_x, window_end_x] = frame_projection.window_x_range(f64::from(*offset));
802        bend.vertex[0] >= window_start_x && bend.vertex[0] < window_end_x
803    });
804    let Some(window_offset) = window_offset else {
805        return;
806    };
807    let window_offset_f64 = f64::from(window_offset);
808    let project = |point: [f64; 2]| {
809        frame_projection.ground_clip_position(window_offset_f64, point[0], point[1])
810    };
811    let previous_clip_position = project(bend.previous);
812    let vertex_clip_position = project(bend.vertex);
813    let next_clip_position = project(bend.next);
814    if previous_clip_position.w <= 0.0
815        || vertex_clip_position.w <= 0.0
816        || next_clip_position.w <= 0.0
817    {
818        return;
819    }
820    let previous_screen = clip_position_to_screen_pixels(previous_clip_position, viewport);
821    let vertex_screen = clip_position_to_screen_pixels(vertex_clip_position, viewport);
822    let next_screen = clip_position_to_screen_pixels(next_clip_position, viewport);
823    let incoming = [
824        vertex_screen[0] - previous_screen[0],
825        vertex_screen[1] - previous_screen[1],
826    ];
827    let outgoing = [
828        next_screen[0] - vertex_screen[0],
829        next_screen[1] - vertex_screen[1],
830    ];
831    let incoming_length = incoming[0].hypot(incoming[1]);
832    let outgoing_length = outgoing[0].hypot(outgoing[1]);
833    if incoming_length < MINIMUM_VISIBLE_SEGMENT_LENGTH_PIXELS
834        || outgoing_length < MINIMUM_VISIBLE_SEGMENT_LENGTH_PIXELS
835    {
836        return;
837    }
838    let incoming_unit = [incoming[0] / incoming_length, incoming[1] / incoming_length];
839    let outgoing_unit = [outgoing[0] / outgoing_length, outgoing[1] / outgoing_length];
840    let half_width_pixels = width_pixels * 0.5;
841    // The bisector of the two segment directions is the direction the wedge
842    // opens along. A reversal (sum near zero) has no bisector: the incoming
843    // perpendicular is its 90-degree stand-in.
844    let sum = [
845        incoming_unit[0] + outgoing_unit[0],
846        incoming_unit[1] + outgoing_unit[1],
847    ];
848    let sum_length = sum[0].hypot(sum[1]);
849    let bisector_unit = if sum_length < MINIMUM_VISIBLE_SEGMENT_LENGTH_PIXELS {
850        [-incoming_unit[1], incoming_unit[0]]
851    } else {
852        [sum[0] / sum_length, sum[1] / sum_length]
853    };
854    let perpendicular_unit = [-bisector_unit[1], bisector_unit[0]];
855    let corner_screens = [
856        [
857            vertex_screen[0] + (bisector_unit[0] + perpendicular_unit[0]) * half_width_pixels,
858            vertex_screen[1] + (bisector_unit[1] + perpendicular_unit[1]) * half_width_pixels,
859        ],
860        [
861            vertex_screen[0] + (-bisector_unit[0] + perpendicular_unit[0]) * half_width_pixels,
862            vertex_screen[1] + (-bisector_unit[1] + perpendicular_unit[1]) * half_width_pixels,
863        ],
864        [
865            vertex_screen[0] + (-bisector_unit[0] - perpendicular_unit[0]) * half_width_pixels,
866            vertex_screen[1] + (-bisector_unit[1] - perpendicular_unit[1]) * half_width_pixels,
867        ],
868        [
869            vertex_screen[0] + (bisector_unit[0] - perpendicular_unit[0]) * half_width_pixels,
870            vertex_screen[1] + (bisector_unit[1] - perpendicular_unit[1]) * half_width_pixels,
871        ],
872    ];
873    let corner_clip_positions = corner_screens.map(|corner_screen_pixels| {
874        screen_pixels_to_clip_position(corner_screen_pixels, vertex_clip_position, viewport)
875    });
876    push_triangle(
877        vertices,
878        &[
879            corner_clip_positions[0],
880            corner_clip_positions[1],
881            corner_clip_positions[2],
882        ],
883        rgba_color,
884    );
885    push_triangle(
886        vertices,
887        &[
888            corner_clip_positions[0],
889            corner_clip_positions[2],
890            corner_clip_positions[3],
891        ],
892        rgba_color,
893    );
894}

896/// Assembles one thickened segment fragment as a screen-space quad: the
897/// perpendicular offset of half the line width, applied in screen pixels
898/// around each endpoint's divided position, so a line keeps its pixel width
899/// regardless of zoom and pitch.
900///
901/// A fragment with a vertex at or behind the camera draws nothing — the
902/// screen-space offset needs a positive homogeneous weight — which is the
903/// documented limitation next to the filled polygons' GPU-clipped near-plane
904/// straddling (see the module notes).
905fn thickened_segment_triangles(
906    frame_projection: &OverlayFrameProjection,
907    viewport: &ViewportPixels,
908    fragment: &SegmentWindowFragment,
909    rgba_color: [f32; 4],
910    width_pixels: f64,
911    vertices: &mut Vec<ClipSpaceOverlayVertex>,
912) {
913    let start_clip_position = frame_projection.ground_clip_position(
914        fragment.window_offset,
915        fragment.start[0],
916        fragment.start[1],
917    );
918    let end_clip_position = frame_projection.ground_clip_position(
919        fragment.window_offset,
920        fragment.end[0],
921        fragment.end[1],
922    );
923    if start_clip_position.w <= 0.0 || end_clip_position.w <= 0.0 {
924        return;
925    }
926    let start_screen = clip_position_to_screen_pixels(start_clip_position, viewport);
927    let end_screen = clip_position_to_screen_pixels(end_clip_position, viewport);
928    let direction = [
929        end_screen[0] - start_screen[0],
930        end_screen[1] - start_screen[1],
931    ];
932    let screen_length = direction[0].hypot(direction[1]);
933    if screen_length < MINIMUM_VISIBLE_SEGMENT_LENGTH_PIXELS {
934        return;
935    }
936    let perpendicular_unit = [-direction[1] / screen_length, direction[0] / screen_length];
937    let half_width_pixels = width_pixels * 0.5;
938    let start_offset = [
939        start_screen[0] + perpendicular_unit[0] * half_width_pixels,
940        start_screen[1] + perpendicular_unit[1] * half_width_pixels,
941    ];
942    let start_opposite = [
943        start_screen[0] - perpendicular_unit[0] * half_width_pixels,
944        start_screen[1] - perpendicular_unit[1] * half_width_pixels,
945    ];
946    let end_offset = [
947        end_screen[0] + perpendicular_unit[0] * half_width_pixels,
948        end_screen[1] + perpendicular_unit[1] * half_width_pixels,
949    ];
950    let end_opposite = [
951        end_screen[0] - perpendicular_unit[0] * half_width_pixels,
952        end_screen[1] - perpendicular_unit[1] * half_width_pixels,
953    ];
954    // Corner order mirrors the tile quads: start-offset, start-opposite,
955    // end-offset, end-opposite, split by the quads' `0,1,2 / 2,1,3` triangle
956    // pattern, each corner re-expanded around its own endpoint's depth.
957    let corner_clip_positions = [
958        (start_offset, start_clip_position),
959        (start_opposite, start_clip_position),
960        (end_offset, end_clip_position),
961        (end_opposite, end_clip_position),
962    ]
963    .map(|(corner_screen_pixels, reference_clip_position)| {
964        screen_pixels_to_clip_position(corner_screen_pixels, reference_clip_position, viewport)
965    });
966    push_triangle(
967        vertices,
968        &[
969            corner_clip_positions[0],
970            corner_clip_positions[1],
971            corner_clip_positions[2],
972        ],
973        rgba_color,
974    );
975    push_triangle(
976        vertices,
977        &[
978            corner_clip_positions[2],
979            corner_clip_positions[1],
980            corner_clip_positions[3],
981        ],
982        rgba_color,
983    );
984}

986/// Assembles one point marker: a regular-octagon triangle fan centered on the
987/// projected anchor (center anchoring; see the module notes), built around a
988/// single clip-space reference so the whole fan stays screen-facing.
989fn point_marker_triangles(
990    frame_projection: &OverlayFrameProjection,
991    viewport: &ViewportPixels,
992    position: &GeographicCoordinate,
993    rgba_color: [f32; 4],
994    radius_pixels: f64,
995    vertices: &mut Vec<ClipSpaceOverlayVertex>,
996) {
997    let normalized_world = coordinate_to_normalized_world(position);
998    let center_clip_position = frame_projection.ground_clip_position(
999        0.0,
1000        frame_projection.wrapped_x(normalized_world[0]),
1001        normalized_world[1],
1002    );
1003    if center_clip_position.w <= 0.0 {
1004        return;
1005    }
1006    let center_screen = clip_position_to_screen_pixels(center_clip_position, viewport);
1007    let center_corner =
1008        screen_pixels_to_clip_position(center_screen, center_clip_position, viewport);
1009    for side_index in 0..POINT_MARKER_FAN_SIDES {
1010        let start_angle = side_index as f64 * TAU / POINT_MARKER_FAN_SIDES as f64;
1011        let end_angle = (side_index + 1) as f64 * TAU / POINT_MARKER_FAN_SIDES as f64;
1012        let ring_corner_start = screen_pixels_to_clip_position(
1013            [
1014                center_screen[0] + radius_pixels * start_angle.cos(),
1015                center_screen[1] + radius_pixels * start_angle.sin(),
1016            ],
1017            center_clip_position,
1018            viewport,
1019        );
1020        let ring_corner_end = screen_pixels_to_clip_position(
1021            [
1022                center_screen[0] + radius_pixels * end_angle.cos(),
1023                center_screen[1] + radius_pixels * end_angle.sin(),
1024            ],
1025            center_clip_position,
1026            viewport,
1027        );
1028        push_triangle(
1029            vertices,
1030            &[center_corner, ring_corner_start, ring_corner_end],
1031            rgba_color,
1032        );
1033    }
1034}

1036/// Assembles one point pin: a teardrop outline anchored with its tip on the
1037/// projected coordinate and a hollow circular center in the head, emitted as
1038/// the annulus between the two outlines.
1039///
1040/// The teardrop is star-shaped around the head center — every ray from the
1041/// center crosses the outline exactly once — so both boundaries are radial
1042/// functions of the same angle and pair up into quads without a bridge. Inside
1043/// the tangent sector the outline is the straight edge running to the tip
1044/// (radius `R / cos(angle - tangentAngle)`); outside it is the head circle
1045/// itself. The hole therefore needs no erase pass, which this pipeline cannot
1046/// perform (see the module notes on holes).
1047fn point_pin_triangles(
1048    frame_projection: &OverlayFrameProjection,
1049    viewport: &ViewportPixels,
1050    position: &GeographicCoordinate,
1051    rgba_color: [f32; 4],
1052    radius_pixels: f64,
1053    vertices: &mut Vec<ClipSpaceOverlayVertex>,
1054) {
1055    let normalized_world = coordinate_to_normalized_world(position);
1056    let tip_clip_position = frame_projection.ground_clip_position(
1057        0.0,
1058        frame_projection.wrapped_x(normalized_world[0]),
1059        normalized_world[1],
1060    );
1061    if tip_clip_position.w <= 0.0 || !radius_pixels.is_finite() || radius_pixels <= 0.0 {
1062        return;
1063    }
1064    let tip_screen = clip_position_to_screen_pixels(tip_clip_position, viewport);
1065    let pin_height_pixels = radius_pixels * POINT_PIN_HEIGHT_RADIUS_FACTOR;
1066    let head_center_screen = [tip_screen[0], tip_screen[1] - pin_height_pixels];
1067    let hollow_radius_pixels = radius_pixels * POINT_PIN_HOLLOW_RADIUS_RATIO;
1068    // The head's tangent points sit this far around the head from the tip
1069    // direction (screen angles grow from +x towards +y, i.e. clockwise in the
1070    // displayed image, so the tip direction is a quarter turn).
1071    let tangent_half_angle = (radius_pixels / pin_height_pixels).acos();
1072    let tip_direction_angle = std::f64::consts::FRAC_PI_2;
1073    let right_tangent_angle = tip_direction_angle - tangent_half_angle;
1074    let left_tangent_angle = tip_direction_angle + tangent_half_angle;
1075    let outline_radius_pixels = |angle: f64| {
1076        if angle > right_tangent_angle && angle < left_tangent_angle {
1077            let edge_angle = if angle <= tip_direction_angle {
1078                right_tangent_angle
1079            } else {
1080                left_tangent_angle
1081            };
1082            radius_pixels / (angle - edge_angle).cos()
1083        } else {
1084            radius_pixels
1085        }
1086    };
1087    // Both outlines share the tip's clip-space reference, so the whole pin
1088    // stays screen-facing at the anchor's depth like a marker fan does.
1089    let corner_clip_position = |radius: f64, angle: f64| {
1090        screen_pixels_to_clip_position(
1091            [
1092                head_center_screen[0] + radius * angle.cos(),
1093                head_center_screen[1] + radius * angle.sin(),
1094            ],
1095            tip_clip_position,
1096            viewport,
1097        )
1098    };
1099    for side_index in 0..POINT_PIN_RING_SIDES {
1100        let start_angle = side_index as f64 * TAU / POINT_PIN_RING_SIDES as f64;
1101        let end_angle = (side_index + 1) as f64 * TAU / POINT_PIN_RING_SIDES as f64;
1102        let inner_start = corner_clip_position(hollow_radius_pixels, start_angle);
1103        let inner_end = corner_clip_position(hollow_radius_pixels, end_angle);
1104        let outer_start = corner_clip_position(outline_radius_pixels(start_angle), start_angle);
1105        let outer_end = corner_clip_position(outline_radius_pixels(end_angle), end_angle);
1106        push_triangle(vertices, &[inner_start, inner_end, outer_end], rgba_color);
1107        push_triangle(vertices, &[inner_start, outer_end, outer_start], rgba_color);
1108    }
1109}

1111/// Assembles a filled polygon: the ring set (outer boundary plus holes)
1112/// triangulates in continuous normalized-world space, every triangle splits
1113/// against the world-copy windows (convex clipping keeps the pieces convex),
1114/// and each clipped piece fans into triangles projected by their genuine
1115/// clip-space corners. Hole rings anchor to the outer ring's world copy (see
1116/// [`OverlayFrameProjection::wrapped_continuous_chain_near`]) so a polygon
1117/// straddling the antimeridian keeps its holes with its exterior. Shared
1118/// (crate-visible) with the location-indicator layer, whose ground-projected
1119/// accuracy ring reuses the exact fill machinery (without holes).
1120pub(crate) fn filled_polygon_triangles(
1121    frame_projection: &OverlayFrameProjection,
1122    ring_coordinates: &[GeographicCoordinate],
1123    hole_ring_coordinates: &[Vec<GeographicCoordinate>],
1124    rgba_color: [f32; 4],
1125    vertices: &mut Vec<ClipSpaceOverlayVertex>,
1126) {
1127    let wrapped_ring = frame_projection.wrapped_continuous_chain(ring_coordinates);
1128    let Some(reference_x) = wrapped_ring.first().map(|vertex| vertex[0]) else {
1129        return;
1130    };
1131    let wrapped_holes: Vec<Vec<[f64; 2]>> = hole_ring_coordinates
1132        .iter()
1133        .map(|hole_ring| frame_projection.wrapped_continuous_chain_near(reference_x, hole_ring))
1134        .collect();
1135    let triangle_minimum_x =
1136        |triangle: &[[f64; 2]; 3]| triangle[0][0].min(triangle[1][0]).min(triangle[2][0]);
1137    let triangle_maximum_x =
1138        |triangle: &[[f64; 2]; 3]| triangle[0][0].max(triangle[1][0]).max(triangle[2][0]);
1139    for triangle in ear_clip_triangulation_with_holes(&wrapped_ring, &wrapped_holes) {
1140        let (triangle_minimum_x, triangle_maximum_x) =
1141            (triangle_minimum_x(&triangle), triangle_maximum_x(&triangle));
1142        for window_offset in WORLD_COPY_WINDOW_OFFSETS {
1143            let window_offset_f64 = f64::from(window_offset);
1144            let [window_start_x, window_end_x] = frame_projection.window_x_range(window_offset_f64);
1145            // A triangle lying entirely outside one of the window's two
1146            // half-planes clips down to the boundary line itself — fewer than
1147            // three corners, and the fan below would draw nothing. Rejecting
1148            // it by its x-extent is exact and saves the whole clip.
1149            if triangle_maximum_x <= window_start_x || triangle_minimum_x >= window_end_x {
1150                continue;
1151            }
1152            // A triangle wholly inside both half-planes passes through the
1153            // clip unchanged — the overwhelmingly common case, since the
1154            // windows are a third of the world apart — so it projects its own
1155            // three corners and skips two Sutherland–Hodgman passes.
1156            let clipped_ring =
1157                if triangle_minimum_x >= window_start_x && triangle_maximum_x <= window_end_x {
1158                    WindowClippedTriangle {
1159                        corners: [triangle[0], triangle[1], triangle[2], [0.0; 2], [0.0; 2]],
1160                        corner_count: 3,
1161                    }
1162                } else {
1163                    clip_triangle_to_x_range(&triangle, window_start_x, window_end_x)
1164                };
1165            if clipped_ring.corner_count < 3 {
1166                continue;
1167            }
1168            // Clipped corners project by their genuine clip-space positions,
1169            // so a triangle straddling the near plane stays for the GPU to
1170            // clip exactly (see the module notes); only an f32-overflowing
1171            // corner drops via `push_triangle`.
1172            for fan_index in 1..clipped_ring.corner_count - 1 {
1173                let corner_clip_positions = [
1174                    clipped_ring.corners[0],
1175                    clipped_ring.corners[fan_index],
1176                    clipped_ring.corners[fan_index + 1],
1177                ]
1178                .map(|corner| {
1179                    let corner_clip_position = frame_projection.ground_clip_position(
1180                        window_offset_f64,
1181                        corner[0],
1182                        corner[1],
1183                    );
1184                    [
1185                        corner_clip_position.x,
1186                        corner_clip_position.y,
1187                        corner_clip_position.z,
1188                        corner_clip_position.w,
1189                    ]
1190                });
1191                push_triangle(vertices, &corner_clip_positions, rgba_color);
1192            }
1193        }
1194    }
1195}

1197/// A triangle clipped against one world-copy window: a convex ring of at most
1198/// [`MAXIMUM_WINDOW_CLIPPED_CORNERS`] corners.
1199///
1200/// The corners live in a fixed array because this runs once per triangle per
1201/// window on every frame — hundreds of thousands of times during a pan — and
1202/// the heap allocation a `Vec` would make per call is a measurable share of
1203/// the frame (PLAN.md, M23). The arithmetic is the same Sutherland–Hodgman the
1204/// allocation-based clipper uses, in the same order, so the drawn vertices are
1205/// unchanged.
1206struct WindowClippedTriangle {
1207    corners: [[f64; 2]; MAXIMUM_WINDOW_CLIPPED_CORNERS],
1208    corner_count: usize,
1209}

1211/// Clips one triangle to `window_start_x <= x < window_end_x` without
1212/// allocating: one Sutherland–Hodgman pass per half-plane over fixed buffers.
1213fn clip_triangle_to_x_range(
1214    triangle: &[[f64; 2]; 3],
1215    window_start_x: f64,
1216    window_end_x: f64,
1217) -> WindowClippedTriangle {
1218    let mut first_clip = [[0.0_f64; 2]; MAXIMUM_WINDOW_CLIPPED_CORNERS];
1219    let first_clip_count =
1220        clip_ring_to_x_half_plane(triangle, true, window_start_x, &mut first_clip);
1221    let mut second_clip = [[0.0_f64; 2]; MAXIMUM_WINDOW_CLIPPED_CORNERS];
1222    let second_clip_count = clip_ring_to_x_half_plane(
1223        &first_clip[..first_clip_count],
1224        false,
1225        window_end_x,
1226        &mut second_clip,
1227    );
1228    WindowClippedTriangle {
1229        corners: second_clip,
1230        corner_count: second_clip_count,
1231    }
1232}

1234/// One Sutherland–Hodgman pass of a convex ring against a vertical half-plane
1235/// (`keep_greater` selects `x >= boundary_x`), writing into a fixed buffer.
1236/// [`clip_polygon_to_x_half_plane`] is the allocating form with the same
1237/// arithmetic; this one is the frame-path copy.
1238fn clip_ring_to_x_half_plane(
1239    polygon: &[[f64; 2]],
1240    keep_greater: bool,
1241    boundary_x: f64,
1242    clipped_polygon: &mut [[f64; 2]; MAXIMUM_WINDOW_CLIPPED_CORNERS],
1243) -> usize {
1244    let mut clipped_corner_count = 0;
1245    for edge_index in 0..polygon.len() {
1246        let edge_start = polygon[edge_index];
1247        let edge_end = polygon[(edge_index + 1) % polygon.len()];
1248        let start_inside = if keep_greater {
1249            edge_start[0] >= boundary_x
1250        } else {
1251            edge_start[0] <= boundary_x
1252        };
1253        let end_inside = if keep_greater {
1254            edge_end[0] >= boundary_x
1255        } else {
1256            edge_end[0] <= boundary_x
1257        };
1258        if start_inside && clipped_corner_count < MAXIMUM_WINDOW_CLIPPED_CORNERS {
1259            clipped_polygon[clipped_corner_count] = edge_start;
1260            clipped_corner_count += 1;
1261        }
1262        if start_inside != end_inside {
1263            let crossing_fraction = (boundary_x - edge_start[0]) / (edge_end[0] - edge_start[0]);
1264            // A crossing at the edge's own endpoint is a vertex on the
1265            // boundary; it already landed through the inside-start pushes of
1266            // the neighboring edges, and pushing it again would degenerate
1267            // the clipped ring into a zero-area fan triangle.
1268            if crossing_fraction > 0.0
1269                && crossing_fraction < 1.0
1270                && clipped_corner_count < MAXIMUM_WINDOW_CLIPPED_CORNERS
1271            {
1272                clipped_polygon[clipped_corner_count] = [
1273                    boundary_x,
1274                    edge_start[1] + (edge_end[1] - edge_start[1]) * crossing_fraction,
1275                ];
1276                clipped_corner_count += 1;
1277            }
1278        }
1279    }
1280    clipped_corner_count
1281}

1283/// Doubled signed area of a ring (the fan-formula sum, without the constant
1284/// factor): positive for one winding, negative for the other, zero for a
1285/// collinear ring.
1286fn ring_doubled_signed_area(ring: &[[f64; 2]]) -> f64 {
1287    ring.iter()
1288        .enumerate()
1289        .map(|(index, vertex)| {
1290            let next = ring[(index + 1) % ring.len()];
1291            vertex[0] * next[1] - next[0] * vertex[1]
1292        })
1293        .sum()
1294}

1296/// Twice the signed area of one triangle (a × b cross-product form): positive
1297/// for counterclockwise winding, negative for clockwise.
1298fn triangle_doubled_signed_area(
1299    first_vertex: [f64; 2],
1300    second_vertex: [f64; 2],
1301    third_vertex: [f64; 2],
1302) -> f64 {
1303    (second_vertex[0] - first_vertex[0]) * (third_vertex[1] - first_vertex[1])
1304        - (second_vertex[1] - first_vertex[1]) * (third_vertex[0] - first_vertex[0])
1305}

1307/// Whether a point lies inside a triangle or on its boundary, by winding-sign
1308/// tests against the triangle's (consistently wound) edges.
1309fn point_lies_within_triangle(
1310    point: [f64; 2],
1311    first_vertex: [f64; 2],
1312    second_vertex: [f64; 2],
1313    third_vertex: [f64; 2],
1314) -> bool {
1315    triangle_doubled_signed_area(point, first_vertex, second_vertex)
1316        >= -EAR_CLIPPING_BOUNDARY_TOLERANCE
1317        && triangle_doubled_signed_area(point, second_vertex, third_vertex)
1318            >= -EAR_CLIPPING_BOUNDARY_TOLERANCE
1319        && triangle_doubled_signed_area(point, third_vertex, first_vertex)
1320            >= -EAR_CLIPPING_BOUNDARY_TOLERANCE
1321}

1323/// Triangulates a simple polygon ring by ear clipping, over the ring's own
1324/// vertices in O(n²) — no external geometry dependency, consistent with the
1325/// crate owning its minimal math.
1326///
1327/// The ring normalizes to one winding (positive doubled signed area) first,
1328/// so input order does not matter. Degenerate rings (fewer than three
1329/// vertices, near-zero area, or a remainder with no clipgable ear — fully
1330/// collinear chains) contribute fewer triangles or none; a numerically stuck
1331/// remainder stops the loop instead of spinning, matching the documented
1332/// "degenerate rings may fill nothing" contract.
1333fn ear_clip_triangulation(ring: &[[f64; 2]]) -> Vec<[[f64; 2]; 3]> {
1334    let mut normalized_ring = ring.to_vec();
1335    if normalized_ring.len() >= 2 && normalized_ring.first() == normalized_ring.last() {
1336        normalized_ring.pop();
1337    }
1338    if normalized_ring.len() < 3 {
1339        return Vec::new();
1340    }
1341    let doubled_signed_area = ring_doubled_signed_area(&normalized_ring);
1342    if doubled_signed_area.abs() < POLYGON_AREA_DEGENERACY_TOLERANCE {
1343        return Vec::new();
1344    }
1345    if doubled_signed_area < 0.0 {
1346        normalized_ring.reverse();
1347    }
1348    let mut remaining_indices: Vec<usize> = (0..normalized_ring.len()).collect();
1349    let mut triangles = Vec::with_capacity(normalized_ring.len() - 2);
1350    while remaining_indices.len() > 3 {
1351        let ear_position = (0..remaining_indices.len())
1352            .find(|&position| is_ring_ear(&normalized_ring, &remaining_indices, position));
1353        let Some(ear_position) = ear_position else {
1354            break;
1355        };
1356        let corner_indices = ear_corner_indices(&remaining_indices, ear_position);
1357        triangles.push([
1358            normalized_ring[corner_indices[0]],
1359            normalized_ring[corner_indices[1]],
1360            normalized_ring[corner_indices[2]],
1361        ]);
1362        remaining_indices.remove(ear_position);
1363    }
1364    if remaining_indices.len() == 3 {
1365        triangles.push([
1366            normalized_ring[remaining_indices[0]],
1367            normalized_ring[remaining_indices[1]],
1368            normalized_ring[remaining_indices[2]],
1369        ]);
1370    }
1371    triangles
1372}

1374/// Triangulates a ring set with holes: the outer boundary plus every hole,
1375/// bridge-joined into one simple ring and ear-clip triangulated in one pass.
1376///
1377/// The hole handling follows the standard bridge-splicing construction (the
1378/// `earcut` algorithm's `eliminateHoles` step) with the crate's own vector
1379/// math: each hole joins the boundary through a zero-width slit, so the
1380/// emitted triangles cover the ring set's interior minus the holes and never
1381/// the holes themselves. Winding normalizes first (outer positive, holes
1382/// negative), so the decoded ring order does not matter.
1383///
1384/// A degenerate outer ring fills nothing, and a hole ring that cannot be
1385/// bridged (fewer than three vertices, zero area, or no visible bridge) is
1386/// skipped rather than approximated; a numerically stuck ear-clipping
1387/// remainder stops early and leaves that remainder unfilled, matching the
1388/// documented "degenerate rings may fill nothing" contract.
1389fn ear_clip_triangulation_with_holes(
1390    outer_ring: &[[f64; 2]],
1391    hole_rings: &[Vec<[f64; 2]>],
1392) -> Vec<[[f64; 2]; 3]> {
1393    if hole_rings.is_empty() {
1394        return ear_clip_triangulation(outer_ring);
1395    }
1396    let normalized_outer = normalized_ring(outer_ring, true);
1397    if normalized_outer.len() < 3 {
1398        return Vec::new();
1399    }
1400    let mut nodes = Vec::new();
1401    let Some(ring_head) = ring_nodes(&normalized_outer, &mut nodes) else {
1402        return Vec::new();
1403    };
1404    let mut holes: Vec<Vec<[f64; 2]>> = hole_rings
1405        .iter()
1406        .map(|hole_ring| normalized_ring(hole_ring, false))
1407        .filter(|hole_ring| hole_ring.len() >= 3)
1408        .collect();
1409    // Left-to-right hole order, matching the bridge search's proven order:
1410    // each hole's westward ray then meets only edges already part of the
1411    // combined boundary.
1412    holes.sort_by(|left, right| {
1413        ring_leftmost_point(left)[0]
1414            .partial_cmp(&ring_leftmost_point(right)[0])
1415            .unwrap_or(std::cmp::Ordering::Equal)
1416    });
1417    for hole_ring in &holes {
1418        let Some(hole_head) = ring_nodes(hole_ring, &mut nodes) else {
1419            continue;
1420        };
1421        // The bridge search validates a bridge from the hole's leftmost
1422        // vertex; the splice must start the hole at that same vertex, or the
1423        // cycle would gain an unvalidated edge across the hole.
1424        let hole_leftmost = hole_leftmost_node(hole_head, &nodes);
1425        let Some(bridge_index) = find_hole_bridge(hole_leftmost, ring_head, &nodes) else {
1426            continue;
1427        };
1428        splice_hole_ring(bridge_index, hole_leftmost, &mut nodes);
1429    }
1430    ear_clip_nodes(&mut nodes, ring_head)
1431}

1433/// One node of the mutable circular ring list the hole-aware triangulator
1434/// clips ears from. Nodes are never freed: removal relinks the neighbors and
1435/// leaves the node unreachable, so arena indices stay valid across removals.
1436struct EarClipNode {
1437    /// The node's position in normalized-world space.
1438    vertex: [f64; 2],
1439    /// Arena index of the previous ring node.
1440    previous_index: usize,
1441    /// Arena index of the next ring node.
1442    next_index: usize,
1443}

1445/// Builds a circular node ring from a ring's vertices, dropping the repeated
1446/// closing vertex and consecutive duplicates. Returns the first node's arena
1447/// index, or `None` when fewer than three distinct vertices remain.
1448fn ring_nodes(points: &[[f64; 2]], nodes: &mut Vec<EarClipNode>) -> Option<usize> {
1449    let mut cleaned: Vec<[f64; 2]> = Vec::with_capacity(points.len());
1450    for &point in points {
1451        if cleaned.last() != Some(&point) {
1452            cleaned.push(point);
1453        }
1454    }
1455    if cleaned.len() >= 2 && cleaned.first() == cleaned.last() {
1456        cleaned.pop();
1457    }
1458    if cleaned.len() < 3 {
1459        return None;
1460    }
1461    let first_index = nodes.len();
1462    for &vertex in &cleaned {
1463        nodes.push(EarClipNode {
1464            vertex,
1465            previous_index: 0,
1466            next_index: 0,
1467        });
1468    }
1469    let vertex_count = cleaned.len();
1470    for position in 0..vertex_count {
1471        let node_index = first_index + position;
1472        nodes[node_index].previous_index =
1473            first_index + (position + vertex_count - 1) % vertex_count;
1474        nodes[node_index].next_index = first_index + (position + 1) % vertex_count;
1475    }
1476    Some(first_index)
1477}

1479/// Normalizes a ring to one winding: drops the repeated closing vertex and
1480/// consecutive duplicates, reverses the ring when its signed area has the
1481/// wrong sign, and returns an empty ring for fewer than three vertices or
1482/// near-zero area.
1483fn normalized_ring(ring: &[[f64; 2]], positive_winding: bool) -> Vec<[f64; 2]> {
1484    let mut normalized: Vec<[f64; 2]> = Vec::with_capacity(ring.len());
1485    for &point in ring {
1486        if normalized.last() != Some(&point) {
1487            normalized.push(point);
1488        }
1489    }
1490    if normalized.len() >= 2 && normalized.first() == normalized.last() {
1491        normalized.pop();
1492    }
1493    if normalized.len() < 3 {
1494        return Vec::new();
1495    }
1496    let doubled_signed_area = ring_doubled_signed_area(&normalized);
1497    if doubled_signed_area.abs() < POLYGON_AREA_DEGENERACY_TOLERANCE {
1498        return Vec::new();
1499    }
1500    if (doubled_signed_area > 0.0) != positive_winding {
1501        normalized.reverse();
1502    }
1503    normalized
1504}

1506/// The ring vertex with the smallest x (the bridge search's ray origin side).
1507fn ring_leftmost_point(ring: &[[f64; 2]]) -> [f64; 2] {
1508    ring.iter()
1509        .copied()
1510        .fold([f64::INFINITY, f64::INFINITY], |leftmost, point| {
1511            if point[0] < leftmost[0] {
1512                point
1513            } else {
1514                leftmost
1515            }
1516        })
1517}

1519/// Splices a hole ring into the outer ring at a bridge vertex: the outer
1520/// boundary turns into the hole, walks the hole ring, crosses back over the
1521/// zero-width bridge, and continues with the outer ring. `hole_head` must be
1522/// a node of the hole ring; the hole enters the cycle after `bridge_index`.
1523/// Returns the hole-head copy, the second cycle's head when the same wiring
1524/// splits a self-touching ring at a diagonal.
1525fn splice_hole_ring(bridge_index: usize, hole_head: usize, nodes: &mut Vec<EarClipNode>) -> usize {
1526    let bridge_copy_index = nodes.len();
1527    nodes.push(EarClipNode {
1528        vertex: nodes[bridge_index].vertex,
1529        previous_index: 0,
1530        next_index: 0,
1531    });
1532    let hole_head_copy_index = nodes.len();
1533    nodes.push(EarClipNode {
1534        vertex: nodes[hole_head].vertex,
1535        previous_index: 0,
1536        next_index: 0,
1537    });
1538    let bridge_next_index = nodes[bridge_index].next_index;
1539    let hole_last_index = nodes[hole_head].previous_index;
1540    // bridge -> hole head ... hole last -> hole head copy -> bridge copy ->
1541    // (original bridge next), the standard splitPolygon wiring.
1542    nodes[bridge_index].next_index = hole_head;
1543    nodes[hole_head].previous_index = bridge_index;
1544    nodes[hole_last_index].next_index = hole_head_copy_index;
1545    nodes[hole_head_copy_index].previous_index = hole_last_index;
1546    nodes[hole_head_copy_index].next_index = bridge_copy_index;
1547    nodes[bridge_copy_index].previous_index = hole_head_copy_index;
1548    nodes[bridge_copy_index].next_index = bridge_next_index;
1549    nodes[bridge_next_index].previous_index = bridge_copy_index;
1550    hole_head_copy_index
1551}

1553/// The hole ring vertex with the smallest x: the bridge search's ray origin
1554/// and the vertex the splice must enter the hole through.
1555fn hole_leftmost_node(hole_head: usize, nodes: &[EarClipNode]) -> usize {
1556    let mut hole_leftmost = hole_head;
1557    let mut hole_node = nodes[hole_head].next_index;
1558    while hole_node != hole_head {
1559        if nodes[hole_node].vertex[0] < nodes[hole_leftmost].vertex[0] {
1560            hole_leftmost = hole_node;
1561        }
1562        hole_node = nodes[hole_node].next_index;
1563    }
1564    hole_leftmost
1565}

1567/// The outer ring node a hole ring should bridge to, or `None` when the hole
1568/// does not overlap the ring at all.
1569///
1570/// This is the `earcut` hole-bridge search: cast a ray westward from the
1571/// hole's leftmost vertex, take the nearest boundary edge it crosses, then
1572/// refine over the vertices of that crossing's wedge to the one with the
1573/// minimum angle that is locally inside the boundary. The refinement is what
1574/// keeps the bridge segment from crossing the ring on concave boundaries.
1575fn find_hole_bridge(
1576    hole_leftmost: usize,
1577    ring_head: usize,
1578    nodes: &[EarClipNode],
1579) -> Option<usize> {
1580    let [hole_x, hole_y] = nodes[hole_leftmost].vertex;
1581
1582    // First pass: the nearest boundary edge to the west of the hole's
1583    // leftmost vertex. The candidate is that edge's endpoint with the lesser
1584    // x, the side the hole approaches from.
1585    let mut nearest_crossing_x = f64::NEG_INFINITY;
1586    let mut bridge_index: Option<usize> = None;
1587    let mut ring_node = ring_head;
1588    loop {
1589        let ring_next = nodes[ring_node].next_index;
1590        let ring_vertex = nodes[ring_node].vertex;
1591        let ring_next_vertex = nodes[ring_next].vertex;
1592        if hole_y <= ring_vertex[1]
1593            && hole_y >= ring_next_vertex[1]
1594            && ring_next_vertex[1] != ring_vertex[1]
1595        {
1596            let crossing_x = ring_vertex[0]
1597                + (hole_y - ring_vertex[1]) * (ring_next_vertex[0] - ring_vertex[0])
1598                    / (ring_next_vertex[1] - ring_vertex[1]);
1599            if crossing_x <= hole_x && crossing_x > nearest_crossing_x {
1600                nearest_crossing_x = crossing_x;
1601                bridge_index = Some(if ring_vertex[0] < ring_next_vertex[0] {
1602                    ring_node
1603                } else {
1604                    ring_next
1605                });
1606                if crossing_x == hole_x {
1607                    return bridge_index;
1608                }
1609            }
1610        }
1611        ring_node = ring_next;
1612        if ring_node == ring_head {
1613            break;
1614        }
1615    }
1616    let initial_bridge_index = bridge_index?;
1617    let initial_bridge_vertex = nodes[initial_bridge_index].vertex;
1618    let stop_index = initial_bridge_index;
1619    let mut minimum_tangent = f64::INFINITY;
1620    let mut ring_node = initial_bridge_index;
1621    loop {
1622        let ring_vertex = nodes[ring_node].vertex;
1623        if hole_x >= ring_vertex[0]
1624            && ring_vertex[0] >= initial_bridge_vertex[0]
1625            && hole_x != ring_vertex[0]
1626            && point_lies_within_earcut_triangle(
1627                [
1628                    if hole_y < initial_bridge_vertex[1] {
1629                        hole_x
1630                    } else {
1631                        nearest_crossing_x
1632                    },
1633                    hole_y,
1634                ],
1635                initial_bridge_vertex,
1636                [
1637                    if hole_y < initial_bridge_vertex[1] {
1638                        nearest_crossing_x
1639                    } else {
1640                        hole_x
1641                    },
1642                    hole_y,
1643                ],
1644                ring_vertex,
1645            )
1646        {
1647            let tangent = (hole_y - ring_vertex[1]).abs() / (hole_x - ring_vertex[0]);
1648            if node_locally_inside(nodes, ring_node, [hole_x, hole_y]) && tangent < minimum_tangent
1649            {
1650                minimum_tangent = tangent;
1651                bridge_index = Some(ring_node);
1652            }
1653        }
1654        ring_node = nodes[ring_node].next_index;
1655        if ring_node == stop_index {
1656            break;
1657        }
1658    }
1659    bridge_index
1660}

1662/// Earcut's point-in-triangle test, written in the winding convention the
1663/// bridge search assumes (outer rings positive).
1664fn point_lies_within_earcut_triangle(
1665    point: [f64; 2],
1666    first: [f64; 2],
1667    second: [f64; 2],
1668    third: [f64; 2],
1669) -> bool {
1670    (third[0] - point[0]) * (first[1] - point[1]) >= (first[0] - point[0]) * (third[1] - point[1])
1671        && (first[0] - point[0]) * (second[1] - point[1])
1672            >= (second[0] - point[0]) * (first[1] - point[1])
1673        && (second[0] - point[0]) * (third[1] - point[1])
1674            >= (third[0] - point[0]) * (second[1] - point[1])
1675}

1677/// Whether the boundary is locally inside at a ring node for a candidate
1678/// bridge endpoint point: for a convex vertex the point must sit between the
1679/// node's two boundary edges, for a reflex vertex outside both.
1680fn node_locally_inside(nodes: &[EarClipNode], node_index: usize, point: [f64; 2]) -> bool {
1681    let previous_vertex = nodes[nodes[node_index].previous_index].vertex;
1682    let vertex = nodes[node_index].vertex;
1683    let next_vertex = nodes[nodes[node_index].next_index].vertex;
1684    if triangle_doubled_signed_area(previous_vertex, vertex, next_vertex) > 0.0 {
1685        triangle_doubled_signed_area(vertex, point, next_vertex) <= 0.0
1686            && triangle_doubled_signed_area(vertex, previous_vertex, point) <= 0.0
1687    } else {
1688        triangle_doubled_signed_area(vertex, point, previous_vertex) > 0.0
1689            || triangle_doubled_signed_area(vertex, next_vertex, point) > 0.0
1690    }
1691}

1698/// Whether the node at `ear_index` is a valid ear: convex, and with no
1699/// reflex boundary vertex inside or on its triangle. Duplicate bridge
1700/// vertices are collinear slits, so the reflex-only containment filter keeps
1701/// them from blocking every ear next to a splice.
1702fn is_ear_node(nodes: &[EarClipNode], ear_index: usize) -> bool {
1703    let previous_index = nodes[ear_index].previous_index;
1704    let next_index = nodes[ear_index].next_index;
1705    let previous_vertex = nodes[previous_index].vertex;
1706    let vertex = nodes[ear_index].vertex;
1707    let next_vertex = nodes[next_index].vertex;
1708    if triangle_doubled_signed_area(previous_vertex, vertex, next_vertex) <= 0.0 {
1709        return false;
1710    }
1711    let minimum_x = previous_vertex[0].min(vertex[0]).min(next_vertex[0]);
1712    let maximum_x = previous_vertex[0].max(vertex[0]).max(next_vertex[0]);
1713    let minimum_y = previous_vertex[1].min(vertex[1]).min(next_vertex[1]);
1714    let maximum_y = previous_vertex[1].max(vertex[1]).max(next_vertex[1]);
1715    let mut point_index = nodes[next_index].next_index;
1716    while point_index != previous_index {
1717        let point = nodes[point_index].vertex;
1718        if point[0] >= minimum_x
1719            && point[0] <= maximum_x
1720            && point[1] >= minimum_y
1721            && point[1] <= maximum_y
1722            && point_lies_within_earcut_triangle(point, previous_vertex, vertex, next_vertex)
1723        {
1724            let point_previous_vertex = nodes[nodes[point_index].previous_index].vertex;
1725            let point_next_vertex = nodes[nodes[point_index].next_index].vertex;
1726            if triangle_doubled_signed_area(point_previous_vertex, point, point_next_vertex) <= 0.0
1727            {
1728                return false;
1729            }
1730        }
1731        point_index = nodes[point_index].next_index;
1732    }
1733    true
1734}

1736/// Unlinks a node from the circular ring list, leaving it unreachable.
1737fn remove_ear_node(nodes: &mut [EarClipNode], node_index: usize) {
1738    let previous_index = nodes[node_index].previous_index;
1739    let next_index = nodes[node_index].next_index;
1740    nodes[previous_index].next_index = next_index;
1741    nodes[next_index].previous_index = previous_index;
1742}

1744/// Ear-clip loops over the assembled node ring, emitting each ear's triangle
1745/// and relinking around it. The traversal advances past the removed ear like
1746/// `earcut`, so well-formed rings stay near linear. Stalls recover in the
1747/// `earcut` order: first drop duplicate and collinear nodes, then cure local
1748/// self-intersections, then split the ring at a valid diagonal and clip both
1749/// halves. A ring that still has no ear stops instead of spinning, leaving
1750/// that remainder unfilled.
1751fn ear_clip_nodes(nodes: &mut Vec<EarClipNode>, ring_head: usize) -> Vec<[[f64; 2]; 3]> {
1752    let mut triangles = Vec::new();
1753    let mut ear_index = ring_head;
1754    let mut stop_index = ring_head;
1755    let mut filtered_once = false;
1756    let mut cured_once = false;
1757    while nodes[ear_index].previous_index != nodes[ear_index].next_index {
1758        let previous_index = nodes[ear_index].previous_index;
1759        let next_index = nodes[ear_index].next_index;
1760        if is_ear_node(nodes, ear_index) {
1761            triangles.push([
1762                nodes[previous_index].vertex,
1763                nodes[ear_index].vertex,
1764                nodes[next_index].vertex,
1765            ]);
1766            remove_ear_node(nodes, ear_index);
1767            ear_index = nodes[next_index].next_index;
1768            stop_index = ear_index;
1769            continue;
1770        }
1771        ear_index = next_index;
1772        if ear_index == stop_index {
1773            if !filtered_once {
1774                filtered_once = true;
1775                if let Some(filtered_head) = filter_ring_nodes(nodes, ear_index) {
1776                    ear_index = filtered_head;
1777                    stop_index = filtered_head;
1778                    continue;
1779                }
1780            }
1781            if !cured_once {
1782                cured_once = true;
1783                let cured_head = cure_local_intersections(nodes, ear_index, &mut triangles);
1784                ear_index = cured_head;
1785                stop_index = cured_head;
1786                continue;
1787            }
1788            if let Some((first_head, second_head)) = split_ring_at_valid_diagonal(nodes, ear_index)
1789            {
1790                triangles.extend(ear_clip_nodes(nodes, first_head));
1791                triangles.extend(ear_clip_nodes(nodes, second_head));
1792                return triangles;
1793            }
1794            break;
1795        }
1796    }
1797    triangles
1798}

1800/// The ear clipper's last resort (`earcut`'s `splitEarcut`): find any pair of
1801/// non-adjacent ring nodes joined by a valid diagonal, split the ring there
1802/// into two smaller cycles, and return both cycle heads. Splitting strictly
1803/// shrinks each piece, so the recursive call always terminates.
1804fn split_ring_at_valid_diagonal(
1805    nodes: &mut Vec<EarClipNode>,
1806    start: usize,
1807) -> Option<(usize, usize)> {
1808    let mut first = start;
1809    loop {
1810        let mut second = nodes[nodes[first].next_index].next_index;
1811        while second != nodes[first].previous_index {
1812            if is_valid_diagonal(nodes, first, second) {
1813                let second_head = splice_hole_ring(first, second, nodes);
1814                return Some((first, second_head));
1815            }
1816            second = nodes[second].next_index;
1817        }
1818        first = nodes[first].next_index;
1819        if first == start {
1820            break;
1821        }
1822    }
1823    None
1824}

1826/// Whether the diagonal between two ring nodes is safe to split at: the nodes
1827/// are non-adjacent, both ends are locally inside, the diagonal's midpoint is
1828/// inside the ring, it crosses no boundary edge, and it does not create
1829/// opposite-facing sectors (`earcut`'s `isValidDiagonal`).
1830fn is_valid_diagonal(nodes: &[EarClipNode], first: usize, second: usize) -> bool {
1831    let first_previous = nodes[first].previous_index;
1832    let first_next = nodes[first].next_index;
1833    let second_previous = nodes[second].previous_index;
1834    let second_next = nodes[second].next_index;
1835    first_next != second
1836        && first_previous != second
1837        && !diagonal_intersects_ring(nodes, first, second)
1838        && node_locally_inside_node(nodes, first, second)
1839        && node_locally_inside_node(nodes, second, first)
1840        && diagonal_middle_inside(nodes, first, second)
1841        && (triangle_doubled_signed_area(
1842            nodes[first_previous].vertex,
1843            nodes[first].vertex,
1844            nodes[second_previous].vertex,
1845        ) != 0.0
1846            || triangle_doubled_signed_area(
1847                nodes[first].vertex,
1848                nodes[second_previous].vertex,
1849                nodes[second].vertex,
1850            ) != 0.0
1851            || (nodes[first].vertex == nodes[second].vertex
1852                && triangle_doubled_signed_area(
1853                    nodes[first_previous].vertex,
1854                    nodes[first].vertex,
1855                    nodes[first_next].vertex,
1856                ) > 0.0
1857                && triangle_doubled_signed_area(
1858                    nodes[second_previous].vertex,
1859                    nodes[second].vertex,
1860                    nodes[second_next].vertex,
1861                ) > 0.0))
1862}

1864/// Whether a candidate split diagonal crosses any ring edge (`earcut`'s
1865/// `intersectsPolygon`).
1866fn diagonal_intersects_ring(nodes: &[EarClipNode], first: usize, second: usize) -> bool {
1867    let mut node = first;
1868    loop {
1869        let next = nodes[node].next_index;
1870        if node != first
1871            && next != first
1872            && node != second
1873            && next != second
1874            && segments_intersect(
1875                nodes[node].vertex,
1876                nodes[next].vertex,
1877                nodes[first].vertex,
1878                nodes[second].vertex,
1879            )
1880        {
1881            return true;
1882        }
1883        node = next;
1884        if node == first {
1885            break;
1886        }
1887    }
1888    false
1889}

1891/// Whether a candidate split diagonal's midpoint lies inside the ring
1892/// (even-odd ray cast; `earcut`'s `middleInside`).
1893fn diagonal_middle_inside(nodes: &[EarClipNode], first: usize, second: usize) -> bool {
1894    let midpoint = [
1895        (nodes[first].vertex[0] + nodes[second].vertex[0]) * 0.5,
1896        (nodes[first].vertex[1] + nodes[second].vertex[1]) * 0.5,
1897    ];
1898    let mut inside = false;
1899    let mut node = first;
1900    loop {
1901        let next = nodes[node].next_index;
1902        let current_vertex = nodes[node].vertex;
1903        let next_vertex = nodes[next].vertex;
1904        if (current_vertex[1] > midpoint[1]) != (next_vertex[1] > midpoint[1])
1905            && next_vertex[1] != current_vertex[1]
1906        {
1907            let boundary_x = (next_vertex[0] - current_vertex[0])
1908                * (midpoint[1] - current_vertex[1])
1909                / (next_vertex[1] - current_vertex[1])
1910                + current_vertex[0];
1911            if midpoint[0] < boundary_x {
1912                inside = !inside;
1913            }
1914        }
1915        node = next;
1916        if node == first {
1917            break;
1918        }
1919    }
1920    inside
1921}

1923/// The ear clipper's stall recovery: removes ring nodes that duplicate their
1924/// next node or are collinear with both neighbors, so the retried scan walks
1925/// a simpler boundary. Returns the head to resume from, or `None` when
1926/// nothing was removable.
1927fn filter_ring_nodes(nodes: &mut [EarClipNode], ring_head: usize) -> Option<usize> {
1928    let mut removed_any = false;
1929    let mut head = ring_head;
1930    loop {
1931        // Snapshot the current cycle so removal never invalidates the walk.
1932        let mut cycle_nodes = Vec::new();
1933        let mut node = head;
1934        loop {
1935            cycle_nodes.push(node);
1936            node = nodes[node].next_index;
1937            if node == head {
1938                break;
1939            }
1940        }
1941        if cycle_nodes.len() <= 3 {
1942            break;
1943        }
1944        let mut removable_node = None;
1945        for &candidate in &cycle_nodes {
1946            let previous_vertex = nodes[nodes[candidate].previous_index].vertex;
1947            let vertex = nodes[candidate].vertex;
1948            let next_vertex = nodes[nodes[candidate].next_index].vertex;
1949            let is_duplicate = vertex == next_vertex;
1950            let longer_edge_squared = (vertex[0] - previous_vertex[0])
1951                .hypot(vertex[1] - previous_vertex[1])
1952                .max((next_vertex[0] - vertex[0]).hypot(next_vertex[1] - vertex[1]))
1953                .powi(2);
1954            let is_collinear = triangle_doubled_signed_area(previous_vertex, vertex, next_vertex)
1955                .abs()
1956                <= EAR_CLIPPING_COLLINEAR_RELATIVE_TOLERANCE * longer_edge_squared;
1957            if is_duplicate || is_collinear {
1958                removable_node = Some(candidate);
1959                break;
1960            }
1961        }
1962        let Some(candidate) = removable_node else {
1963            break;
1964        };
1965        remove_ear_node(nodes, candidate);
1966        removed_any = true;
1967        if candidate == head {
1968            head = nodes[candidate].next_index;
1969        }
1970    }
1971    removed_any.then_some(head)
1972}

1974/// The ear clipper's second stall recovery (`earcut`'s
1975/// `cureLocalIntersections`): when two adjacent boundary edges cross, the
1976/// small triangle they form is emitted directly and the two crossing nodes
1977/// are removed, untangling a locally self-intersecting boundary (common in
1978/// real tile data where a hole touches its exterior at a vertex). Returns the
1979/// head to resume clipping from, after a final collinearity filter.
1980fn cure_local_intersections(
1981    nodes: &mut [EarClipNode],
1982    start: usize,
1983    triangles: &mut Vec<[[f64; 2]; 3]>,
1984) -> usize {
1985    let mut node = start;
1986    let mut head = start;
1987    loop {
1988        let previous_index = nodes[node].previous_index;
1989        let next_index = nodes[node].next_index;
1990        let after_next_index = nodes[next_index].next_index;
1991        if nodes[previous_index].vertex != nodes[after_next_index].vertex
1992            && segments_intersect(
1993                nodes[previous_index].vertex,
1994                nodes[node].vertex,
1995                nodes[next_index].vertex,
1996                nodes[after_next_index].vertex,
1997            )
1998            && node_locally_inside_node(nodes, previous_index, after_next_index)
1999            && node_locally_inside_node(nodes, after_next_index, previous_index)
2000        {
2001            triangles.push([
2002                nodes[previous_index].vertex,
2003                nodes[node].vertex,
2004                nodes[next_index].vertex,
2005            ]);
2006            remove_ear_node(nodes, node);
2007            remove_ear_node(nodes, next_index);
2008            node = after_next_index;
2009            head = after_next_index;
2010        }
2011        node = nodes[node].next_index;
2012        if node == head {
2013            break;
2014        }
2015    }
2016    filter_ring_nodes(nodes, node).unwrap_or(node)
2017}

2040/// Whether two closed segments intersect, including touching and collinear
2041/// overlaps (the `earcut` `intersects` test the local-intersection cure
2042/// relies on).
2043fn segments_intersect(
2044    first_start: [f64; 2],
2045    first_end: [f64; 2],
2046    second_start: [f64; 2],
2047    second_end: [f64; 2],
2048) -> bool {
2049    let first_orientation = sign_of(triangle_doubled_signed_area(
2050        first_start,
2051        first_end,
2052        second_start,
2053    ));
2054    let second_orientation = sign_of(triangle_doubled_signed_area(
2055        first_start,
2056        first_end,
2057        second_end,
2058    ));
2059    let third_orientation = sign_of(triangle_doubled_signed_area(
2060        second_start,
2061        second_end,
2062        first_start,
2063    ));
2064    let fourth_orientation = sign_of(triangle_doubled_signed_area(
2065        second_start,
2066        second_end,
2067        first_end,
2068    ));
2069    if first_orientation != second_orientation && third_orientation != fourth_orientation {
2070        return true;
2071    }
2072    (first_orientation == 0 && point_lies_on_segment(second_start, first_start, first_end))
2073        || (second_orientation == 0 && point_lies_on_segment(second_end, first_start, first_end))
2074        || (third_orientation == 0 && point_lies_on_segment(first_start, second_start, second_end))
2075        || (fourth_orientation == 0 && point_lies_on_segment(first_end, second_start, second_end))
2076}

2475    /// Whether a screen-space triangle contains a point (boundary inclusive).
2476    fn screen_triangle_contains_point(triangle: [[f64; 2]; 3], point: [f64; 2]) -> bool {
2477        let edge_side = |start: [f64; 2], end: [f64; 2]| {
2478            (end[0] - start[0]) * (point[1] - start[1])
2479                - (end[1] - start[1]) * (point[0] - start[0])
2480        };
2481        let sides = [
2482            edge_side(triangle[0], triangle[1]),
2483            edge_side(triangle[1], triangle[2]),
2484            edge_side(triangle[2], triangle[0]),
2485        ];
2486        !(sides.iter().any(|side| *side < 0.0) && sides.iter().any(|side| *side > 0.0))
2487    }

2735            .project_to_screen(&anchor, &viewport, 60.0)
2736            .expect("the camera projects its own scene point");
2737        let head_center_screen = [
2738            anchor_screen[0],
2739            anchor_screen[1]
2740                - DEFAULT_POINT_PIN_RADIUS_PIXELS * super::POINT_PIN_HEIGHT_RADIUS_FACTOR,
2741        ];

2743            DEFAULT_POINT_PIN_RADIUS_PIXELS * super::POINT_PIN_HOLLOW_RADIUS_RATIO;
2744        // The tip is an outline vertex: the ring samples the downward angle.
2745        let tip_appearances = vertices
2746            .iter()
2747            .filter(|vertex| {
2748                let screen_pixels = vertex_screen_pixels(vertex, &viewport);
2749                (screen_pixels[0] - anchor_screen[0]).abs() < SCREEN_PIXEL_TOLERANCE
2750                    && (screen_pixels[1] - anchor_screen[1]).abs() < SCREEN_PIXEL_TOLERANCE
2751            })
2752            .count();
2753        assert!(
2754            tip_appearances >= 2,
2755            "the tip vertex joins both triangles of its ring side: {vertices:?}"
2756        );
2757        let covered = |screen_point: [f64; 2]| {
2758            vertices.chunks_exact(3).any(|triangle| {
2759                let corners =
2760                    [0, 1, 2].map(|corner| vertex_screen_pixels(&triangle[corner], &viewport));
2761                screen_point_in_triangle(screen_point, corners)
2762            })
2763        };
2764        assert!(
2765            !covered(head_center_screen),
2766            "the head center stays unpainted (the hollow)"
2767        );
2768        assert!(
2769            !covered([
2770                head_center_screen[0] + hollow_radius_pixels * 0.5,
2771                head_center_screen[1]

2773            "a point inside the hollow radius stays unpainted"
2774        );

2776            covered([
2777                head_center_screen[0] + hollow_radius_pixels * 1.8,
2778                head_center_screen[1]

2780            "a point between the hollow and the head outline paints"
2781        );
2782        assert!(
2783            !covered([
2784                head_center_screen[0] + DEFAULT_POINT_PIN_RADIUS_PIXELS * 1.25,
2785                head_center_screen[1]
2786            ]),
2787            "a point beyond the head outline stays unpainted"
2788        );

2791    /// Point-in-triangle test on the display's pixel axes; orientation does
2792    /// not matter because the barycentric signs are compared as a set.
2793    fn screen_point_in_triangle(screen_point: [f64; 2], corners: [[f64; 2]; 3]) -> bool {
2794        let edge_signs: Vec<f64> = (0..3)
2795            .map(|edge_index| {
2796                let start = corners[edge_index];
2797                let end = corners[(edge_index + 1) % 3];
2798                (end[0] - start[0]) * (screen_point[1] - start[1])
2799                    - (end[1] - start[1]) * (screen_point[0] - start[0])
2800            })
2801            .collect();
2802        let has_negative = edge_signs
2803            .iter()
2804            .any(|sign| *sign < -SCREEN_PIXEL_TOLERANCE);
2805        let has_positive = edge_signs.iter().any(|sign| *sign > SCREEN_PIXEL_TOLERANCE);
2806        !(has_negative && has_positive)
2807    }
