1//! GPU-free overlay assembly: route polylines, filled polygons, point
2//! markers, and traffic-flow lines projected into clip-space solid-color
3//! triangle lists drawn over the tile quads.
4//!
5//! Overlays are the second draw layer of a captured frame: the same camera
6//! pose and viewport that assemble the tile quads
7//! ([`crate::renderer::Renderer::frame_quads`]) also project overlay shapes
8//! into clip-space triangles via [`overlay_triangle_vertices`], which the
9//! headless renderer draws through a separate solid-color pipeline in the
10//! same render pass after the tiles (see the crate-level seam notes in the
11//! browser-loop documentation: the wasm canvas mirrors the vertex contract
12//! verbatim, so the assembly here is deliberately `wgpu`-free and emits one
13//! flat, non-indexed triangle list).
14//!
15//! Design decisions recorded for this slice:
16//!
17//! * **World copies / antimeridian.** Overlay geometry wraps around the
18//!   camera exactly like tile spans do (see
19//!   [`crate::renderer::FrameProjection`]): the first vertex of a shape sits
20//!   in the camera's world copy via the shortest wrapped delta, every
21//!   following vertex unwraps continuously from it, and each resulting
22//!   segment or triangle is split against the camera-centered world-copy
23//!   windows ([`WORLD_COPY_WINDOW_OFFSETS`], reusing the tile spans' window
24//!   half-width). A shape wider than the covered window range clips to
25//!   nothing beyond it; a single unwrapping step never spans more than half a
26//!   world, so three windows cover every reachable fragment.
27//! * **Frustum clipping.** Assembly relies on the GPU's clip-space frustum
28//!   clip, the same mechanism the tile pipeline relies on for near-plane
29//!   straddling spans; no extra CPU cull runs. Screen-space-thickened shapes
30//!   (route lines, markers, traffic) additionally drop the segments and
31//!   markers with a vertex at or behind the camera because the screen-space
32//!   width offset needs a positive homogeneous weight; filled polygons emit
33//!   genuine clip-space corners and keep near-plane straddling triangles for
34//!   the GPU to clip exactly.
35//! * **Line joins.** A polyline draws as butt-capped segment quads; there are
36//!   deliberately no miter or round joins, so interior joints of a wide line
37//!   show a notch on the outside of sharp angles. Closing that gap needs a
38//!   per-vertex screen-space fan pass, deferred to a later slice together
39//!   with sprite/atlas marker rendering.
40//! * **Polygon holes.** A filled polygon takes one outer ring; hole rings are
41//!   deferred to a later slice (triangulating rings with holes needs a
42//!   bridge-joining step on top of the ear clipping implemented here).
43//! * **Markers.** A marker is a regular-octagon triangle fan centered on the
44//!   projected anchor point (center anchoring, not bottom pinning): the
45//!   projected coordinate is the fan's center, so a marker's center sits
46//!   exactly at its geographic position. Sprites and label callouts are
47//!   explicitly not this slice.
48

308/// Assembles every overlay of a frame into one flat clip-space triangle list
309/// drawn over the tile quads in the same captured frame.
310///
311/// The shapes project through the same camera pose as the tile quads, wrap
312/// around the camera's world copy like the tile spans do (see the module
313/// notes), and leave frustum clipping to the GPU's clip-space clip. An empty
314/// `overlays` slice produces no vertices, and a shape whose geometry turns
315/// out degenerate (behind the camera for screen-thickened shapes, or
316/// numerically unusable anywhere) skips only that shape's own triangles.
317pub fn overlay_triangle_vertices(
318    camera: &MapCamera,
319    viewport: &ViewportPixels,
320    field_of_view_degrees: f64,
321    overlays: &[MapOverlay],
322) -> Vec<ClipSpaceOverlayVertex> {
323    let frame_projection =
324        OverlayFrameProjection::for_frame(camera, viewport, field_of_view_degrees);
325    let mut vertices = Vec::new();
326    for overlay in overlays {
327        match overlay {
328            MapOverlay::RoutePolyline(route_polyline) => {
329                thickened_polyline_triangles(
330                    &frame_projection,
331                    viewport,
332                    &route_polyline.coordinates,
333                    route_polyline.rgba_color,
334                    route_polyline.width_pixels,
335                    &mut vertices,
336                );
337            }
338            MapOverlay::FilledPolygon(filled_polygon) => {
339                filled_polygon_triangles(
340                    &frame_projection,
341                    &filled_polygon.ring_coordinates,
342                    filled_polygon.rgba_color,
343                    &mut vertices,
344                );
345            }
346            MapOverlay::PointMarkers(point_markers) => {
347                for position in &point_markers.positions {
348                    point_marker_triangles(
349                        &frame_projection,
350                        viewport,
351                        position,
352                        point_markers.rgba_color,
353                        point_markers.radius_pixels,
354                        &mut vertices,
355                    );
356                }
357            }
358            MapOverlay::TrafficFlow(traffic_flow) => {
359                for traffic_flow_data in &traffic_flow.flows {
360                    thickened_polyline_triangles(
361                        &frame_projection,
362                        viewport,
363                        &traffic_flow_data.shape_coordinates,
364                        traffic_congestion_rgba(traffic_flow_data.congestion_level),
365                        traffic_flow.width_pixels,
366                        &mut vertices,
367                    );
368                }
369            }
370        }
371    }
372    vertices
373}

375/// Per-frame projection context for overlays: the view-projection matrix, the
376/// world pixel scale, and the camera's normalized-world x, mirroring the tile
377/// renderer's frame projection (see `crate::renderer::FrameProjection`) so
378/// both layers resolve the same world-position convention per frame. Shared
379/// (crate-visible) with the location-indicator layer, which projects through
380/// the same conventions.
381pub(crate) struct OverlayFrameProjection {
382    view_projection_matrix: DMat4,
383    zoom_scale_pixels: f64,
384    center_normalized_x: f64,
385}
386
387impl OverlayFrameProjection {
388    /// Builds the projection context for one frame.
389    pub(crate) fn for_frame(
390        camera: &MapCamera,
391        viewport: &ViewportPixels,
392        field_of_view_degrees: f64,
393    ) -> Self {
394        Self {
395            view_projection_matrix: camera.view_projection_matrix(viewport, field_of_view_degrees),
396            zoom_scale_pixels: zoom_scale_pixels(camera.zoom_level),
397            center_normalized_x: coordinate_to_normalized_world(&camera.center)[0],
398        }
399    }
400
401    /// The wrapped normalized-world x of a raw normalized-world x: the
402    /// position in `(-0.5, 0.5]` around the camera's world copy, shared with
403    /// the tile spans' wrapping rule.
404    pub(crate) fn wrapped_x(&self, normalized_x: f64) -> f64 {
405        self.center_normalized_x + normalized_shortest_delta(self.center_normalized_x, normalized_x)
406    }
407
408    /// Wraps a shape's coordinates into one continuous normalized-world chain:
409    /// the first vertex sits in the camera's world copy and every following
410    /// vertex unwraps from its predecessor by the shortest wrapped delta, so
411    /// chains crossing the antimeridian continue past the copy seam instead
412    /// of tearing back across the viewport (the same wrap-as-one-unit rule
413    /// the tile spans follow).
414    fn wrapped_continuous_chain(&self, coordinates: &[GeographicCoordinate]) -> Vec<[f64; 2]> {
415        let mut points = Vec::with_capacity(coordinates.len());
416        let mut previous_x: Option<f64> = None;
417        for coordinate in coordinates {
418            let normalized_world = coordinate_to_normalized_world(coordinate);
419            let wrapped_x = match previous_x {
420                None => self.wrapped_x(normalized_world[0]),
421                Some(previous_x_value) => {
422                    previous_x_value
423                        + normalized_shortest_delta(previous_x_value, normalized_world[0])
424                }
425            };
426            points.push([wrapped_x, normalized_world[1]]);
427            previous_x = Some(wrapped_x);
428        }
429        points
430    }
431
432    /// The camera-space clip position of a ground point of one world-copy
433    /// window. Camera world space is y-north pixel space (see
434    /// `MapCamera::ground_position_vector`); the window's copy offset is
435    /// subtracted from the wrapped x so the fragment continues the camera's
436    /// own copy past the window edge, exactly as the tile fragments do.
437    pub(crate) fn ground_clip_position(&self, window_offset: f64, x: f64, y: f64) -> DVec4 {
438        let x_pixels = (x - window_offset) * self.zoom_scale_pixels;
439        let y_pixels = -y * self.zoom_scale_pixels;
440        self.view_projection_matrix * DVec4::new(x_pixels, y_pixels, 0.0, 1.0)
441    }
442
443    /// The wrapped normalized-world x-range one world-copy window spans:
444    /// `[window_start_x, window_end_x)` around the camera's world copy,
445    /// shared by the segment fragments' split rule and the location
446    /// indicator's per-window anchor containment.
447    pub(crate) fn window_x_range(&self, window_offset: f64) -> [f64; 2] {
448        let window_start_x =
449            self.center_normalized_x - CAMERA_WORLD_WINDOW_HALF_WIDTH + window_offset;
450        [
451            window_start_x,
452            window_start_x + 2.0 * CAMERA_WORLD_WINDOW_HALF_WIDTH,
453        ]
454    }
455}

457/// Divides a clip position to viewport pixel coordinates (origin top-left, y
458/// down), matching [`MapCamera::project_to_screen`]. Sound only for a
459/// positive homogeneous weight; callers check that first where screen-space
460/// math follows.
461pub(crate) fn clip_position_to_screen_pixels(
462    clip_position: DVec4,
463    viewport: &ViewportPixels,
464) -> [f64; 2] {
465    let normalized_device_x = clip_position.x / clip_position.w;
466    let normalized_device_y = clip_position.y / clip_position.w;
467    [
468        (normalized_device_x + 1.0) * 0.5 * viewport.width_pixels,
469        (1.0 - (normalized_device_y + 1.0) * 0.5) * viewport.height_pixels,
470    ]
471}

473/// Re-expands a screen-pixel offset of one clip-space reference point into a
474/// clip position carrying the reference's depth and homogeneous weight. The
475/// screen offset is applied in normalized-device space around the reference's
476/// divided position, so per-vertex interpolation across a thickened quad ends
477/// at exactly the offset screen corners; the quad's interior follows the
478/// same perspective weighting as the underlying ground line.
479pub(crate) fn screen_pixels_to_clip_position(
480    screen_pixels: [f64; 2],
481    reference_clip_position: DVec4,
482    viewport: &ViewportPixels,
483) -> [f64; 4] {
484    let normalized_device_x = 2.0 * screen_pixels[0] / viewport.width_pixels - 1.0;
485    let normalized_device_y = 1.0 - 2.0 * screen_pixels[1] / viewport.height_pixels;
486    [
487        normalized_device_x * reference_clip_position.w,
488        normalized_device_y * reference_clip_position.w,
489        reference_clip_position.z,
490        reference_clip_position.w,
491    ]
492}

494/// Pushes one solid-color triangle, skipping it entirely when any corner is
495/// non-finite: an f32-overflowing corner would poison the whole draw with
496/// undefined geometry, while skipping keeps the rest of the overlay healthy.
497pub(crate) fn push_triangle(
498    vertices: &mut Vec<ClipSpaceOverlayVertex>,
499    corners: &[[f64; 4]; 3],
500    rgba_color: [f32; 4],
501) {
502    if corners
503        .iter()
504        .any(|corner| corner.iter().any(|component| !component.is_finite()))
505    {
506        return;
507    }
508    for corner in corners {
509        vertices.push(ClipSpaceOverlayVertex {
510            clip_position: [
511                corner[0] as f32,
512                corner[1] as f32,
513                corner[2] as f32,
514                corner[3] as f32,
515            ],
516            rgba_color,
517        });
518    }
519}

534/// Splits a shape segment against the camera-centered world-copy windows.
535///
536/// Each window keeps the segment's intersection with its x-range, matching
537/// [`CAMERA_WORLD_WINDOW_HALF_WIDTH`] around the camera per offset. A
538/// vertical segment (zero x-span) joins the one window containing its x,
539/// half-open on the east so a boundary-exact segment cannot draw twice; a
540/// segment touching a window only at its edge produces no fragment, mirroring
541/// the tile spans' empty-fragment rule.
542fn segment_window_fragments(
543    frame_projection: &OverlayFrameProjection,
544    segment_start: [f64; 2],
545    segment_end: [f64; 2],
546) -> Vec<SegmentWindowFragment> {
547    let span_x = segment_end[0] - segment_start[0];
548    let mut fragments = Vec::new();
549    for window_offset in WORLD_COPY_WINDOW_OFFSETS {
550        let window_offset_f64 = f64::from(window_offset);
551        let [window_start_x, window_end_x] = frame_projection.window_x_range(window_offset_f64);
552        let (start_fraction, end_fraction) = if span_x == 0.0 {
553            let zero_span_inside_window =
554                segment_start[0] >= window_start_x && segment_start[0] < window_end_x;
555            if !zero_span_inside_window {
556                continue;
557            }
558            (0.0, 1.0)
559        } else {
560            let clamped_start_fraction =
561                ((window_start_x - segment_start[0]) / span_x).clamp(0.0, 1.0);
562            let clamped_end_fraction = ((window_end_x - segment_start[0]) / span_x).clamp(0.0, 1.0);
563            if clamped_start_fraction == clamped_end_fraction {
564                continue;
565            }
566            (clamped_start_fraction, clamped_end_fraction)
567        };
568        let fragment_start = [
569            segment_start[0] + span_x * start_fraction,
570            segment_start[1] + (segment_end[1] - segment_start[1]) * start_fraction,
571        ];
572        let fragment_end = [
573            segment_start[0] + span_x * end_fraction,
574            segment_start[1] + (segment_end[1] - segment_start[1]) * end_fraction,
575        ];
576        fragments.push(SegmentWindowFragment {
577            start: fragment_start,
578            end: fragment_end,
579            window_offset: window_offset_f64,
580        });
581    }
582    fragments
583}

618/// Assembles one thickened segment fragment as a screen-space quad: the
619/// perpendicular offset of half the line width, applied in screen pixels
620/// around each endpoint's divided position, so a line keeps its pixel width
621/// regardless of zoom and pitch.
622///
623/// A fragment with a vertex at or behind the camera draws nothing — the
624/// screen-space offset needs a positive homogeneous weight — which is the
625/// documented limitation next to the filled polygons' GPU-clipped near-plane
626/// straddling (see the module notes).
627fn thickened_segment_triangles(
628    frame_projection: &OverlayFrameProjection,
629    viewport: &ViewportPixels,
630    fragment: &SegmentWindowFragment,
631    rgba_color: [f32; 4],
632    width_pixels: f64,
633    vertices: &mut Vec<ClipSpaceOverlayVertex>,
634) {
635    let start_clip_position = frame_projection.ground_clip_position(
636        fragment.window_offset,
637        fragment.start[0],
638        fragment.start[1],
639    );
640    let end_clip_position = frame_projection.ground_clip_position(
641        fragment.window_offset,
642        fragment.end[0],
643        fragment.end[1],
644    );
645    if start_clip_position.w <= 0.0 || end_clip_position.w <= 0.0 {
646        return;
647    }
648    let start_screen = clip_position_to_screen_pixels(start_clip_position, viewport);
649    let end_screen = clip_position_to_screen_pixels(end_clip_position, viewport);
650    let direction = [
651        end_screen[0] - start_screen[0],
652        end_screen[1] - start_screen[1],
653    ];
654    let screen_length = direction[0].hypot(direction[1]);
655    if screen_length < MINIMUM_VISIBLE_SEGMENT_LENGTH_PIXELS {
656        return;
657    }
658    let perpendicular_unit = [-direction[1] / screen_length, direction[0] / screen_length];
659    let half_width_pixels = width_pixels * 0.5;
660    let start_offset = [
661        start_screen[0] + perpendicular_unit[0] * half_width_pixels,
662        start_screen[1] + perpendicular_unit[1] * half_width_pixels,
663    ];
664    let start_opposite = [
665        start_screen[0] - perpendicular_unit[0] * half_width_pixels,
666        start_screen[1] - perpendicular_unit[1] * half_width_pixels,
667    ];
668    let end_offset = [
669        end_screen[0] + perpendicular_unit[0] * half_width_pixels,
670        end_screen[1] + perpendicular_unit[1] * half_width_pixels,
671    ];
672    let end_opposite = [
673        end_screen[0] - perpendicular_unit[0] * half_width_pixels,
674        end_screen[1] - perpendicular_unit[1] * half_width_pixels,
675    ];
676    // Corner order mirrors the tile quads: start-offset, start-opposite,
677    // end-offset, end-opposite, split by the quads' `0,1,2 / 2,1,3` triangle
678    // pattern, each corner re-expanded around its own endpoint's depth.
679    let corner_clip_positions = [
680        (start_offset, start_clip_position),
681        (start_opposite, start_clip_position),
682        (end_offset, end_clip_position),
683        (end_opposite, end_clip_position),
684    ]
685    .map(|(corner_screen_pixels, reference_clip_position)| {
686        screen_pixels_to_clip_position(corner_screen_pixels, reference_clip_position, viewport)
687    });
688    push_triangle(
689        vertices,
690        &[
691            corner_clip_positions[0],
692            corner_clip_positions[1],
693            corner_clip_positions[2],
694        ],
695        rgba_color,
696    );
697    push_triangle(
698        vertices,
699        &[
700            corner_clip_positions[2],
701            corner_clip_positions[1],
702            corner_clip_positions[3],
703        ],
704        rgba_color,
705    );
706}

758/// Assembles a filled polygon: the ring ear-clip triangulates in continuous
759/// normalized-world space, every triangle splits against the world-copy
760/// windows (convex clipping keeps the pieces convex), and each clipped piece
761/// fans into triangles projected by their genuine clip-space corners. Shared
762/// (crate-visible) with the location-indicator layer, whose ground-projected
763/// accuracy ring reuses the exact fill machinery.
764pub(crate) fn filled_polygon_triangles(
765    frame_projection: &OverlayFrameProjection,
766    ring_coordinates: &[GeographicCoordinate],
767    rgba_color: [f32; 4],
768    vertices: &mut Vec<ClipSpaceOverlayVertex>,
769) {
770    let wrapped_ring = frame_projection.wrapped_continuous_chain(ring_coordinates);
771    for triangle in ear_clip_triangulation(&wrapped_ring) {
772        for window_offset in WORLD_COPY_WINDOW_OFFSETS {
773            let window_offset_f64 = f64::from(window_offset);
774            let [window_start_x, window_end_x] = frame_projection.window_x_range(window_offset_f64);
775            let mut clipped_ring = clip_polygon_to_x_half_plane(&triangle, window_start_x, true);
776            clipped_ring = clip_polygon_to_x_half_plane(&clipped_ring, window_end_x, false);
777            if clipped_ring.len() < 3 {
778                continue;
779            }
780            // Clipped corners project by their genuine clip-space positions,
781            // so a triangle straddling the near plane stays for the GPU to
782            // clip exactly (see the module notes); only an f32-overflowing
783            // corner drops via `push_triangle`.
784            for fan_index in 1..clipped_ring.len() - 1 {
785                let corner_clip_positions = [
786                    clipped_ring[0],
787                    clipped_ring[fan_index],
788                    clipped_ring[fan_index + 1],
789                ]
790                .map(|corner| {
791                    let corner_clip_position = frame_projection.ground_clip_position(
792                        window_offset_f64,
793                        corner[0],
794                        corner[1],
795                    );
796                    [
797                        corner_clip_position.x,
798                        corner_clip_position.y,
799                        corner_clip_position.z,
800                        corner_clip_position.w,
801                    ]
802                });
803                push_triangle(vertices, &corner_clip_positions, rgba_color);
804            }
805        }
806    }
807}

809/// Clips a convex polygon to one vertical half-plane (Sutherland–Hodgman):
810/// `keep_greater` selects `x >= boundary_x`, otherwise `x <= boundary_x`.
811///
812/// Every result stays convex (a triangle or the clipped triangle), so callers
813/// fan-triangulate it safely. A degenerate one-component x-span (all shared
814/// boundary) contributes vertices exactly on the boundary; the caller treats
815/// fewer than three remaining vertices as empty.
816fn clip_polygon_to_x_half_plane(
817    polygon: &[[f64; 2]],
818    boundary_x: f64,
819    keep_greater: bool,
820) -> Vec<[f64; 2]> {
821    let mut clipped_polygon = Vec::with_capacity(polygon.len() + 1);
822    for edge_index in 0..polygon.len() {
823        let edge_start = polygon[edge_index];
824        let edge_end = polygon[(edge_index + 1) % polygon.len()];
825        let start_inside = if keep_greater {
826            edge_start[0] >= boundary_x
827        } else {
828            edge_start[0] <= boundary_x
829        };
830        let end_inside = if keep_greater {
831            edge_end[0] >= boundary_x
832        } else {
833            edge_end[0] <= boundary_x
834        };
835        if start_inside {
836            clipped_polygon.push(edge_start);
837        }
838        if start_inside != end_inside {
839            let crossing_fraction = (boundary_x - edge_start[0]) / (edge_end[0] - edge_start[0]);
840            // A crossing at the edge's own endpoint is a vertex on the
841            // boundary; it already landed through the inside-start pushes of
842            // the neighboring edges, and pushing it again would degenerate
843            // the clipped ring into a zero-area fan triangle.
844            if crossing_fraction > 0.0 && crossing_fraction < 1.0 {
845                clipped_polygon.push([
846                    boundary_x,
847                    edge_start[1] + (edge_end[1] - edge_start[1]) * crossing_fraction,
848                ]);
849            }
850        }
851    }
852    clipped_polygon
853}

855/// Doubled signed area of a ring (the fan-formula sum, without the constant
856/// factor): positive for one winding, negative for the other, zero for a
857/// collinear ring.
858fn ring_doubled_signed_area(ring: &[[f64; 2]]) -> f64 {
859    ring.iter()
860        .enumerate()
861        .map(|(index, vertex)| {
862            let next = ring[(index + 1) % ring.len()];
863            vertex[0] * next[1] - next[0] * vertex[1]
864        })
865        .sum()
866}

868/// Twice the signed area of one triangle (a × b cross-product form): positive
869/// for counterclockwise winding, negative for clockwise.
870fn triangle_doubled_signed_area(
871    first_vertex: [f64; 2],
872    second_vertex: [f64; 2],
873    third_vertex: [f64; 2],
874) -> f64 {
875    (second_vertex[0] - first_vertex[0]) * (third_vertex[1] - first_vertex[1])
876        - (second_vertex[1] - first_vertex[1]) * (third_vertex[0] - first_vertex[0])
877}

879/// Whether a point lies inside a triangle or on its boundary, by winding-sign
880/// tests against the triangle's (consistently wound) edges.
881fn point_lies_within_triangle(
882    point: [f64; 2],
883    first_vertex: [f64; 2],
884    second_vertex: [f64; 2],
885    third_vertex: [f64; 2],
886) -> bool {
887    triangle_doubled_signed_area(point, first_vertex, second_vertex)
888        >= -EAR_CLIPPING_BOUNDARY_TOLERANCE
889        && triangle_doubled_signed_area(point, second_vertex, third_vertex)
890            >= -EAR_CLIPPING_BOUNDARY_TOLERANCE
891        && triangle_doubled_signed_area(point, third_vertex, first_vertex)
892            >= -EAR_CLIPPING_BOUNDARY_TOLERANCE
893}

895/// Triangulates a simple polygon ring by ear clipping, over the ring's own
896/// vertices in O(n²) — no external geometry dependency, consistent with the
897/// crate owning its minimal math.
898///
899/// The ring normalizes to one winding (positive doubled signed area) first,
900/// so input order does not matter. Degenerate rings (fewer than three
901/// vertices, near-zero area, or a remainder with no clipgable ear — fully
902/// collinear chains) contribute fewer triangles or none; a numerically stuck
903/// remainder stops the loop instead of spinning, matching the documented
904/// "degenerate rings may fill nothing" contract.
905fn ear_clip_triangulation(ring: &[[f64; 2]]) -> Vec<[[f64; 2]; 3]> {
906    let mut normalized_ring = ring.to_vec();
907    if normalized_ring.len() >= 2 && normalized_ring.first() == normalized_ring.last() {
908        normalized_ring.pop();
909    }
910    if normalized_ring.len() < 3 {
911        return Vec::new();
912    }
913    let doubled_signed_area = ring_doubled_signed_area(&normalized_ring);
914    if doubled_signed_area.abs() < POLYGON_AREA_DEGENERACY_TOLERANCE {
915        return Vec::new();
916    }
917    if doubled_signed_area < 0.0 {
918        normalized_ring.reverse();
919    }
920    let mut remaining_indices: Vec<usize> = (0..normalized_ring.len()).collect();
921    let mut triangles = Vec::with_capacity(normalized_ring.len() - 2);
922    while remaining_indices.len() > 3 {
923        let ear_position = (0..remaining_indices.len())
924            .find(|&position| is_ring_ear(&normalized_ring, &remaining_indices, position));
925        let Some(ear_position) = ear_position else {
926            break;
927        };
928        let corner_indices = ear_corner_indices(&remaining_indices, ear_position);
929        triangles.push([
930            normalized_ring[corner_indices[0]],
931            normalized_ring[corner_indices[1]],
932            normalized_ring[corner_indices[2]],
933        ]);
934        remaining_indices.remove(ear_position);
935    }
936    if remaining_indices.len() == 3 {
937        triangles.push([
938            normalized_ring[remaining_indices[0]],
939            normalized_ring[remaining_indices[1]],
940            normalized_ring[remaining_indices[2]],
941        ]);
942    }
943    triangles
944}

946/// The previous, middle, and next remaining indices around a candidate ear
947/// position, wrapping the current ring order.
948fn ear_corner_indices(remaining_indices: &[usize], position: usize) -> [usize; 3] {
949    [
950        remaining_indices[(position + remaining_indices.len() - 1) % remaining_indices.len()],
951        remaining_indices[position],
952        remaining_indices[(position + 1) % remaining_indices.len()],
953    ]
954}
955
956/// Whether the vertex at `position` of the remaining ring is a valid ear:
957/// convex (positive triangle area in the normalized winding) and with no
958/// other remaining vertex inside or on the candidate triangle.
959fn is_ring_ear(ring: &[[f64; 2]], remaining_indices: &[usize], position: usize) -> bool {
960    let corner_indices = ear_corner_indices(remaining_indices, position);
961    let [previous_vertex, middle_vertex, next_vertex] = [
962        ring[corner_indices[0]],
963        ring[corner_indices[1]],
964        ring[corner_indices[2]],
965    ];
966    if triangle_doubled_signed_area(previous_vertex, middle_vertex, next_vertex) <= 0.0 {
967        return false;
968    }
969    remaining_indices
970        .iter()
971        .filter(|&&vertex_index| !corner_indices.contains(&vertex_index))
972        .all(|&vertex_index| {
973            !point_lies_within_triangle(
974                ring[vertex_index],
975                previous_vertex,
976                middle_vertex,
977                next_vertex,
978            )
979        })
980}
