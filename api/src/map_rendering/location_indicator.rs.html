1//! GPU-free location-indicator assembly: the position fix drawn as a
2//! screen-facing disc with its optional ground-projected accuracy ring and a
3//! course-directed wing, in the same clip-space stream pattern the overlay
4//! layer uses (see [`crate::overlay`]).
5//!
6//! The indicator is the third draw layer of a frame — the explicit draw
7//! order is tiles, then overlays, then the indicator
8//! ([`LOCATION_INDICATOR_TOP_LAYER_DRAW_ORDER`]) — so the fix's visuals
9//! always paint over route lines and polygons without the overlay list
10//! growing or reordering. The vertices reuse the
11//! [`crate::overlay::ClipSpaceOverlayVertex`] type and draw through the
12//! existing overlay solid-color pipeline (a shader/layout-identical third
13//! pass), alpha-blended over what the first two layers drew.
14//!
15//! Layer semantics, recorded for this slice:
16//!
17//! * **Screen anchoring.** The disc and its wing are screen-space shapes
18//!   anchored at the fix's projected position — the fix's pixel stays fixed
19//!   under zoom and pan while the layers beneath move, exactly like point
20//!   markers do (center anchoring, "screen-facing"). At pitch this keeps
21//!   the dot legible where a ground-projected dot would foreshorten to a
22//!   sliver, and the wing rotates with the camera's bearing so a course
23//!   heading reads as the same compass direction it reads on the ground.
24//! * **The guided-course visual.** [`LocationIndicatorVisualKind::GuidedCourseArrow`]
25//!   draws the course wing — a cone of [`HEADING_WEDGE_HALF_ANGLE_DEGREES`]
26//!   half-angle reaching [`LOCATION_COURSE_WEDGE_REACH_PIXELS`] up-course
27//!   from the disc center — underneath the same disc the plain form draws;
28//!   with the course unknown (`None`) it degrades to the plain dot. The
29//!   visual is the survey's "disc ± course cone" reading, not a separate
30//!   arrow sprite.
31//! * **Accuracy ring.** A ground-projected circle of the fix's horizontal
32//!   accuracy — the honest reading of "this fix is this accurate" — drawn
33//!   through the filled-polygon machinery, so it foreshortens into the
34//!   ellipse the ground shows at pitch and splits at world-copy boundaries
35//!   like every other ground shape. A fix more accurate than
36//!   [`MINIMUM_ACCURACY_RING_DRAW_METERS`] draws no ring (a circle inside
37//!   the disc itself carries no information), and an absurdly coarse fix
38//!   clamps to [`MAXIMUM_ACCURACY_RING_DIAMETER_PIXELS`] on screen instead
39//!   of flooding the viewport (the alternative — skipping an over-large
40//!   ring — is documented for the profiling pass to revisit).
41//! * **World copies.** The screen-facing shapes anchor once per visible
42//!   world-copy window like the tile spans' fragments, so at the zoom < 1
43//!   world-copy views the indicator draws in every copy that renders the
44//!   fix; the ground ring splits across windows in the polygon machinery.
45//! * **Behind the camera.** A screen-shape anchor whose homogeneous weight
46//!   is not positive (the fix behind the camera plane) draws no disc or
47//!   wing — the screen-space shapes' documented skip — while the
48//!   ground-projected ring still projects by genuine corners and leaves
49//!   near-plane straddling to the GPU's frustum clip.

208/// Assembles the location-indicator layer of one frame into the flat
209/// clip-space triangle list the overlay pipeline draws. A `None` layer (the
210/// cleared state) produces no vertices; a fix whose anchor projects behind
211/// the camera produces no disc or wing for any window, while its
212/// accuracy ring (a ground-projected polygon) still assembles with genuine
213/// corners for the GPU's frustum clip.
214pub fn location_indicator_vertices(
215    camera: &MapCamera,
216    viewport: &ViewportPixels,
217    field_of_view_degrees: f64,
218    location_indicator_layer: Option<&LocationIndicatorLayer>,
219) -> Vec<ClipSpaceOverlayVertex> {
220    let Some(location_indicator_layer) = location_indicator_layer else {
221        return Vec::new();
222    };
223    let frame_projection =
224        OverlayFrameProjection::for_frame(camera, viewport, field_of_view_degrees);
225    let normalized_world =
226        coordinate_to_normalized_world(&location_indicator_layer.state.coordinate);
227    let wrapped_center_x = frame_projection.wrapped_x(normalized_world[0]);
228    let rgba_color = location_indicator_layer.style.disc_rgba();
229    let mut vertices = Vec::new();
230
231    // The accuracy ring is a ground-projected circle, drawn first so the
232    // screen-facing disc and wing paint over it.
233    fill_accuracy_ring_triangles(
234        camera,
235        &frame_projection,
236        &location_indicator_layer.state,
237        &location_indicator_layer.style,
238        &mut vertices,
239    );
240
241    // The screen-facing shapes anchor once per visible world-copy window,
242    // like the tile spans' fragments: the fix's wrapped x must lie inside
243    // the window's span (the segments' shared half-open rule) before its
244    // anchor projects in that copy.
245    for window_offset in WORLD_COPY_WINDOW_OFFSETS {
246        let window_offset_f64 = f64::from(window_offset);
247        let [window_start_x, window_end_x] = frame_projection.window_x_range(window_offset_f64);
248        if wrapped_center_x < window_start_x || wrapped_center_x >= window_end_x {
249            continue;
250        }
251        let anchor_clip_position = frame_projection.ground_clip_position(
252            window_offset_f64,
253            wrapped_center_x,
254            normalized_world[1],
255        );
256        if anchor_clip_position.w <= 0.0 {
257            continue;
258        }
259        match location_indicator_layer.style.visual_kind {
260            LocationIndicatorVisualKind::TrackingDot => {
261                screen_facing_disc_triangles(
262                    &anchor_clip_position,
263                    viewport,
264                    location_indicator_layer.style.disc_radius_pixels(),
265                    rgba_color,
266                    &mut vertices,
267                );
268            }
269            LocationIndicatorVisualKind::GuidedCourseArrow => {
270                screen_facing_course_wing_triangles(
271                    &anchor_clip_position,
272                    viewport,
273                    camera.bearing_degrees,
274                    location_indicator_layer.state.course_heading_degrees,
275                    rgba_color,
276                    &mut vertices,
277                );
278                screen_facing_disc_triangles(
279                    &anchor_clip_position,
280                    viewport,
281                    location_indicator_layer.style.disc_radius_pixels(),
282                    rgba_color,
283                    &mut vertices,
284                );
285            }
286        }
287    }
288    vertices
289}

291/// The screen-space direction of a course heading under a camera bearing: a
292/// north-up camera shows north up (`[0, -1]` in y-down screen pixels), and
293/// the heading rotates by the camera's bearing so the wing reads as the
294/// same compass direction as the map beneath it.
295fn course_screen_direction(course_heading_degrees: f64, camera_bearing_degrees: f64) -> [f64; 2] {
296    let relative_radians = (course_heading_degrees - camera_bearing_degrees).to_radians();
297    [relative_radians.sin(), -relative_radians.cos()]
298}

300/// Assembles the accuracy ring for the layer's fix when the style shows
301/// rings and the accuracy clears the draw threshold; an over-large ring
302/// clamps to the screen-diameter cap through the camera's ground meters per
303/// screen pixel at the center.
304fn fill_accuracy_ring_triangles(
305    camera: &MapCamera,
306    frame_projection: &OverlayFrameProjection,
307    state: &LocationIndicatorState,
308    style: &LocationIndicatorStyle,
309    vertices: &mut Vec<ClipSpaceOverlayVertex>,
310) {
311    if !style.show_accuracy_ring() {
312        return;
313    }
314    let Some(horizontal_accuracy_meters) = state.horizontal_accuracy_meters else {
315        return;
316    };
317    if !horizontal_accuracy_meters.is_finite()
318        || horizontal_accuracy_meters < MINIMUM_ACCURACY_RING_DRAW_METERS
319    {
320        return;
321    }
322    // The screen-diameter clamp re-expressed in ground meters along the
323    // fix's own parallel: the ring is ground-projected, so the screen cap
324    // must apply to the ground circle the projection draws, and the flat-
325    // centered meters-per-pixel is the shared honest approximation (see the
326    // camera helper's contract text).
327    let clamp_radius_meters =
328        MAXIMUM_ACCURACY_RING_DIAMETER_PIXELS * 0.5 * camera.ground_meters_per_screen_pixel();
329    let radius_meters = horizontal_accuracy_meters.min(clamp_radius_meters);
330    filled_polygon_triangles(
331        frame_projection,
332        &accuracy_ring_coordinates(&state.coordinate, radius_meters),
333        &[],
334        DEFAULT_ACCURACY_RING_RGBA,
335        vertices,
336    );
337}

339/// A flat-earth approximation circle around the fix: the radius maps to
340/// latitude and longitude offsets through the WGS-84 equatorial radius and
341/// the fix's own parallel, which at the sub-kilometer accuracies the
342/// indicator draws is exact to far below a pixel of the ring's own extent.
343fn accuracy_ring_coordinates(
344    fix_coordinate: &GeographicCoordinate,
345    radius_meters: f64,
346) -> Vec<GeographicCoordinate> {
347    let latitude_radius_radians = radius_meters / EARTH_EQUATORIAL_RADIUS_METERS;
348    let longitude_radius_radians =
349        latitude_radius_radians / fix_coordinate.latitude.to_radians().cos();
350    (0..LOCATION_ACCURACY_RING_FAN_SIDES)
351        .map(|side_index| {
352            let side_angle = side_index as f64 * TAU / LOCATION_ACCURACY_RING_FAN_SIDES as f64;
353            GeographicCoordinate::from_degrees(
354                fix_coordinate.latitude + (latitude_radius_radians * side_angle.sin()).to_degrees(),
355                fix_coordinate.longitude
356                    + (longitude_radius_radians * side_angle.cos()).to_degrees(),
357            )
358            .expect("a fix inside the Mercator latitudes always admits a one-radius ring")
359        })
360        .collect()
361}

363/// Draws one screen-facing disc: a [`LOCATION_DISC_FAN_SIDES`]-side triangle
364/// fan centered on the anchor's screen position — the point-marker fan
365/// pattern, screen-facing and center-anchored (see [`crate::overlay`]).
366fn screen_facing_disc_triangles(
367    anchor_clip_position: &DVec4,
368    viewport: &ViewportPixels,
369    radius_pixels: f64,
370    rgba_color: [f32; 4],
371    vertices: &mut Vec<ClipSpaceOverlayVertex>,
372) {
373    let center_screen = clip_position_to_screen_pixels(*anchor_clip_position, viewport);
374    let center_corner =
375        screen_pixels_to_clip_position(center_screen, *anchor_clip_position, viewport);
376    for side_index in 0..LOCATION_DISC_FAN_SIDES {
377        let start_angle = side_index as f64 * TAU / LOCATION_DISC_FAN_SIDES as f64;
378        let end_angle = (side_index + 1) as f64 * TAU / LOCATION_DISC_FAN_SIDES as f64;
379        let ring_corner_start = screen_pixels_to_clip_position(
380            [
381                center_screen[0] + radius_pixels * start_angle.cos(),
382                center_screen[1] + radius_pixels * start_angle.sin(),
383            ],
384            *anchor_clip_position,
385            viewport,
386        );
387        let ring_corner_end = screen_pixels_to_clip_position(
388            [
389                center_screen[0] + radius_pixels * end_angle.cos(),
390                center_screen[1] + radius_pixels * end_angle.sin(),
391            ],
392            *anchor_clip_position,
393            viewport,
394        );
395        push_triangle(
396            vertices,
397            &[center_corner, ring_corner_start, ring_corner_end],
398            rgba_color,
399        );
400    }
401}

403/// Draws the screen-facing course wing: a [`LOCATION_COURSE_WEDGE_FAN_SIDES`]-triangle
404/// fan from the anchor's screen position spanning
405/// ±[`HEADING_WEDGE_HALF_ANGLE_DEGREES`] around the course's screen
406/// direction, reaching [`LOCATION_COURSE_WEDGE_REACH_PIXELS`] up-course. A
407/// fix without a course draws no wing (the guided form degrades to the
408/// plain dot).
409fn screen_facing_course_wing_triangles(
410    anchor_clip_position: &DVec4,
411    viewport: &ViewportPixels,
412    camera_bearing_degrees: f64,
413    course_heading_degrees: Option<f64>,
414    rgba_color: [f32; 4],
415    vertices: &mut Vec<ClipSpaceOverlayVertex>,
416) {
417    let Some(course_heading_degrees) = course_heading_degrees else {
418        return;
419    };
420    let direction = course_screen_direction(course_heading_degrees, camera_bearing_degrees);
421    let direction_angle = direction[1].atan2(direction[0]);
422    let half_wedge_angle = HEADING_WEDGE_HALF_ANGLE_DEGREES.to_radians();
423    let wedge_arc_span = 2.0 * half_wedge_angle;
424    let center_screen = clip_position_to_screen_pixels(*anchor_clip_position, viewport);
425    let center_corner =
426        screen_pixels_to_clip_position(center_screen, *anchor_clip_position, viewport);
427    for segment_index in 0..LOCATION_COURSE_WEDGE_FAN_SIDES {
428        let start_angle = direction_angle - half_wedge_angle
429            + wedge_arc_span * segment_index as f64 / LOCATION_COURSE_WEDGE_FAN_SIDES as f64;
430        let end_angle = direction_angle - half_wedge_angle
431            + wedge_arc_span * (segment_index + 1) as f64 / LOCATION_COURSE_WEDGE_FAN_SIDES as f64;
432        let wing_corner_start = screen_pixels_to_clip_position(
433            [
434                center_screen[0] + LOCATION_COURSE_WEDGE_REACH_PIXELS * start_angle.cos(),
435                center_screen[1] + LOCATION_COURSE_WEDGE_REACH_PIXELS * start_angle.sin(),
436            ],
437            *anchor_clip_position,
438            viewport,
439        );
440        let wing_corner_end = screen_pixels_to_clip_position(
441            [
442                center_screen[0] + LOCATION_COURSE_WEDGE_REACH_PIXELS * end_angle.cos(),
443                center_screen[1] + LOCATION_COURSE_WEDGE_REACH_PIXELS * end_angle.sin(),
444            ],
445            *anchor_clip_position,
446            viewport,
447        );
448        push_triangle(
449            vertices,
450            &[center_corner, wing_corner_start, wing_corner_end],
451            rgba_color,
452        );
453    }
454}
