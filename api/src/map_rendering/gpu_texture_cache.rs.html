1//! Byte-bounded LRU cache of tiles already uploaded to GPU memory.
2//!
3//! Uploads are expensive — every pooled payload is decoded to its full RGBA8
4//! image and copied onto the device — and a steady camera keeps the same
5//! tiles visible for many consecutive frames. This cache stores what
6//! [`crate::headless::HeadlessTileRenderer`] uploaded, keyed by tile and
7//! stamped with the pool's payload revision (see
8//! [`crate::tile_texture_pool::TileTexturePool::payload_revision`]): a pool
9//! insert under the same key replaces the bytes, so a cached texture whose
10//! revision no longer matches is stale and must never be drawn from. Frames
11//! prune stale entries first, then draw from whatever survives.
12//!
13//! Eviction order is least-recently-used, and the budget is a mirror of the
14//! encoded-payload pool's byte budget idea: as with the pool, a single
15//! oversized texture is kept alone rather than flash the map.
16
17use crate::tile_key::TileKey;
18use std::collections::HashMap;
19use wgpu::{BindGroup, Texture};
20
21/// Default GPU byte budget, sized as a heuristic for one complete 16 × 16
22/// tile grid of 256-px RGBA8 tiles — the payload size platform raster tile
23/// sources produce today. Budgets size the cache, not the textures: the
24/// upload path textures whatever a payload decodes to, so a 512-px payload
25/// holds half as many tiles in the same byte budget.
26pub const GPU_TILE_TEXTURE_CACHE_DEFAULT_MAXIMUM_BYTES: u64 =
27    256 * TILE_TEXTURE_PIXELS * TILE_TEXTURE_PIXELS * PIXEL_BYTES;
28
29/// Assumed texel edge of the tile textures the default budget heuristic is
30/// sized for; the actual texture size comes from the decoded payload.
31const TILE_TEXTURE_PIXELS: u64 = 256;
32
33/// Bytes of one RGBA8 texel.
34const PIXEL_BYTES: u64 = 4;
35
36/// One tile's uploaded state: the GPU texture, the bind group that draws it
37/// with the renderer's pipeline, and the identity data that decides whether
38/// the upload is still valid.
39#[derive(Debug)]
40struct GpuTileTexture {
41    pool_revision: u64,
42    texture: Texture,
43    bind_group: BindGroup,
44    texture_bytes: u64,
45    last_accessed_recency: u64,
46}
47
48impl GpuTileTexture {
49    /// Whether this upload still matches the pool's payload: it is stale when
50    /// the payload under the same tile key was replaced, reinserted, or
51    /// evicted since.
52    fn matches_pool_revision(&self, pool_revision: u64) -> bool {
53        self.pool_revision == pool_revision
54    }
55}
56
57/// LRU cache of uploaded tile textures, bounded in GPU bytes.
58#[derive(Debug)]
59pub struct GpuTileTextureCache {
60    maximum_stored_bytes: u64,
61    stored_bytes: u64,
62    recency_counter: u64,
63    /// Diagnostics: uploads ever inserted (each insert decodes and copies a
64    /// payload onto the device), and lookups served by an existing upload.
65    total_upload_count: u64,
66    total_cache_hit_count: u64,
67    entries: HashMap<TileKey, GpuTileTexture>,
68}
69
70impl GpuTileTextureCache {
71    /// Creates an empty cache bounded by `maximum_stored_bytes`.
72    pub fn new(maximum_stored_bytes: u64) -> Self {
73        Self {
74            maximum_stored_bytes,
75            stored_bytes: 0,
76            recency_counter: 0,
77            total_upload_count: 0,
78            total_cache_hit_count: 0,
79            entries: HashMap::new(),
80        }
81    }
82
83    /// Resizes the byte budget; entries beyond the new budget (keeping an
84    /// oversized lone entry) evict immediately, instead of waiting for the
85    /// next insertion that may never come on a steady all-hit scene.
86    pub fn set_maximum_stored_bytes(&mut self, maximum_stored_bytes: u64) {
87        self.maximum_stored_bytes = maximum_stored_bytes;
88        self.evict_until_within_budget();
89    }
90
91    /// Refreshes an entry's recency without counting it as a lookup; reports
92    /// whether the tile is currently uploaded. The renderer touches every
93    /// tile about to be drawn before uploading, so one insert's eviction pass
94    /// inside a frame cannot remove an entry a later quad of the same frame
95    /// reuses.
96    pub(crate) fn touch(&mut self, tile_key: &TileKey) -> bool {
97        if let Some(cache_entry) = self.entries.get_mut(tile_key) {
98            self.recency_counter += 1;
99            cache_entry.last_accessed_recency = self.recency_counter;
100            true
101        } else {
102            false
103        }
104    }
105
106    /// Borrows the upload for the tile, but only when it matches the pool's
107    /// current payload revision — a stale entry stays hidden for pruning.
108    /// Refreshes the entry's recency and the hit diagnostics on a hit.
109    pub(crate) fn lookup(
110        &mut self,
111        tile_key: &TileKey,
112        pool_revision: u64,
113    ) -> Option<(&Texture, &BindGroup)> {
114        let cache_entry = self.entries.get_mut(tile_key)?;
115        if !cache_entry.matches_pool_revision(pool_revision) {
116            return None;
117        }
118        self.recency_counter += 1;
119        cache_entry.last_accessed_recency = self.recency_counter;
120        self.total_cache_hit_count += 1;
121        Some((&cache_entry.texture, &cache_entry.bind_group))
122    }
123
124    /// Stores a fresh upload, evicting least-recently-used entries until the
125    /// budget is honored. A single upload larger than the budget is still
126    /// kept — dropping the only visible tile would flash the map rather than
127    /// save the memory (exactly the pool's rule).
128    pub(crate) fn insert(
129        &mut self,
130        tile_key: TileKey,
131        pool_revision: u64,
132        texture: Texture,
133        bind_group: BindGroup,
134        texture_bytes: u64,
135    ) {
136        self.recency_counter += 1;
137        self.total_upload_count += 1;
138        if let Some(previous_entry) = self.entries.insert(
139            tile_key,
140            GpuTileTexture {
141                pool_revision,
142                texture,
143                bind_group,
144                texture_bytes,
145                last_accessed_recency: self.recency_counter,
146            },
147        ) {
148            self.stored_bytes -= previous_entry.texture_bytes;
149        }
150        self.stored_bytes += texture_bytes;
151        self.evict_until_within_budget();
152    }
153
154    /// Drops every upload whose payload revision no longer matches — the
155    /// prune that keeps GPU bytes from pinning tiles whose bytes the pool
156    /// replaced or no longer holds. Called by the renderer once per frame
157    /// against the pool's current revisions.
158    pub(crate) fn retain_matching(&mut self, mut matches: impl FnMut(&TileKey, u64) -> bool) {
159        self.entries.retain(|tile_key, cache_entry| {
160            let keep = matches(tile_key, cache_entry.pool_revision);
161            if !keep {
162                self.stored_bytes -= cache_entry.texture_bytes;
163            }
164            keep
165        });
166    }
167
168    fn evict_until_within_budget(&mut self) {
169        while self.stored_bytes > self.maximum_stored_bytes && self.entries.len() > 1 {
170            let eviction_candidate = self
171                .entries
172                .iter()
173                .min_by_key(|(_, cache_entry)| cache_entry.last_accessed_recency)
174                .map(|(tile_key, _)| *tile_key);
175            match eviction_candidate {
176                Some(candidate) => {
177                    if let Some(evicted_entry) = self.entries.remove(&candidate) {
178                        self.stored_bytes -= evicted_entry.texture_bytes;
179                    }
180                }
181                None => break,
182            }
183        }
184    }
185
186    /// Total GPU bytes currently held by cached uploads.
187    pub fn stored_bytes(&self) -> u64 {
188        self.stored_bytes
189    }
190
191    /// Number of tile uploads currently cached.
192    pub fn stored_texture_count(&self) -> usize {
193        self.entries.len()
194    }
195
196    /// Uploads ever inserted: each one paid a full decode plus device copy.
197    pub fn total_upload_count(&self) -> u64 {
198        self.total_upload_count
199    }
200
201    /// Lookups served by an existing upload instead of a new one.
202    pub fn total_cache_hit_count(&self) -> u64 {
203        self.total_cache_hit_count
204    }
205}
