1//! Shared vector-tile base pipeline (M22): one scheduler that fetches a
2//! vector tileset through the [`VectorTileProvider`] trait, decodes each tile
3//! once, styles it, and answers the overlay primitives the frame draws beneath
4//! its own overlays.
5//!
6//! Every target serves its vector base through this one implementation, so the
7//! native surfaces and the browser cannot drift in what a tile paints. The
8//! scheduler owns two caches:
9//!
10//! - **Decoded tiles**, keyed by the tile address: a tile's bytes are fetched
11//!   and decoded once and never again while it stays cached; a style change
12//!   restyles the decoded tile instead of refetching it.
13//! - **Styled geometry**, keyed by the tile address, the style revision and the
14//!   style's zoom bucket: the style's zoom span (`min_zoom`/`max_zoom`) and
15//!   width stops are evaluated at the camera's zoom level, so a camera pan
16//!   across a width stop repaints the tile from the style it already carries —
17//!   no refetch, no decode. The zoom level quantizes to
18//!   [`VECTOR_BASE_STYLE_ZOOM_QUANTIZATION_STEPS_PER_LEVEL`] steps per level so
19//!   a continuous zoom gesture cannot grow the styled cache without bound.
20//!
21//! A tile that fails to fetch or decode does not fail the frame: the scheduler
22//! skips it, keeps painting the tiles that succeeded, and reports the failure
23//! in the frame's counts and its last failure detail. The base map is one layer
24//! of a frame, never the frame's reason to fail.
25//!
26//! The scheduler is camera-agnostic: the caller lists the visible tile
27//! addresses (through `MapCamera::visible_tile_keys` or the page's own grid)
28//! and names the camera's zoom level; [`VectorBaseScheduler::served_zoom_level`]
29//! clamps that listing to the tileset's declared zoom span.
30//!
31//! Each resolved frame also carries the drawn tiles' label candidates, read
32//! from the active style's symbol layers. [`VectorBaseFrame::label_layer`]
33//! places them at a camera pose through the SDK's embedded default glyph font,
34//! so a surface that forwards the layer draws the same place names the browser
35//! demo does without any platform asset seam. A frame without candidates
36//! yields `None` and costs the renderer nothing.
37

56/// The decoded-tile cache's default capacity. A 1080p frame asks for a few
57/// dozen tiles, so the cache holds several camera positions' worth of decoded
58/// geometry: a pan back over visited ground repaints without refetching.
59pub const DEFAULT_VECTOR_BASE_MAXIMUM_CACHED_TILE_COUNT: usize = 512;
60
61/// How many style-zoom buckets one zoom level divides into. A bucket is the
62/// granularity at which a restyle repaints: the camera zoom quantizes to the
63/// nearest bucket, so a continuous zoom gesture restyles at most this many
64/// times per level instead of once per frame.
65pub const VECTOR_BASE_STYLE_ZOOM_QUANTIZATION_STEPS_PER_LEVEL: f64 = 4.0;

67/// One resolved vector base frame: the overlay primitives to draw beneath the
68/// caller's overlays plus the honest counts of what the frame drew.
69#[derive(Debug, Clone, PartialEq)]
70pub struct VectorBaseFrame {
71    /// The base overlay primitives, in tile-address order and then style-layer
72    /// order (later entries paint over earlier ones).
73    pub overlays: Vec<MapOverlay>,
74    /// The frame's label candidates, read from the drawn tiles' symbol layers
75    /// under the active style, in tile-address order and then style-layer
76    /// order. Empty when the style declares no symbol layer or no drawn tile
77    /// carries one, so a frame without labels stays label-free.
78    pub label_candidates: Vec<MapLabelCandidate>,
79    /// The distinct tile addresses the frame asked for, after clamping and
80    /// deduplication.
81    pub requested_tile_count: usize,
82    /// How many of those tiles the frame drew.
83    pub drawn_tile_count: usize,
84    /// How many of those tiles failed to fetch or decode this frame.
85    pub failed_tile_count: usize,
86    /// The style the frame painted with.
87    pub style_id: String,
88    /// The style's canvas background, or `None` when the style paints its own
89    /// full ground fill. A surface that hides the raster base while the vector
90    /// base serves clears to this color beneath the overlays, so the gaps
91    /// between the style's translucent fills read as the designed map rather
92    /// than as the platform's clear color.
93    pub background: Option<[f32; 4]>,
94    /// The failure detail of the last tile that failed this frame, if any.
95    pub last_failure_detail: Option<String>,
96}

98impl VectorBaseFrame {
99    /// Whether the frame drew every requested tile and none failed — the
100    /// base map has nothing left to fetch for this camera.
101    pub fn is_idle(&self) -> bool {
102        self.failed_tile_count == 0 && self.drawn_tile_count == self.requested_tile_count
103    }
104
105    /// The frame's label layer at the camera pose, measured and rasterized
106    /// through the SDK's embedded default glyph font ([`default_label_font`]).
107    ///
108    /// `None` when the frame carries no label candidates — a style without
109    /// symbol layers, a raster-only view, or no drawn tile naming anything —
110    /// so a caller that forwards the layer to the renderer costs no buffer,
111    /// no atlas upload and no draw on label-free frames. A frame whose
112    /// candidates all collide off-screen also yields an empty layer, which the
113    /// renderer treats the same way.
114    pub fn label_layer(
115        &self,
116        camera: &MapCamera,
117        viewport: &ViewportPixels,
118        field_of_view_degrees: f64,
119    ) -> Option<LabelLayer> {
120        self.label_layer_with_font(
121            camera,
122            viewport,
123            field_of_view_degrees,
124            default_label_font(),
125        )
126    }
127
128    /// [`Self::label_layer`] against an explicit glyph font: the override for
129    /// a caller that supplies its own glyph source instead of the SDK's
130    /// embedded Latin range.
131    pub fn label_layer_with_font(
132        &self,
133        camera: &MapCamera,
134        viewport: &ViewportPixels,
135        field_of_view_degrees: f64,
136        font: &LabelFont,
137    ) -> Option<LabelLayer> {
138        if self.label_candidates.is_empty() {
139            return None;
140        }
141        Some(LabelLayer::place(
142            camera,
143            viewport,
144            field_of_view_degrees,
145            &self.label_candidates,
146            font,
147        ))
148    }
149}

163/// The shared vector base pipeline: fetches, decodes, styles and caches one
164/// vector tileset.
165pub struct VectorBaseScheduler {
166    /// The tileset the scheduler fetches through.
167    provider: Arc<dyn VectorTileProvider>,
168    /// The active style's identifier (the style name the status reports).
169    style_id: String,
170    /// The active style's layer definitions.
171    style: VectorTileStyle,
172    /// Bumped on every style change; part of the styled cache key so a
173    /// geometry styled under a previous style is never reused.
174    style_revision: u64,
175    /// Decoded tiles by address (fetched and decoded once).
176    decoded_tiles: HashMap<TileKey, Arc<VectorTile>>,
177    /// The decoded cache's insertion order, for eviction.
178    decoded_tile_insertion_order: VecDeque<TileKey>,
179    /// Styled overlay geometry by tile, style revision and zoom bucket.
180    styled_tiles: HashMap<StyledTileCacheKey, Vec<MapOverlay>>,
181    /// The decoded cache's capacity.
182    maximum_cached_tile_count: usize,
183    /// The last frame's last tile failure, for the status readback.
184    last_failure_detail: Option<String>,
185}

241    /// The zoom level the tileset actually serves for a requested zoom level:
242    /// the request clamped into the descriptor's declared span, so a camera
243    /// beyond either edge keeps a base map at the nearest served resolution.
244    /// A descriptor declaring an inverted span clamps to its lower edge rather
245    /// than panicking (the providers refuse such a declaration at
246    /// construction; a hand-written implementor might not).
247    pub fn served_zoom_level(&self, requested_zoom_level: u8) -> u8 {
248        let descriptor = self.provider.source_descriptor();
249        let lowest_served_zoom_level = descriptor.minimum_zoom_level;
250        let highest_served_zoom_level = descriptor.maximum_zoom_level.max(lowest_served_zoom_level);
251        requested_zoom_level
252            .max(lowest_served_zoom_level)
253            .min(highest_served_zoom_level)
254    }

256    /// The last resolved frame's last tile failure detail, if that frame had
257    /// one. Cleared by a frame whose tiles all resolved.
258    pub fn last_failure_detail(&self) -> Option<&str> {
259        self.last_failure_detail.as_deref()
260    }

262    /// Replaces the style with the committed artifact a style id names, keeping
263    /// every decoded tile: the next frame restyles the cached tiles under the
264    /// new style without fetching them again. An unknown id reports the
265    /// artifact error and keeps the current style in force.
266    pub fn set_style(&mut self, style_id: &str) -> Result<(), VectorStyleArtifactError> {
267        let style = vector_tile_style_from_id(style_id)?;
268        self.style = style;
269        self.style_id = style_id.to_string();
270        self.style_revision += 1;
271        // A different style never reuses the previous style's geometry.
272        self.styled_tiles.clear();
273        Ok(())
274    }

276    /// Resolves the base map for the given visible tile addresses and camera
277    /// zoom level: every missing tile is fetched and decoded once, every tile
278    /// is styled at the camera zoom's bucket (from the decoded cache when the
279    /// bucket or the style changed), and the frame's overlay primitives are
280    /// returned in tile-address order. A tile that fails to fetch or decode is
281    /// skipped and counted; the frame never fails as a whole.
282    pub async fn resolve_frame(
283        &mut self,
284        visible_tile_keys: &[TileKey],
285        style_zoom_level: f64,
286    ) -> VectorBaseFrame {
287        let mut requested_tile_keys = visible_tile_keys.to_vec();
288        requested_tile_keys.sort();
289        requested_tile_keys.dedup();
290        let quantized_style_zoom = quantized_style_zoom(style_zoom_level);
291        let evaluated_style_zoom =
292            quantized_style_zoom as f64 / VECTOR_BASE_STYLE_ZOOM_QUANTIZATION_STEPS_PER_LEVEL;
293        let mut frame = VectorBaseFrame {
294            overlays: Vec::new(),
295            label_candidates: Vec::new(),
296            requested_tile_count: requested_tile_keys.len(),
297            drawn_tile_count: 0,
298            failed_tile_count: 0,
299            style_id: self.style_id.clone(),
300            background: self.style.background,
301            last_failure_detail: None,
302        };
303        for tile_key in requested_tile_keys {
304            let decoded_tile = match self.decoded_tile(tile_key).await {
305                Ok(decoded_tile) => decoded_tile,
306                Err(failure_detail) => {
307                    frame.failed_tile_count += 1;
308                    frame.last_failure_detail = Some(failure_detail);
309                    continue;
310                }
311            };
312            let styled_key = StyledTileCacheKey {
313                tile_key,
314                style_revision: self.style_revision,
315                quantized_style_zoom,
316            };
317            if !self.styled_tiles.contains_key(&styled_key) {
318                let overlays = vector_tile_overlays_at_zoom(
319                    &decoded_tile,
320                    VectorTileAddress {
321                        zoom_level: tile_key.zoom_level,
322                        tile_x: tile_key.tile_x,
323                        tile_y: tile_key.tile_y,
324                    },
325                    &self.style,
326                    evaluated_style_zoom,
327                );
328                self.styled_tiles.insert(styled_key, overlays);
329            }
330            if let Some(styled_overlays) = self.styled_tiles.get(&styled_key) {
331                frame.overlays.extend(styled_overlays.iter().cloned());
332            }
333            // Labels read from the same decoded tile at the same evaluated
334            // zoom, so a frame's text and geometry always agree on the camera
335            // pose. The walk is GPU-free and allocates nothing when the style
336            // declares no symbol layer; it is recomputed per frame rather than
337            // cached because it is cheap beside the styled geometry and shares
338            // no state with the overlay cache.
339            frame
340                .label_candidates
341                .extend(label_candidates_for_vector_tile(
342                    &decoded_tile,
343                    VectorTileAddress {
344                        zoom_level: tile_key.zoom_level,
345                        tile_x: tile_key.tile_x,
346                        tile_y: tile_key.tile_y,
347                    },
348                    &self.style,
349                    evaluated_style_zoom,
350                ));
351            frame.drawn_tile_count += 1;
352        }
353        self.last_failure_detail = frame.last_failure_detail.clone();
354        frame
355    }

357    /// The decoded tile for the address, fetching and decoding it only when the
358    /// cache does not already hold it.
359    async fn decoded_tile(&mut self, tile_key: TileKey) -> Result<Arc<VectorTile>, String> {
360        if let Some(decoded_tile) = self.decoded_tiles.get(&tile_key) {
361            return Ok(Arc::clone(decoded_tile));
362        }
363        let tile_data = self
364            .provider
365            .fetch_vector_tile(tile_key.tile_x, tile_key.tile_y, tile_key.zoom_level)
366            .await
367            .map_err(|fetch_error| {
368                format!(
369                    "the vector base tile {}/{}/{} failed to fetch: {fetch_error}",
370                    tile_key.zoom_level, tile_key.tile_x, tile_key.tile_y
371                )
372            })?;
373        let decoded_tile =
374            Arc::new(decode_wire_vector_tile(tile_data).map_err(|decode_error| {
375                format!(
376                    "the vector base tile {}/{}/{} failed to decode: {decode_error}",
377                    tile_key.zoom_level, tile_key.tile_x, tile_key.tile_y
378                )
379            })?);
380        self.store_decoded_tile(tile_key, Arc::clone(&decoded_tile));
381        Ok(decoded_tile)
382    }
383
384    /// Inserts a decoded tile and evicts the oldest decoded tiles beyond the
385    /// cache's capacity, together with every styled geometry entry they own.
386    fn store_decoded_tile(&mut self, tile_key: TileKey, decoded_tile: Arc<VectorTile>) {
387        if self.decoded_tiles.insert(tile_key, decoded_tile).is_none() {
388            self.decoded_tile_insertion_order.push_back(tile_key);
389        }
390        while self.decoded_tiles.len() > self.maximum_cached_tile_count {
391            let Some(evicted_tile_key) = self.decoded_tile_insertion_order.pop_front() else {
392                break;
393            };
394            self.decoded_tiles.remove(&evicted_tile_key);
395            self.styled_tiles
396                .retain(|styled_key, _| styled_key.tile_key != evicted_tile_key);
397        }
398    }
399}

401/// Quantizes a camera zoom level to the styled cache's bucket (the nearest
402/// [`VECTOR_BASE_STYLE_ZOOM_QUANTIZATION_STEPS_PER_LEVEL`]-th of a level).
403fn quantized_style_zoom(style_zoom_level: f64) -> i64 {
404    (style_zoom_level * VECTOR_BASE_STYLE_ZOOM_QUANTIZATION_STEPS_PER_LEVEL).round() as i64
405}

407/// Unwraps the wire envelope and decodes the tile: a gzip-wrapped body is
408/// decompressed first, so the decoder only ever sees plain MVT protobuf.
409fn decode_wire_vector_tile(tile_data: VectorTileData) -> Result<VectorTile, String> {
410    let plain_tile_bytes = match tile_data.tile_encoding {
411        VectorTileEncoding::MapboxVectorTile => tile_data.tile_bytes,
412        VectorTileEncoding::GzippedMapboxVectorTile => {
413            let mut gzip_decoder = GzDecoder::new(tile_data.tile_bytes.as_slice());
414            let mut unwrapped_bytes = Vec::new();
415            gzip_decoder
416                .read_to_end(&mut unwrapped_bytes)
417                .map_err(|decompress_error| {
418                    format!(
419                        "the gzip-wrapped vector tile payload failed to decompress: \
420                         {decompress_error}"
421                    )
422                })?;
423            unwrapped_bytes
424        }
425    };
426    decode_vector_tile(&plain_tile_bytes).map_err(|decode_error| decode_error.to_string())
427}

508            tile_y: u32,
509            zoom_level: u8,
510        ) -> Result<VectorTileData, SDKError> {
511            *self.fetch_count.lock().unwrap() += 1;
512            let tile_key = TileKey::new(tile_x, tile_y, zoom_level)?;
513            if self.failed_tile_keys.contains(&tile_key) {
514                return Err(SDKError::ResponseParsingFailure(format!(
515                    "the scripted provider refuses {zoom_level}/{tile_x}/{tile_y}"
516                )));
517            }
518            let tile_bytes = self
519                .served_tile_bytes
520                .get(&tile_key)
521                .cloned()
522                .ok_or_else(|| {
523                    SDKError::ResponseParsingFailure(format!(
524                        "the scripted provider serves no {zoom_level}/{tile_x}/{tile_y}"
525                    ))
526                })?;
527            VectorTileData::from_wire_bytes(tile_x, tile_y, zoom_level, tile_bytes)
528        }
