1//! Shared vector-tile base pipeline (M22): one scheduler that fetches a
2//! vector tileset through the [`VectorTileProvider`] trait, decodes each tile
3//! once, styles it, and answers the overlay primitives the frame draws beneath
4//! its own overlays.
5//!
6//! Every target serves its vector base through this one implementation, so the
7//! native surfaces and the browser cannot drift in what a tile paints. The
8//! scheduler owns two caches:
9//!
10//! - **Decoded tiles**, keyed by the tile address: a tile's bytes are fetched
11//!   and decoded once and never again while it stays cached; a style change
12//!   restyles the decoded tile instead of refetching it.
13//! - **Styled geometry**, keyed by the tile address, the style revision and the
14//!   style's zoom bucket: the style's zoom span (`min_zoom`/`max_zoom`) and
15//!   width stops are evaluated at the camera's zoom level, so a camera pan
16//!   across a width stop repaints the tile from the style it already carries —
17//!   no refetch, no decode. The zoom level quantizes to
18//!   [`VECTOR_BASE_STYLE_ZOOM_QUANTIZATION_STEPS_PER_LEVEL`] steps per level so
19//!   a continuous zoom gesture cannot grow the styled cache without bound.
20//!
21//! A tile that fails to fetch or decode does not fail the frame: the scheduler
22//! skips it, keeps painting the tiles that succeeded, and reports the failure
23//! in the frame's counts and its last failure detail. The base map is one layer
24//! of a frame, never the frame's reason to fail.
25//!
26//! The scheduler is camera-agnostic: the caller lists the visible tile
27//! addresses (through `MapCamera::visible_tile_keys` or the page's own grid)
28//! and names the camera's zoom level; [`VectorBaseScheduler::served_zoom_level`]
29//! clamps that listing to the tileset's declared zoom span.
30//!
31//! Each resolved frame also carries the drawn tiles' label candidates, read
32//! from the active style's symbol layers. [`VectorBaseFrame::label_layer`]
33//! places them at a camera pose through the SDK's embedded default glyph font,
34//! so a surface that forwards the layer draws the same place names the browser
35//! demo does without any platform asset seam. A frame without candidates
36//! yields `None` and costs the renderer nothing.
37

56/// The decoded-tile cache's default capacity. A 1080p frame asks for a few
57/// dozen tiles, so the cache holds several camera positions' worth of decoded
58/// geometry: a pan back over visited ground repaints without refetching.
59pub const DEFAULT_VECTOR_BASE_MAXIMUM_CACHED_TILE_COUNT: usize = 512;
60
61/// The styled-geometry cache's default capacity, in styled tiles. One decoded
62/// tile can be styled once per zoom bucket the camera passed through, so this
63/// cache is bounded on its own rather than only by decoded-tile eviction: four
64/// buckets' worth of the decoded capacity covers a pan and a zoom while a
65/// tile stays resident, and nothing beyond that is retained.
66pub const DEFAULT_VECTOR_BASE_MAXIMUM_CACHED_STYLED_TILE_COUNT: usize =
67    4 * DEFAULT_VECTOR_BASE_MAXIMUM_CACHED_TILE_COUNT;
68
69/// How many style-zoom buckets one zoom level divides into. A bucket is the
70/// granularity at which a restyle repaints: the camera zoom quantizes to the
71/// nearest bucket, so a continuous zoom gesture restyles at most this many
72/// times per level instead of once per frame.
73pub const VECTOR_BASE_STYLE_ZOOM_QUANTIZATION_STEPS_PER_LEVEL: f64 = 4.0;

75/// One resolved vector base frame: the overlay primitives to draw beneath the
76/// caller's overlays plus the honest counts of what the frame drew.
77#[derive(Debug, Clone, PartialEq)]
78pub struct VectorBaseFrame {
79    /// The base overlay primitives, in tile-address order and then style-layer
80    /// order (later entries paint over earlier ones).
81    pub overlays: Vec<MapOverlay>,
82    /// The frame's label candidates, read from the drawn tiles' symbol layers
83    /// under the active style, in tile-address order and then style-layer
84    /// order. Empty when the style declares no symbol layer or no drawn tile
85    /// carries one, so a frame without labels stays label-free.
86    pub label_candidates: Vec<MapLabelCandidate>,
87    /// The distinct tile addresses the frame asked for, after clamping and
88    /// deduplication.
89    pub requested_tile_count: usize,
90    /// How many of those tiles the frame drew.
91    pub drawn_tile_count: usize,
92    /// How many of those tiles failed to fetch or decode this frame.
93    pub failed_tile_count: usize,
94    /// The style the frame painted with.
95    pub style_id: String,
96    /// The style's canvas background, or `None` when the style paints its own
97    /// full ground fill. A surface that hides the raster base while the vector
98    /// base serves clears to this color beneath the overlays, so the gaps
99    /// between the style's translucent fills read as the designed map rather
100    /// than as the platform's clear color.
101    pub background: Option<[f32; 4]>,
102    /// The failure detail of the last tile that failed this frame, if any.
103    pub last_failure_detail: Option<String>,
104}

106impl VectorBaseFrame {
107    /// Whether the frame drew every requested tile and none failed — the
108    /// base map has nothing left to fetch for this camera.
109    pub fn is_idle(&self) -> bool {
110        self.failed_tile_count == 0 && self.drawn_tile_count == self.requested_tile_count
111    }
112
113    /// The frame's label layer at the camera pose, measured and rasterized
114    /// through the SDK's embedded default glyph font ([`default_label_font`]).
115    ///
116    /// `None` when the frame carries no label candidates — a style without
117    /// symbol layers, a raster-only view, or no drawn tile naming anything —
118    /// so a caller that forwards the layer to the renderer costs no buffer,
119    /// no atlas upload and no draw on label-free frames. A frame whose
120    /// candidates all collide off-screen also yields an empty layer, which the
121    /// renderer treats the same way.
122    pub fn label_layer(
123        &self,
124        camera: &MapCamera,
125        viewport: &ViewportPixels,
126        field_of_view_degrees: f64,
127    ) -> Option<LabelLayer> {
128        self.label_layer_with_font(
129            camera,
130            viewport,
131            field_of_view_degrees,
132            default_label_font(),
133        )
134    }
135
136    /// [`Self::label_layer`] against an explicit glyph font: the override for
137    /// a caller that supplies its own glyph source instead of the SDK's
138    /// embedded Latin range.
139    pub fn label_layer_with_font(
140        &self,
141        camera: &MapCamera,
142        viewport: &ViewportPixels,
143        field_of_view_degrees: f64,
144        font: &LabelFont,
145    ) -> Option<LabelLayer> {
146        if self.label_candidates.is_empty() {
147            return None;
148        }
149        Some(LabelLayer::place(
150            camera,
151            viewport,
152            field_of_view_degrees,
153            &self.label_candidates,
154            font,
155        ))
156    }
157}

171/// The shared vector base pipeline: fetches, decodes, styles and caches one
172/// vector tileset.
173pub struct VectorBaseScheduler {
174    /// The tileset the scheduler fetches through.
175    provider: Arc<dyn VectorTileProvider>,
176    /// The active style's identifier (the style name the status reports).
177    style_id: String,
178    /// The active style's layer definitions.

181    /// geometry styled under a previous style is never reused.

190    styled_tile_insertion_order: VecDeque<StyledTileCacheKey>,
191    /// The styled cache's capacity.
192    maximum_cached_styled_tile_count: usize,
193    /// The decoded cache's capacity.
194    maximum_cached_tile_count: usize,
195    /// The last frame's last tile failure, for the status readback.
196    last_failure_detail: Option<String>,
197}

255    /// The zoom level the tileset actually serves for a requested zoom level:
256    /// the request clamped into the descriptor's declared span, so a camera
257    /// beyond either edge keeps a base map at the nearest served resolution.
258    /// A descriptor declaring an inverted span clamps to its lower edge rather
259    /// than panicking (the providers refuse such a declaration at
260    /// construction; a hand-written implementor might not).
261    pub fn served_zoom_level(&self, requested_zoom_level: u8) -> u8 {
262        let descriptor = self.provider.source_descriptor();
263        let lowest_served_zoom_level = descriptor.minimum_zoom_level;
264        let highest_served_zoom_level = descriptor.maximum_zoom_level.max(lowest_served_zoom_level);
265        requested_zoom_level
266            .max(lowest_served_zoom_level)
267            .min(highest_served_zoom_level)
268    }

276    /// Replaces the style with the committed artifact a style id names, keeping
277    /// every decoded tile: the next frame restyles the cached tiles under the
278    /// new style without fetching them again. An unknown id reports the
279    /// artifact error and keeps the current style in force.
280    pub fn set_style(&mut self, style_id: &str) -> Result<(), VectorStyleArtifactError> {
281        let style = vector_tile_style_from_id(style_id)?;
282        self.style = style;
283        self.style_id = style_id.to_string();
284        self.style_revision += 1;
285        // A different style never reuses the previous style's geometry.
286        self.styled_tiles.clear();
287        self.styled_tile_insertion_order.clear();
288        Ok(())
289    }

291    /// Resolves the base map for the given visible tile addresses and camera
292    /// zoom level: every missing tile is fetched and decoded once, every tile
293    /// is styled at the camera zoom's bucket (from the decoded cache when the
294    /// bucket or the style changed), and the frame's overlay primitives are
295    /// returned in tile-address order. A tile that fails to fetch or decode is
296    /// skipped and counted; the frame never fails as a whole.
297    pub async fn resolve_frame(
298        &mut self,
299        visible_tile_keys: &[TileKey],
300        style_zoom_level: f64,
301    ) -> VectorBaseFrame {
302        let mut requested_tile_keys = visible_tile_keys.to_vec();
303        requested_tile_keys.sort();
304        requested_tile_keys.dedup();
305        let quantized_style_zoom = quantized_style_zoom(style_zoom_level);
306        let evaluated_style_zoom =
307            quantized_style_zoom as f64 / VECTOR_BASE_STYLE_ZOOM_QUANTIZATION_STEPS_PER_LEVEL;
308        let mut frame = VectorBaseFrame {
309            overlays: Vec::new(),
310            label_candidates: Vec::new(),
311            requested_tile_count: requested_tile_keys.len(),
312            drawn_tile_count: 0,
313            failed_tile_count: 0,
314            style_id: self.style_id.clone(),
315            background: self.style.background,
316            last_failure_detail: None,
317        };
318        for tile_key in requested_tile_keys {
319            let decoded_tile = match self.decoded_tile(tile_key).await {
320                Ok(decoded_tile) => decoded_tile,
321                Err(failure_detail) => {
322                    frame.failed_tile_count += 1;
323                    frame.last_failure_detail = Some(failure_detail);
324                    continue;
325                }
326            };
327            let styled_key = StyledTileCacheKey {
328                tile_key,
329                style_revision: self.style_revision,
330                quantized_style_zoom,
331            };
332            if !self.styled_tiles.contains_key(&styled_key) {
333                let overlays = vector_tile_overlays_at_zoom(
334                    &decoded_tile,
335                    VectorTileAddress {
336                        zoom_level: tile_key.zoom_level,
337                        tile_x: tile_key.tile_x,
338                        tile_y: tile_key.tile_y,
339                    },
340                    &self.style,
341                    evaluated_style_zoom,
342                );
343                self.store_styled_tile(styled_key, overlays);
344            }
345            if let Some(styled_overlays) = self.styled_tiles.get(&styled_key) {
346                frame.overlays.extend(styled_overlays.iter().cloned());
347            }
348            // Labels read from the same decoded tile at the same evaluated
349            // zoom, so a frame's text and geometry always agree on the camera
350            // pose. The walk is GPU-free and allocates nothing when the style
351            // declares no symbol layer; it is recomputed per frame rather than
352            // cached because it is cheap beside the styled geometry and shares
353            // no state with the overlay cache.
354            frame
355                .label_candidates
356                .extend(label_candidates_for_vector_tile(
357                    &decoded_tile,
358                    VectorTileAddress {
359                        zoom_level: tile_key.zoom_level,
360                        tile_x: tile_key.tile_x,
361                        tile_y: tile_key.tile_y,
362                    },
363                    &self.style,
364                    evaluated_style_zoom,
365                ));
366            frame.drawn_tile_count += 1;
367        }
368        self.last_failure_detail = frame.last_failure_detail.clone();
369        frame
370    }

372    /// The decoded tile for the address, fetching and decoding it only when the
373    /// cache does not already hold it.
374    async fn decoded_tile(&mut self, tile_key: TileKey) -> Result<Arc<VectorTile>, String> {
375        if let Some(decoded_tile) = self.decoded_tiles.get(&tile_key) {
376            return Ok(Arc::clone(decoded_tile));
377        }
378        let tile_data = self
379            .provider
380            .fetch_vector_tile(tile_key.tile_x, tile_key.tile_y, tile_key.zoom_level)
381            .await
382            .map_err(|fetch_error| {
383                format!(
384                    "the vector base tile {}/{}/{} failed to fetch: {fetch_error}",
385                    tile_key.zoom_level, tile_key.tile_x, tile_key.tile_y
386                )
387            })?;
388        let decoded_tile =
389            Arc::new(decode_wire_vector_tile(tile_data).map_err(|decode_error| {
390                format!(
391                    "the vector base tile {}/{}/{} failed to decode: {decode_error}",
392                    tile_key.zoom_level, tile_key.tile_x, tile_key.tile_y
393                )
394            })?);
395        self.store_decoded_tile(tile_key, Arc::clone(&decoded_tile));
396        Ok(decoded_tile)
397    }
398
399    /// Inserts a decoded tile and evicts the oldest decoded tiles beyond the
400    /// cache's capacity, together with every styled geometry entry they own.
401    /// Stores styled geometry, evicting the least recently inserted entry
402    /// once the styled cache is over its own capacity.
403    ///
404    /// The decoded-tile eviction below drops a tile's styled entries, but a
405    /// tile stays decoded for many frames while the camera may style it once
406    /// per zoom bucket; this cache is therefore bounded on its own too.
407    fn store_styled_tile(&mut self, styled_key: StyledTileCacheKey, overlays: Vec<MapOverlay>) {
408        if self.styled_tiles.insert(styled_key, overlays).is_none() {
409            self.styled_tile_insertion_order.push_back(styled_key);
410        }
411        while self.styled_tiles.len() > self.maximum_cached_styled_tile_count {
412            let Some(evicted_styled_key) = self.styled_tile_insertion_order.pop_front() else {
413                break;
414            };
415            self.styled_tiles.remove(&evicted_styled_key);
416        }
417    }
418
419    fn store_decoded_tile(&mut self, tile_key: TileKey, decoded_tile: Arc<VectorTile>) {
420        if self.decoded_tiles.insert(tile_key, decoded_tile).is_none() {
421            self.decoded_tile_insertion_order.push_back(tile_key);
422        }
423        while self.decoded_tiles.len() > self.maximum_cached_tile_count {
424            let Some(evicted_tile_key) = self.decoded_tile_insertion_order.pop_front() else {
425                break;
426            };
427            self.decoded_tiles.remove(&evicted_tile_key);
428            self.styled_tiles
429                .retain(|styled_key, _| styled_key.tile_key != evicted_tile_key);
430            self.styled_tile_insertion_order
431                .retain(|styled_key| styled_key.tile_key != evicted_tile_key);
432        }
433    }
434}

436/// Quantizes a camera zoom level to the styled cache's bucket (the nearest
437/// [`VECTOR_BASE_STYLE_ZOOM_QUANTIZATION_STEPS_PER_LEVEL`]-th of a level).
438fn quantized_style_zoom(style_zoom_level: f64) -> i64 {
439    (style_zoom_level * VECTOR_BASE_STYLE_ZOOM_QUANTIZATION_STEPS_PER_LEVEL).round() as i64
440}

442/// Unwraps the wire envelope and decodes the tile: a gzip-wrapped body is
443/// decompressed first, so the decoder only ever sees plain MVT protobuf.
444fn decode_wire_vector_tile(tile_data: VectorTileData) -> Result<VectorTile, String> {
445    let plain_tile_bytes = match tile_data.tile_encoding {
446        VectorTileEncoding::MapboxVectorTile => tile_data.tile_bytes,
447        VectorTileEncoding::GzippedMapboxVectorTile => {
448            let mut gzip_decoder = GzDecoder::new(tile_data.tile_bytes.as_slice());
449            let mut unwrapped_bytes = Vec::new();
450            gzip_decoder
451                .read_to_end(&mut unwrapped_bytes)
452                .map_err(|decompress_error| {
453                    format!(
454                        "the gzip-wrapped vector tile payload failed to decompress: \
455                         {decompress_error}"
456                    )
457                })?;
458            unwrapped_bytes
459        }
460    };
461    decode_vector_tile(&plain_tile_bytes).map_err(|decode_error| decode_error.to_string())
462}

534    #[async_trait]
535    impl VectorTileProvider for ScriptedVectorTileProvider {
536        fn source_descriptor(&self) -> VectorTileSourceDescriptor {

540        async fn fetch_vector_tile(
541            &self,
542            tile_x: u32,
543            tile_y: u32,
544            zoom_level: u8,
545        ) -> Result<VectorTileData, SDKError> {
546            *self.fetch_count.lock().unwrap() += 1;
547            let tile_key = TileKey::new(tile_x, tile_y, zoom_level)?;
548            if self.failed_tile_keys.contains(&tile_key) {
549                return Err(SDKError::ResponseParsingFailure(format!(
550                    "the scripted provider refuses {zoom_level}/{tile_x}/{tile_y}"
551                )));
552            }
553            let tile_bytes = self
554                .served_tile_bytes
555                .get(&tile_key)
556                .cloned()
557                .ok_or_else(|| {
558                    SDKError::ResponseParsingFailure(format!(
559                        "the scripted provider serves no {zoom_level}/{tile_x}/{tile_y}"
560                    ))
561                })?;
562            VectorTileData::from_wire_bytes(tile_x, tile_y, zoom_level, tile_bytes)
563        }
564    }

677    #[test]
678    fn the_styled_cache_is_bounded_on_its_own() {
679        // One decoded tile can be styled once per zoom bucket, so the styled
680        // cache cannot rely on decoded-tile eviction alone: storing more
681        // entries than the cap drops the oldest.
682        let provider = ScriptedVectorTileProvider::serving_tile(
683            SOURDOUGH_TILE_KEY,
684            SOURDOUGH_TILE_BYTES.to_vec(),
685        );
686        let mut scheduler = VectorBaseScheduler::new(
687            Arc::clone(&provider) as Arc<dyn VectorTileProvider>,
688            "test-highways",
689            highways_style_with_width_stops(),
690        );
691        scheduler.maximum_cached_styled_tile_count = 3;
692        for style_revision in 0_u64..6 {
693            scheduler.store_styled_tile(
694                StyledTileCacheKey {
695                    tile_key: SOURDOUGH_TILE_KEY,
696                    style_revision,
697                    quantized_style_zoom: 140,
698                },
699                Vec::new(),
700            );
701        }
702        assert_eq!(
703            scheduler.styled_tiles.len(),
704            3,
705            "the styled cache never exceeds its own capacity"
706        );
707        assert!(
708            !scheduler
709                .styled_tiles
710                .keys()
711                .any(|key| key.style_revision == 0),
712            "the oldest styled entry is the one evicted"
713        );
714    }
