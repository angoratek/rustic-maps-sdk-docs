1//! GPU-free frame assembly: prepared tiles mapped to drawable quads.
2//!
3//! The pure half of the renderer. A [`Renderer`] consumes the camera pose, the
4//! viewport, and the tiles the [`crate::tile_scheduler::TileScheduler`]
5//! prepared for the frame, and produces the list of textured quads to draw in
6//! normalized world coordinates, including parent-fallback quads for tiles
7//! that are not yet pooled.
8//!
9//! Quads are projected into per-fragment clip-space vertex arrays: a world
10//! span is wrapped around the camera as one unit (never corner by corner) and
11//! split into at most one vertex array per world copy it reaches, so spans
12//! crossing the antimeridian render contiguously instead of tearing. See
13//! [`FrameProjection`].
14//!
15//! Everything in this module is free of `wgpu` types, so the quad assembly,
16//! the fallback texture window math, and the clip-space projection of the quad
17//! corners are all unit-testable without a GPU.
18

27/// Backdrop color behind the map. Opaque black: a tile that never arrives

31/// Normalized-device-coordinate bounds of the viewport in both axes. Quads
32/// whose entire projected bounding box lies outside these bounds are culled
33/// during assembly.
34pub const VIEWPORT_NORMALIZED_DEVICE_BOUNDS: [f32; 2] = [1.0, 1.0];

36/// How many world copies one tile span can reach after wrapping. A span is at
37/// most one world wide (the zoom-0 root tile), so wrapping it next to the
38/// camera-centered window can cross at most one copy boundary.
39const WORLD_COPY_FRAGMENTS_MAXIMUM: usize = 2;
40
41/// Half-width of the camera-centered world window in normalized-world units:
42/// wrapped span positions beyond it belong to the adjacent world copy, where
43/// coordinates continue from the window edge. Shared with the overlay
44/// assembly (`crate::overlay`), which splits its shapes against the same
45/// world-copy windows.
46pub(crate) const CAMERA_WORLD_WINDOW_HALF_WIDTH: f64 = 0.5;

48/// Normalized-world rectangle one tile occupies: x grows eastward and y grows
49/// southward across `[0, 1]` (web-mercator convention), so the tile grid
50/// subdivides the unit square from the north.
51#[derive(Debug, Clone, Copy, PartialEq)]
52pub struct TileWorldSpan {
53    /// Western edge (inclusive).
54    pub minimum_x: f64,
55    /// Northern edge (inclusive).
56    pub minimum_y: f64,
57    /// Eastern edge (exclusive).
58    pub maximum_x: f64,
59    /// Southern edge (exclusive).
60    pub maximum_y: f64,
61}
62
63impl TileWorldSpan {
64    /// The normalized-world span of a single tile at its own zoom level:
65    /// tile `(tx, ty, z)` covers `x ∈ [tx / 2^z, (tx + 1) / 2^z]` and the
66    /// matching y range.
67    pub fn covering_tile(tile_key: TileKey) -> Self {
68        let tiles_per_axis = 2f64.powi(i32::from(tile_key.zoom_level));
69        let tile_span = 1.0 / tiles_per_axis;
70        Self {
71            minimum_x: tile_key.tile_x as f64 * tile_span,
72            minimum_y: tile_key.tile_y as f64 * tile_span,
73            maximum_x: (tile_key.tile_x as f64 + 1.0) * tile_span,
74            maximum_y: (tile_key.tile_y as f64 + 1.0) * tile_span,
75        }
76    }
77
78    /// Horizontal extent of the span.
79    pub fn width(&self) -> f64 {
80        self.maximum_x - self.minimum_x
81    }
82
83    /// Vertical extent of the span.
84    pub fn height(&self) -> f64 {
85        self.maximum_y - self.minimum_y
86    }
87
88    /// Whether another span sits fully inside this span (boundaries
89    /// inclusive). Tile spans of an ancestor chain meet at shared edges.
90    pub fn contains(&self, other: &TileWorldSpan) -> bool {
91        self.minimum_x <= other.minimum_x
92            && self.minimum_y <= other.minimum_y
93            && other.maximum_x <= self.maximum_x
94            && other.maximum_y <= self.maximum_y
95    }
96}

98/// The sub-window of a texture that a quad samples: u grows rightward with
99/// the texture columns and v grows downward with the texture rows, so at v =
100/// 0 the image's top row — the northern edge of a tile — is sampled. All
101/// components lie in `[0, 1]`.
102#[derive(Debug, Clone, Copy, PartialEq)]
103pub struct TextureWindow {
104    /// Leftmost sampled column.
105    pub minimum_u: f64,
106    /// Topmost (northernmost) sampled row.
107    pub minimum_v: f64,
108    /// Rightmost sampled column.
109    pub maximum_u: f64,
110    /// Bottommost (southernmost) sampled row.
111    pub maximum_v: f64,
112}
113
114/// One textured quad to draw for a frame, in normalized world coordinates.
115#[derive(Debug, Clone, Copy, PartialEq)]
116pub struct TileQuad {
117    /// Tile whose pooled payload supplies the pixels. This is the requested
118    /// tile itself, or its nearest pooled ancestor for a fallback draw.
119    pub texture_tile_key: TileKey,
120    /// The normalized-world rectangle the quad covers — always the requested
121    /// tile's own span, so a fallback draws ancestor pixels at the child's
122    /// geographic position.
123    pub world_span: TileWorldSpan,
124    /// Which part of the texture the quad samples (the full texture for a
125    /// present tile; the ancestor's covering window for a fallback).
126    pub texture_window: TextureWindow,
127}
128
129impl TileQuad {
130    /// The quad covering a pooled tile's full own texture.
131    fn present_tile(tile_key: TileKey) -> Self {
132        Self {
133            texture_tile_key: tile_key,
134            world_span: TileWorldSpan::covering_tile(tile_key),
135            texture_window: TextureWindow {
136                minimum_u: 0.0,
137                minimum_v: 0.0,
138                maximum_u: 1.0,
139                maximum_v: 1.0,
140            },
141        }
142    }
143
144    /// The quad drawing a fallback ancestor's pixels over the requested
145    /// child's own span. Only the covering sub-window of the ancestor texture
146    /// is sampled, so the child's geographic footprint shows exactly the
147    /// ancestor content that covers it, and nothing outside it.
148    fn fallback_tile(requested_tile_key: TileKey, fallback_tile_key: TileKey) -> Self {
149        let requested_span = TileWorldSpan::covering_tile(requested_tile_key);
150        let fallback_span = TileWorldSpan::covering_tile(fallback_tile_key);
151        Self {
152            texture_tile_key: fallback_tile_key,
153            world_span: requested_span,
154            texture_window: TextureWindow {
155                minimum_u: (requested_span.minimum_x - fallback_span.minimum_x)
156                    / fallback_span.width(),
157                minimum_v: (requested_span.minimum_y - fallback_span.minimum_y)
158                    / fallback_span.height(),
159                maximum_u: (requested_span.maximum_x - fallback_span.minimum_x)
160                    / fallback_span.width(),
161                maximum_v: (requested_span.maximum_y - fallback_span.minimum_y)
162                    / fallback_span.height(),
163            },
164        }
165    }
166}

168/// One quad corner as consumed by the GPU pipeline: a position in clip space
169/// (homogeneous coordinates; the perspective divide happens on the GPU) and
170/// the texture coordinates to sample there. `repr(C)` keeps the field order
171/// the vertex buffer layout expects.
172#[repr(C)]
173#[derive(Debug, Clone, Copy)]
174pub struct ClipSpaceTileVertex {
175    /// Position in clip space, `[x, y, z, w]`.
176    pub clip_position: [f32; 4],
177    /// Texture-coordinate sample point.
178    pub texture_coordinates: [f32; 2],
179}
180
181/// Per-pass projection context: the view-projection matrix, the world pixel
182/// scale, and the camera's normalized-world x, built once per frame and
183/// threaded through fragment projection and culling. Rebuilding the matrix
184/// per quad was an order of magnitude of redundant `f64` matrix math.
185#[derive(Debug, Clone, Copy)]
186struct FrameProjection {
187    view_projection_matrix: DMat4,
188    zoom_scale_pixels: f64,
189    center_normalized_x: f64,
190}
191
192impl FrameProjection {
193    /// Builds the projection context for one frame.
194    fn for_frame(
195        camera: &MapCamera,
196        viewport: &ViewportPixels,
197        field_of_view_degrees: f64,
198    ) -> Self {
199        Self {
200            view_projection_matrix: camera.view_projection_matrix(viewport, field_of_view_degrees),
201            zoom_scale_pixels: zoom_scale_pixels(camera.zoom_level),
202            center_normalized_x: coordinate_to_normalized_world(&camera.center)[0],
203        }
204    }
205
206    /// Projects a quad into zero, one, or two clip-space vertex arrays — one
207    /// per world-copy fragment the span reaches around the camera's copy.
208    ///
209    /// The span wraps as one unit: the minimum corner's wrap delta (shared
210    /// with the camera, in `(-0.5, 0.5]`) fixes the span's world copy, and
211    /// every remaining x position advances continuously from it. Wrapping
212    /// corners independently tears any span crossing the copy seam, because
213    /// the far corner flicks to the opposite edge and inverts or collapses
214    /// the quad — a zoom-0 root tile spanned the whole world and collapsed to
215    /// a single point.
216    ///
217    /// Each fragment is the span's intersection with one camera-centered
218    /// world copy window and carries the proportionally split texture window,
219    /// so seams draw each copy's slice of the texture at the right screen
220    /// place. A fragment whose every corner lies behind the camera is
221    /// dropped; that is the only case losing coverage, since anything with a
222    /// corner in front still reaches viewable ground.
223    fn quad_fragments(&self, quad: &TileQuad) -> Vec<[ClipSpaceTileVertex; 4]> {
224        let span = &quad.world_span;
225        let window = &quad.texture_window;
226        let span_width = span.width();
227        let minimum_wrapped_x = self.center_normalized_x
228            + crate::camera::normalized_shortest_delta(self.center_normalized_x, span.minimum_x);
229        let wrapped_maximum_x = minimum_wrapped_x + span_width;
230
231        let mut fragments = Vec::with_capacity(WORLD_COPY_FRAGMENTS_MAXIMUM);
232        for copy_index in 0..WORLD_COPY_FRAGMENTS_MAXIMUM {
233            let copy_offset = copy_index as f64;
234            let copy_window_start =
235                self.center_normalized_x - CAMERA_WORLD_WINDOW_HALF_WIDTH + copy_offset;
236            let copy_window_end = copy_window_start + 2.0 * CAMERA_WORLD_WINDOW_HALF_WIDTH;
237            let fragment_start_x = minimum_wrapped_x.max(copy_window_start);
238            let fragment_end_x = wrapped_maximum_x.min(copy_window_end);
239            if fragment_end_x <= fragment_start_x {
240                continue;
241            }
242            // The fragment's sub-window of the texture, split linearly by its
243            // fraction of the span so the pixels keep their geographic place.
244            let start_span_fraction = (fragment_start_x - minimum_wrapped_x) / span_width;
245            let end_span_fraction = (fragment_end_x - minimum_wrapped_x) / span_width;
246            let u_range = window.maximum_u - window.minimum_u;
247            let fragment_window = TextureWindow {
248                minimum_u: window.minimum_u + u_range * start_span_fraction,
249                minimum_v: window.minimum_v,
250                maximum_u: window.minimum_u + u_range * end_span_fraction,
251                maximum_v: window.maximum_v,
252            };
253            let fragment = [
254                (
255                    fragment_start_x,
256                    span.minimum_y,
257                    fragment_window.minimum_u,
258                    fragment_window.minimum_v,
259                ),
260                (
261                    fragment_end_x,
262                    span.minimum_y,
263                    fragment_window.maximum_u,
264                    fragment_window.minimum_v,
265                ),
266                (
267                    fragment_start_x,
268                    span.maximum_y,
269                    fragment_window.minimum_u,
270                    fragment_window.maximum_v,
271                ),
272                (
273                    fragment_end_x,
274                    span.maximum_y,
275                    fragment_window.maximum_u,
276                    fragment_window.maximum_v,
277                ),
278            ]
279            .map(|(wrapped_x, world_y, texture_u, texture_v)| {
280                // Camera world space is y-north pixel space (see
281                // `MapCamera::ground_position_vector`); copy `copy_index`
282                // continues coordinates past the camera window by subtracting
283                // the copy offset, placing the span's wrapped remainder at the
284                // screen edge the wrap points to.
285                let x_pixels = (wrapped_x - copy_offset) * self.zoom_scale_pixels;
286                let y_pixels = -world_y * self.zoom_scale_pixels;
287                let clip_position =
288                    self.view_projection_matrix * DVec4::new(x_pixels, y_pixels, 0.0, 1.0);
289                ClipSpaceTileVertex {
290                    clip_position: [
291                        clip_position.x as f32,
292                        clip_position.y as f32,
293                        clip_position.z as f32,
294                        clip_position.w as f32,
295                    ],
296                    texture_coordinates: [texture_u as f32, texture_v as f32],
297                }
298            });
299            if fragment.iter().all(|vertex| vertex.clip_position[3] <= 0.0) {
300                continue;
301            }
302            fragments.push(fragment);
303        }
304        fragments
305    }
306
307    /// Whether any part of a quad's fragments reaches the viewport.
308    fn quad_reaches_viewport(&self, quad: &TileQuad) -> bool {
309        self.quad_fragments(quad)
310            .iter()
311            .any(fragment_reaches_viewport)
312    }
313}

315/// Whether any part of one fragment's projected area reaches the viewport.
316///
317/// Any corner with `w <= 0` disables culling for the fragment. Corners behind
318/// the near plane divide (per-corner, before the GPU's homogeneous clip) into
319/// coordinates far outside the bounds, and the fragment's genuinely visible
320/// ground strip — the part the GPU's exact frustum clip cuts into view — can
321/// sit inside the viewport while every surviving corner's divided bounding
322/// box does not. The measured failure band at a 75-degree field of view is a
323/// few degrees wide around 32 degrees of pitch; culling such a strip left a
324/// visible map hole. A conservative `true` costs an extra draw the GPU clips.
325fn fragment_reaches_viewport(fragment: &[ClipSpaceTileVertex; 4]) -> bool {
326    if fragment.iter().any(|vertex| vertex.clip_position[3] <= 0.0) {
327        return true;
328    }
329    let (mut minimum_screen_x, mut maximum_screen_x) = (f32::INFINITY, f32::NEG_INFINITY);
330    let (mut minimum_screen_y, mut maximum_screen_y) = (f32::INFINITY, f32::NEG_INFINITY);
331    for vertex in fragment {
332        let clip_position = vertex.clip_position;
333        // Every corner is in front (checked above), so the divide is sound.
334        let screen_x = clip_position[0] / clip_position[3];
335        let screen_y = clip_position[1] / clip_position[3];
336        minimum_screen_x = minimum_screen_x.min(screen_x);
337        maximum_screen_x = maximum_screen_x.max(screen_x);
338        minimum_screen_y = minimum_screen_y.min(screen_y);
339        maximum_screen_y = maximum_screen_y.max(screen_y);
340    }
341    let [bound_x, bound_y] = VIEWPORT_NORMALIZED_DEVICE_BOUNDS;
342    minimum_screen_x <= bound_x
343        && maximum_screen_x >= -bound_x
344        && minimum_screen_y <= bound_y
345        && maximum_screen_y >= -bound_y
346}

348/// Quad assembler for one frame: turns prepared tiles into draw items and
349/// projects their corners into clip space. Construct it once and reuse it
350/// across frames; it holds only the field of view.
351#[derive(Debug, Clone, Copy, PartialEq)]
352pub struct Renderer {
353    field_of_view_degrees: f64,
354}
355
356impl Default for Renderer {
357    fn default() -> Self {
358        Self {
359            field_of_view_degrees: crate::camera::DEFAULT_FIELD_OF_VIEW_DEGREES,
360        }
361    }
362}
363
364impl Renderer {
365    /// Builds a quad assembler with a non-default field of view.
366    pub fn new(field_of_view_degrees: f64) -> Self {
367        Self {
368            field_of_view_degrees,
369        }
370    }
371
372    /// The field of view every frame assembly of this renderer assumes; the
373    /// overlay assembly of the same frame needs the same projection pose.
374    pub fn field_of_view_degrees(&self) -> f64 {
375        self.field_of_view_degrees
376    }
377
378    /// Assembles the frame's textured quads, culled to the viewport.
379    ///
380    /// `Present` tiles contribute a quad with their full texture; tiles
381    /// resolved as [`TileReadiness::FallsBackTo`] contribute a quad over the
382    /// requested tile's own span sampling the nearest pooled ancestor's
383    /// texture. `Unavailable` and `Deferred` tiles draw nothing this frame —
384    /// the scheduler will resolve them in a later frame. A pooled payload
385    /// that left the pool between scheduling and assembly — or one the GPU
386    /// stage later finds undecodable, which it skips and evicts so the
387    /// scheduler refetches (see the headless renderer) — draws nothing rather
388    /// than guessing.
389    ///
390    /// Culling is per fragment and conservative around the near plane; see
391    /// [`FrameProjection::quad_fragments`]. The view-projection matrix is
392    /// built once per frame and threaded through projection and culling.
393    pub fn frame_quads(
394        &self,
395        camera: &MapCamera,
396        viewport: &ViewportPixels,
397        prepared_tiles: &[PreparedTile],
398        pool: &TileTexturePool,
399    ) -> Vec<TileQuad> {
400        let frame_projection =
401            FrameProjection::for_frame(camera, viewport, self.field_of_view_degrees);
402        prepared_tiles
403            .iter()
404            .filter_map(|prepared_tile| {
405                match prepared_tile.readiness {
406                    TileReadiness::Present if pool.contains(&prepared_tile.requested_tile_key) => {
407                        Some(TileQuad::present_tile(prepared_tile.requested_tile_key))
408                    }
409                    TileReadiness::FallsBackTo {
410                        fallback_tile_key,
411                        zoom_level_difference,
412                    } if pool.contains(&fallback_tile_key) => {
413                        debug_assert_eq!(
414                            u32::from(fallback_tile_key.zoom_level)
415                                + u32::from(zoom_level_difference),
416                            u32::from(prepared_tile.requested_tile_key.zoom_level)
417                        );
418                        Some(TileQuad::fallback_tile(
419                            prepared_tile.requested_tile_key,
420                            fallback_tile_key,
421                        ))
422                    }
423                    TileReadiness::Present
424                    | TileReadiness::FallsBackTo { .. }
425                    | TileReadiness::Unavailable
426                    | TileReadiness::Deferred => None,
427                }
428                .filter(|quad| frame_projection.quad_reaches_viewport(quad))
429            })
430            .collect()
431    }
432
433    /// Projects every quad of a frame into per-fragment vertex arrays,
434    /// preserving quad order: entry `i` holds the 0, 1, or 2 world-copy
435    /// fragments of quad `i`. Pairs with [`Self::frame_quads`], whose quads
436    /// are already viewport-culled. The view-projection matrix is built once.
437    pub fn quad_fragment_vertices(
438        &self,
439        camera: &MapCamera,
440        viewport: &ViewportPixels,
441        quads: &[TileQuad],
442    ) -> Vec<Vec<[ClipSpaceTileVertex; 4]>> {
443        let frame_projection =
444            FrameProjection::for_frame(camera, viewport, self.field_of_view_degrees);
445        quads
446            .iter()
447            .map(|quad| frame_projection.quad_fragments(quad))
448            .collect()
449    }
450
451    /// Flattens [`Self::quad_fragment_vertices`] into one vertex array per
452    /// fragment, dropping quads that lie entirely behind the camera.
453    pub fn quad_vertices(
454        &self,
455        camera: &MapCamera,
456        viewport: &ViewportPixels,
457        quads: &[TileQuad],
458    ) -> Vec<[ClipSpaceTileVertex; 4]> {
459        self.quad_fragment_vertices(camera, viewport, quads)
460            .into_iter()
461            .flatten()
462            .collect()
463    }
464
465    /// Convenience variant of [`Self::quad_fragment_vertices`] for a single
466    /// quad; it rebuilds the projection per call, so frame-scale callers pass
467    /// through the batched methods instead.
468    pub fn tile_vertices_in_clip_space(
469        &self,
470        camera: &MapCamera,
471        viewport: &ViewportPixels,
472        quad: &TileQuad,
473    ) -> Vec<[ClipSpaceTileVertex; 4]> {
474        self.quad_fragment_vertices(camera, viewport, std::slice::from_ref(quad))
475            .into_iter()
476            .next()
477            .unwrap_or_default()
478    }
479}

546    /// Divides a vertex's clip coordinates to its normalized-device x/y.
547    fn divided_position(vertex: &super::ClipSpaceTileVertex) -> [f64; 2] {
548        [
549            f64::from(vertex.clip_position[0]) / f64::from(vertex.clip_position[3]),
550            f64::from(vertex.clip_position[1]) / f64::from(vertex.clip_position[3]),

554    /// Asserts a fragment's corners sit at the expected normalized-device
555    /// positions and sample the expected texture coordinates.
556    fn assert_fragment(
557        fragment: &[super::ClipSpaceTileVertex; 4],
558        expected_positions: [[f64; 2]; 4],
559        expected_uvs: [[f64; 2]; 4],
560        context: &str,
561    ) {
562        for (corner_index, vertex) in fragment.iter().enumerate() {
563            let divided = divided_position(vertex);
564            assert!(
565                (divided[0] - expected_positions[corner_index][0]).abs() < CLIP_POSITION_TOLERANCE
566                    && (divided[1] - expected_positions[corner_index][1]).abs()
567                        < CLIP_POSITION_TOLERANCE,
568                "{context}: corner {corner_index} divided {divided:?} vs {expected:?}",

797    #[test]
798    fn the_root_tile_fills_the_viewport_across_the_world_copy_seam() {
799        // Regression: wrapping each corner independently collapsed the
800        // zoom-0 root tile hovered from x = 0.3 — the east corner wrapped
801        // back onto the west corner's position, degenerating a whole-world
802        // span to one point. The span must wrap as a unit at its minimum
803        // corner and split into the two world-copy fragments that together
804        // cover the viewport, east content forward (never mirrored).
805        let renderer = Renderer::default();
806        let camera = camera_above([0.3, 0.5], 0.0);
807        let viewport = square_viewport();
808        let root_quad = TileQuad::present_tile(tile_key(0, 0, 0));
809        let fragments = renderer.tile_vertices_in_clip_space(&camera, &viewport, &root_quad);
810        assert_eq!(
811            fragments.len(),
812            2,
813            "the root span reaches two world copies around the camera"
814        );
815        // Fragment 0 is the camera's own copy: the span's x [0, 0.8) at
816        // normalized device x [-0.6, 1.0]; fragment 1 continues the wrapped
817        // remainder [0.8, 1.0] at device x [-1.0, -0.6]. Corner order as
818        // everywhere else: northwest, northeast, southwest, southeast.
819        let expected_positions = [
820            [[-0.6, 1.0], [1.0, 1.0], [-0.6, -1.0], [1.0, -1.0]],
821            [[-1.0, 1.0], [-0.6, 1.0], [-1.0, -1.0], [-0.6, -1.0]],
822        ];
823        let expected_uvs = [
824            [[0.0, 0.0], [0.8, 0.0], [0.0, 1.0], [0.8, 1.0]],
825            [[0.8, 0.0], [1.0, 0.0], [0.8, 1.0], [1.0, 1.0]],
826        ];
827        for (fragment_index, fragment) in fragments.iter().enumerate() {
828            for vertex in fragment {
829                assert!(vertex.clip_position[3] > 0.0, "corner behind the camera");
830            }
831            assert_fragment(
832                fragment,
833                expected_positions[fragment_index],
834                expected_uvs[fragment_index],
835                "root tile fragment {fragment_index}",
836            );
837            // Non-mirrored orientation: texture u and device x must grow
838            // together across each fragment's west and east corners.
839            let [west, east] = [
840                divided_position(&fragment[0]),
841                divided_position(&fragment[1]),
842            ];
843            assert!(
844                east[0] > west[0],
845                "fragment {fragment_index} drew eastward texture content westward"
846            );
847        }
848        // The two fragments together cover the whole viewport horizontally.
849        let mut fragments_minimum_x = f64::INFINITY;
850        let mut fragments_maximum_x = f64::NEG_INFINITY;
851        for fragment in &fragments {
852            for vertex in fragment {
853                let divided = divided_position(vertex);
854                fragments_minimum_x = fragments_minimum_x.min(divided[0]);
855                fragments_maximum_x = fragments_maximum_x.max(divided[0]);
856            }
857        }
858        assert!((fragments_minimum_x + 1.0).abs() < CLIP_POSITION_TOLERANCE);
859        assert!((fragments_maximum_x - 1.0).abs() < CLIP_POSITION_TOLERANCE);
860        // The quad survives frame assembly, feeding both fragments downstream.
861        let mut payload_pool = TileTexturePool::new(1 << 20);
862        payload_pool.insert(tile_key(0, 0, 0), vec![1]);
863        let quads = renderer.frame_quads(
864            &camera,
865            &viewport,
866            &prepared(tile_key(0, 0, 0), TileReadiness::Present),
867            &payload_pool,
868        );
869        assert_eq!(quads.len(), 1);
870        assert_eq!(
871            renderer.quad_vertices(&camera, &viewport, &quads).len(),
872            2,
873            "both fragments reach the vertex stage"
874        );
875    }

921    #[test]
922    fn a_near_plane_straddling_tile_is_not_culled_out_of_view() {
923        // Regression for bounding-box culling over divided coordinates: a
924        // tile straddling the near plane keeps genuinely visible ground inside
925        // the viewport while every surviving corner may divide far outside one
926        // screen edge, so the old per-fragment bounding-box test dropped it.
927        // For every swept pitch, any neighborhood tile with a sample-projected
928        // ground point inside the viewport must survive assembly.
929        //
930        // The pixel geometry is zoom-invariant (tiles are 512 pixels and the
931        // camera altitude is fixed), so one zoom covers the mechanism; the
932        // default 60-degree field of view keeps straddling regions outside the
933        // frustum, so the sweep runs at the wider field of view where the
934        // defect measured.
935        let renderer = Renderer::new(CULL_BAND_FIELD_OF_VIEW_DEGREES);
936        let viewport = square_viewport();
937        let tiles_per_axis = 2f64.powi(i32::from(CULL_BAND_ZOOM_LEVEL));
938        let camera_phase_offset = 0.5 / 8.0 / tiles_per_axis;
939        let anchor_world = [0.5 + camera_phase_offset, 0.5 + camera_phase_offset];
940        for pitch_step in 0..CULL_BAND_PITCH_STEPS {
941            let pitch_degrees = CULL_BAND_MINIMUM_PITCH_DEGREES
942                + f64::from(pitch_step) * CULL_BAND_PITCH_STEP_DEGREES;
943            let camera = MapCamera::new(
944                normalized_world_to_coordinate(anchor_world),
945                f64::from(CULL_BAND_ZOOM_LEVEL),
946                0.0,
947                pitch_degrees,
948            );
949            let anchor_tile = tile_key_for_coordinate(&camera.center, CULL_BAND_ZOOM_LEVEL);
950            let zoom_tile_maximum = (1u32 << CULL_BAND_ZOOM_LEVEL) - 1;
951            let mut neighborhood_tiles = Vec::new();
952            for delta_x in -CULL_BAND_TILE_NEIGHBORHOOD..=CULL_BAND_TILE_NEIGHBORHOOD {
953                for delta_y in -CULL_BAND_TILE_NEIGHBORHOOD..=CULL_BAND_TILE_NEIGHBORHOOD {
954                    let tile_x = (i64::from(anchor_tile.tile_x) + delta_x)
955                        .clamp(0, i64::from(zoom_tile_maximum));
956                    let tile_y = (i64::from(anchor_tile.tile_y) + delta_y)
957                        .clamp(0, i64::from(zoom_tile_maximum));
958                    neighborhood_tiles.push(
959                        TileKey::new(tile_x as u32, tile_y as u32, anchor_tile.zoom_level).unwrap(),
960                    );
961                }
962            }
963            let mut payload_pool = TileTexturePool::new(1 << 20);
964            for tile_key in &neighborhood_tiles {
965                payload_pool.insert(*tile_key, vec![1, 2, 3]);
966            }
967            let prepared_tiles = neighborhood_tiles
968                .iter()
969                .map(|tile_key| PreparedTile {
970                    requested_tile_key: *tile_key,
971                    readiness: TileReadiness::Present,
972                })
973                .collect::<Vec<PreparedTile>>();
974            let quads = renderer.frame_quads(&camera, &viewport, &prepared_tiles, &payload_pool);
975            for tile_key in &neighborhood_tiles {
976                let span = TileWorldSpan::covering_tile(*tile_key);
977                let mut has_inside_viewport_sample = false;
978                for sample_column in 0..CULL_BAND_SAMPLES_PER_TILE_EDGE {
979                    for sample_row in 0..CULL_BAND_SAMPLES_PER_TILE_EDGE {
980                        let sample_x = f64::from(sample_column as u32 + 1)
981                            / f64::from(CULL_BAND_SAMPLES_PER_TILE_EDGE as u32 + 1);
982                        let sample_y = f64::from(sample_row as u32 + 1)
983                            / f64::from(CULL_BAND_SAMPLES_PER_TILE_EDGE as u32 + 1);
984                        let sample_world = [
985                            span.minimum_x + sample_x * span.width(),
986                            span.minimum_y + sample_y * span.height(),
987                        ];
988                        let sample_ground = normalized_world_to_coordinate(sample_world);
989                        if let Some(screen_position) = camera.project_to_screen(
990                            &sample_ground,
991                            &viewport,
992                            CULL_BAND_FIELD_OF_VIEW_DEGREES,
993                        ) {
994                            if screen_position[0] >= 0.0
995                                && screen_position[0] <= viewport.width_pixels
996                                && screen_position[1] >= 0.0
997                                && screen_position[1] <= viewport.height_pixels
998                            {
999                                has_inside_viewport_sample = true;
1000                            }
1001                        }
1002                    }
1003                }
1004                if has_inside_viewport_sample {
1005                    assert!(
1006                        quads.iter().any(|quad| {
1007                            quad.texture_tile_key == *tile_key && quad.world_span == span
1008                        }),
1009                        "pitch {pitch_degrees}: tile {tile_key:?} projects sample ground \
1010                         points inside the viewport but assembly culled it"
1011                    );
1012                }
1013            }
1014        }
1015    }
