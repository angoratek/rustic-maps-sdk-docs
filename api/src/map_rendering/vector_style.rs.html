1//! Minimal vector-tile styling (M9, the fill/line/circle slice): a decoded
2//! [`map_vector_tiles::VectorTile`] becomes the renderer's existing overlay
3//! primitives — filled polygons for polygon features, polylines for line
4//! features, point markers for point features — under a small explicit style.
5//!
6//! The style is deliberately data-only: a layer entry names the source layer,
7//! the geometry kind it paints, an optional one-attribute text filter and the
8//! paint values (RGBA, line width, marker radius). No expression language, no
9//! per-symbol layout, no glyph atlas — those are later slices. The committed
10//! artifact [`DEFAULT_VECTOR_STYLE_ARTIFACT_JSON`] carries the default style;
11//! [`default_vector_tile_style`] parses it and [`in_code_default_vector_tile_style`]
12//! is the documented fallback when the artifact is missing or malformed.
13//!
14//! Polygon features paint every exterior ring with its interior rings
15//! subtracted through the overlay's hole-aware fill (see
16//! [`crate::overlay::FilledPolygonOverlay`]), and the conversion groups its
17//! primitives into one [`VectorOverlayBatch`] per style layer, the seam a
18//! dedicated vector pipeline would consume later.
19//!
20//! Coordinates convert from tile-local extent space (`x` right, `y` down, the
21//! MVT convention) through the tile's web-mercator address to geographic
22//! degrees, so the overlays project exactly like every other map coordinate.
23

83/// One style layer: source layer, painted kind, optional filter and paint,
84/// plus the zoom span and width stops the layer is evaluated at.
85#[derive(Debug, Clone, PartialEq)]
86pub struct VectorLayerStyle {
87    /// The decoded layer name the style paints (`water`, `highways`, …).
88    pub source_layer: String,
89    /// The geometry kind the style paints.
90    pub geometry_kind: VectorStyleGeometryKind,
91    /// An optional one-attribute text filter.
92    pub filter: Option<VectorAttributeFilter>,
93    /// Fill RGBA (used by [`VectorStyleGeometryKind::Fill`]).
94    pub fill_rgba: [f32; 4],
95    /// Line RGBA (used by [`VectorStyleGeometryKind::Line`]).
96    pub line_rgba: [f32; 4],
97    /// Line width in screen pixels.
98    pub line_width_pixels: f64,
99    /// Marker RGBA (used by [`VectorStyleGeometryKind::Circle`]).
100    pub circle_rgba: [f32; 4],
101    /// Marker radius in screen pixels.
102    pub circle_radius_pixels: f64,
103    /// The lowest zoom level (inclusive) the layer paints at; below it the
104    /// layer is invisible, exactly like a MapLibre layer's `minzoom`.
105    pub min_zoom_level: f64,
106    /// The highest zoom level (exclusive) the layer paints at; at or above it
107    /// the layer is invisible (MapLibre's `maxzoom`).
108    pub max_zoom_level: f64,
109    /// The layer's width stops, ascending by zoom. Empty keeps
110    /// `line_width_pixels` at every zoom.
111    pub width_zoom_stops: Vec<VectorWidthZoomStop>,
112}
113
114impl VectorLayerStyle {
115    /// Builds a fill layer with the given paint.
116    pub fn fill(source_layer: impl Into<String>, rgba_color: [f32; 4]) -> Self {
117        Self::base(source_layer, VectorStyleGeometryKind::Fill, rgba_color)
118    }
119
120    /// Builds a line layer with the given paint and width.
121    pub fn line(source_layer: impl Into<String>, rgba_color: [f32; 4], width_pixels: f64) -> Self {
122        let mut layer_style = Self::base(source_layer, VectorStyleGeometryKind::Line, rgba_color);
123        layer_style.line_rgba = rgba_color;
124        layer_style.line_width_pixels = width_pixels;
125        layer_style
126    }
127
128    /// Builds a circle layer with the given paint and radius.
129    pub fn circle(
130        source_layer: impl Into<String>,
131        rgba_color: [f32; 4],
132        radius_pixels: f64,
133    ) -> Self {
134        let mut layer_style = Self::base(source_layer, VectorStyleGeometryKind::Circle, rgba_color);
135        layer_style.circle_rgba = rgba_color;
136        layer_style.circle_radius_pixels = radius_pixels;
137        layer_style
138    }
139
140    /// Sets the layer's zoom span (MapLibre `minzoom`/`maxzoom`): the layer
141    /// paints only while `min_zoom_level <= zoom < max_zoom_level`.
142    pub fn with_zoom_span(mut self, min_zoom_level: f64, max_zoom_level: f64) -> Self {
143        self.min_zoom_level = min_zoom_level;
144        self.max_zoom_level = max_zoom_level;
145        self
146    }
147
148    /// Adds a width stop (see [`VectorWidthZoomStop`]); stops are kept
149    /// ascending by zoom level.
150    pub fn with_width_zoom_stop(mut self, zoom_level: f64, width_pixels: f64) -> Self {
151        self.width_zoom_stops.push(VectorWidthZoomStop {
152            zoom_level,
153            width_pixels,
154        });
155        self.width_zoom_stops
156            .sort_by(|first, second| first.zoom_level.total_cmp(&second.zoom_level));
157        self
158    }
159
160    /// Whether the layer paints at `zoom_level`.
161    pub fn is_visible_at_zoom(&self, zoom_level: f64) -> bool {
162        zoom_level >= self.min_zoom_level && zoom_level < self.max_zoom_level
163    }
164
165    /// The line width the layer paints at `zoom_level`: the fixed width when
166    /// the layer carries no stops, the linearly interpolated stop value
167    /// between two stops, or the nearest stop's value outside the stop range.
168    pub fn width_pixels_at_zoom(&self, zoom_level: f64) -> f64 {
169        let Some(first_stop) = self.width_zoom_stops.first() else {
170            return self.line_width_pixels;
171        };
172        if zoom_level <= first_stop.zoom_level {
173            return first_stop.width_pixels;
174        }
175        for stop_pair in self.width_zoom_stops.windows(2) {
176            let [lower_stop, upper_stop] = [stop_pair[0], stop_pair[1]];
177            if zoom_level < upper_stop.zoom_level {
178                let span = upper_stop.zoom_level - lower_stop.zoom_level;
179                if span <= 0.0 {
180                    return upper_stop.width_pixels;
181                }
182                let fraction = (zoom_level - lower_stop.zoom_level) / span;
183                return lower_stop.width_pixels
184                    + (upper_stop.width_pixels - lower_stop.width_pixels) * fraction;
185            }
186        }
187        self.width_zoom_stops
188            .last()
189            .map(|last_stop| last_stop.width_pixels)
190            .unwrap_or(self.line_width_pixels)
191    }
192
193    /// Adds the one-attribute text filter.
194    pub fn with_filter(
195        mut self,
196        attribute_name: impl Into<String>,
197        expected_text: impl Into<String>,
198    ) -> Self {
199        self.filter = Some(VectorAttributeFilter {
200            attribute_name: attribute_name.into(),
201            expected_text: expected_text.into(),
202        });
203        self
204    }
205
206    fn base(
207        source_layer: impl Into<String>,
208        geometry_kind: VectorStyleGeometryKind,
209        rgba_color: [f32; 4],
210    ) -> Self {
211        Self {
212            source_layer: source_layer.into(),
213            geometry_kind,
214            filter: None,
215            fill_rgba: rgba_color,
216            line_rgba: rgba_color,
217            line_width_pixels: DEFAULT_VECTOR_LINE_WIDTH_PIXELS,
218            circle_rgba: rgba_color,
219            circle_radius_pixels: DEFAULT_VECTOR_CIRCLE_RADIUS_PIXELS,
220            min_zoom_level: DEFAULT_VECTOR_STYLE_MINIMUM_ZOOM_LEVEL,
221            max_zoom_level: DEFAULT_VECTOR_STYLE_MAXIMUM_ZOOM_LEVEL,
222            width_zoom_stops: Vec::new(),
223        }
224    }
225}

324/// Parses the committed style artifact a style id names. An unknown id and a
325/// malformed artifact both report the typed artifact error rather than a
326/// guess; only the default style has an in-code fallback.
327pub fn vector_tile_style_from_id(
328    style_id: &str,
329) -> Result<VectorTileStyle, VectorStyleArtifactError> {
330    let Some(artifact_json) = vector_style_artifact_json(style_id) else {
331        return Err(VectorStyleArtifactError {
332            detail: format!(
333                "no committed vector style carries the id {style_id:?}; known ids are \
334                 {VECTOR_STYLE_ID_CARTO_LIGHT}, {VECTOR_STYLE_ID_CARTO_DARK}, \
335                 {VECTOR_STYLE_ID_OPENMAPTILES_LIGHT}, {VECTOR_STYLE_ID_OPENMAPTILES_DARK}, \
336                 {VECTOR_STYLE_ID_BRIGHT_LIGHT}, {VECTOR_STYLE_ID_FIORD_DARK}, \
337                 {VECTOR_STYLE_ID_POSITRON_LIGHT}"
338            ),
339        });
340    };
341    vector_tile_style_from_json(artifact_json)
342}

612/// Parses a data-only style artifact into a [`VectorTileStyle`].
613///
614/// The artifact names no providers and carries no URLs: a layer entry has a
615/// `source_layer`, a `geometry_kind` (`fill`, `line` or `circle`), an `rgba`
616/// paint, an optional one-attribute `filter`, and the kind's `width_pixels`
617/// or `radius_pixels` (the documented defaults apply when omitted).
618pub fn vector_tile_style_from_json(
619    artifact_json: &str,
620) -> Result<VectorTileStyle, VectorStyleArtifactError> {
621    let artifact: VectorStyleArtifact =
622        serde_json::from_str(artifact_json).map_err(|parse_error| VectorStyleArtifactError {
623            detail: parse_error.to_string(),
624        })?;
625    Ok(VectorTileStyle {
626        background: artifact.background,
627        layers: artifact
628            .layers
629            .into_iter()
630            .map(VectorStyleArtifactLayer::into_layer_style)
631            .collect(),
632    })
633}

635/// The style constructed in code, the documented fallback when the artifact
636/// is missing or malformed. It mirrors
637/// [`DEFAULT_VECTOR_STYLE_ARTIFACT_JSON`] so the two paths paint the same
638/// demo, which a test pins.
639pub fn in_code_default_vector_tile_style() -> VectorTileStyle {
640    let mut layers = Vec::new();
641    // Ground cover first: land use, natural areas and leisure surfaces paint
642    // under everything else, in the artifact's filtered-fill order.
643    for area_fill in FALLBACK_AREA_FILLS {
644        layers.push(
645            VectorLayerStyle::fill(area_fill.source_layer, area_fill.rgba_color)
646                .with_filter(area_fill.attribute_name, area_fill.expected_text),
647        );
648    }
649    layers.push(VectorLayerStyle::fill("water", FALLBACK_WATER_FILL_RGBA));
650    layers.push(VectorLayerStyle::line(
651        "waterways",
652        FALLBACK_WATER_FILL_RGBA,
653        FALLBACK_WATERWAY_LINE_WIDTH_PIXELS,
654    ));
655    layers.push(VectorLayerStyle::fill(
656        "buildings",
657        FALLBACK_BUILDING_FILL_RGBA,
658    ));
659    layers.push(VectorLayerStyle::line(
660        "railways",
661        FALLBACK_RAILWAY_LINE_RGBA,
662        FALLBACK_RAILWAY_LINE_WIDTH_PIXELS,
663    ));
664    // Road casings paint as one block under every fill, so intersecting
665    // roads read as continuous ribbons instead of cased-on-cased seams.
666    for road_tier in FALLBACK_ROAD_TIERS {
667        let Some(casing_rgba_color) = road_tier.casing_rgba_color else {
668            continue;
669        };
670        for highway_class in road_tier.highway_classes {
671            layers.push(
672                VectorLayerStyle::line(
673                    "highways",
674                    casing_rgba_color,
675                    road_tier.casing_width_pixels,
676                )
677                .with_filter("highway", *highway_class),
678            );
679        }
680    }
681    for road_tier in FALLBACK_ROAD_TIERS {
682        for highway_class in road_tier.highway_classes {
683            layers.push(
684                VectorLayerStyle::line(
685                    "highways",
686                    road_tier.fill_rgba_color,
687                    road_tier.fill_width_pixels,
688                )
689                .with_filter("highway", *highway_class),
690            );
691        }
692    }
693    layers.push(VectorLayerStyle::line(
694        "boundaries",
695        FALLBACK_BOUNDARY_LINE_RGBA,
696        FALLBACK_BOUNDARY_LINE_WIDTH_PIXELS,
697    ));
698    layers.push(VectorLayerStyle::circle(
699        "places",
700        FALLBACK_PLACE_CIRCLE_RGBA,
701        FALLBACK_PLACE_CIRCLE_RADIUS_PIXELS,
702    ));
703    VectorTileStyle {
704        background: None,
705        layers,
706    }
707}

768/// One artifact width stop of a line layer.

778/// The artifact's geometry-kind spelling (`fill`, `line`, `circle`).
779#[derive(Debug, Deserialize)]

781enum VectorStyleArtifactGeometryKind {
782    /// Polygon features, painted as holes-aware filled rings.
783    Fill,

785    Line,
786    /// Point features, painted as screen-facing markers.

790/// The artifact's one-attribute text filter.
791#[derive(Debug, Deserialize)]
792#[serde(deny_unknown_fields)]
793struct VectorStyleArtifactFilter {
794    /// The attribute key to match.
795    attribute_name: String,
796    /// The exact text value to match.
797    expected_text: String,
798}

800impl VectorStyleArtifactLayer {
801    /// Builds the renderer's layer style from the artifact entry.
802    fn into_layer_style(self) -> VectorLayerStyle {
803        let mut layer_style = match self.geometry_kind {
804            VectorStyleArtifactGeometryKind::Fill => {
805                VectorLayerStyle::fill(self.source_layer, self.rgba)
806            }
807            VectorStyleArtifactGeometryKind::Line => VectorLayerStyle::line(
808                self.source_layer,
809                self.rgba,
810                self.width_pixels
811                    .unwrap_or(DEFAULT_VECTOR_LINE_WIDTH_PIXELS),
812            ),
813            VectorStyleArtifactGeometryKind::Circle => VectorLayerStyle::circle(
814                self.source_layer,
815                self.rgba,
816                self.radius_pixels
817                    .unwrap_or(DEFAULT_VECTOR_CIRCLE_RADIUS_PIXELS),
818            ),
819        };
820        if let Some(filter) = self.filter {
821            layer_style = layer_style.with_filter(filter.attribute_name, filter.expected_text);
822        }
823        layer_style = layer_style.with_zoom_span(
824            self.min_zoom
825                .unwrap_or(DEFAULT_VECTOR_STYLE_MINIMUM_ZOOM_LEVEL),
826            self.max_zoom
827                .unwrap_or(DEFAULT_VECTOR_STYLE_MAXIMUM_ZOOM_LEVEL),
828        );
829        for width_stop in self.width_zoom_stops {
830            layer_style =
831                layer_style.with_width_zoom_stop(width_stop.zoom, width_stop.width_pixels);
832        }
833        layer_style
834    }
835}

837/// Converts one tile-local extent point into a geographic coordinate through
838/// the tile's web-mercator address. Vertices at the extent boundary are legal
839/// (they land on the next tile's edge); the latitude clamp keeps the projection
840/// finite at the world's polar edges.
841pub fn tile_local_point_to_coordinate(
842    tile_address: VectorTileAddress,
843    layer_extent: u64,
844    point: &VectorTileGeometryPoint,
845) -> GeographicCoordinate {
846    let extent = layer_extent.max(1) as f64;
847    let tiles_per_axis = (1u64 << tile_address.zoom_level) as f64;
848    let world_x =
849        (f64::from(tile_address.tile_x) + point.x_coordinate as f64 / extent) / tiles_per_axis;
850    let world_y =
851        (f64::from(tile_address.tile_y) + point.y_coordinate as f64 / extent) / tiles_per_axis;
852    let longitude_degrees = world_x * WORLD_LONGITUDE_SPAN_DEGREES + WORLD_WEST_LONGITUDE_DEGREES;
853    let latitude_degrees = (PI * (1.0 - 2.0 * world_y)).sinh().atan().to_degrees();
854    GeographicCoordinate {
855        latitude: latitude_degrees.clamp(
856            -MERCATOR_MAXIMUM_LATITUDE_DEGREES,
857            MERCATOR_MAXIMUM_LATITUDE_DEGREES,
858        ),
859        longitude: longitude_degrees,
860    }
861}

863/// Converts one decoded tile into one overlay batch per style layer that
864/// painted at least one feature.
865///
866/// A layer style paints only features of its declared geometry kind whose
867/// filter (when set) matches. Polygon features contribute one filled overlay
868/// per exterior ring (positive shoelace area in the tile's `y`-down space),
869/// carrying that exterior's following interior rings (negative area) as the
870/// fill's subtracted holes. Every feature keeps its own overlay primitive, so
871/// per-feature geometry stays representable; the batch is the style grouping
872/// a dedicated vector pipeline would consume.
873pub fn overlay_primitives_for_vector_tile(
874    tile: &VectorTile,
875    tile_address: VectorTileAddress,
876    style: &VectorTileStyle,
877) -> Vec<VectorOverlayBatch> {
878    overlay_primitives_for_vector_tile_at_zoom(
879        tile,
880        tile_address,
881        style,
882        f64::from(tile_address.zoom_level),
883    )
884}

886/// The same conversion as [`overlay_primitives_for_vector_tile`], with the
887/// style's zoom span and width stops evaluated at an explicit `style_zoom_level`
888/// rather than at the tile's own zoom level.
889///
890/// The base-map scheduler styles a cached tile at the camera's fractional zoom
891/// level, so a pan across a width stop restyles the already-fetched tile
892/// instead of fetching it again; the tile's address still converts every
893/// coordinate, because the geometry belongs to the tile, not to the zoom the
894/// paint is evaluated at.
895pub fn overlay_primitives_for_vector_tile_at_zoom(
896    tile: &VectorTile,
897    tile_address: VectorTileAddress,
898    style: &VectorTileStyle,
899    style_zoom_level: f64,
900) -> Vec<VectorOverlayBatch> {
901    let mut batches = Vec::new();
902    for layer_style in &style.layers {
903        let overlays = overlays_for_layer_style(tile, tile_address, layer_style, style_zoom_level);
904        if !overlays.is_empty() {
905            batches.push(VectorOverlayBatch {
906                layer_style: layer_style.clone(),
907                overlays,
908            });
909        }
910    }
911    batches
912}

944/// Every overlay one style layer paints from the tile, in feature order.
945fn overlays_for_layer_style(
946    tile: &VectorTile,
947    tile_address: VectorTileAddress,
948    layer_style: &VectorLayerStyle,
949    style_zoom_level: f64,
950) -> Vec<MapOverlay> {
951    let mut overlays = Vec::new();
952    if !layer_style.is_visible_at_zoom(style_zoom_level) {
953        return overlays;
954    }
955    let painted_width_pixels = layer_style.width_pixels_at_zoom(style_zoom_level);
956    for layer in tile
957        .layers
958        .iter()
959        .filter(|layer| layer.name == layer_style.source_layer)
960    {
961        for feature in &layer.features {
962            if !feature_matches_filter(feature, layer_style.filter.as_ref()) {
963                continue;
964            }
965            match (&feature.geometry, layer_style.geometry_kind) {
966                (VectorTileGeometry::Polygons(rings), VectorStyleGeometryKind::Fill) => {
967                    overlays.extend(filled_polygon_overlays_for_rings(
968                        rings,
969                        layer.extent,
970                        tile_address,
971                        layer_style.fill_rgba,
972                    ));
973                }
974                (VectorTileGeometry::LineStrings(line_strings), VectorStyleGeometryKind::Line) => {
975                    for line_string in line_strings {
976                        let coordinates = line_string
977                            .iter()
978                            .map(|point| {
979                                tile_local_point_to_coordinate(tile_address, layer.extent, point)
980                            })
981                            .collect();
982                        overlays.push(MapOverlay::RoutePolyline(RoutePolylineOverlay {
983                            coordinates,
984                            rgba_color: layer_style.line_rgba,
985                            width_pixels: painted_width_pixels,
986                        }));
987                    }
988                }
989                (VectorTileGeometry::Points(points), VectorStyleGeometryKind::Circle) => {
990                    let positions = points
991                        .iter()
992                        .map(|point| {
993                            tile_local_point_to_coordinate(tile_address, layer.extent, point)
994                        })
995                        .collect();
996                    overlays.push(MapOverlay::PointMarkers(PointMarkersOverlay {
997                        positions,
998                        rgba_color: layer_style.circle_rgba,
999                        radius_pixels: layer_style.circle_radius_pixels,
1000                    }));
1001                }
1002                // A style paints only its declared kind; a feature of
1003                // another kind is simply not this layer's paint.
1004                _ => {}
1005            }
1006        }
1007    }
1008    overlays
1009}

1011/// Groups a polygon feature's rings into filled overlays: each exterior ring
1012/// owns its immediately following interior rings, the specification's order.
1013/// A leading interior ring (malformed ring order) has no exterior to subtract
1014/// from and is skipped rather than painted as a fill of its own.
1015fn filled_polygon_overlays_for_rings(
1016    rings: &[VectorTileLinearRing],
1017    layer_extent: u64,
1018    tile_address: VectorTileAddress,
1019    rgba_color: [f32; 4],
1020) -> Vec<MapOverlay> {
1021    let mut overlays = Vec::new();
1022    let mut current_exterior_index: Option<usize> = None;
1023    for ring in rings {
1024        let ring_coordinates: Vec<GeographicCoordinate> = ring
1025            .vertices
1026            .iter()
1027            .map(|point| tile_local_point_to_coordinate(tile_address, layer_extent, point))
1028            .collect();
1029        if ring_is_exterior(ring) {
1030            overlays.push(MapOverlay::FilledPolygon(FilledPolygonOverlay {
1031                ring_coordinates,
1032                hole_ring_coordinates: Vec::new(),
1033                rgba_color,
1034            }));
1035            current_exterior_index = Some(overlays.len() - 1);
1036        } else if let Some(exterior_index) = current_exterior_index {
1037            if let Some(MapOverlay::FilledPolygon(overlay)) = overlays.get_mut(exterior_index) {
1038                overlay.hole_ring_coordinates.push(ring_coordinates);
1039            }
1040        }
1041    }
1042    overlays
1043}

1050/// Whether the feature carries the filter's attribute with the expected text.
1051fn feature_matches_filter(
1052    feature: &VectorTileFeature,
1053    filter: Option<&VectorAttributeFilter>,
1054) -> bool {
1055    let Some(attribute_filter) = filter else {
1056        return true;
1057    };
1058    feature
1059        .attributes
1060        .iter()
1061        .any(|(attribute_name, attribute_value)| {
1062            attribute_name == &attribute_filter.attribute_name
1063                && matches!(
1064                    attribute_value,
1065                    VectorTileAttributeValue::Text(text) if text == &attribute_filter.expected_text
1066                )
1067        })
1068}

1070/// The ring's signed shoelace area in tile-local space; exterior rings are
1071/// positive (the specification's `y`-down convention), holes negative.
1072fn ring_is_exterior(ring: &VectorTileLinearRing) -> bool {
1073    let vertices = &ring.vertices;
1074    if vertices.len() < 3 {
1075        return false;
1076    }
1077    let mut doubled_area = 0.0;
1078    for vertex_index in 0..vertices.len() {
1079        let current = &vertices[vertex_index];
1080        let next = &vertices[(vertex_index + 1) % vertices.len()];
1081        doubled_area += current.x_coordinate as f64 * next.y_coordinate as f64
1082            - next.x_coordinate as f64 * current.y_coordinate as f64;
1083    }
1084    doubled_area > 0.0
1085}

1574    /// Tolerance of the hole containment check against the exterior boundary:
1575    /// real tile data can emit a hole that touches its exterior at a vertex
1576    /// (common at coastlines), which is legal even-odd fill but not strictly
1577    /// interior. Degrees at tile scale: well below a millimetre.
1578    const RING_BOUNDARY_TOLERANCE_DEGREES: f64 = 1e-9;
1579
1580    /// Whether a coordinate lies inside a polygon ring or on its boundary
1581    /// (even-odd ray cast in longitude/latitude, plus a boundary tolerance).
1582    fn point_lies_within_or_on_ring(
1583        point: &GeographicCoordinate,
1584        ring_coordinates: &[GeographicCoordinate],
1585    ) -> bool {
1586        point_lies_within_ring(point, ring_coordinates)
1587            || ring_coordinates
1588                .iter()
1589                .enumerate()
1590                .any(|(index, edge_start)| {
1591                    let edge_end = &ring_coordinates[(index + 1) % ring_coordinates.len()];
1592                    let edge_longitude = edge_end.longitude - edge_start.longitude;
1593                    let edge_latitude = edge_end.latitude - edge_start.latitude;
1594                    let edge_length_squared =
1595                        edge_longitude * edge_longitude + edge_latitude * edge_latitude;
1596                    let offset_longitude = point.longitude - edge_start.longitude;
1597                    let offset_latitude = point.latitude - edge_start.latitude;
1598                    let projection = if edge_length_squared == 0.0 {
1599                        0.0
1600                    } else {
1601                        ((offset_longitude * edge_longitude + offset_latitude * edge_latitude)
1602                            / edge_length_squared)
1603                            .clamp(0.0, 1.0)
1604                    };
1605                    let closest_longitude = edge_start.longitude + projection * edge_longitude;
1606                    let closest_latitude = edge_start.latitude + projection * edge_latitude;
1607                    (point.longitude - closest_longitude).hypot(point.latitude - closest_latitude)
1608                        <= RING_BOUNDARY_TOLERANCE_DEGREES
1609                })
1610    }

1612    /// Whether a coordinate lies inside a polygon ring (even-odd ray cast in
1613    /// longitude/latitude).
1614    fn point_lies_within_ring(
1615        point: &GeographicCoordinate,
1616        ring_coordinates: &[GeographicCoordinate],
1617    ) -> bool {
1618        let mut inside = false;
1619        let mut previous_index = ring_coordinates.len().saturating_sub(1);
1620        for current_index in 0..ring_coordinates.len() {
1621            let current = ring_coordinates[current_index];
1622            let previous = ring_coordinates[previous_index];
1623            let crosses_parallel =
1624                (current.latitude > point.latitude) != (previous.latitude > point.latitude);
1625            if crosses_parallel {
1626                let boundary_longitude = (previous.longitude - current.longitude)
1627                    * (point.latitude - current.latitude)
1628                    / (previous.latitude - current.latitude)
1629                    + current.longitude;
1630                if point.longitude < boundary_longitude {
1631                    inside = !inside;
1632                }
1633            }
1634            previous_index = current_index;
1635        }
1636        inside
1637    }

1640    /// do not clip at the tile seam, and label layers (`places`) anchor their

1642    /// anchor is about a quarter-tile out; the check allows a third of a tile.

1645    fn assert_tile_bounds_contain(coordinate: &GeographicCoordinate) {
1646        let north_west = tile_local_point_to_coordinate(
1647            SOURDOUGH_TILE_ADDRESS,
1648            4096,
1649            &VectorTileGeometryPoint {
1650                x_coordinate: 0,
1651                y_coordinate: 0,
1652            },
1653        );
1654        let south_east = tile_local_point_to_coordinate(

1656            4096,
1657            &VectorTileGeometryPoint {
1658                x_coordinate: 4096,
1659                y_coordinate: 4096,
1660            },
1661        );
