1//! Minimal vector-tile styling (M9, the fill/line/circle slice): a decoded
2//! [`map_vector_tiles::VectorTile`] becomes the renderer's existing overlay
3//! primitives — filled polygons for polygon features, polylines for line
4//! features, point markers for point features — under a small explicit style.
5//!
6//! The style is deliberately data-only: a layer entry names the source layer,
7//! the geometry kind it paints, an optional one-attribute text filter and the
8//! paint values (RGBA, line width, marker radius). No expression language, no
9//! per-symbol layout, no glyph atlas — those are later slices. The committed
10//! artifact [`DEFAULT_VECTOR_STYLE_ARTIFACT_JSON`] carries the default style;
11//! [`default_vector_tile_style`] parses it and [`in_code_default_vector_tile_style`]
12//! is the documented fallback when the artifact is missing or malformed.
13//!
14//! Polygon features paint every exterior ring with its interior rings
15//! subtracted through the overlay's hole-aware fill (see
16//! [`crate::overlay::FilledPolygonOverlay`]), and the conversion groups its
17//! primitives into one [`VectorOverlayBatch`] per style layer, the seam a
18//! dedicated vector pipeline would consume later.
19//!
20//! Coordinates convert from tile-local extent space (`x` right, `y` down, the
21//! MVT convention) through the tile's web-mercator address to geographic
22//! degrees, so the overlays project exactly like every other map coordinate.
23

72/// One style layer: source layer, painted kind, optional filter and paint.
73#[derive(Debug, Clone, PartialEq)]
74pub struct VectorLayerStyle {
75    /// The decoded layer name the style paints (`water`, `highways`, …).
76    pub source_layer: String,
77    /// The geometry kind the style paints.
78    pub geometry_kind: VectorStyleGeometryKind,
79    /// An optional one-attribute text filter.
80    pub filter: Option<VectorAttributeFilter>,
81    /// Fill RGBA (used by [`VectorStyleGeometryKind::Fill`]).
82    pub fill_rgba: [f32; 4],
83    /// Line RGBA (used by [`VectorStyleGeometryKind::Line`]).
84    pub line_rgba: [f32; 4],
85    /// Line width in screen pixels.
86    pub line_width_pixels: f64,
87    /// Marker RGBA (used by [`VectorStyleGeometryKind::Circle`]).
88    pub circle_rgba: [f32; 4],
89    /// Marker radius in screen pixels.
90    pub circle_radius_pixels: f64,
91}
92
93impl VectorLayerStyle {
94    /// Builds a fill layer with the given paint.
95    pub fn fill(source_layer: impl Into<String>, rgba_color: [f32; 4]) -> Self {
96        Self::base(source_layer, VectorStyleGeometryKind::Fill, rgba_color)
97    }
98
99    /// Builds a line layer with the given paint and width.
100    pub fn line(source_layer: impl Into<String>, rgba_color: [f32; 4], width_pixels: f64) -> Self {
101        let mut layer_style = Self::base(source_layer, VectorStyleGeometryKind::Line, rgba_color);
102        layer_style.line_rgba = rgba_color;
103        layer_style.line_width_pixels = width_pixels;
104        layer_style
105    }
106
107    /// Builds a circle layer with the given paint and radius.
108    pub fn circle(
109        source_layer: impl Into<String>,
110        rgba_color: [f32; 4],
111        radius_pixels: f64,
112    ) -> Self {
113        let mut layer_style = Self::base(source_layer, VectorStyleGeometryKind::Circle, rgba_color);
114        layer_style.circle_rgba = rgba_color;
115        layer_style.circle_radius_pixels = radius_pixels;
116        layer_style
117    }
118
119    /// Adds the one-attribute text filter.
120    pub fn with_filter(
121        mut self,
122        attribute_name: impl Into<String>,
123        expected_text: impl Into<String>,
124    ) -> Self {
125        self.filter = Some(VectorAttributeFilter {
126            attribute_name: attribute_name.into(),
127            expected_text: expected_text.into(),
128        });
129        self
130    }
131
132    fn base(
133        source_layer: impl Into<String>,
134        geometry_kind: VectorStyleGeometryKind,
135        rgba_color: [f32; 4],
136    ) -> Self {
137        Self {
138            source_layer: source_layer.into(),
139            geometry_kind,
140            filter: None,
141            fill_rgba: rgba_color,
142            line_rgba: rgba_color,
143            line_width_pixels: DEFAULT_VECTOR_LINE_WIDTH_PIXELS,
144            circle_rgba: rgba_color,
145            circle_radius_pixels: DEFAULT_VECTOR_CIRCLE_RADIUS_PIXELS,
146        }
147    }
148}

202/// The style artifacts a style id can name; `None` names no committed style.
203fn vector_style_artifact_json(style_id: &str) -> Option<&'static str> {
204    match style_id {
205        VECTOR_STYLE_ID_CARTO_LIGHT => Some(DEFAULT_VECTOR_STYLE_ARTIFACT_JSON),
206        VECTOR_STYLE_ID_CARTO_DARK => Some(CARTO_DARK_STYLE_ARTIFACT_JSON),
207        VECTOR_STYLE_ID_OPENMAPTILES_LIGHT => Some(OPENMAPTILES_LIGHT_STYLE_ARTIFACT_JSON),
208        VECTOR_STYLE_ID_OPENMAPTILES_DARK => Some(OPENMAPTILES_DARK_STYLE_ARTIFACT_JSON),
209        _ => None,
210    }
211}

213/// Parses the committed style artifact a style id names. An unknown id and a
214/// malformed artifact both report the typed artifact error rather than a
215/// guess; only the default style has an in-code fallback.
216pub fn vector_tile_style_from_id(
217    style_id: &str,
218) -> Result<VectorTileStyle, VectorStyleArtifactError> {
219    let Some(artifact_json) = vector_style_artifact_json(style_id) else {
220        return Err(VectorStyleArtifactError {
221            detail: format!(
222                "no committed vector style carries the id {style_id:?}; known ids are \
223                 {VECTOR_STYLE_ID_CARTO_LIGHT}, {VECTOR_STYLE_ID_CARTO_DARK}, \
224                 {VECTOR_STYLE_ID_OPENMAPTILES_LIGHT}, {VECTOR_STYLE_ID_OPENMAPTILES_DARK}"
225            ),
226        });
227    };
228    vector_tile_style_from_json(artifact_json)
229}

478// Walking and cycling paths are deliberately absent: this style carries no
479// zoom-dependent width or visibility, and at the demo's zoom a sidewalk
480// network would paint over the street grid. A symbol/zoom-function style
481// slice is the documented later seam.
482
483/// Why a style artifact could not be parsed into a [`VectorTileStyle`].
484#[derive(Debug, Clone, PartialEq, Eq)]
485pub struct VectorStyleArtifactError {
486    /// The underlying parser detail (a malformed artifact reports the JSON
487    /// error and its location).
488    pub detail: String,
489}
490
491impl std::fmt::Display for VectorStyleArtifactError {
492    fn fmt(&self, formatter: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
493        write!(formatter, "vector style artifact invalid: {}", self.detail)
494    }
495}

497impl std::error::Error for VectorStyleArtifactError {}

499/// Parses a data-only style artifact into a [`VectorTileStyle`].
500///
501/// The artifact names no providers and carries no URLs: a layer entry has a
502/// `source_layer`, a `geometry_kind` (`fill`, `line` or `circle`), an `rgba`
503/// paint, an optional one-attribute `filter`, and the kind's `width_pixels`
504/// or `radius_pixels` (the documented defaults apply when omitted).
505pub fn vector_tile_style_from_json(
506    artifact_json: &str,
507) -> Result<VectorTileStyle, VectorStyleArtifactError> {
508    let artifact: VectorStyleArtifact =
509        serde_json::from_str(artifact_json).map_err(|parse_error| VectorStyleArtifactError {
510            detail: parse_error.to_string(),
511        })?;
512    Ok(VectorTileStyle {
513        layers: artifact
514            .layers
515            .into_iter()
516            .map(VectorStyleArtifactLayer::into_layer_style)
517            .collect(),
518    })
519}

521/// The style constructed in code, the documented fallback when the artifact
522/// is missing or malformed. It mirrors
523/// [`DEFAULT_VECTOR_STYLE_ARTIFACT_JSON`] so the two paths paint the same
524/// demo, which a test pins.
525pub fn in_code_default_vector_tile_style() -> VectorTileStyle {
526    let mut layers = Vec::new();
527    // Ground cover first: land use, natural areas and leisure surfaces paint
528    // under everything else, in the artifact's filtered-fill order.
529    for area_fill in FALLBACK_AREA_FILLS {
530        layers.push(
531            VectorLayerStyle::fill(area_fill.source_layer, area_fill.rgba_color)
532                .with_filter(area_fill.attribute_name, area_fill.expected_text),
533        );
534    }
535    layers.push(VectorLayerStyle::fill("water", FALLBACK_WATER_FILL_RGBA));
536    layers.push(VectorLayerStyle::line(
537        "waterways",
538        FALLBACK_WATER_FILL_RGBA,
539        FALLBACK_WATERWAY_LINE_WIDTH_PIXELS,
540    ));
541    layers.push(VectorLayerStyle::fill(
542        "buildings",
543        FALLBACK_BUILDING_FILL_RGBA,
544    ));
545    layers.push(VectorLayerStyle::line(
546        "railways",
547        FALLBACK_RAILWAY_LINE_RGBA,
548        FALLBACK_RAILWAY_LINE_WIDTH_PIXELS,
549    ));
550    // Road casings paint as one block under every fill, so intersecting
551    // roads read as continuous ribbons instead of cased-on-cased seams.
552    for road_tier in FALLBACK_ROAD_TIERS {
553        let Some(casing_rgba_color) = road_tier.casing_rgba_color else {
554            continue;
555        };
556        for highway_class in road_tier.highway_classes {
557            layers.push(
558                VectorLayerStyle::line(
559                    "highways",
560                    casing_rgba_color,
561                    road_tier.casing_width_pixels,
562                )
563                .with_filter("highway", *highway_class),
564            );
565        }
566    }
567    for road_tier in FALLBACK_ROAD_TIERS {
568        for highway_class in road_tier.highway_classes {
569            layers.push(
570                VectorLayerStyle::line(
571                    "highways",
572                    road_tier.fill_rgba_color,
573                    road_tier.fill_width_pixels,
574                )
575                .with_filter("highway", *highway_class),
576            );
577        }
578    }
579    layers.push(VectorLayerStyle::line(
580        "boundaries",
581        FALLBACK_BOUNDARY_LINE_RGBA,
582        FALLBACK_BOUNDARY_LINE_WIDTH_PIXELS,
583    ));
584    layers.push(VectorLayerStyle::circle(
585        "places",
586        FALLBACK_PLACE_CIRCLE_RGBA,
587        FALLBACK_PLACE_CIRCLE_RADIUS_PIXELS,
588    ));
589    VectorTileStyle { layers }
590}

606/// The artifact's parsed shape; every field is data, never a code path.
607#[derive(Debug, Deserialize)]
608#[serde(deny_unknown_fields)]
609struct VectorStyleArtifact {
610    /// The artifact's layer entries, applied in order.
611    layers: Vec<VectorStyleArtifactLayer>,
612}
613
614/// One artifact layer entry, in the artifact's kind-specific paint form.
615#[derive(Debug, Deserialize)]
616#[serde(deny_unknown_fields)]
617struct VectorStyleArtifactLayer {
618    /// The decoded layer name the style paints.
619    source_layer: String,
620    /// The geometry kind the style paints.
621    geometry_kind: VectorStyleArtifactGeometryKind,
622    /// The layer's RGBA paint.
623    rgba: [f32; 4],
624    /// An optional one-attribute text filter.
625    #[serde(default)]
626    filter: Option<VectorStyleArtifactFilter>,
627    /// Line width in screen pixels (line layers; the documented default when
628    /// omitted).
629    #[serde(default)]
630    width_pixels: Option<f64>,
631    /// Marker radius in screen pixels (circle layers; the documented default
632    /// when omitted).
633    #[serde(default)]
634    radius_pixels: Option<f64>,
635}
636
637/// The artifact's geometry-kind spelling (`fill`, `line`, `circle`).
638#[derive(Debug, Deserialize)]
639#[serde(rename_all = "snake_case")]
640enum VectorStyleArtifactGeometryKind {
641    /// Polygon features, painted as holes-aware filled rings.
642    Fill,
643    /// LineString features, painted as thickened polylines.
644    Line,
645    /// Point features, painted as screen-facing markers.
646    Circle,
647}
648
649/// The artifact's one-attribute text filter.
650#[derive(Debug, Deserialize)]
651#[serde(deny_unknown_fields)]
652struct VectorStyleArtifactFilter {
653    /// The attribute key to match.
654    attribute_name: String,
655    /// The exact text value to match.
656    expected_text: String,
657}
658
659impl VectorStyleArtifactLayer {
660    /// Builds the renderer's layer style from the artifact entry.
661    fn into_layer_style(self) -> VectorLayerStyle {
662        let mut layer_style = match self.geometry_kind {
663            VectorStyleArtifactGeometryKind::Fill => {
664                VectorLayerStyle::fill(self.source_layer, self.rgba)
665            }
666            VectorStyleArtifactGeometryKind::Line => VectorLayerStyle::line(
667                self.source_layer,
668                self.rgba,
669                self.width_pixels
670                    .unwrap_or(DEFAULT_VECTOR_LINE_WIDTH_PIXELS),
671            ),
672            VectorStyleArtifactGeometryKind::Circle => VectorLayerStyle::circle(
673                self.source_layer,
674                self.rgba,
675                self.radius_pixels
676                    .unwrap_or(DEFAULT_VECTOR_CIRCLE_RADIUS_PIXELS),
677            ),
678        };
679        if let Some(filter) = self.filter {
680            layer_style = layer_style.with_filter(filter.attribute_name, filter.expected_text);
681        }
682        layer_style
683    }
684}

686/// Converts one tile-local extent point into a geographic coordinate through
687/// the tile's web-mercator address. Vertices at the extent boundary are legal
688/// (they land on the next tile's edge); the latitude clamp keeps the projection
689/// finite at the world's polar edges.
690pub fn tile_local_point_to_coordinate(
691    tile_address: VectorTileAddress,
692    layer_extent: u64,
693    point: &VectorTileGeometryPoint,
694) -> GeographicCoordinate {
695    let extent = layer_extent.max(1) as f64;
696    let tiles_per_axis = (1u64 << tile_address.zoom_level) as f64;
697    let world_x =
698        (f64::from(tile_address.tile_x) + point.x_coordinate as f64 / extent) / tiles_per_axis;
699    let world_y =
700        (f64::from(tile_address.tile_y) + point.y_coordinate as f64 / extent) / tiles_per_axis;
701    let longitude_degrees = world_x * WORLD_LONGITUDE_SPAN_DEGREES + WORLD_WEST_LONGITUDE_DEGREES;
702    let latitude_degrees = (PI * (1.0 - 2.0 * world_y)).sinh().atan().to_degrees();
703    GeographicCoordinate {
704        latitude: latitude_degrees.clamp(
705            -MERCATOR_MAXIMUM_LATITUDE_DEGREES,
706            MERCATOR_MAXIMUM_LATITUDE_DEGREES,
707        ),
708        longitude: longitude_degrees,
709    }
710}

712/// Converts one decoded tile into one overlay batch per style layer that
713/// painted at least one feature.
714///
715/// A layer style paints only features of its declared geometry kind whose
716/// filter (when set) matches. Polygon features contribute one filled overlay
717/// per exterior ring (positive shoelace area in the tile's `y`-down space),
718/// carrying that exterior's following interior rings (negative area) as the
719/// fill's subtracted holes. Every feature keeps its own overlay primitive, so
720/// per-feature geometry stays representable; the batch is the style grouping
721/// a dedicated vector pipeline would consume.
722pub fn overlay_primitives_for_vector_tile(
723    tile: &VectorTile,
724    tile_address: VectorTileAddress,
725    style: &VectorTileStyle,
726) -> Vec<VectorOverlayBatch> {
727    let mut batches = Vec::new();
728    for layer_style in &style.layers {
729        let overlays = overlays_for_layer_style(tile, tile_address, layer_style);
730        if !overlays.is_empty() {
731            batches.push(VectorOverlayBatch {
732                layer_style: layer_style.clone(),
733                overlays,
734            });
735        }
736    }
737    batches
738}
739
740/// Converts one decoded tile into its flat overlay primitive list under the
741/// style (the batches of [`overlay_primitives_for_vector_tile`] in style
742/// order, concatenated).
743pub fn vector_tile_overlays(
744    tile: &VectorTile,
745    tile_address: VectorTileAddress,
746    style: &VectorTileStyle,
747) -> Vec<MapOverlay> {
748    overlay_primitives_for_vector_tile(tile, tile_address, style)
749        .into_iter()
750        .flat_map(|batch| batch.overlays)
751        .collect()
752}
753
754/// Every overlay one style layer paints from the tile, in feature order.
755fn overlays_for_layer_style(
756    tile: &VectorTile,
757    tile_address: VectorTileAddress,
758    layer_style: &VectorLayerStyle,
759) -> Vec<MapOverlay> {
760    let mut overlays = Vec::new();
761    for layer in tile
762        .layers
763        .iter()
764        .filter(|layer| layer.name == layer_style.source_layer)
765    {
766        for feature in &layer.features {
767            if !feature_matches_filter(feature, layer_style.filter.as_ref()) {
768                continue;
769            }
770            match (&feature.geometry, layer_style.geometry_kind) {
771                (VectorTileGeometry::Polygons(rings), VectorStyleGeometryKind::Fill) => {
772                    overlays.extend(filled_polygon_overlays_for_rings(
773                        rings,
774                        layer.extent,
775                        tile_address,
776                        layer_style.fill_rgba,
777                    ));
778                }
779                (VectorTileGeometry::LineStrings(line_strings), VectorStyleGeometryKind::Line) => {
780                    for line_string in line_strings {
781                        let coordinates = line_string
782                            .iter()
783                            .map(|point| {
784                                tile_local_point_to_coordinate(tile_address, layer.extent, point)
785                            })
786                            .collect();
787                        overlays.push(MapOverlay::RoutePolyline(RoutePolylineOverlay {
788                            coordinates,
789                            rgba_color: layer_style.line_rgba,
790                            width_pixels: layer_style.line_width_pixels,
791                        }));
792                    }
793                }
794                (VectorTileGeometry::Points(points), VectorStyleGeometryKind::Circle) => {
795                    let positions = points
796                        .iter()
797                        .map(|point| {
798                            tile_local_point_to_coordinate(tile_address, layer.extent, point)
799                        })
800                        .collect();
801                    overlays.push(MapOverlay::PointMarkers(PointMarkersOverlay {
802                        positions,
803                        rgba_color: layer_style.circle_rgba,
804                        radius_pixels: layer_style.circle_radius_pixels,
805                    }));
806                }
807                // A style paints only its declared kind; a feature of
808                // another kind is simply not this layer's paint.
809                _ => {}
810            }
811        }
812    }
813    overlays
814}

816/// Groups a polygon feature's rings into filled overlays: each exterior ring
817/// owns its immediately following interior rings, the specification's order.
818/// A leading interior ring (malformed ring order) has no exterior to subtract
819/// from and is skipped rather than painted as a fill of its own.
820fn filled_polygon_overlays_for_rings(
821    rings: &[VectorTileLinearRing],
822    layer_extent: u64,
823    tile_address: VectorTileAddress,
824    rgba_color: [f32; 4],
825) -> Vec<MapOverlay> {
826    let mut overlays = Vec::new();
827    let mut current_exterior_index: Option<usize> = None;
828    for ring in rings {
829        let ring_coordinates: Vec<GeographicCoordinate> = ring
830            .vertices
831            .iter()
832            .map(|point| tile_local_point_to_coordinate(tile_address, layer_extent, point))
833            .collect();
834        if ring_is_exterior(ring) {
835            overlays.push(MapOverlay::FilledPolygon(FilledPolygonOverlay {
836                ring_coordinates,
837                hole_ring_coordinates: Vec::new(),
838                rgba_color,
839            }));
840            current_exterior_index = Some(overlays.len() - 1);
841        } else if let Some(exterior_index) = current_exterior_index {
842            if let Some(MapOverlay::FilledPolygon(overlay)) = overlays.get_mut(exterior_index) {
843                overlay.hole_ring_coordinates.push(ring_coordinates);
844            }
845        }
846    }
847    overlays
848}

850/// The feature's geometry kind (the decoder's own classification).
851pub fn feature_geometry_kind(feature: &VectorTileFeature) -> VectorTileGeometryType {
852    feature.geometry_type
853}
854
855/// Whether the feature carries the filter's attribute with the expected text.
856fn feature_matches_filter(
857    feature: &VectorTileFeature,
858    filter: Option<&VectorAttributeFilter>,
859) -> bool {
860    let Some(attribute_filter) = filter else {
861        return true;
862    };
863    feature
864        .attributes
865        .iter()
866        .any(|(attribute_name, attribute_value)| {
867            attribute_name == &attribute_filter.attribute_name
868                && matches!(
869                    attribute_value,
870                    VectorTileAttributeValue::Text(text) if text == &attribute_filter.expected_text
871                )
872        })
873}
874
875/// The ring's signed shoelace area in tile-local space; exterior rings are
876/// positive (the specification's `y`-down convention), holes negative.
877fn ring_is_exterior(ring: &VectorTileLinearRing) -> bool {
878    let vertices = &ring.vertices;
879    if vertices.len() < 3 {
880        return false;
881    }
882    let mut doubled_area = 0.0;
883    for vertex_index in 0..vertices.len() {
884        let current = &vertices[vertex_index];
885        let next = &vertices[(vertex_index + 1) % vertices.len()];
886        doubled_area += current.x_coordinate as f64 * next.y_coordinate as f64
887            - next.x_coordinate as f64 * current.y_coordinate as f64;
888    }
889    doubled_area > 0.0
890}

1267    /// Tolerance of the hole containment check against the exterior boundary:
1268    /// real tile data can emit a hole that touches its exterior at a vertex
1269    /// (common at coastlines), which is legal even-odd fill but not strictly
1270    /// interior. Degrees at tile scale: well below a millimetre.
1271    const RING_BOUNDARY_TOLERANCE_DEGREES: f64 = 1e-9;
1272
1273    /// Whether a coordinate lies inside a polygon ring or on its boundary
1274    /// (even-odd ray cast in longitude/latitude, plus a boundary tolerance).
1275    fn point_lies_within_or_on_ring(
1276        point: &GeographicCoordinate,
1277        ring_coordinates: &[GeographicCoordinate],
1278    ) -> bool {
1279        point_lies_within_ring(point, ring_coordinates)
1280            || ring_coordinates
1281                .iter()
1282                .enumerate()
1283                .any(|(index, edge_start)| {
1284                    let edge_end = &ring_coordinates[(index + 1) % ring_coordinates.len()];
1285                    let edge_longitude = edge_end.longitude - edge_start.longitude;
1286                    let edge_latitude = edge_end.latitude - edge_start.latitude;
1287                    let edge_length_squared =
1288                        edge_longitude * edge_longitude + edge_latitude * edge_latitude;
1289                    let offset_longitude = point.longitude - edge_start.longitude;
1290                    let offset_latitude = point.latitude - edge_start.latitude;
1291                    let projection = if edge_length_squared == 0.0 {
1292                        0.0
1293                    } else {
1294                        ((offset_longitude * edge_longitude + offset_latitude * edge_latitude)
1295                            / edge_length_squared)
1296                            .clamp(0.0, 1.0)
1297                    };
1298                    let closest_longitude = edge_start.longitude + projection * edge_longitude;
1299                    let closest_latitude = edge_start.latitude + projection * edge_latitude;
1300                    (point.longitude - closest_longitude).hypot(point.latitude - closest_latitude)
1301                        <= RING_BOUNDARY_TOLERANCE_DEGREES
1302                })
1303    }
1304
1305    /// Whether a coordinate lies inside a polygon ring (even-odd ray cast in
1306    /// longitude/latitude).
1307    fn point_lies_within_ring(
1308        point: &GeographicCoordinate,
1309        ring_coordinates: &[GeographicCoordinate],
1310    ) -> bool {
1311        let mut inside = false;
1312        let mut previous_index = ring_coordinates.len().saturating_sub(1);
1313        for current_index in 0..ring_coordinates.len() {
1314            let current = ring_coordinates[current_index];
1315            let previous = ring_coordinates[previous_index];
1316            let crosses_parallel =
1317                (current.latitude > point.latitude) != (previous.latitude > point.latitude);
1318            if crosses_parallel {
1319                let boundary_longitude = (previous.longitude - current.longitude)
1320                    * (point.latitude - current.latitude)
1321                    / (previous.latitude - current.latitude)
1322                    + current.longitude;
1323                if point.longitude < boundary_longitude {
1324                    inside = !inside;
1325                }
1326            }
1327            previous_index = current_index;
1328        }
1329        inside
1330    }

1332    /// MVT producers buffer geometry beyond the nominal extent so features
1333    /// do not clip at the tile seam, and label layers (`places`) anchor their
1334    /// points well past the edge for placement. This fixture's farthest
1335    /// anchor is about a quarter-tile out; the check allows a third of a tile.
1336    const TILE_BUFFER_TOLERANCE_FRACTION: f64 = 0.34;
1337
1338    fn assert_tile_bounds_contain(coordinate: &GeographicCoordinate) {
1339        let north_west = tile_local_point_to_coordinate(
1340            SOURDOUGH_TILE_ADDRESS,
1341            4096,
1342            &VectorTileGeometryPoint {
1343                x_coordinate: 0,
1344                y_coordinate: 0,
1345            },
1346        );
1347        let south_east = tile_local_point_to_coordinate(
1348            SOURDOUGH_TILE_ADDRESS,
1349            4096,
1350            &VectorTileGeometryPoint {
1351                x_coordinate: 4096,
1352                y_coordinate: 4096,
1353            },
1354        );
1355        let longitude_margin =
1356            (south_east.longitude - north_west.longitude) * TILE_BUFFER_TOLERANCE_FRACTION;
1357        let latitude_margin =
1358            (north_west.latitude - south_east.latitude) * TILE_BUFFER_TOLERANCE_FRACTION;
1359        assert!(
1360            coordinate.longitude >= north_west.longitude - longitude_margin,
1361            "longitude {} outside [{}, {}] (margin {longitude_margin})",
1362            coordinate.longitude,
1363            north_west.longitude,
1364            south_east.longitude
1365        );
1366        assert!(
1367            coordinate.longitude <= south_east.longitude + longitude_margin,
1368            "longitude {} outside [{}, {}] (margin {longitude_margin})",
1369            coordinate.longitude,
1370            north_west.longitude,
1371            south_east.longitude
1372        );
1373        assert!(
1374            coordinate.latitude <= north_west.latitude + latitude_margin,
1375            "latitude {} outside [{}, {}] (margin {latitude_margin})",
1376            coordinate.latitude,
1377            south_east.latitude,
1378            north_west.latitude
1379        );
1380        assert!(
1381            coordinate.latitude >= south_east.latitude - latitude_margin,
1382            "latitude {} outside [{}, {}] (margin {latitude_margin})",
1383            coordinate.latitude,
1384            south_east.latitude,
1385            north_west.latitude
1386        );
1387    }
