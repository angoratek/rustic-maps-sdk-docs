1//! Minimal vector-tile styling (M9, the fill/line/circle slice): a decoded
2//! [`map_vector_tiles::VectorTile`] becomes the renderer's existing overlay
3//! primitives — filled polygons for polygon features, polylines for line
4//! features, point markers for point features — under a small explicit style.
5//!
6//! The style is deliberately data-only: a layer entry names the source layer,
7//! the geometry kind it paints, an optional one-attribute text filter and the
8//! paint values (RGBA, line width, marker radius). No expression language, no
9//! per-symbol layout, no glyph atlas — those are later slices. The committed
10//! artifact [`DEFAULT_VECTOR_STYLE_ARTIFACT_JSON`] carries the default style;
11//! [`default_vector_tile_style`] parses it and [`in_code_default_vector_tile_style`]
12//! is the documented fallback when the artifact is missing or malformed.
13//!
14//! Polygon features paint every exterior ring with its interior rings
15//! subtracted through the overlay's hole-aware fill (see
16//! [`crate::overlay::FilledPolygonOverlay`]), and the conversion groups its
17//! primitives into one [`VectorOverlayBatch`] per style layer, the seam a
18//! dedicated vector pipeline would consume later.
19//!
20//! Coordinates convert from tile-local extent space (`x` right, `y` down, the
21//! MVT convention) through the tile's web-mercator address to geographic
22//! degrees, so the overlays project exactly like every other map coordinate.
23

72/// One style layer: source layer, painted kind, optional filter and paint.
73#[derive(Debug, Clone, PartialEq)]
74pub struct VectorLayerStyle {
75    /// The decoded layer name the style paints (`water`, `highways`, …).
76    pub source_layer: String,
77    /// The geometry kind the style paints.
78    pub geometry_kind: VectorStyleGeometryKind,
79    /// An optional one-attribute text filter.
80    pub filter: Option<VectorAttributeFilter>,
81    /// Fill RGBA (used by [`VectorStyleGeometryKind::Fill`]).
82    pub fill_rgba: [f32; 4],
83    /// Line RGBA (used by [`VectorStyleGeometryKind::Line`]).
84    pub line_rgba: [f32; 4],
85    /// Line width in screen pixels.
86    pub line_width_pixels: f64,
87    /// Marker RGBA (used by [`VectorStyleGeometryKind::Circle`]).
88    pub circle_rgba: [f32; 4],
89    /// Marker radius in screen pixels.
90    pub circle_radius_pixels: f64,
91}

93impl VectorLayerStyle {
94    /// Builds a fill layer with the given paint.
95    pub fn fill(source_layer: impl Into<String>, rgba_color: [f32; 4]) -> Self {
96        Self::base(source_layer, VectorStyleGeometryKind::Fill, rgba_color)
97    }
98
99    /// Builds a line layer with the given paint and width.
100    pub fn line(source_layer: impl Into<String>, rgba_color: [f32; 4], width_pixels: f64) -> Self {
101        let mut layer_style = Self::base(source_layer, VectorStyleGeometryKind::Line, rgba_color);
102        layer_style.line_rgba = rgba_color;
103        layer_style.line_width_pixels = width_pixels;
104        layer_style
105    }
106
107    /// Builds a circle layer with the given paint and radius.
108    pub fn circle(
109        source_layer: impl Into<String>,
110        rgba_color: [f32; 4],
111        radius_pixels: f64,
112    ) -> Self {
113        let mut layer_style = Self::base(source_layer, VectorStyleGeometryKind::Circle, rgba_color);
114        layer_style.circle_rgba = rgba_color;
115        layer_style.circle_radius_pixels = radius_pixels;
116        layer_style
117    }
118
119    /// Adds the one-attribute text filter.
120    pub fn with_filter(
121        mut self,
122        attribute_name: impl Into<String>,
123        expected_text: impl Into<String>,
124    ) -> Self {
125        self.filter = Some(VectorAttributeFilter {
126            attribute_name: attribute_name.into(),
127            expected_text: expected_text.into(),
128        });
129        self
130    }
131
132    fn base(
133        source_layer: impl Into<String>,
134        geometry_kind: VectorStyleGeometryKind,
135        rgba_color: [f32; 4],
136    ) -> Self {
137        Self {
138            source_layer: source_layer.into(),
139            geometry_kind,
140            filter: None,
141            fill_rgba: rgba_color,
142            line_rgba: rgba_color,
143            line_width_pixels: DEFAULT_VECTOR_LINE_WIDTH_PIXELS,
144            circle_rgba: rgba_color,
145            circle_radius_pixels: DEFAULT_VECTOR_CIRCLE_RADIUS_PIXELS,
146        }
147    }
148}

224/// Parses a data-only style artifact into a [`VectorTileStyle`].
225///
226/// The artifact names no providers and carries no URLs: a layer entry has a
227/// `source_layer`, a `geometry_kind` (`fill`, `line` or `circle`), an `rgba`
228/// paint, an optional one-attribute `filter`, and the kind's `width_pixels`
229/// or `radius_pixels` (the documented defaults apply when omitted).
230pub fn vector_tile_style_from_json(
231    artifact_json: &str,
232) -> Result<VectorTileStyle, VectorStyleArtifactError> {
233    let artifact: VectorStyleArtifact =
234        serde_json::from_str(artifact_json).map_err(|parse_error| VectorStyleArtifactError {
235            detail: parse_error.to_string(),
236        })?;
237    Ok(VectorTileStyle {
238        layers: artifact
239            .layers
240            .into_iter()
241            .map(VectorStyleArtifactLayer::into_layer_style)
242            .collect(),
243    })
244}

332#[derive(Debug, Deserialize)]
333#[serde(rename_all = "snake_case")]
334enum VectorStyleArtifactGeometryKind {
335    /// Polygon features, painted as holes-aware filled rings.
336    Fill,
337    /// LineString features, painted as thickened polylines.

343/// The artifact's one-attribute text filter.
344#[derive(Debug, Deserialize)]
345#[serde(deny_unknown_fields)]

347    /// The attribute key to match.
348    attribute_name: String,
349    /// The exact text value to match.
350    expected_text: String,

353impl VectorStyleArtifactLayer {
354    /// Builds the renderer's layer style from the artifact entry.
355    fn into_layer_style(self) -> VectorLayerStyle {
356        let mut layer_style = match self.geometry_kind {
357            VectorStyleArtifactGeometryKind::Fill => {
358                VectorLayerStyle::fill(self.source_layer, self.rgba)
359            }
360            VectorStyleArtifactGeometryKind::Line => VectorLayerStyle::line(
361                self.source_layer,
362                self.rgba,
363                self.width_pixels
364                    .unwrap_or(DEFAULT_VECTOR_LINE_WIDTH_PIXELS),
365            ),
366            VectorStyleArtifactGeometryKind::Circle => VectorLayerStyle::circle(
367                self.source_layer,
368                self.rgba,
369                self.radius_pixels
370                    .unwrap_or(DEFAULT_VECTOR_CIRCLE_RADIUS_PIXELS),
371            ),
372        };
373        if let Some(filter) = self.filter {
374            layer_style = layer_style.with_filter(filter.attribute_name, filter.expected_text);
375        }
376        layer_style
377    }
378}

380/// Converts one tile-local extent point into a geographic coordinate through
381/// the tile's web-mercator address. Vertices at the extent boundary are legal
382/// (they land on the next tile's edge); the latitude clamp keeps the projection
383/// finite at the world's polar edges.
384pub fn tile_local_point_to_coordinate(
385    tile_address: VectorTileAddress,
386    layer_extent: u64,
387    point: &VectorTileGeometryPoint,
388) -> GeographicCoordinate {
389    let extent = layer_extent.max(1) as f64;
390    let tiles_per_axis = (1u64 << tile_address.zoom_level) as f64;
391    let world_x =
392        (f64::from(tile_address.tile_x) + point.x_coordinate as f64 / extent) / tiles_per_axis;
393    let world_y =
394        (f64::from(tile_address.tile_y) + point.y_coordinate as f64 / extent) / tiles_per_axis;
395    let longitude_degrees = world_x * WORLD_LONGITUDE_SPAN_DEGREES + WORLD_WEST_LONGITUDE_DEGREES;
396    let latitude_degrees = (PI * (1.0 - 2.0 * world_y)).sinh().atan().to_degrees();
397    GeographicCoordinate {
398        latitude: latitude_degrees.clamp(
399            -MERCATOR_MAXIMUM_LATITUDE_DEGREES,
400            MERCATOR_MAXIMUM_LATITUDE_DEGREES,
401        ),
402        longitude: longitude_degrees,
403    }
404}

406/// Converts one decoded tile into one overlay batch per style layer that
407/// painted at least one feature.
408///
409/// A layer style paints only features of its declared geometry kind whose
410/// filter (when set) matches. Polygon features contribute one filled overlay
411/// per exterior ring (positive shoelace area in the tile's `y`-down space),
412/// carrying that exterior's following interior rings (negative area) as the
413/// fill's subtracted holes. Every feature keeps its own overlay primitive, so
414/// per-feature geometry stays representable; the batch is the style grouping
415/// a dedicated vector pipeline would consume.
416pub fn overlay_primitives_for_vector_tile(
417    tile: &VectorTile,
418    tile_address: VectorTileAddress,
419    style: &VectorTileStyle,
420) -> Vec<VectorOverlayBatch> {
421    let mut batches = Vec::new();
422    for layer_style in &style.layers {
423        let overlays = overlays_for_layer_style(tile, tile_address, layer_style);
424        if !overlays.is_empty() {
425            batches.push(VectorOverlayBatch {
426                layer_style: layer_style.clone(),
427                overlays,
428            });
429        }
430    }
431    batches
432}

434/// Converts one decoded tile into its flat overlay primitive list under the
435/// style (the batches of [`overlay_primitives_for_vector_tile`] in style
436/// order, concatenated).
437pub fn vector_tile_overlays(
438    tile: &VectorTile,
439    tile_address: VectorTileAddress,
440    style: &VectorTileStyle,
441) -> Vec<MapOverlay> {
442    overlay_primitives_for_vector_tile(tile, tile_address, style)
443        .into_iter()
444        .flat_map(|batch| batch.overlays)
445        .collect()
446}

448/// Every overlay one style layer paints from the tile, in feature order.
449fn overlays_for_layer_style(
450    tile: &VectorTile,
451    tile_address: VectorTileAddress,
452    layer_style: &VectorLayerStyle,
453) -> Vec<MapOverlay> {
454    let mut overlays = Vec::new();
455    for layer in tile
456        .layers
457        .iter()
458        .filter(|layer| layer.name == layer_style.source_layer)
459    {
460        for feature in &layer.features {
461            if !feature_matches_filter(feature, layer_style.filter.as_ref()) {
462                continue;
463            }
464            match (&feature.geometry, layer_style.geometry_kind) {
465                (VectorTileGeometry::Polygons(rings), VectorStyleGeometryKind::Fill) => {
466                    overlays.extend(filled_polygon_overlays_for_rings(
467                        rings,
468                        layer.extent,
469                        tile_address,
470                        layer_style.fill_rgba,
471                    ));
472                }
473                (VectorTileGeometry::LineStrings(line_strings), VectorStyleGeometryKind::Line) => {
474                    for line_string in line_strings {
475                        let coordinates = line_string
476                            .iter()
477                            .map(|point| {
478                                tile_local_point_to_coordinate(tile_address, layer.extent, point)
479                            })
480                            .collect();
481                        overlays.push(MapOverlay::RoutePolyline(RoutePolylineOverlay {
482                            coordinates,
483                            rgba_color: layer_style.line_rgba,
484                            width_pixels: layer_style.line_width_pixels,
485                        }));
486                    }
487                }
488                (VectorTileGeometry::Points(points), VectorStyleGeometryKind::Circle) => {
489                    let positions = points
490                        .iter()
491                        .map(|point| {
492                            tile_local_point_to_coordinate(tile_address, layer.extent, point)
493                        })
494                        .collect();
495                    overlays.push(MapOverlay::PointMarkers(PointMarkersOverlay {
496                        positions,
497                        rgba_color: layer_style.circle_rgba,
498                        radius_pixels: layer_style.circle_radius_pixels,
499                    }));
500                }
501                // A style paints only its declared kind; a feature of
502                // another kind is simply not this layer's paint.
503                _ => {}
504            }
505        }
506    }
507    overlays
508}

510/// Groups a polygon feature's rings into filled overlays: each exterior ring
511/// owns its immediately following interior rings, the specification's order.
512/// A leading interior ring (malformed ring order) has no exterior to subtract
513/// from and is skipped rather than painted as a fill of its own.
514fn filled_polygon_overlays_for_rings(
515    rings: &[VectorTileLinearRing],
516    layer_extent: u64,
517    tile_address: VectorTileAddress,
518    rgba_color: [f32; 4],
519) -> Vec<MapOverlay> {
520    let mut overlays = Vec::new();
521    let mut current_exterior_index: Option<usize> = None;
522    for ring in rings {
523        let ring_coordinates: Vec<GeographicCoordinate> = ring
524            .vertices
525            .iter()
526            .map(|point| tile_local_point_to_coordinate(tile_address, layer_extent, point))
527            .collect();
528        if ring_is_exterior(ring) {
529            overlays.push(MapOverlay::FilledPolygon(FilledPolygonOverlay {
530                ring_coordinates,
531                hole_ring_coordinates: Vec::new(),
532                rgba_color,
533            }));
534            current_exterior_index = Some(overlays.len() - 1);
535        } else if let Some(exterior_index) = current_exterior_index {
536            if let Some(MapOverlay::FilledPolygon(overlay)) = overlays.get_mut(exterior_index) {
537                overlay.hole_ring_coordinates.push(ring_coordinates);
538            }
539        }
540    }
541    overlays
542}

549/// Whether the feature carries the filter's attribute with the expected text.
550fn feature_matches_filter(
551    feature: &VectorTileFeature,
552    filter: Option<&VectorAttributeFilter>,
553) -> bool {
554    let Some(attribute_filter) = filter else {
555        return true;
556    };
557    feature
558        .attributes
559        .iter()
560        .any(|(attribute_name, attribute_value)| {
561            attribute_name == &attribute_filter.attribute_name
562                && matches!(
563                    attribute_value,
564                    VectorTileAttributeValue::Text(text) if text == &attribute_filter.expected_text
565                )
566        })
567}

569/// The ring's signed shoelace area in tile-local space; exterior rings are
570/// positive (the specification's `y`-down convention), holes negative.
571fn ring_is_exterior(ring: &VectorTileLinearRing) -> bool {
572    let vertices = &ring.vertices;
573    if vertices.len() < 3 {
574        return false;
575    }
576    let mut doubled_area = 0.0;
577    for vertex_index in 0..vertices.len() {
578        let current = &vertices[vertex_index];
579        let next = &vertices[(vertex_index + 1) % vertices.len()];
580        doubled_area += current.x_coordinate as f64 * next.y_coordinate as f64
581            - next.x_coordinate as f64 * current.y_coordinate as f64;
582    }
583    doubled_area > 0.0
584}

733    fn the_water_layer_assembles_into_finite_overlay_geometry() {
734        // The fixture's water layer includes the 150-ring multipolygon, so
735        // this runs the bridge splicing and ear clipping over the real
736        // messy data end to end, then checks the assembly still emits finite
737        // clip-space triangles.
738        use crate::camera::{DEFAULT_FIELD_OF_VIEW_DEGREES, MapCamera, ViewportPixels};
739        use crate::overlay::overlay_triangle_vertices;
740        use crate::projection::normalized_world_to_coordinate;
741
742        let tile = decode_vector_tile(SOURDOUGH_TILE_BYTES).unwrap();
743        let style = VectorTileStyle {
744            layers: vec![VectorLayerStyle::fill("water", [0.2, 0.5, 0.9, 0.4])],
745        };
746        let overlays = vector_tile_overlays(&tile, SOURDOUGH_TILE_ADDRESS, &style);
747        let tiles_per_axis = f64::from(1u32 << SOURDOUGH_TILE_ADDRESS.zoom_level);
748        let camera_center = normalized_world_to_coordinate([
749            (f64::from(SOURDOUGH_TILE_ADDRESS.tile_x) + 0.5) / tiles_per_axis,
750            (f64::from(SOURDOUGH_TILE_ADDRESS.tile_y) + 0.5) / tiles_per_axis,
751        ]);
752        let camera = MapCamera::new(camera_center, 6.0, 0.0, 0.0);
753        let viewport = ViewportPixels::new(512.0, 512.0).unwrap();
754        let vertices =
755            overlay_triangle_vertices(&camera, &viewport, DEFAULT_FIELD_OF_VIEW_DEGREES, &overlays);
756        assert!(
757            !vertices.is_empty(),

760        assert!(

901    /// Tolerance of the hole containment check against the exterior boundary:
902    /// real tile data can emit a hole that touches its exterior at a vertex
903    /// (common at coastlines), which is legal even-odd fill but not strictly
904    /// interior. Degrees at tile scale: well below a millimetre.
905    const RING_BOUNDARY_TOLERANCE_DEGREES: f64 = 1e-9;
906
907    /// Whether a coordinate lies inside a polygon ring or on its boundary
908    /// (even-odd ray cast in longitude/latitude, plus a boundary tolerance).
909    fn point_lies_within_or_on_ring(
910        point: &GeographicCoordinate,
911        ring_coordinates: &[GeographicCoordinate],
912    ) -> bool {
913        point_lies_within_ring(point, ring_coordinates)
914            || ring_coordinates
915                .iter()
916                .enumerate()
917                .any(|(index, edge_start)| {
918                    let edge_end = &ring_coordinates[(index + 1) % ring_coordinates.len()];
919                    let edge_longitude = edge_end.longitude - edge_start.longitude;
920                    let edge_latitude = edge_end.latitude - edge_start.latitude;
921                    let edge_length_squared =
922                        edge_longitude * edge_longitude + edge_latitude * edge_latitude;
923                    let offset_longitude = point.longitude - edge_start.longitude;
924                    let offset_latitude = point.latitude - edge_start.latitude;
925                    let projection = if edge_length_squared == 0.0 {
926                        0.0
927                    } else {
928                        ((offset_longitude * edge_longitude + offset_latitude * edge_latitude)
929                            / edge_length_squared)
930                            .clamp(0.0, 1.0)
931                    };
932                    let closest_longitude = edge_start.longitude + projection * edge_longitude;
933                    let closest_latitude = edge_start.latitude + projection * edge_latitude;
934                    (point.longitude - closest_longitude).hypot(point.latitude - closest_latitude)
935                        <= RING_BOUNDARY_TOLERANCE_DEGREES
936                })
937    }

939    /// Whether a coordinate lies inside a polygon ring (even-odd ray cast in
940    /// longitude/latitude).
941    fn point_lies_within_ring(
942        point: &GeographicCoordinate,
943        ring_coordinates: &[GeographicCoordinate],
944    ) -> bool {
945        let mut inside = false;
946        let mut previous_index = ring_coordinates.len().saturating_sub(1);
947        for current_index in 0..ring_coordinates.len() {
948            let current = ring_coordinates[current_index];
949            let previous = ring_coordinates[previous_index];
950            let crosses_parallel =
951                (current.latitude > point.latitude) != (previous.latitude > point.latitude);
952            if crosses_parallel {
953                let boundary_longitude = (previous.longitude - current.longitude)
954                    * (point.latitude - current.latitude)
955                    / (previous.latitude - current.latitude)
956                    + current.longitude;
957                if point.longitude < boundary_longitude {
958                    inside = !inside;
959                }
960            }
961            previous_index = current_index;
962        }
963        inside
964    }

966    /// MVT producers buffer geometry beyond the nominal extent so features
967    /// do not clip at the tile seam, and label layers (`places`) anchor their
968    /// points well past the edge for placement. This fixture's farthest
969    /// anchor is about a quarter-tile out; the check allows a third of a tile.
970    const TILE_BUFFER_TOLERANCE_FRACTION: f64 = 0.34;
971
972    fn assert_tile_bounds_contain(coordinate: &GeographicCoordinate) {
973        let north_west = tile_local_point_to_coordinate(
974            SOURDOUGH_TILE_ADDRESS,
975            4096,
976            &VectorTileGeometryPoint {
977                x_coordinate: 0,
978                y_coordinate: 0,
979            },
980        );
981        let south_east = tile_local_point_to_coordinate(
982            SOURDOUGH_TILE_ADDRESS,
983            4096,
984            &VectorTileGeometryPoint {
985                x_coordinate: 4096,
986                y_coordinate: 4096,
987            },
988        );
989        let longitude_margin =
990            (south_east.longitude - north_west.longitude) * TILE_BUFFER_TOLERANCE_FRACTION;
991        let latitude_margin =
992            (north_west.latitude - south_east.latitude) * TILE_BUFFER_TOLERANCE_FRACTION;
993        assert!(
994            coordinate.longitude >= north_west.longitude - longitude_margin,
995            "longitude {} outside [{}, {}] (margin {longitude_margin})",
996            coordinate.longitude,
997            north_west.longitude,
998            south_east.longitude
999        );
1000        assert!(
1001            coordinate.longitude <= south_east.longitude + longitude_margin,
1002            "longitude {} outside [{}, {}] (margin {longitude_margin})",
1003            coordinate.longitude,
1004            north_west.longitude,
1005            south_east.longitude
1006        );
1007        assert!(
1008            coordinate.latitude <= north_west.latitude + latitude_margin,
1009            "latitude {} outside [{}, {}] (margin {latitude_margin})",
1010            coordinate.latitude,
1011            south_east.latitude,
1012            north_west.latitude
1013        );
1014        assert!(
1015            coordinate.latitude >= south_east.latitude - latitude_margin,
1016            "latitude {} outside [{}, {}] (margin {latitude_margin})",
1017            coordinate.latitude,
1018            south_east.latitude,
1019            north_west.latitude
1020        );
1021    }
