1//! Minimal vector-tile styling (M9, the fill/line/circle slice): a decoded
2//! [`map_vector_tiles::VectorTile`] becomes the renderer's existing overlay
3//! primitives — filled polygons for polygon features, polylines for line
4//! features, point markers for point features — under a small explicit style.
5//!
6//! The style is deliberately data-only: a layer entry names the source layer,
7//! the geometry kind it paints, an optional one-attribute text filter and the
8//! paint values (RGBA, line width, marker radius). No expression language, no
9//! per-symbol layout, no glyph atlas — those are later slices. Hole rings are
10//! skipped because the overlay fill draws an outer ring only; the style
11//! therefore paints exteriors and documents the limitation.
12//!
13//! Coordinates convert from tile-local extent space (`x` right, `y` down, the
14//! MVT convention) through the tile's web-mercator address to geographic
15//! degrees, so the overlays project exactly like every other map coordinate.
16

25/// The longitude span the web-mercator world covers, in degrees.
26const WORLD_LONGITUDE_SPAN_DEGREES: f64 = 360.0;
27/// The longitude at the world's western edge, in degrees.
28const WORLD_WEST_LONGITUDE_DEGREES: f64 = -180.0;
29/// The latitude the mercator projection approaches at the world's edges.
30pub const MERCATOR_MAXIMUM_LATITUDE_DEGREES: f64 = 85.051_128_779_806_59;

32/// The tile address a decoded tile belongs to (its web-mercator position).
33#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
34pub struct VectorTileAddress {
35    /// The tile's zoom level.
36    pub zoom_level: u8,
37    /// The tile's column at that zoom.
38    pub tile_x: u32,
39    /// The tile's row at that zoom.
40    pub tile_y: u32,
41}

43/// The primitive a style layer paints.
44#[derive(Debug, Clone, Copy, PartialEq, Eq)]
45pub enum VectorStyleGeometryKind {
46    /// Polygon features as filled outer rings.

48    /// LineString features as screen-space-thickened polylines.

54/// A one-attribute text filter: the feature must carry the named attribute
55/// with exactly this text value.

60    /// The exact text value to match.

64/// One style layer: source layer, painted kind, optional filter and paint.
65#[derive(Debug, Clone, PartialEq)]
66pub struct VectorLayerStyle {
67    /// The decoded layer name the style paints (`water`, `highways`, …).
68    pub source_layer: String,
69    /// The geometry kind the style paints.
70    pub geometry_kind: VectorStyleGeometryKind,
71    /// An optional one-attribute text filter.
72    pub filter: Option<VectorAttributeFilter>,
73    /// Fill RGBA (used by [`VectorStyleGeometryKind::Fill`]).
74    pub fill_rgba: [f32; 4],
75    /// Line RGBA (used by [`VectorStyleGeometryKind::Line`]).
76    pub line_rgba: [f32; 4],
77    /// Line width in screen pixels.
78    pub line_width_pixels: f64,
79    /// Marker RGBA (used by [`VectorStyleGeometryKind::Circle`]).
80    pub circle_rgba: [f32; 4],
81    /// Marker radius in screen pixels.
82    pub circle_radius_pixels: f64,
83}
84
85impl VectorLayerStyle {
86    /// Builds a fill layer with the given paint.
87    pub fn fill(source_layer: impl Into<String>, rgba_color: [f32; 4]) -> Self {
88        Self::base(source_layer, VectorStyleGeometryKind::Fill, rgba_color)
89    }
90
91    /// Builds a line layer with the given paint and width.
92    pub fn line(source_layer: impl Into<String>, rgba_color: [f32; 4], width_pixels: f64) -> Self {
93        let mut layer_style = Self::base(source_layer, VectorStyleGeometryKind::Line, rgba_color);
94        layer_style.line_rgba = rgba_color;
95        layer_style.line_width_pixels = width_pixels;
96        layer_style
97    }
98
99    /// Builds a circle layer with the given paint and radius.
100    pub fn circle(
101        source_layer: impl Into<String>,
102        rgba_color: [f32; 4],
103        radius_pixels: f64,
104    ) -> Self {
105        let mut layer_style = Self::base(source_layer, VectorStyleGeometryKind::Circle, rgba_color);
106        layer_style.circle_rgba = rgba_color;
107        layer_style.circle_radius_pixels = radius_pixels;
108        layer_style
109    }
110
111    /// Adds the one-attribute text filter.
112    pub fn with_filter(
113        mut self,
114        attribute_name: impl Into<String>,
115        expected_text: impl Into<String>,
116    ) -> Self {
117        self.filter = Some(VectorAttributeFilter {
118            attribute_name: attribute_name.into(),
119            expected_text: expected_text.into(),
120        });
121        self
122    }
123
124    fn base(
125        source_layer: impl Into<String>,
126        geometry_kind: VectorStyleGeometryKind,
127        rgba_color: [f32; 4],
128    ) -> Self {
129        Self {
130            source_layer: source_layer.into(),
131            geometry_kind,
132            filter: None,
133            fill_rgba: rgba_color,
134            line_rgba: rgba_color,
135            line_width_pixels: DEFAULT_VECTOR_LINE_WIDTH_PIXELS,
136            circle_rgba: rgba_color,
137            circle_radius_pixels: DEFAULT_VECTOR_CIRCLE_RADIUS_PIXELS,
138        }
139    }
140}

154/// Converts one tile-local extent point into a geographic coordinate through
155/// the tile's web-mercator address. Vertices at the extent boundary are legal
156/// (they land on the next tile's edge); the latitude clamp keeps the projection
157/// finite at the world's polar edges.
158pub fn tile_local_point_to_coordinate(
159    tile_address: VectorTileAddress,
160    layer_extent: u64,
161    point: &VectorTileGeometryPoint,
162) -> GeographicCoordinate {
163    let extent = layer_extent.max(1) as f64;
164    let tiles_per_axis = (1u64 << tile_address.zoom_level) as f64;
165    let world_x =
166        (f64::from(tile_address.tile_x) + point.x_coordinate as f64 / extent) / tiles_per_axis;
167    let world_y =
168        (f64::from(tile_address.tile_y) + point.y_coordinate as f64 / extent) / tiles_per_axis;
169    let longitude_degrees = world_x * WORLD_LONGITUDE_SPAN_DEGREES + WORLD_WEST_LONGITUDE_DEGREES;
170    let latitude_degrees = (PI * (1.0 - 2.0 * world_y)).sinh().atan().to_degrees();
171    GeographicCoordinate {
172        latitude: latitude_degrees.clamp(
173            -MERCATOR_MAXIMUM_LATITUDE_DEGREES,
174            MERCATOR_MAXIMUM_LATITUDE_DEGREES,
175        ),
176        longitude: longitude_degrees,
177    }
178}

180/// Converts one decoded tile into overlay primitives under the style.
181///
182/// A layer style paints only features of its declared geometry kind whose
183/// filter (when set) matches. Polygon features contribute each exterior ring
184/// (positive shoelace area in the tile's `y`-down space); interior rings are
185/// skipped because the overlay fill has no hole support yet. Every feature
186/// contributes its own overlay, so per-feature paint stays representable.
187pub fn vector_tile_overlays(
188    tile: &VectorTile,
189    tile_address: VectorTileAddress,
190    style: &VectorTileStyle,
191) -> Vec<MapOverlay> {
192    let mut overlays = Vec::new();
193    for layer_style in &style.layers {
194        for layer in tile
195            .layers
196            .iter()
197            .filter(|layer| layer.name == layer_style.source_layer)
198        {
199            for feature in &layer.features {
200                if !feature_matches_filter(feature, layer_style.filter.as_ref()) {
201                    continue;
202                }
203                match (&feature.geometry, layer_style.geometry_kind) {
204                    (VectorTileGeometry::Polygons(rings), VectorStyleGeometryKind::Fill) => {
205                        for exterior_ring in rings.iter().filter(|ring| ring_is_exterior(ring)) {
206                            let ring_coordinates = exterior_ring
207                                .vertices
208                                .iter()
209                                .map(|point| {
210                                    tile_local_point_to_coordinate(
211                                        tile_address,
212                                        layer.extent,
213                                        point,
214                                    )
215                                })
216                                .collect();
217                            overlays.push(MapOverlay::FilledPolygon(FilledPolygonOverlay {
218                                ring_coordinates,
219                                rgba_color: layer_style.fill_rgba,
220                            }));
221                        }
222                    }
223                    (
224                        VectorTileGeometry::LineStrings(line_strings),
225                        VectorStyleGeometryKind::Line,
226                    ) => {
227                        for line_string in line_strings {
228                            let coordinates = line_string
229                                .iter()
230                                .map(|point| {
231                                    tile_local_point_to_coordinate(
232                                        tile_address,
233                                        layer.extent,
234                                        point,
235                                    )
236                                })
237                                .collect();
238                            overlays.push(MapOverlay::RoutePolyline(RoutePolylineOverlay {
239                                coordinates,
240                                rgba_color: layer_style.line_rgba,
241                                width_pixels: layer_style.line_width_pixels,
242                            }));
243                        }
244                    }
245                    (VectorTileGeometry::Points(points), VectorStyleGeometryKind::Circle) => {
246                        let positions = points
247                            .iter()
248                            .map(|point| {
249                                tile_local_point_to_coordinate(tile_address, layer.extent, point)
250                            })
251                            .collect();
252                        overlays.push(MapOverlay::PointMarkers(PointMarkersOverlay {
253                            positions,
254                            rgba_color: layer_style.circle_rgba,
255                            radius_pixels: layer_style.circle_radius_pixels,
256                        }));
257                    }
258                    // A style paints only its declared kind; a feature of
259                    // another kind is simply not this layer's paint.
260                    _ => {}
261                }
262            }
263        }
264    }
265    overlays
266}

273/// Whether the feature carries the filter's attribute with the expected text.
274fn feature_matches_filter(
275    feature: &VectorTileFeature,
276    filter: Option<&VectorAttributeFilter>,
277) -> bool {
278    let Some(attribute_filter) = filter else {
279        return true;
280    };
281    feature
282        .attributes
283        .iter()
284        .any(|(attribute_name, attribute_value)| {
285            attribute_name == &attribute_filter.attribute_name
286                && matches!(
287                    attribute_value,
288                    VectorTileAttributeValue::Text(text) if text == &attribute_filter.expected_text
289                )
290        })
291}

293/// The ring's signed shoelace area in tile-local space; exterior rings are
294/// positive (the specification's `y`-down convention), holes negative.
295fn ring_is_exterior(ring: &VectorTileLinearRing) -> bool {
296    let vertices = &ring.vertices;
297    if vertices.len() < 3 {
298        return false;
299    }
300    let mut doubled_area = 0.0;
301    for vertex_index in 0..vertices.len() {
302        let current = &vertices[vertex_index];
303        let next = &vertices[(vertex_index + 1) % vertices.len()];
304        doubled_area += current.x_coordinate as f64 * next.y_coordinate as f64
305            - next.x_coordinate as f64 * current.y_coordinate as f64;
306    }
307    doubled_area > 0.0
308}

413    /// MVT producers buffer geometry beyond the nominal extent so features
414    /// do not clip at the tile seam, and label layers (`places`) anchor their
415    /// points well past the edge for placement. This fixture's farthest
416    /// anchor is about a quarter-tile out; the check allows a third of a tile.
417    const TILE_BUFFER_TOLERANCE_FRACTION: f64 = 0.34;
418
419    fn assert_tile_bounds_contain(coordinate: &GeographicCoordinate) {
420        let north_west = tile_local_point_to_coordinate(
421            SOURDOUGH_TILE_ADDRESS,
422            4096,
423            &VectorTileGeometryPoint {
424                x_coordinate: 0,
425                y_coordinate: 0,
426            },
427        );
428        let south_east = tile_local_point_to_coordinate(
429            SOURDOUGH_TILE_ADDRESS,
430            4096,
431            &VectorTileGeometryPoint {
432                x_coordinate: 4096,
433                y_coordinate: 4096,
434            },
435        );
436        let longitude_margin =
437            (south_east.longitude - north_west.longitude) * TILE_BUFFER_TOLERANCE_FRACTION;
438        let latitude_margin =
439            (north_west.latitude - south_east.latitude) * TILE_BUFFER_TOLERANCE_FRACTION;
440        assert!(
441            coordinate.longitude >= north_west.longitude - longitude_margin,
442            "longitude {} outside [{}, {}] (margin {longitude_margin})",
443            coordinate.longitude,
444            north_west.longitude,
445            south_east.longitude
446        );
447        assert!(
448            coordinate.longitude <= south_east.longitude + longitude_margin,
449            "longitude {} outside [{}, {}] (margin {longitude_margin})",
450            coordinate.longitude,
451            north_west.longitude,
452            south_east.longitude
453        );
454        assert!(
455            coordinate.latitude <= north_west.latitude + latitude_margin,
456            "latitude {} outside [{}, {}] (margin {latitude_margin})",
457            coordinate.latitude,
458            south_east.latitude,
459            north_west.latitude
460        );
461        assert!(
462            coordinate.latitude >= south_east.latitude - latitude_margin,
463            "latitude {} outside [{}, {}] (margin {latitude_margin})",
464            coordinate.latitude,
465            south_east.latitude,
466            north_west.latitude
467        );
468    }
