1//! The gesture-input camera seam: the delta vocabulary a platform's gesture
2//! recognizer speaks to the SDK (docs/ReferenceAppUX.md §5.1's owner split —
3//! the platform owns gesture recognition, the SDK owns camera math).
4//!
5//! The platform loop recognizes gestures (pointer capture, momentum, pinch →
6//! deltas) and forwards plain deltas; this seam owns every camera adjustment
7//! those deltas imply, reusing the anchored camera's own clamping and
8//! projection math rather than duplicating it. The web playground's
9//! pointer/wheel loop is the reference consumer (M3), and the native
10//! platforms wire their Compose/SwiftUI gesture modifiers onto the same
11//! methods in M5. What stays platform-side by design: gesture recognition
12//! itself, delta shaping (e.g. wheel ticks to zoom levels), inertia after a
13//! gesture ends, and any documented zoom-range policy beyond the camera's
14//! own clamp. Pitch has no delta method yet: no consumer drives tilt by
15//! gesture (the playground's plane toggle applies an absolute pitch through
16//! the camera object); add the method when a pitch gesture consumer exists.
17//!
18//! The focal-point correction is exact, not iterative: a screen pixel's
19//! ground ray direction depends only on bearing, pitch, field of view and
20//! viewport — never on zoom or camera position — so after a zoom or rotate
21//! delta, translating the camera anchor by the ground offset between where
22//! the candidate camera projects the focal pixel and the held ground point
23//! puts the world point back under the focal pixel at any pitch.
24

29/// The gesture-delta camera seam. Every method takes the pose in, answers the
30/// resulting pose out, and never mutates: [`MapCamera`] is small and `Copy`, so
31/// callers keep single ownership of the live pose while applying deltas.
32///
33/// Pan, focal zoom and rotation are the deltas the reference consumer's loop
34/// produces today; the camera's own clamps (zoom `[0, 24]`, pitch
35/// `[0, 75]°`, bearing wrapped into `[0, 360)`) and Mercator latitude bounds
36/// always hold on the answer.
37pub trait CameraInteraction {
38    /// Pans by a pointer travel of `delta_pixels` (screen pixels, x right /
39    /// y down): the ground point the pointer just moved over becomes the new
40    /// camera center, so map content tracks the pointer exactly. Answers the
41    /// camera unchanged when the delta's ground ray misses the plane (above
42    /// the horizon at steep pitch) — the whole-delta rule the reference
43    /// consumer applies.
44    fn pan_by_screen_pixels(
45        &self,
46        delta_pixels: [f64; 2],
47        viewport: &ViewportPixels,
48        field_of_view_degrees: f64,
49    ) -> MapCamera;
50
51    /// Zooms by `delta_zoom_levels` (levels; positive zooms in) keeping the
52    /// world point under `focal_point` fixed under it. Answers the zoom
53    /// without the focal correction when the focal ray misses the ground
54    /// plane (above the horizon at steep pitch). Zoom clamps to the camera's
55    /// own level range before the focal correction applies, so a clamped
56    /// zoom still holds the focal point.
57    fn zoom_by_levels_around_screen_point(
58        &self,
59        delta_zoom_levels: f64,
60        focal_point: [f64; 2],
61        viewport: &ViewportPixels,
62        field_of_view_degrees: f64,
63    ) -> MapCamera;
64
65    /// Rotates the bearing by `delta_bearing_degrees` (degrees clockwise,
66    /// wrapping into `[0, 360)`) keeping the world point under `focal_point`
67    /// fixed under it. Answers the rotation without the focal correction
68    /// when the focal ray misses the ground plane — rotating about the focal
69    /// pixel, the touch-point turn a two-finger gesture implies.
70    fn rotate_by_degrees_around_screen_point(
71        &self,
72        delta_bearing_degrees: f64,
73        focal_point: [f64; 2],
74        viewport: &ViewportPixels,
75        field_of_view_degrees: f64,
76    ) -> MapCamera;
77}

79impl CameraInteraction for MapCamera {
80    fn pan_by_screen_pixels(
81        &self,
82        delta_pixels: [f64; 2],
83        viewport: &ViewportPixels,
84        field_of_view_degrees: f64,
85    ) -> MapCamera {
86        let viewport_center = viewport_center_pixels(viewport);
87        let ground_before =
88            self.screen_to_ground_coordinate(viewport_center, viewport, field_of_view_degrees);
89        let ground_after = self.screen_to_ground_coordinate(
90            [
91                viewport_center[0] - delta_pixels[0],
92                viewport_center[1] - delta_pixels[1],
93            ],
94            viewport,
95            field_of_view_degrees,
96        );
97        let (ground_before, ground_after) = match (ground_before, ground_after) {
98            (Some(before), Some(after)) => (before, after),
99            // A ray above the horizon carries no ground to pan with; the
100            // whole delta drops (the reference consumer's rule).
101            _ => return *self,
102        };
103        let center_shifted =
104            normalized_world_shifted_by_ground_delta(&self.center, &ground_before, &ground_after);
105        MapCamera::new(
106            center_shifted,
107            self.zoom_level,
108            self.bearing_degrees,
109            self.pitch_degrees,
110        )
111    }
112
113    fn zoom_by_levels_around_screen_point(
114        &self,
115        delta_zoom_levels: f64,
116        focal_point: [f64; 2],
117        viewport: &ViewportPixels,
118        field_of_view_degrees: f64,
119    ) -> MapCamera {
120        let held_ground =
121            self.screen_to_ground_coordinate(focal_point, viewport, field_of_view_degrees);
122        let zoomed = MapCamera::new(
123            self.center,
124            self.zoom_level + delta_zoom_levels,
125            self.bearing_degrees,
126            self.pitch_degrees,
127        );
128        camera_holding_focal_ground_point(
129            zoomed,
130            held_ground,
131            focal_point,
132            viewport,
133            field_of_view_degrees,
134        )
135    }
136
137    fn rotate_by_degrees_around_screen_point(
138        &self,
139        delta_bearing_degrees: f64,
140        focal_point: [f64; 2],
141        viewport: &ViewportPixels,
142        field_of_view_degrees: f64,
143    ) -> MapCamera {
144        let held_ground =
145            self.screen_to_ground_coordinate(focal_point, viewport, field_of_view_degrees);
146        let rotated = MapCamera::new(
147            self.center,
148            self.zoom_level,
149            self.bearing_degrees + delta_bearing_degrees,
150            self.pitch_degrees,
151        );
152        camera_holding_focal_ground_point(
153            rotated,
154            held_ground,
155            focal_point,
156            viewport,
157            field_of_view_degrees,
158        )
159    }
160}

162/// The viewport center in pixels (origin top-left, y down).
163fn viewport_center_pixels(viewport: &ViewportPixels) -> [f64; 2] {
164    [viewport.width_pixels * 0.5, viewport.height_pixels * 0.5]
165}
166
167/// Normalized-world ground shift below which a delta is treated as the
168/// identity: the base coordinate stays verbatim instead of a Mercator round
169/// trip whose last-ulp inversion noise would otherwise move the center
170/// (e.g. a zero wheel delta, or a rotation about the viewport center). A
171/// shift this small is ~0.04 millimeters on the ground.
172const IDENTITY_GROUND_DELTA_TOLERANCE: f64 = 1e-12;
173
174/// The `base` coordinate shifted by the normalized-world delta from `before`
175/// to `after`: longitude takes the shortest wrap-around path, latitude rides
176/// the Mercator difference and clamps at the map square like any conversion.
177/// An (effectively) zero delta answers `base` verbatim — the conversion
178/// round trip alone would drift the last mantissa digits of the latitude.
179fn normalized_world_shifted_by_ground_delta(
180    base: &GeographicCoordinate,
181    before: &GeographicCoordinate,
182    after: &GeographicCoordinate,
183) -> GeographicCoordinate {
184    let base_world = coordinate_to_normalized_world(base);
185    let before_world = coordinate_to_normalized_world(before);
186    let after_world = coordinate_to_normalized_world(after);
187    let x_delta = crate::camera::normalized_shortest_delta(before_world[0], after_world[0]);
188    let y_delta = after_world[1] - before_world[1];
189    if x_delta.abs() <= IDENTITY_GROUND_DELTA_TOLERANCE
190        && y_delta.abs() <= IDENTITY_GROUND_DELTA_TOLERANCE
191    {
192        return *base;
193    }
194    normalized_world_to_coordinate([base_world[0] + x_delta, base_world[1] + y_delta])
195}

197/// The `candidate` camera shifted so the world point `held_ground` stays
198/// under `focal_point`: the pixel ray of `focal_point` keeps its direction
199/// across the shift (bearing, pitch, field of view and viewport are
200/// unchanged), so the anchor translation that puts `held_ground` back under
201/// the focal pixel is the ground offset between held and where `candidate`
202/// unprojects the focal pixel. Answers `candidate` unchanged when either ray
203/// misses the ground plane — the focal correction is then impossible, not
204/// skipped-on-purpose.
205fn camera_holding_focal_ground_point(
206    candidate: MapCamera,
207    held_ground: Option<GeographicCoordinate>,
208    focal_point: [f64; 2],
209    viewport: &ViewportPixels,
210    field_of_view_degrees: f64,
211) -> MapCamera {
212    let candidate_focal_ground =
213        candidate.screen_to_ground_coordinate(focal_point, viewport, field_of_view_degrees);
214    let (held_ground, candidate_focal_ground) = match (held_ground, candidate_focal_ground) {
215        (Some(held), Some(focal_ground)) => (held, focal_ground),
216        _ => return candidate,
217    };
218    let center_shifted = normalized_world_shifted_by_ground_delta(
219        &candidate.center,
220        &candidate_focal_ground,
221        &held_ground,
222    );
223    MapCamera::new(
224        center_shifted,
225        candidate.zoom_level,
226        candidate.bearing_degrees,
227        candidate.pitch_degrees,
228    )
229}

240    fn coordinate(latitude: f64, longitude: f64) -> GeographicCoordinate {
241        GeographicCoordinate::from_degrees(latitude, longitude).unwrap()

244    fn viewport() -> ViewportPixels {
245        ViewportPixels::new(1024.0, 768.0).unwrap()
246    }
247
248    /// Headroom for the focal point's world point landing back under its
249    /// pixel after a delta; matrix round trips at zoom 14 permit
250    /// sub-hundredth-pixel drift (the same class the camera tests carry).
251    const FOCAL_HOLD_TOLERANCE_PIXELS: f64 = 1e-2;
252    /// Headroom for the zoom delta arriving exactly (clamps pass the level
253    /// through `MapCamera::new` unmodified in range).
254    const ZOOM_LEVEL_TOLERANCE: f64 = 1e-9;
255    /// Headroom for bearing wrap arithmetic on the rotate answers.
256    const BEARING_DEGREES_TOLERANCE: f64 = 1e-9;
257
258    /// The fixture camera: the recorded Zurich neighborhood at the
259    /// playground's initial zoom, pitched and turned so the corrections run
260    /// under a real perspective, not only top-down.
261    fn fixture_camera() -> MapCamera {
262        MapCamera::new(coordinate(47.3769, 8.5417), 14.0, 25.0, 30.0)
263    }
264
265    /// The focal pixel of the shared tests: a quarter of the viewport in
266    /// from the top left, visibly off the center so the hold correction has
267    /// work to do.
268    fn focal_point(viewport: &ViewportPixels) -> [f64; 2] {
269        [viewport.width_pixels * 0.25, viewport.height_pixels * 0.25]
270    }
271
272    fn assert_points_agree(first: [f64; 2], second: [f64; 2], tolerance_pixels: f64) {
273        assert!(
274            (first[0] - second[0]).hypot(first[1] - second[1]) <= tolerance_pixels,
275            "screen points must agree within {tolerance_pixels} px: {first:?} vs {second:?}",
276        );
277    }

279    #[test]
280    fn pan_tracks_the_pointers_ground_point_to_the_viewport_center() {
281        let camera = fixture_camera();
282        let viewport = viewport();
283        let delta_pixels = [64.0, -48.0];
284        let viewport_center = viewport_center_pixels(&viewport);
285        let pointer_start = [
286            viewport_center[0] - delta_pixels[0],
287            viewport_center[1] - delta_pixels[1],
288        ];
289        let pointer_ground = camera
290            .screen_to_ground_coordinate(pointer_start, &viewport, DEFAULT_FIELD_OF_VIEW_DEGREES)
291            .unwrap();
292        let panned =
293            camera.pan_by_screen_pixels(delta_pixels, &viewport, DEFAULT_FIELD_OF_VIEW_DEGREES);
294        // The pointer traveled from `pointer_start` to the center; the ground
295        // point under its start must now sit at the center.
296        assert_points_agree(
297            panned
298                .project_to_screen(&pointer_ground, &viewport, DEFAULT_FIELD_OF_VIEW_DEGREES)
299                .unwrap(),
300            viewport_center,
301            FOCAL_HOLD_TOLERANCE_PIXELS,
302        );
303        // Zoom, bearing and pitch ride through unchanged.
304        assert_eq!(panned.zoom_level, camera.zoom_level);
305        assert_eq!(panned.bearing_degrees, camera.bearing_degrees);
306        assert_eq!(panned.pitch_degrees, camera.pitch_degrees);
307    }

328    #[test]
329    fn focal_zoom_keeps_the_world_point_under_the_focal_pixel() {
330        let camera = fixture_camera();
331        let viewport = viewport();
332        let focal = focal_point(&viewport);
333        let held_ground = camera
334            .screen_to_ground_coordinate(focal, &viewport, DEFAULT_FIELD_OF_VIEW_DEGREES)
335            .unwrap();
336        let zoomed = camera.zoom_by_levels_around_screen_point(
337            1.0,
338            focal,
339            &viewport,
340            DEFAULT_FIELD_OF_VIEW_DEGREES,
341        );
342        assert!((zoomed.zoom_level - (camera.zoom_level + 1.0)).abs() < ZOOM_LEVEL_TOLERANCE);
343        assert_points_agree(
344            zoomed
345                .project_to_screen(&held_ground, &viewport, DEFAULT_FIELD_OF_VIEW_DEGREES)
346                .unwrap(),
347            focal,
348            FOCAL_HOLD_TOLERANCE_PIXELS,
349        );
350        // The center moved toward the held point's ground position (a
351        // zoom-in pans the anchor toward the point being held) but zoom,
352        // bearing and pitch carry through.
353        assert_eq!(zoomed.bearing_degrees, camera.bearing_degrees);
354        assert_eq!(zoomed.pitch_degrees, camera.pitch_degrees);
355        let camera_world = crate::projection::coordinate_to_normalized_world(&camera.center);
356        let zoomed_world = crate::projection::coordinate_to_normalized_world(&zoomed.center);
357        // The fixture focal sits left of and above the center (northwest on
358        // the ground at bearing 25°), so the anchor must move the same way.
359        assert!(zoomed_world[0] < camera_world[0] || zoomed_world[1] < camera_world[1]);
360    }

362    #[test]
363    fn focal_zoom_clamps_to_the_camera_zoom_range() {
364        let camera = MapCamera::new(coordinate(47.3769, 8.5417), 14.0, 0.0, 0.0);
365        let viewport = viewport();
366        // Zoom-in clamp against an off-center focal: top-down, so the whole
367        // held-point correction is affine and stays inside the map square.
368        let held_ground = camera
369            .screen_to_ground_coordinate(
370                focal_point(&viewport),
371                &viewport,
372                DEFAULT_FIELD_OF_VIEW_DEGREES,
373            )
374            .unwrap();
375        let zoomed_in = camera.zoom_by_levels_around_screen_point(
376            100.0,
377            focal_point(&viewport),
378            &viewport,
379            DEFAULT_FIELD_OF_VIEW_DEGREES,
380        );
381        assert_eq!(zoomed_in.zoom_level, MAXIMUM_ZOOM_LEVEL);
382        assert_points_agree(
383            zoomed_in
384                .project_to_screen(&held_ground, &viewport, DEFAULT_FIELD_OF_VIEW_DEGREES)
385                .unwrap(),
386            focal_point(&viewport),
387            FOCAL_HOLD_TOLERANCE_PIXELS,
388        );
389        // Zoom-out clamp about the viewport center: at minimum zoom the view
390        // footprint spans whole world copies, where the held-point
391        // correction is out of square — a center focal keeps the test honest
392        // and asserts the clamp itself.
393        let zoomed_out = camera.zoom_by_levels_around_screen_point(
394            -100.0,
395            viewport_center_pixels(&viewport),
396            &viewport,
397            DEFAULT_FIELD_OF_VIEW_DEGREES,
398        );
399        assert_eq!(zoomed_out.zoom_level, MINIMUM_ZOOM_LEVEL);
400        assert_eq!(zoomed_out.center, camera.center);
401        assert_eq!(zoomed_out.bearing_degrees, camera.bearing_degrees);
402        assert_eq!(zoomed_out.pitch_degrees, camera.pitch_degrees);
403    }
