1//! Headless `wgpu` rendering of raster tiles plus deterministic frame
2//! capture.
3//!
4//! The renderer draws one textured quad per pooled tile for a single frame
5//! into an off-screen render-attachment texture and reads the pixels back
6//! into a [`CapturedFrame`]; an optional overlay layer
7//! ([`crate::overlay::MapOverlay`]) draws after the quads in the same render
8//! pass through a separate solid-color pipeline. Pooled payloads that fail to
9//! decode are skipped
10//! for the frame and evicted from the pool so the scheduler refetches them —
11//! one corrupt tile never poisons the frames after it; only structural
12//! failures (an overflowing draw list, a failed context acquisition or
13//! capture) are errors. No surface is requested, so this works on any native
14//! backend (Metal, Vulkan, DirectX): device acquisition lives behind the
15//! object-safe [`GraphicsContextSource`] seam, keeping `wgpu` types out of
16//! the scheduling core and letting tests substitute their own context source
17//! or skip rendering when no adapter exists.
18//!
19//! This module is native-only: blocking on the asynchronous `wgpu`
20//! initialization calls and polling the device is not valid on wasm, where
21//! the browser's own async runtime drives `webgpu`.
22//!
23//! # Capturing snapshots (MapLibre `snapshot()` / `snapshotSync()`)
24//!
25//! MapLibre renders the current frame into a bitmap asynchronously
26//! (`snapshot()`) or synchronously (`snapshotSync()`), and GL JS fires its
27//! `idle` event precisely when no tile loading or animation work remains —
28//! the pair exists so captures are deterministic (await the idle state, then
29//! snapshot without racing late work). Reading that pattern against what the
30//! headless path actually is:
31//!
32//! * Every capture here is already snapshot-shaped: frames compose into an
33//!   off-screen render attachment that is never presented to a window, and
34//!   [`capture_frame_bytes`] reads the result back — the same separation of
35//!   scene composition from presented output the MapLibre snapshot relies
36//!   on, with no capture machinery of a live view to sidestep.
37//! * There is no async/sync split to port, because no asynchronous runtime
38//!   drives the device: a capture is inherently blocking on the readback
39//!   ([`capture_frame_bytes`] escalates to a blocking device poll), so an
40//!   async `snapshot()` twin would be ceremony over the same call.
41//! * What is actually missing for deterministic captures is the *idle
42//!   gate*. It is implemented as
43//!   [`HeadlessTileRenderer::snapshot_frame_when_idle`], which refuses to
44//!   render while pending tile work remains — returning
45//!   [`HeadlessRenderError::PendingTileWork`] immediately, without touching
46//!   the GPU — and otherwise behaves exactly like the ordinary capture. That
47//!   makes it the non-blocking half: the refusal decision costs no device
48//!   work and never waits for pending work to finish; completing the work is
49//!   the caller's next prepare frame. MapLibre's un-gated synchronous
50//!   variant is the ordinary capture
51//!   ([`HeadlessTileRenderer::render_captured_frame_with_overlays`]) — it
52//!   always renders and captures, however much tile work is still pending.
53//!
54//! The idle gate covers tile fetch state only; camera animations are
55//! composed outside this renderer entirely (see
56//! [`crate::tile_scheduler`] on the full idle semantics).

107/// WGSL pipeline: a single textured-quad shader. Vertex positions arrive
108/// already in clip space (the camera transform runs on the CPU in `f64`; see
109/// [`crate::renderer::Renderer::tile_vertices_in_clip_space`]), so the vertex
110/// stage is a pass-through and the fragment stage samples the tile texture.
111const TILE_QUAD_SHADER_SOURCE: &str = "
112struct VertexOutput {
113    @builtin(position) clip_position: vec4<f32>,
114    @location(0) texture_coordinates: vec2<f32>,
115};
116
117@group(0)
118@binding(0)
119var tile_texture: texture_2d<f32>;
120
121@group(0)
122@binding(1)
123var tile_sampler: sampler;
124
125@vertex
126fn vertex_main(@location(0) clip_position: vec4<f32>, @location(1) texture_coordinates: vec2<f32>) -> VertexOutput {
127    var output: VertexOutput;
128    output.clip_position = clip_position;
129    output.texture_coordinates = texture_coordinates;
130    return output;
131}
132
133@fragment
134fn fragment_main(vertex: VertexOutput) -> @location(0) vec4<f32> {
135    return textureSample(tile_texture, tile_sampler, vertex.texture_coordinates);
136}
137";
138
139/// WGSL pipeline for the overlay layer: solid-color triangles in clip space,
140/// drawn after the tile quads in the same render pass through standard alpha
141/// blending (translucent overlay colors composite over the base tiles).
142/// Vertex positions arrive already in clip space — the camera transform runs
143/// on the CPU in `f64` (see [`crate::overlay::overlay_triangle_vertices`] and
144/// the documented wasm canvas mirror: same vertex layout and shader text
145/// there) — so the vertex stage is a pass-through and the fragment stage
146/// writes the interpolated vertex color.
147const OVERLAY_SHADER_SOURCE: &str = "
148struct VertexOutput {
149    @builtin(position) clip_position: vec4<f32>,
150    @location(0) rgba_color: vec4<f32>,
151};
152
153@vertex
154fn vertex_main(@location(0) clip_position: vec4<f32>, @location(1) rgba_color: vec4<f32>) -> VertexOutput {
155    var output: VertexOutput;
156    output.clip_position = clip_position;
157    output.rgba_color = rgba_color;
158    return output;
159}
160
161@fragment
162fn fragment_main(vertex: VertexOutput) -> @location(0) vec4<f32> {
163    return vertex.rgba_color;
164}
165";

167/// An adapter/device/queue triple acquired for headless rendering.
168#[derive(Debug)]
169pub struct GraphicsContext {
170    /// Logical device the renderer creates resources on.
171    pub device: Device,
172    /// Queue the renderer submits command buffers through.
173    pub queue: Queue,
174    /// Adapter description, naming the backend that renders the frames.
175    pub adapter_information: AdapterInfo,
176}

178/// Why no graphics context could be acquired.
179#[derive(Debug, Clone, PartialEq, Eq)]
180pub enum GraphicsContextError {
181    /// No rendering adapter matched the request (for example on a machine
182    /// without a usable GPU backend).
183    AdapterUnavailable(String),
184    /// An adapter was found but the device request failed.
185    DeviceRequestFailed(String),
186}
187
188impl std::fmt::Display for GraphicsContextError {
189    fn fmt(&self, formatter: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
190        match self {
191            GraphicsContextError::AdapterUnavailable(detail) => {
192                write!(formatter, "graphics adapter unavailable: {detail}")
193            }
194            GraphicsContextError::DeviceRequestFailed(detail) => {
195                write!(formatter, "graphics device request failed: {detail}")
196            }
197        }
198    }
199}

201/// The dependency-inversion seam for device acquisition: tests substitute
202/// their own source (a scripted renderer, a pinned backend, or a context
203/// created elsewhere) instead of relying on environment adapter discovery.
204pub trait GraphicsContextSource: Send + Sync {
205    /// Acquires a device/queue pair for headless rendering, or explains why
206    /// none was available.
207    fn request_graphics_context(&self) -> Result<GraphicsContext, GraphicsContextError>;
208}

210/// The production source: requests the first available adapter with no
211/// compatible surface, blocking on it with `pollster` (native only; see the
212/// module documentation).
213pub struct WgpuGraphicsContextSource;
214
215impl GraphicsContextSource for WgpuGraphicsContextSource {
216    fn request_graphics_context(&self) -> Result<GraphicsContext, GraphicsContextError> {
217        let instance = wgpu::Instance::new(wgpu::InstanceDescriptor::new_without_display_handle());
218        let adapter = pollster::block_on(instance.request_adapter(&wgpu::RequestAdapterOptions {
219            power_preference: wgpu::PowerPreference::default(),
220            compatible_surface: None,
221            force_fallback_adapter: false,
222            apply_limit_buckets: false,
223        }))
224        .map_err(|error| GraphicsContextError::AdapterUnavailable(error.to_string()))?;
225
226        let (device, queue) = pollster::block_on(adapter.request_device(&wgpu::DeviceDescriptor {
227            label: Some("map-rendering-headless"),
228            required_features: wgpu::Features::empty(),
229            // See `surface.rs` for why the downlevel defaults are requested
230            // rather than `Limits::default()`: some Metal adapters cap
231            // `max_inter_stage_shader_variables` at 15, one below the
232            // default request, which would reject the device outright.
233            required_limits: wgpu::Limits::downlevel_defaults(),
234            experimental_features: wgpu::ExperimentalFeatures::disabled(),
235            memory_hints: wgpu::MemoryHints::default(),
236            trace: wgpu::Trace::Off,
237        }))
238        .map_err(|error| GraphicsContextError::DeviceRequestFailed(error.to_string()))?;
239
240        Ok(GraphicsContext {
241            device,
242            queue,
243            adapter_information: adapter.get_info(),
244        })
245    }
246}

248/// Why one frame could not be rendered or captured.
249#[derive(Debug, Clone, PartialEq, Eq)]
250pub enum HeadlessRenderError {
251    /// The graphics context could not be acquired at all: either no adapter
252    /// matched ([`GraphicsContextError::AdapterUnavailable`]) or the device
253    /// request failed. Kept distinguishable from a per-frame capture failure
254    /// so callers can tell "no GPU here" from "this frame broke".
255    GraphicsContextUnavailable(GraphicsContextError),
256    /// The draw list would overflow the 16-bit index buffer.
257    DrawListTooLarge {
258        /// Draw quads (world-copy fragments) assembled for the frame.
259        quad_count: usize,
260        /// Draw quads the index buffer supports.
261        maximum_quad_count: usize,
262    },
263    /// The rendered frame could not be captured back to the CPU.
264    FrameCapture(String),
265    /// The idle-gated snapshot refused to render: pending tile work remains,
266    /// so the frame's pixels could still change through scheduled work.
267    /// Reports how many requested tiles are not pooled at their own zoom
268    /// level — verdicts other than `Present`, or `Present` verdicts whose
269    /// payload left the pool since. Nothing was rendered; retrying after the
270    /// scheduler's next prepare frame resolves the pending tiles.
271    PendingTileWork {
272        /// Requested tiles still lacking their own pooled payload.
273        pending_tile_count: usize,
274    },
275}
276
277impl From<GraphicsContextError> for HeadlessRenderError {
278    fn from(error: GraphicsContextError) -> Self {
279        HeadlessRenderError::GraphicsContextUnavailable(error)
280    }
281}

283/// One rendered frame copied back from the GPU: tightly packed RGBA8,
284/// row-major, origin top-left, no row padding.
285#[derive(Debug, Clone, PartialEq, Eq)]
286pub struct CapturedFrame {
287    /// Frame width in pixels.
288    pub width_pixels: u32,
289    /// Frame height in pixels.
290    pub height_pixels: u32,
291    /// `width * height * 4` RGBA bytes, rows tightly packed.
292    pub rgba_bytes: Vec<u8>,
293}
294
295/// Outcome of the staging buffer's map callback, handed to the frame reader
296/// over a channel to bridge the `wgpu` callback into synchronous code.
297type BufferMapOutcome = Result<(), String>;
298
299/// The row stride granularity texture-to-buffer copies require: every row of
300/// a copied texture starts on a 256-byte boundary
301/// ([`wgpu::COPY_BYTES_PER_ROW_ALIGNMENT`]), so partial rows round up to it.
302const ROW_ALIGNMENT_BYTES: usize = 256;
303
304/// Bytes per buffer row after the mandatory copy alignment above: each row of
305/// a texture-to-buffer copy starts on a 256-byte boundary, so partial rows
306/// round up.
307fn aligned_bytes_per_row(width_pixels: u32) -> usize {
308    let unpadded_row_bytes = width_pixels as usize * PIXEL_BYTES as usize;
309    unpadded_row_bytes.div_ceil(ROW_ALIGNMENT_BYTES) * ROW_ALIGNMENT_BYTES
310}

560    /// The idle-gated snapshot: refuses to render while pending tile work
561    /// remains, and otherwise captures exactly the ordinary frame (see the
562    /// module documentation for how MapLibre's snapshot pair maps onto the
563    /// headless path).
564    ///
565    /// The gate is the idle render-state signal applied to this frame's own
566    /// request set at capture time, so it needs no scheduler around: a
567    /// requested tile is pending when its prepared verdict is anything but
568    /// [`crate::tile_scheduler::TileReadiness::Present`] — a parent fallback
569    /// still owes the child's
570    /// detail, an `Unavailable` fetch is retried later, a `Deferred` tile is
571    /// fetched later — and also when a `Present` verdict's payload left the
572    /// pool between the prepare frame and this snapshot, whose area would
573    /// otherwise capture as a hole pixels the next fetch could fill. When
574    /// the verdict is [`crate::prepared_frame_is_idle`] and every pooled
575    /// payload is still in the pool, the frame's pixels can no longer change
576    /// through scheduled work and the frame renders and captures — byte-for-
577    /// byte the ordinary capture's behavior, so the two are intercomparable.
578    ///
579    /// The refusal is decided before any GPU work: an unresolved frame
580    /// returns [`HeadlessRenderError::PendingTileWork`] immediately and
581    /// never waits for the pending work to complete. An empty request set is
582    /// idle by vacuity, so an empty-frame snapshot renders the clear color
583    /// (as does the ordinary capture).
584    pub fn snapshot_frame_when_idle(
585        &mut self,
586        camera: &MapCamera,
587        viewport: &ViewportPixels,
588        prepared_tiles: &[PreparedTile],
589        pool: &mut TileTexturePool,
590        overlays: &[MapOverlay],
591    ) -> Result<CapturedFrame, HeadlessRenderError> {
592        let pending_tile_count = idle_gate::pending_tile_count(prepared_tiles, pool);
593        if pending_tile_count > 0 {
594            return Err(HeadlessRenderError::PendingTileWork { pending_tile_count });
595        }
596        self.render_captured_frame_with_overlays(camera, viewport, prepared_tiles, pool, overlays)
597    }

599    /// Renders one frame — the pooled tiles' textured quads plus the overlay
600    /// layer on top — and captures its pixels. The three-layer variant adds
601    /// the location-indicator layer; this signature keeps the two-layer
602    /// contract byte-stable by delegating with no indicator.
603    pub fn render_captured_frame_with_overlays(
604        &mut self,
605        camera: &MapCamera,
606        viewport: &ViewportPixels,
607        prepared_tiles: &[PreparedTile],
608        pool: &mut TileTexturePool,
609        overlays: &[MapOverlay],
610    ) -> Result<CapturedFrame, HeadlessRenderError> {
611        self.render_captured_frame_layers(camera, viewport, prepared_tiles, pool, overlays, None)
612    }
613
614    /// The frame's explicit three-layer variant: the pooled tiles' textured
615    /// quads, then the overlay layer, then the location-indicator layer
616    /// (the indicator draws last, the frame's explicit top layer — see
617    /// [`crate::location_indicator::LOCATION_INDICATOR_TOP_LAYER_DRAW_ORDER`]).
618    ///
619    /// The frame's quads are assembled GPU-free by
620    /// [`crate::renderer::Renderer::frame_quads`]; each pooled payload is
621    /// uploaded once — either freshly decoded, or reused from
622    /// [`GpuTileTextureCache`] when the pool's payload revision still matches
623    /// the upload's stamp — in prepared order, so later detailed tiles paint
624    /// over any earlier fallback content they overlap. Uploads whose bytes
625    /// the pool replaced or dropped since are pruned at frame start. Each
626    /// quad may project into up to two world-copy fragments, which are
627    /// separate draw quads sharing the quad's texture. A payload that fails
628    /// to decode — or one that left the pool between assembly and upload —
629    /// skips its quads and, when corrupt, is evicted from the pool so the
630    /// scheduler refetches it; the rest of the frame renders normally.
631    ///
632    /// After the tile draws, the overlay layer — the clip-space solid-color
633    /// triangles assembled GPU-free by [`crate::overlay::overlay_triangle_vertices`]
634    /// through the same camera pose — draws through a second pipeline into
635    /// the same render attachment, alpha-blended over the tiles. An empty
636    /// overlay list contributes no draw at all, keeping such frames identical
637    /// to the tiles-only pipeline's.
638    ///
639    /// The location-indicator layer draws third — the clip-space solid-color
640    /// triangles assembled GPU-free by
641    /// [`crate::location_indicator::location_indicator_vertices`] through the
642    /// same camera pose, drawn through the same overlay pipeline
643    /// (shader/layout-identical) and alpha-blended over what the first two
644    /// layers drew. A `None` indicator creates no buffer and contributes no
645    /// draw, keeping indicator-free frames identical to the two-layer
646    /// pipeline's.
647    pub fn render_captured_frame_layers(
648        &mut self,
649        camera: &MapCamera,
650        viewport: &ViewportPixels,
651        prepared_tiles: &[PreparedTile],
652        pool: &mut TileTexturePool,
653        overlays: &[MapOverlay],
654        location_indicator_layer: Option<&crate::location_indicator::LocationIndicatorLayer>,
655    ) -> Result<CapturedFrame, HeadlessRenderError> {
656        let quads = self
657            .quad_composer
658            .frame_quads(camera, viewport, prepared_tiles, pool);
659
660        // Drop uploads the pool invalidated — bytes replaced or evicted since
661        // their texture was uploaded — before deciding what needs uploading.
662        self.gpu_texture_cache
663            .retain_matching(|tile_key, pool_revision| {
664                pool.payload_revision(tile_key) == Some(pool_revision)
665            });
666        // Touch the recency of everything this frame draws before any upload,
667        // so an upload's eviction pass below cannot remove an entry a later
668        // quad of this same frame reuses.
669        for quad in &quads {
670            self.gpu_texture_cache.touch(&quad.texture_tile_key);
671        }
672
673        // The device and queue handles are cheap reference-counted clones;
674        // borrowing them out of the context instead would pin the shared
675        // borrow of `self` across the mutable upload path below.
676        let device = self.graphics_context.device.clone();
677        let queue = self.graphics_context.queue.clone();
678
679        // One vertex array per world-copy fragment, each tagged with the quad
680        // whose texture it draws.
681        let quad_fragment_vertices = self
682            .quad_composer
683            .quad_fragment_vertices(camera, viewport, &quads);
684        let mut fragment_vertices = Vec::new();
685        let mut fragment_texture_positions = Vec::with_capacity(quad_fragment_vertices.len());
686        for (quad_index, quad_fragments) in quad_fragment_vertices.iter().enumerate() {
687            for quad_fragment in quad_fragments {
688                fragment_texture_positions.push(quad_index);
689                fragment_vertices.push(*quad_fragment);
690            }
691        }
692        if fragment_vertices.len() > MAXIMUM_QUADS_PER_FRAME {
693            return Err(HeadlessRenderError::DrawListTooLarge {
694                quad_count: fragment_vertices.len(),
695                maximum_quad_count: MAXIMUM_QUADS_PER_FRAME,
696            });
697        }
698
699        // Upload once per quad; every world-copy fragment of the quad draws
700        // with the same texture bind group. A quad whose payload is missing
701        // or undecodable contributes no draw (see `upload_tile_texture`).
702        let mut tile_textures_and_bind_groups = Vec::with_capacity(quads.len());
703        for quad in &quads {
704            tile_textures_and_bind_groups.push(self.upload_tile_texture(
705                &device,
706                quad.texture_tile_key,
707                pool,
708            )?);
709        }
710
711        let (vertex_bytes, index_bytes) = encoded_fragment_buffers(&fragment_vertices);
712
713        // An all-missing frame has no draw list; its buffers would be empty,
714        // which `wgpu` buffers cannot represent, so only buffers of a frame
715        // that draws are created and the all-missing frame renders as the
716        // clear color alone.
717        let vertex_buffer = (!fragment_vertices.is_empty()).then(|| {
718            create_filled_buffer(
719                &device,
720                &queue,
721                &vertex_bytes,
722                wgpu::BufferUsages::VERTEX,
723                Some("map-rendering-tile-vertices"),
724            )
725        });
726        let index_buffer = (!fragment_vertices.is_empty()).then(|| {
727            create_filled_buffer(
728                &device,
729                &queue,
730                &index_bytes,
731                wgpu::BufferUsages::INDEX,
732                Some("map-rendering-tile-indices"),
733            )
734        });
735
736        // The overlay layer assembles through the same camera pose as the
737        // tile quads and draws after them; an empty overlay list creates no
738        // buffer and contributes no draw.
739        let overlay_vertices = overlay_triangle_vertices(
740            camera,
741            viewport,
742            self.quad_composer.field_of_view_degrees(),
743            overlays,
744        );
745        let overlay_vertex_buffer = (!overlay_vertices.is_empty()).then(|| {
746            create_filled_buffer(
747                &device,
748                &queue,
749                &encoded_overlay_vertices(&overlay_vertices),
750                wgpu::BufferUsages::VERTEX,
751                Some("map-rendering-overlay-vertices"),
752            )
753        });
754
755        // The location-indicator layer assembles through the same camera pose
756        // and draws through the same overlay pipeline after the overlays; a
757        // cleared (None) indicator creates no buffer and draws nothing.
758        let indicator_vertices = crate::location_indicator::location_indicator_vertices(
759            camera,
760            viewport,
761            self.quad_composer.field_of_view_degrees(),
762            location_indicator_layer,
763        );
764        let indicator_vertex_buffer = (!indicator_vertices.is_empty()).then(|| {
765            create_filled_buffer(
766                &device,
767                &queue,
768                &encoded_overlay_vertices(&indicator_vertices),
769                wgpu::BufferUsages::VERTEX,
770                Some("map-rendering-indicator-vertices"),
771            )
772        });
773
774        let width_pixels = viewport.width_pixels as u32;
775        let height_pixels = viewport.height_pixels as u32;
776        let frame_texture = device.create_texture(&wgpu::TextureDescriptor {
777            label: Some("map-rendering-frame"),
778            size: Extent3d {
779                width: width_pixels,
780                height: height_pixels,
781                depth_or_array_layers: 1,
782            },
783            mip_level_count: 1,
784            sample_count: 1,
785            dimension: wgpu::TextureDimension::D2,
786            format: FRAME_TEXTURE_FORMAT,
787            usage: TextureUsages::RENDER_ATTACHMENT | TextureUsages::COPY_SRC,
788            view_formats: &[],
789        });
790        let frame_texture_view = frame_texture.create_view(&wgpu::TextureViewDescriptor::default());
791
792        let mut command_encoder = device.create_command_encoder(&wgpu::CommandEncoderDescriptor {
793            label: Some("map-rendering-frame-encoder"),
794        });
795        {
796            let mut render_pass = command_encoder.begin_render_pass(&wgpu::RenderPassDescriptor {
797                label: Some("map-rendering-frame-pass"),
798                color_attachments: &[Some(wgpu::RenderPassColorAttachment {
799                    view: &frame_texture_view,
800                    resolve_target: None,
801                    depth_slice: None,
802                    ops: wgpu::Operations {
803                        load: wgpu::LoadOp::Clear(wgpu::Color {
804                            r: f64::from(CLEAR_COLOR_RGBA[0]),
805                            g: f64::from(CLEAR_COLOR_RGBA[1]),
806                            b: f64::from(CLEAR_COLOR_RGBA[2]),
807                            a: f64::from(CLEAR_COLOR_RGBA[3]),
808                        }),
809                        store: wgpu::StoreOp::Store,
810                    },
811                })],
812                depth_stencil_attachment: None,
813                timestamp_writes: None,
814                occlusion_query_set: None,
815                multiview_mask: None,
816            });
817            if !fragment_vertices.is_empty() {
818                render_pass.set_pipeline(&self.tile_pipeline);
819                render_pass.set_vertex_buffer(0, vertex_buffer.expect("frame draws").slice(..));
820                render_pass.set_index_buffer(
821                    index_buffer.expect("frame draws").slice(..),
822                    wgpu::IndexFormat::Uint16,
823                );
824                for (fragment_index, texture_position) in
825                    fragment_texture_positions.iter().copied().enumerate()
826                {
827                    let Some((_, bind_group)) = &tile_textures_and_bind_groups[texture_position]
828                    else {
829                        continue;
830                    };
831                    render_pass.set_bind_group(0, bind_group, &[]);
832                    let first_index = u32::try_from(fragment_index * INDICES_PER_QUAD)
833                        .expect("the draw list was bounded above");
834                    render_pass.draw_indexed(
835                        first_index..first_index + INDICES_PER_QUAD as u32,
836                        0,
837                        0..1,
838                    );
839                }
840            }
841            if let Some(overlay_vertex_buffer) = &overlay_vertex_buffer {
842                render_pass.set_pipeline(&self.overlay_pipeline);
843                render_pass.set_vertex_buffer(0, overlay_vertex_buffer.slice(..));
844                let overlay_vertex_count = u32::try_from(overlay_vertices.len())
845                    .expect("the overlay triangle list stays inside u32");
846                render_pass.draw(0..overlay_vertex_count, 0..1);
847            }
848            if let Some(indicator_vertex_buffer) = &indicator_vertex_buffer {
849                render_pass.set_pipeline(&self.overlay_pipeline);
850                render_pass.set_vertex_buffer(0, indicator_vertex_buffer.slice(..));
851                let indicator_vertex_count = u32::try_from(indicator_vertices.len())
852                    .expect("the indicator triangle list stays inside u32");
853                render_pass.draw(0..indicator_vertex_count, 0..1);
854            }
855        }
856
857        capture_frame_bytes(&device, &queue, command_encoder, &frame_texture)
858            .map_err(HeadlessRenderError::FrameCapture)
859    }
860
861    /// Uploads a pooled payload as a renderable texture, touching the pool's
862    /// recency for the tile on both the fresh-load and cached-reuse paths.
863    /// Returns `Ok(None)` — no error — for two skip
864    /// cases: the payload left the pool between assembly and upload (an
865    /// eviction raced the frame), or the payload fails to decode, in which
866    /// case the corrupt bytes are evicted so the scheduler's next fetch
867    /// replaces them and every later frame recovers. Otherwise returns the
868    /// texture and bind group, freshly uploaded on a payload-revision miss or
869    /// reused from the GPU texture cache on a hit.
870    fn upload_tile_texture(
871        &mut self,
872        device: &Device,
873        tile_key: TileKey,
874        pool: &mut TileTexturePool,
875    ) -> Result<Option<(Texture, wgpu::BindGroup)>, HeadlessRenderError> {
876        // The revision stamps the cache entry, so a pool miss also misses the
877        // cache: an upload whose source bytes are gone must never draw.
878        let Some(payload_revision) = pool.payload_revision(&tile_key) else {
879            return Ok(None);
880        };
881        if let Some((texture, bind_group)) =
882            self.gpu_texture_cache.lookup(&tile_key, payload_revision)
883        {
884            // Drawing from the cache is still a use of the pooled bytes: keep
885            // the payload's recency fresh or LRU would evict exactly the
886            // steady-camera tiles the cache exists to avoid refetching.
887            pool.refresh_recency(&tile_key);
888            return Ok(Some((texture.clone(), bind_group.clone())));
889        }
890        let Some(encoded_image_bytes) = pool.lookup(&tile_key).map(<[u8]>::to_vec) else {
891            return Ok(None);
892        };
893        match image::load_from_memory(&encoded_image_bytes) {
894            Ok(decoded_image) => {
895                let (texture, bind_group, texture_bytes) =
896                    self.upload_decoded_texture(device, decoded_image);
897                self.gpu_texture_cache.insert(
898                    tile_key,
899                    payload_revision,
900                    texture.clone(),
901                    bind_group.clone(),
902                    texture_bytes,
903                );
904                Ok(Some((texture, bind_group)))
905            }
906            Err(_) => {
907                pool.evict(&tile_key);
908                Ok(None)
909            }
910        }
911    }
912
913    /// Creates a renderable texture from a decoded image, queues its RGBA8
914    /// pixels onto the device, and builds the tile bind group; the byte
915    /// count feeds the cache budget as the texture's GPU cost.
916    fn upload_decoded_texture(
917        &mut self,
918        device: &Device,
919        decoded_image: DynamicImage,
920    ) -> (Texture, wgpu::BindGroup, u64) {
921        let rgba_image = to_rgba8(decoded_image);
922        let rgba_bytes = rgba_image.as_raw();
923        let (width_pixels, height_pixels) = rgba_image.dimensions();
924
925        let texture = device.create_texture(&wgpu::TextureDescriptor {
926            label: Some("map-rendering-tile-texture"),
927            size: Extent3d {
928                width: width_pixels,
929                height: height_pixels,
930                depth_or_array_layers: 1,
931            },
932            mip_level_count: 1,
933            sample_count: 1,
934            dimension: wgpu::TextureDimension::D2,
935            format: FRAME_TEXTURE_FORMAT,
936            usage: TextureUsages::TEXTURE_BINDING | TextureUsages::COPY_DST,
937            view_formats: &[],
938        });
939        self.graphics_context.queue.write_texture(
940            TexelCopyTextureInfo {
941                texture: &texture,
942                mip_level: 0,
943                origin: wgpu::Origin3d::ZERO,
944                aspect: TextureAspect::All,
945            },
946            rgba_bytes,
947            TexelCopyBufferLayout {
948                offset: 0,
949                bytes_per_row: Some(PIXEL_BYTES * width_pixels),
950                rows_per_image: None,
951            },
952            Extent3d {
953                width: width_pixels,
954                height: height_pixels,
955                depth_or_array_layers: 1,
956            },
957        );
958        let bind_group = self.create_tile_bind_group(device, &texture);
959        let texture_bytes = u64::from(width_pixels) * u64::from(height_pixels) * PIXEL_BYTES as u64;
960        (texture, bind_group, texture_bytes)
961    }
962
963    fn create_tile_bind_group(&self, device: &Device, texture: &Texture) -> wgpu::BindGroup {
964        let texture_view = texture.create_view(&wgpu::TextureViewDescriptor::default());
965        device.create_bind_group(&wgpu::BindGroupDescriptor {
966            label: Some("map-rendering-tile-bind-group"),
967            layout: &self.tile_pipeline.get_bind_group_layout(0),
968            entries: &[
969                wgpu::BindGroupEntry {
970                    binding: 0,
971                    resource: wgpu::BindingResource::TextureView(&texture_view),
972                },
973                wgpu::BindGroupEntry {
974                    binding: 1,
975                    resource: wgpu::BindingResource::Sampler(&self.tile_sampler),
976                },
977            ],
978        })
979    }
980}

991/// The GPU-free gating state of the idle-gated snapshot: how many of a
992/// frame's requested tiles are not pooled at their own zoom level. Kept
993/// free of `wgpu` types (and of the renderer itself) so the refusal logic is
994/// unit-testable and re-derivable without a device; see
995/// [`HeadlessTileRenderer::snapshot_frame_when_idle`].
996mod idle_gate {
997    use crate::tile_scheduler::{PreparedTile, TileReadiness};
998    use crate::tile_texture_pool::TileTexturePool;

1000    /// Counts the frame's requested tiles that are still pending: a verdict
1001    /// other than [`TileReadiness::Present`] — a fallback still owes the
1002    /// child's detail, an unavailable fetch is retried, a deferred tile is
1003    /// fetched — or a `Present` verdict whose payload has left the pool
1004    /// since the frame was prepared.
1005    pub(super) fn pending_tile_count(
1006        prepared_tiles: &[PreparedTile],
1007        pool: &TileTexturePool,
1008    ) -> usize {
1009        prepared_tiles
1010            .iter()
1011            .filter(|prepared_tile| {
1012                prepared_tile.readiness != TileReadiness::Present
1013                    || !pool.contains(&prepared_tile.requested_tile_key)
1014            })
1015            .count()
1016    }
1017}

1019/// Serializes fragment vertices into the vertex and index byte streams the GPU
1020/// buffers expect: little-endian floats, [`CORNERS_PER_QUAD`] corners per
1021/// world-copy fragment, and the two-triangle index pattern `0, 1, 2, 2, 1, 3`
1022/// ([`INDICES_PER_QUAD`] per fragment).
1023fn encoded_fragment_buffers(fragment_vertices: &[[ClipSpaceTileVertex; 4]]) -> (Vec<u8>, Vec<u8>) {
1024    let vertex_stride = std::mem::size_of::<ClipSpaceTileVertex>();
1025    let mut vertex_bytes =
1026        Vec::with_capacity(fragment_vertices.len() * vertex_stride * CORNERS_PER_QUAD);
1027    let mut index_bytes =
1028        Vec::with_capacity(fragment_vertices.len() * INDICES_PER_QUAD * std::mem::size_of::<u16>());
1029    for (fragment_index, vertices) in fragment_vertices.iter().enumerate() {
1030        let base_corner_index = u16::try_from(fragment_index * CORNERS_PER_QUAD)
1031            .expect("the draw list was bounded above");
1032        for component in vertices.iter().flat_map(|vertex| {
1033            vertex
1034                .clip_position
1035                .iter()
1036                .chain(vertex.texture_coordinates.iter())
1037        }) {
1038            vertex_bytes.extend_from_slice(&component.to_le_bytes());
1039        }
1040        for index in [
1041            base_corner_index,
1042            base_corner_index + 1,
1043            base_corner_index + 2,
1044            base_corner_index + 2,
1045            base_corner_index + 1,
1046            base_corner_index + 3,
1047        ] {
1048            index_bytes.extend_from_slice(&index.to_le_bytes());
1049        }
1050    }
1051    (vertex_bytes, index_bytes)
1052}
1053
1054/// Serializes the overlay triangle-list vertices into the vertex byte stream
1055/// the GPU buffer expects: little-endian floats. The overlay draws
1056/// non-indexed (a plain vertex draw over the flat triangle list), so no index
1057/// stream exists for this layer.
1058fn encoded_overlay_vertices(overlay_vertices: &[ClipSpaceOverlayVertex]) -> Vec<u8> {
1059    let mut vertex_bytes = Vec::with_capacity(std::mem::size_of_val(overlay_vertices));
1060    for vertex in overlay_vertices {
1061        for component in vertex.clip_position.iter().chain(vertex.rgba_color.iter()) {
1062            vertex_bytes.extend_from_slice(&component.to_le_bytes());
1063        }
1064    }
1065    vertex_bytes
1066}

1068/// Creates a device buffer pre-filled from host bytes.
1069fn create_filled_buffer(
1070    device: &Device,
1071    queue: &Queue,
1072    bytes: &[u8],
1073    usage: wgpu::BufferUsages,
1074    label: Option<&str>,
1075) -> Buffer {
1076    let buffer = device.create_buffer(&wgpu::BufferDescriptor {
1077        label,
1078        size: bytes.len() as wgpu::BufferAddress,
1079        usage: usage | wgpu::BufferUsages::COPY_DST,
1080        mapped_at_creation: false,
1081    });
1082    queue.write_buffer(&buffer, 0, bytes);
1083    buffer
1084}

1086/// Copies a rendered texture into a staging buffer honoring the 256-byte row
1087/// alignment ([`wgpu::COPY_BYTES_PER_ROW_ALIGNMENT`]), maps the buffer, and
1088/// reads it back as a tightly packed [`CapturedFrame`], stripping the padding
1089/// the alignment inserts between rows.
1090///
1091/// # Reusability
1092///
1093/// Any render-attachment texture (`RENDER_ATTACHMENT | COPY_SRC`, RGBA8
1094/// unorm) of any size is captured. The copy commands are encoded onto the
1095/// provided encoder, which the caller has not submitted yet and which this
1096/// function then finishes and submits; mapping completes through device
1097/// polling — native only (see the module docs).
1098pub fn capture_frame_bytes(
1099    device: &Device,
1100    queue: &Queue,
1101    mut command_encoder: CommandEncoder,
1102    source_texture: &Texture,
1103) -> Result<CapturedFrame, String> {
1104    let texture_size = source_texture.size();
1105    let width_pixels = texture_size.width;
1106    let height_pixels = texture_size.height;
1107
1108    let unpadded_row_bytes = width_pixels as usize * PIXEL_BYTES as usize;
1109    let bytes_per_row = u32::try_from(aligned_bytes_per_row(width_pixels))
1110        .expect("aligned row length stays below u32 for viewport sizes");
1111    let staging_buffer_size = u64::from(bytes_per_row) * u64::from(height_pixels.saturating_sub(1))
1112        + unpadded_row_bytes as u64;
1113    let staging_buffer = device.create_buffer(&wgpu::BufferDescriptor {
1114        label: Some("map-rendering-frame-staging"),
1115        size: staging_buffer_size,
1116        usage: wgpu::BufferUsages::MAP_READ | wgpu::BufferUsages::COPY_DST,
1117        mapped_at_creation: false,
1118    });
1119
1120    command_encoder.copy_texture_to_buffer(
1121        TexelCopyTextureInfo {
1122            texture: source_texture,
1123            mip_level: 0,
1124            origin: wgpu::Origin3d::ZERO,
1125            aspect: TextureAspect::All,
1126        },
1127        TexelCopyBufferInfo {
1128            buffer: &staging_buffer,
1129            layout: TexelCopyBufferLayout {
1130                offset: 0,
1131                bytes_per_row: Some(bytes_per_row),
1132                rows_per_image: None,
1133            },
1134        },
1135        Extent3d {
1136            width: width_pixels,
1137            height: height_pixels,
1138            depth_or_array_layers: 1,
1139        },
1140    );
1141    queue.submit([command_encoder.finish()]);
1142
1143    let (map_result_sender, map_result_receiver) = mpsc::channel::<BufferMapOutcome>();
1144    staging_buffer
1145        .slice(..)
1146        .map_async(MapMode::Read, move |map_result| {
1147            let _ = map_result_sender.send(map_result.map_err(|error| error.to_string()));
1148        });
1149
1150    // The map callback is delivered by polling the device; on native backends
1151    // only a blocking `Wait` poll guarantees the map completes, so the loop
1152    // escalates to one on the first empty receive rather than spinning on
1153    // non-blocking polls.
1154    loop {
1155        match map_result_receiver.try_recv() {
1156            Ok(map_result) => {
1157                map_result?;
1158                break;
1159            }
1160            Err(TryRecvError::Disconnected) => {
1161                return Err(
1162                    "buffer map callback sender dropped before the map completed".to_string(),
1163                );
1164            }
1165            Err(TryRecvError::Empty) => {
1166                device
1167                    .poll(PollType::wait_indefinitely())
1168                    .map_err(|poll_error| poll_error.to_string())?;
1169            }
1170        }
1171    }
1172
1173    let mapped_bytes = staging_buffer
1174        .slice(..)
1175        .get_mapped_range()
1176        .map_err(|error| error.to_string())?;
1177    let mut rgba_bytes = Vec::with_capacity(unpadded_row_bytes * height_pixels as usize);
1178    for row_index in 0..height_pixels as usize {
1179        let row_start = row_index * bytes_per_row as usize;
1180        rgba_bytes.extend_from_slice(&mapped_bytes[row_start..row_start + unpadded_row_bytes]);
1181    }
1182    drop(mapped_bytes);
1183    staging_buffer.unmap();
1184
1185    Ok(CapturedFrame {
1186        width_pixels,
1187        height_pixels,
1188        rgba_bytes,
1189    })
1190}

1293        }

1299            idle_gate::pending_tile_count(&frame, &payload_pool),
