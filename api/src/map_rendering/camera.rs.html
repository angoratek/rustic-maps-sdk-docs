1//! Camera model: geographic center, zoom, bearing, and pitch, projected into
2//! a perspective viewport whose ground magnification matches the zoom scale.
3//!
4//! The camera is the mapbox-style anchored model: the ground point under
5//! [`MapCamera::center`] always sits at the viewport center, the camera hovers
6//! above it at the altitude implied by the zoom level, and bearing and pitch
7//! rotate and tilt the view around that anchor. Projection math runs on
8//! `f64` matrices because world units reach ~8.6e9 pixels at zoom 24.
9

13};

19/// Field of view used when a caller does not configure one.
20pub const DEFAULT_FIELD_OF_VIEW_DEGREES: f64 = 60.0;

22/// Largest usable pitch; steeper tilts push most of the viewport below the
23/// horizon where no ground is visible.
24pub const MAXIMUM_PITCH_DEGREES: f64 = 75.0;
25
26/// Smallest and largest zoom levels the camera accepts.
27pub const MINIMUM_ZOOM_LEVEL: f64 = 0.0;
28pub const MAXIMUM_ZOOM_LEVEL: f64 = 24.0;

30/// Near clip plane distance as a fraction of the far plane. The ground plane
31/// needs only enough near depth to stay numerically sane at high zoom, where
32/// world coordinates reach ~1e9 pixels.
33const NEAR_PLANE_TO_FAR_PLANE_RATIO: f64 = 1e-5;
34
35/// The far clip plane reaches this multiple of the camera altitude, far
36/// beyond any ground detail the anchored zoom implies.
37const FAR_PLANE_ALTITUDE_MULTIPLIER: f64 = 100.0;
38
39/// Screen-pixel NDC depth sentinels for the two-point ray inversion: depth
40/// one sits on the near plane, depth `MAXIMUM_NORMALIZED_DEPTH` on the far
41/// plane (or behind it).
42const MINIMUM_NORMALIZED_DEPTH: f64 = -1.0;
43const MAXIMUM_NORMALIZED_DEPTH: f64 = 1.0;
44
45/// Guard on the inverted clip weights: below this the two reference points
46/// degenerate and the pixel ray is meaningless.
47const MINIMUM_CLIP_WEIGHT: f64 = 1e-12;
48
49/// Squared-length guard rejecting degenerate screen rays whose two reference
50/// points coincide after inversion.
51const MINIMUM_RAY_LENGTH_SQUARED: f64 = 1e-24;
52
53/// Largest ray z-component still counted as descending toward the ground
54/// plane; anything flatter is at or above the horizon, where unprojection
55/// reports no ground.
56const MINIMUM_DOWNWARD_RAY_COMPONENT: f64 = -1e-12;
57
58/// Half-extent of the ground quad the frustum clip starts from, as a multiple
59/// of the far plane — comfortably beyond any ground the clip can leave.
60const GROUND_CLIP_REACH_MULTIPLIER: f64 = 8.0;

139/// The Web Mercator world square's ground-equivalent length: the WGS-84
140/// equatorial circumference the projection stretches onto one world edge.
141pub const EARTH_EQUATORIAL_CIRCUMFERENCE_METERS: f64 = 40_075_016.686;
142
143impl MapCamera {
144    /// Ground meters represented by one screen pixel at the camera's
145    /// center: the equatorial ground meters per pixel at the camera's zoom
146    /// level times the cosine of the center latitude, so the answer shrinks
147    /// toward the poles exactly like the projected ground shrinks in the
148    /// Web Mercator projection. At any nonzero pitch this is the
149    /// flat-centered approximation (the helper folds no unprojection of the
150    /// pitch into the answer) — see the contract's member documentation.
151    pub fn ground_meters_per_screen_pixel(&self) -> f64 {
152        let world_pixels_per_edge = MERCATOR_TILE_SIZE_PIXELS * 2_f64.powf(self.zoom_level);
153        let equatorial_meters_per_pixel =
154            EARTH_EQUATORIAL_CIRCUMFERENCE_METERS / world_pixels_per_edge;
155        equatorial_meters_per_pixel * self.center.latitude.to_radians().cos()
156    }
157    /// Builds a camera, clamping zoom and pitch and wrapping the bearing into
158    /// `[0, 360)`.
159    pub fn new(
160        center: GeographicCoordinate,
161        zoom_level: f64,
162        bearing_degrees: f64,
163        pitch_degrees: f64,
164    ) -> Self {
165        Self {
166            center,
167            zoom_level: zoom_level.clamp(MINIMUM_ZOOM_LEVEL, MAXIMUM_ZOOM_LEVEL),
168            bearing_degrees: bearing_degrees.rem_euclid(360.0),
169            pitch_degrees: pitch_degrees.clamp(0.0, MAXIMUM_PITCH_DEGREES),
170        }
171    }

173    /// Projects a coordinate to viewport pixels (origin top-left, y down).
174    /// Returns `None` when the point lies behind the camera.
175    pub fn project_to_screen(
176        &self,
177        coordinate: &GeographicCoordinate,
178        viewport: &ViewportPixels,
179        field_of_view_degrees: f64,
180    ) -> Option<[f64; 2]> {
181        let view_projection = self.view_projection_matrix(viewport, field_of_view_degrees);
182        let clip_position = view_projection * self.ground_position_vector(coordinate);
183        if clip_position.w <= 0.0 {
184            return None;
185        }
186        let normalized_device = clip_position.truncate() / clip_position.w;
187        Some([
188            (normalized_device.x + 1.0) * 0.5 * viewport.width_pixels,
189            (1.0 - (normalized_device.y + 1.0) * 0.5) * viewport.height_pixels,
190        ])
191    }

193    /// Unprojects a viewport pixel back onto the ground plane. Returns `None`
194    /// when the ray misses the ground (typically below the horizon at pitch).
195    pub fn screen_to_ground_coordinate(
196        &self,
197        screen_point: [f64; 2],
198        viewport: &ViewportPixels,
199        field_of_view_degrees: f64,
200    ) -> Option<GeographicCoordinate> {
201        let (camera_eye, camera_forward, _, far_plane) =
202            self.camera_frame(viewport, field_of_view_degrees);
203        let view_projection_inverse = self
204            .view_projection_matrix(viewport, field_of_view_degrees)
205            .inverse();
206        let normalized_x = 2.0 * screen_point[0] / viewport.width_pixels - 1.0;
207        let normalized_y = 1.0 - 2.0 * screen_point[1] / viewport.height_pixels;
208
209        // Both reference points sit on the pixel's world ray; whichever depth
210        // the inverse produces, the ray direction is oriented toward the
211        // camera forward so the ground walk is unambiguous even when the far
212        // clip plane lies nearer than depth 1 (high zoom).
213        let depth_one_clip = view_projection_inverse
214            * DVec4::new(normalized_x, normalized_y, MINIMUM_NORMALIZED_DEPTH, 1.0);
215        let far_plane_clip = view_projection_inverse
216            * DVec4::new(normalized_x, normalized_y, MAXIMUM_NORMALIZED_DEPTH, 1.0);
217        if depth_one_clip.w.abs() < MINIMUM_CLIP_WEIGHT
218            || far_plane_clip.w.abs() < MINIMUM_CLIP_WEIGHT
219        {
220            return None;
221        }
222        let mut ray_direction = (far_plane_clip / far_plane_clip.w).truncate()
223            - (depth_one_clip / depth_one_clip.w).truncate();
224        if ray_direction.dot(camera_forward) < 0.0 {
225            ray_direction = -ray_direction;
226        }
227        if ray_direction.length_squared() < MINIMUM_RAY_LENGTH_SQUARED {
228            return None;
229        }
230        let ray_direction = ray_direction.normalize();
231        // Only rays that descend toward the ground plane reach it.
232        if ray_direction.z >= MINIMUM_DOWNWARD_RAY_COMPONENT {
233            return None;
234        }
235        // The far clip bounds which ground pixels are actually renderable.
236        let ground_distance = -camera_eye.z / ray_direction.z;
237        if ground_distance > far_plane {
238            return None;
239        }
240        let ground_world = camera_eye + ray_direction * ground_distance;
241        let zoom_scale = zoom_scale_pixels(self.zoom_level);
242        // Convert the y-north camera world space back to tile-convention
243        // normalized world space (y southward).
244        let normalized_world = [ground_world.x / zoom_scale, -ground_world.y / zoom_scale];
245        Some(normalized_world_to_coordinate(normalized_world))
246    }

248    /// The coordinate's position in camera world space: x east, y **north**,
249    /// z up on the ground plane. The tile-convention y-south normalized world
250    /// space of [`crate::projection`] flips to y-north exactly here, so the
251    /// camera frame stays in a right-handed geographic system. Longitude wraps
252    /// to the nearest world copy around the camera center, so a view across
253    /// the antimeridian renders the adjacent world copy.
254    fn ground_position_vector(&self, coordinate: &GeographicCoordinate) -> DVec4 {
255        let zoom_scale = zoom_scale_pixels(self.zoom_level);
256        let normalized_world = coordinate_to_normalized_world(coordinate);
257        let center_normalized_world = coordinate_to_normalized_world(&self.center);
258        let wrapped_x_offset =
259            normalized_shortest_delta(center_normalized_world[0], normalized_world[0]);
260        DVec4::new(
261            (center_normalized_world[0] + wrapped_x_offset) * zoom_scale,
262            -normalized_world[1] * zoom_scale,
263            0.0,
264            1.0,
265        )
266    }

268    /// The camera's eye position, forward direction, up direction, and far
269    /// clip plane in world space. Ground magnification at the center equals
270    /// `zoom_scale_pixels(zoom_level)` pixels per normalized world unit.
271    fn camera_frame(
272        &self,
273        viewport: &ViewportPixels,
274        field_of_view_degrees: f64,
275    ) -> (DVec3, DVec3, DVec3, f64) {
276        let pitch_radians = self
277            .pitch_degrees
278            .clamp(0.0, MAXIMUM_PITCH_DEGREES)
279            .to_radians();
280        let bearing_radians = self.bearing_degrees.to_radians();
281        let field_of_view_radians = field_of_view_degrees.to_radians();
282
283        // Altitude above the anchor in the zoom-scaled pixel frame. World
284        // coordinates already scale with zoom (x pixels per normalized unit),
285        // so the half-viewport of `height_pixels` spans `(height/2)` pixels of
286        // ground at center distance under half the field of view.
287        let camera_altitude = (viewport.height_pixels * 0.5) / (field_of_view_radians * 0.5).tan();
288
289        let bearing_rotation = DMat4::from_rotation_z(-bearing_radians);
290        let anchor = DVec3::new(
291            self.ground_position_vector(&self.center).x,
292            self.ground_position_vector(&self.center).y,
293            0.0,
294        );
295        let camera_eye = anchor
296            + bearing_rotation.transform_point3(DVec3::new(
297                0.0,
298                -camera_altitude * pitch_radians.sin(),
299                camera_altitude * pitch_radians.cos(),
300            ));
301        let camera_forward = bearing_rotation.transform_point3(DVec3::new(
302            0.0,
303            pitch_radians.sin(),
304            -pitch_radians.cos(),
305        ));
306        // At pitch zero the screen-top direction must face north. World space
307        // is converted to y-north at the boundary (see
308        // `ground_position_vector`), so screen-top at pitch zero is +y.
309        let camera_up = bearing_rotation.transform_point3(DVec3::new(
310            0.0,
311            pitch_radians.cos(),
312            pitch_radians.sin(),
313        ));
314        let far_plane = camera_altitude * FAR_PLANE_ALTITUDE_MULTIPLIER;
315        (camera_eye, camera_forward, camera_up, far_plane)
316    }

318    /// Perspective-view matrix matching the anchored camera. Ground
319    /// magnification at the center equals `zoom_scale_pixels(zoom_level)`
320    /// pixels per normalized world unit.
321    pub fn view_projection_matrix(
322        &self,
323        viewport: &ViewportPixels,
324        field_of_view_degrees: f64,
325    ) -> DMat4 {
326        let (camera_eye, camera_forward, camera_up, far_plane) =
327            self.camera_frame(viewport, field_of_view_degrees);
328
329        let view_matrix = DMat4::look_to_rh(camera_eye, camera_forward, camera_up);
330        let near_plane = far_plane * NEAR_PLANE_TO_FAR_PLANE_RATIO;
331        let projection_matrix = DMat4::perspective_rh(
332            field_of_view_degrees.to_radians(),
333            viewport.aspect_ratio(),
334            near_plane,
335            far_plane,
336        );
337        projection_matrix * view_matrix
338    }

350    /// Every tile at `tile_zoom_level` whose area intersects the visible
351    /// ground: the view frustum cut by the ground plane, wrapped to the
352    /// camera's world copy across the antimeridian. The footprint reduces to
353    /// its axis-aligned normalized-world bounding box, so bearing-rotated
354    /// views also list the covering corner tiles. Returns the tiles sorted
355    /// row-major from the north.
356    pub fn visible_tile_keys(
357        &self,
358        viewport: &ViewportPixels,
359        field_of_view_degrees: f64,
360        tile_zoom_level: u8,
361    ) -> Vec<TileKey> {
362        let zoom_scale = zoom_scale_pixels(self.zoom_level);
363        let ground_polygon = self.clipped_ground_polygon_vertices(viewport, field_of_view_degrees);
364        if ground_polygon.is_empty() {
365            return vec![tile_key_for_coordinate(&self.center, tile_zoom_level)];
366        }
367
368        // y grows south in normalized world space; ground points convert from
369        // the y-north camera world space exactly at the camera boundary. Tiles
370        // clamp to the map square in y; x runs on raw world-copy indices and
371        // wraps below, so a view reaching past the antimeridian naturally
372        // lists tiles from both copies.
373        let tiles_per_axis = (1u64 << tile_zoom_level) as f64;
374        let minimum_y = ground_polygon
375            .iter()
376            .map(|vertex| -vertex.y / zoom_scale)
377            .fold(f64::INFINITY, f64::min)
378            .max(0.0);
379        let maximum_y = ground_polygon
380            .iter()
381            .map(|vertex| -vertex.y / zoom_scale)
382            .fold(f64::NEG_INFINITY, f64::max)
383            .min(1.0);
384        let minimum_column =
385            ((minimum_y * tiles_per_axis).floor().max(0.0) as u64).min((tiles_per_axis as u64) - 1);
386        let maximum_column = ((maximum_y * tiles_per_axis - TILE_BOUNDARY_EPSILON)
387            .floor()
388            .max(0.0) as u64)
389            .min((tiles_per_axis as u64) - 1);
390
391        let minimum_x = ground_polygon
392            .iter()
393            .map(|vertex| vertex.x / zoom_scale)
394            .fold(f64::INFINITY, f64::min);
395        let maximum_x = ground_polygon
396            .iter()
397            .map(|vertex| vertex.x / zoom_scale)
398            .fold(f64::NEG_INFINITY, f64::max);
399        let minimum_index = (minimum_x * tiles_per_axis).floor() as i64;
400        // The right-hand boundary stays exclusive, mirroring floor semantics.
401        let maximum_index = (maximum_x * tiles_per_axis - TILE_BOUNDARY_EPSILON).floor() as i64;
402
403        // Raw indices may exceed the grid or go negative near the antimeridian
404        // and at low zoom, where the view reaches past one world copy; both
405        // wrap back onto the grid modulo the copy count.
406        let tiles_per_axis_index = tiles_per_axis as i64;
407        let mut visible_tiles: HashSet<TileKey> = HashSet::new();
408        for row in minimum_column..=maximum_column {
409            for raw_index in minimum_index..=maximum_index {
410                let wrapped_index = raw_index.rem_euclid(tiles_per_axis_index) as u32;
411                if let Ok(tile_key) = TileKey::new(wrapped_index, row as u32, tile_zoom_level) {
412                    visible_tiles.insert(tile_key);
413                }
414            }
415        }
416        let mut visible_tiles: Vec<TileKey> = visible_tiles.into_iter().collect();
417        visible_tiles.sort_by_key(|tile_key| (tile_key.tile_y, tile_key.tile_x));
418        visible_tiles
419    }

421    /// The view frustum clipped to the ground plane, in camera world space
422    /// (x east, y north, z up on the ground). Starts from a ground quad well
423    /// beyond the far plane and clips against the six frustum half-spaces, so
424    /// above-horizon and out-of-range ground drops out exactly rather than by
425    /// sampling.
426    fn clipped_ground_polygon_vertices(
427        &self,
428        viewport: &ViewportPixels,
429        field_of_view_degrees: f64,
430    ) -> Vec<DVec3> {
431        let (camera_eye, camera_forward, camera_up, far_plane) =
432            self.camera_frame(viewport, field_of_view_degrees);
433        let camera_right = camera_forward.cross(camera_up);
434        let field_of_view_radians = field_of_view_degrees.to_radians();
435        let vertical_slope = (field_of_view_radians * 0.5).tan();
436        let horizontal_slope = viewport.aspect_ratio() * vertical_slope;
437        // Side planes pass through the eye; near and far sit along the look
438        // axis. A point stays when `normal · (point − eye) ≥ bias`.
439        let near_plane = far_plane * NEAR_PLANE_TO_FAR_PLANE_RATIO;
440        let frustum_planes: [(DVec3, f64); 6] = [
441            (camera_forward * horizontal_slope + camera_right, 0.0),
442            (camera_forward * horizontal_slope - camera_right, 0.0),
443            (camera_forward * vertical_slope + camera_up, 0.0),
444            (camera_forward * vertical_slope - camera_up, 0.0),
445            (camera_forward, near_plane),
446            (-camera_forward, -far_plane),
447        ];
448
449        let reach = far_plane * GROUND_CLIP_REACH_MULTIPLIER;
450        let mut polygon = vec![
451            DVec3::new(camera_eye.x - reach, camera_eye.y - reach, 0.0),
452            DVec3::new(camera_eye.x + reach, camera_eye.y - reach, 0.0),
453            DVec3::new(camera_eye.x + reach, camera_eye.y + reach, 0.0),
454            DVec3::new(camera_eye.x - reach, camera_eye.y + reach, 0.0),
455        ];
456        for (plane_normal, plane_bias) in frustum_planes {
457            let signed_distance =
458                |vertex: DVec3| plane_normal.dot(vertex - camera_eye) - plane_bias;
459            let mut clipped_polygon = Vec::with_capacity(polygon.len() + 1);
460            for edge_index in 0..polygon.len() {
461                let start_vertex = polygon[edge_index];
462                let end_vertex = polygon[(edge_index + 1) % polygon.len()];
463                let start_distance = signed_distance(start_vertex);
464                let end_distance = signed_distance(end_vertex);
465                let start_inside = start_distance >= 0.0;
466                let end_inside = end_distance >= 0.0;
467                if start_inside {
468                    clipped_polygon.push(start_vertex);
469                }
470                if start_inside != end_inside {
471                    let crossing_fraction = start_distance / (start_distance - end_distance);
472                    clipped_polygon.push(start_vertex.lerp(end_vertex, crossing_fraction));
473                }
474            }
475            polygon = clipped_polygon;
476            if polygon.is_empty() {
477                return polygon;
478            }
479        }
480        polygon
481    }
482}

484/// Camera interpolation over progress `0.0..=1.0` (values outside clamp).
485///
486/// The center interpolates linearly in mercator space along the shortest
487/// longitude path, zoom interpolates in log space, and the bearing takes the
488/// shortest arc.
489pub fn interpolate_camera(
490    start_camera: &MapCamera,
491    target_camera: &MapCamera,
492    progress: f64,
493) -> MapCamera {
494    let eased_progress = smoothstep_progress(progress);
495    // Endpoints return the input verbatim so an anchored start (or settle) is
496    // free of mercator round-trip noise.
497    if eased_progress <= 0.0 {
498        return *start_camera;
499    }
500    if eased_progress >= 1.0 {
501        return *target_camera;
502    }
503    let start_normalized = coordinate_to_normalized_world(&start_camera.center);
504    let target_normalized = coordinate_to_normalized_world(&target_camera.center);
505    let longitude_delta = normalized_shortest_delta(start_normalized[0], target_normalized[0]);
506    let interpolated_center = normalized_world_to_coordinate([
507        start_normalized[0] + longitude_delta * eased_progress,
508        start_normalized[1] + (target_normalized[1] - start_normalized[1]) * eased_progress,
509    ]);
510    let bearing_delta = normalized_shortest_delta(
511        start_camera.bearing_degrees / 360.0,
512        target_camera.bearing_degrees / 360.0,
513    );
514    let interpolated_bearing =
515        (start_camera.bearing_degrees + bearing_delta * 360.0 * eased_progress).rem_euclid(360.0);
516    MapCamera {
517        center: interpolated_center,
518        zoom_level: start_camera.zoom_level
519            + (target_camera.zoom_level - start_camera.zoom_level) * eased_progress,
520        bearing_degrees: interpolated_bearing,
521        pitch_degrees: start_camera.pitch_degrees
522            + (target_camera.pitch_degrees - start_camera.pitch_degrees) * eased_progress,
523    }
524}

526/// Signed wrap-around delta in normalized space: the value in `[-0.5, 0.5)`
527/// such that `start + delta ≡ target (mod 1)`. Shared with the renderer, which
528/// must wrap tile spans into the camera's world copy the same way.
529pub(crate) fn normalized_shortest_delta(start: f64, target: f64) -> f64 {
530    let forward_delta = (target - start).rem_euclid(1.0);
531    if forward_delta > 0.5 {
532        forward_delta - 1.0
533    } else {
534        forward_delta
535    }
536}

538/// Smoothstep progress shaping shared by every camera animation: the eased
539/// progress used to travel along the pose curve.
540fn smoothstep_progress(progress: f64) -> f64 {
541    let clamped_progress = progress.clamp(0.0, 1.0);
542    clamped_progress * clamped_progress * (3.0 - 2.0 * clamped_progress)
543}

546/// derivation, from the leading GL JS implementation): larger values dip the

549/// dip the zoom past the SDK's minimum zoom level.

552/// Default travel speed of the fly-to arc, in screenfulls of the arc per

598/// Shape of the fly-to arc between its endpoints (see [`fly_to_animation`]).
599#[derive(Debug, Clone, Copy, PartialEq)]
600enum FlyToArcShape {
601    /// Center and zoom both travel: the hyperbolic ascent-descent arc.
602    TravelingArc,
603    /// Center holds still while the zoom rises or falls exponentially; the
604    /// sign drives the span toward the target's (wider spans zoom out).
605    PureZoom { span_growth_sign: f64 },
606    /// Endpoints coincide; the animation resolves instantly (zero duration).
607    Instantaneous,
608}

610/// The fly-to pose parameters: everything `camera_at` needs to evaluate the
611/// arc. Distances are measured in "screenfulls" of the initial viewport span
612/// (the fly-to derivation's unit), which is what makes the pose independent of
613/// the viewport except through the path length.
614#[derive(Debug, Clone, Copy)]
615struct FlyToPathPose {
616    initial_visible_span_pixels: f64,
617    start_normalized_center: [f64; 2],
618    normalized_center_delta: [f64; 2],
619    start_zoom_level: f64,
620    /// The flight-path length in pixels at the start zoom's scale (u₁ in the
621    /// fly-to derivation); the traveled center fraction divides by it.
622    flight_path_pixels: f64,
623    /// Zoom-out factor at the arc's start (r₀ in the fly-to derivation).
624    arc_start_zoom_out_factor: f64,
625    arc_zoom_out_curve: f64,
626    /// Total arc length the eased progress travels (S in the derivation).
627    arc_path_length_in_screenfulls: f64,
628    arc_shape: FlyToArcShape,
629}

631/// The camera animation that flies between two cameras instead of cutting
632/// between them: zoom follows the fly-to arc's exponential dip — long travels
633/// zoom out along the way and zoom back in on arrival, short travels slide
634/// almost monotonically — while the center, bearing, and pitch ease normally.
635///
636/// The arc follows the leading GL JS implementation's derivation, adapted to
637/// the SDK's mercator-anchored camera: the visible span `w(s)` shrinks and
638/// grows hyperbolically along the path in units of the initial viewport span,
639/// and the derived duration (absent an override) is the arc length over the
640/// travel speed. The outer easing deliberately stays the SDK's smoothstep
641/// (`interpolate_camera` uses the same), trading the GL JS default cubic-
642/// bezier easing for one animation easing story across both pose curves.
643///
644/// Errors report non-finite or unusable input; use [`CameraAnimation::new`]
645/// for a plain cut-interpolated animation.
646pub fn fly_to_animation(
647    start_camera: &MapCamera,
648    target_camera: &MapCamera,
649    viewport: &ViewportPixels,
650    fly_to_options: FlyToOptions,
651) -> Result<CameraAnimation, String> {
652    if !start_camera.zoom_level.is_finite() || !target_camera.zoom_level.is_finite() {
653        return Err("the start and target zoom levels must be finite".to_string());
654    }
655    if !viewport.width_pixels.is_finite()
656        || !viewport.height_pixels.is_finite()
657        || viewport.width_pixels <= 0.0
658        || viewport.height_pixels <= 0.0
659    {
660        return Err("the viewport extent must be finite and positive".to_string());
661    }
662    if !fly_to_options.zoom_out_curve.is_finite() || fly_to_options.zoom_out_curve <= 0.0 {
663        return Err("the zoom-out curve must be finite and positive".to_string());
664    }
665    if !fly_to_options.travel_speed.is_finite() || fly_to_options.travel_speed <= 0.0 {
666        return Err("the travel speed must be finite and positive".to_string());
667    }
668    if let Some(duration_seconds) = fly_to_options.duration_seconds {
669        if !duration_seconds.is_finite() {
670            return Err("the duration must be finite".to_string());
671        }
672    }
673
674    // The zoom ceiling applies to the arc's end like every transform update.
675    let arc_maximum_zoom = fly_to_options
676        .maximum_zoom_level
677        .clamp(MINIMUM_ZOOM_LEVEL, MAXIMUM_ZOOM_LEVEL);
678    let clamped_target_zoom = target_camera
679        .zoom_level
680        .clamp(MINIMUM_ZOOM_LEVEL, arc_maximum_zoom);
681
682    // w₀ and w₁: visible spans measured in pixels with respect to the initial
683    // zoom's scale. The viewport enters the pose only through w₀, and every
684    // derived quantity is a span ratio from there.
685    let initial_visible_span_pixels = viewport.width_pixels.max(viewport.height_pixels);
686    let zoom_pixel_ratio = (clamped_target_zoom - start_camera.zoom_level).exp2();
687    let final_visible_span_pixels = initial_visible_span_pixels / zoom_pixel_ratio;
688
689    // The flight path in pixels at the start zoom's scale: the shortest
690    // wrapped longitude path, matching the SDK's nearest-world-copy rule.
691    let start_normalized_center = coordinate_to_normalized_world(&start_camera.center);
692    let target_normalized_center = coordinate_to_normalized_world(&target_camera.center);
693    let normalized_center_delta = [
694        normalized_shortest_delta(start_normalized_center[0], target_normalized_center[0]),
695        target_normalized_center[1] - start_normalized_center[1],
696    ];
697    let start_zoom_pixel_scale = zoom_scale_pixels(start_camera.zoom_level);
698    let flight_path_pixels = (normalized_center_delta[0] * start_zoom_pixel_scale)
699        .hypot(normalized_center_delta[1] * start_zoom_pixel_scale);
700
701    // The arc may not zoom the visible span past the SDK's minimum zoom, so
702    // the tightness yields to the w_max bound (GL JS: ρ = min(ρ, √(wmax/u1·2))).
703    let minimum_zoom_pixel_ratio = (MINIMUM_ZOOM_LEVEL - start_camera.zoom_level).exp2();
704    let maximum_visible_span_pixels = initial_visible_span_pixels / minimum_zoom_pixel_ratio;
705    let arc_zoom_out_curve = fly_to_options
706        .zoom_out_curve
707        .min((maximum_visible_span_pixels / flight_path_pixels * 2.0).sqrt());
708    let arc_zoom_out_curve_squared = arc_zoom_out_curve * arc_zoom_out_curve;
709
710    // rᵢ: the zoom-out factor at one arc end; the ascent leg (false) mirrors
711    // the initial span w0, the descent leg (true) mirrors the final span w1.
712    let zoom_out_factor = |descent: bool| -> f64 {
713        let arc_bias = (final_visible_span_pixels * final_visible_span_pixels
714            - initial_visible_span_pixels * initial_visible_span_pixels
715            + (if descent { -1.0 } else { 1.0 })
716                * arc_zoom_out_curve_squared
717                * arc_zoom_out_curve_squared
718                * flight_path_pixels
719                * flight_path_pixels)
720            / (2.0
721                * (if descent {
722                    final_visible_span_pixels
723                } else {
724                    initial_visible_span_pixels
725                })
726                * arc_zoom_out_curve_squared
727                * flight_path_pixels);
728        ((arc_bias * arc_bias + 1.0).sqrt() - arc_bias).ln()
729    };
730    let arc_start_zoom_out_factor = zoom_out_factor(false);
731    let arc_path_length_in_screenfulls =
732        (zoom_out_factor(true) - arc_start_zoom_out_factor) / arc_zoom_out_curve;
733
734    // When the path length cannot support both legs (GL JS: |u₁| below the
735    // path epsilon, or a non-finite S), the flight collapses to a zoom-only
736    // arc along an exponential span — or resolves instantly when the spans
737    // also coincide.
738    let (arc_shape, arc_path_length_in_screenfulls_resolved) = if flight_path_pixels.abs()
739        < FLY_TO_DEGENERATE_PATH_PIXEL_TOLERANCE
740        || !arc_path_length_in_screenfulls.is_finite()
741    {
742        if (initial_visible_span_pixels - final_visible_span_pixels).abs()
743            < FLY_TO_IDENTICAL_SPAN_TOLERANCE
744        {
745            (FlyToArcShape::Instantaneous, 0.0)
746        } else {
747            let span_growth_sign = if final_visible_span_pixels < initial_visible_span_pixels {
748                -1.0
749            } else {
750                1.0
751            };
752            (
753                FlyToArcShape::PureZoom { span_growth_sign },
754                (final_visible_span_pixels / initial_visible_span_pixels)
755                    .ln()
756                    .abs()
757                    / arc_zoom_out_curve,
758            )
759        }
760    } else {
761        (FlyToArcShape::TravelingArc, arc_path_length_in_screenfulls)
762    };
763
764    let animation_duration_seconds = match fly_to_options.duration_seconds {
765        Some(explicit_duration_seconds) => explicit_duration_seconds,
766        None => arc_path_length_in_screenfulls_resolved / fly_to_options.travel_speed,
767    };
768
769    Ok(CameraAnimation {
770        start_camera: *start_camera,
771        target_camera: *target_camera,
772        duration_seconds: animation_duration_seconds,
773        path_curve: CameraPathCurve::FlyToArc(FlyToPathPose {
774            initial_visible_span_pixels,
775            start_normalized_center,
776            normalized_center_delta,
777            start_zoom_level: start_camera.zoom_level,
778            flight_path_pixels,
779            arc_start_zoom_out_factor,
780            arc_zoom_out_curve,
781            arc_path_length_in_screenfulls: arc_path_length_in_screenfulls_resolved,
782            arc_shape,
783        }),
784    })
785}

787impl FlyToPathPose {
788    /// The camera pose along the fly-to arc at eased progress `progress` in
789    /// `0..=1` (values outside clamp); progress zero returns `start` verbatim.
790    fn camera_at_progress(
791        &self,
792        progress: f64,
793        start_camera: &MapCamera,
794        target_camera: &MapCamera,
795    ) -> MapCamera {
796        let eased_progress = smoothstep_progress(progress);
797        if eased_progress <= 0.0 {
798            return *start_camera;
799        }
800        let arc_position = eased_progress * self.arc_path_length_in_screenfulls;
801        // w(s): visible span relative to the initial span; the zoom is the
802        // initial zoom plus the span's reciprocal in log-2 space.
803        let visible_span_fraction = match self.arc_shape {
804            FlyToArcShape::TravelingArc => {
805                self.arc_start_zoom_out_factor.cosh()
806                    / (self.arc_start_zoom_out_factor + self.arc_zoom_out_curve * arc_position)
807                        .cosh()
808            }
809            FlyToArcShape::PureZoom { span_growth_sign } => {
810                (span_growth_sign * self.arc_zoom_out_curve * arc_position).exp()
811            }
812            // A zero-duration animation resolves to the target before the
813            // pose is ever consulted; this branch exists for completeness.
814            FlyToArcShape::Instantaneous => return *target_camera,
815        };
816        let arc_zoom_level = self.start_zoom_level - visible_span_fraction.log2();
817        // u(s): the center's traveled fraction along the shortest wrapped
818        // path; a degenerate (zoom-only) arc holds the start center.
819        let arc_center_fraction = match self.arc_shape {
820            FlyToArcShape::TravelingArc => {
821                self.initial_visible_span_pixels
822                    * ((self.arc_start_zoom_out_factor.cosh()
823                        * (self.arc_start_zoom_out_factor
824                            + self.arc_zoom_out_curve * arc_position)
825                            .tanh()
826                        - self.arc_start_zoom_out_factor.sinh())
827                        / (self.arc_zoom_out_curve * self.arc_zoom_out_curve))
828                    / self.flight_path_pixels
829            }
830            FlyToArcShape::PureZoom { .. } | FlyToArcShape::Instantaneous => 0.0,
831        };
832        let clamped_center_fraction = arc_center_fraction.clamp(0.0, 1.0);
833        let arc_center = if clamped_center_fraction == 0.0 {
834            start_camera.center
835        } else {
836            normalized_world_to_coordinate([
837                self.start_normalized_center[0]
838                    + self.normalized_center_delta[0] * clamped_center_fraction,
839                self.start_normalized_center[1]
840                    + self.normalized_center_delta[1] * clamped_center_fraction,
841            ])
842        };
843        // Angles ease with the progress itself (GL JS: bearing and pitch take
844        // the eased frame fraction, not the arc position).
845        let bearing_delta = normalized_shortest_delta(
846            start_camera.bearing_degrees / 360.0,
847            target_camera.bearing_degrees / 360.0,
848        );
849        let arc_bearing_degrees = (start_camera.bearing_degrees
850            + bearing_delta * 360.0 * eased_progress)
851            .rem_euclid(360.0);
852        let arc_pitch_degrees = start_camera.pitch_degrees
853            + (target_camera.pitch_degrees - start_camera.pitch_degrees) * eased_progress;
854        MapCamera::new(
855            arc_center,
856            arc_zoom_level,
857            arc_bearing_degrees,
858            arc_pitch_degrees,
859        )
860    }
861}

872/// Time-driven camera transition between two cameras.
873pub struct CameraAnimation {
874    start_camera: MapCamera,
875    target_camera: MapCamera,
876    duration_seconds: f64,
877    path_curve: CameraPathCurve,
878}

880impl CameraAnimation {
881    /// Builds an animation; non-positive durations resolve instantly to the
882    /// target camera.
883    pub fn new(start_camera: MapCamera, target_camera: MapCamera, duration_seconds: f64) -> Self {
884        Self {
885            start_camera,
886            target_camera,
887            duration_seconds,
888            path_curve: CameraPathCurve::MercatorSmoothstep,
889        }
890    }
891
892    /// The animation's total duration in seconds; zero for animations that
893    /// resolve instantly (coincident endpoints or an explicit zero/negative
894    /// duration).
895    pub fn duration_seconds(&self) -> f64 {
896        self.duration_seconds
897    }
898
899    /// The camera pose after `elapsed_seconds` from the animation start.
900    pub fn camera_at(&self, elapsed_seconds: f64) -> MapCamera {
901        if self.duration_seconds <= 0.0 || elapsed_seconds >= self.duration_seconds {
902            return self.target_camera;
903        }
904        match self.path_curve {
905            CameraPathCurve::MercatorSmoothstep => interpolate_camera(
906                &self.start_camera,
907                &self.target_camera,
908                elapsed_seconds / self.duration_seconds,
909            ),
910            CameraPathCurve::FlyToArc(path_pose) => path_pose.camera_at_progress(
911                elapsed_seconds / self.duration_seconds,
912                &self.start_camera,
913                &self.target_camera,
914            ),
915        }
916    }

919/// The camera that frames a bounding box inside the viewport's inset region.
920///
921/// Bounds fitting is top-down and north-up by construction (pitch and bearing
922/// zero), matching every leading SDK: perspective fitting is ill-defined and
923/// would double-fit against the frustum clip. The zoom is the largest level
924/// whose ground scale keeps the whole box inside the inset area, clamped to
925/// `maximum_zoom_level`; the center is the box's normalized-world midpoint,
926/// shifted so its ground point lands on the inset region's center under
927/// asymmetric insets. The whole-box-inside-the-inset-area guarantee holds
928/// only for the unclamped fit: when the zoom is clamped (by
929/// `maximum_zoom_level`, or by the minimum zoom for boxes too large for the
930/// world), the box overflows the inset region and the center-anchor
931/// guarantee is the one that survives. Returns an error when the insets or
932/// the maximum zoom are not finite usable values, when the insets are
933/// negative or leave no inset area (`left + right ≥ viewport width` and
934/// likewise vertical), or when the box spans no positive world area
935/// (non-finite or reversed corners).
936pub fn fit_bounds(
937    bounding_box: &BoundingBox,
938    viewport: &ViewportPixels,
939    edge_insets: EdgeInsetsPixels,
940    maximum_zoom_level: f64,
941) -> Result<MapCamera, String> {
942    // An inset must be finite and non-negative; the finite check refuses
943    // `NaN` inputs that would otherwise produce a silent all-`NaN` camera.
944    let insets_are_valid = [
945        edge_insets.left,
946        edge_insets.top,
947        edge_insets.right,
948        edge_insets.bottom,
949    ]
950    .iter()
951    .all(|inset_pixels| inset_pixels.is_finite() && *inset_pixels >= 0.0);
952    if !insets_are_valid {
953        return Err("insets must be finite and non-negative".to_string());
954    }
955    if !maximum_zoom_level.is_finite() {
956        return Err("the maximum zoom level must be finite".to_string());
957    }
958    let inset_available_width = viewport.width_pixels - edge_insets.left - edge_insets.right;
959    let inset_available_height = viewport.height_pixels - edge_insets.top - edge_insets.bottom;
960    if inset_available_width <= 0.0 || inset_available_height <= 0.0 {
961        return Err(format!(
962            "insets {edge_insets:?} leave no inset area in a {}x{} viewport",
963            viewport.width_pixels, viewport.height_pixels
964        ));
965    }
966
967    // A box never spans the antimeridian, so the normalized-world corners
968    // span the box directly with no wrap to reconcile. World y grows
969    // southward, so the south-west corner holds the larger y and the height
970    // span takes the corner difference in the southward direction.
971    let south_west_world = coordinate_to_normalized_world(&bounding_box.south_west);
972    let north_east_world = coordinate_to_normalized_world(&bounding_box.north_east);
973    let world_width = north_east_world[0] - south_west_world[0];
974    let south_world_y = south_west_world[1];
975    let north_world_y = north_east_world[1];
976    let world_height = south_world_y - north_world_y;
977    if !world_width.is_finite()
978        || !world_height.is_finite()
979        || world_width <= 0.0
980        || world_height <= 0.0
981    {
982        return Err(format!(
983            "the bounding box must span a positive, finite world area; got \
984             width {world_width} and height {world_height} (reversed or \
985             non-finite corners?)"
986        ));
987    }
988    // The binding axis decides the scale: the box must fit on both.
989    let fitting_scale =
990        (inset_available_width / world_width).min(inset_available_height / world_height);
991    // Ground magnification at the anchored camera equals the zoom scale
992    // (`zoom_scale_pixels`, pixels per normalized world unit).
993    let unclamped_zoom = (fitting_scale / MERCATOR_TILE_SIZE_PIXELS).log2();
994    let effective_maximum_zoom = maximum_zoom_level.clamp(MINIMUM_ZOOM_LEVEL, MAXIMUM_ZOOM_LEVEL);
995    let fitted_zoom = unclamped_zoom.clamp(MINIMUM_ZOOM_LEVEL, effective_maximum_zoom);
996
997    let world_center = [
998        (south_west_world[0] + north_east_world[0]) * 0.5,
999        (south_world_y + north_world_y) * 0.5,
1000    ];
1001    // An asymmetric inset region's center sits displaced from the viewport
1002    // center; the anchored camera model puts the camera center's ground point
1003    // at the viewport center, so the camera shifts by half the inset
1004    // imbalance, converted to world units at the fitted scale.
1005    let fitted_zoom_scale = zoom_scale_pixels(fitted_zoom);
1006    let inset_center_shift_pixels = [
1007        (edge_insets.left - edge_insets.right) * 0.5,
1008        (edge_insets.top - edge_insets.bottom) * 0.5,
1009    ];
1010    let camera_world = [
1011        world_center[0] - inset_center_shift_pixels[0] / fitted_zoom_scale,
1012        world_center[1] - inset_center_shift_pixels[1] / fitted_zoom_scale,
1013    ];
1014    Ok(MapCamera::new(
1015        normalized_world_to_coordinate(camera_world),
1016        fitted_zoom,
1017        0.0,
1018        0.0,
1019    ))
1020}
