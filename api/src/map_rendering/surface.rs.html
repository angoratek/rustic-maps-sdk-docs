1//! Surface rendering of raster tiles plus overlays onto a live native
2//! window: the same scene composition as the headless renderer, presented
3//! through a real `wgpu::Surface`.
4//!
5//! The renderer draws one textured quad per pooled tile for a single frame
6//! and an optional overlay layer ([`crate::overlay::MapOverlay`]) after the
7//! quads in the same render pass — byte-for-byte the headless pipeline's
8//! geometry and shaders (the private shader sources and encode helpers are
9//! verbatim mirrors of [`crate::headless`], the established cross-seam
10//! mirror pattern the wasm canvas follows) — but the frame renders into the
11//! next acquired surface texture and is presented, not captured. The
12//! pipelines therefore target the *surface's negotiated format* (for example
13//! Bgra8Unorm under Metal), unlike the headless capture's fixed
14//! `FRAME_TEXTURE_FORMAT`; the negotiation happens once at construction from
15//! the documented preference list and every later configuration keeps it.
16//!
17//! # Lifecycle (the platform seams' contract)
18//!
19//! Per-platform raw window handles lower into a [`wgpu::SurfaceTargetUnsafe`]
20//! *outside* this crate, which takes over from there — this module is
21//! platform-agnostic and never names a windowing backend:
22//!
23//! 1. Wrap the unsafe target with
24//!    [`SurfaceRenderTarget::from_unsafe_target`], which creates the
25//!    `wgpu::Instance` and the `'static` [`wgpu::Surface`] against it. The
26//!    raw handle must stay valid for the surface's whole lifetime;
27//!    `wgpu`'s own validity rules apply (see
28//!    [`wgpu::SurfaceTargetUnsafe`]).
29//! 2. Build the renderer with [`SurfaceTileRenderer::new`], acquiring the
30//!    adapter/device/queue through the injected
31//!    [`SurfaceGraphicsContextSource`] (production:
32//!    [`WgpuSurfaceGraphicsContextSource`], which requests an adapter
33//!    *compatible with the surface*, unlike the headless source). The
34//!    surface format is negotiated here.
35//! 3. Size the render area explicitly with
36//!    [`SurfaceTileRenderer::configure_extent`] — resize is a call, not an
37//!    observation: only a configure changes the swap-chain extent. A zero
38//!    extent is refused; an already-configured extent reconfigures only when
39//!    the extent changes (matching the headless-wasm mirror's behavior).
40//! 4. Draw every frame with [`SurfaceTileRenderer::render_frame`]; the
41//!    frame's viewport must match the configured extent exactly. A stale
42//!    surface (window resized or torn down behind the renderer's back)
43//!    reconfigures once and retries the acquisition before failing.
44//! 5. Detach by dropping the renderer: the `wgpu` surface's own release
45//!    semantics are its drop behavior (wgpu 30 has no separate
46//!    un-configure call), so the platform glue's surface view drops the
47//!    renderer — and every [`SurfaceRenderTarget`] it wraps — when its
48//!    window handle goes away.
49//!
50//! Attach and detach of the underlying platform view itself belong to the
51//! caller: dropping the [`SurfaceRenderTarget`] while its raw handle is
52//! reused elsewhere is the handle owner's bug to avoid, so the platform glue
53//! that owns the view outlives the renderer.
54//!
55//! # Native only
56//!
57//! Like [`crate::headless`], this module blocks on the asynchronous `wgpu`
58//! initialization and configuration calls (`pollster`) and is not valid on
59//! wasm, where the browser runtime drives `webgpu` (the wasm seam is the
60//! `GpuMapCanvas` of `map-web-bindings`).

191/// A live presentation target: the `wgpu` instance and the surface created
192/// against it from a platform-provided [`wgpu::SurfaceTargetUnsafe`]. The
193/// surface is `'static` — `wgpu` relies on the raw handle staying valid
194/// while the surface exists, which is the platform glue's responsibility
195/// (see the module documentation on the lifecycle).
196#[derive(Debug)]
197pub struct SurfaceRenderTarget {
198    /// Instance the surface was created against; the surface's backend
199    /// objects borrow from it, so it outlives the surface by construction.
200    pub instance: wgpu::Instance,
201    /// The live surface of a platform window or layer.
202    pub surface: wgpu::Surface<'static>,
203}

205impl SurfaceRenderTarget {
206    /// Creates the `wgpu` instance and takes the surface for an
207    /// already-assembled unsafe target (a raw window handle, a Metal layer
208    /// pointer, or one of the typed variants).
209    ///
210    /// # Safety
211    ///
212    /// The target's raw handle must point to a live platform object and stay
213    /// valid for the surface's whole lifetime — the guarantee the platform
214    /// glue owning that object makes when it assembles the
215    /// [`wgpu::SurfaceTargetUnsafe`]. A handle already returned to the
216    /// platform cannot be validated here; this constructor trusts it (see
217    /// the module documentation on where the guarantee lives).
218    pub fn from_unsafe_target(
219        surface_target: wgpu::SurfaceTargetUnsafe,
220    ) -> Result<Self, SurfaceRenderError> {
221        let instance = wgpu::Instance::new(wgpu::InstanceDescriptor::new_without_display_handle());
222        // SAFETY: the platform glue guarantees the raw handle's validity for
223        // the surface's whole lifetime (see the safety note above).
224        let surface = unsafe { instance.create_surface_unsafe(surface_target) }
225            .map_err(|error| SurfaceRenderError::SurfaceTargetInvalid(error.to_string()))?;
226        Ok(Self { instance, surface })
227    }
228}

260/// The production source: requests the first available adapter compatible
261/// with the wrapping surface, blocking on it with `pollster` (native only;
262/// see the module documentation). The device/queue request itself is the
263/// headless source's call sequence verbatim.
264pub struct WgpuSurfaceGraphicsContextSource;
265
266impl SurfaceGraphicsContextSource for WgpuSurfaceGraphicsContextSource {
267    fn request_surface_graphics_context(
268        &self,
269        surface_target: &SurfaceRenderTarget,
270    ) -> Result<SurfaceGraphicsContext, GraphicsContextError> {
271        let adapter = pollster::block_on(surface_target.instance.request_adapter(
272            &wgpu::RequestAdapterOptions {
273                power_preference: wgpu::PowerPreference::default(),
274                compatible_surface: Some(&surface_target.surface),
275                force_fallback_adapter: false,
276                apply_limit_buckets: false,
277            },
278        ))
279        .map_err(|error| GraphicsContextError::AdapterUnavailable(error.to_string()))?;
280
281        let (device, queue) = pollster::block_on(adapter.request_device(&wgpu::DeviceDescriptor {
282            label: Some("map-rendering-surface"),
283            required_features: wgpu::Features::empty(),
284            // Downlevel defaults keep the request within what every backend
285            // reports: some Metal adapters (iOS simulator runtimes, for
286            // example) cap `max_inter_stage_shader_variables` at 15, one
287            // below the `Limits::default()` request, which would reject the
288            // device outright. The renderer's pipelines use only a handful
289            // of varyings, so the downlevel limits are always sufficient.
290            required_limits: wgpu::Limits::downlevel_defaults(),
291            experimental_features: wgpu::ExperimentalFeatures::disabled(),
292            memory_hints: wgpu::MemoryHints::default(),
293            trace: wgpu::Trace::Off,
294        }))
295        .map_err(|error| GraphicsContextError::DeviceRequestFailed(error.to_string()))?;
296
297        Ok(SurfaceGraphicsContext {
298            adapter_information: adapter.get_info(),
299            adapter,
300            device,
301            queue,
302        })
303    }
304}

306/// Why rendering to a live surface could not proceed.
307#[derive(Debug, Clone, PartialEq, Eq)]
308pub enum SurfaceRenderError {
309    /// The raw window handle could not become a `wgpu` surface (the handle
310    /// was already dead, or the platform backend does not recognize it).
311    SurfaceTargetInvalid(String),
312    /// The graphics context could not be acquired at all: either no adapter
313    /// matched (for example on a machine without a usable GPU backend) or
314    /// the device request failed. The detail is the headless context
315    /// source's own verbatim failure.
316    GraphicsContextUnavailable(GraphicsContextError),
317    /// The surface refused configuration: no preferred format was
318    /// supported, the extent was unusable, or configuration itself failed.
319    SurfaceConfigurationFailed(String),
320    /// The surface texture could not be acquired even after the
321    /// documented reconfigure-and-retry attempt: the surface stayed stale,
322    /// lost or invalid.
323    SurfaceTextureAcquisitionFailed(String),
324    /// The frame's viewport did not match the configured surface extent:
325    /// the projection would not map onto the presented pixels, so nothing
326    /// was drawn.
327    ViewportExtentMismatch {
328        /// Viewport extent the frame asked to draw through.
329        viewport_extent: (u32, u32),
330        /// Extent the surface is currently configured to.
331        configured_extent: (u32, u32),
332    },
333    /// The draw list would overflow the 16-bit index buffer.
334    DrawListTooLarge {
335        /// Draw quads (world-copy fragments) assembled for the frame.
336        quad_count: usize,
337        /// Draw quads the index buffer supports.
338        maximum_quad_count: usize,
339    },
340}

348/// One presented surface frame's tile and overlay accounting, straight out
349/// of the scheduler's readiness verdicts and the overlay assembly — every
350/// requested tile lands in exactly one of the four readiness counts, so
351/// `present + fallback + deferred + unavailable` always equals `requested`.
352/// Drawn pixels this frame come from the `present` and `fallback` tiles;
353/// the `deferred` and `unavailable` tiles are where later frames may still
354/// change pixels (the pending half of the fetch-state taxonomy).
355#[derive(Debug, Clone, Copy, PartialEq, Eq)]
356pub struct SurfaceFrameStatistics {
357    /// Tile requests the frame resolved (one entry per request — a request
358    /// listing duplicates costs one fetch but reports every request).
359    pub requested_tile_count: u32,
360    /// Tiles drawn at their own zoom level.
361    pub present_tile_count: u32,
362    /// Tiles drawn from their nearest pooled ancestor.
363    pub fallback_tile_count: u32,
364    /// Tiles whose fetch stayed inside the per-frame budget deferral.
365    pub deferred_tile_count: u32,
366    /// Tiles the provider rejected this frame.
367    pub unavailable_tile_count: u32,
368    /// Clip-space triangles the overlay layer contributed to the frame.
369    pub overlay_triangle_count: u32,
370}
371
372impl SurfaceFrameStatistics {
373    /// Builds the statistics from a frame's readiness verdicts and its
374    /// overlay vertex stream (a flat, non-indexed triangle list).
375    pub(crate) fn from_frame_readiness_and_overlay_vertices(
376        prepared_tiles: &[PreparedTile],
377        overlay_vertex_count: usize,
378    ) -> Self {
379        let mut statistics = Self {
380            requested_tile_count: prepared_tiles.len() as u32,
381            present_tile_count: 0,
382            fallback_tile_count: 0,
383            deferred_tile_count: 0,
384            unavailable_tile_count: 0,
385            overlay_triangle_count: (overlay_vertex_count / OVERLAY_TRIANGLE_VERTEX_COUNT) as u32,
386        };
387        for prepared_tile in prepared_tiles {
388            match prepared_tile.readiness {
389                TileReadiness::Present => statistics.present_tile_count += 1,
390                TileReadiness::FallsBackTo { .. } => statistics.fallback_tile_count += 1,
391                TileReadiness::Deferred => statistics.deferred_tile_count += 1,
392                TileReadiness::Unavailable => statistics.unavailable_tile_count += 1,
393            }
394        }
395        // The per-frame accounting event (the "renderer" target): one debug
396        // line per presented frame, held behind the subscribers' severity
397        // gate like the "tile_fetch" target's own events.
398        tracing::debug!(
399            target: SURFACE_TRACING_TARGET,
400            requested_tile_count = statistics.requested_tile_count,
401            present_tile_count = statistics.present_tile_count,
402            fallback_tile_count = statistics.fallback_tile_count,
403            deferred_tile_count = statistics.deferred_tile_count,
404            unavailable_tile_count = statistics.unavailable_tile_count,
405            "surface frame resolved"
406        );
407        statistics
408    }
409}

471    /// Acquires a surface-compatible graphics context through `context_source`,
472    /// negotiates the surface format against the target's capabilities, and
473    /// builds the tile and overlay pipelines on the acquired device.
474    pub fn new(
475        surface_target: SurfaceRenderTarget,
476        context_source: &dyn SurfaceGraphicsContextSource,
477    ) -> Result<Self, SurfaceRenderError> {
478        let graphics_context = context_source.request_surface_graphics_context(&surface_target)?;
479        let surface_capabilities = surface_target

490        let overlay_pipeline = build_overlay_pipeline(&graphics_context, surface_format);

544    /// Sizes the render area explicitly — the resize seam of the
545    /// lifecycle: only this call changes the swap-chain extent. A zero
546    /// extent is refused (a surface with no pixels is not configurable);
547    /// reconfiguring happens only when the extent actually changed, so a
548    /// repeated call on extent change is idempotent. The first configure
549    /// adopts the negotiated surface format and the documented present and
550    /// alpha modes; the format is never renegotiated, so a platform surprise
551    /// that drops support for it surfaces here as a configuration failure.
552    pub fn configure_extent(
553        &mut self,
554        width_pixels: u32,
555        height_pixels: u32,
556    ) -> Result<(), SurfaceRenderError> {
557        self.configure_surface_for_extent(width_pixels, height_pixels, false)
558    }

560    /// Configures the surface for an extent, overriding the `wgpu`
561    /// defaults for format, present and alpha mode exactly like the wasm
562    /// canvas does; `force_reconfiguration` is the stale-surface reset of
563    /// the acquisition retry.
564    fn configure_surface_for_extent(
565        &mut self,
566        width_pixels: u32,
567        height_pixels: u32,
568        force_reconfiguration: bool,
569    ) -> Result<(), SurfaceRenderError> {
570        if width_pixels == 0 || height_pixels == 0 {
571            return Err(SurfaceRenderError::SurfaceConfigurationFailed(
572                "the surface extent must be non-zero: a surface with no \
573                 pixels cannot be configured"
574                    .to_string(),
575            ));
576        }
577        if !force_reconfiguration
578            && self.configured_surface_extent() == Some((width_pixels, height_pixels))
579        {
580            return Ok(());
581        }
582        let mut configuration = self
583            .surface_target
584            .surface
585            .get_default_config(&self.graphics_context.adapter, width_pixels, height_pixels)
586            .ok_or_else(|| {
587                SurfaceRenderError::SurfaceConfigurationFailed(format!(
588                    "the surface supports none of the surface formats \
589                     {SURFACE_FORMAT_PREFERENCE:?}"
590                ))
591            })?;
592        configuration.format = self.surface_format;
593        configuration.present_mode = PRESENT_MODE;
594        configuration.alpha_mode =
595            negotiated_alpha_mode(&self.surface_target.surface, &self.graphics_context.adapter)?;
596        configuration.desired_maximum_frame_latency = DESIRED_MAXIMUM_FRAME_LATENCY;
597        self.surface_target
598            .surface
599            .configure(&self.graphics_context.device, &configuration);
600        Ok(())
601    }

603    /// Reads the surface texture at the currently configured extent,
604    /// reconfiguring the surface once on a stale or lost surface between
605    /// attempts (the documented stale-surface retry, mirroring the wasm
606    /// canvas).
607    fn acquire_surface_texture(&mut self) -> Result<wgpu::SurfaceTexture, SurfaceRenderError> {
608        for attempt_index in 0..SURFACE_ACQUISITION_ATTEMPTS {
609            match self.surface_target.surface.get_current_texture() {
610                wgpu::CurrentSurfaceTexture::Success(surface_texture)
611                | wgpu::CurrentSurfaceTexture::Suboptimal(surface_texture) => {
612                    return Ok(surface_texture);
613                }
614                // Outdated, Lost and Validation mean a stale or dead surface
615                // configuration is at fault, so the next attempt reconfigures
616                // (see the `wgpu` guidance on each) and tries again.
617                wgpu::CurrentSurfaceTexture::Outdated
618                | wgpu::CurrentSurfaceTexture::Lost
619                | wgpu::CurrentSurfaceTexture::Validation
620                    if attempt_index + 1 < SURFACE_ACQUISITION_ATTEMPTS =>
621                {
622                    let (width_pixels, height_pixels) = self
623                        .configured_surface_extent()
624                        .expect("checked by the caller before acquiring");
625                    self.configure_surface_for_extent(width_pixels, height_pixels, true)?;
626                }
627                status_variant => {
628                    return Err(SurfaceRenderError::SurfaceTextureAcquisitionFailed(
629                        format!(
630                            "the map surface could not provide a frame texture: \
631                             {status_variant:?}"
632                        ),
633                    ));
634                }
635            }
636        }
637        Err(SurfaceRenderError::SurfaceTextureAcquisitionFailed(
638            "the map surface was never reconfigured into a usable state".to_string(),
639        ))
640    }

642    /// Renders one frame of tiles plus overlays to the live surface and
643    /// presents it. The three-layer variant adds the location-indicator
644    /// layer; this signature keeps the two-layer contract byte-stable by
645    /// delegating with no indicator.
646    pub fn render_frame(
647        &mut self,
648        camera: &MapCamera,
649        viewport: &ViewportPixels,
650        prepared_tiles: &[PreparedTile],
651        pool: &mut TileTexturePool,
652        overlays: &[MapOverlay],
653    ) -> Result<SurfaceFrameStatistics, SurfaceRenderError> {
654        self.render_frame_layers(camera, viewport, prepared_tiles, pool, overlays, None)
655    }

657    /// The frame's explicit three-layer variant: the pooled tiles' textured
658    /// quads, then the overlay layer, then the location-indicator layer
659    /// (the indicator draws last, the frame's explicit top layer — see
660    /// [`crate::location_indicator::LOCATION_INDICATOR_TOP_LAYER_DRAW_ORDER`]).
661    ///
662    /// The frame composition — quad assembly, texture uploads with
663    /// revision-stamped cache reuse, corrupted-payload eviction, the
664    /// tiles-then-overlays-then-indicator pass order — is the headless
665    /// renderer's behavior exactly (see [`HeadlessTileRenderer`'s frame
666    /// documentation][`crate::headless::HeadlessTileRenderer::render_captured_frame_layers`]);
667    /// differences are only the attachment and its outcome: the frame draws
668    /// into the next acquired surface texture, presented instead of
669    /// captured.
670    ///
671    /// Returns the frame's [`SurfaceFrameStatistics`] — the scheduler
672    /// readiness verdicts and the overlay triangles the presented frame
673    /// carried.
674    ///
675    /// Requires a prior [`Self::configure_extent`]; the frame's viewport
676    /// must match the configured extent exactly (pixels are physical). An
677    /// empty frame request set, an empty overlay list, and a cleared
678    /// indicator naturally render the clear color alone.
679    pub fn render_frame_layers(
680        &mut self,
681        camera: &MapCamera,
682        viewport: &ViewportPixels,
683        prepared_tiles: &[PreparedTile],
684        pool: &mut TileTexturePool,
685        overlays: &[MapOverlay],
686        location_indicator_layer: Option<&crate::location_indicator::LocationIndicatorLayer>,
687    ) -> Result<SurfaceFrameStatistics, SurfaceRenderError> {
688        let Some((configured_width_pixels, configured_height_pixels)) =
689            self.configured_surface_extent()
690        else {
691            return Err(SurfaceRenderError::SurfaceConfigurationFailed(
692                "the surface is not configured; call configure_extent \
693                 before rendering frames"
694                    .to_string(),
695            ));
696        };
697        let viewport_extent = (viewport.width_pixels as u32, viewport.height_pixels as u32);
698        if viewport_extent != (configured_width_pixels, configured_height_pixels) {
699            return Err(SurfaceRenderError::ViewportExtentMismatch {
700                viewport_extent,
701                configured_extent: (configured_width_pixels, configured_height_pixels),
702            });
703        }
704
705        let quads = self
706            .quad_composer
707            .frame_quads(camera, viewport, prepared_tiles, pool);
708
709        // Drop uploads the pool invalidated — bytes replaced or evicted since
710        // their texture was uploaded — before deciding what needs uploading.
711        self.gpu_texture_cache
712            .retain_matching(|tile_key, pool_revision| {
713                pool.payload_revision(tile_key) == Some(pool_revision)
714            });
715        // Touch the recency of everything this frame draws before any upload,
716        // so an upload's eviction pass below cannot remove an entry a later
717        // quad of this same frame reuses.
718        for quad in &quads {
719            self.gpu_texture_cache.touch(&quad.texture_tile_key);
720        }
721
722        // The device and queue handles are cheap reference-counted clones;
723        // borrowing them out of the context instead would pin the shared
724        // borrow of `self` across the mutable upload path below.
725        let device = self.graphics_context.device.clone();
726        let queue = self.graphics_context.queue.clone();
727
728        // One vertex array per world-copy fragment, each tagged with the quad
729        // whose texture it draws.
730        let quad_fragment_vertices = self
731            .quad_composer
732            .quad_fragment_vertices(camera, viewport, &quads);
733        let mut fragment_vertices = Vec::new();
734        let mut fragment_texture_positions = Vec::with_capacity(quad_fragment_vertices.len());
735        for (quad_index, quad_fragments) in quad_fragment_vertices.iter().enumerate() {
736            for quad_fragment in quad_fragments {
737                fragment_texture_positions.push(quad_index);
738                fragment_vertices.push(*quad_fragment);
739            }
740        }
741        if fragment_vertices.len() > MAXIMUM_QUADS_PER_FRAME {
742            return Err(SurfaceRenderError::DrawListTooLarge {
743                quad_count: fragment_vertices.len(),
744                maximum_quad_count: MAXIMUM_QUADS_PER_FRAME,
745            });
746        }
747
748        // Upload once per quad; every world-copy fragment of the quad draws
749        // with the same texture bind group. A quad whose payload is missing
750        // or undecodable contributes no draw (see `upload_tile_texture`).
751        let mut tile_textures_and_bind_groups = Vec::with_capacity(quads.len());
752        for quad in &quads {
753            tile_textures_and_bind_groups.push(self.upload_tile_texture(
754                &device,
755                quad.texture_tile_key,
756                pool,
757            )?);
758        }
759
760        let (vertex_bytes, index_bytes) = encoded_fragment_buffers(&fragment_vertices);
761
762        // An all-missing frame has no draw list; its buffers would be empty,
763        // which `wgpu` buffers cannot represent, so only buffers of a frame
764        // that draws are created and the all-missing frame renders as the
765        // clear color alone.
766        let vertex_buffer = (!fragment_vertices.is_empty()).then(|| {
767            create_filled_buffer(
768                &device,
769                &queue,
770                &vertex_bytes,
771                wgpu::BufferUsages::VERTEX,
772                Some("map-rendering-surface-tile-vertices"),
773            )
774        });
775        let index_buffer = (!fragment_vertices.is_empty()).then(|| {
776            create_filled_buffer(
777                &device,
778                &queue,
779                &index_bytes,
780                wgpu::BufferUsages::INDEX,
781                Some("map-rendering-surface-tile-indices"),
782            )
783        });
784
785        // The overlay layer assembles through the same camera pose as the
786        // tile quads and draws after them; an empty overlay list creates no
787        // buffer and contributes no draw.
788        let overlay_vertices = overlay_triangle_vertices(
789            camera,
790            viewport,
791            self.quad_composer.field_of_view_degrees(),
792            overlays,
793        );
794        let overlay_vertex_buffer = (!overlay_vertices.is_empty()).then(|| {
795            create_filled_buffer(
796                &device,
797                &queue,
798                &encoded_overlay_vertices(&overlay_vertices),
799                wgpu::BufferUsages::VERTEX,
800                Some("map-rendering-surface-overlay-vertices"),
801            )
802        });
803
804        // The location-indicator layer assembles through the same camera pose
805        // and draws through the same overlay pipeline after the overlays; a
806        // cleared (None) indicator creates no buffer and draws nothing.
807        let indicator_vertices = crate::location_indicator::location_indicator_vertices(
808            camera,
809            viewport,
810            self.quad_composer.field_of_view_degrees(),
811            location_indicator_layer,
812        );
813        let indicator_vertex_buffer = (!indicator_vertices.is_empty()).then(|| {
814            create_filled_buffer(
815                &device,
816                &queue,
817                &encoded_overlay_vertices(&indicator_vertices),
818                wgpu::BufferUsages::VERTEX,
819                Some("map-rendering-surface-indicator-vertices"),
820            )
821        });
822
823        let surface_texture = self.acquire_surface_texture()?;
824        let frame_texture_view = surface_texture
825            .texture
826            .create_view(&wgpu::TextureViewDescriptor::default());
827
828        let mut command_encoder = device.create_command_encoder(&wgpu::CommandEncoderDescriptor {
829            label: Some("map-rendering-surface-frame-encoder"),
830        });
831        {
832            let mut render_pass = command_encoder.begin_render_pass(&wgpu::RenderPassDescriptor {
833                label: Some("map-rendering-surface-frame-pass"),
834                color_attachments: &[Some(wgpu::RenderPassColorAttachment {
835                    view: &frame_texture_view,
836                    resolve_target: None,
837                    depth_slice: None,
838                    ops: wgpu::Operations {
839                        load: wgpu::LoadOp::Clear(wgpu::Color {
840                            r: f64::from(CLEAR_COLOR_RGBA[0]),
841                            g: f64::from(CLEAR_COLOR_RGBA[1]),
842                            b: f64::from(CLEAR_COLOR_RGBA[2]),
843                            a: f64::from(CLEAR_COLOR_RGBA[3]),
844                        }),
845                        store: wgpu::StoreOp::Store,
846                    },
847                })],
848                depth_stencil_attachment: None,
849                timestamp_writes: None,
850                occlusion_query_set: None,
851                multiview_mask: None,
852            });
853            if !fragment_vertices.is_empty() {
854                render_pass.set_pipeline(&self.tile_pipeline);
855                render_pass.set_vertex_buffer(0, vertex_buffer.expect("frame draws").slice(..));
856                render_pass.set_index_buffer(
857                    index_buffer.expect("frame draws").slice(..),
858                    wgpu::IndexFormat::Uint16,
859                );
860                for (fragment_index, texture_position) in
861                    fragment_texture_positions.iter().copied().enumerate()
862                {
863                    let Some((_, bind_group)) = &tile_textures_and_bind_groups[texture_position]
864                    else {
865                        continue;
866                    };
867                    render_pass.set_bind_group(0, bind_group, &[]);
868                    let first_index = u32::try_from(fragment_index * INDICES_PER_QUAD)
869                        .expect("the draw list was bounded above");
870                    render_pass.draw_indexed(
871                        first_index..first_index + INDICES_PER_QUAD as u32,
872                        0,
873                        0..1,
874                    );
875                }
876            }
877            if let Some(overlay_vertex_buffer) = &overlay_vertex_buffer {
878                render_pass.set_pipeline(&self.overlay_pipeline);
879                render_pass.set_vertex_buffer(0, overlay_vertex_buffer.slice(..));
880                let overlay_vertex_count = u32::try_from(overlay_vertices.len())
881                    .expect("the overlay triangle list stays inside u32");
882                render_pass.draw(0..overlay_vertex_count, 0..1);
883            }
884            if let Some(indicator_vertex_buffer) = &indicator_vertex_buffer {
885                render_pass.set_pipeline(&self.overlay_pipeline);
886                render_pass.set_vertex_buffer(0, indicator_vertex_buffer.slice(..));
887                let indicator_vertex_count = u32::try_from(indicator_vertices.len())
888                    .expect("the indicator triangle list stays inside u32");
889                render_pass.draw(0..indicator_vertex_count, 0..1);
890            }
891        }
892        // The frame's commands are encoded against the acquired texture
893        // already owned exclusively by the present queue; submitting and
894        // presenting them is the composition's only observable outcome.
895        queue.submit([command_encoder.finish()]);
896        self.graphics_context.queue.present(surface_texture);
897        Ok(
898            SurfaceFrameStatistics::from_frame_readiness_and_overlay_vertices(
899                prepared_tiles,
900                overlay_vertices.len(),
901            ),
902        )
903    }

905    /// Uploads a pooled payload as a renderable texture, touching the pool's
906    /// recency for the tile on both the fresh-load and cached-reuse paths.
907    /// Returns `Ok(None)` — no error — for two skip
908    /// cases: the payload left the pool between assembly and upload (an
909    /// eviction raced the frame), or the payload fails to decode, in which
910    /// case the corrupt bytes are evicted so the scheduler's next fetch
911    /// replaces them and every later frame recovers. Otherwise returns the
912    /// texture and bind group, freshly uploaded on a payload-revision miss or
913    /// reused from the GPU texture cache on a hit.
914    ///
915    /// Verbatim mirror of the headless renderer's private upload path; only
916    /// the uploaded texture format differs ([`TILE_TEXTURE_FORMAT`], plain
917    /// RGBA8, instead of the headless capture's frame format — the two are
918    /// the same format there only because headless captures in RGBA8 too).
919    fn upload_tile_texture(
920        &mut self,
921        device: &Device,
922        tile_key: crate::tile_key::TileKey,
923        pool: &mut TileTexturePool,
924    ) -> Result<Option<(Texture, wgpu::BindGroup)>, SurfaceRenderError> {
925        // The revision stamps the cache entry, so a pool miss also misses the
926        // cache: an upload whose source bytes are gone must never draw.
927        let Some(payload_revision) = pool.payload_revision(&tile_key) else {
928            return Ok(None);
929        };
930        if let Some((texture, bind_group)) =
931            self.gpu_texture_cache.lookup(&tile_key, payload_revision)
932        {
933            // Drawing from the cache is still a use of the pooled bytes: keep
934            // the payload's recency fresh or LRU would evict exactly the
935            // steady-camera tiles the cache exists to avoid refetching.
936            pool.refresh_recency(&tile_key);
937            return Ok(Some((texture.clone(), bind_group.clone())));
938        }
939        let Some(encoded_image_bytes) = pool.lookup(&tile_key).map(<[u8]>::to_vec) else {
940            return Ok(None);
941        };
942        match image::load_from_memory(&encoded_image_bytes) {
943            Ok(decoded_image) => {
944                let (texture, bind_group, texture_bytes) =
945                    self.upload_decoded_texture(device, decoded_image);
946                self.gpu_texture_cache.insert(
947                    tile_key,
948                    payload_revision,
949                    texture.clone(),
950                    bind_group.clone(),
951                    texture_bytes,
952                );
953                Ok(Some((texture, bind_group)))
954            }
955            Err(_) => {
956                pool.evict(&tile_key);
957                Ok(None)
958            }
959        }
960    }

962    /// Creates a renderable texture from a decoded image, queues its RGBA8
963    /// pixels onto the device, and builds the tile bind group; the byte
964    /// count feeds the cache budget as the texture's GPU cost. Verbatim
965    /// mirror of the headless renderer's private helper, against
966    /// [`TILE_TEXTURE_FORMAT`].
967    fn upload_decoded_texture(
968        &self,
969        device: &Device,
970        decoded_image: DynamicImage,
971    ) -> (Texture, wgpu::BindGroup, u64) {
972        let rgba_image = to_rgba8(decoded_image);
973        let rgba_bytes = rgba_image.as_raw();
974        let (width_pixels, height_pixels) = rgba_image.dimensions();
975
976        let texture = device.create_texture(&wgpu::TextureDescriptor {
977            label: Some("map-rendering-surface-tile-texture"),
978            size: Extent3d {
979                width: width_pixels,
980                height: height_pixels,
981                depth_or_array_layers: 1,
982            },
983            mip_level_count: 1,
984            sample_count: 1,
985            dimension: wgpu::TextureDimension::D2,
986            format: TILE_TEXTURE_FORMAT,
987            usage: TextureUsages::TEXTURE_BINDING | TextureUsages::COPY_DST,
988            view_formats: &[],
989        });
990        self.graphics_context.queue.write_texture(
991            TexelCopyTextureInfo {
992                texture: &texture,
993                mip_level: 0,
994                origin: wgpu::Origin3d::ZERO,
995                aspect: TextureAspect::All,
996            },
997            rgba_bytes,
998            TexelCopyBufferLayout {
999                offset: 0,
1000                bytes_per_row: Some(PIXEL_BYTES * width_pixels),
1001                rows_per_image: None,
1002            },
1003            Extent3d {
1004                width: width_pixels,
1005                height: height_pixels,
1006                depth_or_array_layers: 1,
1007            },
1008        );
1009        let bind_group = self.create_tile_bind_group(device, &texture);
1010        let texture_bytes = u64::from(width_pixels) * u64::from(height_pixels) * PIXEL_BYTES as u64;
1011        (texture, bind_group, texture_bytes)
1012    }
1013
1014    fn create_tile_bind_group(&self, device: &Device, texture: &Texture) -> wgpu::BindGroup {
1015        let texture_view = texture.create_view(&wgpu::TextureViewDescriptor::default());
1016        device.create_bind_group(&wgpu::BindGroupDescriptor {
1017            label: Some("map-rendering-surface-tile-bind-group"),
1018            layout: &self.tile_pipeline.get_bind_group_layout(0),
1019            entries: &[
1020                wgpu::BindGroupEntry {
1021                    binding: 0,
1022                    resource: wgpu::BindingResource::TextureView(&texture_view),
1023                },
1024                wgpu::BindGroupEntry {
1025                    binding: 1,
1026                    resource: wgpu::BindingResource::Sampler(&self.tile_sampler),
1027                },
1028            ],
1029        })
1030    }

1043/// Serializes fragment vertices into the vertex and index byte streams the GPU
1044/// buffers expect: little-endian floats, [`CORNERS_PER_QUAD`] corners per
1045/// world-copy fragment, and the two-triangle index pattern `0, 1, 2, 2, 1, 3`
1046/// ([`INDICES_PER_QUAD`] per fragment). Verbatim mirror of the headless
1047/// renderer's private helper.
1048fn encoded_fragment_buffers(fragment_vertices: &[[ClipSpaceTileVertex; 4]]) -> (Vec<u8>, Vec<u8>) {
1049    let vertex_stride = std::mem::size_of::<ClipSpaceTileVertex>();
1050    let mut vertex_bytes =
1051        Vec::with_capacity(fragment_vertices.len() * vertex_stride * CORNERS_PER_QUAD);
1052    let mut index_bytes =
1053        Vec::with_capacity(fragment_vertices.len() * INDICES_PER_QUAD * std::mem::size_of::<u16>());
1054    for (fragment_index, vertices) in fragment_vertices.iter().enumerate() {
1055        let base_corner_index = u16::try_from(fragment_index * CORNERS_PER_QUAD)
1056            .expect("the draw list was bounded above");
1057        for component in vertices.iter().flat_map(|vertex| {
1058            vertex
1059                .clip_position
1060                .iter()
1061                .chain(vertex.texture_coordinates.iter())
1062        }) {
1063            vertex_bytes.extend_from_slice(&component.to_le_bytes());
1064        }
1065        for index in [
1066            base_corner_index,
1067            base_corner_index + 1,
1068            base_corner_index + 2,
1069            base_corner_index + 2,
1070            base_corner_index + 1,
1071            base_corner_index + 3,
1072        ] {
1073            index_bytes.extend_from_slice(&index.to_le_bytes());
1074        }
1075    }
1076    (vertex_bytes, index_bytes)
1077}

1079/// Serializes the overlay triangle-list vertices into the vertex byte stream
1080/// the GPU buffer expects: little-endian floats. The overlay draws
1081/// non-indexed (a plain vertex draw over the flat triangle list), so no index
1082/// stream exists for this layer. Verbatim mirror of the headless renderer's
1083/// private helper.
1084fn encoded_overlay_vertices(overlay_vertices: &[ClipSpaceOverlayVertex]) -> Vec<u8> {
1085    let mut vertex_bytes = Vec::with_capacity(std::mem::size_of_val(overlay_vertices));
1086    for vertex in overlay_vertices {
1087        for component in vertex.clip_position.iter().chain(vertex.rgba_color.iter()) {
1088            vertex_bytes.extend_from_slice(&component.to_le_bytes());
1089        }
1090    }
1091    vertex_bytes
1092}

1123        .find(|preferred_format| supported_formats.contains(preferred_format))

1127/// Negotiates the configured alpha mode: the preferred opaque compositing
1128/// when the surface supports it, otherwise the first mode the platform
1129/// reports. Mirrors `preferred_surface_format`'s prefer-then-negotiate
1130/// pattern — a hardware-backed surface, unlike a web canvas, is free to
1131/// restrict the compositing modes it accepts, and refusing such a surface
1132/// (`Android emulator`) would render the SDK on none of that backend's own
1133/// listed surfaces.
1134fn negotiated_alpha_mode(
1135    surface: &wgpu::Surface<'static>,
1136    adapter: &wgpu::Adapter,
1137) -> Result<wgpu::CompositeAlphaMode, SurfaceRenderError> {
1138    let surface_capabilities = surface.get_capabilities(adapter);
1139    if surface_capabilities
1140        .alpha_modes
1141        .contains(&PREFERRED_ALPHA_MODE)
1142    {
1143        return Ok(PREFERRED_ALPHA_MODE);
1144    }
1145    surface_capabilities
1146        .alpha_modes
1147        .first()
1148        .copied()
1149        .ok_or_else(|| {
1150            SurfaceRenderError::SurfaceConfigurationFailed(
1151                "the surface supports no alpha mode at all, which no wgpu \
1152             backend reports; refusing to configure"
1153                    .to_string(),
1154            )
1155        })
1156}

1158/// Builds the textured-quad pipeline of the tile layer against a target
1159/// surface format. Verbatim mirror of the headless pipeline builder; only
1160/// the target format and the labels differ.
1161fn build_tile_pipeline(
1162    graphics_context: &SurfaceGraphicsContext,
1163    surface_format: wgpu::TextureFormat,
1164) -> wgpu::RenderPipeline {
1165    let shader_module =
1166        graphics_context
1167            .device
