1//! Native, memory-mapped consumption seam for `OfflineRoutingNetworkGraph`
2//! archive files.
3//!
4//! This is milestone M7 slice 2's zero-copy handoff: the mapped file answers
5//! byte slices that the fully validating in-memory reader
6//! ([`OfflineRoutingNetworkGraph::from_bytes`]) consumes unchanged, so both
7//! load paths — the file-mapped one here and the in-memory one in
8//! [`crate::archive`] — share the entire typed corruption model. The module
9//! is compiled only for native targets (see the workspace manifest): the
10//! browser loads the graph from in-memory bytes, with no `mmap` in the wasm
11//! path.
12//!
13//! # Alignment validation
14//!
15//! Opening the mapping validates the file-level alignment invariants the
16//! archive format fixes ([`crate::archive`]):
17//!
18//! - the mapped region covers at least [`FILE_HEADER_BYTES`];
19//! - its byte length is a multiple of [`ARCHIVE_RESOURCE_ALIGNMENT`] — the
20//!   deterministic writer always pads the archive's total length to that
21//!   boundary, so a region that breaks it is a truncated or corrupt archive;
22//! - the mapped base address honors the same alignment, which keeps the
23//!   little-endian `u64` table-of-contents field reads naturally aligned.
24//!   Page-granular mapping guarantees this; it is validated, not assumed.
25//!
26//! Deeper structure (table of contents, record widths, cross-references) is
27//! validated by the reader, exactly as for in-memory bytes.
28//!
29//! # Error mapping
30//!
31//! File-level failures (missing, unreadable, mapping refused) surface as
32//! `SDKError::OfflineRoutingFailure(OfflineGraphInvalidReason::ArchiveMalformed)`:
33//! the typed corruption vocabulary deliberately has no I/O cause (the
34//! [`SDKError`] variants serialize across the frozen FFI contract), and an
35//! archive that cannot be mapped is unusable in exactly the way a malformed
36//! one is. A recognized-but-unsupported archive still fails as
37//! `VersionUnsupported` out of the shared reader.
38
39use std::path::Path;

41use geospatial_types::OfflineGraphInvalidReason;
42use geospatial_types::SDKError;
43use memmap2::Mmap;

45use crate::archive::{ARCHIVE_RESOURCE_ALIGNMENT, FILE_HEADER_BYTES, OfflineRoutingNetworkGraph};

47/// A memory-mapped archive file.
48///
49/// The mapped region stays valid for the struct's lifetime and answers byte
50/// slices through [`Self::mapped_bytes`]; the validating reader materializes
51/// the graph records out of them without touching the filesystem again.
52#[derive(Debug)]
53pub struct MemoryMappedArchive {
54    mapped_file: Mmap,
55}
56
57impl MemoryMappedArchive {
58    /// Memory-maps an archive file and validates its alignment invariants.
59    ///
60    /// Structural validation beyond alignment is deferred to the shared
61    /// reader (see the module docs).
62    ///
63    /// # Errors
64    /// `SDKError::OfflineRoutingFailure(ArchiveMalformed)` for every I/O and
65    /// alignment failure (see the module docs); reader-side failures are
66    /// typed unchanged by [`OfflineRoutingNetworkGraph::from_bytes`].
67    pub fn open(archive_path: &Path) -> Result<Self, SDKError> {
68        let source_file =
69            std::fs::File::open(archive_path).map_err(|_| archive_malformed_error())?;
70        // SAFETY: `Mmap::map` has no preconditions beyond a readable file
71        // descriptor; the resulting view is read-only over the whole file
72        // and never mutated in place.
73        let mapped_file =
74            unsafe { Mmap::map(&source_file) }.map_err(|_| archive_malformed_error())?;
75
76        let mapped_bytes: &[u8] = &mapped_file;
77        if mapped_bytes.len() < FILE_HEADER_BYTES
78            || !mapped_bytes
79                .len()
80                .is_multiple_of(ARCHIVE_RESOURCE_ALIGNMENT)
81            || !mapped_bytes
82                .as_ptr()
83                .addr()
84                .is_multiple_of(ARCHIVE_RESOURCE_ALIGNMENT)
85        {
86            return Err(archive_malformed_error());
87        }
88        Ok(Self { mapped_file })
89    }
90
91    /// The mapped archive bytes, exactly as they sit in the file.
92    pub fn mapped_bytes(&self) -> &[u8] {
93        &self.mapped_file
94    }
95}
96
97/// Memory-maps an archive file and fully validates it, returning the
98/// [`OfflineRoutingNetworkGraph`] graph itself (the validation materializes
99/// the typed records from the mapped bytes).
100///
101/// # Errors
102/// See [`MemoryMappedArchive::open`] and
103/// [`OfflineRoutingNetworkGraph::from_bytes`].
104pub fn offline_routing_network_graph_from_mapped_file(
105    archive_path: &Path,
106) -> Result<OfflineRoutingNetworkGraph, SDKError> {
107    OfflineRoutingNetworkGraph::from_bytes(MemoryMappedArchive::open(archive_path)?.mapped_bytes())
108}
109
110/// The typed failure every unmappable or alignment-violating archive maps to
111/// (see the module docs for the mapping rationale).
112fn archive_malformed_error() -> SDKError {
113    SDKError::OfflineRoutingFailure(OfflineGraphInvalidReason::ArchiveMalformed)
114}
