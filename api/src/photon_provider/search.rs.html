1//! Photon search, autocomplete and geocoding.
2//!
3//! Photon (`photon.komoot.io`) answers forward queries from `/api` and
4//! reverse queries from `/reverse`, both as a GeoJSON `FeatureCollection`.
5//! This module maps that document onto the SDK's provider-neutral
6//! [`SearchPlaceProvider`] and [`GeocodingProvider`] contracts.
7

16/// Completion candidates a typing query asks for. The reference applications
17/// show five rows, so the provider never returns more than the consumer
18/// displays.
19pub const AUTOCOMPLETE_RESULT_LIMIT: u32 = 5;
20
21/// Candidates a settled search asks for: a longer list than the type-ahead
22/// for a query the user has finished writing.
23pub const SEARCH_RESULT_LIMIT: u32 = 10;
24
25/// The forward-query endpoint path.
26const SEARCH_ENDPOINT_PATH: &str = "api";
27
28/// The reverse-query endpoint path.
29const REVERSE_ENDPOINT_PATH: &str = "reverse";
30
31/// Photon's property naming the feature class, mapped onto the SDK's
32/// geocode-quality scale.
33const TYPE_PROPERTY_NAME: &str = "type";
34
35/// Photon's display-name property; when present and non-empty it is the
36/// provider's own label for the feature.
37const NAME_PROPERTY_NAME: &str = "name";
38
39/// Address components joined into a display line when the feature carries no
40/// name of its own.
41const STREET_PROPERTY_NAME: &str = "street";
42/// See [`STREET_PROPERTY_NAME`].
43const HOUSE_NUMBER_PROPERTY_NAME: &str = "housenumber";
44/// See [`STREET_PROPERTY_NAME`].
45const POSTCODE_PROPERTY_NAME: &str = "postcode";
46/// See [`STREET_PROPERTY_NAME`].
47const CITY_PROPERTY_NAME: &str = "city";
48/// See [`STREET_PROPERTY_NAME`].
49const STATE_PROPERTY_NAME: &str = "state";
50/// See [`STREET_PROPERTY_NAME`].
51const COUNTRY_PROPERTY_NAME: &str = "country";
52
53/// Separator between the display line's address components.
54const DISPLAY_COMPONENT_SEPARATOR: &str = ", ";
55
56/// Photon place lookup, autocomplete and reverse geocoding over one
57/// deployment.
58pub struct PhotonPlaceProvider {
59    /// The HTTP client every query executes through.

70#[async_trait]
71impl SearchPlaceProvider for PhotonPlaceProvider {
72    async fn search_autocomplete(
73        &self,
74        query_text: &str,
75        search_boundary: Option<BoundingBox>,
76    ) -> Result<Vec<PlaceResult>, SDKError> {
77        self.run_search(query_text, search_boundary, AUTOCOMPLETE_RESULT_LIMIT)
78            .await
79    }
80
81    async fn search_places(
82        &self,
83        query_text: &str,
84        search_boundary: Option<BoundingBox>,
85    ) -> Result<Vec<PlaceResult>, SDKError> {
86        self.run_search(query_text, search_boundary, SEARCH_RESULT_LIMIT)
87            .await
88    }
89}

91#[async_trait]
92impl GeocodingProvider for PhotonPlaceProvider {
93    async fn forward_geocode(&self, address_text: &str) -> Result<Vec<PlaceResult>, SDKError> {
94        self.run_search(address_text, None, SEARCH_RESULT_LIMIT)
95            .await
96    }
97
98    async fn reverse_geocode(
99        &self,
100        coordinate: &GeographicCoordinate,
101    ) -> Result<Vec<PlaceResult>, SDKError> {
102        let query_pairs = vec![
103            ("lon".to_string(), coordinate.longitude.to_string()),
104            ("lat".to_string(), coordinate.latitude.to_string()),
105            ("limit".to_string(), 1.to_string()),
106        ];
107        let payload = self
108            .provider_client
109            .get_json(REVERSE_ENDPOINT_PATH, &query_pairs)
110            .await?;
111        places_from_feature_collection(&payload)
112    }
113}

115impl PhotonPlaceProvider {
116    /// Runs one forward query and maps its features onto place results.
117    async fn run_search(
118        &self,
119        query_text: &str,
120        search_boundary: Option<BoundingBox>,
121        result_limit: u32,
122    ) -> Result<Vec<PlaceResult>, SDKError> {
123        if query_text.trim().is_empty() {
124            return Err(SDKError::InvalidCoordinateInput(
125                "search query text must not be empty".to_string(),
126            ));
127        }
128        let mut query_pairs = vec![
129            ("q".to_string(), query_text.to_string()),
130            ("limit".to_string(), result_limit.to_string()),
131        ];
132        if let Some(bounding_box) = search_boundary {
133            // Photon's bbox filter: minimum longitude, minimum latitude,
134            // maximum longitude, maximum latitude.
135            query_pairs.push((
136                "bbox".to_string(),
137                format!(
138                    "{},{},{},{}",
139                    bounding_box.south_west.longitude,
140                    bounding_box.south_west.latitude,
141                    bounding_box.north_east.longitude,
142                    bounding_box.north_east.latitude
143                ),
144            ));
145        }
146        let payload = self
147            .provider_client
148            .get_json(SEARCH_ENDPOINT_PATH, &query_pairs)
149            .await?;
150        places_from_feature_collection(&payload)
151    }
152}

154/// Maps a Photon `FeatureCollection` onto place results, skipping features
155/// without a usable point.
156fn places_from_feature_collection(payload: &Value) -> Result<Vec<PlaceResult>, SDKError> {
157    let features = payload
158        .get("features")
159        .and_then(Value::as_array)
160        .ok_or_else(|| {
161            SDKError::ResponseParsingFailure(
162                "the Photon response carries no features array".to_string(),
163            )
164        })?;
165    let mut places = Vec::with_capacity(features.len());
166    for feature in features {
167        if let Some(place) = place_from_feature(feature) {
168            places.push(place);
169        }
170    }
171    Ok(places)
172}
173
174/// One feature's place result, or `None` when its geometry carries no point.
175fn place_from_feature(feature: &Value) -> Option<PlaceResult> {
176    let coordinates = feature.get("geometry")?.get("coordinates")?.as_array()?;
177    let longitude = coordinates.first()?.as_f64()?;
178    let latitude = coordinates.get(1)?.as_f64()?;
179    let coordinate = GeographicCoordinate::from_degrees(latitude, longitude).ok()?;
180    let properties = feature.get("properties");
181    let display_text = property_text(properties, NAME_PROPERTY_NAME)
182        .filter(|name| !name.trim().is_empty())
183        .unwrap_or_else(|| address_line(properties));
184    let geocode_quality = property_text(properties, TYPE_PROPERTY_NAME)
185        .map(|feature_type| geocode_quality_for_feature_type(&feature_type))
186        .unwrap_or(GeocodeQuality::Point);
187    Some(PlaceResult {
188        structured_address_text: display_text,
189        coordinate,
190        geocode_quality,
191    })
192}

194/// Joins the address components Photon published into one display line,
195/// skipping the ones it omitted.
196fn address_line(properties: Option<&Value>) -> String {
197    let street_line = match (
198        property_text(properties, HOUSE_NUMBER_PROPERTY_NAME),
199        property_text(properties, STREET_PROPERTY_NAME),
200    ) {
201        (Some(house_number), Some(street)) => Some(format!("{house_number} {street}")),
202        (None, Some(street)) => Some(street),
203        _ => None,
204    };
205    let locality_line = match (
206        property_text(properties, POSTCODE_PROPERTY_NAME),
207        property_text(properties, CITY_PROPERTY_NAME),
208    ) {
209        (Some(postcode), Some(city)) => Some(format!("{postcode} {city}")),
210        (None, Some(city)) => Some(city),
211        _ => None,
212    };
213    [
214        street_line,
215        locality_line,
216        property_text(properties, STATE_PROPERTY_NAME),
217        property_text(properties, COUNTRY_PROPERTY_NAME),
218    ]
219    .into_iter()
220    .flatten()
221    .collect::<Vec<String>>()
222    .join(DISPLAY_COMPONENT_SEPARATOR)
223}
224
225/// Reads one string property, treating a blank value as absent.
226fn property_text(properties: Option<&Value>, property_name: &str) -> Option<String> {
227    properties?
228        .get(property_name)?
229        .as_str()
230        .map(str::to_string)
231        .filter(|value| !value.trim().is_empty())
232}
233
234/// Maps Photon's feature class onto the SDK's geocode-quality scale.
235fn geocode_quality_for_feature_type(feature_type: &str) -> GeocodeQuality {
236    match feature_type {
237        "house" => GeocodeQuality::StreetAddress,
238        "street" => GeocodeQuality::Street,
239        "locality" | "district" | "city" => GeocodeQuality::City,
240        "county" => GeocodeQuality::County,
241        "state" => GeocodeQuality::State,
242        "country" => GeocodeQuality::Country,
243        "postcode" => GeocodeQuality::PostalCode,
244        _ => GeocodeQuality::Point,
245    }
246}

261    /// A transport that records the last request and serves a canned body.
262    struct RecordingTransport {
263        body: serde_json::Value,
264    }
265
266    #[async_trait::async_trait]
267    impl HttpTransport for RecordingTransport {
268        async fn execute(&self, _request: HttpRequest) -> Result<HttpResponse, TransportError> {
269            Ok(HttpResponse {
270                status_code: 200,
271                headers: Vec::new(),
272                body: serde_json::to_vec(&self.body).expect("the fixture serializes"),
273            })
274        }
275    }
276
277    fn fixture(name: &str) -> Value {
278        let fixture_path = format!("{}/fixtures/{name}", env!("CARGO_MANIFEST_DIR"));
279        serde_json::from_str(&std::fs::read_to_string(fixture_path).unwrap()).unwrap()
280    }
281
282    fn provider_with_fixture(name: &str) -> PhotonPlaceProvider {
283        let transport: Arc<dyn HttpTransport> = Arc::new(RecordingTransport {
284            body: fixture(name),
285        });
286        let client = PhotonHttpClient::with_transport(
287            PhotonProviderConfiguration {
288                base_url: "https://photon.example".to_string(),
289                user_agent: DEFAULT_PHOTON_USER_AGENT.to_string(),
290            },
291            0,
292            transport,
293        )
294        .expect("the test configuration is valid");
295        PhotonPlaceProvider::new(Arc::new(client))
296    }
