1//! Configuration and the HTTP facade shared by every Photon endpoint.
2//!
3//! Photon is keyless: a deployment is reached by its base URL, so this client
4//! carries no credential injection. It holds one base URL and one user agent,
5//! performs idempotent GETs through [`provider_transport`], and hands each
6//! response to the shared JSON interpretation below.
7
8use geospatial_types::SDKError;
9use provider_transport::{
10    HTTP_ERROR_STATUS_CODE_THRESHOLD, HttpMethod, HttpRequest, HttpResponse, HttpTransport,
11    ObservedTransport, RetryPolicy, default_transport, execute_idempotent_get,
12};
13use std::sync::Arc;
14use url::Url;

16/// Base URL of the public Photon instance when no configuration is provided.
17/// A self-hosted deployment replaces it with its own address.
18pub const DEFAULT_PHOTON_BASE_URL: &str = "https://photon.komoot.io";
19
20/// The registry key of this provider, used for request observation only.
21pub const PHOTON_PROVIDER_KEY: &str = "photon";
22
23/// User agent sent when the configuration carries none.
24pub const DEFAULT_PHOTON_USER_AGENT: &str =
25    concat!("rustic-maps-photon-provider/", env!("CARGO_PKG_VERSION"));
26
27/// Retries after the first attempt for one idempotent query GET. Retries
28/// apply only to transient transport and server failures.
29pub const DEFAULT_MAXIMUM_RETRY_ATTEMPTS: u32 = 2;
30
31/// Character cap on error bodies echoed into error strings.
32const MAXIMUM_ERROR_BODY_TAIL: usize = 256;
33
34/// Connection settings for one Photon deployment.
35///
36/// Photon needs no key, so the deployment URL is the only address
37/// information the service needs. `base_url` may name the public instance or
38/// a self-hosted one and must be an absolute URL without the API path.
39///
40/// `user_agent` identifies the SDK to the deployment's operator; an empty
41/// value suppresses the header entirely.
42#[derive(Debug, Clone, PartialEq, Eq)]
43pub struct PhotonProviderConfiguration {
44    /// Root URL of the Photon deployment.
45    pub base_url: String,
46    /// `User-Agent` header value sent with every request.
47    pub user_agent: String,
48}
49
50impl Default for PhotonProviderConfiguration {
51    fn default() -> Self {
52        Self {
53            base_url: DEFAULT_PHOTON_BASE_URL.to_string(),
54            user_agent: DEFAULT_PHOTON_USER_AGENT.to_string(),
55        }
56    }
57}

59/// GET-only JSON client against one Photon base URL.
60pub struct PhotonHttpClient {
61    base_url: String,
62    user_agent: String,
63    maximum_retry_attempts: u32,
64    transport: Arc<dyn HttpTransport>,
65}
66
67impl PhotonHttpClient {
68    /// Builds a client around the target's default transport.
69    pub fn new(configuration: PhotonProviderConfiguration) -> Result<Self, SDKError> {
70        Self::with_transport(
71            configuration,
72            DEFAULT_MAXIMUM_RETRY_ATTEMPTS,
73            default_transport()?,
74        )
75    }
76
77    /// Builds a client around an explicit transport and retry budget (the
78    /// registration path and tests use this seam). The supplied transport is
79    /// wrapped for observation, so every request made through this client
80    /// reports one redacted observation whichever construction path built it.
81    pub fn with_transport(
82        configuration: PhotonProviderConfiguration,
83        maximum_retry_attempts: u32,
84        transport: Arc<dyn HttpTransport>,
85    ) -> Result<Self, SDKError> {
86        validate_base_url(&configuration.base_url)?;
87        Ok(Self {
88            base_url: configuration.base_url,
89            user_agent: configuration.user_agent,
90            maximum_retry_attempts,
91            transport: Arc::new(ObservedTransport::new(
92                transport,
93                crate::PLACE_CONTRACT_NAME,
94                PHOTON_PROVIDER_KEY,
95            )),
96        })
97    }
98
99    /// Performs a bounded-retry GET against `{base_url}/{path}` with the
100    /// given query pairs and returns the decoded JSON payload.
101    pub async fn get_json(
102        &self,
103        path: &str,
104        query_pairs: &[(String, String)],
105    ) -> Result<serde_json::Value, SDKError> {
106        let request = self.build_request(path, query_pairs)?;
107        execute_idempotent_get(
108            self.transport.as_ref(),
109            &request,
110            RetryPolicy::new(self.maximum_retry_attempts),
111            interpret_json_response,
112        )
113        .await
114    }
115
116    /// Builds one GET request against `{base_url}/{path}`.
117    fn build_request(
118        &self,
119        path: &str,
120        query_pairs: &[(String, String)],
121    ) -> Result<HttpRequest, SDKError> {
122        let endpoint = format!(
123            "{}/{}",
124            self.base_url.trim_end_matches('/'),
125            path.trim_start_matches('/')
126        );
127        let mut url = Url::parse(&endpoint).map_err(|error| {
128            SDKError::NetworkRequestFailed(format!(
129                "invalid Photon request URL {endpoint}: {error}"
130            ))
131        })?;
132        {
133            let mut query_pairs_mut = url.query_pairs_mut();
134            for (name, value) in query_pairs {
135                query_pairs_mut.append_pair(name, value);
136            }
137        }
138        let headers = if self.user_agent.is_empty() {
139            Vec::new()
140        } else {
141            vec![("user-agent".to_string(), self.user_agent.clone())]
142        };
143        Ok(HttpRequest {
144            method: HttpMethod::Get,
145            url: url.to_string(),
146            headers,
147            body: None,
148        })
149    }
150}
151
152/// Rejects a base URL that is not an absolute http or https URL without a
153/// query or fragment: every request path is appended to it.
154fn validate_base_url(base_url: &str) -> Result<(), SDKError> {
155    let parsed = Url::parse(base_url).map_err(|error| {
156        SDKError::NetworkRequestFailed(format!(
157            "the Photon base URL {base_url} is not an absolute URL: {error}"
158        ))
159    })?;
160    let scheme_is_http = parsed.scheme() == "http" || parsed.scheme() == "https";
161    if !scheme_is_http || parsed.host_str().is_none() {
162        return Err(SDKError::NetworkRequestFailed(format!(
163            "the Photon base URL {base_url} must be an absolute http or https URL"
164        )));
165    }
166    if parsed.query().is_some() || parsed.fragment().is_some() {
167        return Err(SDKError::NetworkRequestFailed(format!(
168            "the Photon base URL {base_url} must not carry a query or fragment"
169        )));
170    }
171    Ok(())
172}

174/// Interprets one Photon response: a successful JSON body passes through, a
175/// non-success status becomes a network failure naming the status, and a
176/// successful non-JSON body is a parsing failure.
177fn interpret_json_response(response: HttpResponse) -> Result<serde_json::Value, SDKError> {
178    let successful_status = response.status_code < HTTP_ERROR_STATUS_CODE_THRESHOLD;
179    let body_text = String::from_utf8_lossy(&response.body).into_owned();
180    match serde_json::from_slice::<serde_json::Value>(&response.body) {
181        Ok(payload) if successful_status => Ok(payload),
182        Ok(_) => Err(SDKError::NetworkRequestFailed(format!(
183            "Photon answered with HTTP status {}: {}",
184            response.status_code,
185            error_body_tail(body_text)
186        ))),
187        Err(parse_error) if successful_status => Err(SDKError::ResponseParsingFailure(format!(
188            "the Photon response body is not valid JSON: {parse_error}"
189        ))),
190        Err(_) => Err(SDKError::NetworkRequestFailed(format!(
191            "Photon answered with HTTP status {}: {}",
192            response.status_code,
193            error_body_tail(body_text)
194        ))),
195    }
196}
197
198/// The tail of an error body, capped so a huge response cannot bloat the
199/// diagnostic text.
200fn error_body_tail(body_text: String) -> String {
201    if body_text.len() <= MAXIMUM_ERROR_BODY_TAIL {
202        body_text
203    } else {
204        body_text[body_text.len() - MAXIMUM_ERROR_BODY_TAIL..].to_string()
205    }
206}

219    /// A transport that answers one canned JSON body.
220    struct CannedTransport {
221        body: &'static [u8],
222        status_code: u16,
223    }
224
225    #[async_trait::async_trait]
226    impl HttpTransport for CannedTransport {
227        async fn execute(&self, _request: HttpRequest) -> Result<HttpResponse, TransportError> {
228            Ok(HttpResponse {
229                status_code: self.status_code,
230                headers: Vec::new(),
231                body: self.body.to_vec(),
232            })
233        }
234    }

236    /// An observer that keeps every observation it receives.
237    #[derive(Default)]
238    struct RecordingObserver {
239        observations: Mutex<Vec<ProviderRequestObservation>>,
240    }
241
242    impl ProviderRequestObserver for RecordingObserver {
243        fn on_provider_request(&self, observation: ProviderRequestObservation) {
244            self.observations
245                .lock()
246                .expect("the observation list is never poisoned")
247                .push(observation);
248        }
249    }

251    #[tokio::test]
252    async fn a_request_emits_one_observation_with_a_redacted_credential() {
253        const PROBE_PATH: &str = "api/diagnostics-probe";
254        const PROBE_CREDENTIAL: &str = "SECRET_PROBE_VALUE";
255
256        let recording_observer = Arc::new(RecordingObserver::default());
257        install_provider_request_observer(Some(recording_observer.clone()));
258        let transport: Arc<dyn HttpTransport> = Arc::new(CannedTransport {
259            body: b"{\"features\":[]}",
260            status_code: 200,
261        });
262        let client =
263            PhotonHttpClient::with_transport(PhotonProviderConfiguration::default(), 0, transport)
264                .expect("the default configuration is valid");
265        client
266            .get_json(
267                PROBE_PATH,
268                &[("api_key".to_string(), PROBE_CREDENTIAL.to_string())],
269            )
270            .await
271            .expect("the canned transport answers a JSON payload");
272
273        let observations = recording_observer
274            .observations
275            .lock()
276            .expect("the observation list is never poisoned")
277            .clone();
278        // The observer slot is process-wide, so tests running in parallel are
279        // observed too; the probe path isolates this call.
280        let matching_observations: Vec<&ProviderRequestObservation> = observations
281            .iter()
282            .filter(|observation| observation.request_path.contains(PROBE_PATH))
283            .collect();
284        assert_eq!(
285            matching_observations.len(),
286            1,
287            "one observation per request"
288        );
289        let observation = matching_observations[0];
290        assert_eq!(observation.provider_contract, crate::PLACE_CONTRACT_NAME);
291        assert_eq!(observation.provider_key, PHOTON_PROVIDER_KEY);
292        assert_eq!(observation.http_method, "GET");
293        assert_eq!(observation.status_class, "2xx");
294        assert!(observation.request_path.contains("api_key=<redacted>"));
295        assert!(!observation.request_path.contains(PROBE_CREDENTIAL));
296        install_provider_request_observer(None);
297    }

300    fn a_base_url_with_a_path_is_accepted_but_one_with_a_query_is_refused() {
301        assert!(validate_base_url("https://photon.komoot.io").is_ok());
302        assert!(validate_base_url("https://example.com/geocoder").is_ok());
303        assert!(validate_base_url("https://example.com/?q=1").is_err());
304        assert!(validate_base_url("ftp://example.com").is_err());
