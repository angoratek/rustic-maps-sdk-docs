1//! Provider registry: one optional slot per provider contract.
2//!
3//! The application registers its provider implementations at startup and the
4//! rest of the SDK resolves them by contract. Nothing in this module depends
5//! on a concrete provider.
6
7use geospatial_traits::{
8    DistanceMatrixProvider, ElevationProvider, GeocodingProvider, OfflineRoutingProvider,
9    RasterTileProvider, RoutingProvider, SearchPlaceProvider, TrafficFlowProvider,
10    TrafficIncidentProvider, VectorTileProvider,
11};
12use geospatial_types::SDKError;
13use std::sync::Arc;
14
15/// Registry of provider implementations keyed by contract, not by vendor.
16///
17/// Each contract has one optional slot. Slots start unconfigured. Callers
18/// resolve them with the `require_*` methods, which report
19/// [`SDKError::ProviderNotConfigured`] when the application did not register
20/// that provider.
21///
22/// The registry is cheap to share after configuration: every provider is
23/// stored behind an [`Arc`].
24#[derive(Default)]
25pub struct ProviderRegistry {
26    geocoding_provider: Option<Arc<dyn GeocodingProvider>>,
27    routing_provider: Option<Arc<dyn RoutingProvider>>,
28    distance_matrix_provider: Option<Arc<dyn DistanceMatrixProvider>>,
29    search_place_provider: Option<Arc<dyn SearchPlaceProvider>>,
30    traffic_incident_provider: Option<Arc<dyn TrafficIncidentProvider>>,
31    traffic_flow_provider: Option<Arc<dyn TrafficFlowProvider>>,
32    elevation_provider: Option<Arc<dyn ElevationProvider>>,
33    raster_tile_provider: Option<Arc<dyn RasterTileProvider>>,
34    vector_tile_provider: Option<Arc<dyn VectorTileProvider>>,
35    offline_routing_provider: Option<Arc<dyn OfflineRoutingProvider>>,
36}

71fn not_configured_error(contract_name: &str) -> SDKError {
72    SDKError::ProviderNotConfigured(contract_name.to_string())
73}

212#[cfg(test)]
213mod tests {
214    use super::ProviderRegistry;
215    use geospatial_traits::mock::{
216        MockGeocodingProvider, MockRasterTileProvider, MockVectorTileProvider,
217    };
218    use geospatial_types::SDKError;
219    use std::sync::Arc;
220
221    #[test]
222    fn unregistered_slots_report_provider_not_configured() {
223        let registry = ProviderRegistry::new();
224        let failure = match registry.require_geocoding_provider() {
225            Err(failure) => failure,
226            Ok(_) => panic!("expected ProviderNotConfigured for GeocodingProvider"),
227        };
228        assert_eq!(
229            failure,
230            SDKError::ProviderNotConfigured("GeocodingProvider".to_string())
231        );
232    }
233
234    #[test]
235    fn registered_slots_resolve_through_trait_objects() {
236        let mut registry = ProviderRegistry::new();
237        registry.register_geocoding_provider(Arc::new(MockGeocodingProvider));
238        registry.register_raster_tile_provider(Arc::new(MockRasterTileProvider));
239
240        assert!(registry.require_geocoding_provider().is_ok());
241        assert!(registry.require_raster_tile_provider().is_ok());
242        assert!(registry.require_routing_provider().is_err());
243    }
244
245    #[test]
246    fn vector_tile_slot_resolves_through_trait_objects() {
247        let mut registry = ProviderRegistry::new();
248        registry.register_vector_tile_provider(Arc::new(MockVectorTileProvider));
249        assert!(registry.require_vector_tile_provider().is_ok());
250        let failure = match registry.require_routing_provider() {
251            Err(failure) => failure,
252            Ok(_) => panic!("expected ProviderNotConfigured for RoutingProvider"),
253        };
254        assert_eq!(
255            failure,
256            SDKError::ProviderNotConfigured("RoutingProvider".to_string())
257        );
258    }
