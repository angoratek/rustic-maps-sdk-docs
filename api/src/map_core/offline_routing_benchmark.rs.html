1//! Deterministic synthetic graphs for measuring the offline routing engine.
2//!
3//! The generators write archive bytes with the same deterministic builder the
4//! graph compiler uses, read them back through the same validating reader and
5//! index them with [`LoadedRoutingGraph::new`]. Measurements therefore run
6//! over the production graph structures, on reproducible inputs, with no
7//! network access.
8//!
9//! Cost columns are chosen so both strategies' heuristics stay admissible:
10//! every distance is at least the endpoints' straight-line distance, and every
11//! duration is `distance / speed` at a speed below the packed column's
12//! saturation limit.
13//!
14//! These functions support performance measurement only. Production routing
15//! always goes through [`crate::offline_routing::a_star_search`].
16

30/// Grid graph side lengths the grid benchmark sweeps, as squares of
31/// `side * side` nodes.
32///
33/// The sizes are small enough for a quick run and large enough that the
34/// settled-set and wall-clock growth of Dijkstra against A*'s corridor is
35/// visible at the top size.
36pub const BENCHMARK_GRID_SIDE_LENGTHS: [u32; 3] = [16, 32, 64];
37
38/// Random-spanning graph node counts the random-spanning benchmark sweeps.
39///
40/// The spanning topology, branching corridors plus shortcut arcs instead of a
41/// lattice, gives Dijkstra a second, differently shaped case.
42pub const BENCHMARK_RANDOM_SPANNING_NODE_COUNTS: [usize; 3] = [512, 2048, 8192];
43
44/// Fixed seed of the random-spanning generator.
45///
46/// The generated graphs are part of a recorded baseline, so every run
47/// regenerates byte-identical archives. `BenchmarkRng` is a plain SplitMix64
48/// generator.
49pub const BENCHMARK_RANDOM_SPANNING_SEED: u64 = 0x5EED_00A5_0F7E_BE1A;
50
51/// Node count of the synthetic graph used to time loading: the validating read
52/// and the routing-index build.
53///
54/// The graph carries more nodes and edges than the small reference graph, so
55/// load regressions are visible.
56pub const BENCHMARK_SPANNING_LOAD_BENCH_NODE_COUNT: usize = 2048;

58// ---------------------------------------------------------------------------
59// Synthetic graph geometry and cost constants
60// ---------------------------------------------------------------------------
61
62/// Grid and lattice spacing in degrees: ~55.6 m along a meridian, i.e. an
63/// urban block length the node lattice is placed on.
64const BENCHMARK_BLOCK_DEGREES: f64 = 0.0005;
65
66/// The street classes and speeds of the synthetic non-arterial arcs, as
67/// `(road class, speed km/h, relative weight)`: a realistic street mix where
68/// most travel happens on slow local streets (the shape contrast the
69/// `Fastest` heuristic prunes on). Speeds are whole archive speed cells and
70/// stay below the packed column's saturation limit, keeping the heuristic
71/// admissible.
72const BENCHMARK_ARC_CLASSES: [(RoadClass, u16, u64); 7] = [
73    (RoadClass::Motorway, 130, 2),
74    (RoadClass::Trunk, 110, 3),
75    (RoadClass::Primary, 85, 6),
76    (RoadClass::Secondary, 60, 12),
77    (RoadClass::Tertiary, 50, 12),
78    (RoadClass::Unclassified, 40, 25),
79    (RoadClass::Residential, 30, 40),
80];
81
82/// The arterial overlay of the grid generator: every stride-th street (each
83/// direction the street runs) carries this fast class/speed — the metro's
84/// major-road spine, the corridor shape A*'s heuristic is for.
85const BENCHMARK_ARTERIAL_STRIDE: u32 = 8;
86/// The arterial arc's class/speed pair (see the street-mix table above).
87const BENCHMARK_ARTERIAL_CLASS: (RoadClass, u16) = (RoadClass::Trunk, 110);
88
89/// The fast motorway-override arc of the benchmark generators (the sparse
90/// fast overlay of the spanning family): at the packed speed column's
91/// saturation limit, so the `Fastest` heuristic is exactly tight on these
92/// arcs.
93const BENCHMARK_MOTORWAY_ARC: (RoadClass, u16) = (RoadClass::Motorway, 155);
94
95/// How many of the spanning family's shortcut arcs per node are the fast
96/// motorway override, in whole percent of `node_count` (the sparse fast
97/// overlay; the rest of the shortcut arcs draw the street mix).
98const BENCHMARK_MOTORWAY_SHORTCUT_PERCENT: u64 = 6;
99
100/// Width (in lattice-neighbor steps) of the random-spanning generator's
101/// edge window: a node's spanning edge reaches a random node among the few
102/// nearest earlier cells, so edges stay geometrically local like a road
103/// network's do.
104const BENCHMARK_SPANNING_PARENT_WINDOW: usize = 8;
105
106/// One-way shortcut arcs per node added on top of the random spanning edges
107/// (the occasional long detour arc that breaks the pure corridor).
108const BENCHMARK_SHORTCUT_ARCS_PER_NODE: usize = 2;
109
110/// Sideways jitter of spanning-lattice node positions, as a fraction of the
111/// block spacing (keeps node coordinates distinguishable without overlap).
112const BENCHMARK_SPANNING_JITTER_FRACTION: f64 = 0.4;
113
114/// Jitter resolution: the generator's jitter draw is `next_below` of this,
115/// centered to a signed fraction of [`BENCHMARK_SPANNING_JITTER_FRACTION`].
116const BENCHMARK_SPANNING_JITTER_STEPS: u64 = 1_001;
117
118/// Whole-meter inflation the generator may add over an arc's straight-line
119/// distance: the distance column's own granularity, so the draw is fully
120/// visible in the archive while the column never drops below the
121/// admissible straight-line bound. The range is deliberately wide (winding
122/// streets travel far more than their straight-line length) so the recorded
123/// graphs' optimal paths are *unique* per endpoint pair — the premise of
124/// the A*/reference chain-equality assertion (an equal-total alternative
125/// optimum would let the two engines' settlement orders diverge on a tie).
126/// The recorded (seed, size) set is verified tie-free by the equality test
127/// itself; growing either means re-verifying.
128const BENCHMARK_DISTANCE_INFLATION_METERS: u64 = 128;
129
130/// Unit conversions for the duration column (`distance / speed`), matching
131/// `offline_routing`'s columns (milliseconds, seconds, km/h).
132const BENCHMARK_MILLISECONDS_PER_SECOND: f64 = 1_000.0;
133/// Unit conversion: seconds per hour.
134const BENCHMARK_SECONDS_PER_HOUR: f64 = 3_600.0;
135/// Unit conversion: meters per kilometer.
136const BENCHMARK_METERS_PER_KILOMETER: f64 = 1_000.0;

138// ---------------------------------------------------------------------------
139// Deterministic PRNG (SplitMix64, no external dependency)
140// ---------------------------------------------------------------------------
141
142/// SplitMix64's golden-gamma increment; the standard `0x9E3779B97F4A7C15`.
143const SPLITMIX64_GOLDEN_GAMMA: u64 = 0x9E37_79B9_7F4A_7C15;
144
145/// A plain SplitMix64 PRNG: the synthetic benchmark graphs must be
146/// byte-identical on every machine and every run, so the generator is
147/// seeded in-source instead of pulling in a random-number dependency.
148#[derive(Debug)]
149struct BenchmarkRng {
150    state: u64,
151}
152
153impl BenchmarkRng {
154    /// A PRNG seeded from the (fixed) generator seed.
155    const fn new(seed: u64) -> Self {
156        Self { state: seed }
157    }
158
159    /// The next SplitMix64 output.
160    fn next_u64(&mut self) -> u64 {
161        self.state = self.state.wrapping_add(SPLITMIX64_GOLDEN_GAMMA);
162        let mut mixed = self.state;
163        mixed = (mixed ^ (mixed >> 30)).wrapping_mul(0xBF58_476D_1CE4_E5B9);
164        mixed = (mixed ^ (mixed >> 27)).wrapping_mul(0x94D0_49BB_1331_11EB);
165        mixed ^ (mixed >> 31)
166    }
167
168    /// The next output reduced below `bound` (the tiny modulo bias of a
169    /// benchmark generator is irrelevant against a fixed recorded graph).
170    fn next_below(&mut self, bound: u64) -> u64 {
171        self.next_u64() % bound
172    }
173}

209/// The canonical query pair of a `node_count`-node benchmark graph: the two
210/// opposing index corners.
211///
212/// This is the worst case for a settled-set comparison: Dijkstra settles
213/// nearly the whole graph, while the haversine corridor does not.

219/// Loads a benchmark graph exactly the way the routing provider loads a
220/// compiled archive: validating in-memory reader, then the routing index.
221fn loaded_graph_from_archive_bytes(archive_bytes: &[u8]) -> Result<LoadedRoutingGraph, SDKError> {

230/// Writes the grid graph's deterministic archive bytes.
231///
232/// # Errors
233/// Returns [`SDKError::OfflineRoutingFailure`] only on a generator bug; indices
234/// are in range by construction.
235pub fn benchmark_grid_archive_bytes(side_length: u32) -> Result<Vec<u8>, SDKError> {
236    let side_as_usize = side_length as usize;
237    let node_count = side_as_usize * side_as_usize;
238    // The seed folds the shape in, so every size is a fixed recorded graph.
239    let mut benchmark_rng =
240        BenchmarkRng::new(BENCHMARK_RANDOM_SPANNING_SEED ^ u64::from(side_length));
241
242    let mut node_records = Vec::with_capacity(node_count);
243    for row in 0..side_as_usize {
244        for column in 0..side_as_usize {
245            node_records.push(NodeRecord::from_geographic_degrees(
246                f64::from(row as u32) * BENCHMARK_BLOCK_DEGREES,
247                f64::from(column as u32) * BENCHMARK_BLOCK_DEGREES,
248            ));
249        }
250    }
251
252    let mut edge_records = Vec::new();
253    for row in 0..side_as_usize {
254        for column in 0..side_as_usize {
255            let node_index = row * side_as_usize + column;
256            if column + 1 < side_as_usize {
257                // An eastbound block: arterial when its street row is one
258                // of the stride-spaced artery rows.
259                let forced_class_and_speed = ((row as u32)
260                    .is_multiple_of(BENCHMARK_ARTERIAL_STRIDE))
261                .then_some(BENCHMARK_ARTERIAL_CLASS);
262                push_two_way_arc(
263                    &mut edge_records,
264                    node_index,
265                    node_index + 1,
266                    &node_records,
267                    &mut benchmark_rng,
268                    forced_class_and_speed,
269                );
270            }
271            if row + 1 < side_as_usize {
272                // A southbound block: arterial when its street column is
273                // one of the stride-spaced artery columns.
274                let forced_class_and_speed = ((column as u32)
275                    .is_multiple_of(BENCHMARK_ARTERIAL_STRIDE))
276                .then_some(BENCHMARK_ARTERIAL_CLASS);
277                push_two_way_arc(
278                    &mut edge_records,
279                    node_index,
280                    node_index + side_as_usize,
281                    &node_records,
282                    &mut benchmark_rng,
283                    forced_class_and_speed,
284                );
285            }
286        }
287    }
288
289    encoded_archive_bytes(node_records, edge_records)
290}

292/// Writes the random-spanning graph's deterministic archive bytes.
293///
294/// # Errors
295/// Returns [`SDKError::OfflineRoutingFailure`] only on a generator bug; indices
296/// are in range by construction.
297pub fn benchmark_random_spanning_archive_bytes(
298    node_count: usize,
299    seed: u64,
300) -> Result<Vec<u8>, SDKError> {
301    let mut benchmark_rng = BenchmarkRng::new(seed);
302    let lattice_side = f64::from(node_count as u32).sqrt().ceil() as usize;
303
304    // Jittered lattice positions: cell placement with a per-node sideways
305    // offset, so straight-line distances vary but stay ordered.
306    let mut node_records = Vec::with_capacity(node_count);
307    for node_index in 0..node_count {
308        let (row, column) = (node_index / lattice_side, node_index % lattice_side);
309        node_records.push(NodeRecord::from_geographic_degrees(
310            f64::from(row as u32) * BENCHMARK_BLOCK_DEGREES + jitter_degrees(&mut benchmark_rng),
311            f64::from(column as u32) * BENCHMARK_BLOCK_DEGREES + jitter_degrees(&mut benchmark_rng),
312        ));
313    }
314
315    let mut edge_records = Vec::new();
316    // The spanning edges, both ways: directed reachability of the whole
317    // graph, on geometrically local random edges (branching corridors
318    // rather than a lattice).
319    for node_index in 1..node_count {
320        let parent_window = BENCHMARK_SPANNING_PARENT_WINDOW.min(node_index);
321        let parent_index = node_index - 1 - benchmark_rng.next_below(parent_window as u64) as usize;
322        push_two_way_arc(
323            &mut edge_records,
324            parent_index,
325            node_index,
326            &node_records,
327            &mut benchmark_rng,
328            None,
329        );
330    }
331    // One-way shortcut arcs detouring across the lattice: mostly the street
332    // mix, plus the sparse fast motorway overlay (the metro's few fast
333    // corridors across a slow local network — the cost anisotropy A*'s
334    // heuristic prunes on).
335    let shortcut_arc_count = node_count * BENCHMARK_SHORTCUT_ARCS_PER_NODE;
336    let motorway_arc_count = node_count * (BENCHMARK_MOTORWAY_SHORTCUT_PERCENT as usize) / 100;
337    for arc_position in 0..shortcut_arc_count {
338        let from_index = benchmark_rng.next_below(node_count as u64) as usize;
339        let to_index = benchmark_rng.next_below(node_count as u64) as usize;
340        if from_index == to_index {
341            continue;
342        }
343        let motorway_arc = arc_position < motorway_arc_count;
344        push_single_arc(
345            &mut edge_records,
346            from_index,
347            to_index,
348            &node_records,
349            &mut benchmark_rng,
350            motorway_arc.then_some(BENCHMARK_MOTORWAY_ARC),
351        );
352    }
353
354    encoded_archive_bytes(node_records, edge_records)
355}

357/// The sideways jitter of a spanning-lattice node, in signed degrees.
358fn jitter_degrees(benchmark_rng: &mut BenchmarkRng) -> f64 {
359    let centered_steps = benchmark_rng.next_below(BENCHMARK_SPANNING_JITTER_STEPS) as f64
360        - f64::from(BENCHMARK_SPANNING_JITTER_STEPS as u32) / 2.0;
361    centered_steps / f64::from(BENCHMARK_SPANNING_JITTER_STEPS as u32)
362        * BENCHMARK_SPANNING_JITTER_FRACTION
363        * BENCHMARK_BLOCK_DEGREES
364}

366/// Appends both directions of one two-way arc as their own edge records
367/// (the archive's stored-pair contract: every traversable direction is its
368/// own `(source, target)` record).
369fn push_two_way_arc(
370    edge_records: &mut Vec<EdgeRecord>,
371    first_node_index: usize,
372    second_node_index: usize,
373    node_records: &[NodeRecord],
374    benchmark_rng: &mut BenchmarkRng,
375    forced_class_and_speed: Option<(RoadClass, u16)>,
376) {
377    push_single_arc(
378        edge_records,
379        first_node_index,
380        second_node_index,
381        node_records,
382        benchmark_rng,
383        forced_class_and_speed,
384    );
385    push_single_arc(
386        edge_records,
387        second_node_index,
388        first_node_index,
389        node_records,
390        benchmark_rng,
391        forced_class_and_speed,
392    );
393}

395/// Appends one directed arc record with per-arc varied, column-honest costs.
396fn push_single_arc(
397    edge_records: &mut Vec<EdgeRecord>,
398    source_node_index: usize,
399    target_node_index: usize,
400    node_records: &[NodeRecord],
401    benchmark_rng: &mut BenchmarkRng,
402    forced_class_and_speed: Option<(RoadClass, u16)>,
403) {
404    let straight_line_meters = straight_line_meters(
405        &node_records[source_node_index],
406        &node_records[target_node_index],
407    );
408    let (road_class, speed_kilometers_per_hour) =
409        forced_class_and_speed.unwrap_or_else(|| benchmark_street_class_and_speed(benchmark_rng));
410    let (distance_meters, duration_milliseconds) = benchmark_cost_columns(
411        straight_line_meters,
412        speed_kilometers_per_hour,
413        benchmark_rng,
414    );
415    edge_records.push(EdgeRecord::new_directed(
416        source_node_index as u32,
417        target_node_index as u32,
418        distance_meters,
419        duration_milliseconds,
420        EdgeProperties {
421            road_class,
422            speed_kilometers_per_hour,
423            forward_allowed: true,
424            backward_allowed: false,
425        },
426    ));
427}

429/// The class/speed pair of a non-arterial synthetic arc, drawn by the
430/// street-mix table's relative weights (arbitrary but fixed per arc; the
431/// class column feeds maneuver runs, not the search costs).
432fn benchmark_street_class_and_speed(benchmark_rng: &mut BenchmarkRng) -> (RoadClass, u16) {
433    let mut class_draw = benchmark_rng.next_below(100);
434    for &(road_class, speed_kilometers_per_hour, weight) in &BENCHMARK_ARC_CLASSES {
435        if class_draw < weight {
436            return (road_class, speed_kilometers_per_hour);
437        }
438        class_draw -= weight;
439    }
440    let (road_class, speed_kilometers_per_hour, _) =
441        BENCHMARK_ARC_CLASSES[BENCHMARK_ARC_CLASSES.len() - 1];
442    (road_class, speed_kilometers_per_hour)
443}

445/// The cost columns of one synthetic arc: distance inflated over the
446/// straight line by a whole-percentage draw (the straight-line heuristic
447/// stays admissible for the `Shortest` strategy), duration = distance over
448/// speed (below the packed column's saturation limit, so the `Fastest`
449/// heuristic stays admissible too).
450fn benchmark_cost_columns(
451    straight_line_meters: f64,
452    speed_kilometers_per_hour: u16,
453    benchmark_rng: &mut BenchmarkRng,
454) -> (u32, u32) {
455    let variation_meters = benchmark_rng.next_below(BENCHMARK_DISTANCE_INFLATION_METERS) as u32;
456    let distance_meters = straight_line_meters.ceil() as u32 + variation_meters;
457    let speed_meters_per_second = f64::from(speed_kilometers_per_hour)
458        * BENCHMARK_METERS_PER_KILOMETER
459        / BENCHMARK_SECONDS_PER_HOUR;
460    let duration_milliseconds = (f64::from(distance_meters) / speed_meters_per_second
461        * BENCHMARK_MILLISECONDS_PER_SECOND)
462        .round() as u32;
463    (distance_meters, duration_milliseconds)
464}

466/// Straight-line (great-circle) distance between two nodes' quantized
467/// coordinates: the admissibility floor the distance column inflates over.
468fn straight_line_meters(from_node: &NodeRecord, to_node: &NodeRecord) -> f64 {
469    geographic_coordinate_of(from_node)
470        .haversine_distance_meters(&geographic_coordinate_of(to_node))
471}
472
473/// A quantized node record as a `GeographicCoordinate` (always in range).
474fn geographic_coordinate_of(node_record: &NodeRecord) -> GeographicCoordinate {
475    GeographicCoordinate::from_degrees(
476        node_record.latitude_degrees(),
477        node_record.longitude_degrees(),
478    )
479    .expect("benchmark node coordinates are within the canonical ranges")
480}

482/// Encodes a generated archive through the same deterministic builder the
483/// graph compiler uses, with the empty auxiliary resources a routing-only
484/// archive carries.
485fn encoded_archive_bytes(
486    node_records: Vec<NodeRecord>,
487    edge_records: Vec<EdgeRecord>,
488) -> Result<Vec<u8>, SDKError> {
489    OfflineRoutingNetworkGraphBuilder::new()
490        .with_nodes(node_records)
491        .with_edges(edge_records)
492        .with_lane_guidances(Vec::new())
493        .with_instruction_strings(Vec::new())
494        .with_instruction_lookup_indices(vec![INSTRUCTION_ABSENT_INDEX; 0])
495        .with_turn_restrictions(Vec::new())
496        .encode_deterministic()
497}

553    #[test]
554    fn astar_and_dijkstra_reference_agree_on_the_synthetic_benchmark_graphs() {
555        // The result-equality half of the benchmark claim: over the same
556        // loaded graphs, A* and the Dijkstra reference return the same
557        // optimal chain at the same total cost (per-edge costs vary, so the
558        // optimum is unique), and A* settles no more nodes than Dijkstra —
559        // the measured-better-or-equal expansion sanity of the "A* beats
560        // Dijkstra" claim (all deterministic, noise-free).
561        let mut benchmark_graphs = Vec::new();
562        for side_length in [TEST_GRID_SIDE_LENGTH, TEST_THIRD_GRID_SIDE_LENGTH] {
563            benchmark_graphs.push((
564                format!("grid_{side_length}"),
565                benchmark_grid_loaded_graph(side_length).expect("the test grid graph loads"),
566                side_length * side_length,
567            ));
568        }
569        for node_count in [TEST_SPANNING_NODE_COUNT, TEST_SECOND_SPANNING_NODE_COUNT] {

576        }

619    /// The coarse CI budget alarm: the asserted generous upper bound on the
620    /// A* solve of the fixed benchmark graphs.
621    ///
622    /// Shared-runner wall clock is noisy, so this is deliberately NOT a
623    /// sharp gate: the bound sits three orders of magnitude above the
624    /// locally measured (sub-millisecond) solves and exists to catch
625    /// *gross* regressions (a lost heuristic, an O(n²) relaxation) — the
626    /// criterion numbers uploaded by the benchmark workflow are the
627    /// recorded trend baseline, not a CI pass/fail. If this bound ever
628    /// fires while nothing structural changed, it is runner noise; re-run
629    /// before bisecting.
630    const COARSE_BUDGET_ALARM_MILLISECONDS: u128 = 1_000;
631
632    #[test]
633    fn astar_stays_within_the_coarse_solve_budget_alarm_on_the_fixed_graphs() {
634        let grid_graph = benchmark_grid_loaded_graph(64).expect("the fixed grid graph loads");
635        let spanning_graph =
636            benchmark_random_spanning_loaded_graph(8_192, BENCHMARK_RANDOM_SPANNING_SEED)
637                .expect("the fixed spanning graph loads");
638        let fixed_queries = [
639            (
640                "grid_64x64",
641                &grid_graph,
642                benchmark_corner_query_pair(64 * 64),
643            ),
644            (
645                "spanning_8192",
646                &spanning_graph,
647                benchmark_corner_query_pair(8_192),
648            ),
649        ];
650
651        for (graph_name, loaded_graph, (start_node_index, end_node_index)) in fixed_queries {
652            let solve_started_at = std::time::Instant::now();
653            let astar_route = a_star_search(
654                loaded_graph,
655                start_node_index,
656                end_node_index,
657                TraversalCost::TimeSeconds,
658            )
659            .unwrap_or_else(|| panic!("{graph_name} connects the fixed query pair"));
660            let solve_elapsed_milliseconds = solve_started_at.elapsed().as_millis();
661            assert!(
662                solve_elapsed_milliseconds < COARSE_BUDGET_ALARM_MILLISECONDS,
663                "{graph_name}: the A* solve took {solve_elapsed_milliseconds} ms, above the \
664                 {} ms coarse budget alarm (a gross-regression signal — re-run before \
665                 bisecting, shared-runner wall clock is noisy)",
666                COARSE_BUDGET_ALARM_MILLISECONDS
667            );
668            // The alarm graph is also sanity-sized: the solve is fully
669            // answered (non-empty chain ending at the requested terminal).
670            assert_eq!(
671                astar_route
672                    .node_chain_indices
673                    .last()
674                    .expect("non-empty chain"),
675                &end_node_index
676            );
677        }
678    }
