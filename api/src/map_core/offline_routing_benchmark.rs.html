1//! Benchmark support for the offline routing engine — the M7 slice 6 gate
2//! (`PLAN.md` M7 item 3; `PRD.md` §4.6's "Dijkstra upgraded to A*" claim,
3//! measured in [`crate::offline_routing`]).
4//!
5//! **Not SDK surface**: this module exists for the test suite and the
6//! criterion benches of this crate (`benches/offline_search_benchmark.rs`,
7//! run natively only, wasm is gated lib-only); production routing never
8//! calls anything here, and routing always goes through
9//! [`crate::offline_routing::a_star_search`].
10//!
11//! Synthetic graphs are built the same way production loads build the
12//! router state: archive **bytes** are written with the same deterministic
13//! builder, read back through the same validating reader, and indexed by
14//! the same [`LoadedRoutingGraph::new`] — so the benchmark measures the
15//! real loaded-graph structures, on deterministic seeds, with no network.
16//!
17//! Costs are column-honest so both strategies' heuristics stay admissible:
18//! the distance column is always at least the endpoints' straight-line
19//! distance, and the duration column is always `distance / speed` at a
20//! speed below the packed column's saturation limit.
21

35/// Grid graph side lengths the criterion suite sweeps (squares of
36/// `side * side` nodes). Small enough for a fast job, large enough that the
37/// settled-set and wall-clock growth of Dijkstra against A*'s corridor is
38/// visible at the top size.
39pub const BENCHMARK_GRID_SIDE_LENGTHS: [u32; 3] = [16, 32, 64];

41/// Random-spanning graph node counts the criterion suite sweeps. The
42/// spanning topology (branching corridors plus shortcut arcs instead of a
43/// lattice) is the second, differently-shaped case for Dijkstra.
44pub const BENCHMARK_RANDOM_SPANNING_NODE_COUNTS: [usize; 3] = [512, 2048, 8192];
45
46/// Fixed seed of the random-spanning generator: the benchmark graphs are
47/// part of the recorded baseline, so the suite regenerates byte-identical
48/// archives every run (`BenchmarkRng` is a plain SplitMix64).
49pub const BENCHMARK_RANDOM_SPANNING_SEED: u64 = 0x5EED_00A5_0F7E_BE1A;

51/// Node count of the synthetic graph the load benches time the validating
52/// read and the routing-index build over (a larger column count than the

56// ---------------------------------------------------------------------------
57// Synthetic graph geometry and cost constants
58// ---------------------------------------------------------------------------
59
60/// Grid and lattice spacing in degrees: ~55.6 m along a meridian, i.e. an
61/// urban block length the node lattice is placed on.
62const BENCHMARK_BLOCK_DEGREES: f64 = 0.0005;
63
64/// The street classes and speeds of the synthetic non-arterial arcs, as
65/// `(road class, speed km/h, relative weight)`: a realistic street mix where
66/// most travel happens on slow local streets (the shape contrast the
67/// `Fastest` heuristic prunes on). Speeds are whole archive speed cells and
68/// stay below the packed column's saturation limit, keeping the heuristic
69/// admissible.
70const BENCHMARK_ARC_CLASSES: [(RoadClass, u16, u64); 7] = [
71    (RoadClass::Motorway, 130, 2),
72    (RoadClass::Trunk, 110, 3),
73    (RoadClass::Primary, 85, 6),
74    (RoadClass::Secondary, 60, 12),
75    (RoadClass::Tertiary, 50, 12),
76    (RoadClass::Unclassified, 40, 25),
77    (RoadClass::Residential, 30, 40),
78];
79
80/// The arterial overlay of the grid generator: every stride-th street (each
81/// direction the street runs) carries this fast class/speed — the metro's
82/// major-road spine, the corridor shape A*'s heuristic is for.
83const BENCHMARK_ARTERIAL_STRIDE: u32 = 8;
84/// The arterial arc's class/speed pair (see the street-mix table above).
85const BENCHMARK_ARTERIAL_CLASS: (RoadClass, u16) = (RoadClass::Trunk, 110);
86
87/// The fast motorway-override arc of the benchmark generators (the sparse
88/// fast overlay of the spanning family): at the packed speed column's
89/// saturation limit, so the `Fastest` heuristic is exactly tight on these
90/// arcs.
91const BENCHMARK_MOTORWAY_ARC: (RoadClass, u16) = (RoadClass::Motorway, 155);
92
93/// How many of the spanning family's shortcut arcs per node are the fast
94/// motorway override, in whole percent of `node_count` (the sparse fast
95/// overlay; the rest of the shortcut arcs draw the street mix).
96const BENCHMARK_MOTORWAY_SHORTCUT_PERCENT: u64 = 6;
97
98/// Width (in lattice-neighbor steps) of the random-spanning generator's
99/// edge window: a node's spanning edge reaches a random node among the few
100/// nearest earlier cells, so edges stay geometrically local like a road
101/// network's do.
102const BENCHMARK_SPANNING_PARENT_WINDOW: usize = 8;
103
104/// One-way shortcut arcs per node added on top of the random spanning edges
105/// (the occasional long detour arc that breaks the pure corridor).
106const BENCHMARK_SHORTCUT_ARCS_PER_NODE: usize = 2;
107
108/// Sideways jitter of spanning-lattice node positions, as a fraction of the
109/// block spacing (keeps node coordinates distinguishable without overlap).
110const BENCHMARK_SPANNING_JITTER_FRACTION: f64 = 0.4;
111
112/// Jitter resolution: the generator's jitter draw is `next_below` of this,
113/// centered to a signed fraction of [`BENCHMARK_SPANNING_JITTER_FRACTION`].
114const BENCHMARK_SPANNING_JITTER_STEPS: u64 = 1_001;
115
116/// Whole-meter inflation the generator may add over an arc's straight-line
117/// distance: the distance column's own granularity, so the draw is fully
118/// visible in the archive while the column never drops below the
119/// admissible straight-line bound. The range is deliberately wide (winding
120/// streets travel far more than their straight-line length) so the recorded
121/// graphs' optimal paths are *unique* per endpoint pair — the premise of
122/// the A*/reference chain-equality assertion (an equal-total alternative
123/// optimum would let the two engines' settlement orders diverge on a tie).
124/// The recorded (seed, size) set is verified tie-free by the equality test
125/// itself; growing either means re-verifying.
126const BENCHMARK_DISTANCE_INFLATION_METERS: u64 = 128;
127
128/// Unit conversions for the duration column (`distance / speed`), matching
129/// `offline_routing`'s columns (milliseconds, seconds, km/h).
130const BENCHMARK_MILLISECONDS_PER_SECOND: f64 = 1_000.0;
131/// Unit conversion: seconds per hour.
132const BENCHMARK_SECONDS_PER_HOUR: f64 = 3_600.0;
133/// Unit conversion: meters per kilometer.
134const BENCHMARK_METERS_PER_KILOMETER: f64 = 1_000.0;
135
136// ---------------------------------------------------------------------------
137// Deterministic PRNG (SplitMix64, no external dependency)
138// ---------------------------------------------------------------------------
139
140/// SplitMix64's golden-gamma increment; the standard `0x9E3779B97F4A7C15`.
141const SPLITMIX64_GOLDEN_GAMMA: u64 = 0x9E37_79B9_7F4A_7C15;
142
143/// A plain SplitMix64 PRNG: the synthetic benchmark graphs must be
144/// byte-identical on every machine and every run, so the generator is
145/// seeded in-source instead of pulling in a random-number dependency.
146#[derive(Debug)]
147struct BenchmarkRng {
148    state: u64,
149}
150
151impl BenchmarkRng {
152    /// A PRNG seeded from the (fixed) generator seed.
153    const fn new(seed: u64) -> Self {
154        Self { state: seed }
155    }
156
157    /// The next SplitMix64 output.
158    fn next_u64(&mut self) -> u64 {
159        self.state = self.state.wrapping_add(SPLITMIX64_GOLDEN_GAMMA);
160        let mut mixed = self.state;
161        mixed = (mixed ^ (mixed >> 30)).wrapping_mul(0xBF58_476D_1CE4_E5B9);
162        mixed = (mixed ^ (mixed >> 27)).wrapping_mul(0x94D0_49BB_1331_11EB);
163        mixed ^ (mixed >> 31)
164    }
165
166    /// The next output reduced below `bound` (the tiny modulo bias of a
167    /// benchmark generator is irrelevant against a fixed recorded graph).
168    fn next_below(&mut self, bound: u64) -> u64 {
169        self.next_u64() % bound
170    }
171}

202/// The canonical query pair of an `node_count`-node benchmark graph: the

211/// Loads a benchmark graph exactly the way the routing provider loads a
212/// compiled archive: validating in-memory reader, then the routing index.
213fn loaded_graph_from_archive_bytes(archive_bytes: &[u8]) -> Result<LoadedRoutingGraph, SDKError> {

222/// Writes the grid graph's deterministic archive bytes.
223///
224/// # Errors
225/// Only on a synthetic-generator bug (indices are in-range by construction).
226pub fn benchmark_grid_archive_bytes(side_length: u32) -> Result<Vec<u8>, SDKError> {
227    let side_as_usize = side_length as usize;
228    let node_count = side_as_usize * side_as_usize;
229    // The seed folds the shape in, so every size is a fixed recorded graph.
230    let mut benchmark_rng =
231        BenchmarkRng::new(BENCHMARK_RANDOM_SPANNING_SEED ^ u64::from(side_length));
232
233    let mut node_records = Vec::with_capacity(node_count);
234    for row in 0..side_as_usize {
235        for column in 0..side_as_usize {
236            node_records.push(NodeRecord::from_geographic_degrees(
237                f64::from(row as u32) * BENCHMARK_BLOCK_DEGREES,
238                f64::from(column as u32) * BENCHMARK_BLOCK_DEGREES,
239            ));
240        }
241    }
242
243    let mut edge_records = Vec::new();
244    for row in 0..side_as_usize {
245        for column in 0..side_as_usize {
246            let node_index = row * side_as_usize + column;
247            if column + 1 < side_as_usize {
248                // An eastbound block: arterial when its street row is one
249                // of the stride-spaced artery rows.
250                let forced_class_and_speed = ((row as u32)
251                    .is_multiple_of(BENCHMARK_ARTERIAL_STRIDE))
252                .then_some(BENCHMARK_ARTERIAL_CLASS);
253                push_two_way_arc(
254                    &mut edge_records,
255                    node_index,
256                    node_index + 1,
257                    &node_records,
258                    &mut benchmark_rng,
259                    forced_class_and_speed,
260                );
261            }
262            if row + 1 < side_as_usize {
263                // A southbound block: arterial when its street column is
264                // one of the stride-spaced artery columns.
265                let forced_class_and_speed = ((column as u32)
266                    .is_multiple_of(BENCHMARK_ARTERIAL_STRIDE))
267                .then_some(BENCHMARK_ARTERIAL_CLASS);
268                push_two_way_arc(
269                    &mut edge_records,
270                    node_index,
271                    node_index + side_as_usize,
272                    &node_records,
273                    &mut benchmark_rng,
274                    forced_class_and_speed,
275                );
276            }
277        }
278    }
279
280    encoded_archive_bytes(node_records, edge_records)
281}

283/// Writes the random-spanning graph's deterministic archive bytes.
284///
285/// # Errors
286/// Only on a synthetic-generator bug (indices are in-range by construction).
287pub fn benchmark_random_spanning_archive_bytes(
288    node_count: usize,
289    seed: u64,
290) -> Result<Vec<u8>, SDKError> {
291    let mut benchmark_rng = BenchmarkRng::new(seed);
292    let lattice_side = f64::from(node_count as u32).sqrt().ceil() as usize;
293
294    // Jittered lattice positions: cell placement with a per-node sideways
295    // offset, so straight-line distances vary but stay ordered.
296    let mut node_records = Vec::with_capacity(node_count);
297    for node_index in 0..node_count {
298        let (row, column) = (node_index / lattice_side, node_index % lattice_side);
299        node_records.push(NodeRecord::from_geographic_degrees(
300            f64::from(row as u32) * BENCHMARK_BLOCK_DEGREES + jitter_degrees(&mut benchmark_rng),
301            f64::from(column as u32) * BENCHMARK_BLOCK_DEGREES + jitter_degrees(&mut benchmark_rng),
302        ));
303    }
304
305    let mut edge_records = Vec::new();
306    // The spanning edges, both ways: directed reachability of the whole
307    // graph, on geometrically local random edges (branching corridors
308    // rather than a lattice).
309    for node_index in 1..node_count {
310        let parent_window = BENCHMARK_SPANNING_PARENT_WINDOW.min(node_index);
311        let parent_index = node_index - 1 - benchmark_rng.next_below(parent_window as u64) as usize;
312        push_two_way_arc(
313            &mut edge_records,
314            parent_index,
315            node_index,
316            &node_records,
317            &mut benchmark_rng,
318            None,
319        );
320    }
321    // One-way shortcut arcs detouring across the lattice: mostly the street
322    // mix, plus the sparse fast motorway overlay (the metro's few fast
323    // corridors across a slow local network — the cost anisotropy A*'s
324    // heuristic prunes on).
325    let shortcut_arc_count = node_count * BENCHMARK_SHORTCUT_ARCS_PER_NODE;
326    let motorway_arc_count = node_count * (BENCHMARK_MOTORWAY_SHORTCUT_PERCENT as usize) / 100;
327    for arc_position in 0..shortcut_arc_count {
328        let from_index = benchmark_rng.next_below(node_count as u64) as usize;
329        let to_index = benchmark_rng.next_below(node_count as u64) as usize;
330        if from_index == to_index {
331            continue;
332        }
333        let motorway_arc = arc_position < motorway_arc_count;
334        push_single_arc(
335            &mut edge_records,
336            from_index,
337            to_index,
338            &node_records,
339            &mut benchmark_rng,
340            motorway_arc.then_some(BENCHMARK_MOTORWAY_ARC),
341        );
342    }
343
344    encoded_archive_bytes(node_records, edge_records)
345}

347/// The sideways jitter of a spanning-lattice node, in signed degrees.
348fn jitter_degrees(benchmark_rng: &mut BenchmarkRng) -> f64 {
349    let centered_steps = benchmark_rng.next_below(BENCHMARK_SPANNING_JITTER_STEPS) as f64
350        - f64::from(BENCHMARK_SPANNING_JITTER_STEPS as u32) / 2.0;
351    centered_steps / f64::from(BENCHMARK_SPANNING_JITTER_STEPS as u32)
352        * BENCHMARK_SPANNING_JITTER_FRACTION
353        * BENCHMARK_BLOCK_DEGREES
354}

356/// Appends both directions of one two-way arc as their own edge records
357/// (the archive's stored-pair contract: every traversable direction is its
358/// own `(source, target)` record).
359fn push_two_way_arc(
360    edge_records: &mut Vec<EdgeRecord>,
361    first_node_index: usize,
362    second_node_index: usize,
363    node_records: &[NodeRecord],
364    benchmark_rng: &mut BenchmarkRng,
365    forced_class_and_speed: Option<(RoadClass, u16)>,
366) {
367    push_single_arc(
368        edge_records,
369        first_node_index,
370        second_node_index,
371        node_records,
372        benchmark_rng,
373        forced_class_and_speed,
374    );
375    push_single_arc(
376        edge_records,
377        second_node_index,
378        first_node_index,
379        node_records,
380        benchmark_rng,
381        forced_class_and_speed,
382    );
383}

385/// Appends one directed arc record with per-arc varied, column-honest costs.
386fn push_single_arc(
387    edge_records: &mut Vec<EdgeRecord>,
388    source_node_index: usize,
389    target_node_index: usize,
390    node_records: &[NodeRecord],
391    benchmark_rng: &mut BenchmarkRng,
392    forced_class_and_speed: Option<(RoadClass, u16)>,
393) {
394    let straight_line_meters = straight_line_meters(
395        &node_records[source_node_index],
396        &node_records[target_node_index],
397    );
398    let (road_class, speed_kilometers_per_hour) =
399        forced_class_and_speed.unwrap_or_else(|| benchmark_street_class_and_speed(benchmark_rng));
400    let (distance_meters, duration_milliseconds) = benchmark_cost_columns(
401        straight_line_meters,
402        speed_kilometers_per_hour,
403        benchmark_rng,
404    );
405    edge_records.push(EdgeRecord::new_directed(
406        source_node_index as u32,
407        target_node_index as u32,
408        distance_meters,
409        duration_milliseconds,
410        EdgeProperties {
411            road_class,
412            speed_kilometers_per_hour,
413            forward_allowed: true,
414            backward_allowed: false,
415        },
416    ));
417}

419/// The class/speed pair of a non-arterial synthetic arc, drawn by the
420/// street-mix table's relative weights (arbitrary but fixed per arc; the
421/// class column feeds maneuver runs, not the search costs).
422fn benchmark_street_class_and_speed(benchmark_rng: &mut BenchmarkRng) -> (RoadClass, u16) {
423    let mut class_draw = benchmark_rng.next_below(100);
424    for &(road_class, speed_kilometers_per_hour, weight) in &BENCHMARK_ARC_CLASSES {
425        if class_draw < weight {
426            return (road_class, speed_kilometers_per_hour);
427        }
428        class_draw -= weight;
429    }
430    let (road_class, speed_kilometers_per_hour, _) =
431        BENCHMARK_ARC_CLASSES[BENCHMARK_ARC_CLASSES.len() - 1];
432    (road_class, speed_kilometers_per_hour)
433}

435/// The cost columns of one synthetic arc: distance inflated over the
436/// straight line by a whole-percentage draw (the straight-line heuristic
437/// stays admissible for the `Shortest` strategy), duration = distance over
438/// speed (below the packed column's saturation limit, so the `Fastest`
439/// heuristic stays admissible too).
440fn benchmark_cost_columns(
441    straight_line_meters: f64,
442    speed_kilometers_per_hour: u16,
443    benchmark_rng: &mut BenchmarkRng,
444) -> (u32, u32) {
445    let variation_meters = benchmark_rng.next_below(BENCHMARK_DISTANCE_INFLATION_METERS) as u32;
446    let distance_meters = straight_line_meters.ceil() as u32 + variation_meters;
447    let speed_meters_per_second = f64::from(speed_kilometers_per_hour)
448        * BENCHMARK_METERS_PER_KILOMETER
449        / BENCHMARK_SECONDS_PER_HOUR;
450    let duration_milliseconds = (f64::from(distance_meters) / speed_meters_per_second
451        * BENCHMARK_MILLISECONDS_PER_SECOND)
452        .round() as u32;
453    (distance_meters, duration_milliseconds)
454}

456/// Straight-line (great-circle) distance between two nodes' quantized
457/// coordinates: the admissibility floor the distance column inflates over.
458fn straight_line_meters(from_node: &NodeRecord, to_node: &NodeRecord) -> f64 {
459    geographic_coordinate_of(from_node)
460        .haversine_distance_meters(&geographic_coordinate_of(to_node))
461}
462
463/// A quantized node record as a `GeographicCoordinate` (always in range).
464fn geographic_coordinate_of(node_record: &NodeRecord) -> GeographicCoordinate {
465    GeographicCoordinate::from_degrees(
466        node_record.latitude_degrees(),
467        node_record.longitude_degrees(),
468    )
469    .expect("benchmark node coordinates are within the canonical ranges")
470}

472/// Encodes a generated archive through the same deterministic builder the
473/// compiler writes with (and the same empty auxiliary resources the
474/// routing-only schema carries in this slice).
475fn encoded_archive_bytes(
476    node_records: Vec<NodeRecord>,
477    edge_records: Vec<EdgeRecord>,
478) -> Result<Vec<u8>, SDKError> {
479    OfflineRoutingNetworkGraphBuilder::new()
480        .with_nodes(node_records)
481        .with_edges(edge_records)
482        .with_lane_guidances(Vec::new())
483        .with_instruction_strings(Vec::new())
484        .with_instruction_lookup_indices(vec![INSTRUCTION_ABSENT_INDEX; 0])
485        .with_turn_restrictions(Vec::new())
486        .encode_deterministic()
487}

543    #[test]
544    fn astar_and_dijkstra_reference_agree_on_the_synthetic_benchmark_graphs() {
545        // The result-equality half of the benchmark claim: over the same
546        // loaded graphs, A* and the Dijkstra reference return the same
547        // optimal chain at the same total cost (per-edge costs vary, so the
548        // optimum is unique), and A* settles no more nodes than Dijkstra —
549        // the measured-better-or-equal expansion sanity of the "A* beats
550        // Dijkstra" claim (all deterministic, noise-free).
551        let mut benchmark_graphs = Vec::new();
552        for side_length in [TEST_GRID_SIDE_LENGTH, TEST_THIRD_GRID_SIDE_LENGTH] {

554                format!("grid_{side_length}"),
555                benchmark_grid_loaded_graph(side_length).expect("the test grid graph loads"),

557            ));
558        }
559        for node_count in [TEST_SPANNING_NODE_COUNT, TEST_SECOND_SPANNING_NODE_COUNT] {
560            benchmark_graphs.push((
561                format!("spanning_{node_count}"),
562                benchmark_random_spanning_loaded_graph(node_count, BENCHMARK_RANDOM_SPANNING_SEED)
563                    .expect("the test spanning graph loads"),
564                node_count as u32,
565            ));
566        }
567
568        for (graph_name, loaded_graph, node_count) in &benchmark_graphs {
569            for (start_node_index, end_node_index) in benchmark_query_pairs(*node_count) {

609    /// The coarse CI budget alarm: the asserted generous upper bound on the
610    /// A* solve of the fixed benchmark graphs.
611    ///
612    /// Shared-runner wall clock is noisy, so this is deliberately NOT a
613    /// sharp gate: the bound sits three orders of magnitude above the
614    /// locally measured (sub-millisecond) solves and exists to catch
615    /// *gross* regressions (a lost heuristic, an O(n²) relaxation) — the
616    /// criterion numbers uploaded by the benchmark workflow are the
617    /// recorded trend baseline, not a CI pass/fail. If this bound ever
618    /// fires while nothing structural changed, it is runner noise; re-run
619    /// before bisecting.
620    const COARSE_BUDGET_ALARM_MILLISECONDS: u128 = 1_000;
621
622    #[test]
623    fn astar_stays_within_the_coarse_solve_budget_alarm_on_the_fixed_graphs() {
624        let grid_graph = benchmark_grid_loaded_graph(64).expect("the fixed grid graph loads");
625        let spanning_graph =
626            benchmark_random_spanning_loaded_graph(8_192, BENCHMARK_RANDOM_SPANNING_SEED)
627                .expect("the fixed spanning graph loads");
628        let fixed_queries = [
629            (
630                "grid_64x64",
631                &grid_graph,
632                benchmark_corner_query_pair(64 * 64),
633            ),
634            (
635                "spanning_8192",
636                &spanning_graph,
637                benchmark_corner_query_pair(8_192),
638            ),
639        ];
640
641        for (graph_name, loaded_graph, (start_node_index, end_node_index)) in fixed_queries {
642            let solve_started_at = std::time::Instant::now();
643            let astar_route = a_star_search(
644                loaded_graph,
645                start_node_index,
646                end_node_index,
647                TraversalCost::TimeSeconds,
648            )
649            .unwrap_or_else(|| panic!("{graph_name} connects the fixed query pair"));
650            let solve_elapsed_milliseconds = solve_started_at.elapsed().as_millis();
651            assert!(
652                solve_elapsed_milliseconds < COARSE_BUDGET_ALARM_MILLISECONDS,
653                "{graph_name}: the A* solve took {solve_elapsed_milliseconds} ms, above the \
654                 {} ms coarse budget alarm (a gross-regression signal — re-run before \
655                 bisecting, shared-runner wall clock is noisy)",
656                COARSE_BUDGET_ALARM_MILLISECONDS
657            );
658            // The alarm graph is also sanity-sized: the solve is fully
659            // answered (non-empty chain ending at the requested terminal).
660            assert_eq!(
661                astar_route
662                    .node_chain_indices
663                    .last()
664                    .expect("non-empty chain"),
665                &end_node_index
666            );
667        }
668    }
