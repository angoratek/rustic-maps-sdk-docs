1//! The exported map surface view: the FFI face of the `rustic-maps-rendering`
2//! surface renderer, attaching a platform window surface, driving its
3//! lifecycle and drawing camera-tiled frames with the overlay list over
4//! them.
5//!
6//! # The view-state locking discipline
7//!
8//! `draw` awaits the tile fetches, so no lock may be held across the await
9//! while the exported future must be `Send`. The view therefore keeps its
10//! surface state (renderer, tile scheduler, extent) in a dedicated slot the
11//! whole draw *owns* for its duration: `draw` takes the session out of the
12//! slot under one brief lock, awaits and renders with no lock held, and
13//! restores it afterwards — under a second brief lock that also honors a
14//! `detach` racing the draw. The same take-out serves `resize`, so resize
15//! works while a fetch is in flight unless its own brief configure collides
16//! with a render in progress. Overlays live in their own lock, taken only
17//! to clone the shared snapshot the frame draws. The consequences are
18//! documented on the methods (serialized draws, a mid-draw resize surfacing
19//! the busy state).
20//!
21//! # Cancellation-safe ownership
22//!
23//! The borrowed session and vector-base scheduler move into the spawned frame
24//! work, so the restore cannot live on the caller's future: a caller that
25//! drops its future mid-draw would otherwise strand the slot `Busy`. A
26//! [`FrameOwnership`] guard inside the work restores the borrowed state when
27//! the frame ends — normally, on an early error, or while unwinding from a
28//! panic — holding `Arc` handles to the view's slots so it can write them
29//! back. A racing `detach` wins: the guard drops the session and the slot
30//! stays honestly detached. The spawned work always runs to completion on the
31//! crate-owned runtime even when the awaiting future is gone, so the slot is
32//! recoverable after every way a frame can end.

164/// The presented state one settled-check compares against: the camera the
165/// last frame drew (exactly), the overlay list revision as drawn, the
166/// location layer, the vector base source it drew (if any), and the last
167/// frame's pending tile counts (fallback, deferred or unavailable raster
168/// tiles and failed vector base tiles may still change pixels when their
169/// detail arrives).
170#[derive(Debug, Clone)]
171struct LastDrawnFrame {
172    /// The camera the last frame drew, compared exactly with
173    /// `is_frame_settled`'s camera.
174    camera: rustic_maps_rendering::MapCamera,
175    /// Whether the last frame still owed cross-fade frames: a running
176    /// zoom transition keeps the presented pixels changing without any tile
177    /// left pending.
178    zoom_transition_running: bool,
179    /// The overlay store's mutation revision when the frame snapshotted its
180    /// list: any later mutation makes the presented state unsettled until the
181    /// next frame draws, so the check compares one integer instead of the
182    /// whole overlay list.
183    overlay_revision: u64,
184    /// The location layer the last frame drew.
185    location_indicator: Option<LocationIndicatorLayer>,
186    /// The device pixel ratio the last frame drew its labels and overlay
187    /// screen sizes at; a ratio change resizes both, so the frame is not
188    /// settled against a different one.
189    device_pixel_ratio: f64,
190    /// The vector base source the last frame drew (no source drew none),
191    /// compared with the source installed at the settle check.
192    vector_base_source: Option<VectorBaseSourceSummary>,
193    /// How many of the last frame's requested tiles may still change pixels
194    /// (the fallback, deferred or unavailable counts' sum, plus the vector
195    /// base's failed tiles): a settled frame draws none of them.
196    pending_tile_count: u32,
197    /// Whether the last frame was drawn in the vector base's motion-graded
198    /// mode. A graded frame deliberately skipped style and label assembly, so
199    /// the presented pixels are not the resolved frame even when the camera is
200    /// back at rest: the settle check keeps the pump drawing until a still
201    /// frame resolves them.
202    motion_graded: bool,
203}

205impl LastDrawnFrame {
206    /// Whether the frame's presented pixels may still change: a pending

208    /// frame whose settled assembly is still owed.
209    fn drawn_pixels_may_still_change(&self) -> bool {
210        self.pending_tile_count > 0 || self.zoom_transition_running || self.motion_graded
211    }

214/// Prefix of every identifier `add_overlay` generates: the identifier is
215/// this prefix plus the view's monotonically increasing identifier index.
216const OVERLAY_IDENTIFIER_PREFIX: &str = "overlay-";
217
218/// Index of the first identifier `add_overlay` generates (`overlay-1`). The
219/// counter only ever increases, so one view never reuses an identifier.
220const FIRST_OVERLAY_IDENTIFIER_INDEX: u64 = 1;
221
222/// Milliseconds in one second, converting the cross-fade clock's elapsed
223/// duration into the milliseconds its state machine reads.
224const MILLISECONDS_PER_SECOND: f64 = 1_000.0;
225
226/// The refusal detail `update_overlay` throws for an identifier that does
227/// not address any current overlay; the identifier itself is appended.
228const UNKNOWN_OVERLAY_IDENTIFIER_DETAIL_PREFIX: &str = "no overlay is addressed by identifier ";

230/// The view's overlay state: the drawn list in draw order plus the
231/// identifier generator. `set_overlays` replaces the whole list and drops
232/// every identifier (the documented bulk-call semantics), while
233/// `add_overlay` appends and hands out the next identifier; the counter
234/// survives both calls (and detach/attach) so an identifier is never reused
235/// within one view's lifetime.
236///
237/// The drawn overlays live behind one [`Arc`]: a frame's snapshot is an `Arc`
238/// clone, so an unchanged overlay list costs no payload copy per frame. A
239/// mutation after the snapshot was taken copies the list once
240/// ([`Arc::make_mut`]), leaving the in-flight frame drawing what it snapshotted.
241#[derive(Debug)]
242struct OverlayStore {
243    /// The validated overlays drawn in every frame, in draw order.
244    drawn_overlays: Arc<Vec<RenderingMapOverlay>>,
245    /// The identifier addressing each drawn overlay at the same position;
246    /// `None` for entries supplied through `set_overlays`.
247    overlay_identifiers: Vec<Option<String>>,
248    /// The index the next `add_overlay` identifier carries.
249    next_identifier_index: u64,
250    /// A monotonically increasing stamp bumped by every mutation, so the
251    /// settle check compares one integer instead of the whole list.
252    revision: u64,
253}
254
255/// The revision a store that has never been mutated carries.
256const FIRST_OVERLAY_REVISION: u64 = 0;
257
258impl Default for OverlayStore {
259    /// Builds an empty store whose first generated identifier is
260    /// `overlay-1` and whose revision is [`FIRST_OVERLAY_REVISION`].
261    fn default() -> Self {
262        Self {
263            drawn_overlays: Arc::new(Vec::new()),
264            overlay_identifiers: Vec::new(),
265            next_identifier_index: FIRST_OVERLAY_IDENTIFIER_INDEX,
266            revision: FIRST_OVERLAY_REVISION,
267        }
268    }
269}
270
271impl OverlayStore {
272    /// The shared snapshot of the native overlays in draw order — the list a
273    /// frame draws. Cloning the snapshot is cloning one reference count, so a
274    /// frame over unchanged overlays never copies the payload.
275    fn drawn_overlays(&self) -> Arc<Vec<RenderingMapOverlay>> {
276        Arc::clone(&self.drawn_overlays)
277    }
278
279    /// The current mutation revision; the settle check compares it with the
280    /// revision the last drawn frame snapshotted.
281    fn revision(&self) -> u64 {
282        self.revision
283    }
284
285    /// Advances the mutation revision, saturating at [`u64::MAX`] rather than
286    /// wrapping back to a revision an earlier frame recorded.
287    fn advance_revision(&mut self) {
288        self.revision = self.revision.saturating_add(1);
289    }
290
291    /// Replaces the whole drawn list with unaddressed entries (the
292    /// `set_overlays` bulk semantics): every identifier generated earlier
293    /// stops addressing anything.
294    fn replace_all(&mut self, overlays: Vec<RenderingMapOverlay>) {
295        self.overlay_identifiers = vec![None; overlays.len()];
296        self.drawn_overlays = Arc::new(overlays);
297        self.advance_revision();
298    }
299
300    /// Appends one addressed overlay and answers its generated identifier.
301    fn add(&mut self, overlay: RenderingMapOverlay) -> String {
302        let overlay_identifier =
303            format!("{OVERLAY_IDENTIFIER_PREFIX}{}", self.next_identifier_index);
304        self.next_identifier_index = self.next_identifier_index.saturating_add(1);
305        Arc::make_mut(&mut self.drawn_overlays).push(overlay);
306        self.overlay_identifiers
307            .push(Some(overlay_identifier.clone()));
308        self.advance_revision();
309        overlay_identifier
310    }
311
312    /// Replaces the overlay the identifier addresses, keeping its position
313    /// in the draw order; an identifier that addresses nothing refuses with
314    /// a precise detail (an addressed-overlay input failure, reported as
315    /// `InvalidRenderInput` like every overlay-addressing input refusal).
316    fn update(
317        &mut self,
318        overlay_identifier: &str,
319        overlay: RenderingMapOverlay,
320    ) -> Result<(), SdkError> {
321        let Some(overlay_index) = self
322            .overlay_identifiers
323            .iter()
324            .position(|entry_identifier| entry_identifier.as_deref() == Some(overlay_identifier))
325        else {
326            return Err(SdkError::InvalidRenderInput {
327                detail: format!(
328                    "{UNKNOWN_OVERLAY_IDENTIFIER_DETAIL_PREFIX}`{overlay_identifier}`; \
329                     `add_overlay` answers the identifiers in force"
330                ),
331            });
332        };
333        Arc::make_mut(&mut self.drawn_overlays)[overlay_index] = overlay;
334        self.advance_revision();
335        Ok(())
336    }
337
338    /// Removes the overlay the identifier addresses and answers whether it
339    /// was present.
340    fn remove(&mut self, overlay_identifier: &str) -> bool {
341        let Some(overlay_index) = self
342            .overlay_identifiers
343            .iter()
344            .position(|entry_identifier| entry_identifier.as_deref() == Some(overlay_identifier))
345        else {
346            return false;
347        };
348        Arc::make_mut(&mut self.drawn_overlays).remove(overlay_index);
349        self.overlay_identifiers.remove(overlay_index);
350        self.advance_revision();
351        true
352    }
353
354    /// Clears the whole drawn list (addressed and unaddressed entries alike).
355    fn clear(&mut self) {
356        Arc::make_mut(&mut self.drawn_overlays).clear();
357        self.overlay_identifiers.clear();
358        self.advance_revision();
359    }
360}

528/// The occupancy of the view's slot: the idle drawing state, a draw that
529/// took the state out and will restore it, or the detached state.
530enum SurfaceSlot {
531    // Boxed: the drawing state is many hundreds of bytes (the renderer's
532    // pipelines and caches), several times the empty variants' size, and the
533    // slot switches states without the state usually being read — the
534    // box keeps the enum dispatch cheap.
535    Idle(Box<SurfaceSession>),
536    /// A draw holds the state; its restore only writes back into this
537    /// state, so a racing `detach` keeps the draw's state dropped.
538    Busy,
539    /// No surface state: the view renders nothing until the next attach.
540    Detached,
541}

543/// The occupancy of the vector base source's slot: the installed
544/// scheduler, a draw that took it out and will restore it, or no source.
545/// Unlike [`SurfaceSlot`], an install or clear racing a draw is not refused:
546/// it is remembered in the view's pending-change slot and applied when the
547/// draw restores the scheduler.
548enum VectorBaseSlot {
549    // Boxed for the same reason the surface state is: the scheduler's caches
550    // dwarf the empty variants.
551    Installed(Box<VectorBaseScheduler>),
552    /// A draw holds the scheduler for its duration.
553    Busy,
554    /// No vector base source: frames draw raster tiles and overlays alone.
555    Absent,
556}

558/// A base-map switch that landed while a draw owned the vector base slot.
559/// The draw's restore applies it once the frame has drawn, so the request
560/// succeeds instead of being refused; the newest request replaces any
561/// earlier one.
562enum PendingVectorBaseChange {
563    /// Drop the base when the frame finishes.
564    Clear,
565    /// Install this scheduler and summary when the frame finishes.
566    Install(Box<VectorBaseScheduler>, VectorBaseSourceSummary),
567}

576    style_id: String,
577    /// The tileset's name.

579    /// The attribution line the tileset's terms require.
580    attribution_text: String,

582    maximum_zoom_level: u8,

645/// The view state a draw hands to the spawned frame work: `Arc` handles to
646/// the slots the frame borrows and the accounting it writes.
647///
648/// The spawned work owns these handles for the frame's whole lifetime, which
649/// is what makes the ownership transfer cancellation-safe: the work restores
650/// the borrowed session and scheduler through [`FrameOwnership`] even when the
651/// foreign caller drops the future awaiting it. Cloning this struct clones
652/// reference counts only.
653#[derive(Clone)]
654struct FrameSharedState {
655    /// The SDK state, for the provider-selection accounting a base-map switch
656    /// writes.
657    map_sdk: Arc<MapSdk>,
658    /// The surface state's occupancy.
659    surface_slot: Arc<Mutex<SurfaceSlot>>,
660    /// The vector base scheduler's occupancy.
661    vector_base: Arc<Mutex<VectorBaseSlot>>,
662    /// What the view draws a vector base with.
663    vector_base_summary: Arc<Mutex<Option<VectorBaseSourceSummary>>>,
664    /// The last resolved vector base frame's accounting.
665    last_vector_base_frame: Arc<Mutex<Option<LastVectorBaseFrame>>>,
666    /// The base-map switch a call requested while a draw owned the scheduler.
667    pending_vector_base_change: Arc<Mutex<Option<PendingVectorBaseChange>>>,
668    /// The raster provider a `reload_tile_source` call resolved while the
669    /// frame owned the surface state, applied by the restore.
670    pending_raster_tile_reload: Arc<Mutex<Option<Arc<dyn RasterTileProvider>>>>,
671    /// The presented state the settle check compares against.
672    last_drawn_frame: Arc<Mutex<Option<LastDrawnFrame>>>,
673}
674
675impl FrameSharedState {
676    /// Records the resolved vector base frame's accounting for the status
677    /// readback. Taken before the scheduler restore, so a base-map switch
678    /// applied there resets it instead of pairing the new source's summary
679    /// with the previous source's tile counts.
680    fn record_vector_base_frame(&self, vector_base_frame: &VectorBaseFrame) {
681        *crate::panic_boundary::state_lock_guard(
682            &self.last_vector_base_frame,
683            "the surface view's last vector base frame",
684        ) = Some(LastVectorBaseFrame::from_vector_base_frame(
685            vector_base_frame,
686        ));
687    }
688
689    /// Records the presented state one completed frame drew: the camera it
690    /// drew (exactly), the overlay revision it snapshotted, the location
691    /// layer and device pixel ratio it drew, the vector base source it drew,
692    /// and the pending tile counts whose later detail may still change pixels.
693    #[allow(clippy::too_many_arguments)]
694    fn record_last_drawn_frame(
695        &self,
696        camera: rustic_maps_rendering::MapCamera,
697        overlay_revision: u64,
698        location_indicator: Option<LocationIndicatorLayer>,
699        device_pixel_ratio: f64,
700        vector_base_source: Option<VectorBaseSourceSummary>,
701        frame_statistics: &FrameStatistics,
702        vector_base_frame: Option<&VectorBaseFrame>,
703        zoom_transition_running: bool,
704    ) {
705        let vector_base_failed_tile_count =
706            u32::try_from(vector_base_frame.map_or(0, |frame| frame.failed_tile_count))
707                .unwrap_or(u32::MAX);
708        let vector_base_pending_tile_count =
709            u32::try_from(vector_base_frame.map_or(0, |frame| frame.pending_tile_count))
710                .unwrap_or(u32::MAX);
711        let motion_graded =
712            vector_base_frame.is_some_and(|vector_base_frame| vector_base_frame.motion_graded);
713        *crate::panic_boundary::state_lock_guard(
714            &self.last_drawn_frame,
715            "the surface view's last drawn frame",
716        ) = Some(LastDrawnFrame {
717            camera,
718            zoom_transition_running,
719            overlay_revision,
720            location_indicator,
721            device_pixel_ratio,
722            vector_base_source,
723            pending_tile_count: frame_statistics.fallback_tile_count
724                + frame_statistics.deferred_tile_count
725                + frame_statistics.unavailable_tile_count
726                + vector_base_failed_tile_count
727                + vector_base_pending_tile_count,
728            motion_graded,
729        });
730    }
731
732    /// Restores the drawn session into the slot, but only when the slot still
733    /// expects it: a racing `detach` left the slot [`SurfaceSlot::Detached`]
734    /// and the drawn state stays dropped. A reload requested while the frame
735    /// owned the state is applied first: the restored session takes a fresh
736    /// scheduler over the resolved provider, and the presented frame is
737    /// marked unsettled so the pump draws the re-planned tiles.
738    fn restore_surface_session(&self, mut surface_session: Box<SurfaceSession>) {
739        let pending_raster_tile_reload = crate::panic_boundary::state_lock_guard(
740            &self.pending_raster_tile_reload,
741            "the surface view's pending raster-tile reload",
742        )
743        .take();
744        let mut slot_state =
745            crate::panic_boundary::state_lock_guard(&self.surface_slot, "the surface view's slot");
746        if !matches!(*slot_state, SurfaceSlot::Busy) {
747            return;
748        }
749        if let Some(raster_tile_provider) = pending_raster_tile_reload {
750            surface_session.replace_raster_tile_scheduler(raster_tile_provider);
751            *crate::panic_boundary::state_lock_guard(
752                &self.last_drawn_frame,
753                "the surface view's last drawn frame",
754            ) = None;
755        }
756        *slot_state = SurfaceSlot::Idle(surface_session);
757    }
758
759    /// Restores the drawn scheduler into the slot and applies the base-map
760    /// switch a call requested while the draw owned it: a pending clear
761    /// leaves the slot absent and drops what the draw borrowed, a pending
762    /// install replaces the drawn scheduler and its summary. The pending
763    /// change is taken in the same step, so it can never reach a later draw.
764    /// Without a pending change the drawn scheduler is written back, but only
765    /// while the slot still expects it.
766    fn restore_vector_base_scheduler(
767        &self,
768        vector_base_scheduler: Option<Box<VectorBaseScheduler>>,
769    ) {
770        let mut slot_state = crate::panic_boundary::state_lock_guard(
771            &self.vector_base,
772            "the surface view's vector base",
773        );
774        if !matches!(*slot_state, VectorBaseSlot::Busy) {
775            return;
776        }
777        match crate::panic_boundary::state_lock_guard(
778            &self.pending_vector_base_change,
779            "the surface view's pending vector base change",
780        )
781        .take()
782        {
783            Some(PendingVectorBaseChange::Clear) => {
784                *slot_state = VectorBaseSlot::Absent;
785                *crate::panic_boundary::state_lock_guard(
786                    &self.vector_base_summary,
787                    "the surface view's vector base summary",
788                ) = None;
789                *crate::panic_boundary::state_lock_guard(
790                    &self.last_vector_base_frame,
791                    "the surface view's last vector base frame",
792                ) = None;
793                self.map_sdk
794                    .clear_provider_selection(crate::records::ProviderContract::VectorTiles);
795            }
796            Some(PendingVectorBaseChange::Install(
797                pending_vector_base_scheduler,
798                pending_source_summary,
799            )) => {
800                self.map_sdk.record_provider_selection(
801                    crate::records::ProviderContract::VectorTiles,
802                    &pending_source_summary.source_name,
803                    &pending_source_summary.source_name,
804                );
805                *crate::panic_boundary::state_lock_guard(
806                    &self.vector_base_summary,
807                    "the surface view's vector base summary",
808                ) = Some(pending_source_summary);
809                *crate::panic_boundary::state_lock_guard(
810                    &self.last_vector_base_frame,
811                    "the surface view's last vector base frame",
812                ) = None;
813                *slot_state = VectorBaseSlot::Installed(pending_vector_base_scheduler);
814            }
815            None => {
816                *slot_state = match vector_base_scheduler {
817                    Some(vector_base_scheduler) => VectorBaseSlot::Installed(vector_base_scheduler),
818                    None => VectorBaseSlot::Absent,
819                };
820            }
821        }
822    }
823}

825/// Owns the surface session and vector base scheduler for one frame and
826/// restores them into the view's shared slots when the frame ends.
827///
828/// The guard runs *inside* the spawned frame work, so the ownership transfer
829/// survives every way a frame can end: a completed frame, an early error, a
830/// racing detach (which leaves the slot honestly [`SurfaceSlot::Detached`]),
831/// and a panic that unwinds the work. The caller's future may be dropped
832/// without the frame stranding the slot [`SurfaceSlot::Busy`]; the spawned
833/// work still runs to completion and drops this guard.
834struct FrameOwnership {
835    /// The shared slots the restore writes into.
836    shared: FrameSharedState,
837    /// The borrowed session; taken by the restore so a second call is a
838    /// no-op.
839    surface_session: Option<Box<SurfaceSession>>,
840    /// The borrowed vector base scheduler (`None`: no base was installed).
841    vector_base_scheduler: Option<Box<VectorBaseScheduler>>,
842    /// Whether the restore already ran, so [`Drop`] stays idempotent.
843    restored: bool,
844}
845
846impl FrameOwnership {
847    /// Takes ownership of the borrowed session and scheduler for one frame.
848    fn new(
849        shared: FrameSharedState,
850        surface_session: Box<SurfaceSession>,
851        vector_base_scheduler: Option<Box<VectorBaseScheduler>>,
852    ) -> Self {
853        Self {
854            shared,
855            surface_session: Some(surface_session),
856            vector_base_scheduler,
857            restored: false,
858        }
859    }
860
861    /// The borrowed parts the frame draws through; the guard keeps the
862    /// ownership so a panic still restores them.
863    fn borrowed_parts(&mut self) -> (&mut SurfaceSession, Option<&mut VectorBaseScheduler>) {
864        (
865            crate::panic_boundary::required_invariant(
866                self.surface_session.as_deref_mut(),
867                "the frame ownership guard holds the session until it restores",
868            ),
869            self.vector_base_scheduler.as_deref_mut(),
870        )
871    }
872
873    /// Restores the session and scheduler into their slots. Idempotent, so the
874    /// frame work calls it on its normal path and [`Drop`] calls it while
875    /// unwinding from a panic.
876    fn restore(&mut self) {
877        if self.restored {
878            return;
879        }
880        self.restored = true;
881        if let Some(surface_session) = self.surface_session.take() {
882            self.shared.restore_surface_session(surface_session);
883        }
884        self.shared
885            .restore_vector_base_scheduler(self.vector_base_scheduler.take());
886    }
887}
888
889impl Drop for FrameOwnership {
890    /// Restores the borrowed state if the frame work did not reach its own
891    /// restore, which is the panic path.
892    fn drop(&mut self) {
893        self.restore();
894    }
895}

1143    /// Attaches the platform surface (`attach`'s documented UDL method),
1144    /// building the whole drawing machinery — surface target, renderer,
1145    /// configured extent, tile scheduler — before installing it, so a busy
1146    /// slot races nothing. Attaching an attached view is refused.
1147    pub fn attach(
1148        &self,
1149        surface_handle: SurfaceHandle,
1150        width_pixels: u32,
1151        height_pixels: u32,
1152    ) -> Result<(), SdkError> {
1153        let surface_session =
1154            self.build_surface_session(surface_handle, width_pixels, height_pixels)?;
1155        let mut slot_state = self.surface_slot_lock();
1156        match *slot_state {
1157            // Replacing an attached view is refused: the platform glue's
1158            // lifecycle is detach-then-attach.
1159            SurfaceSlot::Idle(_) => {
1160                return Err(SdkError::RenderingSurfaceUnavailable {
1161                    detail: "the map surface view is already attached; detach first".to_string(),
1162                });
1163            }
1164            // A draw holding the state will restore it; attaching mid-draw
1165            // loses and retries after the draw (and, if intended, a detach).
1166            SurfaceSlot::Busy => {
1167                return Err(SdkError::RenderingSurfaceUnavailable {
1168                    detail: "a frame draw is in flight; attach after the frame draws".to_string(),
1169                });
1170            }
1171            // The only installable state.
1172            SurfaceSlot::Detached => {}
1173        }
1174        *slot_state = SurfaceSlot::Idle(Box::new(surface_session));
1175        drop(slot_state);
1176        self.camera_state_lock().viewport = Some(ViewportSize {
1177            width_pixels: f64::from(width_pixels),
1178            height_pixels: f64::from(height_pixels),
1179        });
1180        Ok(())
1181    }

1184    /// non-zero). Works while a fetch is in flight — the scheduler is not
1185    /// touched — and surfaces the busy state only against an active render.
1186    pub fn resize(&self, width_pixels: u32, height_pixels: u32) -> Result<(), SdkError> {
1187        if width_pixels == 0 || height_pixels == 0 {
1188            return Err(SdkError::InvalidRenderInput {
1189                detail: format!(
1190                    "resize extent {width_pixels}x{height_pixels} must be \
1191                     non-zero; a platform view between layouts waits for the \
1192                     next laid out size"
1193                ),
1194            });
1195        }
1196        let mut slot_state = self.surface_slot_lock();

1202                    .map_err(|surface_error| rendering_failure_to_sdk_error(&surface_error))?;
1203                surface_session.width_pixels = width_pixels;
1204                surface_session.height_pixels = height_pixels;

1344    /// Reads whether the view's presented state would not change if another
1345    /// frame drew right now (`is_frame_settled`'s documented UDL method): the
1346    /// camera equals the last drawn camera exactly, the overlay list carries
1347    /// the revision the last frame drew, and the location layer, device pixel
1348    /// ratio and vector base source are unchanged since that draw, and the
1349    /// last frame resolved no pending pixels (raster or vector base). Any
1350    /// overlay mutation since the last draw marks the frame unsettled, even
1351    /// when it restores equal content, until the next frame draws. A pure
1352    /// query — it never draws, fetches or mutates state. It answers `false`
1353    /// while a draw is in flight (the presented state is then changing) and
1354    /// before a frame has ever drawn, and throws when the view is detached.
1355    pub fn is_frame_settled(&self, camera: Arc<MapCamera>) -> Result<bool, SdkError> {
1356        match &*self.surface_slot_lock() {
1357            SurfaceSlot::Detached => {
1358                return Err(SdkError::RenderingSurfaceUnavailable {
1359                    detail: SURFACE_DETACHED_DETAIL.to_string(),
1360                });
1361            }
1362            SurfaceSlot::Busy => return Ok(false),
1363            SurfaceSlot::Idle(_) => {}
1364        }
1365        let Some(last_drawn_frame) = self.last_drawn_frame_lock().clone() else {
1366            // No frame has drawn yet: the presented state is whatever the
1367            // platform left, and the first draw would change it.
1368            return Ok(false);
1369        };
1370        let drawn_overlay_revision = self.overlays_lock().revision();
1371        let location_indicator = self.location_indicator_lock().clone();
1372        let vector_base_source = self.vector_base_summary_lock().clone();
1373        if last_drawn_frame.camera != *camera.native_camera()
1374            || drawn_overlay_revision != last_drawn_frame.overlay_revision
1375            || location_indicator != last_drawn_frame.location_indicator
1376            || self.stored_device_pixel_ratio() != last_drawn_frame.device_pixel_ratio
1377            || vector_base_source != last_drawn_frame.vector_base_source
1378        {
1379            return Ok(false);
1380        }
1381        Ok(!last_drawn_frame.drawn_pixels_may_still_change())
1382    }

1384    /// Re-resolves the raster-tile provider and drops the scheduler's pooled
1385    /// raster state (`reload_tile_source`'s documented UDL method), so a
1386    /// provider, credential or source change reaches the following frames
1387    /// without an attach/detach cycle.
1388    ///
1389    /// The next draw builds a fresh scheduler and payload pool over the
1390    /// current provider: pooled payloads, in-flight warm fetches and the
1391    /// retained cross-fade level from the previous source are dropped, so no
1392    /// tile fetched from it can draw again, and the frame reports its
1393    /// previously pooled tiles as missing while it re-plans them. Overlays,
1394    /// the location layer, the camera, the device pixel ratio and the vector
1395    /// base survive the reload. The view is left unsettled so the platform's
1396    /// pump draws the re-planned frame.
1397    ///
1398    /// A call made while a frame draws is applied when that frame finishes,
1399    /// and the following frames use the new provider; a later call replaces
1400    /// the earlier one. While detached there is no pooled state to clear and
1401    /// the next attach resolves the current provider, so the call succeeds
1402    /// and changes nothing.
1403    ///
1404    /// # Errors
1405    /// Throws `ProviderNotConfigured` when the view is attached and no
1406    /// raster-tile provider is registered; the previous source stays in
1407    /// force.
1408    pub fn reload_tile_source(&self) -> Result<(), SdkError> {
1409        // The pending slot is taken before the surface slot, the same order
1410        // the frame restore uses, so a reload racing a restore can never
1411        // deadlock on the two locks.
1412        let mut pending_raster_tile_reload = self.pending_raster_tile_reload_lock();
1413        let mut slot_state = self.surface_slot_lock();
1414        match &*slot_state {
1415            SurfaceSlot::Detached => return Ok(()),
1416            SurfaceSlot::Busy | SurfaceSlot::Idle(_) => {}
1417        }
1418        let raster_tile_provider = self.map_sdk.required_raster_tile_provider()?;
1419        match &mut *slot_state {
1420            SurfaceSlot::Busy => {
1421                *pending_raster_tile_reload = Some(raster_tile_provider);
1422            }
1423            SurfaceSlot::Idle(surface_session) => {
1424                surface_session.replace_raster_tile_scheduler(raster_tile_provider);
1425                // The presented pixels will change when the re-planned tiles
1426                // arrive, so no earlier frame can answer settled.
1427                *self.last_drawn_frame_lock() = None;
1428            }
1429            SurfaceSlot::Detached => unreachable!("the detached slot returned above"),
1430        }
1431        Ok(())
1432    }

1518    /// (`set_provider_vector_tile_source_from_document`'s documented UDL
1519    /// method).

1522    /// style-resource fetch, so the credential stays in the provider client
1523    /// and the request carries the provider's own transport, pacing and retry
1524    /// policy. A document that compiled is kept per provider and address, so
1525    /// installing the same family again costs no request; a document that
1526    /// fails to fetch, decode or compile throws and leaves the previous base
1527    /// in force.
1528    ///
1529    /// # Parameters
1530    /// - `source` — the configured provider key and the style label the
1531    ///   status reports.
1532    /// - `style_document_url` — the absolute address of the provider's style
1533    ///   document, without a credential.
1534    ///
1535    /// # Errors
1536    /// Throws when no vector-tile provider serves the requested key, when the
1537    /// provider registers no style-resource fetch, when the request fails, or
1538    /// when the served document is not a compilable style.
1539    pub async fn set_provider_vector_tile_source_from_document(
1540        &self,
1541        source: ProviderVectorTileSource,
1542        style_document_url: String,
1543    ) -> Result<(), SdkError> {
1544        if let Some((installed_provider_key, _)) = self
1545            .map_sdk
1546            .selected_provider(crate::records::ProviderContract::VectorTiles)
1547            && installed_provider_key != source.provider_key
1548        {
1549            return Err(SdkError::ProviderNotConfigured {
1550                contract_name: format!(
1551                    "vector tiles from provider {}; the installed vector-tile provider is {}",
1552                    source.provider_key, installed_provider_key
1553                ),
1554            });
1555        }
1556        let vector_tile_provider = self.map_sdk.required_vector_tile_provider()?;
1557        let style_resources = self.style_resource_client(&source.provider_key)?;
1558        let source_descriptor = vector_tile_provider.source_descriptor();
1559        let style_document = match self
1560            .cached_provider_style_document(&source.provider_key, &style_document_url)
1561        {
1562            Some(style_document) => style_document,
1563            None => {
1564                let document_url = style_document_url.clone();
1565                let document_bytes =
1566                    drive(async move { style_resources.fetch_style_resource(&document_url).await })
1567                        .await?;
1568                String::from_utf8(document_bytes).map_err(|utf8_error| {
1569                    SdkError::ResponseParsingFailure {
1570                        detail: format!(
1571                            "the style document the vector base source {} served is not UTF-8 \
1572                             text: {utf8_error}",
1573                            source_descriptor.source_name
1574                        ),
1575                    }
1576                })?
1577            }
1578        };
1579        let vector_base_scheduler = VectorBaseScheduler::from_style_document_json(
1580            vector_tile_provider,
1581            &source.style_id,
1582            &style_document,
1583        )
1584        .map_err(|style_error| SdkError::ResponseParsingFailure {
1585            detail: format!(
1586                "the style document the vector base source {} served does not compile: \
1587                     {style_error}",
1588                source_descriptor.source_name
1589            ),
1590        })?;
1591        // The style's own labels draw through the glyph endpoint it declares,
1592        // fetched with the provider's credential. A style that declares no
1593        // endpoint, or whose ranges the install could not complete, keeps the
1594        // embedded Latin range; a glyph address that resolves off the
1595        // document's origin refuses the install before any request.
1596        let vector_base_scheduler = match self
1597            .hosted_label_font(
1598                &source.provider_key,
1599                &style_document_url,
1600                &vector_base_scheduler,
1601            )
1602            .await?
1603        {
1604            Some(label_font) => vector_base_scheduler.with_label_font(label_font),
1605            None => vector_base_scheduler,
1606        };
1607        // The style's own icons draw from the sprite it declares, fetched the
1608        // same way. The renderer's sprite seam is synchronous, so the install
1609        // prefetches and hands over the decoded atlas; a style without a
1610        // sprite, or one the install could not complete, draws its text
1611        // without icons, and an off-origin sprite base refuses the install.
1612        #[cfg(not(target_arch = "wasm32"))]
1613        let vector_base_scheduler = match self
1614            .hosted_sprite_atlas(
1615                &source.provider_key,
1616                &style_document_url,
1617                &vector_base_scheduler,
1618            )
1619            .await?
1620        {
1621            Some((sprite_url, sprite_atlas)) => {
1622                vector_base_scheduler.with_prefetched_sprite_atlas(sprite_url, sprite_atlas)
1623            }
1624            None => vector_base_scheduler,
1625        };
1626        // Remember the document only after it compiled, so a malformed
1627        // document is fetched again rather than remembered as usable.
1628        self.remember_provider_style_document(
1629            source.provider_key.clone(),
1630            style_document_url,
1631            style_document,
1632        );
1633        let vector_base_scheduler =
1634            self.vector_base_scheduler_with_payload_cache(vector_base_scheduler);
1635        let source_summary =
1636            VectorBaseSourceSummary::from_scheduler(source.style_id, &vector_base_scheduler);
1637        self.install_vector_base_scheduler(vector_base_scheduler, source_summary, None)
1638    }

1909    /// to the drawing frame when one owns the slot. `selection` names the

1913        &self,

1915        source_summary: VectorBaseSourceSummary,
1916        selection: Option<(String, String)>,
1917    ) -> Result<(), SdkError> {

2037    /// Draws one frame for the camera (`draw`'s documented UDL method):
2038    /// lists the camera's visible tiles, resolves them straight from the
2039    /// pool's state (no await on the network), plans the detached warm fetch
2040    /// of the misses — the visible set plus the prefetch planner's ahead
2041    /// ring — and spawns it fire-and-forget through the crate-owned
2042    /// runtime, then renders tiles, overlays and the location layer, and
2043    /// presents. A never-awaited frame, with pending detail arriving on
2044    /// later frames' resolves (which `is_frame_settled`'s pump resumes on).
2045    /// Returns the frame's tile and overlay statistics so the platform
2046    /// render loop can report exactly what a frame drew (the round-9
2047    /// contract ask), and `is_frame_settled` can compare the pending counts.
2048    ///
2049    /// The frame's work hands to the crate-owned runtime through [`drive`],
2050    /// together with the taken-out session, the shared slot handles and the
2051    /// overlay and indicator snapshots, so the poller awaits only a
2052    /// runtime-independent handback. The spawned work owns the session and
2053    /// scheduler through a [`FrameOwnership`] guard that restores them when
2054    /// the frame ends, so dropping the caller's future mid-draw cannot strand
2055    /// the slot busy: the work still runs to completion and restores the
2056    /// session (or leaves the slot honestly detached when a `detach` raced).
2057    /// A panicking frame surfaces as the same panic on a caller still awaiting
2058    /// it, and the guard's drop during unwinding still restores the slot.
2059    pub async fn draw(&self, camera: Arc<MapCamera>) -> Result<FrameStatistics, SdkError> {
2060        let native_camera = *camera.native_camera();
2061        let surface_session = self.take_surface_session()?;
2062        let (overlays, overlay_revision) = {
2063            let overlay_store = self.overlays_lock();
2064            (overlay_store.drawn_overlays(), overlay_store.revision())
2065        };
2066        let location_indicator = self.location_indicator_lock().clone();
2067        let device_pixel_ratio = self.stored_device_pixel_ratio();
2068        let drawn_vector_base_source = self.vector_base_summary_lock().clone();
2069        // The camera is the platform's gesture seam: a frame drawn at a pose
2070        // other than the last presented one is a frame in motion (a pointer
2071        // pan, a pinch, a rotation or an eased camera move), so the vector
2072        // base grades its assembly and the settle check below keeps the pump
2073        // drawing until a still frame resolves it.
2074        let motion_graded = self
2075            .last_drawn_frame_lock()
2076            .as_ref()
2077            .is_some_and(|last_drawn_frame| last_drawn_frame.camera != native_camera);
2078        let vector_base_scheduler = self.take_vector_base_scheduler();
2079        let shared_state = self.frame_shared_state();
2080        drive(async move {
2081            let mut frame_ownership = FrameOwnership::new(
2082                FrameSharedState::clone(&shared_state),
2083                surface_session,
2084                vector_base_scheduler,
2085            );
2086            let draw_outcome = {
2087                let (surface_session, vector_base_scheduler) = frame_ownership.borrowed_parts();
2088                Self::draw_frame(
2089                    surface_session,
2090                    &native_camera,
2091                    vector_base_scheduler,
2092                    &overlays,
2093                    location_indicator.as_ref(),
2094                    device_pixel_ratio,
2095                    motion_graded,
2096                )
2097                .await
2098            };
2099            // The drawn frame's accounting describes the source that frame
2100            // drew. Recording it before the restore lets a base-map switch
2101            // applied there reset it, instead of pairing the new source's
2102            // summary with the previous source's tile counts.
2103            if let Ok((_, Some(drawn_vector_base_frame), _)) = &draw_outcome {
2104                shared_state.record_vector_base_frame(drawn_vector_base_frame);
2105            }
2106            // The presented state is recorded before the restore too: a
2107            // raster reload applied there drops it, so the restored view
2108            // answers unsettled and the pump draws the re-planned frame.
2109            if let Ok((frame_statistics, vector_base_frame, zoom_transition_running)) =
2110                &draw_outcome
2111            {

2113                    native_camera,
2114                    overlay_revision,
2115                    location_indicator,
2116                    device_pixel_ratio,
2117                    drawn_vector_base_source,
2118                    frame_statistics,
2119                    vector_base_frame.as_deref(),
2120                    *zoom_transition_running,
2121                );
2122            }
2123            frame_ownership.restore();
2124            draw_outcome
2125        })
2126        .await
2127        .map(|(frame_statistics, _vector_base_frame, _zoom_transition_running)| frame_statistics)
2128    }

2157    /// its await. (The navigation
2158    /// corridor leg of the prefetch planner stays unwired here: the surface
2159    /// view has no navigation session to pull a corridor from; the
2160    /// ring-ahead and fallback layers still prefetch.)
2161    async fn draw_frame(
2162        surface_session: &mut SurfaceSession,
2163        native_camera: &rustic_maps_rendering::MapCamera,
2164        vector_base_scheduler: Option<&mut VectorBaseScheduler>,
2165        overlays: &[RenderingMapOverlay],
2166        location_indicator_layer: Option<&LocationIndicatorLayer>,
2167        device_pixel_ratio: f64,
2168        motion_graded: bool,
2169    ) -> Result<(FrameStatistics, Option<Arc<VectorBaseFrame>>, bool), SdkError> {
2170        let viewport = RenderingViewportPixels::new(
2171            f64::from(surface_session.width_pixels),
2172            f64::from(surface_session.height_pixels),
2173        )
2174        .map_err(|failure_message| SdkError::RenderingSurfaceUnavailable {
2175            detail: format!(
2176                "{SURFACE_SEAM_LABEL} surface extent became unusable: {failure_message}"
2177            ),
2178        })?;
2179        // A styled vector base replaces the raster base instead of tinting it:
2180        // while one is installed the raster layer is hidden, so the imagery and
2181        // its baked labels cannot show through the style's translucent fills.
2182        // The browser canvas takes the same decision (`setBaseMapVisible`).
2183        let raster_base_serves = vector_base_scheduler.is_none();
2184        let (prepared_tiles, retained_tile_level, zoom_transition_running) = if raster_base_serves {
2185            let tile_zoom_level = native_camera.tile_zoom_level();
2186            let visible_tile_keys = native_camera.visible_tile_keys(
2187                &viewport,
2188                DEFAULT_FIELD_OF_VIEW_DEGREES,
2189                tile_zoom_level,

2200                None,
2201                surface_session.tile_scheduler.texture_pool(),
2202                &surface_session.tile_scheduler.warming_tile_keys(),
2203            );

2205                .tile_scheduler

2209            }

2211                .tile_scheduler

2216            // arrive. The transition also stays unsettled, so the platform's
2217            // pump draws the fade out instead of leaving it half way.

2220                .zoom_transition_clock_origin
2221                .elapsed()

2227                ready_tile_fraction,

2230                transition_frame

2232                    .map(|retained_tile_zoom_level| RetainedTileLevel {
2233                        tile_zoom_level: retained_tile_zoom_level,

2237                prepared_tiles,
2238                retained_tile_level,

2241        } else {

2243        };

2245        // plan (when one serves) and composes beneath the caller's overlays.
2246        let mut style_transition_running = false;
2247        // The glyph font the style's own place names draw with, when the

2249        let mut style_label_font: Option<Arc<LabelFont>> = None;
2250        let vector_base_frame = match vector_base_scheduler {

2256                    &viewport,
2257                    DEFAULT_FIELD_OF_VIEW_DEGREES,

2259                );
2260                // The fetch runs detached and the frame draws the cache, so a
2261                // first frame over unvisited ground presents the style's

2264                // The visible plan delivers each tile the moment it lands, so
2265                // a later pump frame paints it without waiting for the whole
2266                // visible set; the native pump already redraws while a frame
2267                // is unsettled, so the callback needs no wake of its own.

2269                    .prefetch_missing_tiles_with_delivery(&vector_base_tile_keys, |_| {})

2271                    spawn_detached(prefetch);
2272                }
2273                // The style's declared transitions blend from whichever style

2282                // surface never clones the overlay list to compare or draw it.

2284                    .cached_frame_at_time_in_motion_shared(

2292                // back a parent fallback) warms after the presentable frame
2293                // is built, so the next pan starts from the cache. The plan
2294                // is detached and cancellable; it never runs inside a draw.
2295                if let Some(ring_prefetch) =
2296                    vector_base_scheduler.prefetch_ring_tiles(&vector_base_tile_keys)
2297                {
2298                    spawn_detached(ring_prefetch);
2299                }

2301            }

2303        };

2305        // overlay list paints over them. Both layers stay borrowed: the
2306        // renderer scales each declared point/CSS size at emission, so an
2307        // unchanged overlay list is never deep-cloned per frame and every

2310        let base_overlays: &[RenderingMapOverlay] =

2321        // base, a style without symbol layers, or a frame whose labels all

2323        // buffer, no atlas upload and no draw — the unlabeled frame stays
2324        // identical.
2325        let label_layer = vector_base_frame
2326            .as_ref()
2327            .and_then(|drawn_vector_base_frame| {
2328                let previous_label_placements = &surface_session.previous_label_placements;
2329                match style_label_font.as_deref() {
2330                    Some(label_font) => drawn_vector_base_frame
2331                        .label_layer_with_font_and_device_pixel_ratio_and_history(
2332                            native_camera,
2333                            &viewport,
2334                            DEFAULT_FIELD_OF_VIEW_DEGREES,
2335                            label_font,
2336                            device_pixel_ratio,
2337                            previous_label_placements,
2338                        ),
2339                    None => drawn_vector_base_frame
2340                        .label_layer_with_device_pixel_ratio_and_history(
2341                            native_camera,
2342                            &viewport,
2343                            DEFAULT_FIELD_OF_VIEW_DEGREES,
2344                            device_pixel_ratio,
2345                            previous_label_placements,
2346                        ),
2347                }
2348            });

2350        // keep their spots through a pan instead of flickering as newcomers

2357            .as_ref()

2362                .map(|layer| layer.labels().to_vec())

2369        let surface_clear_color = vector_base_frame
2370            .as_ref()
2371            .and_then(|frame| frame.background)
2372            .unwrap_or(CLEAR_COLOR_RGBA);

2374            .surface_renderer

2377            .surface_renderer

2395                    .is_some_and(|vector_base_frame| vector_base_frame.zoom_transition_running);

2398                    vector_base_frame,

2400                        || vector_zoom_transition_running

2408    /// handle lowers to a surface target, the renderer acquires its
2409    /// context and negotiates its format, the extent configures, and the

2454/// Maps the surface renderer's failures onto the binding error model: the
2455/// frame's own input problems (an overflowing draw list, a mismatched
2456/// viewport) are render-input failures, while everything about the surface
2457/// and graphics context itself is a rendering-surface failure. Both carry
2458/// the renderer's verbatim detail.
2459fn rendering_failure_to_sdk_error(surface_render_error: &SurfaceRenderError) -> SdkError {
2460    match surface_render_error {
2461        SurfaceRenderError::DrawListTooLarge {
2462            quad_count,
2463            maximum_quad_count,
2464        } => SdkError::InvalidRenderInput {
2465            detail: format!(
2466                "the frame's draw list of {quad_count} quads exceeds the \
2467                 supported {maximum_quad_count}; reduce the viewport extent \
2468                 or the fetch zoom"
2469            ),
2470        },
2471        SurfaceRenderError::ViewportExtentMismatch {
2472            viewport_extent,
2473            configured_extent,
2474        } => SdkError::InvalidRenderInput {
2475            detail: format!(
2476                "the frame's viewport extent {viewport_extent:?} does not \
2477                 match the configured surface extent {configured_extent:?}"
2478            ),
2479        },
2480        other_surface_failure => SdkError::RenderingSurfaceUnavailable {
2481            detail: other_surface_failure.to_string(),
2482        },
2483    }
2484}
