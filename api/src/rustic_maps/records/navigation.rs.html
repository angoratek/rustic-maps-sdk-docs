5/// Pace chosen by the application for a simulated trip. The SDK carries no
6/// default pace; validation happens in `rustic-maps-navigation`'s route walker.
7#[derive(Debug, Clone, Copy, PartialEq)]
8pub struct SimulatedTripPace {
9    /// Baseline travel speed in meters per second.
10    pub speed_meters_per_second: f64,
11    /// Multiplier applied to the baseline speed.
12    pub speed_multiplier: f64,
13}
14
15impl SimulatedTripPace {
16    /// Builds the native pace from the mirror record. Values cross unvalidated
17    /// (foreign code may hand anything across); the walker refuses a pace that
18    /// cannot drive a trip.
19    pub(crate) fn to_native_pace(self) -> rustic_maps_navigation::SimulatedTripPace {
20        rustic_maps_navigation::SimulatedTripPace {
21            speed_meters_per_second: self.speed_meters_per_second,
22            speed_multiplier: self.speed_multiplier,
23        }
24    }
25}

59/// High-level phase of a trip that has already been started.
60#[derive(Debug, Clone, Copy, PartialEq, Eq)]
61pub enum NavigationPhase {
62    /// Following the route's step sequence.
63    Navigating,
64    /// The deviation detector judged the last graded fix off route; progress
65    /// is frozen at the last on-route values until a graded fix recovers or a
66    /// reroute replaces the route.
67    OffRoute,
68    /// The final step has been completed; location updates no longer change
69    /// the state.
70    Arrived,
71}
72
73impl NavigationPhase {
74    /// Builds the mirror enum from the native navigation phase.
75    pub(crate) fn from_native_navigation_phase(
76        native_navigation_phase: rustic_maps_navigation::NavigationPhase,
77    ) -> Self {
78        match native_navigation_phase {
79            rustic_maps_navigation::NavigationPhase::Navigating => Self::Navigating,
80            rustic_maps_navigation::NavigationPhase::OffRoute => Self::OffRoute,
81            rustic_maps_navigation::NavigationPhase::Arrived => Self::Arrived,
82        }
83    }
84}

86/// Which grade of deviation a recorded deviation carries.
87///
88/// The two grades of the two-grade deviation distinction: off the *current
89/// step* yet still on the route line versus completely off the route. Only
90/// the fully-off grade freezes navigation; the wrong-step grade never freezes
91/// on its own.
92#[derive(Debug, Clone, Copy, PartialEq, Eq)]
93pub enum DeviationGradeKind {
94    /// The fix projects onto neither the current step nor any future step of
95    /// the remaining route line: the freezing grade. The trip's progress is
96    /// frozen until a graded fix recovers or a reroute replaces the route.
97    FullyOffRoute,
98    /// The fix is off the current step but projects onto a future step of the
99    /// remaining route line: "on route, wrong step". The trip keeps
100    /// navigating and counting down, and is never frozen by this grade alone.
101    OffStepOnRoute,
102}
103
104impl DeviationGradeKind {
105    /// Builds the mirror enum from the native deviation grade.
106    pub(crate) fn from_native_deviation_grade(
107        native_deviation_grade: rustic_maps_navigation::DeviationGrade,
108    ) -> Self {
109        match native_deviation_grade {
110            rustic_maps_navigation::DeviationGrade::FullyOffRoute => Self::FullyOffRoute,
111            rustic_maps_navigation::DeviationGrade::OffStepOnRoute => Self::OffStepOnRoute,
112        }
113    }
114}

189/// The full navigable state of an active trip, read-only for foreign code.
190///
191/// The steps of [`Self::route`] are carried as traversal-ordered indices
192/// (`step_positions`), never duplicated; the route's steps themselves arrive
193/// through the [`super::Route`] record, which carries no lane guidance. The
194/// live step-advance condition and deviation detector are not records and
195/// cannot cross the FFI; they are configured through the navigation policies
196/// on the controller, matching the native session's pairing.
197#[derive(Debug, Clone, PartialEq)]
198pub struct TripNavigationState {
199    /// Current phase of the trip.
200    pub phase: NavigationPhase,
201    /// The route being navigated.
202    pub route: Route,
203    /// All steps across every leg, flattened in traversal order.
204    pub step_positions: Vec<RouteStepPosition>,
205    /// Index into `step_positions` of the step currently traversed.
206    pub current_step_position: u64,
207    /// Progress along the remaining trip.
208    pub progress: TripProgress,
209    /// Position snapped onto the current step's shape; `None` until the first
210    /// location update has been processed.
211    pub snapped_coordinate: Option<GeographicCoordinate>,
212    /// Active deviation, as judged by the deviation detector. Set exactly
213    /// while the phase is `OffRoute`.
214    pub deviation: Option<DeviationInfo>,
215    /// Whether a reroute recalculation is currently in flight (the reroute
216    /// gate's observable readback). This is a controller wall-clock
217    /// fact, not reducer state, so the pure conversion below cannot know it;
218    /// it always builds `false` here and the only caller — the exported
219    /// controller's trip-state readback — overlays its gate's live value.
220    pub is_calculating_new_route: bool,
221}
222
223impl TripNavigationState {
224    /// Builds the mirror record from the native trip state.
225    pub(crate) fn from_native_trip_navigation_state(
226        native_trip_navigation_state: &rustic_maps_navigation::TripNavigationState,
227    ) -> Self {
228        Self {
229            phase: NavigationPhase::from_native_navigation_phase(
230                native_trip_navigation_state.phase,
231            ),
232            route: Route::from_native_route_data(&native_trip_navigation_state.route),
233            step_positions: native_trip_navigation_state
234                .step_positions
235                .iter()
236                .map(RouteStepPosition::from_native_route_step_position)
237                .collect(),
238            current_step_position: native_trip_navigation_state.current_step_position as u64,
239            progress: TripProgress::from_native_trip_progress(
240                &native_trip_navigation_state.progress,
241            ),
242            snapped_coordinate: native_trip_navigation_state
243                .snapped_coordinate
244                .as_ref()
245                .map(GeographicCoordinate::from_native_coordinate),
246            deviation: native_trip_navigation_state
247                .deviation
248                .as_ref()
249                .map(DeviationInfo::from_native_deviation_info),
250            // A pure conversion cannot know the controller's wall-clock gate
251            // state; the controller's trip-state readback overlays the live
252            // value.
253            is_calculating_new_route: false,
254        }
255    }
256}

285/// Which form a distance phrase takes. UDL enums carry no payloads, so the
286/// phrase mirrors its three forms under this kind plus the nullable fields of
287/// the [`DistancePhrase`] record.
288#[derive(Debug, Clone, Copy, PartialEq, Eq)]
289pub enum DistancePhraseKind {
290    /// The position is closer than half a near-band increment to the maneuver
291    /// point; no distance is phrased and composed text drops the prefix. The
292    /// record's magnitude is `None` on this kind.
293    Immediate,
294    /// A rounded distance magnitude in a specific unit; the record's
295    /// magnitude and unit are set on this kind.
296    Quantity,
297    /// No distance is phrased because the freezing (fully-off-route) deviation
298    /// grade holds: the projection is frozen, so any countdown composed off it
299    /// would be a phantom measurement. The record's magnitude is `None` on
300    /// this kind, and the composed text channels carry no distance prefix.
301    Suppressed,
302}
303
304/// The unit-system-aware phrasing of the distance remaining to a maneuver,
305/// rounded into the native distance phrasing bands.
306#[derive(Debug, Clone, Copy, PartialEq)]
307pub struct DistancePhrase {
308    /// Which form the phrase takes (see [`DistancePhraseKind`]).
309    pub phrase_kind: DistancePhraseKind,
310    /// Rounded magnitude of the phrase (half-up, per the native bands);
311    /// `None` for the immediate and suppressed forms.
312    pub magnitude: Option<f64>,
313    /// Unit the magnitude is expressed in; `None` for the immediate and
314    /// suppressed forms.
315    pub magnitude_unit: Option<DistanceUnit>,
316}
317
318impl DistancePhrase {
319    /// Builds the mirror record from the native distance phrase.
320    pub(crate) fn from_native_distance_phrase(
321        native_distance_phrase: &rustic_maps_navigation::DistancePhrase,
322    ) -> Self {
323        match *native_distance_phrase {
324            rustic_maps_navigation::DistancePhrase::Immediate => Self {
325                phrase_kind: DistancePhraseKind::Immediate,
326                magnitude: None,
327                magnitude_unit: None,
328            },
329            rustic_maps_navigation::DistancePhrase::Quantity { magnitude, unit } => Self {
330                phrase_kind: DistancePhraseKind::Quantity,
331                magnitude: Some(magnitude),
332                magnitude_unit: Some(DistanceUnit::from_native_distance_unit(unit)),
333            },
334            rustic_maps_navigation::DistancePhrase::Suppressed => Self {
335                phrase_kind: DistancePhraseKind::Suppressed,
336                magnitude: None,
337                magnitude_unit: None,
338            },
339        }
340    }
341}

343/// Which form an instruction text takes. UDL enums carry no payloads, so the

356/// The composed text channels of one maneuver: provider-authored text is
357/// carried verbatim beside the composed strings (the composed display variant

360/// provider variant falls back to the other, still provider-authored one).
361#[derive(Debug, Clone, PartialEq)]

368    pub provider_speech_text: Option<String>,
369    /// Banner variant: the live distance prefix plus the display variant.
370    pub composed_display_text: Option<String>,
371    /// Text-to-speech variant: the live distance prefix plus the speech
372    /// variant.
373    pub composed_speech_text: Option<String>,
374}

573/// One raw position fix delivered to the navigation controller; mirrors the

590}

606/// Which step-advance strategy a navigation policy configures.
607#[derive(Debug, Clone, Copy, PartialEq, Eq)]
608pub enum StepAdvancePolicyKind {
609    /// Steps change only when driven programmatically; this binding surface
610    /// exposes no programmatic step advance, so the trip tracks progress
611    /// without changing steps.
612    Manual,
613    /// Advances when the position fix comes within the threshold of the end
614    /// of the current step, eager-advance semantics
615    /// (`DistanceToEndOfStepCondition`). Thresholds from
616    /// [crate::navigation::default_step_advance_policy] are the documented
617    /// defaults.
618    DistanceToEndOfStep,
619    /// Advances when the position fix moves farther from the current step's
620    /// polyline than the threshold, delayed-advance semantics
621    /// (`DistanceFromStepCondition`).
622    DistanceFromStep,
623}
624
625/// The step-advance policy of a trip: mirrors the native decision objects
626/// that cannot cross the FFI as one tagged record. The `Manual` kind ignores
627/// the threshold fields; both distance kinds require finite, non-negative
628/// threshold values (refused otherwise at the controller boundary).
629#[derive(Debug, Clone, Copy, PartialEq)]
630pub struct StepAdvancePolicy {
631    /// Which step-advance strategy the policy configures.
632    pub policy_kind: StepAdvancePolicyKind,
633    /// Distance, in meters, of the advance threshold (the distance to the
634    /// step's end, or from the step's polyline, depending on the kind).
635    pub advance_threshold_meters: f64,
636    /// Fixes with a reported horizontal accuracy worse than this, in meters,
637    /// never advance.
638    pub minimum_horizontal_accuracy_meters: f64,
639}

641/// Which deviation-detection strategy a deviation policy configures.
642#[derive(Debug, Clone, Copy, PartialEq, Eq)]
643pub enum DeviationPolicyKind {
644    /// Flags a deviation when the raw fix is farther from the remaining route
645    /// line than the configured threshold, widened by the fix's reported
646    /// accuracy (`DistanceDeviationDetector`).
647    DistanceThreshold,
648}
649
650/// The deviation-detection policy of a trip: mirrors the native decision
651/// object that cannot cross the FFI. The threshold must be finite and
652/// non-negative (refused otherwise at the controller boundary).
653#[derive(Debug, Clone, Copy, PartialEq)]
654pub struct DeviationPolicy {
655    /// Which deviation-detection strategy the policy configures.
656    pub policy_kind: DeviationPolicyKind,
657    /// Distance, in meters, from the remaining route line beyond which a fix
658    /// is judged off route (before any accuracy widening, strictly greater).
659    pub deviation_threshold_meters: f64,
660}
661
662/// The reroute-recalculation hysteresis policy of a controller: mirrors the
663/// native decision values of
664/// `rustic_maps_navigation::ReroutePolicy`. Both thresholds must be finite and
665/// non-negative (refused otherwise at the controller boundary).
666#[derive(Debug, Clone, Copy, PartialEq)]
667pub struct ReroutePolicy {
668    /// Minimum wall-clock seconds between the beginnings of two reroute
669    /// calculations.
670    pub recalculation_cooldown_seconds: f64,
671    /// Minimum displacement, in meters, between the raw position fix current
672    /// at a calculation's beginning and the one current at the previous
673    /// calculation's beginning.
674    pub recalculation_movement_meters: f64,
675}

677/// What triggered a speech-instruction line (the speech-instruction topic).
678#[derive(Debug, Clone, Copy, PartialEq, Eq)]
679pub enum SpeechInstructionKind {
680    /// A trigger band was crossed this fix (the far, moderate or near band).
681    ManeuverApproach,
682    /// The immediate band: the maneuver is imminent.
683    ImmediateManeuver,
684    /// Guidance just recovered from off-route; guidance resumed.
685    OffRouteRecovery,
686    /// The trip arrived.
687    Arrival,
688}
689
690/// One speech-trigger line the navigation controller's pacing policy judged
691/// newsworthy; the speech text is the composed payload's verbatim channel.
692#[derive(Debug, Clone, PartialEq)]
693pub struct SpeechInstruction {
694    /// The verbatim composed speech text (never rewritten by the trigger
695    /// policy).
696    pub utterance_speech_text: String,
697    /// The live distance to the maneuver, in meters, at emission time.
698    pub trigger_distance_meters: f64,
699    /// The utterance identity (a controller-ordered integer, monotonic across
700    /// the controller's lifetime): a stable monotonic
701    /// id composed at emission — strictly increasing over every line the
702    /// composing planner's owning controller has emitted since construction,
703    /// never reused, and stable for the single delivery of one line.
704    /// Identical text re-emitted is a new id; dedup and flush decisions stay
705    /// at the platform speaker.
706    pub utterance_id: u64,
707    /// What triggered this line.
708    pub instruction_kind: SpeechInstructionKind,
709}
710
711impl SpeechInstruction {
712    /// Builds the mirror record from the native utterance.
713    pub(crate) fn from_native_speech_utterance(
714        native_utterance: &rustic_maps_navigation::SpeechUtterance,
715    ) -> Self {
716        Self {
717            utterance_speech_text: native_utterance.utterance_speech_text.clone(),
718            trigger_distance_meters: native_utterance.trigger_distance_meters,
719            utterance_id: native_utterance.utterance_id,
720            instruction_kind: match native_utterance.kind {
721                rustic_maps_navigation::SpeechUtteranceKind::ManeuverApproach => {
722                    SpeechInstructionKind::ManeuverApproach
723                }
724                rustic_maps_navigation::SpeechUtteranceKind::ImmediateManeuver => {
725                    SpeechInstructionKind::ImmediateManeuver
726                }
727                rustic_maps_navigation::SpeechUtteranceKind::OffRouteRecovery => {
728                    SpeechInstructionKind::OffRouteRecovery
729                }
730                rustic_maps_navigation::SpeechUtteranceKind::Arrival => {
731                    SpeechInstructionKind::Arrival
732                }
733            },
734        }
735    }
