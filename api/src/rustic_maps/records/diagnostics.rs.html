6/// One presented frame's tile and overlay accounting (`MapView::draw`'s
7/// return), mirroring `rustic-maps-rendering`'s per-frame statistics: every requested
8/// tile lands in exactly one readiness count, and drawn pixels come from the
9/// present and fallback tiles while the deferred and unavailable ones are
10/// where later frames may still change pixels.
11#[derive(Debug, Clone, Copy, PartialEq, Eq)]
12pub struct FrameStatistics {
13    /// Tile requests the frame resolved (per request; duplicates in a
14    /// request listing cost one fetch but report per request).
15    pub requested_tile_count: u32,
16    /// Tiles drawn at their own zoom level.
17    pub present_tile_count: u32,
18    /// Tiles drawn from their nearest pooled ancestor.
19    pub fallback_tile_count: u32,
20    /// Tiles whose fetch was deferred by the per-frame budget.
21    pub deferred_tile_count: u32,
22    /// Tiles the provider rejected this frame.
23    pub unavailable_tile_count: u32,
24    /// Clip-space triangles the overlay layer contributed to the frame.
25    pub overlay_triangle_count: u32,
26}
27
28impl FrameStatistics {
29    /// Builds the mirror record from the native frame statistics.
30    #[cfg(not(target_arch = "wasm32"))]
31    pub(crate) fn from_native_frame_statistics(
32        native_frame_statistics: &SurfaceFrameStatistics,
33    ) -> Self {
34        Self {
35            requested_tile_count: native_frame_statistics.requested_tile_count,
36            present_tile_count: native_frame_statistics.present_tile_count,
37            fallback_tile_count: native_frame_statistics.fallback_tile_count,
38            deferred_tile_count: native_frame_statistics.deferred_tile_count,
39            unavailable_tile_count: native_frame_statistics.unavailable_tile_count,
40            overlay_triangle_count: native_frame_statistics.overlay_triangle_count,
41        }
42    }
43}

45/// The serving traffic-flow cache's counters beside the budget they obey
46/// (`MapSdk::traffic_flow_cache_statistics`'s return; see
47/// `rustic_maps.udl`'s `TrafficFlowCacheStatistics` for the field contract).
48/// The hits and misses prove a repeated pane was answered without asking the
49/// provider, and `maximum_entry_count` bounds how many cells the cache may
50/// hold at once.
51#[derive(Debug, Clone, Copy, PartialEq, Eq)]
52pub struct TrafficFlowCacheStatistics {
53    /// Requests answered entirely from fresh cells.
54    pub hit_count: u64,
55    /// Requests that had to ask the provider for at least one cell.
56    pub miss_count: u64,
57    /// Cells currently held.
58    pub entry_count: u64,
59    /// Cells dropped to stay inside the entry budget.
60    pub eviction_count: u64,
61    /// Flow segments currently held across every cell.
62    pub segment_count: u64,
63    /// Zoom level the cells are keyed at.
64    pub grid_zoom_level: u32,
65    /// Cells the cache may hold before the least recently used is evicted.
66    pub maximum_entry_count: u64,
67    /// Seconds a stored cell stays fresh.
68    pub time_to_live_seconds: u64,
69}
70
71impl From<rustic_maps_core::TrafficFlowCacheReadback> for TrafficFlowCacheStatistics {
72    fn from(readback: rustic_maps_core::TrafficFlowCacheReadback) -> Self {
73        Self {
74            hit_count: readback.statistics.hit_count,
75            miss_count: readback.statistics.miss_count,
76            entry_count: u64::try_from(readback.statistics.entry_count).unwrap_or(u64::MAX),
77            eviction_count: readback.statistics.eviction_count,
78            segment_count: u64::try_from(readback.statistics.segment_count).unwrap_or(u64::MAX),
79            grid_zoom_level: u32::from(readback.configuration.grid_zoom_level),
80            maximum_entry_count: u64::try_from(readback.configuration.maximum_entry_count)
81                .unwrap_or(u64::MAX),
82            time_to_live_seconds: readback.configuration.time_to_live_seconds,
83        }
84    }
85}

87/// The installed tile payload cache's exact counters and occupancy
88/// (`MapSdk::tile_cache_statistics`'s return; see `rustic_maps.udl`'s
89/// `TileCacheStatistics` for the field contract).
90///
91/// Both the raster base and the vector base fetch through the same installed
92/// store, so one counter pair covers every payload lookup: a hit was answered
93/// without a provider request, and a miss went to the provider (or degraded to
94/// one after a store read failure).
95#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
96pub struct TileCacheStatistics {
97    /// Payload lookups the store answered without a provider request.
98    pub hit_count: u64,
99    /// Payload lookups the store could not answer.
100    pub miss_count: u64,
101    /// Bytes currently stored, or `None` when the store's occupancy could not
102    /// be read.
103    pub stored_byte_count: Option<u64>,
104}
105

111            miss_count: statistics.miss_count,
112            stored_byte_count: None,
113        }
114    }

117/// The effective raster-tile caching terms of the installed cache
118/// (`MapSdk::raster_tile_cache_terms`'s return), shaped so a developer can
119/// display the caching posture: the category default, the provider's terms
120/// ceiling and the value currently in force (see
121/// `rustic_maps.udl`'s `TileCacheTerms` for the field contract).
122#[derive(Debug, Clone, Copy, PartialEq, Eq)]
123pub struct TileCacheTerms {
124    /// Time-to-live currently in force on the installed cache, in seconds:
125    /// the category default, or the last accepted override (clamped to the
126    /// ceiling when the override exceeded it).
127    pub effective_time_to_live_seconds: u64,
128    /// The installed provider's declared caching-terms ceiling in seconds,
129    /// or `None` when the provider declares no ceiling — the requested
130    /// time-to-live then applies verbatim. For the MapQuest raster tiles
131    /// this is the documented terms ceiling (thirty calendar days).
132    pub provider_terms_ceiling_seconds: Option<u64>,
133    /// The cache category's default time-to-live for raster tiles, in
134    /// seconds (thirty days, the imagery provider's declared ceiling); the
135    /// value a freshly registered cache starts at.
136    pub default_time_to_live_seconds: u64,
137}

139/// Severity of an SDK diagnostic event, mirroring the `tracing` levels the
140/// crates log under.
141#[derive(Debug, Clone, Copy, PartialEq, Eq)]
142pub enum SdkLogSeverity {
143    /// Very fine-grained frame-path instrumentation.
144    Trace,
145    /// Fine-grained internal state transitions.
146    Debug,
147    /// Coarse-grained progress.
148    Info,
149    /// A problem this event explains.
150    Warn,
151    /// A failure this event explains.
152    Error,
153}

155/// One completed provider request, as a diagnostics event reports it.
156///
157/// The record is already redacted: the request path keeps its structure with
158/// query values removed, and no credential or response body is carried.
159#[derive(Debug, Clone, PartialEq, Eq)]
160pub struct ProviderRequestTrace {
161    /// The registry key of the provider that served the call.
162    pub provider_key: String,
163    /// The contract the call served, for example `RoutingProvider`.
164    pub provider_contract: String,
165    /// The HTTP method as an uppercase word (`GET`, `POST`).
166    pub http_method: String,
167    /// The redacted request path, with query names and no values.
168    pub request_path: String,
169    /// `2xx`, `3xx`, `4xx`, `5xx` or the transport-failure marker.
170    pub status_class: String,
171    /// The derived outcome word (`success`, `client-error`, …).
172    pub outcome: String,
173    /// Wall-clock duration of the attempt in milliseconds.
174    pub duration_milliseconds: u64,
175    /// Response payload size in bytes; zero when no response arrived.
176    pub response_bytes: u64,
177    /// The redacted response prefix captured behind the excerpt switch,
178    /// capped at 512 bytes.
179    pub response_excerpt: Option<String>,
180}

182/// One structured SDK internal log event, as delivered to a foreign
183/// diagnostics listener or read back by a flush. Coordinates in event messages
184/// are rounded at the sink; the API key never appears in a message.
185#[derive(Debug, Clone, PartialEq)]
186pub struct LogEvent {
187    /// Wall-clock milliseconds since the Unix epoch, as the ring recorded
188    /// the event.
189    pub timestamp_milliseconds_since_epoch: u64,
190    /// The event's severity.
191    pub severity: SdkLogSeverity,
192    /// The named per-subsystem target that logged it.
193    pub subsystem_target: String,
194    /// The one-line already-scrubbed message.
195    pub message_line: String,
196    /// The typed request facts when this event reports a completed provider
197    /// call; `None` for every other event.
198    pub provider_request_trace: Option<ProviderRequestTrace>,
199}

201/// Which shape of foreign provider failure a diagnostic captured. The two
202/// shapes follow the two ways a foreign implementation of a
203/// `[Trait, WithForeign]` provider contract can fail: typed (it threw one of
204/// the contract's `SdkError` variants) or unclassified (it failed without a
205/// mapped error, the round-11 conversion's subject).
206#[derive(Debug, Clone, Copy, PartialEq, Eq)]
207pub enum ProviderFailureClassification {
208    /// The foreign implementation failed without a mapped error (an
209    /// exception the contract does not spell): the failure text is the
210    /// uniffi callback machinery's verbatim reason for it.
211    UnclassifiedCallbackFailure,
212    /// The foreign implementation threw one of the contract's `SdkError`
213    /// variants, which round-trips typed through the seam: the failure text
214    /// is the thrown binding error's verbatim display form.
215    MappedContractError,
216}

218/// One failed foreign-callback provider call, as retained in
219/// `MapSdk::latest_provider_failure_diagnostics`'s bounded per-object ring
220/// (the provider-failure diagnostic seam). Every
221/// field is knowable at the adapter boundary; no data is invented beyond it
222/// (see the UDL documentation for the classification limitation).
223#[derive(Debug, Clone, PartialEq, Eq)]
224pub struct ProviderFailureDiagnostic {
225    /// Wall-clock milliseconds since the Unix epoch, as the seam recorded
226    /// the failure.
227    pub timestamp_milliseconds_since_epoch: u64,
228    /// Which failure shape this entry captured.
229    pub failure_classification: ProviderFailureClassification,
230    /// The provider contract the foreign code implements, e.g.
231    /// `GeocodingProvider`.
232    pub provider_contract_name: String,
233    /// The contract method whose call failed, e.g. `forward_geocode`.
234    pub provider_method_name: String,
235    /// The verbatim failure text: the uniffi callback machinery's reason for
236    /// the unclassified shape, or the thrown binding error's display form
237    /// for the mapped shape.
238    pub failure_detail: String,
239}
