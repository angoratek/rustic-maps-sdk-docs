8/// Lowest valid overlay color channel (a channel is also refused when NaN,
9/// outside the closed range by construction). Native-only with the surface
10/// view's overlay converter.
11#[cfg(not(target_arch = "wasm32"))]
12const OVERLAY_MINIMUM_COLOR_CHANNEL: f64 = 0.0;
13/// Highest valid overlay color channel. Native-only with the surface view's
14/// overlay converter.
15#[cfg(not(target_arch = "wasm32"))]
16const OVERLAY_MAXIMUM_COLOR_CHANNEL: f64 = 1.0;
17
18/// Solid RGBA overlay style, every channel in `[0, 1]`; the rustic-maps-rendering
19/// pipelines consume `f32` channels, converted in the constructor.
20#[derive(Debug, Clone, Copy, PartialEq)]
21pub struct OverlayColor {
22    /// Red channel in `[0, 1]`.
23    pub red: f64,
24    /// Green channel in `[0, 1]`.
25    pub green: f64,
26    /// Blue channel in `[0, 1]`.
27    pub blue: f64,
28    /// Alpha channel in `[0, 1]` (the overlay pipeline alpha-blends).
29    pub alpha: f64,
30}
31
32impl OverlayColor {
33    /// Builds the native `f32` RGBA style array from the mirror record,
34    /// refusing any channel outside the documented range. Native-only with
35    /// the surface view's overlay converter.
36    #[cfg(not(target_arch = "wasm32"))]
37    pub(crate) fn to_native_rgba(self, style_description: &str) -> Result<[f32; 4], SdkError> {
38        for (channel_name, channel_value) in [
39            ("red", self.red),
40            ("green", self.green),
41            ("blue", self.blue),
42            ("alpha", self.alpha),
43        ] {
44            if !((OVERLAY_MINIMUM_COLOR_CHANNEL..=OVERLAY_MAXIMUM_COLOR_CHANNEL)
45                .contains(&channel_value))
46            {
47                return Err(SdkError::InvalidRenderInput {
48                    detail: format!(
49                        "{style_description} color channel {channel_name} \
50                         {channel_value} outside [0, 1]"
51                    ),
52                });
53            }
54        }
55        Ok([
56            self.red as f32,
57            self.green as f32,
58            self.blue as f32,
59            self.alpha as f32,
60        ])
61    }
62}

64/// Which overlay shape one descriptor configures — the kind tag of the
65/// wasm canvas' plain-object overlay names.
66#[derive(Debug, Clone, Copy, PartialEq, Eq)]
67pub enum MapOverlayKind {
68    /// A routed path drawn as a thick line.
69    RoutePolyline,
70    /// A filled polygon: one outer ring, ear-clip triangulated.
71    FilledPolygon,
72    /// Screen-facing point markers (octagon fans).
73    PointMarkers,
74    /// Congestion-colored traffic-flow lines.
75    TrafficFlow,
76    /// Screen-facing teardrop pins anchored at their tip.
77    PointPins,
78}
79
80/// One overlay shape drawn over the map surface's tiles in every frame,
81/// selected by `kind`. Each kind reads exactly its own payload fields and
82/// ignores the others (the wasm canvas carries the same tolerance); the
83/// kind-required payload must be present.
84#[derive(Debug, Clone, PartialEq)]
85pub struct MapOverlay {
86    /// Which overlay shape this descriptor configures and which payload
87    /// fields carry it.
88    pub kind: MapOverlayKind,
89    /// Route polyline: the traversal-order polyline. Required for the kind.
90    pub coordinates: Option<Vec<GeographicCoordinate>>,
91    /// Route polyline, filled polygon and point markers: solid color
92    /// override, or `None` for the kind's documented default.
93    pub rgba_color: Option<OverlayColor>,
94    /// Route polyline and traffic flow: line width in screen pixels
95    /// (positive and finite), or `None` for the documented default.
96    pub width_pixels: Option<f64>,
97    /// Filled polygon: the outer ring without a repeated closing vertex.
98    /// Required for the kind.
99    pub ring_coordinates: Option<Vec<GeographicCoordinate>>,
100    /// Point markers and point pins: the geographic anchor of every marker
101    /// (a pin's anchor is its tip). Required for both kinds.
102    pub point_marker_positions: Option<Vec<GeographicCoordinate>>,
103    /// Point markers and point pins: marker or head radius in screen pixels
104    /// (positive and finite), or `None` for the documented default.
105    pub point_marker_radius_pixels: Option<f64>,
106    /// Traffic flow: the flow observations to paint. Required for the kind.
107    pub traffic_flow_segments: Option<Vec<TrafficFlowData>>,
108}
109
110impl MapOverlay {
111    /// Validates a screen-pixel size field (a line width or marker radius):
112    /// it must be positive and finite, like the wasm canvas' validation of
113    /// the same fields. Native-only with the surface view's overlay
114    /// converter.
115    #[cfg(not(target_arch = "wasm32"))]
116    pub(crate) fn validated_size_pixels(
117        size_pixels: f64,
118        size_description: &str,
119    ) -> Result<f64, SdkError> {
120        if !size_pixels.is_finite() || size_pixels.is_sign_negative() || size_pixels == 0.0 {
121            return Err(SdkError::InvalidRenderInput {
122                detail: format!("{size_description} must be positive and finite"),
123            });
124        }
125        Ok(size_pixels)
126    }
127
128    /// Builds the native overlay from the descriptor, refusing an
129    /// incompatible kind/payload pair or an unusable style field. A `None`
130    /// style field keeps the documented rustic-maps-rendering default of the kind.
131    /// Native-only with the surface view that draws it.
132    #[cfg(not(target_arch = "wasm32"))]
133    pub(crate) fn into_native_overlay(self) -> Result<rustic_maps_rendering::MapOverlay, SdkError> {
134        match self.kind {
135            MapOverlayKind::RoutePolyline => {
136                let mut route_polyline = rustic_maps_rendering::RoutePolylineOverlay::new(
137                    validated_coordinates(self.coordinates, "route polyline")?,
138                );
139                if let Some(rgba_color) = self.rgba_color {
140                    route_polyline.rgba_color = rgba_color.to_native_rgba("route polyline")?;
141                }
142                if let Some(width_pixels) = self.width_pixels {
143                    route_polyline.width_pixels =
144                        Self::validated_size_pixels(width_pixels, "route polyline width")?;
145                }
146                Ok(rustic_maps_rendering::MapOverlay::RoutePolyline(
147                    route_polyline,
148                ))
149            }
150            MapOverlayKind::FilledPolygon => {
151                let mut filled_polygon = rustic_maps_rendering::FilledPolygonOverlay::new(
152                    validated_coordinates(self.ring_coordinates, "filled polygon ring")?,
153                );
154                if let Some(rgba_color) = self.rgba_color {
155                    filled_polygon.rgba_color = rgba_color.to_native_rgba("filled polygon")?;
156                }
157                Ok(rustic_maps_rendering::MapOverlay::FilledPolygon(
158                    filled_polygon,
159                ))
160            }
161            MapOverlayKind::PointMarkers => {
162                let mut point_markers = rustic_maps_rendering::PointMarkersOverlay::new(
163                    validated_coordinates(self.point_marker_positions, "point markers")?,
164                );
165                if let Some(rgba_color) = self.rgba_color {
166                    point_markers.rgba_color = rgba_color.to_native_rgba("point markers")?;
167                }
168                if let Some(radius_pixels) = self.point_marker_radius_pixels {
169                    point_markers.radius_pixels =
170                        Self::validated_size_pixels(radius_pixels, "point marker radius")?;
171                }
172                Ok(rustic_maps_rendering::MapOverlay::PointMarkers(
173                    point_markers,
174                ))
175            }
176            MapOverlayKind::PointPins => {
177                let mut point_pins = rustic_maps_rendering::PointPinsOverlay::new(
178                    validated_coordinates(self.point_marker_positions, "point pins")?,
179                );
180                if let Some(rgba_color) = self.rgba_color {
181                    point_pins.rgba_color = rgba_color.to_native_rgba("point pins")?;
182                }
183                if let Some(radius_pixels) = self.point_marker_radius_pixels {
184                    point_pins.radius_pixels =
185                        Self::validated_size_pixels(radius_pixels, "point pin radius")?;
186                }
187                Ok(rustic_maps_rendering::MapOverlay::PointPins(point_pins))
188            }
189            MapOverlayKind::TrafficFlow => {
190                let mut traffic_flow = rustic_maps_rendering::TrafficFlowOverlay::new(
191                    self.traffic_flow_segments
192                        .ok_or(SdkError::InvalidRenderInput {
193                            detail: "traffic flow descriptor needs traffic_flow_segments"
194                                .to_string(),
195                        })?
196                        .into_iter()
197                        .map(|traffic_flow_segment| {
198                            traffic_flow_segment.into_native_traffic_flow_data()
199                        })
200                        .collect::<Result<Vec<_>, SdkError>>()?,
201                );
202                if let Some(width_pixels) = self.width_pixels {
203                    traffic_flow.width_pixels =
204                        Self::validated_size_pixels(width_pixels, "traffic flow width")?;
205                }
206                Ok(rustic_maps_rendering::MapOverlay::TrafficFlow(traffic_flow))
207            }
208        }
209    }
210}

212/// Converts the kind-required coordinate payload: missing payload is
213/// refused, and present payload validates every coordinate the overlay
214/// seam validates — the same canonical-degree boundary the projection
215/// surface applies (`SdkError::InvalidCoordinateInput` for an out-of-range
216/// latitude or longitude, matching the wasm canvas' overlay validation).
217/// Native-only with the surface view's overlay converter.
218#[cfg(not(target_arch = "wasm32"))]
219fn validated_coordinates(
220    coordinates: Option<Vec<GeographicCoordinate>>,
221    payload_description: &str,
222) -> Result<Vec<geospatial_types::GeographicCoordinate>, SdkError> {
223    let mirror_coordinates = coordinates.ok_or(SdkError::InvalidRenderInput {
224        detail: format!("{payload_description} descriptor needs its coordinates"),
225    })?;
226    mirror_coordinates
227        .into_iter()
228        .map(crate::projection::validated_coordinate)
229        .collect()
230}

232/// The platform window surface handle of `MapView::attach`: exactly
233/// one platform pointer is set. Pointers lower as `u64` (Kotlin `ULong`,
234/// Swift `UInt64`), where a plain `u64 0` stands in for the foreign
235/// languages' null pointer without ever dereferencing one.
236#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
237pub struct SurfaceHandle {
238    /// macOS: pointer to the `NSView` the surface renders into.
239    pub appkit_view_pointer: Option<u64>,
240    /// iOS: pointer to the `UIView` the surface renders into.
241    pub ios_view_pointer: Option<u64>,
242    /// Android: pointer to the `ANativeWindow` obtained from the platform
243    /// `Surface`.
244    pub android_native_window_pointer: Option<u64>,
245}

247/// Which visual form the renderer-owned location layer draws.

249pub enum LocationIndicatorVisualKind {
250    /// The soft tracking dot (no course known or no route): accent core,

259/// Style of the location layer; `None` fields keep the documented
260/// `rustic-maps-rendering` defaults (the overlay-style pattern).
261#[derive(Debug, Clone, Copy, PartialEq)]
262pub struct LocationIndicatorStyle {
263    /// Which visual form this layer draws.
264    pub visual_kind: LocationIndicatorVisualKind,
265    /// Accent-core radius in screen pixels; `None` keeps the default.
266    pub radius_pixels: Option<f64>,
267    /// Accent fill color of the core disc and navigation dart; `None` keeps
268    /// the default.
269    pub rgba_color: Option<OverlayColor>,
270    /// Ring color: the ring around the puck and the navigation dart's
271    /// outline; `None` keeps the default white.
272    pub ring_rgba_color: Option<OverlayColor>,
273    /// Halo color drawn under the puck; `None` keeps the default.
274    pub halo_rgba_color: Option<OverlayColor>,
275    /// Accuracy ring drawn around the puck; `None` keeps the default.
276    pub show_accuracy_ring: Option<bool>,
277}

279/// One frame's location-layer state (the presentable subset of a position
280/// fix): position, optional course over ground, optional horizontal accuracy.
281#[derive(Debug, Clone, Copy, PartialEq)]
282pub struct LocationIndicatorState {
283    /// Fix position.
284    pub coordinate: GeographicCoordinate,
285    /// Course over ground in degrees clockwise from true north; `None` when
286    /// the source does not report one.
287    pub course_heading_degrees: Option<f64>,
288    /// Horizontal accuracy radius in meters; the accuracy ring renders when
289    /// set and the style shows rings.
290    pub horizontal_accuracy_meters: Option<f64>,
291}

293impl LocationIndicatorVisualKind {
294    /// Builds the native visual kind tag from the mirror record.
295    /// Native-only with the surface view that draws it.
296    #[cfg(not(target_arch = "wasm32"))]
297    pub(crate) fn to_native_visual_kind(
298        self,
299    ) -> rustic_maps_rendering::LocationIndicatorVisualKind {
300        match self {
301            Self::TrackingDot => rustic_maps_rendering::LocationIndicatorVisualKind::TrackingDot,
302            Self::GuidedCourseArrow => {
303                rustic_maps_rendering::LocationIndicatorVisualKind::GuidedCourseArrow
304            }
305        }
306    }

309impl LocationIndicatorState {
310    /// Validates the fix and builds the native location-layer state: the
311    /// coordinate must be canonical degrees (the same boundary the overlay
312    /// seam applies), while the optional scalar fields pass through
313    /// verbatim (the renderer's clamps own their ranges). Native-only with
314    /// the surface view that draws it.
315    #[cfg(not(target_arch = "wasm32"))]
316    pub(crate) fn into_native_location_indicator_state(
317        self,
318    ) -> Result<rustic_maps_rendering::LocationIndicatorState, SdkError> {
319        Ok(rustic_maps_rendering::LocationIndicatorState {
320            coordinate: crate::projection::validated_coordinate(self.coordinate)?,
321            course_heading_degrees: self.course_heading_degrees,
322            horizontal_accuracy_meters: self.horizontal_accuracy_meters,
323        })
324    }
325}

327impl LocationIndicatorStyle {
328    /// Validates the numeric style fields and builds the native style; the
329    /// `None` fields keep the documented rustic-maps-rendering defaults (mirroring
330    /// the overlay descriptor's style validation: a radius must be positive
331    /// and finite, a color channel inside `[0, 1]`). Native-only with the
332    /// surface view that draws it.
333    #[cfg(not(target_arch = "wasm32"))]
334    pub(crate) fn into_native_location_indicator_style(
335        self,
336    ) -> Result<rustic_maps_rendering::LocationIndicatorStyle, SdkError> {
337        if let Some(radius_pixels) = self.radius_pixels {
338            MapOverlay::validated_size_pixels(radius_pixels, "location indicator disc radius")?;
339        }
340        let rgba_color = match self.rgba_color {
341            Some(rgba_color) => Some(rgba_color.to_native_rgba("location indicator")?),
342            None => None,
343        };
344        let ring_rgba_color = match self.ring_rgba_color {
345            Some(ring_rgba_color) => {
346                Some(ring_rgba_color.to_native_rgba("location indicator ring")?)
347            }
348            None => None,
349        };
350        let halo_rgba_color = match self.halo_rgba_color {
351            Some(halo_rgba_color) => {
352                Some(halo_rgba_color.to_native_rgba("location indicator halo")?)
353            }
354            None => None,
355        };
356        Ok(rustic_maps_rendering::LocationIndicatorStyle {
357            visual_kind: self.visual_kind.to_native_visual_kind(),
358            radius_pixels: self.radius_pixels,
359            rgba_color,
360            ring_rgba_color,
361            halo_rgba_color,
362            show_accuracy_ring: self.show_accuracy_ring,
363        })
364    }
