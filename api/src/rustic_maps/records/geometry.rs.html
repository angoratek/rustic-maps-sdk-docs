5/// A position on the WGS 84 ellipsoid in decimal degrees.
6#[derive(Debug, Clone, Copy, PartialEq)]
7pub struct GeographicCoordinate {
8    /// Latitude in decimal degrees, valid within `[-90, 90]`.
9    pub latitude: f64,
10    /// Longitude in decimal degrees, valid within `[-180, 180]`.
11    pub longitude: f64,
12}
13
14impl GeographicCoordinate {
15    /// Builds the mirror record from a validated native coordinate.
16    pub(crate) fn from_native_coordinate(
17        native_coordinate: &geospatial_types::GeographicCoordinate,
18    ) -> Self {
19        Self {
20            latitude: native_coordinate.latitude,
21            longitude: native_coordinate.longitude,
22        }
23    }
24
25    /// Builds the native coordinate from the mirror record. The mirror never
26    /// carries canonical-degree guarantees (foreign implementations may hand
27    /// any value across), so this is a plain field copy without validation;
28    /// the call paths that need canonical degrees validate on their own
29    /// boundary.
30    pub(crate) fn to_native_coordinate(self) -> geospatial_types::GeographicCoordinate {
31        geospatial_types::GeographicCoordinate {
32            latitude: self.latitude,
33            longitude: self.longitude,
34        }
35    }
36}

38/// Rectangular area between a south-west and a north-east corner; boxes
39/// never span the antimeridian (encoded as two adjacent boxes instead).
40#[derive(Debug, Clone, Copy, PartialEq)]
41pub struct BoundingBox {
42    /// South-west corner (minimum latitude and longitude).
43    pub south_west: GeographicCoordinate,
44    /// North-east corner (maximum latitude and longitude).
45    pub north_east: GeographicCoordinate,
46}
47
48impl BoundingBox {
49    /// Builds the mirror record from a native bounding box.
50    pub(crate) fn from_native_bounding_box(
51        native_bounding_box: &geospatial_types::BoundingBox,
52    ) -> Self {
53        Self {
54            south_west: GeographicCoordinate::from_native_coordinate(
55                &native_bounding_box.south_west,
56            ),
57            north_east: GeographicCoordinate::from_native_coordinate(
58                &native_bounding_box.north_east,
59            ),
60        }
61    }
62
63    /// Builds the native bounding box from the mirror record. As with the
64    /// coordinate converters this is a plain field copy without box
65    /// validation; the call paths needing canonical boxes validate on their
66    /// own boundary.
67    pub(crate) fn to_native_bounding_box(self) -> geospatial_types::BoundingBox {
68        geospatial_types::BoundingBox {
69            south_west: self.south_west.to_native_coordinate(),
70            north_east: self.north_east.to_native_coordinate(),
71        }
72    }
73}

77pub struct ViewportSize {

84/// Screen-space insets kept clear of map content: distances from each
85/// viewport edge in pixels, so UI overlays — route cards, attribution, sheet
86/// panels — never cover the content the camera guarantees visible.
87#[derive(Debug, Clone, Copy)]
88pub struct EdgeInsets {
89    /// Pixels kept clear from the left viewport edge.
90    pub left: f64,
91    /// Pixels kept clear from the top viewport edge.
92    pub top: f64,
93    /// Pixels kept clear from the right viewport edge.
94    pub right: f64,
95    /// Pixels kept clear from the bottom viewport edge.
96    pub bottom: f64,
97}
98
99impl EdgeInsets {
100    /// Builds the native inset set from the mirror record, for the native
101    /// `fit_bounds` constructor.
102    pub(crate) fn to_native_edge_insets(self) -> rustic_maps_rendering::EdgeInsetsPixels {
103        rustic_maps_rendering::EdgeInsetsPixels {
104            left: self.left,
105            top: self.top,
106            right: self.right,
107            bottom: self.bottom,
108        }
109    }
110}

112/// Position on the viewport in physical pixels (origin top-left, y down).
113#[derive(Debug, Clone, Copy)]
114pub struct ScreenPoint {
115    /// Horizontal position in pixels.
116    pub x: f64,
117    /// Vertical position in pixels.
118    pub y: f64,
119}
120
121impl ScreenPoint {
122    /// Builds a screen point from the native projection's two-component
123    /// result in viewport pixels.
124    pub(crate) fn from_screen_pixels(screen_pixels: [f64; 2]) -> Self {
125        Self {
126            x: screen_pixels[0],
127            y: screen_pixels[1],
128        }
129    }
130}

132/// A world position in normalized world coordinates, both axes in `[0, 1]`
133/// (x eastward from the antimeridian, y southward from the north pole).
134#[derive(Debug, Clone, Copy)]
135pub struct NormalizedWorldCoordinate {
136    /// Eastward position from the antimeridian.
137    pub x: f64,
138    /// Southward position from the north pole.
139    pub y: f64,
140}
141
142impl NormalizedWorldCoordinate {
143    /// Builds a normalized world coordinate from the native projection's
144    /// two-component result.
145    pub(crate) fn from_normalized_world_components(normalized_world_components: [f64; 2]) -> Self {
146        Self {
147            x: normalized_world_components[0],
148            y: normalized_world_components[1],
149        }
150    }
151}

153/// One web-mercator XYZ tile address.
154#[derive(Debug, Clone, Copy, PartialEq, Eq)]
155pub struct TileKey {
156    /// Tile column, in `0..2^zoom_level`.
157    pub tile_x: u32,
158    /// Tile row, counted from the north, in `0..2^zoom_level`.
159    pub tile_y: u32,
160    /// Zoom level this tile belongs to.
161    pub zoom_level: u8,
162}
163
164impl TileKey {
165    /// Builds a tile key record from the native `TileKey`.
166    pub(crate) fn from_native_tile_key(tile_key: RenderingTileKey) -> Self {
167        Self {
168            tile_x: tile_key.tile_x,
169            tile_y: tile_key.tile_y,
170            zoom_level: tile_key.zoom_level,
171        }
172    }
