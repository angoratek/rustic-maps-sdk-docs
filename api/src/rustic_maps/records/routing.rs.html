6/// Mode of transport used to calculate a route.
7#[derive(Debug, Clone, Copy, PartialEq, Eq)]
8pub enum TransportationMode {
9    /// Private car routing.
10    Automobile,
11    /// Bicycle routing.
12    Bicycle,
13    /// Walking routing.
14    Pedestrian,
15}
16
17impl TransportationMode {
18    /// Builds the native transportation mode from the mirror enum.
19    pub(crate) fn to_native_transportation_mode(self) -> geospatial_types::TransportationMode {
20        match self {
21            Self::Automobile => geospatial_types::TransportationMode::Automobile,
22            Self::Bicycle => geospatial_types::TransportationMode::Bicycle,
23            Self::Pedestrian => geospatial_types::TransportationMode::Pedestrian,
24        }
25    }
26
27    /// Builds the mirror enum from the native transportation mode — the
28    /// foreign provider adapters' direction, handing a native routing request
29    /// to an implementation written in application code.
30    pub(crate) fn from_native_transportation_mode(
31        native_transportation_mode: geospatial_types::TransportationMode,
32    ) -> Self {
33        match native_transportation_mode {
34            geospatial_types::TransportationMode::Automobile => Self::Automobile,
35            geospatial_types::TransportationMode::Bicycle => Self::Bicycle,
36            geospatial_types::TransportationMode::Pedestrian => Self::Pedestrian,
37        }
38    }
39}

71/// What a routing request optimizes for; the strategy applies to automobile
72/// routing only (the current providers have no time-based cost model for the
73/// other modes).
74#[derive(Debug, Clone, Copy, PartialEq, Eq)]
75pub enum RouteStrategy {
76    /// Minimize expected travel time.
77    Fastest,
78    /// Minimize traveled distance.
79    Shortest,
80}
81
82impl RouteStrategy {
83    /// Builds the native route strategy from the mirror enum.
84    pub(crate) fn to_native_route_strategy(self) -> geospatial_types::route::RouteStrategy {
85        match self {
86            Self::Fastest => geospatial_types::route::RouteStrategy::Fastest,
87            Self::Shortest => geospatial_types::route::RouteStrategy::Shortest,
88        }
89    }
90
91    /// Builds the mirror enum from the native route strategy (the foreign
92    /// provider adapters' direction).
93    pub(crate) fn from_native_route_strategy(
94        native_route_strategy: geospatial_types::route::RouteStrategy,
95    ) -> Self {
96        match native_route_strategy {
97            geospatial_types::route::RouteStrategy::Fastest => Self::Fastest,
98            geospatial_types::route::RouteStrategy::Shortest => Self::Shortest,
99        }
100    }
101}

141/// Options controlling how a route is calculated.
142#[derive(Debug, Clone, Copy, PartialEq, Eq)]
143pub struct RouteOptions {
144    /// Mode of transport.
145    pub transportation_mode: TransportationMode,
146    /// Whether routing optimizes travel time or distance.
147    pub route_strategy: RouteStrategy,
148    /// Display unit convention the provider should annotate.
149    pub unit_system: UnitSystem,
150    /// Features to avoid.
151    pub avoid_options: AvoidOptions,
152    /// Departure instant as seconds since the Unix epoch; `None` for
153    /// "depart now".
154    pub departure_timestamp_seconds: Option<u64>,
155}
156
157impl RouteOptions {
158    /// Builds the native options from the mirror record, consuming it (the
159    /// record is a `Copy` value).
160    pub(crate) fn to_native_route_options(self) -> geospatial_types::RouteOptions {
161        geospatial_types::RouteOptions {
162            transportation_mode: self.transportation_mode.to_native_transportation_mode(),
163            route_strategy: self.route_strategy.to_native_route_strategy(),
164            unit_system: self.unit_system.to_native_unit_system(),
165            avoid_options: self.avoid_options.to_native_avoid_options(),
166            departure_timestamp_seconds: self.departure_timestamp_seconds,
167        }
168    }
169
170    /// Builds the mirror record from the native options — the foreign
171    /// provider adapters' direction, handing a native routing request to an
172    /// implementation written in application code.
173    pub(crate) fn from_native_route_options(
174        native_route_options: &geospatial_types::RouteOptions,
175    ) -> Self {
176        Self {
177            transportation_mode: TransportationMode::from_native_transportation_mode(
178                native_route_options.transportation_mode,
179            ),
180            route_strategy: RouteStrategy::from_native_route_strategy(
181                native_route_options.route_strategy,
182            ),
183            unit_system: UnitSystem::from_native_unit_system(native_route_options.unit_system),
184            avoid_options: AvoidOptions::from_native_avoid_options(
185                &native_route_options.avoid_options,
186            ),
187            departure_timestamp_seconds: native_route_options.departure_timestamp_seconds,
188        }
189    }
190}

192/// Which transport modes and strategies the serving routing provider can
193/// honour, as reported by `MapSdk::route_capabilities`.
194#[derive(Debug, Clone, PartialEq)]
195pub struct RouteCapabilities {
196    /// The transport modes the provider can route for.
197    pub supported_transportation_modes: Vec<TransportationMode>,
198    /// Whether the provider can minimize traveled distance for automobile
199    /// routing.
200    pub supports_shortest_route_strategy: bool,
201}
202
203impl RouteCapabilities {
204    /// Builds the mirror record from the native capability answer.
205    pub(crate) fn from_native_route_capabilities(
206        native_route_capabilities: &geospatial_types::route::RouteCapabilities,
207    ) -> Self {
208        Self {
209            supported_transportation_modes: native_route_capabilities
210                .supported_transportation_modes
211                .iter()
212                .map(|mode| TransportationMode::from_native_transportation_mode(*mode))
213                .collect(),
214            supports_shortest_route_strategy: native_route_capabilities
215                .supports_shortest_route_strategy,
216        }
217    }
218}

220/// One maneuver within a route leg. The native maneuver classification
221/// crosses as a stable snake-case text label (see
222/// [`maneuver_type_text`]) and this record carries no lane guidance.
223#[derive(Debug, Clone, PartialEq)]
224pub struct RouteStep {
225    /// Distance of this step in meters.
226    pub distance_meters: f64,
227    /// Travel time of this step in seconds.
228    pub duration_seconds: f64,
229    /// Maneuver classification as a stable snake-case text label; the
230    /// turn direction and the one-based roundabout exit index are carried
231    /// in the label (`turn-left`, `roundabout-2`).
232    pub maneuver_type: String,
233    /// Human-readable instruction in the display language.
234    pub instruction_display_text: String,
235    /// Instruction tuned for text-to-speech engines.
236    pub instruction_speech_text: String,
237    /// Step geometry from the previous step end to this maneuver point.
238    pub shape: Vec<GeographicCoordinate>,
239}
240
241/// Stable snake-case text labels of the native maneuver classification;
242/// turn directions and the roundabout exit index are embedded in the label,
243/// because the native classification carries payloads UDL enums cannot.
244///
245/// The spelling is owned by [`geospatial_types::ManeuverType::stable_label`]
246/// (the same single spelling the serde serialization of route records emits)
247/// — this converter delegates so the FFI surface and the serialized route
248/// records cannot drift apart.
249pub(crate) fn maneuver_type_text(native_maneuver_type: &geospatial_types::ManeuverType) -> String {
250    native_maneuver_type.stable_label()
251}
252
253/// Builds the native maneuver classification back from one of the stable
254/// snake-case labels [`maneuver_type_text`] emitted; the inverse of that
255/// labeling, needed because route records cross the FFI as labels. Unknown
256/// labels are refused at the caller's boundary (never guessed).
257pub(crate) fn native_maneuver_type_from_text(
258    maneuver_type_text: &str,
259) -> Result<geospatial_types::ManeuverType, SdkError> {
260    geospatial_types::ManeuverType::from_stable_label(maneuver_type_text).ok_or_else(|| {
261        SdkError::InvalidNavigationInput {
262            detail: format!("unrecognized step maneuver label `{maneuver_type_text}`"),
263        }
264    })
265}
266
267impl RouteStep {
268    /// Builds the mirror record from a native route step.
269    pub(crate) fn from_native_route_step(native_route_step: &geospatial_types::RouteStep) -> Self {
270        Self {
271            distance_meters: native_route_step.distance_meters,
272            duration_seconds: native_route_step.duration_seconds,
273            maneuver_type: maneuver_type_text(&native_route_step.maneuver_type),
274            instruction_display_text: native_route_step.instruction_display_text.clone(),
275            instruction_speech_text: native_route_step.instruction_speech_text.clone(),
276            shape: native_route_step
277                .shape
278                .iter()
279                .map(GeographicCoordinate::from_native_coordinate)
280                .collect(),
281        }
282    }
283
284    /// Builds the native route step from the mirror record, refusing a
285    /// maneuver label the converter cannot map. Lane guidance is not part of
286    /// this surface (see [`Self::from_native_route_step`]), so steps rebuilt
287    /// across the FFI arrive without it — mirroring the forward direction.
288    pub(crate) fn into_native_route_step(self) -> Result<geospatial_types::RouteStep, SdkError> {
289        Ok(geospatial_types::RouteStep {
290            distance_meters: self.distance_meters,
291            duration_seconds: self.duration_seconds,
292            maneuver_type: native_maneuver_type_from_text(&self.maneuver_type)?,
293            instruction_display_text: self.instruction_display_text,
294            instruction_speech_text: self.instruction_speech_text,
295            shape: self
296                .shape
297                .into_iter()
298                .map(|shape_coordinate| shape_coordinate.to_native_coordinate())
299                .collect(),
300            lane_guidances: Vec::new(),
301        })
302    }
303}

305/// Leg between two consecutive waypoints.
306#[derive(Debug, Clone, PartialEq)]
307pub struct RouteLeg {
308    /// Leg distance in meters.
309    pub distance_meters: f64,
310    /// Leg travel time in seconds.
311    pub duration_seconds: f64,
312    /// Maneuver steps in traversal order.
313    pub steps: Vec<RouteStep>,
314}
315
316impl RouteLeg {
317    /// Builds the mirror record from a native route leg.
318    pub(crate) fn from_native_route_leg(native_route_leg: &geospatial_types::RouteLeg) -> Self {
319        Self {
320            distance_meters: native_route_leg.distance_meters,
321            duration_seconds: native_route_leg.duration_seconds,
322            steps: native_route_leg
323                .steps
324                .iter()
325                .map(RouteStep::from_native_route_step)
326                .collect(),
327        }
328    }
329
330    /// Builds the native route leg from the mirror record, refusing a step
331    /// with an unrecognized maneuver label.
332    pub(crate) fn into_native_route_leg(self) -> Result<geospatial_types::RouteLeg, SdkError> {
333        Ok(geospatial_types::RouteLeg {
334            distance_meters: self.distance_meters,
335            duration_seconds: self.duration_seconds,
336            steps: self
337                .steps
338                .into_iter()
339                .map(|mirror_step| mirror_step.into_native_route_step())
340                .collect::<Result<Vec<_>, SdkError>>()?,
341        })
342    }
343}

345/// A complete calculated route.
346#[derive(Debug, Clone, PartialEq)]
347pub struct Route {
348    /// Total distance across all legs, in meters.
349    pub distance_meters: f64,
350    /// Total travel time across all legs, in seconds.
351    pub duration_seconds: f64,
352    /// Geographic bounds of the full route geometry.
353    pub bounding_box: BoundingBox,
354    /// The full route geometry in traversal order.
355    pub geometry: Vec<GeographicCoordinate>,
356    /// Legs, one per consecutive waypoint pair.
357    pub legs: Vec<RouteLeg>,
358}

360impl Route {
361    /// Builds the mirror record from a native route.

363        Self {
364            distance_meters: native_route_data.distance_meters,
365            duration_seconds: native_route_data.duration_seconds,

375                .map(RouteLeg::from_native_route_leg)

377        }
378    }

380    /// Builds the native route from the mirror record, refusing a step with

384            distance_meters: self.distance_meters,
385            duration_seconds: self.duration_seconds,
386            bounding_box: self.bounding_box.to_native_bounding_box(),
387            geometry: self
388                .geometry
389                .into_iter()
390                .map(GeographicCoordinate::to_native_coordinate)
391                .collect(),
392            legs: self
393                .legs
394                .into_iter()
395                .map(|mirror_leg| mirror_leg.into_native_route_leg())
396                .collect::<Result<Vec<_>, SdkError>>()?,
397        })
398    }

401/// One origin-destination cell of a distance matrix.
402#[derive(Debug, Clone, Copy, PartialEq)]
403pub struct DistanceMatrixEntry {
404    /// Origin index into the request's origins.
405    pub origin_index: u32,
406    /// Destination index into the request's destinations.
407    pub destination_index: u32,
408    /// Distance for this cell in meters.
409    pub distance_meters: f64,
410    /// Travel time for this cell in seconds.
411    pub duration_seconds: f64,
412}
413
414/// Travel-time matrix for origins x destinations.
415#[derive(Debug, Clone, PartialEq)]
416pub struct DistanceMatrix {
417    /// Entries in row-major origin-major order.
418    pub entries: Vec<DistanceMatrixEntry>,
419}
420
421impl DistanceMatrix {
422    /// Builds the mirror record from a native distance matrix.
423    pub(crate) fn from_native_distance_matrix(
424        native_distance_matrix: &geospatial_types::DistanceMatrix,
425    ) -> Self {
426        Self {
427            entries: native_distance_matrix
428                .entries
429                .iter()
430                .map(|native_entry| DistanceMatrixEntry {
431                    origin_index: native_entry.origin_index,
432                    destination_index: native_entry.destination_index,
433                    distance_meters: native_entry.distance_meters,
434                    duration_seconds: native_entry.duration_seconds,
435                })
436                .collect(),
437        }
438    }
439
440    /// Builds the native distance matrix from the mirror record — the foreign
441    /// provider adapters' direction. The cell values copy verbatim: a
442    /// provider's reported distance and duration are its answer, and the
443    /// routing call that requested the matrix is where input validity is
444    /// enforced.
445    pub(crate) fn into_native_distance_matrix(self) -> geospatial_types::DistanceMatrix {
446        geospatial_types::DistanceMatrix {
447            entries: self
448                .entries
449                .into_iter()
450                .map(|entry| geospatial_types::DistanceMatrixEntry {
451                    origin_index: entry.origin_index,
452                    destination_index: entry.destination_index,
453                    distance_meters: entry.distance_meters,
454                    duration_seconds: entry.duration_seconds,
455                })
456                .collect(),
457        }
458    }
