6/// Class of disruption reported in a traffic incident.
7#[derive(Debug, Clone, Copy, PartialEq, Eq)]
8pub enum TrafficIncidentType {
9    /// Road construction.
10    Construction,
11    /// Traffic-impacting event.
12    Event,
13    /// Collision.
14    Accident,
15    /// Weather-related disruption.
16    Weather,
17    /// Lane or road closure.
18    Closure,
19}
20
21impl TrafficIncidentType {
22    /// Builds the mirror enum from the native incident type.
23    pub(crate) fn from_native_traffic_incident_type(
24        native_type: geospatial_types::TrafficIncidentType,
25    ) -> Self {
26        match native_type {
27            geospatial_types::TrafficIncidentType::Construction => Self::Construction,
28            geospatial_types::TrafficIncidentType::Event => Self::Event,
29            geospatial_types::TrafficIncidentType::Accident => Self::Accident,
30            geospatial_types::TrafficIncidentType::Weather => Self::Weather,
31            geospatial_types::TrafficIncidentType::Closure => Self::Closure,
32        }
33    }
34
35    /// Builds the native incident type from the mirror enum (the foreign
36    /// provider adapters' direction).
37    pub(crate) fn to_native_traffic_incident_type(self) -> geospatial_types::TrafficIncidentType {
38        match self {
39            Self::Construction => geospatial_types::TrafficIncidentType::Construction,
40            Self::Event => geospatial_types::TrafficIncidentType::Event,
41            Self::Accident => geospatial_types::TrafficIncidentType::Accident,
42            Self::Weather => geospatial_types::TrafficIncidentType::Weather,
43            Self::Closure => geospatial_types::TrafficIncidentType::Closure,
44        }
45    }
46}

111/// One reported disruption inside a queried bounding box.
112#[derive(Debug, Clone, PartialEq)]
113pub struct TrafficIncident {
114    /// Provider-stable incident identifier.
115    pub identifier: String,
116    /// Class of disruption.
117    pub incident_type: TrafficIncidentType,
118    /// Severity ranking.
119    pub severity: TrafficIncidentSeverity,
120    /// Provider display text for this incident.
121    pub short_description_text: String,
122    /// Position where the incident starts.
123    pub start_coordinate: GeographicCoordinate,
124    /// Position where the incident ends, if reported.
125    pub end_coordinate: Option<GeographicCoordinate>,
126}
127
128impl TrafficIncident {
129    /// Builds the mirror record from a native traffic incident.
130    pub(crate) fn from_native_traffic_incident(
131        native_traffic_incident: &geospatial_types::TrafficIncident,
132    ) -> Self {
133        Self {
134            identifier: native_traffic_incident.identifier.clone(),
135            incident_type: TrafficIncidentType::from_native_traffic_incident_type(
136                native_traffic_incident.incident_type,
137            ),
138            severity: TrafficIncidentSeverity::from_native_traffic_incident_severity(
139                native_traffic_incident.severity,
140            ),
141            short_description_text: native_traffic_incident.short_description_text.clone(),
142            start_coordinate: GeographicCoordinate::from_native_coordinate(
143                &native_traffic_incident.start_coordinate,
144            ),
145            end_coordinate: native_traffic_incident
146                .end_coordinate
147                .map(|native_end_coordinate| {
148                    GeographicCoordinate::from_native_coordinate(&native_end_coordinate)
149                }),
150        }
151    }
152
153    /// Builds the native traffic incident from the mirror record — the
154    /// foreign provider adapters' direction. The coordinates validate like
155    /// every other geometry a provider returns, so an out-of-range incident
156    /// position refuses the call rather than entering the map pipeline.
157    pub(crate) fn into_native_traffic_incident(
158        self,
159    ) -> Result<geospatial_types::TrafficIncident, SdkError> {
160        Ok(geospatial_types::TrafficIncident {
161            identifier: self.identifier,
162            incident_type: self.incident_type.to_native_traffic_incident_type(),
163            severity: self.severity.to_native_traffic_incident_severity(),
164            short_description_text: self.short_description_text,
165            start_coordinate: crate::projection::validated_coordinate(self.start_coordinate)?,
166            end_coordinate: self
167                .end_coordinate
168                .map(crate::projection::validated_coordinate)
169                .transpose()?,
170        })
171    }
172}

174/// Flow condition for one road segment (or polyline of road segments) inside
175/// the queried box, in kilometers per hour when the provider reports speeds
176/// and as a congestion classification otherwise.
177#[derive(Debug, Clone, PartialEq)]
178pub struct TrafficFlowData {
179    /// Segment geometry, ordered along the road.
180    pub shape_coordinates: Vec<GeographicCoordinate>,
181    /// Currently observed speed in kilometers per hour, when reported.
182    pub current_speed_kilometers_per_hour: Option<f64>,
183    /// Unconstrained (free-flow) speed in kilometers per hour, when reported.
184    pub free_flow_speed_kilometers_per_hour: Option<f64>,
185    /// Congestion classification from the speed ratio or provider buckets.
186    pub congestion_level: TrafficFlowCongestionLevel,
187}
188
189impl TrafficFlowData {
190    /// Builds the mirror record from a native traffic flow observation.
191    pub(crate) fn from_native_traffic_flow_data(
192        native_traffic_flow_data: &geospatial_types::TrafficFlowData,
193    ) -> Self {
194        Self {
195            shape_coordinates: native_traffic_flow_data
196                .shape_coordinates
197                .iter()
198                .map(GeographicCoordinate::from_native_coordinate)
199                .collect(),
200            current_speed_kilometers_per_hour: native_traffic_flow_data
201                .current_speed_kilometers_per_hour,
202            free_flow_speed_kilometers_per_hour: native_traffic_flow_data
203                .free_flow_speed_kilometers_per_hour,
204            congestion_level: TrafficFlowCongestionLevel::from_native_congestion_level(
205                native_traffic_flow_data.congestion_level,
206            ),
207        }
208    }
209
210    /// Builds the native traffic flow observation from the mirror record —
211    /// the surface-view converter's and the foreign provider adapters'
212    /// direction, feeding the overlay assembly. The shape coordinates
213    /// validate like every other overlay payload; the speeds and the
214    /// congestion classification copy verbatim (the congestion color lookup
215    /// is total over the classification).
216    pub(crate) fn into_native_traffic_flow_data(
217        self,
218    ) -> Result<geospatial_types::TrafficFlowData, SdkError> {
219        Ok(geospatial_types::TrafficFlowData {
220            shape_coordinates: self
221                .shape_coordinates
222                .into_iter()
223                .map(crate::projection::validated_coordinate)
224                .collect::<Result<Vec<_>, SdkError>>()?,
225            current_speed_kilometers_per_hour: self.current_speed_kilometers_per_hour,
226            free_flow_speed_kilometers_per_hour: self.free_flow_speed_kilometers_per_hour,
227            congestion_level: self.congestion_level.to_native_congestion_level(),
228        })
229    }
230}
