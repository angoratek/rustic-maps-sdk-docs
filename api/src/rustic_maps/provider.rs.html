1//! The UniFFI-facing async provider surface: one [`MapSdk`] object resolving
2//! the implemented `geospatial-traits` contracts through the `rustic-maps-core`
3//! [`ProviderRegistry`], one exported `async` method per contract call
4//! mirroring the trait surface one-to-one.
5//!
6//! Provider registration has two paths. The native integrator path stays
7//! foreign-free: the `register_*` methods here wrap the registry for
8//! providers built in Rust, including MapQuest providers constructed from a
9//! consumer key. The foreign integrator path injects implementations of the
10//! exported callback contracts through the `register_*_from_foreign`
11//! methods, which wrap them in the [`crate::foreign_providers`] adapters
12//! before the same registry call. Calls into an unregistered contract throw
13//! [`SdkError::ProviderNotConfigured`] with the contract name.
14

62/// Async provider entry point of the SDK: resolves every call of the
63/// implemented provider contracts through the [`ProviderRegistry`].
64///
65/// The exported methods mirror the provider contracts one-to-one; each
66/// validates its inputs (binding coordinates must be canonical degrees) and
67/// converts the native result into the binding records in [`crate::records`].
68/// Provider registration stays native (this crate's `register_*` methods).
69///
70/// The registry lock is taken only to clone a provider slot out of the
71/// registry; the guard never spans an await, so every exported future stays
72/// `Send`. Each exported method resolves its provider synchronously and
73/// then hands the call's work to the crate-owned tokio runtime through
74/// [`drive`], whose returned future only awaits a completion channel — the
75/// foreign executor polls a runtime-independent handback, because the
76/// provider futures need a live reactor no foreign polling thread runs (see
77/// the `runtime` module).
78#[derive(Debug)]
79pub struct MapSdk {
80    provider_registry: Mutex<ProviderRegistry>,
81    /// The tile cache currently installed behind the raster-tile contract
82    /// slot with the terms its settings surface reads and tunes, or `None`
83    /// while the raster-tile contract serves uncached (never registered
84    /// with a tile cache, or re-registered without one — re-registration
85    /// replaces the slots and the tuning with them). The wasm build has no
86    /// disk cache store, so it carries no installation slot at all.
87    #[cfg(not(target_arch = "wasm32"))]
88    tile_cache_installation: Mutex<Option<TileCacheInstallation>>,
89    /// The provider-failure diagnostic seam's ring: the bounded record of
90    /// every failed foreign-callback provider call registered on
91    /// this object, read back through
92    /// [`MapSdk::latest_provider_failure_diagnostics`]. The foreign-callback
93    /// adapters registered here hold a handle to it.
94    provider_failure_diagnostics: Arc<crate::provider_diagnostics::ProviderFailureDiagnosticsRing>,
95    /// Which provider key an application chose for each capability, kept
96    /// beside the registry because the registry answers only whether a slot
97    /// is filled. [`Self::provider_registrations`] reads both, so a settings
98    /// surface names the serving provider rather than guessing.
99    provider_selections: Mutex<Vec<crate::provider_catalog::ProviderSelection>>,
100}

201    /// Chooses the provider that serves one capability.
202    ///
203    /// The key names a provider this build carries; the call registers that
204    /// provider's implementation for the requested capability and leaves every
205    /// other capability untouched, so an application can serve geocoding from
206    /// one provider and routing from another. Nothing is registered until the
207    /// provider constructs successfully, so a refused call leaves the
208    /// previously serving provider in place.
209    ///
210    /// # Parameters
211    /// - `contract` — the capability whose provider is being chosen.
212    /// - `provider_key` — the stable key of a provider built into this SDK.
213    /// - `credential` — the provider's credential; empty is accepted only for
214    ///   a provider that needs none.
215    ///
216    /// # Errors
217    /// Throws `ProviderNotConfigured` when the key is unknown to this build or
218    /// serves no implementation of `contract`, and `ProviderRequestRejected`
219    /// when the provider declares a credential and the supplied one is blank.
220    /// A provider's own construction failure is passed through verbatim.
221    pub fn configure_provider(
222        &self,
223        contract: ProviderContract,
224        provider_key: String,
225        credential: String,
226    ) -> Result<(), SdkError> {
227        let Some(catalog_entry) = crate::provider_catalog::catalog_entry_for_key(&provider_key)
228        else {
229            return Err(SdkError::ProviderNotConfigured {
230                contract_name: format!("provider key {provider_key} is not built into this SDK"),
231            });
232        };
233        if !catalog_entry.served_contracts.contains(&contract) {
234            return Err(SdkError::ProviderNotConfigured {
235                contract_name: crate::provider_catalog::contract_name_of(contract),
236            });
237        }
238        let trimmed_credential = credential.trim();
239        if catalog_entry.credential_required && trimmed_credential.is_empty() {
240            return Err(SdkError::ProviderRequestRejected {
241                detail: format!(
242                    "the provider {} requires a credential before it can serve requests",
243                    catalog_entry.display_name
244                ),
245            });
246        }
247        (catalog_entry.install)(self, contract, trimmed_credential)?;
248        self.record_provider_selection(
249            contract,
250            catalog_entry.provider_key,
251            catalog_entry.display_name,
252        );
253        Ok(())
254    }

256    /// Reports which provider serves each capability, in the order a settings
257    /// surface lists them.
258    ///
259    /// Every capability appears, including the ones with no provider
260    /// installed; those carry an empty key and name with `registered` false.
261    /// The readback never mutates the registry.
262    pub fn provider_registrations(&self) -> Vec<crate::records::ProviderRegistration> {
263        let selections = crate::panic_boundary::state_lock_guard(
264            &self.provider_selections,
265            "the provider selection record",
266        )
267        .clone();
268        let provider_registry = crate::panic_boundary::state_lock_guard(
269            &self.provider_registry,
270            "the provider registry",
271        );
272        rustic_maps_core::ProviderContract::ALL
273            .into_iter()
274            .map(|core_contract| {
275                let contract = ProviderContract::from_core_contract(core_contract);
276                let selection = selections
277                    .iter()
278                    .find(|selection| selection.contract == contract);
279                crate::provider_catalog::registration_entry(
280                    contract,
281                    selection,
282                    provider_registry.is_configured(core_contract),
283                )
284            })
285            .collect()
286    }

288    /// Subscribes a listener to the diagnostics live stream: the listener
289    /// receives every SDK event that meets its minimum severity as
290    /// it is recorded, fire-and-forget. A subscription at
291    /// [`SdkLogSeverity::Debug`] or [`SdkLogSeverity::Trace`] also lowers the
292    /// sink's severity gate, which starts the GPU-hot render path's
293    /// fine-grained instrumentation (the gate never raises back). Duplicate
294    /// registrations deliver duplicate events (the caller's subscription
295    /// lifecycle owns the listener).
296    pub fn subscribe_to_diagnostic_events(
297        &self,
298        listener: Arc<dyn crate::diagnostics::SdkDiagnosticEventListener>,
299        minimum_severity: SdkLogSeverity,
300    ) {
301        crate::diagnostics::DiagnosticsHandle::shared().subscribe(listener, minimum_severity);
302    }

304    /// Replays the most recent events the diagnostics rings recorded: at
305    /// most `maximum_count` events oldest first, every
306    /// subsystem's ring interleaved in recording order, filtered by
307    /// `minimum_severity`. The flush is non-destructive — the rings are
308    /// never emptied, so a later flush can redeliver the same events.
309    /// Delivered events are fired-and-forget, never awaited.
310    pub fn flush_recent_diagnostic_events(
311        &self,
312        listener: Arc<dyn crate::diagnostics::SdkDiagnosticEventListener>,
313        minimum_severity: SdkLogSeverity,
314        maximum_count: u32,
315    ) -> u32 {
316        let recent_events = crate::diagnostics::DiagnosticsHandle::shared()
317            .recent_events(minimum_severity, maximum_count);
318        let flush_count = recent_events.len() as u32;
319        for event in recent_events {
320            listener.on_diagnostic_event(event);
321        }
322        flush_count
323    }

671    /// The shared MapQuest installation constructing the whole provider set;
672    /// the plain [`Self::set_provider_credential`] path leaves the
673    /// raster-tile contract uncached, the with-tile-cache path wraps it in
674    /// the caller's store. A fresh registration always starts at the tiles
675    /// category's default time-to-live (thirty days, clamped through the
676    /// provider's terms ceiling) and the settings
677    /// surface tunes it afterwards — re-registration replaces the tuning
678    /// with the slots.
679    #[cfg(not(target_arch = "wasm32"))]
680    fn register_mapquest_providers_internal(
681        &self,
682        mapquest_api_key: String,
683        tile_cache_wiring: Option<TileCacheWiring>,
684    ) -> Result<(), SdkError> {
685        let uncached_raster_tile_provider =
686            self.register_mapquest_provider_set(mapquest_api_key)?;
687        let tile_cache_installation = tile_cache_wiring.map(|tile_cache_wiring| {
688            let default_time_to_live_seconds =
689                rustic_maps_cache::ResponseCategory::RasterTiles.default_time_to_live_seconds();
690            let clamped_time_to_live_seconds = clamped_to_provider_terms_ceiling(
691                default_time_to_live_seconds,
692                tile_cache_wiring.provider_terms_ceiling_seconds,
693            );
694            self.register_raster_tile_provider(Arc::new(
695                rustic_maps_cache::CachedRasterTileProvider::new(
696                    uncached_raster_tile_provider.clone(),
697                    tile_cache_wiring.response_store.clone(),
698                )
699                .with_time_to_live_seconds(clamped_time_to_live_seconds),
700            ));
701            TileCacheInstallation {
702                raster_tile_provider: uncached_raster_tile_provider.clone(),
703                response_store: tile_cache_wiring.response_store,
704                time_to_live_seconds: clamped_time_to_live_seconds,
705                provider_terms_ceiling_seconds: tile_cache_wiring.provider_terms_ceiling_seconds,
706            }
707        });
708        if tile_cache_installation.is_none() {
709            self.register_raster_tile_provider(uncached_raster_tile_provider);
710        }
711        *crate::panic_boundary::state_lock_guard(
712            &self.tile_cache_installation,
713            "the tile-cache installation state",
714        ) = tile_cache_installation;
715        Ok(())
716    }

732    /// Registers the seven native MapQuest provider implementations from one
733    /// key exactly as [`Self::set_provider_credential`], wrapping the
734    /// raster-tile contract in the disk response cache opened (or created)
735    /// as a SQLite database at `tile_cache_database_path` under an absolute
736    /// byte budget
737    /// of `tile_cache_absolute_maximum_bytes`. Cached tile responses follow
738    /// the tiles category's time-to-live (thirty days by default, the
739    /// imagery provider's declared ceiling, applied as
740    /// [`rustic_maps_cache::ResponseCategory::RasterTiles`]'s default); a cache
741    /// miss goes to the network and back-fills the cache, so pre-warmed
742    /// tiles keep rendering whenever the network is unavailable.
743    ///
744    /// The path must be non-empty and the byte budget positive; a cache
745    /// open or quota failure throws with its verbatim detail. Re-registration
746    /// replaces the slots (including the cached raster-tile slot).
747    ///
748    /// The disk store is native-only: a wasm build has no SQLite response
749    /// store, so the call validates its inputs and then throws
750    /// [`SdkError::CacheOperationFailure`] with that target limitation.
751    pub fn set_provider_credential_with_tile_cache(
752        &self,
753        mapquest_api_key: String,
754        tile_cache_database_path: String,
755        tile_cache_absolute_maximum_bytes: u64,
756    ) -> Result<(), SdkError> {
757        if tile_cache_database_path.is_empty() {
758            return Err(SdkError::InvalidCoordinateInput {
759                detail: "the tile cache database path must be non-empty".to_string(),
760            });
761        }
762        if tile_cache_absolute_maximum_bytes == 0 {
763            return Err(SdkError::InvalidCoordinateInput {
764                detail: "the tile cache byte budget must be positive".to_string(),
765            });
766        }
767        #[cfg(not(target_arch = "wasm32"))]
768        {
769            let tile_cache_response_store = rustic_maps_cache::SqliteResponseCache::open(
770                rustic_maps_cache::SqliteResponseCacheConfiguration {
771                    database_path: std::path::PathBuf::from(tile_cache_database_path),
772                    quota: rustic_maps_cache::CacheQuota {
773                        absolute_maximum_bytes: Some(tile_cache_absolute_maximum_bytes),
774                        ..Default::default()
775                    },
776                    ..Default::default()
777                },
778            )
779            .map_err(SdkError::from)?;
780            self.register_mapquest_providers_internal(
781                mapquest_api_key,
782                Some(TileCacheWiring {
783                    response_store: tile_cache_response_store,
784                    // The MapQuest raster-tile terms (PRD §6.7) are this
785                    // installation's declared caching-terms ceiling: the
786                    // settings surface clamps every requested time-to-live
787                    // through it.
788                    provider_terms_ceiling_seconds: Some(
789                        mapquest_provider::tiles::MAPQUEST_RASTER_TILE_CACHE_CEILING_SECONDS,
790                    ),
791                }),
792            )
793        }
794        #[cfg(target_arch = "wasm32")]
795        {
796            let _ = mapquest_api_key;
797            Err(SdkError::CacheOperationFailure {
798                detail: "the disk tile cache is unavailable on this target; \
799                         a wasm build has no SQLite response store"
800                    .to_string(),
801            })
802        }
803    }

843    /// Overrides the installed raster-tile cache's time-to-live, in seconds,
844    /// re-wrapping the cached provider under the requested value through
845    /// [`rustic_maps_cache::CachedRasterTileProvider::with_time_to_live_seconds`]. A
846    /// value above the installed provider's declared terms ceiling is
847    /// accepted with the ceiling clamping what the cache stores (the ceiling
848    /// bounds what the cache may retain, never a preference);
849    /// [`Self::raster_tile_cache_terms`] reports the effective value. A zero
850    /// value is refused: a zero time-to-live expires every entry on capture,
851    /// a cache that stores nothing. Requires a tile cache installed through
852    /// [`Self::set_provider_credential_with_tile_cache`] — the
853    /// byte budget is registration-time (the store's quota is fixed at open)
854    /// and retunes only through re-registration.
855    pub fn set_raster_tile_cache_time_to_live(
856        &self,
857        time_to_live_seconds: u64,
858    ) -> Result<(), SdkError> {
859        if time_to_live_seconds == 0 {
860            return Err(SdkError::InvalidCoordinateInput {
861                detail: "the raster-tile cache time-to-live must be positive (seconds)".to_string(),
862            });
863        }
864        #[cfg(not(target_arch = "wasm32"))]
865        {
866            let (rebuilt_raster_tile_provider, clamped_time_to_live_seconds) = {
867                let installation = crate::panic_boundary::state_lock_guard(
868                    &self.tile_cache_installation,
869                    "the tile-cache installation state",
870                );
871                let tile_cache_installation =
872                    installation
873                        .as_ref()
874                        .ok_or(SdkError::ProviderNotConfigured {
875                            contract_name: TILE_CACHE_NOT_CONFIGURED_CONTRACT_NAME.to_string(),
876                        })?;
877                let clamped_time_to_live_seconds = clamped_to_provider_terms_ceiling(
878                    time_to_live_seconds,
879                    tile_cache_installation.provider_terms_ceiling_seconds,
880                );
881                (
882                    Arc::new(
883                        rustic_maps_cache::CachedRasterTileProvider::new(
884                            tile_cache_installation.raster_tile_provider.clone(),
885                            tile_cache_installation.response_store.clone(),
886                        )
887                        .with_time_to_live_seconds(clamped_time_to_live_seconds),
888                    ),
889                    clamped_time_to_live_seconds,
890                )
891            };
892            // The rebuild and the registry write stay outside the
893            // installation lock: the registry's guard is never taken while
894            // the installation's is held, and vice versa, so the two locks
895            // cannot dead lock.
896            self.register_raster_tile_provider(rebuilt_raster_tile_provider);
897            if let Some(tile_cache_installation) = crate::panic_boundary::state_lock_guard(
898                &self.tile_cache_installation,
899                "the tile-cache installation state",
900            )
901            .as_mut()
902            {
903                tile_cache_installation.time_to_live_seconds = clamped_time_to_live_seconds;
904            }
905            Ok(())
906        }
907        // No installable disk cache exists on wasm, so the native
908        // not-configured refusal is the only outcome there.
909        #[cfg(target_arch = "wasm32")]
910        {
911            Err(SdkError::ProviderNotConfigured {
912                contract_name: TILE_CACHE_NOT_CONFIGURED_CONTRACT_NAME.to_string(),
913            })
914        }
915    }

1328/// What the shared MapQuest installation needs to wire the raster-tile
1329/// contract into the disk cache: the opened response store, and the
1330/// provider's declared caching-terms ceiling, which the tile time-to-live
1331/// clamps through. Native-only: the wasm build installs no disk cache.
1332#[cfg(not(target_arch = "wasm32"))]
1333struct TileCacheWiring {
1334    /// The opened response store the cached wrapper consults.
1335    response_store: Arc<rustic_maps_cache::SqliteResponseCache>,
1336    /// The provider's declared caching-terms ceiling in seconds; `None`
1337    /// leaves every requested time-to-live applying verbatim.
1338    provider_terms_ceiling_seconds: Option<u64>,
1339}
1340
1341/// Clamps a requested tile-cache time-to-live into what the installed
1342/// provider's terms allow: the ceiling bounds what the cache may
1343/// retain, so a request above it stores at the ceiling and the terms
1344/// readback reports the effective value — the request is never refused,
1345/// because the compliance bound is enforced by what the cache stores, not by
1346/// how the call returns. Without a declared ceiling the requested value
1347/// applies verbatim. Native-only with the disk-cache installation it serves.
1348#[cfg(not(target_arch = "wasm32"))]
1349fn clamped_to_provider_terms_ceiling(
1350    requested_time_to_live_seconds: u64,
1351    provider_terms_ceiling_seconds: Option<u64>,
1352) -> u64 {
1353    requested_time_to_live_seconds
1354        .min(provider_terms_ceiling_seconds.unwrap_or(NO_PROVIDER_TERMS_CEILING_SECONDS))
1355}
