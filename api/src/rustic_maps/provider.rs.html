1//! The UniFFI-facing async provider surface: one [`MapSdk`] object resolving
2//! the implemented `geospatial-traits` contracts through the `rustic-maps-core`
3//! [`ProviderRegistry`], one exported `async` method per contract call
4//! mirroring the trait surface one-to-one.
5//!
6//! Provider registration has two paths. The native integrator path stays
7//! foreign-free: the `register_*` methods here wrap the registry for
8//! providers built in Rust, including MapQuest providers constructed from a
9//! consumer key. The foreign integrator path injects implementations of the
10//! exported callback contracts through the `register_*_from_foreign`
11//! methods, which wrap them in the [`crate::foreign_providers`] adapters
12//! before the same registry call. Calls into an unregistered contract throw
13//! [`SdkError::ProviderNotConfigured`] with the contract name.

65/// Async provider entry point of the SDK: resolves every call of the
66/// implemented provider contracts through the [`ProviderRegistry`].
67///
68/// The exported methods mirror the provider contracts one-to-one; each
69/// validates its inputs (binding coordinates must be canonical degrees) and
70/// converts the native result into the binding records in [`crate::records`].
71/// Provider registration stays native (this crate's `register_*` methods).
72///
73/// The registry lock is taken only to clone a provider slot out of the
74/// registry; the guard never spans an await, so every exported future stays
75/// `Send`. Each exported method resolves its provider synchronously and
76/// then hands the call's work to the crate-owned tokio runtime through
77/// [`drive`], whose returned future only awaits a completion channel — the
78/// foreign executor polls a runtime-independent handback, because the
79/// provider futures need a live reactor no foreign polling thread runs (see
80/// the `runtime` module).
81#[derive(Debug)]
82pub struct MapSdk {
83    provider_registry: Mutex<ProviderRegistry>,
84    /// The tile cache currently installed behind the raster-tile contract
85    /// slot with the terms its settings surface reads and tunes, or `None`
86    /// while the raster-tile contract serves uncached (never registered
87    /// with a tile cache, or re-registered without one — re-registration
88    /// replaces the slots and the tuning with them). The wasm build has no
89    /// disk cache store, so it carries no installation slot at all.
90    #[cfg(not(target_arch = "wasm32"))]
91    tile_cache_installation: Mutex<Option<TileCacheInstallation>>,
92    /// The provider-failure diagnostic seam's ring: the bounded record of
93    /// every failed foreign-callback provider call registered on
94    /// this object, read back through
95    /// [`MapSdk::latest_provider_failure_diagnostics`]. The foreign-callback
96    /// adapters registered here hold a handle to it.
97    provider_failure_diagnostics: Arc<crate::provider_diagnostics::ProviderFailureDiagnosticsRing>,
98    /// Which provider key an application chose for each capability, kept
99    /// beside the registry because the registry answers only whether a slot
100    /// is filled. [`Self::provider_registrations`] reads both, so a settings
101    /// surface names the serving provider rather than guessing.
102    provider_selections: Mutex<Vec<crate::provider_catalog::ProviderSelection>>,
103    /// The in-flight offline region downloads, keyed by their region
104    /// identifier. Keeping the registry on the SDK (not on the installation)
105    /// means every `OfflineManager` built over the same `MapSdk` sees the same
106    /// downloads; replacing the installation signals every entry to cancel so
107    /// no download keeps writing into a store the SDK stopped serving. The
108    /// wasm build has no disk store and no region downloads.
109    #[cfg(not(target_arch = "wasm32"))]
110    region_downloads: Mutex<HashMap<String, Arc<RegionDownloadCancellation>>>,
111}

113/// The cancellation signal of one in-flight offline region download. A
114/// download races its prepare future against the receiver of this channel and
115/// drops the future when the signal arrives; dropping is cancellation-safe by
116/// the tile-pack manager's contract (each tile fetch is atomic, the store
117/// writes are transactional, and the membership names the missing tiles).
118#[cfg(not(target_arch = "wasm32"))]
119pub(crate) struct RegionDownloadCancellation {
120    /// `true` once the download must stop. The receiver lives inside the
121    /// download's `tokio::select!`; the sender lives in the SDK registry.
122    cancellation_signal: tokio::sync::watch::Sender<bool>,
123}
124
125#[cfg(not(target_arch = "wasm32"))]
126impl std::fmt::Debug for RegionDownloadCancellation {
127    fn fmt(&self, formatter: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
128        formatter
129            .debug_struct("RegionDownloadCancellation")
130            .finish_non_exhaustive()
131    }
132}

134#[cfg(not(target_arch = "wasm32"))]
135impl RegionDownloadCancellation {
136    /// Builds a fresh, unsignalled cancellation channel for one download.
137    fn new() -> Self {
138        let (cancellation_signal, _receiver) = tokio::sync::watch::channel(false);
139        Self {
140            cancellation_signal,
141        }
142    }
143
144    /// Subscribes a receiver the download races against its prepare future.
145    fn subscribe(&self) -> tokio::sync::watch::Receiver<bool> {
146        self.cancellation_signal.subscribe()
147    }
148
149    /// Signals the download to stop. Idempotent: a second signal is a no-op.
150    fn signal_cancellation(&self) {
151        // The receiver existing for the whole download is the normal case;
152        // a receiver that already went away means the download finished and
153        // there is nothing left to signal.
154        let _ = self.cancellation_signal.send(true);
155    }
156}

158/// The tile cache currently installed behind the raster-tile contract slot,
159/// holding what the settings surface needs to read the effective terms and
160/// to rewrap the underlying provider under a new time-to-live
161/// (`rustic_maps_cache::CachedRasterTileProvider` re-wraps per override; the store's
162/// quota stays fixed at open). Native-only: the wasm build has no SQLite
163/// response store to install.
164#[cfg(not(target_arch = "wasm32"))]
165struct TileCacheInstallation {
166    /// The underlying, uncached raster-tile provider the cached wrapper
167    /// re-wraps on every time-to-live override.
168    raster_tile_provider: Arc<dyn RasterTileProvider>,
169    /// The SQLite response store the cached wrapper consults. Kept as the
170    /// concrete store so the vector-tile payload cache can share it under a
171    /// second typed view; the store serves any serializable record.
172    response_store: Arc<rustic_maps_cache::SqliteResponseCache>,
173    /// Time-to-live currently in force on the installed cached wrapper, in
174    /// seconds.
175    time_to_live_seconds: u64,
176    /// The installed provider's declared caching-terms ceiling in seconds;
177    /// `None` when the provider declares no ceiling (an override then
178    /// applies verbatim).
179    provider_terms_ceiling_seconds: Option<u64>,
180    /// The installed provider's declared maximum raster-tile zoom level;
181    /// `None` when the provider declares none, which leaves the tile-pack
182    /// manager's structural bound in force.
183    raster_tile_maximum_zoom_level: Option<u8>,
184}

205    }

208#[cfg(not(target_arch = "wasm32"))]
209impl TileCacheInstallation {
210    /// Builds a tile-pack manager over this installation's current snapshot:
211    /// the uncached provider wrapped in the cached pipeline under the current
212    /// time-to-live, the same store, the provider's terms ceiling and its
213    /// declared maximum zoom level.
214    ///
215    /// The manager is rebuilt per exported call rather than stored: an
216    /// override of the time-to-live rewraps the cached provider, and a stored
217    /// manager would keep holding the previous wrapper and its time-to-live.
218    fn tile_pack_manager(&self) -> rustic_maps_cache::TilePackManager {
219        let cached_response_store: Arc<
220            dyn rustic_maps_core::CachedResponseStore<geospatial_types::RasterTileData>,
221        > = self.response_store.clone();
222        let cached_raster_tile_provider = rustic_maps_cache::CachedRasterTileProvider::new(
223            Arc::clone(&self.raster_tile_provider),
224            cached_response_store,
225        )
226        .with_time_to_live_seconds(self.time_to_live_seconds)
227        .with_blocking_work_executor(crate::runtime::blocking_work_executor());
228        let tile_pack_manager = rustic_maps_cache::TilePackManager::new(
229            Arc::new(cached_raster_tile_provider),
230            Arc::clone(&self.response_store),
231            self.provider_terms_ceiling_seconds,
232        )
233        .with_blocking_work_executor(crate::runtime::blocking_work_executor());
234        match self.raster_tile_maximum_zoom_level {
235            Some(raster_tile_maximum_zoom_level) => {
236                tile_pack_manager.with_maximum_zoom_level(raster_tile_maximum_zoom_level)
237            }
238            None => tile_pack_manager,
239        }
240    }
241}

307    /// Chooses the provider that serves one capability.
308    ///
309    /// The key names a provider this build carries; the call registers that
310    /// provider's implementation for the requested capability and leaves every
311    /// other capability untouched, so an application can serve geocoding from
312    /// one provider and routing from another. Nothing is registered until the
313    /// provider constructs successfully, so a refused call leaves the
314    /// previously serving provider in place.
315    ///
316    /// # Parameters
317    /// - `contract` — the capability whose provider is being chosen.
318    /// - `provider_key` — the stable key of a provider built into this SDK.
319    /// - `credential` — the provider's credential; empty is accepted only for
320    ///   a provider that needs none.
321    ///
322    /// # Errors
323    /// Throws `ProviderNotConfigured` when the key is unknown to this build or
324    /// serves no implementation of `contract`, and `ProviderRequestRejected`
325    /// when the provider declares a credential and the supplied one is blank.
326    /// A provider's own construction failure is passed through verbatim.
327    pub fn configure_provider(
328        &self,
329        contract: ProviderContract,
330        provider_key: String,
331        credential: String,
332    ) -> Result<(), SdkError> {
333        let Some(catalog_entry) = crate::provider_catalog::catalog_entry_for_key(&provider_key)
334        else {
335            return Err(SdkError::ProviderNotConfigured {
336                contract_name: format!("provider key {provider_key} is not built into this SDK"),
337            });
338        };
339        if !catalog_entry.served_contracts.contains(&contract) {
340            return Err(SdkError::ProviderNotConfigured {
341                contract_name: crate::provider_catalog::contract_name_of(contract),
342            });
343        }
344        let trimmed_credential = credential.trim();
345        if catalog_entry.credential_required && trimmed_credential.is_empty() {
346            return Err(SdkError::ProviderRequestRejected {
347                detail: format!(
348                    "the provider {} requires a credential before it can serve requests",
349                    catalog_entry.display_name
350                ),
351            });
352        }
353        (catalog_entry.install)(self, contract, trimmed_credential)?;
354        self.record_provider_selection(
355            contract,
356            catalog_entry.provider_key,
357            catalog_entry.display_name,
358        );
359        Ok(())
360    }

362    /// Reports which provider serves each capability, in the order a settings
363    /// surface lists them.
364    ///
365    /// Every capability appears, including the ones with no provider
366    /// installed; those carry an empty key and name with `registered` false.
367    /// The readback never mutates the registry.
368    pub fn provider_registrations(&self) -> Vec<crate::records::ProviderRegistration> {
369        let selections = crate::panic_boundary::state_lock_guard(
370            &self.provider_selections,
371            "the provider selection record",
372        )
373        .clone();
374        let provider_registry = crate::panic_boundary::state_lock_guard(
375            &self.provider_registry,
376            "the provider registry",
377        );
378        rustic_maps_core::ProviderContract::ALL
379            .into_iter()
380            .map(|core_contract| {
381                let contract = ProviderContract::from_core_contract(core_contract);
382                let selection = selections
383                    .iter()
384                    .find(|selection| selection.contract == contract);
385                crate::provider_catalog::registration_entry(
386                    contract,
387                    selection,
388                    provider_registry.is_configured(core_contract),
389                )
390            })
391            .collect()
392    }

410    /// Replays the most recent events the diagnostics rings recorded: at
411    /// most `maximum_count` events oldest first, every
412    /// subsystem's ring interleaved in recording order, filtered by
413    /// `minimum_severity`. The flush is non-destructive — the rings are
414    /// never emptied, so a later flush can redeliver the same events.
415    /// Delivered events are fired-and-forget, never awaited.
416    pub fn flush_recent_diagnostic_events(
417        &self,
418        listener: Arc<dyn crate::diagnostics::SdkDiagnosticEventListener>,
419        minimum_severity: SdkLogSeverity,
420        maximum_count: u32,
421    ) -> u32 {
422        let recent_events = crate::diagnostics::DiagnosticsHandle::shared()
423            .recent_events(minimum_severity, maximum_count);
424        let flush_count = recent_events.len() as u32;
425        for event in recent_events {
426            listener.on_diagnostic_event(event);
427        }
428        flush_count
429    }

532    /// Registers the traffic-flow implementation on the native integrator
533    /// path (the UDL surface carries no provider parameters).
534    ///
535    /// The registered provider is wrapped in the grid-keyed, memory-bounded
536    /// traffic cache: a traffic layer asks for the segments under its camera
537    /// on every settle, and the cache answers a pane it already holds instead
538    /// of asking the provider again. The wrapper is transparent — the same
539    /// segments, the same failures — and a re-registration replaces both the
540    /// provider and its cache.
541    pub fn register_traffic_flow_provider(
542        &self,
543        traffic_flow_provider: Arc<dyn TrafficFlowProvider>,
544    ) {
545        let traffic_flow_current_time_seconds: rustic_maps_core::TrafficFlowCurrentTimeSeconds =
546            Arc::new(rustic_maps_core::system_clock_time_seconds);
547        let provider_for_cache = Arc::clone(&traffic_flow_provider);
548        let mut provider_registry = crate::panic_boundary::state_lock_guard(
549            &self.provider_registry,
550            "the provider registry's contract slots",
551        );
552        match rustic_maps_core::GridCachedTrafficFlowProvider::new(
553            provider_for_cache,
554            rustic_maps_core::TrafficFlowCacheConfiguration::default(),
555            traffic_flow_current_time_seconds,
556        ) {
557            Ok(cached_traffic_flow_provider) => {
558                // The registry keeps the cache handle beside the wrapped
559                // provider, so the statistics readback can report the
560                // counters the provider is actually answered from.
561                let traffic_flow_cache = cached_traffic_flow_provider.cache();
562                provider_registry.register_traffic_flow_provider_with_cache(
563                    Arc::new(cached_traffic_flow_provider),
564                    traffic_flow_cache,
565                );
566            }
567            // The default configuration is valid by construction, so this
568            // arm is unreachable in practice; serving uncached is the
569            // honest fallback if it ever is not.
570            Err(cache_configuration_error) => {
571                tracing::warn!(
572                    target: "rustic_maps::providers",
573                    error = %cache_configuration_error,
574                    "the traffic flow cache configuration was refused; the provider serves uncached"
575                );
576                provider_registry.register_traffic_flow_provider(traffic_flow_provider);
577            }
578        }
579    }

798        for contract in catalog_entry.served_contracts {

806        // any earlier cache installation and its terms are dropped.
807        #[cfg(not(target_arch = "wasm32"))]
808        {
809            self.cancel_all_region_downloads();

818    /// Registers every provider contract the catalog entry named by
819    /// `provider_key` serves from one credential, exactly as
820    /// [`Self::set_provider_credential`] does for the full-service entry,
821    /// wrapping the raster-tile contract in the disk response cache opened
822    /// (or created) as a SQLite database at `tile_cache_database_path` under
823    /// an absolute byte budget of `tile_cache_absolute_maximum_bytes`.
824    /// Cached tile responses follow the tiles category's time-to-live (thirty
825    /// days by default, clamped to the catalog entry's declared caching-terms
826    /// ceiling, applied as
827    /// [`rustic_maps_cache::ResponseCategory::RasterTiles`]'s default); a cache
828    /// miss goes to the network and back-fills the cache, so pre-warmed
829    /// tiles keep rendering whenever the network is unavailable.
830    ///
831    /// The key must name a provider built into this build that serves the
832    /// raster-tile contract and declares caching terms, the path must be
833    /// non-empty and the byte budget positive, and a provider that requires a
834    /// credential must receive a non-blank one; each is refused with its typed
835    /// error. A cache open or quota failure throws with its verbatim detail.
836    /// Re-registration replaces the slots (including the cached raster-tile
837    /// slot).
838    ///
839    /// The disk store is native-only: a wasm build has no SQLite response
840    /// store, so the call validates its inputs and then throws
841    /// [`SdkError::CacheOperationFailure`] with that target limitation.
842    pub fn set_provider_credential_with_tile_cache(
843        &self,
844        provider_key: String,
845        credential: String,
846        tile_cache_database_path: String,
847        tile_cache_absolute_maximum_bytes: u64,
848    ) -> Result<(), SdkError> {
849        if tile_cache_database_path.is_empty() {
850            return Err(SdkError::InvalidCoordinateInput {
851                detail: "the tile cache database path must be non-empty".to_string(),
852            });
853        }
854        if tile_cache_absolute_maximum_bytes == 0 {
855            return Err(SdkError::InvalidCoordinateInput {
856                detail: "the tile cache byte budget must be positive".to_string(),
857            });
858        }
859        let Some(catalog_entry) = crate::provider_catalog::catalog_entry_for_key(&provider_key)
860        else {
861            return Err(SdkError::ProviderNotConfigured {
862                contract_name: format!("provider key {provider_key} is not built into this SDK"),
863            });
864        };
865        if !catalog_entry
866            .served_contracts
867            .contains(&ProviderContract::RasterTiles)
868        {
869            return Err(SdkError::ProviderRequestRejected {
870                detail: format!(
871                    "the provider {} serves no raster tiles, so it cannot install a tile cache",
872                    catalog_entry.display_name
873                ),
874            });
875        }
876        let Some(provider_terms_ceiling_seconds) = catalog_entry.raster_tile_cache_ceiling_seconds
877        else {
878            return Err(SdkError::ProviderRequestRejected {
879                detail: format!(
880                    "the provider {} declares no raster-tile caching terms, so it cannot \
881                     install a tile cache",
882                    catalog_entry.display_name
883                ),
884            });
885        };
886        let trimmed_credential = credential.trim();
887        if catalog_entry.credential_required && trimmed_credential.is_empty() {
888            return Err(SdkError::ProviderRequestRejected {
889                detail: format!(
890                    "the provider {} requires a credential before it can serve requests",
891                    catalog_entry.display_name
892                ),
893            });
894        }
895        #[cfg(not(target_arch = "wasm32"))]
896        {
897            let tile_cache_response_store = rustic_maps_cache::SqliteResponseCache::open(
898                rustic_maps_cache::SqliteResponseCacheConfiguration {
899                    database_path: std::path::PathBuf::from(tile_cache_database_path),
900                    quota: rustic_maps_cache::CacheQuota {
901                        absolute_maximum_bytes: Some(tile_cache_absolute_maximum_bytes),
902                        ..Default::default()
903                    },
904                    ..Default::default()
905                },
906            )
907            .map_err(SdkError::from)?;
908            (catalog_entry.install_all)(self, trimmed_credential)?;
909            let uncached_raster_tile_provider = self.required_raster_tile_provider()?;
910            let default_time_to_live_seconds =
911                rustic_maps_cache::ResponseCategory::RasterTiles.default_time_to_live_seconds();
912            let clamped_time_to_live_seconds = clamped_to_provider_terms_ceiling(
913                default_time_to_live_seconds,
914                Some(provider_terms_ceiling_seconds),
915            );
916            self.register_raster_tile_provider(Arc::new(
917                rustic_maps_cache::CachedRasterTileProvider::new(
918                    Arc::clone(&uncached_raster_tile_provider),
919                    tile_cache_response_store.clone(),
920                )
921                .with_time_to_live_seconds(clamped_time_to_live_seconds)
922                .with_blocking_work_executor(crate::runtime::blocking_work_executor()),
923            ));
924            for contract in catalog_entry.served_contracts {
925                self.record_provider_selection(
926                    *contract,
927                    catalog_entry.provider_key,
928                    catalog_entry.display_name,
929                );
930            }
931            // A replaced installation never keeps serving an in-flight pack
932            // download: signal every download before the swap so none keeps
933            // writing into the store the SDK is about to stop serving.
934            self.cancel_all_region_downloads();
935            *crate::panic_boundary::state_lock_guard(
936                &self.tile_cache_installation,
937                "the tile-cache installation state",
938            ) = Some(TileCacheInstallation {
939                raster_tile_provider: uncached_raster_tile_provider,
940                response_store: tile_cache_response_store,
941                time_to_live_seconds: clamped_time_to_live_seconds,
942                provider_terms_ceiling_seconds: Some(provider_terms_ceiling_seconds),
943                raster_tile_maximum_zoom_level: catalog_entry.raster_tile_maximum_zoom_level,
944            });
945            Ok(())
946        }
947        #[cfg(target_arch = "wasm32")]
948        {
949            let _ = (
950                trimmed_credential,
951                catalog_entry,
952                provider_terms_ceiling_seconds,
953            );
954            Err(SdkError::CacheOperationFailure {
955                detail: "the disk tile cache is unavailable on this target; \
956                         a wasm build has no SQLite response store"
957                    .to_string(),
958            })
959        }
960    }

1009    /// [`Self::raster_tile_cache_terms`] reports the effective value. A zero

1018    ) -> Result<(), SdkError> {

1020            return Err(SdkError::InvalidCoordinateInput {
1021                detail: "the raster-tile cache time-to-live must be positive (seconds)".to_string(),
1022            });
1023        }
1024        #[cfg(not(target_arch = "wasm32"))]
1025        {
1026            let (rebuilt_raster_tile_provider, clamped_time_to_live_seconds) = {
1027                let installation = crate::panic_boundary::state_lock_guard(
1028                    &self.tile_cache_installation,
1029                    "the tile-cache installation state",
1030                );
1031                let tile_cache_installation =
1032                    installation
1033                        .as_ref()
1034                        .ok_or(SdkError::ProviderNotConfigured {
1035                            contract_name: TILE_CACHE_NOT_CONFIGURED_CONTRACT_NAME.to_string(),
1036                        })?;
1037                let clamped_time_to_live_seconds = clamped_to_provider_terms_ceiling(
1038                    time_to_live_seconds,
1039                    tile_cache_installation.provider_terms_ceiling_seconds,
1040                );
1041                (
1042                    Arc::new(
1043                        rustic_maps_cache::CachedRasterTileProvider::new(
1044                            tile_cache_installation.raster_tile_provider.clone(),
1045                            tile_cache_installation.response_store.clone(),
1046                        )

1048                        .with_blocking_work_executor(crate::runtime::blocking_work_executor()),

1051                )
1052            };

1054            // installation lock: the registry's guard is never taken while
1055            // the installation's is held, and vice versa, so the two locks
1056            // cannot dead lock.
1057            self.register_raster_tile_provider(rebuilt_raster_tile_provider);
1058            if let Some(tile_cache_installation) = crate::panic_boundary::state_lock_guard(
1059                &self.tile_cache_installation,
1060                "the tile-cache installation state",
1061            )
1062            .as_mut()
1063            {
1064                tile_cache_installation.time_to_live_seconds = clamped_time_to_live_seconds;
1065            }
1066            Ok(())
1067        }
1068        // No installable disk cache exists on wasm, so the native
1069        // not-configured refusal is the only outcome there.
1070        #[cfg(target_arch = "wasm32")]
1071        {
1072            Err(SdkError::ProviderNotConfigured {
1073                contract_name: TILE_CACHE_NOT_CONFIGURED_CONTRACT_NAME.to_string(),
1074            })
1075        }

1174    /// Registers one in-flight region download and answers the cancellation
1175    /// receiver the download races against its prepare future. A second
1176    /// download of the same region while one is in flight is refused with a
1177    /// typed cache failure naming the region; different regions may run
1178    /// concurrently.
1179    #[cfg(not(target_arch = "wasm32"))]
1180    pub(crate) fn begin_region_download(
1181        &self,
1182        region_id: &str,
1183    ) -> Result<
1184        (
1185            tokio::sync::watch::Receiver<bool>,
1186            Arc<RegionDownloadCancellation>,
1187        ),
1188        SdkError,
1189    > {
1190        let mut region_downloads = crate::panic_boundary::state_lock_guard(
1191            &self.region_downloads,
1192            "the in-flight region download registry",
1193        );
1194        if region_downloads.contains_key(region_id) {
1195            return Err(SdkError::CacheOperationFailure {
1196                detail: format!(
1197                    "a download for the region {region_id:?} is already in flight; \
1198                     cancel it or wait for it to finish"
1199                ),
1200            });
1201        }
1202        let region_download_cancellation = Arc::new(RegionDownloadCancellation::new());
1203        let cancellation_receiver = region_download_cancellation.subscribe();
1204        region_downloads.insert(
1205            region_id.to_string(),
1206            Arc::clone(&region_download_cancellation),
1207        );
1208        Ok((cancellation_receiver, region_download_cancellation))
1209    }
