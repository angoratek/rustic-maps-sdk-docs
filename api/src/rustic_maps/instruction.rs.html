1//! The instruction-text composition the contract exposes to every reference
2//! application.
3//!
4//! The rule itself lives in [`map_refapp_core`]: the composed text first, the
5//! provider-authored text second, the honest unavailable line on the display
6//! channel and silence on the speech channel, plus the roundabout exit
7//! suffix. This module only lowers those answers across the UniFFI boundary;
8//! no platform keeps its own copy of the rule.
9
10/// The display text of one maneuver instruction.
11///
12/// The composed display text wins; when it is absent the provider-authored
13/// display text renders verbatim; when neither exists the honest `No
14/// instruction text provided` line renders.
15///
16/// # Parameters
17/// - `composed_display_text` — the banner variant composed by the contract,
18///   when present.
19/// - `provider_display_text` — the provider-authored display text, verbatim,
20///   when present.
21///
22/// # Returns
23/// The display line, never empty for absent provider text.
24pub fn instruction_display_text(
25    composed_display_text: Option<String>,
26    provider_display_text: Option<String>,
27) -> String {
28    map_refapp_core::instruction_display_text(
29        composed_display_text.as_deref(),
30        provider_display_text.as_deref(),
31    )
32}
33
34/// The speech text of one maneuver instruction.
35///
36/// The composed speech text wins; when it is absent the provider-authored
37/// speech text is returned verbatim. When neither exists the maneuver is
38/// never spoken, so the result is `None`.
39///
40/// # Parameters
41/// - `composed_speech_text` — the text-to-speech variant composed by the
42///   contract, when present.
43/// - `provider_speech_text` — the provider-authored speech text, verbatim,
44///   when present.
45///
46/// # Returns
47/// The line to speak verbatim, or `None` when the provider authored none.
48pub fn instruction_speech_text(
49    composed_speech_text: Option<String>,
50    provider_speech_text: Option<String>,
51) -> Option<String> {
52    map_refapp_core::instruction_speech_text(
53        composed_speech_text.as_deref(),
54        provider_speech_text.as_deref(),
55    )
56    .map(str::to_string)
57}
58
59/// The turn banner's full instruction line.
60///
61/// The line is the display text followed by the roundabout exit suffix: ` (exit
62/// <index>)` for a roundabout maneuver and nothing for every other maneuver.
63///
64/// # Parameters
65/// - `composed_display_text` — the banner variant composed by the contract,
66///   when present.
67/// - `provider_display_text` — the provider-authored display text, verbatim,
68///   when present.
69/// - `roundabout_exit_index` — the structured exit index of a roundabout
70///   maneuver, when the contract carries one.
71///
72/// # Returns
73/// The banner line, including the honest unavailable line when the provider
74/// authored no display text.
75pub fn instruction_banner_text(
76    composed_display_text: Option<String>,
77    provider_display_text: Option<String>,
78    roundabout_exit_index: Option<u8>,
79) -> String {
80    map_refapp_core::instruction_banner_text(
81        composed_display_text.as_deref(),
82        provider_display_text.as_deref(),
83        roundabout_exit_index,
84    )
85}

87#[cfg(test)]
88mod tests {
89    use super::{instruction_banner_text, instruction_display_text, instruction_speech_text};
90
91    fn text(value: &str) -> Option<String> {
92        Some(value.to_string())
93    }
94
95    #[test]
96    fn display_text_priority_and_honest_unavailable_line() {
97        assert_eq!(
98            instruction_display_text(text("In 300 meters, Turn left"), text("Turn left")),
99            "In 300 meters, Turn left"
100        );
101        assert_eq!(
102            instruction_display_text(None, text("Turn left")),
103            "Turn left"
104        );
105        assert_eq!(
106            instruction_display_text(None, None),
107            "No instruction text provided"
108        );
109    }
110
111    #[test]
112    fn speech_text_priority_and_silence_without_provider_text() {
113        assert_eq!(
114            instruction_speech_text(text("In 300 meters, turn left"), text("turn left")),
115            Some("In 300 meters, turn left".to_string())
116        );
117        assert_eq!(
118            instruction_speech_text(None, text("turn left")),
119            Some("turn left".to_string())
120        );
121        assert_eq!(instruction_speech_text(None, None), None);
122    }
123
124    #[test]
125    fn banner_text_appends_the_roundabout_exit_suffix() {
126        assert_eq!(
127            instruction_banner_text(text("Take the roundabout"), None, Some(2)),
128            "Take the roundabout (exit 2)"
129        );
130        assert_eq!(
131            instruction_banner_text(None, text("Take the roundabout"), None),
132            "Take the roundabout"
133        );
134    }
135}
