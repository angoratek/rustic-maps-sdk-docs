1//! The one mapping between the shared provider registry's `served_contracts`
2//! names and the contract's capability enum.
3//!
4//! The shared registry spells every capability in snake_case (`geocoding`,
5//! `distance_matrix`, …) while the frozen contract names the same capabilities
6//! as [`ProviderContract`] variants. The mapping lives here, in the contract,
7//! and the platform surfaces resolve names through these two functions: the
8//! Kotlin registration plan and descriptor accessors, and the Swift registry
9//! catalog and registration path, whose generated bindings carry both calls.
10//! Adding a capability to the registry therefore never requires a per-surface
11//! table.
12//!
13//! [`ProviderContract::registry_name`] is the single table; both functions
14//! read it, and the module's tests prove that every variant has exactly one
15//! name and that parsing round-trips.
16
17use crate::records::ProviderContract;
18
19/// The capability contract a registry `served_contracts` entry denotes.
20///
21/// # Parameters
22/// - `registry_name` — one entry of a provider descriptor's `served_contracts`
23///   list, spelled as the shared registry spells it (`geocoding`,
24///   `distance_matrix`, …).
25///
26/// # Returns
27/// The matching [`ProviderContract`], or `None` when this build's contract
28/// does not define the name. A registry artifact newer than the SDK may
29/// declare a capability this build cannot serve; callers skip it rather than
30/// refuse the whole descriptor.
31///
32/// # Errors
33/// This call cannot fail.
34pub fn provider_contract_from_registry_name(registry_name: String) -> Option<ProviderContract> {
35    ProviderContract::ALL
36        .into_iter()
37        .find(|contract| contract.registry_name() == registry_name.as_str())
38}
39
40/// The shared registry's `served_contracts` name for a capability contract.
41///
42/// # Parameters
43/// - `contract` — the capability to name.
44///
45/// # Returns
46/// The registry's snake_case storage name (`geocoding`, `distance_matrix`, …),
47/// matching the spelling the shared registry uses.
48///
49/// # Errors
50/// This call cannot fail: every [`ProviderContract`] variant carries exactly
51/// one registry name.
52pub fn registry_name_for_provider_contract(contract: ProviderContract) -> String {
53    contract.registry_name().to_owned()
54}

56#[cfg(test)]
57mod tests {
58    use super::{provider_contract_from_registry_name, registry_name_for_provider_contract};
59    use crate::records::ProviderContract;
60
61    /// The registry spellings the shared artifact declares, in the contract's
62    /// stable order. The test compares the contract-owned names against this
63    /// list, so a renamed variant or a typo in either direction fails here.
64    const EXPECTED_REGISTRY_NAMES: [&str; ProviderContract::ALL.len()] = [
65        "geocoding",
66        "routing",
67        "distance_matrix",
68        "place_search",
69        "traffic_incidents",
70        "traffic_flow",
71        "elevation",
72        "raster_tiles",
73        "vector_tiles",
74        "offline_routing",
75    ];
76
77    #[test]
78    fn every_contract_has_exactly_one_registry_name() {
79        for (contract, expected_name) in ProviderContract::ALL
80            .into_iter()
81            .zip(EXPECTED_REGISTRY_NAMES)
82        {
83            let registry_name = registry_name_for_provider_contract(contract);
84            assert_eq!(
85                registry_name, expected_name,
86                "{contract:?} must use the registry's spelling"
87            );
88            assert_eq!(
89                EXPECTED_REGISTRY_NAMES
90                    .iter()
91                    .filter(|candidate| **candidate == expected_name)
92                    .count(),
93                1,
94                "{expected_name:?} must name exactly one contract"
95            );
96        }
97    }
98
99    #[test]
100    fn registry_names_round_trip_through_every_contract() {
101        for contract in ProviderContract::ALL {
102            let registry_name = registry_name_for_provider_contract(contract);
103            assert_eq!(
104                provider_contract_from_registry_name(registry_name.clone()),
105                Some(contract),
106                "{registry_name:?} must parse back to {contract:?}"
107            );
108        }
109    }
110
111    #[test]
112    fn a_registry_name_the_contract_does_not_define_has_no_contract() {
113        assert_eq!(
114            provider_contract_from_registry_name("a-future-capability".to_owned()),
115            None
116        );
117        assert_eq!(provider_contract_from_registry_name(String::new()), None);
118        assert_eq!(
119            provider_contract_from_registry_name("Geocoding".to_owned()),
120            None
121        );
122    }
123}
