1//! The provider catalog: which provider keys this build carries, the
2//! capabilities each serves, and the construction call for each.
3//!
4//! The registry in `rustic-maps-core` resolves calls per capability, so an
5//! application can serve one capability from one provider and the next from
6//! another. This module is the seam that makes that choice reachable from an
7//! application: a key names a provider the binary was built with, and
8//! [`ProviderCatalogEntry::install`] registers exactly the requested
9//! capability, leaving every other capability's slot untouched.
10//!
11//! A provider's name appears here, in its own crate, in its sample-data
12//! directory and in the shared registry descriptor — nowhere else. Adding a
13//! provider is a crate, a descriptor, and one entry in [`PROVIDER_CATALOG`].
14
15use crate::MapSdk;
16use crate::error::SdkError;
17use crate::records::{ProviderContract, ProviderRegistration};
18use std::sync::Arc;
19
20/// One provider this build can construct, keyed by its stable registry key.
21pub(crate) struct ProviderCatalogEntry {
22    /// The provider's stable key, matching its shared-registry descriptor.
23    pub(crate) provider_key: &'static str,
24    /// The provider's display name for a settings surface.
25    pub(crate) display_name: &'static str,
26    /// Whether the provider refuses to serve without a credential.
27    pub(crate) credential_required: bool,
28    /// The capabilities [`Self::install`] can register.
29    pub(crate) served_contracts: &'static [ProviderContract],
30    /// Registers the provider implementation for one capability on the SDK.
31    pub(crate) install: fn(&MapSdk, ProviderContract, &str) -> Result<(), SdkError>,
32    /// Registers every capability in [`Self::served_contracts`] from one
33    /// credential, sharing whatever provider state the implementations share.
34    pub(crate) install_all: fn(&MapSdk, &str) -> Result<(), SdkError>,
35    /// The provider's declared raster-tile caching-terms ceiling in seconds,
36    /// or `None` when the provider declares no ceiling or serves no raster
37    /// tiles. The tile-cache settings surface clamps every requested
38    /// time-to-live through it. The wasm build installs no disk tile cache,
39    /// so it never reads the value.
40    #[cfg_attr(target_arch = "wasm32", allow(dead_code))]
41    pub(crate) raster_tile_cache_ceiling_seconds: Option<u64>,
42    /// The deepest raster-tile zoom level the provider declares it serves, or
43    /// `None` when the provider declares no maximum. An installed tile cache
44    /// carries it into the tile-pack manager, which refuses a region deeper
45    /// than the ceiling before downloading anything. The wasm build installs
46    /// no disk tile cache, so it never reads the value.
47    #[cfg_attr(target_arch = "wasm32", allow(dead_code))]
48    pub(crate) raster_tile_maximum_zoom_level: Option<u8>,
49}

101/// The catalog entry a credential-only registration installs: the first
102/// credentialed provider, which the catalog orders first because a
103/// credential-only call addresses the full-service provider.
104pub(crate) fn credentialed_catalog_entry() -> Option<&'static ProviderCatalogEntry> {
105    PROVIDER_CATALOG
106        .iter()
107        .find(|entry| entry.credential_required)
108}

110/// The credential a keyless provider reads as an optional deployment address.
111///
112/// A self-hosted routing engine has no key: the application supplies the base
113/// URL of the deployment it uses, or leaves it empty to reach the crate's
114/// documented local default.
115fn deployment_base_url_or_default(credential: &str, default_base_url: &str) -> String {
116    let trimmed_credential = credential.trim();
117    if trimmed_credential.is_empty() {
118        default_base_url.to_string()
119    } else {
120        trimmed_credential.to_string()
121    }
122}

124/// Registers the OSRM routing implementation for its one capability.
125///
126/// The credential, when non-empty, is the deployment's base URL; an empty one
127/// targets the crate's documented local default. A malformed URL is refused
128/// by the provider crate before anything is registered.
129fn install_osrm_contract(
130    map_sdk: &MapSdk,
131    contract: ProviderContract,
132    credential: &str,
133) -> Result<(), SdkError> {
134    if contract != ProviderContract::Routing {
135        return Err(SdkError::ProviderNotConfigured {
136            contract_name: contract_name_of(contract),
137        });
138    }
139    let configuration = osrm_provider::OsrmProviderConfiguration {
140        base_url: deployment_base_url_or_default(credential, osrm_provider::DEFAULT_OSRM_BASE_URL),
141        ..Default::default()
142    };
143    let provider = osrm_provider::OsrmRoutingProvider::new(configuration)?;
144    map_sdk.register_routing_provider(Arc::new(provider));
145    Ok(())
146}
147
148/// Registers the OSRM routing implementation for every capability it serves.
149fn install_all_osrm_contracts(map_sdk: &MapSdk, credential: &str) -> Result<(), SdkError> {
150    install_osrm_contract(map_sdk, ProviderContract::Routing, credential)
151}

153/// Registers the Valhalla routing implementation for its one capability.
154///
155/// The credential, when non-empty, is the deployment's base URL; an empty one

157fn install_valhalla_contract(
158    map_sdk: &MapSdk,

185/// Finds the catalog entry for a provider key.
186pub(crate) fn catalog_entry_for_key(provider_key: &str) -> Option<&'static ProviderCatalogEntry> {
187    PROVIDER_CATALOG
188        .iter()
189        .find(|entry| entry.provider_key == provider_key)
190}

192/// Registers one capability from the provider set the shared client serves.
193///
194/// The client is constructed per call, so a credential change takes effect
195/// immediately and a refused credential leaves the previously serving
196/// provider in place (nothing is registered until the client builds).
197fn install_mapquest_contract(
198    map_sdk: &MapSdk,
199    contract: ProviderContract,
200    credential: &str,
201) -> Result<(), SdkError> {
202    register_mapquest_contract(map_sdk, contract, mapquest_provider_client(credential)?)
203}
204
205/// Registers every capability the provider set serves from one shared client,
206/// so the set keeps one rate limiter and one envelope validator.
207fn install_all_mapquest_contracts(map_sdk: &MapSdk, credential: &str) -> Result<(), SdkError> {
208    let provider_client = mapquest_provider_client(credential)?;
209    for contract in [
210        ProviderContract::Geocoding,
211        ProviderContract::Routing,
212        ProviderContract::DistanceMatrix,
213        ProviderContract::PlaceSearch,
214        ProviderContract::TrafficIncidents,
215        ProviderContract::TrafficFlow,
216        ProviderContract::RasterTiles,
217    ] {
218        register_mapquest_contract(map_sdk, contract, Arc::clone(&provider_client))?;
219    }
220    Ok(())
221}

223/// Builds the provider set's shared client from a credential.
224fn mapquest_provider_client(
225    credential: &str,
226) -> Result<Arc<mapquest_provider::MapQuestProviderClient>, SdkError> {
227    Ok(Arc::new(
228        mapquest_provider::MapQuestProviderClient::new(
229            mapquest_provider::MapQuestProviderConfiguration {
230                api_key: credential.to_string(),
231                ..Default::default()
232            },
233        )
234        .map_err(SdkError::from)?,
235    ))
236}

238/// Registers one capability of the provider set onto the SDK.
239fn register_mapquest_contract(
240    map_sdk: &MapSdk,
241    contract: ProviderContract,
242    provider_client: Arc<mapquest_provider::MapQuestProviderClient>,
243) -> Result<(), SdkError> {
244    match contract {
245        ProviderContract::Geocoding => {
246            map_sdk.register_geocoding_provider(Arc::new(
247                mapquest_provider::MapQuestGeocodingProvider::new(provider_client),
248            ));
249            Ok(())
250        }
251        ProviderContract::Routing => {
252            map_sdk.register_routing_provider(Arc::new(
253                mapquest_provider::MapQuestRoutingProvider::new(provider_client),
254            ));
255            Ok(())
256        }
257        ProviderContract::DistanceMatrix => {
258            map_sdk.register_distance_matrix_provider(Arc::new(
259                mapquest_provider::MapQuestDistanceMatrixProvider::new(provider_client),
260            ));
261            Ok(())
262        }
263        ProviderContract::PlaceSearch => {
264            map_sdk.register_search_place_provider(Arc::new(
265                mapquest_provider::MapQuestSearchPlaceProvider::new(provider_client),
266            ));
267            Ok(())
268        }
269        ProviderContract::TrafficIncidents => {
270            map_sdk.register_traffic_incident_provider(Arc::new(
271                mapquest_provider::MapQuestTrafficIncidentProvider::new(provider_client),
272            ));
273            Ok(())
274        }
275        ProviderContract::TrafficFlow => {
276            map_sdk.register_traffic_flow_provider(Arc::new(
277                mapquest_provider::MapQuestTrafficFlowProvider::new(provider_client),
278            ));
279            Ok(())
280        }
281        ProviderContract::RasterTiles => {
282            map_sdk.register_raster_tile_provider(Arc::new(
283                mapquest_provider::MapQuestRasterTileProvider::new(provider_client),
284            ));
285            Ok(())
286        }
287        // The catalog's served-contract list gates every other capability
288        // before this call, so this arm is the compiler asking for totality
289        // rather than a reachable path.
290        ProviderContract::Elevation
291        | ProviderContract::VectorTiles
292        | ProviderContract::OfflineRouting => Err(SdkError::ProviderNotConfigured {
293            contract_name: contract_name_of(contract),
294        }),
295    }
296}

298/// The name a capability reports in a not-configured detail.
299pub(crate) fn contract_name_of(contract: ProviderContract) -> String {
300    contract.to_core_contract().contract_name().to_string()
301}

303/// Builds one readback entry from the registry state and the recorded
304/// selection.
305pub(crate) fn registration_entry(
306    contract: ProviderContract,
307    selection: Option<&ProviderSelection>,
308    registry_configured: bool,
309) -> ProviderRegistration {
310    let registered = registry_configured || selection.is_some();
311    ProviderRegistration {
312        contract,
313        provider_key: selection
314            .map(|selection| selection.provider_key.clone())
315            .unwrap_or_default(),
316        display_name: selection
317            .map(|selection| selection.display_name.clone())
318            .unwrap_or_default(),
319        registered,
320    }
321}

323/// The provider an application chose for one capability, kept so the readback
324/// can name it: the registry itself answers only whether its slot is filled.
325#[derive(Debug, Clone, PartialEq)]
326pub(crate) struct ProviderSelection {
327    /// The capability the selection applies to.
328    pub(crate) contract: ProviderContract,
329    /// The provider's stable key.
330    pub(crate) provider_key: String,
331    /// The provider's display name.
332    pub(crate) display_name: String,
333}

335#[cfg(test)]
336mod tests {
337    use super::{PROVIDER_CATALOG, catalog_entry_for_key};
338
339    #[test]
340    fn the_catalog_declares_the_mapquest_raster_zoom_ceiling() {
341        let mapquest_entry =
342            catalog_entry_for_key("mapquest").expect("the full-service provider is catalogued");
343        assert_eq!(
344            mapquest_entry.raster_tile_maximum_zoom_level,
345            Some(mapquest_provider::tiles::MAPQUEST_RASTER_TILE_MAXIMUM_ZOOM_LEVEL),
346            "the credentialed provider's declared ceiling reaches the tile-pack manager"
347        );
348        assert_eq!(
349            mapquest_entry.raster_tile_maximum_zoom_level,
350            Some(20),
351            "the documented MapQuest Tiles API maximum is twenty"
352        );
353    }
354
355    #[test]
356    fn catalog_entries_without_raster_tiles_declare_no_zoom_ceiling() {
357        for provider_key in ["osrm", "valhalla"] {
358            let catalog_entry = catalog_entry_for_key(provider_key)
359                .expect("the single-capability routing providers are catalogued");
360            assert_eq!(
361                catalog_entry.raster_tile_maximum_zoom_level, None,
362                "{provider_key} serves no raster tiles and declares no zoom ceiling"
363            );
364        }
365        assert_eq!(
366            PROVIDER_CATALOG
367                .iter()
368                .filter(|entry| entry.raster_tile_maximum_zoom_level.is_some())
369                .count(),
370            1,
371            "only the raster-serving provider declares a zoom ceiling"
372        );
373    }
