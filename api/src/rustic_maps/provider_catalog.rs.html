1//! The provider catalog: which provider keys this build carries, the
2//! capabilities each serves, and the construction call for each.
3//!
4//! The registry in `rustic-maps-core` resolves calls per capability, so an
5//! application can serve one capability from one provider and the next from
6//! another. This module is the seam that makes that choice reachable from an
7//! application: a key names a provider the binary was built with, and
8//! [`ProviderCatalogEntry::install`] registers exactly the requested
9//! capability, leaving every other capability's slot untouched.
10//!
11//! A provider's name appears here, in its own crate, in its sample-data
12//! directory and in the shared registry descriptor — nowhere else. Adding a
13//! provider is a crate, a descriptor, and one entry in [`PROVIDER_CATALOG`].
14
15use crate::MapSdk;
16use crate::error::SdkError;
17use crate::records::{ProviderContract, ProviderRegistration};
18use std::sync::Arc;

20/// One provider this build can construct, keyed by its stable registry key.
21pub(crate) struct ProviderCatalogEntry {
22    /// The provider's stable key, matching its shared-registry descriptor.
23    pub(crate) provider_key: &'static str,
24    /// The provider's display name for a settings surface.
25    pub(crate) display_name: &'static str,
26    /// Whether the provider refuses to serve without a credential.
27    pub(crate) credential_required: bool,
28    /// The capabilities [`Self::install`] can register.
29    pub(crate) served_contracts: &'static [ProviderContract],
30    /// Registers the provider implementation for one capability on the SDK.
31    pub(crate) install: fn(&MapSdk, ProviderContract, &str) -> Result<(), SdkError>,
32    /// Registers every capability in [`Self::served_contracts`] from one
33    /// credential, sharing whatever provider state the implementations share.
34    pub(crate) install_all: fn(&MapSdk, &str) -> Result<(), SdkError>,
35    /// The provider's declared raster-tile caching-terms ceiling in seconds,
36    /// or `None` when the provider declares no ceiling or serves no raster
37    /// tiles. The tile-cache settings surface clamps every requested
38    /// time-to-live through it. The wasm build installs no disk tile cache,
39    /// so it never reads the value.
40    #[cfg_attr(target_arch = "wasm32", allow(dead_code))]
41    pub(crate) raster_tile_cache_ceiling_seconds: Option<u64>,
42    /// The deepest raster-tile zoom level the provider declares it serves, or
43    /// `None` when the provider declares no maximum. An installed tile cache
44    /// carries it into the tile-pack manager, which refuses a region deeper
45    /// than the ceiling before downloading anything. The wasm build installs
46    /// no disk tile cache, so it never reads the value.
47    #[cfg_attr(target_arch = "wasm32", allow(dead_code))]
48    pub(crate) raster_tile_maximum_zoom_level: Option<u8>,
49}

111/// The catalog entry a credential-only registration installs: the first
112/// credentialed provider, which the catalog orders first because a
113/// credential-only call addresses the full-service provider.
114pub(crate) fn credentialed_catalog_entry() -> Option<&'static ProviderCatalogEntry> {
115    PROVIDER_CATALOG
116        .iter()
117        .find(|entry| entry.credential_required)
118}

120/// The credential a keyless provider reads as an optional deployment address.
121///
122/// A self-hosted routing engine has no key: the application supplies the base
123/// URL of the deployment it uses, or leaves it empty to reach the crate's
124/// documented local default.
125fn deployment_base_url_or_default(credential: &str, default_base_url: &str) -> String {
126    let trimmed_credential = credential.trim();
127    if trimmed_credential.is_empty() {
128        default_base_url.to_string()
129    } else {
130        trimmed_credential.to_string()
131    }
132}

134/// Registers the OSRM routing implementation for its one capability.
135///
136/// The credential, when non-empty, is the deployment's base URL; an empty one
137/// targets the crate's documented local default. A malformed URL is refused
138/// by the provider crate before anything is registered.
139fn install_osrm_contract(
140    map_sdk: &MapSdk,
141    contract: ProviderContract,
142    credential: &str,
143) -> Result<(), SdkError> {
144    if contract != ProviderContract::Routing {
145        return Err(SdkError::ProviderNotConfigured {
146            contract_name: contract_name_of(contract),
147        });
148    }
149    let configuration = osrm_provider::OsrmProviderConfiguration {
150        base_url: deployment_base_url_or_default(credential, osrm_provider::DEFAULT_OSRM_BASE_URL),
151        ..Default::default()
152    };
153    let provider = osrm_provider::OsrmRoutingProvider::new(configuration)?;
154    map_sdk.register_routing_provider(Arc::new(provider));
155    Ok(())
156}
157
158/// Registers the OSRM routing implementation for every capability it serves.
159fn install_all_osrm_contracts(map_sdk: &MapSdk, credential: &str) -> Result<(), SdkError> {
160    install_osrm_contract(map_sdk, ProviderContract::Routing, credential)
161}

163/// Registers the Photon geocoding and place-search implementations for one
164/// capability each.
165///
166/// The credential, when non-empty, is the deployment's base URL; an empty one
167/// targets the public Photon instance the crate documents. A malformed URL is
168/// refused by the provider crate before anything is registered.
169fn install_photon_contract(
170    map_sdk: &MapSdk,
171    contract: ProviderContract,
172    credential: &str,
173) -> Result<(), SdkError> {
174    let serves_contract = matches!(
175        contract,
176        ProviderContract::Geocoding | ProviderContract::PlaceSearch
177    );
178    if !serves_contract {
179        return Err(SdkError::ProviderNotConfigured {
180            contract_name: contract_name_of(contract),
181        });
182    }
183    let configuration = photon_provider::PhotonProviderConfiguration {
184        base_url: deployment_base_url_or_default(
185            credential,
186            photon_provider::DEFAULT_PHOTON_BASE_URL,
187        ),
188        ..Default::default()
189    };
190    let client = Arc::new(photon_provider::PhotonHttpClient::new(configuration)?);
191    let provider = Arc::new(photon_provider::PhotonPlaceProvider::new(client));
192    match contract {
193        ProviderContract::Geocoding => map_sdk.register_geocoding_provider(provider),
194        ProviderContract::PlaceSearch => map_sdk.register_search_place_provider(provider),
195        _ => {
196            return Err(SdkError::ProviderNotConfigured {
197                contract_name: contract_name_of(contract),
198            });
199        }
200    }
201    Ok(())
202}

204/// Registers the Photon implementations for every capability it serves.
205fn install_all_photon_contracts(map_sdk: &MapSdk, credential: &str) -> Result<(), SdkError> {
206    install_photon_contract(map_sdk, ProviderContract::Geocoding, credential)?;
207    install_photon_contract(map_sdk, ProviderContract::PlaceSearch, credential)
208}

242/// Finds the catalog entry for a provider key.
243pub(crate) fn catalog_entry_for_key(provider_key: &str) -> Option<&'static ProviderCatalogEntry> {
244    PROVIDER_CATALOG
245        .iter()
246        .find(|entry| entry.provider_key == provider_key)
247}

249/// Registers one capability from the provider set the shared client serves.
250///
251/// The client is constructed per call, so a credential change takes effect
252/// immediately and a refused credential leaves the previously serving
253/// provider in place (nothing is registered until the client builds).
254fn install_mapquest_contract(
255    map_sdk: &MapSdk,
256    contract: ProviderContract,
257    credential: &str,
258) -> Result<(), SdkError> {
259    register_mapquest_contract(map_sdk, contract, mapquest_provider_client(credential)?)
260}
261
262/// Registers every capability the provider set serves from one shared client,
263/// so the set keeps one rate limiter and one envelope validator.
264fn install_all_mapquest_contracts(map_sdk: &MapSdk, credential: &str) -> Result<(), SdkError> {
265    let provider_client = mapquest_provider_client(credential)?;
266    for contract in [
267        ProviderContract::Geocoding,
268        ProviderContract::Routing,
269        ProviderContract::DistanceMatrix,
270        ProviderContract::PlaceSearch,
271        ProviderContract::TrafficIncidents,
272        ProviderContract::TrafficFlow,
273        ProviderContract::RasterTiles,
274    ] {
275        register_mapquest_contract(map_sdk, contract, Arc::clone(&provider_client))?;
276    }
277    Ok(())
278}

280/// Builds the provider set's shared client from a credential.
281fn mapquest_provider_client(
282    credential: &str,
283) -> Result<Arc<mapquest_provider::MapQuestProviderClient>, SdkError> {
284    Ok(Arc::new(
285        mapquest_provider::MapQuestProviderClient::new(
286            mapquest_provider::MapQuestProviderConfiguration {
287                api_key: credential.to_string(),
288                ..Default::default()
289            },
290        )
291        .map_err(SdkError::from)?,
292    ))
293}

295/// Registers one capability of the provider set onto the SDK.
296fn register_mapquest_contract(
297    map_sdk: &MapSdk,
298    contract: ProviderContract,
299    provider_client: Arc<mapquest_provider::MapQuestProviderClient>,
300) -> Result<(), SdkError> {
301    match contract {
302        ProviderContract::Geocoding => {
303            map_sdk.register_geocoding_provider(Arc::new(
304                mapquest_provider::MapQuestGeocodingProvider::new(provider_client),
305            ));
306            Ok(())
307        }
308        ProviderContract::Routing => {
309            map_sdk.register_routing_provider(Arc::new(
310                mapquest_provider::MapQuestRoutingProvider::new(provider_client),
311            ));
312            Ok(())
313        }
314        ProviderContract::DistanceMatrix => {
315            map_sdk.register_distance_matrix_provider(Arc::new(
316                mapquest_provider::MapQuestDistanceMatrixProvider::new(provider_client),
317            ));
318            Ok(())
319        }
320        ProviderContract::PlaceSearch => {
321            map_sdk.register_search_place_provider(Arc::new(
322                mapquest_provider::MapQuestSearchPlaceProvider::new(provider_client),
323            ));
324            Ok(())
325        }
326        ProviderContract::TrafficIncidents => {
327            map_sdk.register_traffic_incident_provider(Arc::new(
328                mapquest_provider::MapQuestTrafficIncidentProvider::new(provider_client),
329            ));
330            Ok(())
331        }
332        ProviderContract::TrafficFlow => {
333            map_sdk.register_traffic_flow_provider(Arc::new(
334                mapquest_provider::MapQuestTrafficFlowProvider::new(provider_client),
335            ));
336            Ok(())
337        }
338        ProviderContract::RasterTiles => {
339            map_sdk.register_raster_tile_provider(Arc::new(
340                mapquest_provider::MapQuestRasterTileProvider::new(provider_client),
341            ));
342            Ok(())
343        }
344        // The catalog's served-contract list gates every other capability
345        // before this call, so this arm is the compiler asking for totality
346        // rather than a reachable path.
347        ProviderContract::Elevation
348        | ProviderContract::VectorTiles
349        | ProviderContract::OfflineRouting => Err(SdkError::ProviderNotConfigured {
350            contract_name: contract_name_of(contract),
351        }),
352    }
353}

355/// The name a capability reports in a not-configured detail.
356pub(crate) fn contract_name_of(contract: ProviderContract) -> String {
357    contract.to_core_contract().contract_name().to_string()

360/// Builds one readback entry from the registry state and the recorded
361/// selection.
362pub(crate) fn registration_entry(
363    contract: ProviderContract,
364    selection: Option<&ProviderSelection>,
365    registry_configured: bool,
366) -> ProviderRegistration {
367    let registered = registry_configured || selection.is_some();
368    ProviderRegistration {
369        contract,
370        provider_key: selection
371            .map(|selection| selection.provider_key.clone())
372            .unwrap_or_default(),
373        display_name: selection
374            .map(|selection| selection.display_name.clone())
375            .unwrap_or_default(),
376        registered,
377    }
378}

380/// The provider an application chose for one capability, kept so the readback
381/// can name it: the registry itself answers only whether its slot is filled.
382#[derive(Debug, Clone, PartialEq)]
383pub(crate) struct ProviderSelection {
384    /// The capability the selection applies to.
385    pub(crate) contract: ProviderContract,
386    /// The provider's stable key.
387    pub(crate) provider_key: String,
388    /// The provider's display name.
389    pub(crate) display_name: String,
390}

393mod tests {
394    use super::{PROVIDER_CATALOG, catalog_entry_for_key};

396    #[test]
397    fn the_catalog_declares_the_mapquest_raster_zoom_ceiling() {
398        let mapquest_entry =
399            catalog_entry_for_key("mapquest").expect("the full-service provider is catalogued");
400        assert_eq!(
401            mapquest_entry.raster_tile_maximum_zoom_level,
402            Some(mapquest_provider::tiles::MAPQUEST_RASTER_TILE_MAXIMUM_ZOOM_LEVEL),
403            "the credentialed provider's declared ceiling reaches the tile-pack manager"
404        );
405        assert_eq!(
406            mapquest_entry.raster_tile_maximum_zoom_level,
407            Some(20),
408            "the documented MapQuest Tiles API maximum is twenty"
409        );
410    }
411
412    #[test]
413    fn the_photon_entry_serves_geocoding_and_place_search_keylessly() {
414        let photon_entry =
415            catalog_entry_for_key("photon").expect("the Photon provider is catalogued");
416        assert!(!photon_entry.credential_required);
417        assert!(
418            photon_entry
419                .served_contracts
420                .contains(&crate::records::ProviderContract::Geocoding)
421        );
422        assert!(
423            photon_entry
424                .served_contracts
425                .contains(&crate::records::ProviderContract::PlaceSearch)
426        );
427        assert_eq!(photon_entry.raster_tile_maximum_zoom_level, None);
428    }
429
430    #[test]
431    fn catalog_entries_without_raster_tiles_declare_no_zoom_ceiling() {
432        for provider_key in ["osrm", "valhalla"] {
433            let catalog_entry = catalog_entry_for_key(provider_key)
434                .expect("the single-capability routing providers are catalogued");
435            assert_eq!(
436                catalog_entry.raster_tile_maximum_zoom_level, None,
437                "{provider_key} serves no raster tiles and declares no zoom ceiling"
438            );
439        }
440        assert_eq!(
441            PROVIDER_CATALOG
442                .iter()
443                .filter(|entry| entry.raster_tile_maximum_zoom_level.is_some())
444                .count(),
445            1,
446            "only the raster-serving provider declares a zoom ceiling"
447        );
448    }
