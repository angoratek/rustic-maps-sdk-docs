1//! The diagnostics sink: the `SdkDiagnosticEventListener`
2//! foreign-callback contract foreign code implements to receive the SDK's
3//! structured internal log events, plus the ring-buffer `tracing` subscriber
4//! and the sink's privacy scrubbing behind
5//! `MapSdk::subscribe_to_diagnostic_events` and
6//! `MapSdk::flush_recent_diagnostic_events`.
7//!
8//! # Architecture
9//!
10//! The SDK crates emit plain `tracing` events under per-subsystem targets
11//! (`tile_fetch`, `renderer`, `navigation`, `binding_camera`, `cache`);
12//! [`DiagnosticsHandle::shared`] is the only writer into the bounded rings —
13//! a [`DiagnosticsSubscriber`] installed as the process's global subscriber
14//! at the first `MapSdk` construction — and the FFI seams are two read-only
15//! views of it. Delivery is fire-and-forget: an event is copied into the
16//! ring and to any live listener and the emitting thread moves on
17//! immediately, so a host's slow listener never slows the code whose events
18//! it observes.
19//!
20//! The state is process-wide because a `tracing` global subscriber is: a
21//! second `MapSdk` in the same process shares the single rings-and-listeners
22//! state — the documented, honest shape of a process-wide logging facade. A
23//! host that already installed its own global subscriber keeps it (the host
24//! then owns the process's logging policy, and the rings and the live
25//! stream stay empty — the flush seam answers an empty list).
26//!
27//! # The severity gate
28//!
29//! The rings record at [`DIAGNOSTIC_IDLE_RECORDING_SEVERITY`] or above while
30//! no fine-grained subscription is live. Trace- and debug-severity events —
31//! the GPU-hot render path's instrumentation — are recorded and dispatched
32//! only once a subscription requested them: the subscriber's `enabled`
33//! check consults the gate before any argument formatting runs, so an idle
34//! foreign listener leaves the frame path at near-zero logging cost. The
35//! gate never raises back: once fine-grained events have been subscribed,
36//! they keep recording while the process lives (the rings are bounded).
37//!
38//! # Privacy (scrubbing at the sink)
39//!
40//! A structured event's `latitude` and `longitude` fields are rounded to
41//! [`DIAGNOSTIC_COORDINATE_DECIMALS`] decimals when the line is composed
42//! (≈ 110 m at the equator, a coarse privacy granularity), in exactly one
43//! place — the subscriber's event visitor — regardless of which crate emitted.
44//! Call sites never embed raw coordinates in free-text messages, and the
45//! API key never appears in any event.
46

57/// The capacity of each subsystem's diagnostics ring, in events (bounded so
58/// a chatty subsystem cannot grow the process's memory unboundedly).
59pub const DIAGNOSTIC_RING_CAPACITY: usize = 512;
60
61/// Decimals coordinate values are rounded to at the sink for coarse privacy
62/// granularity (≈ 110 m at the equator).
63pub const DIAGNOSTIC_COORDINATE_DECIMALS: usize = 3;

65/// The severity the rings record down to while no fine-grained subscription
66/// is live: only the coarse (informational and above) events pass, keeping
67/// the GPU-hot frame path's trace instrumentation off when nothing listens.
68pub const DIAGNOSTIC_IDLE_RECORDING_SEVERITY: SdkLogSeverity = SdkLogSeverity::Info;

70/// The named severity ranks (higher is more severe; the intuitive — not the
71/// `tracing` — direction). Only relative order is meaningful.
72pub(crate) const TRACE_SEVERITY_RANK: u8 = 0;
73pub(crate) const DEBUG_SEVERITY_RANK: u8 = 1;
74pub(crate) const INFO_SEVERITY_RANK: u8 = 2;
75pub(crate) const WARN_SEVERITY_RANK: u8 = 3;
76pub(crate) const ERROR_SEVERITY_RANK: u8 = 4;

78/// The numeric severity rank of an `SdkLogSeverity` (higher is more severe).
79pub(crate) const fn severity_rank(severity: SdkLogSeverity) -> u8 {
80    match severity {
81        SdkLogSeverity::Trace => TRACE_SEVERITY_RANK,
82        SdkLogSeverity::Debug => DEBUG_SEVERITY_RANK,
83        SdkLogSeverity::Info => INFO_SEVERITY_RANK,
84        SdkLogSeverity::Warn => WARN_SEVERITY_RANK,
85        SdkLogSeverity::Error => ERROR_SEVERITY_RANK,
86    }
87}

89/// The numeric severity rank of a `tracing` level, in `severity_rank`'s

125/// Whether an event with this module path (or tracing target) originates
126/// from a SDK crate — the sink observes the SDK, not its host.
127pub(crate) fn is_sdk_module_path(module_path_or_target: &str) -> bool {

133// Re-entrancy state of the current thread's event recording: a listener
134// callback emitting its own `tracing` events must not recurse into the
135// sink (the listener's events are the host's, not the SDK's to record).
136thread_local! {
137    static EVENT_RECORDING_IN_FLIGHT: Cell<bool> = const { Cell::new(false) };
138}
139
140/// Marks the current thread as inside a listener dispatch for its lifetime;
141/// while marked, the subscriber drops events (which stops the recursion a
142/// listener-emitted event would otherwise cause).
143struct EventRecordingMarker {
144    _no_construction_without_enter: (),
145}
146
147impl EventRecordingMarker {
148    /// Enters the marked state unless a marker is already live on this
149    /// thread (re-entrancy), in which case `None` says "drop this event".
150    fn enter() -> Option<Self> {
151        EVENT_RECORDING_IN_FLIGHT.with(|in_flight| {
152            if in_flight.get() {
153                None
154            } else {
155                in_flight.set(true);
156                Some(Self {
157                    _no_construction_without_enter: (),
158                })
159            }
160        })
161    }
162}
163
164impl Drop for EventRecordingMarker {
165    fn drop(&mut self) {
166        EVENT_RECORDING_IN_FLIGHT.with(|in_flight| in_flight.set(false));
167    }
168}

170/// One ring entry: the sink's already-scrubbed event (the subsystem target

181/// One live foreign diagnostics listener registration (the subscription
182/// `MapSdk::subscribe_to_diagnostic_events` created).
183struct DiagnosticsListenerRegistration {
184    listener: Arc<dyn SdkDiagnosticEventListener>,
185    minimum_severity: SdkLogSeverity,
186}

188/// The process-wide diagnostics state: the per-subsystem bounded rings, the
189/// live subscription list, and the severity gate, behind the single
190/// `tracing` subscriber. Constructed through [`DiagnosticsHandle::shared`]
191/// (the FFI objects take the leaked handle).
192pub struct DiagnosticsHandle {
193    rings: Mutex<HashMap<String, VecDeque<RecordedEvent>>>,
194    gate: AtomicU8,
195    listeners: Mutex<Vec<DiagnosticsListenerRegistration>>,
196    ring_sequence: AtomicU64,
197}
198
199impl DiagnosticsHandle {
200    /// The process-wide state, installing the global subscriber on first
201    /// use. A host-installed global subscriber keeps precedence (documented
202    /// in the module head): the rings and the live stream then stay empty
203    /// and the flush seam answers an empty list.
204    pub(crate) fn shared() -> &'static DiagnosticsHandle {
205        static SHARED_HANDLE: OnceLock<&'static DiagnosticsHandle> = OnceLock::new();
206        SHARED_HANDLE.get_or_init(|| {
207            let handle: &'static DiagnosticsHandle = Box::leak(Box::new(DiagnosticsHandle {
208                rings: Mutex::new(HashMap::new()),
209                gate: AtomicU8::new(GATE_DEFAULT_RANK),
210                listeners: Mutex::new(Vec::new()),
211                ring_sequence: AtomicU64::new(1),
212            }));
213            let _ = tracing::subscriber::set_global_default(DiagnosticsSubscriber {
214                handle,
215                next_span_id: AtomicU64::new(1),
216            });
217            handle
218        })
219    }
220
221    /// Subscribes a listener to the live event stream; also lowers the
222    /// recording gate to the subscription's minimum severity (the gate never
223    /// raises back, so a later coarser subscription keeps recording the
224    /// finer-grained events).
225    pub(crate) fn subscribe(
226        &'static self,
227        listener: Arc<dyn SdkDiagnosticEventListener>,
228        minimum_severity: SdkLogSeverity,
229    ) {
230        crate::panic_boundary::state_lock_guard(
231            &self.listeners,
232            "the diagnostics listener registry",
233        )
234        .push(DiagnosticsListenerRegistration {
235            listener,
236            minimum_severity,
237        });
238        self.lower_gate_to(severity_rank(minimum_severity));
239    }
240
241    /// Lowers the gate to `rank` if it is finer than the current gate (the
242    /// gate never raises back once fine-grained events have been requested).
243    fn lower_gate_to(&self, rank: u8) {
244        if rank < self.gate.load(Ordering::Relaxed) {
245            self.gate.store(rank, Ordering::Relaxed);
246        }
247    }
248
249    /// Whether the sink records an event of `severity_rank_value` emitted
250    /// from `module_path_or_target`: the event must come from an SDK crate's
251    /// module path (not the host's) and pass the severity gate. The
252    /// subscriber's `enabled` consults exactly this, so no argument
253    /// formatting happens for a rejected event.
254    pub(crate) fn should_record(
255        &self,
256        module_path_or_target: &str,
257        severity_rank_value: u8,
258    ) -> bool {
259        is_sdk_module_path(module_path_or_target)
260            && severity_rank_value >= self.gate.load(Ordering::Relaxed)
261    }
262
263    /// Reads the most recent events across every subsystem's ring, oldest
264    /// first, filtered by `minimum_severity` and capped at `maximum_count`
265    /// (the flush seam; the rings are never emptied, so a later flush can
266    /// redeliver).
267    pub(crate) fn recent_events(
268        &self,
269        minimum_severity: SdkLogSeverity,
270        maximum_count: u32,
271    ) -> Vec<LogEvent> {
272        let minimum_rank = severity_rank(minimum_severity);
273        let mut matched_events: Vec<(String, u64, SdkLogSeverity, u64, String)> = {
274            let rings =
275                crate::panic_boundary::state_lock_guard(&self.rings, "the diagnostics event rings");
276            rings
277                .iter()
278                .flat_map(|(subsystem_target, ring)| {
279                    ring.iter()
280                        .filter(|event| severity_rank(event.severity) >= minimum_rank)
281                        .map(|event| {
282                            (
283                                subsystem_target.clone(),
284                                event.timestamp_milliseconds_since_epoch,
285                                event.severity,
286                                event.sequence,
287                                event.message_line.clone(),
288                            )
289                        })
290                })
291                .collect()
292        };
293        // Oldest first across subsystems, the process-wide sequence breaking
294        // same-millisecond ties; the newest `maximum_count` events are the
295        // requested prefix.
296        matched_events
297            .sort_unstable_by_key(|(_, timestamp, _, sequence, _)| (*timestamp, *sequence));
298        matched_events
299            .into_iter()
300            .rev()
301            .take(maximum_count as usize)
302            .rev()
303            .map(
304                |(
305                    subsystem_target,
306                    timestamp_milliseconds_since_epoch,
307                    severity,
308                    _,
309                    message_line,
310                )| {
311                    LogEvent {
312                        timestamp_milliseconds_since_epoch,
313                        severity,
314                        subsystem_target,
315                        message_line,
316                    }
317                },
318            )
319            .collect()
320    }
321
322    /// Records one scrubbed event into the named subsystem's ring (ring
323    /// retention: at [`DIAGNOSTIC_RING_CAPACITY`] the oldest event drops).
324    fn record(&self, subsystem_target: &str, event: RecordedEvent) {
325        let mut rings =
326            crate::panic_boundary::state_lock_guard(&self.rings, "the diagnostics event rings");
327        let ring = rings
328            .entry(subsystem_target.to_string())
329            .or_insert_with(|| VecDeque::with_capacity(DIAGNOSTIC_RING_CAPACITY));
330        if ring.len() >= DIAGNOSTIC_RING_CAPACITY {
331            ring.pop_front();
332        }
333        ring.push_back(event);
334    }
335
336    /// Dispatches a scrubbed event to every live listener whose minimum
337    /// severity it meets: listener handles are cloned under the listeners
338    /// lock and the callbacks run fire-and-forget, with no lock held across
339    /// a callback and the thread's re-entrancy flag dropping listener-emitted
340    /// events.
341    fn dispatch_live(&self, event: &LogEvent) {
342        let Some(_recording_marker) = EventRecordingMarker::enter() else {
343            return;
344        };
345        let minimum_rank = severity_rank(event.severity);
346        let matched_listeners: Vec<Arc<dyn SdkDiagnosticEventListener>> = {
347            let listeners = crate::panic_boundary::state_lock_guard(
348                &self.listeners,
349                "the diagnostics listener registry",
350            );
351            listeners
352                .iter()
353                .filter(|registration| severity_rank(registration.minimum_severity) <= minimum_rank)
354                .map(|registration| Arc::clone(&registration.listener))
355                .collect()
356        };
357        // No lock is held across the callbacks; the callbacks are
358        // fire-and-forget (a callback's failure cannot fail the caller).
359        for listener in matched_listeners {
360            listener.on_diagnostic_event(event.clone());
361        }
362    }
363}

365/// The `tracing` subscriber that is the diagnostics sink: SDK-crate events
366/// passing the severity gate are scrubbed, ring-recorded and fire-and-forget
367/// dispatched. Spans are not supported (the sink records flat events); a
368/// span's `new_span` hands out a unique id and span bookkeeping is ignored,
369/// so events carry all of their structure themselves.
370struct DiagnosticsSubscriber {
371    handle: &'static DiagnosticsHandle,
372    next_span_id: AtomicU64,
373}
374
375impl Subscriber for DiagnosticsSubscriber {
376    /// The gate check — called by `tracing` before any event argument is
377    /// formatted, which is what keeps the GPU-hot render path at near-zero
378    /// logging cost while the gate is closed to a coarse severity.
379    fn enabled(&self, metadata: &Metadata<'_>) -> bool {
380        let severity_rank_value = severity_rank_for_tracing_level(metadata.level());
381        self.handle.should_record(
382            metadata.module_path().unwrap_or(metadata.target()),
383            severity_rank_value,
384        )
385    }
386
387    fn new_span(&self, _attributes: &tracing::span::Attributes<'_>) -> Id {
388        // The sink supports no span structure; ids exist to satisfy the
389        // subscriber contract and are never dereferenced.
390        Id::from_u64(self.next_span_id.fetch_add(1, Ordering::Relaxed))
391    }
392
393    fn record(&self, _span: &Id, _values: &tracing::span::Record<'_>) {}
394
395    fn record_follows_from(&self, _span: &Id, _follows: &Id) {}
396
397    fn event(&self, event: &Event<'_>) {
398        if !self.enabled(event.metadata()) {
399            return;
400        }
401        let mut visitor = EventLineVisitor::default();
402        event.record(&mut visitor);
403        let EventLineVisitorCollected {
404            message,
405            latitude,
406            longitude,
407            structured_field_pairs,
408        } = visitor.into_collected();
409        // Scrubbing at the sink: the composed line rounds the coordinate
410        // fields to the sink's decimals.
411        let mut message_line = message.unwrap_or_default();
412        if let Some(latitude) = latitude {
413            message_line.push_str(" latitude=");
414            message_line.push_str(&EventLineVisitor::scrub_coordinate(latitude));
415        }
416        if let Some(longitude) = longitude {
417            message_line.push_str(" longitude=");
418            message_line.push_str(&EventLineVisitor::scrub_coordinate(longitude));
419        }
420        for (name, value) in structured_field_pairs {
421            message_line.push_str(&format!(" {name}={value}"));
422        }
423        let severity = tracing_level_severity(event.metadata().level());
424        let timestamp_milliseconds_since_epoch = unix_epoch_milliseconds();
425        let subsystem_target = event.metadata().target().to_string();
426        self.handle.record(
427            &subsystem_target,
428            RecordedEvent {
429                timestamp_milliseconds_since_epoch,
430                severity,
431                sequence: self.handle.ring_sequence.fetch_add(1, Ordering::Relaxed),
432                message_line: message_line.clone(),
433            },
434        );
435        self.handle.dispatch_live(&LogEvent {
436            timestamp_milliseconds_since_epoch,
437            severity,
438            subsystem_target,
439            message_line,
440        });
441    }
442
443    fn enter(&self, _span: &Id) {}
444
445    fn exit(&self, _span: &Id) {}
446
447    /// No span structure: the sink has no current span (events are flat).
448    fn current_span(&self) -> tracing_core::span::Current {
449        tracing_core::span::Current::none()
450    }
451}

457        "TRACE" => SdkLogSeverity::Trace,
458        "DEBUG" => SdkLogSeverity::Debug,

465/// Wall-clock milliseconds since the Unix epoch (the `tracing` sink rings'
466/// timestamp and the provider-failure diagnostic seam's timestamp).
467pub(crate) fn unix_epoch_milliseconds() -> u64 {
468    SystemTime::now()
469        .duration_since(UNIX_EPOCH)
470        .map(|duration| duration.as_millis() as u64)
471        .unwrap_or(0)
472}

474/// The sink's event visitor collecting a `tracing` event's fields. The
475/// privacy scrub — coordinate fields rounded to
476/// [`DIAGNOSTIC_COORDINATE_DECIMALS`] decimals — happens at the sink's
477/// composing step, whatever crate emitted the event.
478#[derive(Default)]
479struct EventLineVisitor {
480    message: Option<String>,
481    latitude: Option<f64>,
482    longitude: Option<f64>,
483    structured_field_pairs: Vec<(String, String)>,
484}
485
486struct EventLineVisitorCollected {
487    message: Option<String>,
488    latitude: Option<f64>,
489    longitude: Option<f64>,
490    structured_field_pairs: Vec<(String, String)>,
491}
492
493impl EventLineVisitor {
494    /// Rounds a coordinate value to the sink's decimals.
495    fn scrub_coordinate(value: f64) -> String {
496        format!("{:.*}", DIAGNOSTIC_COORDINATE_DECIMALS, value)
497    }
498
499    /// Hands the collected fields to the sink's composing step.
500    fn into_collected(self) -> EventLineVisitorCollected {
501        EventLineVisitorCollected {
502            message: self.message,
503            latitude: self.latitude,
504            longitude: self.longitude,
505            structured_field_pairs: self.structured_field_pairs,
506        }
507    }
508
509    /// Scrubs a textual coordinate value: parsed as an f64 it rounds to the
510    /// sink's decimals; un-parseable it is replaced with the placeholder
511    /// (the raw text never reaches the composed line).
512    fn scrub_coordinate_text(text: &str) -> String {
513        match text.parse::<f64>() {
514            Ok(value) => Self::scrub_coordinate(value),
515            Err(_) => SCRUBBED_FIELD_PLACEHOLDER.to_string(),
516        }
517    }
518}
519
520impl Visit for EventLineVisitor {
521    fn record_f64(&mut self, field: &Field, value: f64) {
522        if COORDINATE_FIELD_NAMES.contains(&field.name()) {
523            match field.name() {
524                "latitude" => self.latitude = Some(value),
525                _ => self.longitude = Some(value),
526            }
527            return;
528        }
529        self.structured_field_pairs
530            .push((field.name().to_string(), format!("{value}")));
531    }
532
533    fn record_str(&mut self, field: &Field, value: &str) {
534        if field.name() == "message" {
535            self.message = Some(value.to_string());
536            return;
537        }
538        if COORDINATE_FIELD_NAMES.contains(&field.name()) {
539            self.structured_field_pairs
540                .push((field.name().to_string(), Self::scrub_coordinate_text(value)));
541            return;
542        }
543        self.structured_field_pairs
544            .push((field.name().to_string(), value.to_string()));
545    }
546
547    fn record_debug(&mut self, field: &Field, value: &dyn std::fmt::Debug) {
548        if field.name() == "message" {
549            self.message = Some(format!("{value:?}"));
550            return;
551        }
552        if COORDINATE_FIELD_NAMES.contains(&field.name()) {
553            self.structured_field_pairs.push((
554                field.name().to_string(),
555                Self::scrub_coordinate_text(&format!("{value:?}")),
556            ));
557            return;
558        }
559        self.structured_field_pairs
560            .push((field.name().to_string(), format!("{value:?}")));
561    }
562}

564/// The exported [`SdkDiagnosticEventListener`] contract: foreign code
565/// implements it to receive the SDK's structured internal log events —
566/// either streamed live (through
567/// `MapSdk::subscribe_to_diagnostic_events`, which also lowers the sink's
568/// severity gate) or replayed from the bounded recent-events rings (through
569/// `MapSdk::flush_recent_diagnostic_events`). Callbacks are
570/// fire-and-forget: the sink cannot fail on a listener's failure, and a
571/// listener emitting its own events is not recursed into.
572///
573/// Coordinates in event messages are rounded at the sink to
574/// [`DIAGNOSTIC_COORDINATE_DECIMALS`] decimals; the API key never appears
575/// in a message.
576pub trait SdkDiagnosticEventListener: Send + Sync {
577    /// Receives one structured log event of the SDK's internals.
578    fn on_diagnostic_event(&self, event: LogEvent);
579
580    /// Internally required by the uniffi foreign-handle machinery (the
581    /// foreign-handle probe of the uniffi 0.31 trait-object lowering);
582    /// Rust-side implementations keep the default, which reports a
583    /// Rust-implemented object.
584    #[doc(hidden)]
585    fn uniffi_foreign_handle(&self) -> Option<::uniffi::Handle> {
586        Option::None
587    }
588}

606    /// The wall clock reads from the platform's real time source on every
607    /// target: a plausible Unix-epoch millisecond value that never decreases

838    /// A diagnostics listener keeping every event it receives.
839    #[derive(Default)]
840    struct RecordingBridgeListener {
841        events: Mutex<Vec<LogEvent>>,
842    }
843
844    impl SdkDiagnosticEventListener for RecordingBridgeListener {
845        fn on_diagnostic_event(&self, event: LogEvent) {
846            self.events
847                .lock()
848                .expect("the bridge test's event list is never poisoned")
849                .push(event);
850        }
851    }
852
853    impl RecordingBridgeListener {
854        /// The provider-request observations the listener received for one
855        /// probe provider.
856        fn observations_for(&self, provider_key: &str) -> Vec<LogEvent> {
857            let provider_key_field = format!("provider_key={provider_key}");
858            self.events
859                .lock()
860                .expect("the bridge test's event list is never poisoned")
861                .iter()
862                .filter(|event| {
863                    event.subsystem_target == provider_transport::PROVIDER_REQUEST_DIAGNOSTIC_TARGET
864                        && event.message_line.contains(&provider_key_field)
865                })
866                .cloned()
867                .collect()
868        }
869    }

871    /// One provider request reaches both diagnostics seams: the observation
872    /// the transports emit becomes one Info event under the transport's
873    /// target, streamed to a live subscription and replayable from the
874    /// bounded ring. The redacted path is the observation's, so the
875    /// credential never appears.
876    #[tokio::test]
877    async fn a_provider_request_observation_reaches_the_subscription_and_the_ring() {
878        const PROBE_PROVIDER_KEY: &str = "diagnostic-probe-provider";
879        const PROBE_CONTRACT: &str = "RoutingProvider";
880        const PROBE_CREDENTIAL: &str = "SECRET_PROBE_VALUE";
881
882        // The SDK construction installs the diagnostics subscriber and the
883        // transport observation bridge, so this exercises the real seam.
884        let map_sdk = crate::provider::MapSdk::new();
885        let subscription_listener = Arc::new(RecordingBridgeListener::default());
886        map_sdk.subscribe_to_diagnostic_events(
887            Arc::clone(&subscription_listener) as Arc<dyn SdkDiagnosticEventListener>,
888            SdkLogSeverity::Info,
889        );
890
891        let observed_transport = provider_transport::ObservedTransport::new(
892            Arc::new(FixedObservationTransport {
893                status_code: 200,
894                body: b"{}".to_vec(),
895            }),
896            PROBE_CONTRACT,
897            PROBE_PROVIDER_KEY,
898        );
899        let request = provider_transport::HttpRequest {
900            method: provider_transport::HttpMethod::Get,
901            url: format!("https://provider.invalid/route/v1/driving/a;b?key={PROBE_CREDENTIAL}"),
902            headers: Vec::new(),
903            body: None,
904        };
905        let response = provider_transport::HttpTransport::execute(&observed_transport, request)
906            .await
907            .expect("the fixed transport always answers");
908        assert_eq!(response.status_code, 200);
909
910        let streamed_observations = subscription_listener.observations_for(PROBE_PROVIDER_KEY);
911        assert_eq!(
912            streamed_observations.len(),
913            1,
914            "one observation is streamed per request"
915        );
916        let observation_line = &streamed_observations[0];
917        assert_eq!(observation_line.severity, SdkLogSeverity::Info);
918        for expected_fragment in [
919            "provider request completed",
920            "provider_contract=RoutingProvider",
921            "http_method=GET",
922            "request_path=/route/v1/driving/a;b?key=<redacted>",
923            "status_class=2xx",
924            "outcome=success",
925            "response_bytes=2",
926        ] {
927            assert!(
928                observation_line.message_line.contains(expected_fragment),
929                "the observation line carries {expected_fragment:?}: {}",
930                observation_line.message_line
931            );
932        }
933        assert!(
934            observation_line
935                .message_line
936                .contains("duration_milliseconds="),
937            "the observation reports its duration: {}",
938            observation_line.message_line
939        );
940        assert!(
941            !observation_line.message_line.contains(PROBE_CREDENTIAL),
942            "the credential never reaches the diagnostics line: {}",
943            observation_line.message_line
944        );
945
946        // The bounded ring holds the same event for a later flush.
947        let flush_listener = Arc::new(RecordingBridgeListener::default());
948        let flushed_count = map_sdk.flush_recent_diagnostic_events(
949            Arc::clone(&flush_listener) as Arc<dyn SdkDiagnosticEventListener>,
950            SdkLogSeverity::Info,
951            u32::MAX,
952        );
953        assert!(
954            flushed_count >= 1,
955            "the flush delivered the recorded events"
956        );
957        assert_eq!(
958            flush_listener.observations_for(PROBE_PROVIDER_KEY).len(),
959            1,
960            "the ring replays the request observation"
961        );
962    }

964    /// The live dispatch delivers only to listeners that meet the event's
965    /// severity and fire-and-forgets; a listener re-emitting its own
966    /// `tracing` event while being called is not recursed into (the marker
967    /// drops the re-emitted event).
968    #[test]
969    fn dispatch_live_matches_severity_and_survives_re_emit() {
970        use std::sync::Mutex as TestMutex;
971        struct RecordingTestListener {
972            received: TestMutex<Vec<String>>,
973        }
974        impl SdkDiagnosticEventListener for RecordingTestListener {
975            fn on_diagnostic_event(&self, event: LogEvent) {
976                if let Ok(mut received) = self.received.lock() {
977                    received.push(event.message_line);
978                }
979                // Re-emit through the subscriber while being called: the
980                // marking drops the re-emitted event (no recursion).
981                tracing::info!(target: "rustic_maps::tests", "listener re-emits");
982            }
983        }
984
985        let listener = Arc::new(RecordingTestListener {
986            received: TestMutex::new(Vec::new()),
987        });
988        let handle: &'static DiagnosticsHandle = Box::leak(Box::new(fresh_handle()));
989        handle.subscribe(
990            Arc::clone(&listener) as Arc<dyn SdkDiagnosticEventListener>,
991            SdkLogSeverity::Info,
992        );
993        handle.dispatch_live(&LogEvent {
994            timestamp_milliseconds_since_epoch: 0,
995            severity: SdkLogSeverity::Debug,
996            subsystem_target: "rustic_maps::tests".to_string(),
997            message_line: "below the subscription".to_string(),
998        });
999        handle.dispatch_live(&LogEvent {
1000            timestamp_milliseconds_since_epoch: 1,
1001            severity: SdkLogSeverity::Info,
1002            subsystem_target: "rustic_maps::tests".to_string(),
1003            message_line: "delivered".to_string(),
1004        });
1005        let received = listener
1006            .received
1007            .lock()
1008            .expect("the test listener's mutex is only held by this test");
1009        assert_eq!(
1010            received.as_slice(),
1011            ["delivered"],
1012            "only severity-meeting events are dispatched"
1013        );
