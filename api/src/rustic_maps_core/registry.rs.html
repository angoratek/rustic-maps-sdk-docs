1//! Provider registry: one optional slot per provider contract.
2//!
3//! The application registers its provider implementations at startup and the
4//! rest of the SDK resolves them by contract. Nothing in this module depends
5//! on a concrete provider.
6
7use geospatial_traits::{
8    DistanceMatrixProvider, ElevationProvider, GeocodingProvider, OfflineRoutingProvider,
9    RasterTileProvider, RoutingProvider, SearchPlaceProvider, TrafficFlowProvider,
10    TrafficIncidentProvider, VectorTileProvider,
11};
12use geospatial_types::SDKError;
13use std::sync::Arc;
14
15/// The contract a registry slot serves.
16///
17/// Each variant names one capability an application can configure
18/// independently, which is what lets a map draw its imagery from one provider
19/// while another plans its routes. [`ProviderContract::contract_name`] is the
20/// name the [`SDKError::ProviderNotConfigured`] detail carries.
21#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, PartialOrd, Ord)]
22pub enum ProviderContract {
23    /// Forward and reverse address lookup.
24    Geocoding,
25    /// Route calculation through ordered waypoints.
26    Routing,
27    /// Travel-time and distance matrix calculation.
28    DistanceMatrix,
29    /// Place prediction and lookup by free text.
30    PlaceSearch,
31    /// Traffic incident reports.
32    TrafficIncidents,
33    /// Traffic flow along road segments.
34    TrafficFlow,
35    /// Terrain elevation lookup.
36    Elevation,
37    /// Raster map imagery.
38    RasterTiles,
39    /// Vector map tiles.
40    VectorTiles,
41    /// Route calculation on a compiled offline graph.
42    OfflineRouting,
43}
44
45impl ProviderContract {
46    /// Every contract the registry defines, in the order a settings surface
47    /// lists them: the request/response capabilities first, then the two map
48    /// imagery contracts, then offline routing.
49    pub const ALL: [Self; 10] = [
50        Self::Geocoding,
51        Self::Routing,
52        Self::DistanceMatrix,
53        Self::PlaceSearch,
54        Self::TrafficIncidents,
55        Self::TrafficFlow,
56        Self::Elevation,
57        Self::RasterTiles,
58        Self::VectorTiles,
59        Self::OfflineRouting,
60    ];
61
62    /// The name this contract reports in a
63    /// [`SDKError::ProviderNotConfigured`] detail.
64    pub fn contract_name(self) -> &'static str {
65        match self {
66            Self::Geocoding => "GeocodingProvider",
67            Self::Routing => "RoutingProvider",
68            Self::DistanceMatrix => "DistanceMatrixProvider",
69            Self::PlaceSearch => "SearchPlaceProvider",
70            Self::TrafficIncidents => "TrafficIncidentProvider",
71            Self::TrafficFlow => "TrafficFlowProvider",
72            Self::Elevation => "ElevationProvider",
73            Self::RasterTiles => "RasterTileProvider",
74            Self::VectorTiles => "VectorTileProvider",
75            Self::OfflineRouting => "OfflineRoutingProvider",
76        }
77    }
78}

80/// Registry of provider implementations keyed by contract, not by vendor.
81///
82/// Each contract has one optional slot. Slots start unconfigured. Callers
83/// resolve them with the `require_*` methods, which report
84/// [`SDKError::ProviderNotConfigured`] when the application did not register
85/// that provider.
86///
87/// The registry is cheap to share after configuration: every provider is
88/// stored behind an [`Arc`].
89#[derive(Default)]
90pub struct ProviderRegistry {
91    geocoding_provider: Option<Arc<dyn GeocodingProvider>>,
92    routing_provider: Option<Arc<dyn RoutingProvider>>,
93    distance_matrix_provider: Option<Arc<dyn DistanceMatrixProvider>>,
94    search_place_provider: Option<Arc<dyn SearchPlaceProvider>>,
95    traffic_incident_provider: Option<Arc<dyn TrafficIncidentProvider>>,
96    traffic_flow_provider: Option<Arc<dyn TrafficFlowProvider>>,
97    elevation_provider: Option<Arc<dyn ElevationProvider>>,
98    raster_tile_provider: Option<Arc<dyn RasterTileProvider>>,
99    vector_tile_provider: Option<Arc<dyn VectorTileProvider>>,
100    offline_routing_provider: Option<Arc<dyn OfflineRoutingProvider>>,
101}

136fn not_configured_error(contract_name: &str) -> SDKError {
137    SDKError::ProviderNotConfigured(contract_name.to_string())
138}

277    /// Whether the contract's slot holds a provider.
278    ///
279    /// The check answers without constructing anything, so a settings surface
280    /// can list which capabilities are served before a call is attempted.
281    pub fn is_configured(&self, contract: ProviderContract) -> bool {
282        match contract {
283            ProviderContract::Geocoding => self.geocoding_provider.is_some(),
284            ProviderContract::Routing => self.routing_provider.is_some(),
285            ProviderContract::DistanceMatrix => self.distance_matrix_provider.is_some(),
286            ProviderContract::PlaceSearch => self.search_place_provider.is_some(),
287            ProviderContract::TrafficIncidents => self.traffic_incident_provider.is_some(),
288            ProviderContract::TrafficFlow => self.traffic_flow_provider.is_some(),
289            ProviderContract::Elevation => self.elevation_provider.is_some(),
290            ProviderContract::RasterTiles => self.raster_tile_provider.is_some(),
291            ProviderContract::VectorTiles => self.vector_tile_provider.is_some(),
292            ProviderContract::OfflineRouting => self.offline_routing_provider.is_some(),
293        }
294    }

296#[cfg(test)]
297mod tests {
298    use super::ProviderRegistry;
299    use geospatial_traits::mock::{
300        MockGeocodingProvider, MockRasterTileProvider, MockVectorTileProvider,
301    };
302    use geospatial_types::SDKError;
303    use std::sync::Arc;
304
305    #[test]
306    fn unregistered_slots_report_provider_not_configured() {
307        let registry = ProviderRegistry::new();
308        let failure = match registry.require_geocoding_provider() {
309            Err(failure) => failure,
310            Ok(_) => panic!("expected ProviderNotConfigured for GeocodingProvider"),
311        };
312        assert_eq!(
313            failure,
314            SDKError::ProviderNotConfigured("GeocodingProvider".to_string())
315        );
316    }
317
318    #[test]
319    fn registered_slots_resolve_through_trait_objects() {
320        let mut registry = ProviderRegistry::new();
321        registry.register_geocoding_provider(Arc::new(MockGeocodingProvider));
322        registry.register_raster_tile_provider(Arc::new(MockRasterTileProvider));
323
324        assert!(registry.require_geocoding_provider().is_ok());
325        assert!(registry.require_raster_tile_provider().is_ok());
326        assert!(registry.require_routing_provider().is_err());
327    }
328
329    #[test]
330    fn vector_tile_slot_resolves_through_trait_objects() {
331        let mut registry = ProviderRegistry::new();
332        registry.register_vector_tile_provider(Arc::new(MockVectorTileProvider));
333        assert!(registry.require_vector_tile_provider().is_ok());
334        let failure = match registry.require_routing_provider() {
335            Err(failure) => failure,
336            Ok(_) => panic!("expected ProviderNotConfigured for RoutingProvider"),
337        };
338        assert_eq!(
339            failure,
340            SDKError::ProviderNotConfigured("RoutingProvider".to_string())
341        );
342    }
