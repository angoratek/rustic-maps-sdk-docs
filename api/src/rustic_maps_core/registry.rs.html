1//! Provider registry: one optional slot per provider contract.
2//!
3//! The application registers its provider implementations at startup and the
4//! rest of the SDK resolves them by contract. Nothing in this module depends
5//! on a concrete provider.
6
7use crate::traffic_flow_cache::{TrafficFlowCacheReadback, TrafficFlowGridCache};
8use geospatial_traits::{
9    DistanceMatrixProvider, ElevationProvider, GeocodingProvider, OfflineRoutingProvider,
10    RasterTileProvider, RoutingProvider, SearchPlaceProvider, TrafficFlowProvider,
11    TrafficIncidentProvider, VectorTileProvider,
12};
13use geospatial_types::SDKError;
14use std::sync::Arc;

16/// The contract a registry slot serves.
17///
18/// Each variant names one capability an application can configure
19/// independently, which is what lets a map draw its imagery from one provider
20/// while another plans its routes. [`ProviderContract::contract_name`] is the
21/// name the [`SDKError::ProviderNotConfigured`] detail carries.
22#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, PartialOrd, Ord)]
23pub enum ProviderContract {
24    /// Forward and reverse address lookup.
25    Geocoding,
26    /// Route calculation through ordered waypoints.
27    Routing,
28    /// Travel-time and distance matrix calculation.
29    DistanceMatrix,
30    /// Place prediction and lookup by free text.
31    PlaceSearch,
32    /// Traffic incident reports.
33    TrafficIncidents,
34    /// Traffic flow along road segments.
35    TrafficFlow,
36    /// Terrain elevation lookup.
37    Elevation,
38    /// Raster map imagery.
39    RasterTiles,
40    /// Vector map tiles.
41    VectorTiles,
42    /// Route calculation on a compiled offline graph.
43    OfflineRouting,
44}
45
46impl ProviderContract {
47    /// Every contract the registry defines, in the order a settings surface
48    /// lists them: the request/response capabilities first, then the two map
49    /// imagery contracts, then offline routing.
50    pub const ALL: [Self; 10] = [
51        Self::Geocoding,
52        Self::Routing,
53        Self::DistanceMatrix,
54        Self::PlaceSearch,
55        Self::TrafficIncidents,
56        Self::TrafficFlow,
57        Self::Elevation,
58        Self::RasterTiles,
59        Self::VectorTiles,
60        Self::OfflineRouting,
61    ];
62
63    /// The name this contract reports in a
64    /// [`SDKError::ProviderNotConfigured`] detail.
65    pub fn contract_name(self) -> &'static str {
66        match self {
67            Self::Geocoding => "GeocodingProvider",
68            Self::Routing => "RoutingProvider",
69            Self::DistanceMatrix => "DistanceMatrixProvider",
70            Self::PlaceSearch => "SearchPlaceProvider",
71            Self::TrafficIncidents => "TrafficIncidentProvider",
72            Self::TrafficFlow => "TrafficFlowProvider",
73            Self::Elevation => "ElevationProvider",
74            Self::RasterTiles => "RasterTileProvider",
75            Self::VectorTiles => "VectorTileProvider",
76            Self::OfflineRouting => "OfflineRoutingProvider",
77        }
78    }
79}

81/// Registry of provider implementations keyed by contract, not by vendor.
82///
83/// Each contract has one optional slot. Slots start unconfigured. Callers
84/// resolve them with the `require_*` methods, which report
85/// [`SDKError::ProviderNotConfigured`] when the application did not register
86/// that provider.
87///
88/// The registry is cheap to share after configuration: every provider is
89/// stored behind an [`Arc`].
90#[derive(Default)]
91pub struct ProviderRegistry {
92    geocoding_provider: Option<Arc<dyn GeocodingProvider>>,
93    routing_provider: Option<Arc<dyn RoutingProvider>>,
94    distance_matrix_provider: Option<Arc<dyn DistanceMatrixProvider>>,
95    search_place_provider: Option<Arc<dyn SearchPlaceProvider>>,
96    traffic_incident_provider: Option<Arc<dyn TrafficIncidentProvider>>,
97    traffic_flow_provider: Option<Arc<dyn TrafficFlowProvider>>,
98    /// The grid cache wrapping the serving traffic-flow provider, kept so a
99    /// diagnostics readback can report its counters and budget. `None` while
100    /// the provider owns its own freshness policy.
101    traffic_flow_cache: Option<Arc<TrafficFlowGridCache>>,
102    elevation_provider: Option<Arc<dyn ElevationProvider>>,
103    raster_tile_provider: Option<Arc<dyn RasterTileProvider>>,
104    vector_tile_provider: Option<Arc<dyn VectorTileProvider>>,
105    offline_routing_provider: Option<Arc<dyn OfflineRoutingProvider>>,
106}

141fn not_configured_error(contract_name: &str) -> SDKError {
142    SDKError::ProviderNotConfigured(contract_name.to_string())
143}

145impl ProviderRegistry {
146    /// Creates a registry with every contract slot unconfigured.
147    pub fn new() -> Self {

151    /// Registers the geocoding implementation, replacing any earlier one.
152    pub fn register_geocoding_provider(&mut self, provider: Arc<dyn GeocodingProvider>) {
153        self.geocoding_provider = Some(provider);
154    }
155
156    /// Registers the routing implementation, replacing any earlier one.
157    pub fn register_routing_provider(&mut self, provider: Arc<dyn RoutingProvider>) {
158        self.routing_provider = Some(provider);

163        self.distance_matrix_provider = Some(provider);

174        &mut self,

176    ) {
177        self.traffic_incident_provider = Some(provider);
178    }
179
180    /// Registers the traffic-flow implementation, replacing any earlier one.
181    ///
182    /// A provider registered here answers uncached: it owns its own freshness
183    /// policy, so the registry holds no cache to report.
184    pub fn register_traffic_flow_provider(&mut self, provider: Arc<dyn TrafficFlowProvider>) {
185        self.traffic_flow_provider = Some(provider);
186        self.traffic_flow_cache = None;
187    }
188
189    /// Registers the traffic-flow implementation behind the grid cache that
190    /// answers it, replacing any earlier provider and cache. The statistics
191    /// readback reports that cache's counters while this provider serves.
192    pub fn register_traffic_flow_provider_with_cache(
193        &mut self,
194        provider: Arc<dyn TrafficFlowProvider>,
195        cache: Arc<TrafficFlowGridCache>,
196    ) {
197        self.traffic_flow_provider = Some(provider);
198        self.traffic_flow_cache = Some(cache);
199    }
200
201    /// The serving traffic-flow cache's counters and budget, or `None` while
202    /// no traffic-flow provider is registered or the provider answers
203    /// uncached.
204    pub fn traffic_flow_cache_statistics(&self) -> Option<TrafficFlowCacheReadback> {
205        self.traffic_flow_cache
206            .as_ref()
207            .map(|cache| TrafficFlowCacheReadback {
208                statistics: cache.statistics(),
209                configuration: cache.configuration(),
210            })
211    }

310    /// Whether the contract's slot holds a provider.
311    ///
312    /// The check answers without constructing anything, so a settings surface
313    /// can list which capabilities are served before a call is attempted.
314    pub fn is_configured(&self, contract: ProviderContract) -> bool {
315        match contract {
316            ProviderContract::Geocoding => self.geocoding_provider.is_some(),
317            ProviderContract::Routing => self.routing_provider.is_some(),
318            ProviderContract::DistanceMatrix => self.distance_matrix_provider.is_some(),
319            ProviderContract::PlaceSearch => self.search_place_provider.is_some(),
320            ProviderContract::TrafficIncidents => self.traffic_incident_provider.is_some(),
321            ProviderContract::TrafficFlow => self.traffic_flow_provider.is_some(),
322            ProviderContract::Elevation => self.elevation_provider.is_some(),
323            ProviderContract::RasterTiles => self.raster_tile_provider.is_some(),
324            ProviderContract::VectorTiles => self.vector_tile_provider.is_some(),
325            ProviderContract::OfflineRouting => self.offline_routing_provider.is_some(),
326        }
327    }

330mod tests {
331    use super::ProviderRegistry;
332    use geospatial_traits::mock::{
333        MockGeocodingProvider, MockRasterTileProvider, MockVectorTileProvider,
334    };
335    use geospatial_types::SDKError;

338    #[test]
339    fn unregistered_slots_report_provider_not_configured() {
340        let registry = ProviderRegistry::new();
341        let failure = match registry.require_geocoding_provider() {
342            Err(failure) => failure,
343            Ok(_) => panic!("expected ProviderNotConfigured for GeocodingProvider"),
344        };
345        assert_eq!(
346            failure,
347            SDKError::ProviderNotConfigured("GeocodingProvider".to_string())
348        );
349    }
350
351    #[test]
352    fn registered_slots_resolve_through_trait_objects() {
353        let mut registry = ProviderRegistry::new();
354        registry.register_geocoding_provider(Arc::new(MockGeocodingProvider));
355        registry.register_raster_tile_provider(Arc::new(MockRasterTileProvider));
356
357        assert!(registry.require_geocoding_provider().is_ok());
358        assert!(registry.require_raster_tile_provider().is_ok());
359        assert!(registry.require_routing_provider().is_err());
360    }
361
362    #[test]
363    fn vector_tile_slot_resolves_through_trait_objects() {
364        let mut registry = ProviderRegistry::new();
365        registry.register_vector_tile_provider(Arc::new(MockVectorTileProvider));
366        assert!(registry.require_vector_tile_provider().is_ok());
367        let failure = match registry.require_routing_provider() {
368            Err(failure) => failure,
369            Ok(_) => panic!("expected ProviderNotConfigured for RoutingProvider"),
370        };
371        assert_eq!(
372            failure,
373            SDKError::ProviderNotConfigured("RoutingProvider".to_string())
374        );
375    }
376
377    #[test]
378    fn the_traffic_flow_cache_readback_follows_the_serving_provider() {
379        use crate::traffic_flow_cache::{
380            GridCachedTrafficFlowProvider, TrafficFlowCacheConfiguration,
381        };
382
383        let mut registry = ProviderRegistry::new();
384        assert!(
385            registry.traffic_flow_cache_statistics().is_none(),
386            "an empty registry has no cache to report"
387        );
388
389        let caching_provider = GridCachedTrafficFlowProvider::new(
390            Arc::new(geospatial_traits::mock::MockTrafficFlowProvider),
391            TrafficFlowCacheConfiguration::default(),
392            Arc::new(|| 0),
393        )
394        .expect("the default configuration must be accepted");
395        let cache = caching_provider.cache();
396        registry.register_traffic_flow_provider_with_cache(Arc::new(caching_provider), cache);
397        let readback = registry
398            .traffic_flow_cache_statistics()
399            .expect("a cached registration reports its cache");
400        assert_eq!(readback.statistics.miss_count, 0);
401        assert_eq!(
402            readback.configuration,
403            TrafficFlowCacheConfiguration::default()
404        );
405
406        // A provider that owns its own freshness policy replaces both the
407        // provider and the cache the readback reported.
408        registry.register_traffic_flow_provider(Arc::new(
409            geospatial_traits::mock::MockTrafficFlowProvider,
410        ));
411        assert!(
412            registry.traffic_flow_cache_statistics().is_none(),
413            "an uncached provider leaves no cache statistics behind"
414        );
415    }
