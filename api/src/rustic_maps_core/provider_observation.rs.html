1//! Provider-request observation: the record a console shows for one provider
2//! call, and the sink the SDK installs to receive it.
3//!
4//! The observation is built here rather than in a provider crate so the
5//! redaction rule has one home: a request path keeps its structure, query
6//! values are dropped except the documented non-secret parameters, and the
7//! response snippet is capped at [`RESPONSE_SNIPPET_MAXIMUM_BYTES`] on a
8//! character boundary. A credential reaches this module as part of the request
9//! and never leaves it — the tests feed one through a request and assert the
10//! observation carries neither the value nor the parameter pair.
11//!
12//! The sink is process-wide because the provider crates are stateless with
13//! respect to the SDK: the native binding and the browser façade each install
14//! one observer at startup, and a provider crate used on its own (a unit test,
15//! a stand-alone integration) runs with none installed and pays no cost.
16

48/// Query parameters whose values may appear in an observation because they

57/// Whether observations capture the response prefix. Off by default: a
58/// host opts in when it shows request rows, and the switch reaches the
59/// provider layer's own excerpt capture so a hosted provider hands the body
60/// over only while it is on.

63/// Enables or disables the bounded response excerpt on provider
64/// observations, and on the provider layer's own observation seam.

89/// One provider call, already redacted.
90#[derive(Debug, Clone, PartialEq, Eq)]
91pub struct ProviderRequestObservation {
92    /// The provider contract the call served, for example `RoutingProvider`.
93    pub provider_contract: String,
94    /// The registry key of the serving provider.
95    pub provider_key: String,
96    /// The HTTP method as a two-to-four-letter word (`GET`, `POST`).
97    pub http_method: String,
98    /// The redacted request path, query names included and values removed.
99    pub request_path: String,
100    /// `2xx`, `3xx`, `4xx`, `5xx` or [`TRANSPORT_FAILURE_STATUS_CLASS`].
101    pub status_class: String,
102    /// Wall-clock duration of the attempt, in milliseconds.
103    pub duration_milliseconds: u64,
104    /// Response payload size in bytes; zero when no response arrived.
105    pub response_bytes: u64,
106    /// The redacted response prefix captured behind the excerpt switch, capped
107    /// at [`RESPONSE_SNIPPET_MAXIMUM_BYTES`]; empty while capture is off or no
108    /// response body arrived.
109    pub response_excerpt: String,
110}
111
112impl ProviderRequestObservation {
113    /// The outcome word for this observation, derived from its status class:
114    /// [`OUTCOME_SUCCESS`], [`OUTCOME_REDIRECT`], [`OUTCOME_CLIENT_ERROR`],
115    /// [`OUTCOME_SERVER_ERROR`], [`OUTCOME_TRANSPORT_FAILURE`] or
116    /// [`OUTCOME_UNCLASSIFIED`].
117    pub fn outcome(&self) -> &'static str {
118        outcome_text(&self.status_class)
119    }
120}

122/// Receives one observation per completed provider call.
123pub trait ProviderRequestObserver: Send + Sync {
124    /// Records one provider call. Called synchronously on the request's
125    /// task, so an implementation must not block.
126    fn on_provider_request(&self, observation: ProviderRequestObservation);
127}

129/// The installed observer, if any.
130static INSTALLED_OBSERVER: OnceLock<Mutex<Option<Arc<dyn ProviderRequestObserver>>>> =
131    OnceLock::new();
132
133/// Installs the observer that receives every provider call made afterwards.
134///
135/// Replacing an installed observer is allowed: a binding constructs its SDK
136/// more than once in the same process, and the last installed observer is the
137/// one that matters. Passing no observer (`None`) stops delivery.
138pub fn install_provider_request_observer(observer: Option<Arc<dyn ProviderRequestObserver>>) {
139    let installed_observer = INSTALLED_OBSERVER.get_or_init(|| Mutex::new(None));
140    *installed_observer
141        .lock()
142        .unwrap_or_else(std::sync::PoisonError::into_inner) = observer;
143}

145/// A [`ProviderRequestObserver`] that publishes every observation as one
146/// structured `tracing` event under [`PROVIDER_REQUEST_DIAGNOSTIC_TARGET`].
147///
148/// The event carries the provider key, the contract, the HTTP method, the
149/// redacted request path, the status class, the duration, the response byte
150/// count and the derived outcome. Because it is the observation the transport
151/// already redacted, a credential cannot reach the event; the SDK's
152/// diagnostics sink records the event like any other SDK log line, so a
153/// request observation is visible through the existing diagnostics
154/// subscription and flush seams.
155///
156/// A provider crate used on its own never installs this bridge and pays no
157/// logging cost; each SDK seam installs it once at construction through
158/// [`install_tracing_observation_bridge`].
159#[derive(Debug, Default)]
160pub struct TracingObservationBridge;
161
162impl ProviderRequestObserver for TracingObservationBridge {
163    fn on_provider_request(&self, observation: ProviderRequestObservation) {
164        let outcome = observation.outcome();
165        tracing::info!(
166            target: PROVIDER_REQUEST_DIAGNOSTIC_TARGET,
167            provider_key = observation.provider_key.as_str(),
168            provider_contract = observation.provider_contract.as_str(),
169            http_method = observation.http_method.as_str(),
170            request_path = observation.request_path.as_str(),
171            status_class = observation.status_class.as_str(),
172            outcome,
173            duration_milliseconds = observation.duration_milliseconds,
174            response_bytes = observation.response_bytes,
175            response_excerpt = observation.response_excerpt.as_str(),
176            "provider request completed"
177        );
178    }
179}

181/// Installs [`TracingObservationBridge`] as the process's provider-request
182/// observer, replacing any observer installed earlier (the slot's documented
183/// last-one-wins rule).
184pub fn install_tracing_observation_bridge() {
185    install_provider_request_observer(Some(Arc::new(TracingObservationBridge)));

188/// Builds one redacted observation for a completed attempt and hands it to
189/// the installed observer. Returns immediately when none is installed.
190///
191/// # Parameters
192/// - `provider_contract` — the contract the call served.
193/// - `provider_key` — the serving provider's registry key.
194/// - `http_method` — the request's method.
195/// - `request_url` — the fully built request URL, before redaction.
196/// - `outcome` — the status code and body, when a response arrived.
197/// - `duration` — how long the attempt took.
198pub fn observe_provider_request(
199    provider_contract: &str,
200    provider_key: &str,
201    http_method: HttpMethod,
202    request_url: &str,
203    outcome: Option<(u16, &[u8])>,
204    duration: Duration,
205) {
206    deliver_provider_request_observation(provider_request_observation(
207        provider_contract,
208        provider_key,
209        http_method,
210        request_url,
211        outcome,
212        duration,
213    ));
214}

216/// Builds one [`ProviderRequestObservation`] without delivering it.
217///
218/// A caller that reports through a different transport — the provider layer
219/// has its own observer — can build the record here and hand it to
220/// [`deliver_provider_request_observation`], so both transports publish the
221/// same vocabulary.
222pub fn provider_request_observation(
223    provider_contract: &str,
224    provider_key: &str,
225    http_method: HttpMethod,
226    request_url: &str,
227    outcome: Option<(u16, &[u8])>,
228    duration: Duration,
229) -> ProviderRequestObservation {
230    let (status_class, response_bytes, response_excerpt) = match outcome {
231        Some((status_code, body)) => (
232            status_class_text(status_code),
233            body.len() as u64,
234            if response_excerpt_capture_enabled() {
235                response_snippet_text(body)
236            } else {
237                String::new()
238            },
239        ),
240        None => (TRANSPORT_FAILURE_STATUS_CLASS.to_string(), 0, String::new()),
241    };
242    ProviderRequestObservation {
243        provider_contract: provider_contract.to_string(),
244        provider_key: provider_key.to_string(),
245        http_method: http_method_text(http_method).to_string(),
246        request_path: redacted_request_path(request_url),
247        status_class,
248        duration_milliseconds: duration.as_millis().min(u64::MAX as u128) as u64,
249        response_bytes,
250        response_excerpt,
251    }
252}

254/// Hands one observation to the installed observer, if any.
255pub fn deliver_provider_request_observation(observation: ProviderRequestObservation) {
256    if let Some(observer) = installed_observer() {
257        observer.on_provider_request(observation);
258    }
259}

261/// Reduces a request URL to what an observation may show: the path, plus the
262/// names of its query parameters with values removed except the parameters
263/// whose value describes the request shape.
264pub fn redacted_request_path(request_url: &str) -> String {
265    let Ok(parsed_url) = Url::parse(request_url) else {
266        // An unparseable URL never reaches a provider in practice; refuse to
267        // echo it verbatim, because it may carry a credential in any shape.
268        return String::from("<unparseable request url>");
269    };
270    let mut redacted_path = parsed_url.path().to_string();
271    let mut redacted_parameters: Vec<String> = parsed_url
272        .query_pairs()
273        .map(|(parameter_name, parameter_value)| {
274            if OBSERVABLE_QUERY_PARAMETER_NAMES.contains(&parameter_name.as_ref()) {
275                format!("{parameter_name}={parameter_value}")
276            } else {
277                format!("{parameter_name}=<redacted>")
278            }
279        })
280        .collect();
281    redacted_parameters.sort();
282    if !redacted_parameters.is_empty() {
283        redacted_path.push('?');
284        redacted_path.push_str(&redacted_parameters.join("&"));
285    }
286    redacted_path
287}

289/// Caps a response body at [`RESPONSE_SNIPPET_MAXIMUM_BYTES`] on a character
290/// boundary, so a multi-byte glyph is never split.
291pub fn response_snippet_text(response_body: &[u8]) -> String {
292    let text = String::from_utf8_lossy(response_body);
293    if text.len() <= RESPONSE_SNIPPET_MAXIMUM_BYTES {
294        return text.into_owned();
295    }
296    let mut snippet_end = RESPONSE_SNIPPET_MAXIMUM_BYTES;
297    while !text.is_char_boundary(snippet_end) {
298        snippet_end -= 1;
299    }
300    text[..snippet_end].to_string()
301}

303/// The status class an observation reports for one HTTP status code.
304fn status_class_text(status_code: u16) -> String {
305    let status_class_prefix = status_code / 100;
306    format!("{status_class_prefix}xx")
307}

309/// The outcome word an observation reports for one status class.
310fn outcome_text(status_class: &str) -> &'static str {
311    match status_class {
312        "2xx" => OUTCOME_SUCCESS,
313        "3xx" => OUTCOME_REDIRECT,
314        "4xx" => OUTCOME_CLIENT_ERROR,
315        "5xx" => OUTCOME_SERVER_ERROR,
316        TRANSPORT_FAILURE_STATUS_CLASS => OUTCOME_TRANSPORT_FAILURE,
317        _ => OUTCOME_UNCLASSIFIED,
318    }
319}

321/// The word an observation reports for one HTTP method.
322fn http_method_text(http_method: HttpMethod) -> &'static str {
323    match http_method {
324        HttpMethod::Get => "GET",
325        HttpMethod::Post => "POST",
326    }
327}

339// The provider layer's request observer, rendered in this module's vocabulary.
340//
341// The SDK's provider diagnostics are built from one observation per
342// completed provider call: the contract it served, the provider's registry
343// key, the method, the redacted path, the status class, the duration and the
344// response size. The provider layer reports the same facts through its own
345// [`everymap_core::client::RequestObserver`], so an adapter lets a provider
346// built on that layer keep feeding the existing diagnostics ring, the typed
347// `ProviderRequestTrace` and the browser's event stream unchanged.
348//
349// The adapter pairs each request with its response, because the provider
350// layer reports the two separately and the SDK's vocabulary wants one
351// record. A response without a recorded request is dropped rather than
352// reported with invented fields.
353
354use everymap_core::client::{RequestObservation, RequestObserver, ResponseObservation};
355use std::sync::PoisonError;
356use web_time::Instant;
357
358/// One request the provider layer has started and not yet answered.
359struct InFlightRequest {
360    /// The method the SDK's vocabulary names.
361    http_method: HttpMethod,
362    /// The redacted request URL the provider layer handed over.
363    ///
364    /// The provider layer redacts credential values with its own marker; the
365    /// SDK re-renders the path from this URL, so the marker never reaches the
366    /// diagnostic line.
367    request_url: String,
368    /// When the transport started the attempt.
369    started_at: Instant,
370}
371
372/// Publishes one provider's request attempts into the SDK's diagnostics.
373pub struct EveryMapRequestObserver {
374    /// The contract the calls through this client serve.
375    provider_contract: String,
376    /// The provider's registry key.
377    provider_key: String,
378    /// The attempt awaiting its response.
379    in_flight: Mutex<Option<InFlightRequest>>,
380}
381
382impl EveryMapRequestObserver {
383    /// Builds the observer for one provider and one contract.
384    ///
385    /// # Parameters
386    /// - `provider_contract` — the contract name the SDK reports.
387    /// - `provider_key` — the provider's registry key.
388    pub fn new(provider_contract: &str, provider_key: &str) -> Self {
389        Self {
390            provider_contract: provider_contract.to_string(),
391            provider_key: provider_key.to_string(),
392            in_flight: Mutex::new(None),
393        }
394    }
395
396    /// Renders one completed attempt in the SDK's vocabulary.
397    ///
398    /// The provider layer reports no response body, so the byte count is zero
399    /// and the snippet empty; both are facts about what was observed rather
400    /// than invented values.
401    fn translated_observation(
402        &self,
403        in_flight: InFlightRequest,
404        status: Option<u16>,
405        response_excerpt: Option<&str>,
406        elapsed_upstream: Duration,
407    ) -> ProviderRequestObservation {
408        let response_body = response_excerpt.map(str::as_bytes).unwrap_or(&[]);
409        let outcome = status.map(|status_code| (status_code, response_body));
410        provider_request_observation(
411            &self.provider_contract,
412            &self.provider_key,
413            in_flight.http_method,
414            &in_flight.request_url,
415            outcome,
416            duration_or_zero(in_flight.started_at.elapsed(), elapsed_upstream),
417        )
418    }
419}
420
421impl RequestObserver for EveryMapRequestObserver {
422    fn on_request(&self, observation: &RequestObservation<'_>) {
423        let mut in_flight = self
424            .in_flight
425            .lock()
426            .unwrap_or_else(PoisonError::into_inner);
427        *in_flight = Some(InFlightRequest {
428            http_method: sdk_http_method(observation.method),
429            request_url: observation.url.to_string(),
430            started_at: Instant::now(),
431        });
432    }
433
434    fn on_response(&self, observation: &ResponseObservation<'_>) {
435        let Some(in_flight) = self
436            .in_flight
437            .lock()
438            .unwrap_or_else(PoisonError::into_inner)
439            .take()
440        else {
441            return;
442        };
443        deliver_provider_request_observation(self.translated_observation(
444            in_flight,
445            observation.status,
446            observation.response_excerpt,
447            observation.elapsed,
448        ));
449    }
450}

452/// The SDK method word for one provider-layer method.
453///
454/// The SDK's vocabulary models the two methods its providers use; a provider
455/// layer method outside that pair is reported as a POST, which is what the
456/// request carries a body for.
457fn sdk_http_method(method: everymap_core::http::Method) -> HttpMethod {
458    match method {
459        everymap_core::http::Method::Get => HttpMethod::Get,
460        _ => HttpMethod::Post,
461    }
462}

464/// The attempt duration, preferring the provider layer's own measurement.
465fn duration_or_zero(measured_here: Duration, measured_upstream: Duration) -> Duration {
466    if measured_upstream.is_zero() {
467        measured_here
468    } else {
469        measured_upstream
470    }
471}

688    #[test]
689    fn an_observation_reaches_the_installed_observer() {
690        let _observer_lock = lock_installed_observer();
691        let recording_observer = Arc::new(RecordingObserver::default());
692        install_provider_request_observer(Some(recording_observer.clone()));
693        set_response_excerpt_capture(false);
694        observe_provider_request(
695            "RoutingProvider",
696            "test-provider",
697            HttpMethod::Get,
698            "https://provider.invalid/route/v1/driving/a;b?key=SECRET_VALUE",
699            Some((200, b"{}".as_slice())),
700            Duration::from_millis(12),
701        );
702        // The excerpt is off by default: the size is reported, the body is not.
703        let observations = recording_observer
704            .observations
705            .lock()
706            .expect("the observation list is never poisoned")
707            .clone();
708        assert_eq!(observations.len(), 1);
709        assert_eq!(observations[0].response_bytes, 2);
710        assert!(observations[0].response_excerpt.is_empty());
711        set_response_excerpt_capture(true);
712        observe_provider_request(
713            "RoutingProvider",
714            "test-provider",
715            HttpMethod::Get,
716            "https://provider.invalid/route/v1/driving/a;b?key=SECRET_VALUE",
717            Some((200, b"{}".as_slice())),
718            Duration::from_millis(12),
719        );
720        set_response_excerpt_capture(false);
721        let observations = recording_observer
722            .observations
723            .lock()
724            .expect("the observation list is never poisoned")
725            .clone();
726        assert_eq!(observations.len(), 2);
727        let observation = &observations[1];
728        assert_eq!(observation.provider_contract, "RoutingProvider");
729        assert_eq!(observation.provider_key, "test-provider");
730        assert_eq!(observation.http_method, "GET");
731        assert_eq!(observation.status_class, "2xx");
732        assert_eq!(observation.response_bytes, 2);
733        assert_eq!(observation.response_excerpt, "{}");
734        assert!(observation.duration_milliseconds >= 12);
735        assert!(!observation.request_path.contains("SECRET_VALUE"));
736    }
