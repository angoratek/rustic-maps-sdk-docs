1//! The typed expression tree, its parser and its static analysis.
2//!
3//! The parser accepts the specification's array form and, through the
4//! legacy-function normaliser, the deprecated `{stops, type, base, property,
5//! default}` object form. Both normalize into the same tree, so evaluation has
6//! one path.
7//!
8//! Parsing validates the rules that do not depend on runtime data: operator
9//! arity, numeric and strictly ascending interpolation stops, and the rule
10//! that `["zoom"]` may only be the direct input of a top-level `step` or
11//! `interpolate`. [`Expression::analysis`] classifies an expression as
12//! constant, camera-driven, data-driven or both, so a caller can reject a
13//! data-driven property where the specification forbids one before it is ever
14//! evaluated.
15

50/// The expression operators this crate models.
51#[derive(Debug, Clone, PartialEq)]
52pub enum ExpressionKind {
53    /// A constant JSON value, produced by a scalar or by `literal`.
54    Literal(Value),
55    /// A property lookup; `object` optionally overrides the feature's
56    /// properties, as the legacy `get` form allows.
57    Get {
58        /// The property key.
59        key: Box<Expression>,
60        /// The object to read from instead of the feature's properties.
61        object: Option<Box<Expression>>,
62    },
63    /// Whether a property exists.
64    Has {
65        /// The property key.
66        key: Box<Expression>,
67    },
68    /// The feature's geometry kind.
69    GeometryType,
70    /// The feature's identifier.
71    FeatureId,
72    /// The feature's properties as an object.
73    Properties,
74    /// The current zoom level.
75    Zoom,
76    /// A value from the style's global state.
77    GlobalState {
78        /// The state key.
79        key: Box<Expression>,
80    },
81    /// A value from the feature's state (parsed; not implemented in this
82    /// phase).
83    FeatureState {
84        /// The state key.
85        key: Box<Expression>,
86        /// The value used when the state has no entry.
87        default: Option<Box<Expression>>,
88    },
89    /// A lexical binding followed by a body.
90    Let {
91        /// The named bindings, evaluated in order.
92        bindings: Vec<(String, Expression)>,
93        /// The expression that reads them.
94        body: Box<Expression>,
95    },
96    /// A reference to a `let` binding.
97    Variable(String),
98    /// Whether every child is true.
99    All(Vec<Expression>),
100    /// Whether any child is true.
101    Any(Vec<Expression>),
102    /// Logical negation.
103    Not(Box<Expression>),
104    /// Equality following the specification's coercion rules.
105    Equal(Box<Expression>, Box<Expression>),
106    /// Inequality.
107    NotEqual(Box<Expression>, Box<Expression>),
108    /// Numeric or lexical less than.
109    Less(Box<Expression>, Box<Expression>),
110    /// Less than or equal.
111    LessEqual(Box<Expression>, Box<Expression>),
112    /// Greater than.
113    Greater(Box<Expression>, Box<Expression>),
114    /// Greater than or equal.
115    GreaterEqual(Box<Expression>, Box<Expression>),
116    /// A conditional chain with a fallback.
117    Case {
118        /// Condition/output pairs, tried in order.
119        branches: Vec<(Expression, Expression)>,
120        /// The value when no condition is true.
121        otherwise: Box<Expression>,
122    },
123    /// A value match with a fallback.
124    Match {
125        /// The value to compare.
126        input: Box<Expression>,
127        /// Label groups and their outputs, tried in order.
128        labels: Vec<(Vec<Value>, Expression)>,
129        /// The value when no label group matches.
130        otherwise: Box<Expression>,
131    },
132    /// The first non-null child.
133    Coalesce(Vec<Expression>),
134    /// A stepped function.
135    Step {
136        /// The input, usually `["zoom"]` or `["get", …]`.
137        input: Box<Expression>,
138        /// The output below the first stop.
139        first_output: Box<Expression>,
140        /// Strictly ascending `(input, output)` stops.
141        stops: Vec<(f64, Expression)>,
142    },
143    /// An interpolated function.
144    Interpolate {
145        /// The curve between stops.
146        interpolation: Interpolation,
147        /// The input, usually `["zoom"]` or `["get", …]`.
148        input: Box<Expression>,
149        /// Strictly ascending `(input, output)` stops.
150        stops: Vec<(f64, Expression)>,
151    },
152    /// Converts a value to a string.
153    ToString(Box<Expression>),
154    /// Converts a value to a number.
155    ToNumber(Box<Expression>),
156    /// Converts a value to a boolean.
157    ToBoolean(Box<Expression>),
158    /// Converts a value to a colour.
159    ToColor(Box<Expression>),
160    /// An `rgb()` colour from three `0..=255` channels.
161    Rgb {
162        /// Red channel.
163        red: Box<Expression>,
164        /// Green channel.
165        green: Box<Expression>,
166        /// Blue channel.
167        blue: Box<Expression>,
168    },
169    /// An `rgba()` colour.
170    Rgba {
171        /// Red channel.
172        red: Box<Expression>,
173        /// Green channel.
174        green: Box<Expression>,
175        /// Blue channel.
176        blue: Box<Expression>,
177        /// Alpha channel in `0.0..=1.0`.
178        alpha: Box<Expression>,
179    },
180    /// String concatenation.
181    Concat(Vec<Expression>),
182    /// Upper-cases a string.
183    Upcase(Box<Expression>),
184    /// Lower-cases a string.
185    Downcase(Box<Expression>),
186    /// The number of characters in a string or entries in an array.
187    Length(Box<Expression>),
188    /// The entry at an index in a string or array.
189    At {
190        /// The string or array.
191        input: Box<Expression>,
192        /// The index; negative counts from the end.
193        index: Box<Expression>,
194    },
195    /// Whether a string contains a substring or an array contains a value.

245        /// The base.

252    /// The largest child.

254    /// Rounds to the nearest integer.

259    Ceil(Box<Expression>),

263    Sqrt(Box<Expression>),

265    NaturalLog(Box<Expression>),

272    /// Cosine.

300/// How an expression depends on its inputs.
301#[derive(Debug, Clone, Copy, PartialEq, Eq)]
302pub enum ExpressionAnalysis {
303    /// The expression has the same value for every feature and zoom.
304    Constant,
305    /// The expression depends on the zoom level only.
306    Camera,
307    /// The expression depends on the feature only.
308    DataDriven,
309    /// The expression depends on both the feature and the zoom level.
310    CameraAndData,
311}

313impl Expression {
314    /// Parses one expression from a JSON value.
315    ///
316    /// # Errors
317    /// An unknown operator, an argument of the wrong arity or type, a stop
318    /// array that is not strictly ascending, or a JSON object that is neither
319    /// an expression nor a legacy function reports [`ExpressionError`].
320    pub fn parse(json_value: &JsonValue) -> Result<Self, ExpressionError> {
321        match json_value {
322            JsonValue::Array(array) => parse_array(array),
323            JsonValue::Object(object) => {
324                if object.contains_key("stops")
325                    || object.contains_key("property")
326                    || object.contains_key("type")
327                {
328                    let normalized = legacy_function::normalize_legacy_function(json_value)?;
329                    Self::parse(&normalized)
330                } else {
331                    Err(ExpressionError::new(
332                        "a JSON object is not an expression unless it is a legacy function",
333                    ))
334                }
335            }
336            scalar => Ok(Self {
337                kind: ExpressionKind::Literal(Value::from(scalar)),
338            }),
339        }
340    }
341
342    /// Parses one expression from JSON text.
343    ///
344    /// # Errors
345    /// Invalid JSON and any [`Expression::parse`] error are reported.
346    pub fn parse_str(json_text: &str) -> Result<Self, ExpressionError> {
347        let json_value: JsonValue = serde_json::from_str(json_text).map_err(|json_error| {
348            ExpressionError::new(format!("expression is not valid JSON: {json_error}"))
349        })?;
350        Self::parse(&json_value)
351    }

353    /// Classifies the expression by its inputs.
354    pub fn analysis(&self) -> ExpressionAnalysis {
355        analyze(self)
356    }

368    /// Every direct child expression, in evaluation order.
369    pub fn children(&self) -> Vec<&Expression> {
370        let mut children = Vec::new();
371        match &self.kind {
372            ExpressionKind::Literal(_)
373            | ExpressionKind::GeometryType
374            | ExpressionKind::FeatureId
375            | ExpressionKind::Properties
376            | ExpressionKind::Zoom
377            | ExpressionKind::Variable(_)
378            | ExpressionKind::Pi
379            | ExpressionKind::Euler
380            | ExpressionKind::Ln2 => {}
381            ExpressionKind::Get { key, object } => {
382                children.push(key.as_ref());
383                if let Some(object) = object {
384                    children.push(object.as_ref());
385                }
386            }
387            ExpressionKind::Has { key }
388            | ExpressionKind::GlobalState { key }
389            | ExpressionKind::ToString(key)
390            | ExpressionKind::ToNumber(key)
391            | ExpressionKind::ToBoolean(key)
392            | ExpressionKind::ToColor(key)
393            | ExpressionKind::Upcase(key)
394            | ExpressionKind::Downcase(key)
395            | ExpressionKind::Length(key)
396            | ExpressionKind::Round(key)
397            | ExpressionKind::Floor(key)
398            | ExpressionKind::Ceil(key)
399            | ExpressionKind::Abs(key)
400            | ExpressionKind::Sqrt(key)
401            | ExpressionKind::NaturalLog(key)
402            | ExpressionKind::Log10(key)
403            | ExpressionKind::Log2(key)
404            | ExpressionKind::Sin(key)
405            | ExpressionKind::Cos(key)
406            | ExpressionKind::Tan(key)
407            | ExpressionKind::Asin(key)
408            | ExpressionKind::Acos(key)
409            | ExpressionKind::Atan(key) => children.push(key.as_ref()),
410            ExpressionKind::FeatureState { key, default } => {
411                children.push(key.as_ref());
412                if let Some(default) = default {
413                    children.push(default.as_ref());
414                }
415            }
416            ExpressionKind::Let { bindings, body } => {
417                for (_, binding) in bindings {
418                    children.push(binding);
419                }
420                children.push(body.as_ref());
421            }
422            ExpressionKind::All(expressions)
423            | ExpressionKind::Any(expressions)
424            | ExpressionKind::Coalesce(expressions)
425            | ExpressionKind::Concat(expressions)
426            | ExpressionKind::Add(expressions)
427            | ExpressionKind::Multiply(expressions)
428            | ExpressionKind::Min(expressions)
429            | ExpressionKind::Max(expressions) => {
430                children.extend(expressions.iter());
431            }
432            ExpressionKind::Not(child)
433            | ExpressionKind::Equal(child, _)
434            | ExpressionKind::NotEqual(child, _)
435            | ExpressionKind::Less(child, _)
436            | ExpressionKind::LessEqual(child, _)
437            | ExpressionKind::Greater(child, _)
438            | ExpressionKind::GreaterEqual(child, _) => {
439                children.push(child.as_ref());
440                match &self.kind {
441                    ExpressionKind::Equal(_, right)
442                    | ExpressionKind::NotEqual(_, right)
443                    | ExpressionKind::Less(_, right)
444                    | ExpressionKind::LessEqual(_, right)
445                    | ExpressionKind::Greater(_, right)
446                    | ExpressionKind::GreaterEqual(_, right) => children.push(right.as_ref()),
447                    _ => {}
448                }
449            }
450            ExpressionKind::Case {
451                branches,
452                otherwise,
453            } => {
454                for (condition, output) in branches {
455                    children.push(condition);
456                    children.push(output);
457                }
458                children.push(otherwise.as_ref());
459            }
460            ExpressionKind::Match {
461                input,
462                labels,
463                otherwise,
464            } => {
465                children.push(input.as_ref());
466                for (_, output) in labels {
467                    children.push(output);
468                }
469                children.push(otherwise.as_ref());
470            }
471            ExpressionKind::Step {
472                input,
473                first_output,
474                stops,
475            } => {
476                children.push(input.as_ref());
477                children.push(first_output.as_ref());
478                for (_, output) in stops {
479                    children.push(output);
480                }
481            }
482            ExpressionKind::Interpolate { input, stops, .. } => {
483                children.push(input.as_ref());
484                for (_, output) in stops {
485                    children.push(output);
486                }
487            }
488            ExpressionKind::Rgb { red, green, blue } => {
489                children.push(red.as_ref());
490                children.push(green.as_ref());
491                children.push(blue.as_ref());
492            }
493            ExpressionKind::Rgba {
494                red,
495                green,
496                blue,
497                alpha,
498            } => {
499                children.push(red.as_ref());
500                children.push(green.as_ref());
501                children.push(blue.as_ref());
502                children.push(alpha.as_ref());
503            }
504            ExpressionKind::At { input, index } => {
505                children.push(input.as_ref());
506                children.push(index.as_ref());
507            }
508            ExpressionKind::In { needle, haystack } => {
509                children.push(needle.as_ref());
510                children.push(haystack.as_ref());
511            }
512            ExpressionKind::IndexOf { haystack, needle } => {
513                children.push(haystack.as_ref());
514                children.push(needle.as_ref());
515            }
516            ExpressionKind::Slice { input, start, end } => {
517                children.push(input.as_ref());
518                children.push(start.as_ref());
519                if let Some(end) = end {
520                    children.push(end.as_ref());
521                }
522            }
523            ExpressionKind::Subtract {
524                minuend,
525                subtrahend,
526            } => {
527                children.push(minuend.as_ref());
528                children.push(subtrahend.as_ref());
529            }
530            ExpressionKind::Divide { dividend, divisor } => {
531                children.push(dividend.as_ref());
532                children.push(divisor.as_ref());
533            }
534            ExpressionKind::Modulo { dividend, divisor } => {
535                children.push(dividend.as_ref());
536                children.push(divisor.as_ref());
537            }
538            ExpressionKind::Power { base, exponent } => {
539                children.push(base.as_ref());
540                children.push(exponent.as_ref());
541            }
542            ExpressionKind::Format(sections) => {
543                for section in sections {
544                    children.push(&section.value);
545                }
546            }
547            ExpressionKind::Unsupported { arguments, .. } => {
548                children.extend(arguments.iter());
549            }
550        }
551        children
552    }
553}

555/// Whether a JSON value is an expression array: an array whose first member
556/// is a known operator name.
557///
558/// A bare array such as `["Open Sans Regular", "Noto Sans"]` is a constant
559/// list, not an expression, which is why the operator name is checked rather
560/// than merely the first member's type.
561pub fn is_expression_array(json_value: &JsonValue) -> bool {
562    json_value
563        .as_array()
564        .and_then(|array| array.first())
565        .and_then(JsonValue::as_str)
566        .is_some_and(is_known_operator)
567}

652fn parse_array(array: &[JsonValue]) -> Result<Expression, ExpressionError> {
653    let Some(operator) = array.first().and_then(JsonValue::as_str) else {
654        return Err(ExpressionError::new(
655            "an expression array must start with an operator string",
656        ));
657    };
658    let arguments = &array[1..];
659    let kind = match operator {
660        "literal" => {
661            expect_exact_arity(operator, arguments, 1)?;
662            ExpressionKind::Literal(Value::from(&arguments[0]))
663        }
664        "get" => {
665            expect_arity(operator, arguments, 1, 2)?;
666            ExpressionKind::Get {
667                key: Box::new(parse_argument(arguments, 0)?),
668                object: match arguments.get(1) {
669                    Some(object) => Some(Box::new(Expression::parse(object)?)),
670                    None => None,
671                },
672            }
673        }
674        "has" => {
675            expect_exact_arity(operator, arguments, 1)?;
676            ExpressionKind::Has {
677                key: Box::new(parse_argument(arguments, 0)?),
678            }
679        }
680        "geometry-type" => {
681            expect_exact_arity(operator, arguments, 0)?;
682            ExpressionKind::GeometryType
683        }
684        "id" => {
685            expect_exact_arity(operator, arguments, 0)?;
686            ExpressionKind::FeatureId
687        }
688        "properties" => {
689            expect_exact_arity(operator, arguments, 0)?;
690            ExpressionKind::Properties
691        }
692        "zoom" => {
693            expect_exact_arity(operator, arguments, 0)?;
694            ExpressionKind::Zoom
695        }
696        "global-state" => {
697            expect_exact_arity(operator, arguments, 1)?;
698            ExpressionKind::GlobalState {
699                key: Box::new(parse_argument(arguments, 0)?),
700            }
701        }
702        "feature-state" => {
703            expect_arity(operator, arguments, 1, 2)?;
704            ExpressionKind::FeatureState {
705                key: Box::new(parse_argument(arguments, 0)?),
706                default: match arguments.get(1) {
707                    Some(default) => Some(Box::new(Expression::parse(default)?)),
708                    None => None,
709                },
710            }
711        }
712        "let" => parse_let(arguments)?,
713        "var" => {
714            expect_exact_arity(operator, arguments, 1)?;
715            ExpressionKind::Variable(required_string(arguments, 0, operator)?)
716        }
717        "all" => ExpressionKind::All(parse_arguments(arguments)?),
718        "any" => ExpressionKind::Any(parse_arguments(arguments)?),
719        "!" => {
720            expect_exact_arity(operator, arguments, 1)?;
721            ExpressionKind::Not(Box::new(parse_argument(arguments, 0)?))
722        }
723        "==" => binary(operator, arguments, ExpressionKind::Equal)?,
724        "!=" => binary(operator, arguments, ExpressionKind::NotEqual)?,
725        "<" => binary(operator, arguments, ExpressionKind::Less)?,
726        "<=" => binary(operator, arguments, ExpressionKind::LessEqual)?,
727        ">" => binary(operator, arguments, ExpressionKind::Greater)?,
728        ">=" => binary(operator, arguments, ExpressionKind::GreaterEqual)?,
729        "case" => parse_case(arguments)?,
730        "match" => parse_match(arguments)?,
731        "coalesce" => {
732            if arguments.is_empty() {
733                return Err(ExpressionError::new("coalesce needs at least one argument"));
734            }
735            ExpressionKind::Coalesce(parse_arguments(arguments)?)
736        }
737        "step" => parse_step(arguments)?,
738        "interpolate" => parse_interpolate(arguments)?,
739        "to-string" => unary(operator, arguments, ExpressionKind::ToString)?,
740        "to-number" => unary(operator, arguments, ExpressionKind::ToNumber)?,
741        "to-boolean" => unary(operator, arguments, ExpressionKind::ToBoolean)?,
742        "to-color" => unary(operator, arguments, ExpressionKind::ToColor)?,
743        "rgb" => {
744            expect_exact_arity(operator, arguments, 3)?;
745            ExpressionKind::Rgb {
746                red: Box::new(parse_argument(arguments, 0)?),
747                green: Box::new(parse_argument(arguments, 1)?),
748                blue: Box::new(parse_argument(arguments, 2)?),
749            }
750        }
751        "rgba" => {
752            expect_exact_arity(operator, arguments, 4)?;
753            ExpressionKind::Rgba {
754                red: Box::new(parse_argument(arguments, 0)?),
755                green: Box::new(parse_argument(arguments, 1)?),
756                blue: Box::new(parse_argument(arguments, 2)?),
757                alpha: Box::new(parse_argument(arguments, 3)?),
758            }
759        }
760        "concat" => {
761            if arguments.is_empty() {
762                return Err(ExpressionError::new("concat needs at least one argument"));
763            }
764            ExpressionKind::Concat(parse_arguments(arguments)?)
765        }
766        "upcase" => unary(operator, arguments, ExpressionKind::Upcase)?,
767        "downcase" => unary(operator, arguments, ExpressionKind::Downcase)?,
768        "length" => unary(operator, arguments, ExpressionKind::Length)?,
769        "at" => {
770            expect_exact_arity(operator, arguments, 2)?;
771            ExpressionKind::At {
772                input: Box::new(parse_argument(arguments, 0)?),
773                index: Box::new(parse_argument(arguments, 1)?),
774            }
775        }
776        "in" => {
777            expect_exact_arity(operator, arguments, 2)?;
778            ExpressionKind::In {
779                needle: Box::new(parse_argument(arguments, 0)?),
780                haystack: Box::new(parse_argument(arguments, 1)?),
781            }
782        }
783        "index-of" => {
784            expect_exact_arity(operator, arguments, 2)?;
785            ExpressionKind::IndexOf {
786                haystack: Box::new(parse_argument(arguments, 0)?),
787                needle: Box::new(parse_argument(arguments, 1)?),
788            }
789        }
790        "slice" => {
791            expect_arity(operator, arguments, 2, 3)?;
792            ExpressionKind::Slice {
793                input: Box::new(parse_argument(arguments, 0)?),
794                start: Box::new(parse_argument(arguments, 1)?),
795                end: match arguments.get(2) {
796                    Some(end) => Some(Box::new(Expression::parse(end)?)),
797                    None => None,
798                },
799            }
800        }
801        "+" => {
802            require_at_least(operator, arguments, 1)?;
803            ExpressionKind::Add(parse_arguments(arguments)?)
804        }
805        "-" => {
806            expect_exact_arity(operator, arguments, 2)?;
807            ExpressionKind::Subtract {
808                minuend: Box::new(parse_argument(arguments, 0)?),
809                subtrahend: Box::new(parse_argument(arguments, 1)?),
810            }
811        }
812        "*" => {
813            require_at_least(operator, arguments, 1)?;
814            ExpressionKind::Multiply(parse_arguments(arguments)?)
815        }
816        "/" => {
817            expect_exact_arity(operator, arguments, 2)?;
818            ExpressionKind::Divide {
819                dividend: Box::new(parse_argument(arguments, 0)?),
820                divisor: Box::new(parse_argument(arguments, 1)?),
821            }
822        }
823        "%" => {
824            expect_exact_arity(operator, arguments, 2)?;
825            ExpressionKind::Modulo {
826                dividend: Box::new(parse_argument(arguments, 0)?),
827                divisor: Box::new(parse_argument(arguments, 1)?),
828            }
829        }
830        "^" => {
831            expect_exact_arity(operator, arguments, 2)?;
832            ExpressionKind::Power {
833                base: Box::new(parse_argument(arguments, 0)?),
834                exponent: Box::new(parse_argument(arguments, 1)?),
835            }
836        }
837        "min" => {
838            require_at_least(operator, arguments, 1)?;
839            ExpressionKind::Min(parse_arguments(arguments)?)
840        }
841        "max" => {
842            require_at_least(operator, arguments, 1)?;
843            ExpressionKind::Max(parse_arguments(arguments)?)
844        }
845        "round" => unary(operator, arguments, ExpressionKind::Round)?,
846        "floor" => unary(operator, arguments, ExpressionKind::Floor)?,
847        "ceil" => unary(operator, arguments, ExpressionKind::Ceil)?,
848        "abs" => unary(operator, arguments, ExpressionKind::Abs)?,
849        "sqrt" => unary(operator, arguments, ExpressionKind::Sqrt)?,
850        "ln" => unary(operator, arguments, ExpressionKind::NaturalLog)?,
851        "log10" => unary(operator, arguments, ExpressionKind::Log10)?,
852        "log2" => unary(operator, arguments, ExpressionKind::Log2)?,
853        "sin" => unary(operator, arguments, ExpressionKind::Sin)?,
854        "cos" => unary(operator, arguments, ExpressionKind::Cos)?,
855        "tan" => unary(operator, arguments, ExpressionKind::Tan)?,
856        "asin" => unary(operator, arguments, ExpressionKind::Asin)?,
857        "acos" => unary(operator, arguments, ExpressionKind::Acos)?,
858        "atan" => unary(operator, arguments, ExpressionKind::Atan)?,
859        "pi" => {
860            expect_exact_arity(operator, arguments, 0)?;
861            ExpressionKind::Pi
862        }
863        "e" => {
864            expect_exact_arity(operator, arguments, 0)?;
865            ExpressionKind::Euler
866        }
867        "ln2" => {
868            expect_exact_arity(operator, arguments, 0)?;
869            ExpressionKind::Ln2
870        }
871        "format" => ExpressionKind::Format(parse_format(arguments)?),
872        unsupported => ExpressionKind::Unsupported {
873            operator: unsupported.to_string(),
874            arguments: parse_arguments(arguments)?,
875        },
876    };
877    Ok(Expression { kind })
878}

880fn parse_argument(arguments: &[JsonValue], index: usize) -> Result<Expression, ExpressionError> {
881    let argument = arguments.get(index).ok_or_else(|| {
882        ExpressionError::new(format!("expression argument {} is missing", index + 1))
883    })?;
884    Expression::parse(argument)
885}

887fn parse_arguments(arguments: &[JsonValue]) -> Result<Vec<Expression>, ExpressionError> {
888    arguments.iter().map(Expression::parse).collect()
889}

891fn required_string(
892    arguments: &[JsonValue],
893    index: usize,
894    operator: &str,
895) -> Result<String, ExpressionError> {
896    arguments
897        .get(index)
898        .and_then(JsonValue::as_str)
899        .map(str::to_string)
900        .ok_or_else(|| {
901            ExpressionError::new(format!(
902                "{operator} argument {} must be a string",
903                index + 1
904            ))
905        })
906}

908fn expect_exact_arity(
909    operator: &str,
910    arguments: &[JsonValue],
911    expected: usize,
912) -> Result<(), ExpressionError> {
913    if arguments.len() == expected {
914        Ok(())
915    } else {
916        Err(ExpressionError::new(format!(
917            "{operator} takes {expected} argument(s), not {}",
918            arguments.len()
919        )))
920    }
921}

923fn expect_arity(
924    operator: &str,
925    arguments: &[JsonValue],
926    minimum: usize,
927    maximum: usize,
928) -> Result<(), ExpressionError> {
929    if (minimum..=maximum).contains(&arguments.len()) {
930        Ok(())
931    } else {
932        Err(ExpressionError::new(format!(
933            "{operator} takes {minimum} to {maximum} arguments, not {}",
934            arguments.len()
935        )))
936    }
937}

939fn require_at_least(
940    operator: &str,
941    arguments: &[JsonValue],
942    minimum: usize,
943) -> Result<(), ExpressionError> {
944    if arguments.len() >= minimum {
945        Ok(())
946    } else {
947        Err(ExpressionError::new(format!(
948            "{operator} takes at least {minimum} argument(s)"
949        )))
950    }
951}

962fn binary(
963    operator: &str,
964    arguments: &[JsonValue],

974fn parse_let(arguments: &[JsonValue]) -> Result<ExpressionKind, ExpressionError> {
975    if arguments.len() < 3 || arguments.len().is_multiple_of(2) {
976        return Err(ExpressionError::new(
977            "let takes name/expression pairs followed by a body expression",
978        ));
979    }
980    let mut bindings = Vec::new();
981    let binding_count = (arguments.len() - 1) / 2;
982    for binding_index in 0..binding_count {
983        let name = required_string(arguments, binding_index * 2, "let")?;
984        let value = parse_argument(arguments, binding_index * 2 + 1)?;
985        bindings.push((name, value));
986    }
987    let body = parse_argument(arguments, arguments.len() - 1)?;
988    Ok(ExpressionKind::Let {
989        bindings,
990        body: Box::new(body),
991    })
992}

994fn parse_case(arguments: &[JsonValue]) -> Result<ExpressionKind, ExpressionError> {
995    if arguments.len() < 3 || arguments.len().is_multiple_of(2) {
996        return Err(ExpressionError::new(
997            "case takes condition/output pairs followed by a fallback expression",
998        ));
999    }
1000    let mut branches = Vec::new();
1001    let branch_count = (arguments.len() - 1) / 2;
1002    for branch_index in 0..branch_count {
1003        branches.push((
1004            parse_argument(arguments, branch_index * 2)?,
1005            parse_argument(arguments, branch_index * 2 + 1)?,
1006        ));
1007    }
1008    Ok(ExpressionKind::Case {
1009        branches,
1010        otherwise: Box::new(parse_argument(arguments, arguments.len() - 1)?),
1011    })
1012}

1014fn parse_match(arguments: &[JsonValue]) -> Result<ExpressionKind, ExpressionError> {
1015    if arguments.len() < 4 || (arguments.len() - 1).is_multiple_of(2) {
1016        return Err(ExpressionError::new(
1017            "match takes an input, label/output pairs and a fallback expression",
1018        ));
1019    }
1020    let input = parse_argument(arguments, 0)?;
1021    let mut labels = Vec::new();
1022    let pair_count = (arguments.len() - 2) / 2;
1023    for pair_index in 0..pair_count {
1024        let label_value = arguments
1025            .get(1 + pair_index * 2)
1026            .ok_or_else(|| ExpressionError::new("match label is missing"))?;
1027        let label_group = match label_value {
1028            JsonValue::Array(array) => array.iter().map(Value::from).collect(),
1029            scalar => vec![Value::from(scalar)],
1030        };
1031        let output = parse_argument(arguments, 2 + pair_index * 2)?;
1032        labels.push((label_group, output));
1033    }
1034    Ok(ExpressionKind::Match {
1035        input: Box::new(input),
1036        labels,
1037        otherwise: Box::new(parse_argument(arguments, arguments.len() - 1)?),
1038    })
1039}

1041fn parse_step(arguments: &[JsonValue]) -> Result<ExpressionKind, ExpressionError> {
1042    if arguments.len() < 2 || !(arguments.len() - 2).is_multiple_of(2) {
1043        return Err(ExpressionError::new(
1044            "step takes an input, the value below the first stop, and input/output stop pairs",
1045        ));
1046    }
1047    let input = parse_argument(arguments, 0)?;
1048    let first_output = parse_argument(arguments, 1)?;
1049    let stops = parse_stop_pairs(arguments, "step")?;
1050    Ok(ExpressionKind::Step {
1051        input: Box::new(input),
1052        first_output: Box::new(first_output),
1053        stops,
1054    })
1055}

1057fn parse_interpolate(arguments: &[JsonValue]) -> Result<ExpressionKind, ExpressionError> {
1058    if arguments.len() < 4 || !(arguments.len() - 2).is_multiple_of(2) {
1059        return Err(ExpressionError::new(
1060            "interpolate takes an interpolation, an input, and input/output stop pairs",
1061        ));
1062    }
1063    let interpolation = parse_interpolation(&arguments[0])?;
1064    let input = parse_argument(arguments, 1)?;
1065    let stops = parse_stop_pairs(arguments, "interpolate")?;
1066    if stops.len() < 2 {
1067        return Err(ExpressionError::new("interpolate needs at least two stops"));
1068    }
1069    Ok(ExpressionKind::Interpolate {
1070        interpolation,
1071        input: Box::new(input),
1072        stops,
1073    })
1074}

1076fn parse_stop_pairs(
1077    arguments: &[JsonValue],
1078    operator: &str,
1079) -> Result<Vec<(f64, Expression)>, ExpressionError> {
1080    let mut stops = Vec::new();
1081    let mut previous_input: Option<f64> = None;
1082    let mut index = 2;
1083    while index + 1 < arguments.len() {
1084        let stop_input = arguments[index].as_f64().ok_or_else(|| {
1085            ExpressionError::new(format!(
1086                "{operator} stop inputs must be numeric literals (found {})",
1087                arguments[index]
1088            ))
1089        })?;
1090        if previous_input.is_some_and(|previous| stop_input <= previous) {
1091            return Err(ExpressionError::new(format!(
1092                "{operator} stop inputs must be strictly ascending"
1093            )));
1094        }
1095        previous_input = Some(stop_input);
1096        let output = Expression::parse(&arguments[index + 1])?;
1097        stops.push((stop_input, output));
1098        index += 2;
1099    }
1100    Ok(stops)
1101}

1103fn parse_interpolation(json_value: &JsonValue) -> Result<Interpolation, ExpressionError> {
1104    if let Some(name) = json_value.as_str() {
1105        return match name {
1106            "linear" => Ok(Interpolation::Linear),
1107            other => Err(ExpressionError::new(format!(
1108                "interpolation {other:?} is not one this phase implements"
1109            ))),
1110        };
1111    }
1112    let array = json_value
1113        .as_array()
1114        .ok_or_else(|| ExpressionError::new("interpolation must be an array or a string"))?;
1115    let Some(name) = array.first().and_then(JsonValue::as_str) else {
1116        return Err(ExpressionError::new(
1117            "interpolation must start with its name",
1118        ));
1119    };
1120    let arguments = &array[1..];
1121    match name {
1122        "linear" => {
1123            expect_exact_arity("linear", arguments, 0)?;
1124            Ok(Interpolation::Linear)
1125        }
1126        "exponential" => {
1127            expect_exact_arity("exponential", arguments, 1)?;
1128            let base = arguments[0].as_f64().ok_or_else(|| {
1129                ExpressionError::new("exponential interpolation needs a numeric base")
1130            })?;
1131            if base <= 0.0 {
1132                return Err(ExpressionError::new(
1133                    "exponential interpolation base must be positive",
1134                ));
1135            }
1136            Ok(Interpolation::Exponential { base })
1137        }
1138        "cubic-bezier" => {
1139            expect_exact_arity("cubic-bezier", arguments, 4)?;
1140            let coordinates: Option<Vec<f64>> = arguments.iter().map(JsonValue::as_f64).collect();
1141            let coordinates = coordinates
1142                .ok_or_else(|| ExpressionError::new("cubic-bezier needs four numbers"))?;
1143            Ok(Interpolation::CubicBezier {
1144                x1: coordinates[0],
1145                y1: coordinates[1],
1146                x2: coordinates[2],
1147                y2: coordinates[3],
1148            })
1149        }
1150        other => Err(ExpressionError::new(format!(
1151            "interpolation {other:?} is not one this phase implements"
1152        ))),
1153    }
1154}

1156fn parse_format(arguments: &[JsonValue]) -> Result<Vec<FormatSection>, ExpressionError> {
1157    let mut sections: Vec<FormatSection> = Vec::new();
1158    for argument in arguments {
1159        match argument {
1160            JsonValue::Object(options) => {
1161                let Some(section) = sections.last_mut() else {
1162                    return Err(ExpressionError::new(
1163                        "a format section's options must follow its text",
1164                    ));
1165                };
1166                if let Some(scale) = options.get("scale") {
1167                    section.scale = Some(Expression::parse(scale)?);
1168                }
1169                if let Some(font) = options.get("font") {
1170                    section.font = Some(Expression::parse(font)?);
1171                }
1172                if let Some(text_color) = options.get("text-color") {
1173                    section.text_color = Some(Expression::parse(text_color)?);
1174                }
1175            }
1176            scalar => sections.push(FormatSection {
1177                value: Expression::parse(scalar)?,
1178                scale: None,
1179                font: None,
1180                text_color: None,
1181            }),
1182        }
1183    }
1184    if sections.is_empty() {
1185        return Err(ExpressionError::new("format needs at least one section"));
1186    }
1187    Ok(sections)
1188}

1190fn analyze(expression: &Expression) -> ExpressionAnalysis {
1191    match &expression.kind {
1192        ExpressionKind::Literal(_)
1193        | ExpressionKind::GeometryType
1194        | ExpressionKind::FeatureId
1195        | ExpressionKind::Properties
1196        | ExpressionKind::Variable(_)
1197        | ExpressionKind::Pi
1198        | ExpressionKind::Euler
1199        | ExpressionKind::Ln2 => ExpressionAnalysis::Constant,
1200        ExpressionKind::Zoom => ExpressionAnalysis::Camera,
1201        ExpressionKind::Get { key, object } => {
1202            let mut analysis = analyze(key);
1203            if let Some(object) = object {
1204                analysis = combine_analysis(analysis, analyze(object));
1205            }
1206            combine_analysis(ExpressionAnalysis::DataDriven, analysis)
1207        }
1208        ExpressionKind::Has { key } => {
1209            combine_analysis(ExpressionAnalysis::DataDriven, analyze(key))
1210        }
1211        ExpressionKind::GlobalState { key } | ExpressionKind::FeatureState { key, .. } => {
1212            combine_analysis(ExpressionAnalysis::DataDriven, analyze(key))
1213        }
1214        ExpressionKind::Unsupported { .. } => ExpressionAnalysis::DataDriven,
1215        _ => {
1216            let mut analysis = ExpressionAnalysis::Constant;
1217            for child in expression.children() {
1218                analysis = combine_analysis(analysis, analyze(child));
1219            }
1220            analysis
1221        }
1222    }
1223}
1224
1225fn combine_analysis(first: ExpressionAnalysis, second: ExpressionAnalysis) -> ExpressionAnalysis {
1226    use ExpressionAnalysis::{Camera, CameraAndData, Constant, DataDriven};
1227    match (first, second) {
1228        (Constant, other) | (other, Constant) => other,
1229        (Camera, Camera) | (DataDriven, DataDriven) => first,
1230        (Camera, DataDriven) | (DataDriven, Camera) => CameraAndData,
1231        (CameraAndData, _) | (_, CameraAndData) => CameraAndData,
1232    }
1233}

1235fn validate_zoom(
1236    expression: &Expression,
1237    zoom_is_direct_input: bool,
1238) -> Result<(), ExpressionError> {
1239    match &expression.kind {
1240        ExpressionKind::Zoom => {
1241            if zoom_is_direct_input {
1242                Ok(())
1243            } else {
1244                Err(ExpressionError::new(
1245                    "[\"zoom\"] may only be the direct input of a top-level step or interpolate",
1246                ))
1247            }
1248        }
1249        ExpressionKind::Step {
1250            input,
1251            first_output,
1252            stops,
1253        } => {
1254            validate_zoom(input, true)?;
1255            validate_zoom(first_output, false)?;
1256            for (_, output) in stops {
1257                validate_zoom(output, false)?;
1258            }
1259            Ok(())
1260        }
1261        ExpressionKind::Interpolate { input, stops, .. } => {
1262            validate_zoom(input, true)?;
1263            for (_, output) in stops {
1264                validate_zoom(output, false)?;
1265            }
1266            Ok(())
1267        }
1268        _ => {
1269            for child in expression.children() {
1270                validate_zoom(child, false)?;
1271            }
1272            Ok(())
1273        }
1274    }
