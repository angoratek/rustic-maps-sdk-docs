1//! Parse, validation and evaluation diagnostics for style documents.
2//!
3//! Every error is a value, never a panic: a style author receives the
4//! operator or property name, the failing text and a sentence explaining the
5//! rule that was broken. Evaluation failures are separated from parse
6//! failures because the renderer treats a property that fails at runtime as
7//! "use the property's documented default" rather than as an unloadable
8//! document.
9
10use thiserror::Error;
11
12/// A CSS colour that cannot be parsed.
13#[derive(Debug, Clone, PartialEq, Eq, Error)]
14#[error("invalid colour {text:?}: {detail}")]
15pub struct ColorParseError {
16    /// The text that was offered as a colour.
17    pub text: String,
18    /// The rule the text violated.
19    pub detail: String,
20}
21
22impl ColorParseError {
23    /// Builds a colour diagnostic.
24    pub fn new(text: impl Into<String>, detail: impl Into<String>) -> Self {
25        Self {
26            text: text.into(),
27            detail: detail.into(),
28        }
29    }
30}

87/// A whole style document that cannot be parsed.
88#[derive(Debug, Clone, PartialEq, Eq, Error)]
89pub enum StyleError {
90    /// The document text is not valid JSON.
91    #[error("style document is not valid JSON: {detail}")]
92    Json {
93        /// The underlying JSON parser message.
94        detail: String,
95    },
96    /// The document root is not a JSON object.
97    #[error("style document must be a JSON object")]
98    DocumentNotObject,
99    /// The document declares a version other than 8.
100    #[error("style document version {version} is not supported; version 8 is required")]
101    UnsupportedVersion {
102        /// The version the document declares.
103        version: u64,
104    },
105    /// A required structural member is absent or has the wrong type.
106    #[error("style document member {member:?} is invalid: {detail}")]
107    Member {
108        /// The member name.
109        member: String,
110        /// The rule the member violated.
111        detail: String,
112    },
113    /// A layer has no `id` or no `type`.
114    #[error("layer at index {layer_index} is invalid: {detail}")]
115    Layer {
116        /// The layer's document position.
117        layer_index: usize,
118        /// The rule the layer violated.
119        detail: String,
120    },
121    /// A source declaration is invalid.
122    #[error("source {source_id:?} is invalid: {detail}")]
123    Source {
124        /// The source id the document uses.
125        source_id: String,
126        /// The rule the source violated.
127        detail: String,
128    },
129    /// An expression failed to parse.
130    #[error("expression invalid: {0}")]
131    Expression(#[from] ExpressionError),
132    /// A filter failed to normalise.
133    #[error("filter invalid: {0}")]
134    Filter(#[from] FilterError),
135    /// A colour failed to parse.
136    #[error("colour invalid: {0}")]
137    Color(#[from] ColorParseError),
138}

140/// The part of a document an unsupported property belongs to.
141#[derive(Debug, Clone, PartialEq, Eq)]
142pub enum UnsupportedScope {
143    /// A root property.
144    Document,
145    /// A layer property, named by the layer's `id`.
146    Layer {
147        /// The layer id.
148        layer_id: String,
149    },
150    /// A source property, named by the source id.
151    Source {
152        /// The source id.
153        source_id: String,
154    },
155}
156
157impl UnsupportedScope {
158    /// A short human-readable description of the scope.
159    pub fn description(&self) -> String {
160        match self {
161            Self::Document => "document".to_string(),
162            Self::Layer { layer_id } => format!("layer {layer_id:?}"),
163            Self::Source { source_id } => format!("source {source_id:?}"),
164        }
165    }
166}

168/// A property the parser recognizes but this phase cannot apply.
169///
170/// Unknown and unimplemented properties are reported rather than silently
171/// dropped, so a caller can tell the difference between "the style paints
172/// this" and "the style asked for something the SDK does not implement yet".
173#[derive(Debug, Clone, PartialEq, Eq)]
174pub struct UnsupportedProperty {
175    /// The document part that carries the property.
176    pub scope: UnsupportedScope,
177    /// The property's JSON member name.
178    pub property_name: String,
179    /// Why the property is not applied.
180    pub detail: String,
181}
182
183impl UnsupportedProperty {
184    /// Builds a diagnostic for one property.
185    pub fn new(
186        scope: UnsupportedScope,
187        property_name: impl Into<String>,
188        detail: impl Into<String>,
189    ) -> Self {
190        Self {
191            scope,
192            property_name: property_name.into(),
193            detail: detail.into(),
194        }
195    }
