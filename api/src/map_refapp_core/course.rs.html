1//! Circular course-over-ground smoothing.
2//!
3//! A course-over-ground is an angle, so an arithmetic median is wrong around
4//! north. The smoother keeps a fixed window of recent courses and reports the
5//! windowed sample that minimises the sum of angular distances to every other
6//! sample in the window: the circular median.
7
8/// One full circle in degrees, the modulus bearings wrap around.
9pub const FULL_CIRCLE_DEGREES: f64 = 360.0;
10
11/// Half a circle in degrees, the shortest-way wrap boundary.
12pub const HALF_CIRCLE_DEGREES: f64 = 180.0;
13
14/// The default number of recent fixes the median smooths over.
15///
16/// Five fixes smooth the walk demo's stepwise courses without lagging turns
17/// noticeably at the accelerated replay cadence.
18pub const COURSE_SMOOTHING_WINDOW_FIXES: usize = 5;
19
20/// The angular distance between two bearings, in degrees, always in
21/// `[0, 180]`.
22pub fn angular_distance_degrees(first_degrees: f64, second_degrees: f64) -> f64 {
23    let absolute_difference = (first_degrees - second_degrees).abs() % FULL_CIRCLE_DEGREES;
24    if absolute_difference > HALF_CIRCLE_DEGREES {
25        FULL_CIRCLE_DEGREES - absolute_difference
26    } else {
27        absolute_difference
28    }
29}
30
31/// A fixed-window circular median over recent course-over-ground samples.
32#[derive(Debug, Clone, PartialEq)]
33pub struct CourseMedianSmoother {
34    window_fix_count: usize,
35    recent_course_degrees: Vec<f64>,
36}
37
38impl CourseMedianSmoother {
39    /// Creates a smoother over `window_fix_count` recent samples.
40    ///
41    /// A zero-length window is valid and reports no median; it is useful only
42    /// to a caller that intentionally disables smoothing.
43    pub fn new(window_fix_count: usize) -> Self {
44        Self {
45            window_fix_count,
46            recent_course_degrees: Vec::with_capacity(window_fix_count),
47        }
48    }
49
50    /// Creates a smoother over the default window.
51    pub fn with_default_window() -> Self {
52        Self::new(COURSE_SMOOTHING_WINDOW_FIXES)
53    }
54
55    /// The window length in fixes.
56    pub fn window_fix_count(&self) -> usize {
57        self.window_fix_count
58    }
59
60    /// Adds one fix's course-over-ground, dropping the window's oldest sample
61    /// once the window is full.
62    pub fn add(&mut self, course_degrees: f64) {
63        self.recent_course_degrees.push(course_degrees);
64        if self.recent_course_degrees.len() > self.window_fix_count {
65            self.recent_course_degrees.remove(0);
66        }
67    }
68
69    /// The circular median, or `None` while the window holds no sample.
70    ///
71    /// Ties keep the earliest sample in the window. A window containing a
72    /// non-finite sample reports `None` for that comparison; a later finite
73    /// sample can still become the median.
74    pub fn median(&self) -> Option<f64> {
75        let mut best_median = None;
76        let mut best_distance_sum = f64::INFINITY;
77        for candidate_degrees in &self.recent_course_degrees {
78            let distance_sum = self
79                .recent_course_degrees
80                .iter()
81                .map(|other_degrees| angular_distance_degrees(*candidate_degrees, *other_degrees))
82                .sum::<f64>();
83            if distance_sum < best_distance_sum {
84                best_distance_sum = distance_sum;
85                best_median = Some(*candidate_degrees);
86            }
87        }
88        best_median
89    }
90
91    /// Empties the window.
92    pub fn reset(&mut self) {
93        self.recent_course_degrees.clear();
94    }
95
96    /// The windowed samples, oldest first.
97    pub fn samples(&self) -> &[f64] {
98        &self.recent_course_degrees
99    }
100}
101
102impl Default for CourseMedianSmoother {
103    fn default() -> Self {
104        Self::with_default_window()
105    }
106}
