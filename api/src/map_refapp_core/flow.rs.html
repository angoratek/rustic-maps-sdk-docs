1//! The reference-application flow vocabulary and reducer.
2//!
3//! The reducer is a total pure function:
4//!
5//! ```
6//! use map_refapp_core::{FlowEvent, FlowState, advance_flow};
7//! let transition = advance_flow(FlowState::HomePeek, FlowEvent::StartTrip);
8//! assert_eq!(transition.state, FlowState::Navigating);
9//! ```
10//!
11//! It answers one question: given the flow state and an event, what should the
12//! user see next? It does not own the SDK handle, the route record, the camera
13//! pose, the credential store, the speech engine or the view lifecycle. Every
14//! event from every state returns either a new state or an explicit unchanged
15//! transition, whose [`FlowTransition::unchanged`] flag is `true`.
16//!
17//! Settings and diagnostics are overlays, not flow positions. Their events
18//! leave the flow state untouched and report the overlay outcome, because
19//! opening either surface never moves the visitor within the plan or
20//! navigation flow. [`FlowState::ProviderSettings`] and
21//! [`FlowState::Diagnostics`] therefore stay in the vocabulary for a
22//! platform's own overlay model, but [`advance_flow`] never returns them.
23
24/// One reference-application flow position.
25#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
26pub enum FlowState {
27    /// The home sheet's peek.
28    HomePeek,
29    /// The two-field search view.
30    PlanSearch,
31    /// A plan request is in flight.
32    Planning,
33    /// A route is present and can start.
34    RoutePlanned,
35    /// A simulated or device-driven trip is running.
36    Navigating,
37    /// The trip reached its end.
38    Arrived,
39    /// The provider-settings overlay vocabulary; the reducer keeps the flow
40    /// position beneath the overlay and never returns this state.
41    ProviderSettings,
42    /// The diagnostics overlay vocabulary; the reducer keeps the flow
43    /// position beneath the overlay and never returns this state.
44    Diagnostics,
45}

319/// The reducer's answer.
320#[derive(Debug, Clone, Copy, PartialEq, Eq)]
321pub struct FlowTransition {
322    /// The flow state after the event.
323    pub state: FlowState,
324    /// What the user should see.
325    pub outcome: FlowOutcome,
326    /// The copy IDs the platform must render for this transition.
327    pub copy_ids: &'static [&'static str],
328    /// Whether the event left the flow state unchanged.
329    pub unchanged: bool,
330}
331
332impl FlowTransition {
333    /// Whether the event left the flow state unchanged.
334    pub const fn is_unchanged(&self) -> bool {
335        self.unchanged
336    }

376const fn changed(
377    state: FlowState,
378    outcome: FlowOutcome,
379    copy_ids: &'static [&'static str],
380) -> FlowTransition {
381    FlowTransition {
382        state,
383        outcome,
384        copy_ids,
385        unchanged: false,
386    }

389const fn unchanged_with(
390    state: FlowState,
391    outcome: FlowOutcome,
392    copy_ids: &'static [&'static str],
393) -> FlowTransition {
394    FlowTransition {
395        state,
396        outcome,
397        copy_ids,
398        unchanged: true,
399    }

402const fn unchanged(state: FlowState) -> FlowTransition {
403    unchanged_with(state, FlowOutcome::Unchanged, COPY_NONE)
404}

406/// The settings and diagnostics overlays never move the flow position.
407fn overlay_transition(state: FlowState, event: FlowEvent) -> Option<FlowTransition> {
408    let transition = match event {
409        FlowEvent::OpenSettings => unchanged_with(state, FlowOutcome::OpenSettings, COPY_NONE),
410        FlowEvent::ProviderSelected => {
411            unchanged_with(state, FlowOutcome::ProviderSelected, COPY_PROVIDER_ROW)
412        }
413        FlowEvent::CredentialApplied => unchanged_with(
414            state,
415            FlowOutcome::CredentialApplied,
416            COPY_CREDENTIAL_APPLIED,
417        ),
418        FlowEvent::OpenDiagnostics => {
419            unchanged_with(state, FlowOutcome::OpenDiagnostics, COPY_DIAGNOSTICS_TITLE)
420        }
421        FlowEvent::DebugDetailToggled => {
422            unchanged_with(state, FlowOutcome::DebugDetailToggled, COPY_DEBUG_TOGGLE)
423        }
424        FlowEvent::Clear => unchanged_with(state, FlowOutcome::Clear, COPY_CLEAR),
425        FlowEvent::Close => unchanged_with(state, FlowOutcome::Close, COPY_CLOSE),
426        _ => return None,
427    };
428    Some(transition)
429}

431/// The total flow reducer.
432///
433/// Assignment never plans: only [`FlowEvent::PlanRequested`] enters
434/// [`FlowState::Planning`]. One trip runs at a time: [`FlowEvent::StartTrip`]
435/// moves to [`FlowState::Navigating`] from the home peek or the route card.
436/// Diagnostics and provider settings are overlays, so their events return an
437/// unchanged transition at the current flow position.
438pub fn advance_flow(state: FlowState, event: FlowEvent) -> FlowTransition {
439    if let Some(overlay) = overlay_transition(state, event) {
440        return overlay;
441    }
442    match state {
443        FlowState::HomePeek => match event {
444            FlowEvent::OpenSearch(field) => changed(
445                FlowState::PlanSearch,
446                FlowOutcome::OpenSearch(field),
447                open_search_copy(field),
448            ),
449            FlowEvent::StartTrip => {
450                changed(FlowState::Navigating, FlowOutcome::StartTrip, COPY_DRIVE)
451            }
452            _ => unchanged(state),
453        },
454        FlowState::PlanSearch => match event {
455            FlowEvent::FieldTextChanged(field) => {
456                unchanged_with(state, FlowOutcome::FieldTextChanged(field), COPY_NONE)
457            }
458            FlowEvent::SearchRequested(field) => {
459                unchanged_with(state, FlowOutcome::SearchRequested(field), COPY_SEARCHING)
460            }
461            FlowEvent::ResultAssigned(field) => {
462                unchanged_with(state, FlowOutcome::ResultAssigned(field), COPY_PLAN_READY)
463            }
464            FlowEvent::MyLocationRequested(field) => unchanged_with(
465                state,
466                FlowOutcome::MyLocationRequested(field),
467                COPY_MY_LOCATION,
468            ),
469            FlowEvent::SwapRequested => unchanged_with(state, FlowOutcome::Swap, COPY_SWAP),
470            FlowEvent::PlanRequested => changed(
471                FlowState::Planning,
472                FlowOutcome::PlanRequested,
473                COPY_PLANNING,
474            ),
475            FlowEvent::Done => changed(FlowState::HomePeek, FlowOutcome::Done, COPY_PEEK),
476            _ => unchanged(state),
477        },
478        FlowState::Planning => match event {
479            FlowEvent::PlanSucceeded => changed(
480                FlowState::RoutePlanned,
481                FlowOutcome::PlanSucceeded,
482                COPY_ROUTE_PLANNED,
483            ),
484            FlowEvent::PlanFailed => changed(
485                FlowState::PlanSearch,
486                FlowOutcome::PlanFailed,
487                COPY_PLAN_FAILED,
488            ),
489            _ => unchanged(state),
490        },
491        FlowState::RoutePlanned => match event {
492            FlowEvent::StartTrip => {
493                changed(FlowState::Navigating, FlowOutcome::StartTrip, COPY_DRIVE)
494            }
495            FlowEvent::EditRoute => {
496                changed(FlowState::PlanSearch, FlowOutcome::EditRoute, COPY_EDIT)
497            }
498            FlowEvent::PlanRequested => changed(
499                FlowState::Planning,
500                FlowOutcome::PlanRequested,
501                COPY_PLANNING,
502            ),
503            _ => unchanged(state),
504        },
505        FlowState::Navigating => match event {
506            FlowEvent::Fix => unchanged_with(state, FlowOutcome::Fix, COPY_FIX),
507            FlowEvent::SpeedChanged => {
508                unchanged_with(state, FlowOutcome::SpeedChanged, COPY_SPEED_CHANGED)
509            }
510            FlowEvent::Pause => unchanged_with(state, FlowOutcome::Pause, COPY_AFTER_PAUSE),
511            FlowEvent::Resume => unchanged_with(state, FlowOutcome::Resume, COPY_AFTER_RESUME),
512            FlowEvent::Exit => changed(FlowState::HomePeek, FlowOutcome::Exit, COPY_EXIT),
513            FlowEvent::Arrived => changed(FlowState::Arrived, FlowOutcome::Arrived, COPY_ARRIVED),
514            _ => unchanged(state),
515        },
516        FlowState::Arrived => match event {
517            FlowEvent::Exit => changed(FlowState::HomePeek, FlowOutcome::Exit, COPY_EXIT),
518            _ => unchanged(state),
519        },
520        FlowState::ProviderSettings | FlowState::Diagnostics => unchanged(state),
521    }
522}

524const fn open_search_copy(field: FlowField) -> &'static [&'static str] {
525    match field {
526        FlowField::Origin => COPY_OPEN_SEARCH_ORIGIN,
527        FlowField::Destination => COPY_OPEN_SEARCH_DESTINATION,
