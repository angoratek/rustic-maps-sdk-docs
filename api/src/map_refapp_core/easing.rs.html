1//! Easing primitives used by the reference applications' camera moves.
2//!
3//! The platform frame pump supplies every timestamp; these helpers are pure.
4//! All durations are milliseconds so the same values travel across the four
5//! targets without a nanosecond conversion at each call site.
6
7use crate::course::{FULL_CIRCLE_DEGREES, HALF_CIRCLE_DEGREES};
8
9/// The follow camera bearing's exponential chase time constant.
10pub const FOLLOW_BEARING_CHASE_TIME_CONSTANT_MILLISECONDS: f64 = 150.0;
11
12/// The chase snaps to its target under this remaining angular distance; the
13/// eased value is visually identical to the target.
14pub const FOLLOW_BEARING_SNAP_DISTANCE_DEGREES: f64 = 0.05;
15
16/// The first frame of a chase carries one display frame's worth of catching
17/// up: one 60 Hz frame.
18pub const CHASE_FIRST_FRAME_DELTA_MILLISECONDS: f64 = 1000.0 / 60.0;
19
20/// Smoothstep easing (`3t² − 2t³`): eased in and eased out.
21///
22/// The input is clamped into `[0, 1]`, so a caller may pass an unclamped
23/// progress fraction.
24pub fn smoothstep_eased_progress(linear_progress: f64) -> f64 {
25    let clamped_progress = linear_progress.clamp(0.0, 1.0);
26    clamped_progress * clamped_progress * (3.0 - 2.0 * clamped_progress)
27}
28
29/// The camera bearing's smallest off-north angular distance in degrees,
30/// always in `[0, 180]`.
31pub fn bearing_off_north_degrees(bearing_degrees: f64) -> f64 {
32    let normalized_degrees =
33        ((bearing_degrees % FULL_CIRCLE_DEGREES) + FULL_CIRCLE_DEGREES) % FULL_CIRCLE_DEGREES;
34    if normalized_degrees <= HALF_CIRCLE_DEGREES {
35        normalized_degrees
36    } else {
37        FULL_CIRCLE_DEGREES - normalized_degrees
38    }
39}
40
41/// The shortest signed angular difference `to − from` in degrees, wrapped to
42/// `[-180, 180]`.
43pub fn signed_angular_difference_degrees(to_degrees: f64, from_degrees: f64) -> f64 {
44    let mut difference = (to_degrees - from_degrees) % FULL_CIRCLE_DEGREES;
45    if difference > HALF_CIRCLE_DEGREES {
46        difference -= FULL_CIRCLE_DEGREES;
47    }
48    if difference < -HALF_CIRCLE_DEGREES {
49        difference += FULL_CIRCLE_DEGREES;
50    }
51    difference
52}

54/// A linear-in-time, smoothstep-eased value move begun with a frame-clock
55/// timestamp.
56///
57/// The compass's eased north-up return and the 2D/3D pitch swing both run as
58/// this timed ease over the design language's one eased-camera-move duration.
59#[derive(Debug, Clone, PartialEq)]
60pub struct SmoothstepTimedEase {
61    start_milliseconds: Option<f64>,
62    from_value: f64,
63    to_value: f64,
64    duration_milliseconds: f64,
65}
66
67impl SmoothstepTimedEase {
68    /// Creates an idle ease with no target.
69    pub fn new() -> Self {
70        Self {
71            start_milliseconds: None,
72            from_value: 0.0,
73            to_value: 0.0,
74            duration_milliseconds: 0.0,
75        }
76    }
77
78    /// Begins, or restarts, an ease from one value to another.
79    pub fn begin(
80        &mut self,
81        start_milliseconds: f64,
82        from_value: f64,
83        to_value: f64,
84        duration_milliseconds: f64,
85    ) {
86        self.start_milliseconds = Some(start_milliseconds);
87        self.from_value = from_value;
88        self.to_value = to_value;
89        self.duration_milliseconds = duration_milliseconds;
90    }
91
92    /// Stops an in-flight ease without applying anything.
93    pub fn reset(&mut self) {
94        self.start_milliseconds = None;
95    }
96
97    /// Whether an ease is in flight.
98    pub fn is_running(&self) -> bool {
99        self.start_milliseconds.is_some()
100    }
101
102    /// The ease's target value.
103    pub fn target_value(&self) -> f64 {
104        self.to_value
105    }
106
107    /// Advances the ease at `now_milliseconds`.
108    ///
109    /// Returns the current eased value and whether the move finished on this
110    /// call. An idle ease reports its target and `true`.
111    pub fn advance(&mut self, now_milliseconds: f64) -> (f64, bool) {
112        let Some(start_milliseconds) = self.start_milliseconds else {
113            return (self.to_value, true);
114        };
115        let elapsed_milliseconds = now_milliseconds - start_milliseconds;
116        if self.duration_milliseconds <= 0.0 || elapsed_milliseconds >= self.duration_milliseconds {
117            self.start_milliseconds = None;
118            return (self.to_value, true);
119        }
120        let progress = elapsed_milliseconds / self.duration_milliseconds;
121        let eased_value = self.from_value
122            + (self.to_value - self.from_value) * smoothstep_eased_progress(progress);
123        (eased_value, false)
124    }
125}
126
127impl Default for SmoothstepTimedEase {
128    fn default() -> Self {
129        Self::new()
130    }
131}

133/// An exponential chase toward a moving target.
134///
135/// The follow camera's bearing chases the median-smoothed course this way: a
136/// time-constant chase settles on its own when the target stops moving, and
137/// snaps when the remaining angular distance falls under
138/// [`FOLLOW_BEARING_SNAP_DISTANCE_DEGREES`].
139#[derive(Debug, Clone, PartialEq)]
140pub struct ExponentialValueChase {
141    time_constant_milliseconds: f64,
142    last_frame_milliseconds: Option<f64>,
143}
144
145impl ExponentialValueChase {
146    /// Creates a chase with the given time constant in milliseconds.
147    pub fn new(time_constant_milliseconds: f64) -> Self {
148        Self {
149            time_constant_milliseconds,
150            last_frame_milliseconds: None,
151        }
152    }
153
154    /// Creates a chase with the follow camera's token time constant.
155    pub fn with_follow_bearing_time_constant() -> Self {
156        Self::new(FOLLOW_BEARING_CHASE_TIME_CONSTANT_MILLISECONDS)
157    }
158
159    /// The chase's time constant in milliseconds.
160    pub fn time_constant_milliseconds(&self) -> f64 {
161        self.time_constant_milliseconds
162    }
163
164    /// Advances the chase toward `target` at `now_milliseconds`, returning the
165    /// next current value.
166    ///
167    /// The first call after a reset folds one display frame's worth of
168    /// catching up. A frame clock that goes backwards is treated as a first
169    /// frame. A non-finite or non-positive time constant snaps to the target
170    /// rather than producing a meaningless value.
171    pub fn advance(&mut self, current: f64, target: f64, now_milliseconds: f64) -> f64 {
172        let last_milliseconds = self.last_frame_milliseconds;
173        self.last_frame_milliseconds = Some(now_milliseconds);
174        let delta_time_milliseconds = match last_milliseconds {
175            Some(last) if now_milliseconds >= last => now_milliseconds - last,
176            _ => CHASE_FIRST_FRAME_DELTA_MILLISECONDS,
177        };
178        let delta_degrees = signed_angular_difference_degrees(target, current);
179        if delta_degrees.abs() <= FOLLOW_BEARING_SNAP_DISTANCE_DEGREES
180            || !self.time_constant_milliseconds.is_finite()
181            || self.time_constant_milliseconds <= 0.0
182        {
183            return target;
184        }
185        let follow_factor =
186            1.0 - (-delta_time_milliseconds / self.time_constant_milliseconds).exp();
187        current + delta_degrees * follow_factor
188    }
189
190    /// Clears the frame clock so the next call is treated as a first frame.
191    pub fn reset(&mut self) {
192        self.last_frame_milliseconds = None;
193    }
194}
