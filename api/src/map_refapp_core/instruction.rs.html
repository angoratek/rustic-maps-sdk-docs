1//! The instruction-text composition the reference applications present.
2//!
3//! The turn banner, the then-line and the speech channel read the same
4//! provider-neutral payload from the SDK contract, and all three apply one
5//! rule: the composed text first, the provider-authored text second, and — on
6//! the display channel — the honest unavailable line when the provider
7//! authored no text at all. The banner appends the structured roundabout exit
8//! index when the maneuver carries one.
9//!
10//! Provider text is carried verbatim: the rule chooses between channels and
11//! appends the exit suffix; it never rewrites, shortens or invents text.
12
13/// The honest display line rendered for a maneuver whose provider authored no
14/// instruction text.
15pub const UNAVAILABLE_INSTRUCTION_TEXT: &str = "No instruction text provided";
16
17/// The roundabout exit suffix's fixed text before the exit index.
18const ROUNDABOUT_EXIT_SUFFIX_PREFIX: &str = " (exit ";
19
20/// The roundabout exit suffix's fixed text after the exit index.
21const ROUNDABOUT_EXIT_SUFFIX_TERMINATOR: &str = ")";
22
23/// Returns the display text of one maneuver instruction.
24///
25/// The composed display text wins; when it is absent the provider-authored
26/// display text renders verbatim; when neither exists the honest
27/// [`UNAVAILABLE_INSTRUCTION_TEXT`] line renders. The contract's `Unavailable`
28/// instruction-text kind carries null text fields, so it follows the same
29/// path and renders the honest line rather than an invented instruction.
30///
31/// # Parameters
32/// - `composed_display_text` — the banner variant the contract composed (the
33///   live distance prefix plus the provider display text), when present.
34/// - `provider_display_text` — the provider-authored display text, verbatim,
35///   when present.
36///
37/// # Returns
38/// The display line for the banner and then-line, never empty for absent
39/// provider text.
40pub fn instruction_display_text(
41    composed_display_text: Option<&str>,
42    provider_display_text: Option<&str>,
43) -> String {
44    composed_display_text
45        .or(provider_display_text)
46        .unwrap_or(UNAVAILABLE_INSTRUCTION_TEXT)
47        .to_string()
48}
49
50/// Returns the speech text of one maneuver instruction.
51///
52/// The composed speech text wins; when it is absent the provider-authored
53/// speech text is returned verbatim. When neither exists the maneuver is
54/// never spoken: speaking an invented line is not an option, so the answer is
55/// `None`.
56///
57/// # Parameters
58/// - `composed_speech_text` — the text-to-speech variant the contract
59///   composed (the live distance prefix plus the provider speech text), when
60///   present.
61/// - `provider_speech_text` — the provider-authored speech text, verbatim,
62///   when present.
63///
64/// # Returns
65/// The line to speak verbatim, or `None` when the provider authored none.
66pub fn instruction_speech_text<'text>(
67    composed_speech_text: Option<&'text str>,
68    provider_speech_text: Option<&'text str>,
69) -> Option<&'text str> {
70    composed_speech_text.or(provider_speech_text)
71}
72
73/// Returns the banner's roundabout exit suffix.
74///
75/// # Parameters
76/// - `roundabout_exit_index` — the structured exit index of a roundabout
77///   maneuver, starting at one; every other maneuver has `None`.
78///
79/// # Returns
80/// `" (exit <index>)"` for a roundabout maneuver, or the empty string for
81/// every other maneuver.
82pub fn roundabout_exit_suffix(roundabout_exit_index: Option<u8>) -> String {
83    match roundabout_exit_index {
84        Some(exit_index) => format!(
85            "{ROUNDABOUT_EXIT_SUFFIX_PREFIX}{exit_index}{ROUNDABOUT_EXIT_SUFFIX_TERMINATOR}"
86        ),
87        None => String::new(),
88    }
89}

91/// Returns the turn banner's full instruction line.
92///
93/// The line is the display text ([`instruction_display_text`]) followed by the
94/// roundabout exit suffix ([`roundabout_exit_suffix`]). The then-line renders
95/// the display text alone; the suffix belongs to the banner only.
96///
97/// # Parameters
98/// - `composed_display_text` — the banner variant the contract composed, when
99///   present.
100/// - `provider_display_text` — the provider-authored display text, verbatim,
101///   when present.
102/// - `roundabout_exit_index` — the structured exit index of a roundabout
103///   maneuver, when the contract carries one.
104///
105/// # Returns
106/// The banner line, including the honest unavailable line when the provider
107/// authored no display text.
108pub fn instruction_banner_text(
109    composed_display_text: Option<&str>,
110    provider_display_text: Option<&str>,
111    roundabout_exit_index: Option<u8>,
112) -> String {
113    let mut banner_text = instruction_display_text(composed_display_text, provider_display_text);
114    banner_text.push_str(&roundabout_exit_suffix(roundabout_exit_index));
115    banner_text
116}
