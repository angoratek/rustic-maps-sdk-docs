1//! The recorded-trace cursor and dispatch cadence.
2//!
3//! A recorded trace replays its samples verbatim. The cursor owns which sample
4//! is next; the timer and the speed choice stay platform-side. The dispatch
5//! interval derives from the trace's recorded sample interval and the playback
6//! acceleration factor the caller folds the selected speed multiplier into.
7
8/// Milliseconds in one second.
9pub const MILLISECONDS_PER_SECOND: f64 = 1000.0;
10
11/// The shortest dispatch interval, in milliseconds, so a fast speed multiplier
12/// never spins the platform timer.
13pub const MINIMUM_TRACE_DISPATCH_INTERVAL_MILLISECONDS: u64 = 1;
14
15/// A forward-only cursor over a recorded trace's samples.
16#[derive(Debug, Clone, PartialEq, Eq)]
17pub struct TraceCursor {
18    sample_count: usize,
19    next_sample_index: usize,
20}
21
22impl TraceCursor {
23    /// Creates a cursor at the first sample of a trace with `sample_count`
24    /// samples.
25    pub fn new(sample_count: usize) -> Self {
26        Self {
27            sample_count,
28            next_sample_index: 0,
29        }
30    }
31
32    /// The trace's sample count.
33    pub fn sample_count(&self) -> usize {
34        self.sample_count
35    }
36
37    /// The next sample's index, or `None` when every sample was delivered.
38    pub fn next_sample_index(&self) -> Option<usize> {
39        (self.next_sample_index < self.sample_count).then_some(self.next_sample_index)
40    }
41
42    /// The next sample's index without consuming it.
43    pub fn peek(&self) -> Option<usize> {
44        self.next_sample_index()
45    }
46
47    /// Consumes and returns the next sample's index, or `None` at the end.
48    pub fn advance(&mut self) -> Option<usize> {
49        let sample_index = self.next_sample_index()?;
50        self.next_sample_index += 1;
51        Some(sample_index)
52    }
53
54    /// Whether every sample was delivered.
55    pub fn is_finished(&self) -> bool {
56        self.next_sample_index >= self.sample_count
57    }
58
59    /// The number of samples not yet delivered.
60    pub fn remaining_sample_count(&self) -> usize {
61        self.sample_count.saturating_sub(self.next_sample_index)
62    }
63
64    /// Rewinds the cursor to the first sample.
65    pub fn reset(&mut self) {
66        self.next_sample_index = 0;
67    }
68
69    /// Moves the cursor to `sample_index`, clamped to the trace's bounds.
70    pub fn seek_to_sample_index(&mut self, sample_index: usize) {
71        self.next_sample_index = sample_index.min(self.sample_count);
72    }
73}
74
75/// The replay dispatch interval for a recorded trace, in whole milliseconds.
76///
77/// The recorded sample interval is divided by the playback acceleration
78/// factor, then floored at
79/// [`MINIMUM_TRACE_DISPATCH_INTERVAL_MILLISECONDS`]. A non-finite or
80/// non-positive input cannot produce a meaningful cadence and therefore takes
81/// the minimum interval rather than panicking or returning zero.
82pub fn trace_dispatch_interval_milliseconds(
83    recorded_sample_interval_seconds: f64,
84    playback_acceleration_factor: f64,
85) -> u64 {
86    if !recorded_sample_interval_seconds.is_finite()
87        || recorded_sample_interval_seconds <= 0.0
88        || !playback_acceleration_factor.is_finite()
89        || playback_acceleration_factor <= 0.0
90    {
91        return MINIMUM_TRACE_DISPATCH_INTERVAL_MILLISECONDS;
92    }
93    let dispatch_seconds = recorded_sample_interval_seconds / playback_acceleration_factor;
94    let dispatch_milliseconds = dispatch_seconds * MILLISECONDS_PER_SECOND;
95    if !dispatch_milliseconds.is_finite() || dispatch_milliseconds <= 0.0 {
96        return MINIMUM_TRACE_DISPATCH_INTERVAL_MILLISECONDS;
97    }
98    dispatch_milliseconds.max(MINIMUM_TRACE_DISPATCH_INTERVAL_MILLISECONDS as f64) as u64
99}
