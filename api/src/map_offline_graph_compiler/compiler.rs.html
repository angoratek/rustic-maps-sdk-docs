1//! The compiler pipeline: OpenStreetMap PBF extract → deterministic
2//! [`OfflineRoutingNetworkGraph`] archive bytes.
3//!
4//! # Determinism contract
5//!
6//! Two runs over the same PBF produce byte-identical archives. The ordering
7//! rules that make this true:
8//!
9//! - nodes keep their surviving graph nodes in ascending OSM node id order;
10//! - edges sort by (`source_node_index`, `target_node_index`, originating
11//!   OSM way id);
12//! - lane guidance records (none emitted — see [`LaneGuidanceRecord`]'s
13//!   documentation for the schema refusal) would sort by edge index;
14//! - turn restrictions sort by (`from_edge_index`, `via_node_index`,
15//!   `to_edge_index`), and instruction strings are emitted in that same
16//!   restriction order;
17//! - the writer's canonical resource order and zero-filled alignment gaps
18//!   are fixed in [`crate::archive`].
19//!
20//! # Slices
21//!
22//! The pipeline collects nodes, drivable ways and restriction relations in
23//! one sequential pass ([`profile`] applies the routing rules), then
24//! resolves graph topology and restrictions in memory. Scale limits are a
25//! documented M7 slice-2 seam: everything is held in memory (metro-scale
26//! extracts only) and unresolved restrictions are skipped rather than
27//! failing the compile.
28

45/// Haversine computation uses this mean Earth radius; segment distances use
46/// the quantized archive coordinates so they are exactly re-derivable from
47/// the archive itself.
48const EARTH_RADIUS_METERS: f64 = 6_371_000.0;
49const METERS_PER_KILOMETER: f64 = 1_000.0;
50const SECONDS_PER_HOUR: f64 = 3_600.0;
51const MILLISECONDS_PER_SECOND: f64 = 1_000.0;

53/// Compiler-side failure: parsing, I/O, or graph overflow.
54#[derive(Debug)]
55pub enum CompilerError {
56    /// Reading the input PBF failed.
57    InputReadFailure(io::Error),
58    /// The PBF parser rejected the extract.
59    OsmParsingFailure(String),
60    /// The extract exceeded this minimal compiler's addressable size.
61    GraphCapacityExceeded(String),
62}
63
64impl std::fmt::Display for CompilerError {
65    fn fmt(&self, formatter: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
66        match self {
67            Self::InputReadFailure(error) => {
68                write!(formatter, "input read failed: {error}")
69            }
70            Self::OsmParsingFailure(detail) => {
71                write!(formatter, "OSM PBF parsing failed: {detail}")
72            }
73            Self::GraphCapacityExceeded(detail) => {
74                write!(
75                    formatter,
76                    "graph exceeds this compiler's capacity: {detail}"
77                )
78            }
79        }
80    }
81}

85/// Compiles an OpenStreetMap PBF extract into deterministic archive bytes.
86///
87/// # Errors
88/// See [`CompilerError`]; note archive-side validation failures surface as
89/// `geospatial_types::SDKError` only on the read path ([`crate::archive`]).
90pub fn compile_osm_extract_bytes(osm_pbf_bytes: &[u8]) -> Result<Vec<u8>, CompilerError> {
91    let element_reader = ElementReader::new(Cursor::new(osm_pbf_bytes.to_vec()));
92    compile_from_element_reader(element_reader)
93}
94
95/// Reads a PBF file from disk and compiles it.
96///
97/// # Errors
98/// See [`CompilerError`].
99pub fn compile_osm_extract_file(input_path: &std::path::Path) -> Result<Vec<u8>, CompilerError> {
100    let osm_pbf_bytes = std::fs::read(input_path).map_err(CompilerError::InputReadFailure)?;
101    compile_osm_extract_bytes(&osm_pbf_bytes)
102}

112/// One drivable way after profile evaluation.

116    node_osm_ids: Vec<i64>,

122/// One well-formed `type=restriction` relation that maps to a known
123/// [`RestrictionKind`].
124#[derive(Debug)]
125struct TurnRestrictionCandidate {
126    relation_osm_id: i64,
127    from_way_osm_id: i64,
128    via_node_osm_id: i64,
129    to_way_osm_id: i64,
130    restriction_kind: RestrictionKind,
131}

133fn compile_from_element_reader(
134    element_reader: ElementReader<Cursor<Vec<u8>>>,
135) -> Result<Vec<u8>, CompilerError> {
136    let mut collected_elements = CollectedElements::default();
137    element_reader
138        .for_each(|element| collect_element(element, &mut collected_elements))
139        .map_err(|pbf_error| CompilerError::OsmParsingFailure(pbf_error.to_string()))?;
140    build_archive(collected_elements)
141}

143/// Applies the [`crate::profile`] rules to one OSM element and collects it.
144fn collect_element(element: Element<'_>, collected_elements: &mut CollectedElements) {
145    match element {
146        Element::Node(node) => {
147            collected_elements
148                .node_positions
149                .insert(node.id(), (node.lat(), node.lon()));
150        }
151        Element::DenseNode(node) => {
152            collected_elements
153                .node_positions
154                .insert(node.id(), (node.lat(), node.lon()));
155        }
156        Element::Way(way) => {
157            if let Some(collected_way) = collect_way(way.id(), way.refs().collect(), way.tags()) {
158                collected_elements.drivable_ways.push(collected_way);
159            }
160        }
161        Element::Relation(relation) => {
162            if let Some(candidate) =
163                collect_restriction_candidate(relation.id(), relation.tags(), relation.members())
164            {
165                collected_elements
166                    .turn_restriction_candidates
167                    .push(candidate);
168            }
169        }
170    }
171}

173/// Way tag snapshot the profile rules evaluate against.
174struct WayTagSummary {
175    highway: Option<String>,
176    oneway: Option<String>,
177    access: Option<String>,
178    maxspeed: Option<String>,
179}
180
181fn collect_way_summary<'a>(tags: impl Iterator<Item = (&'a str, &'a str)>) -> WayTagSummary {
182    let mut summary = WayTagSummary {
183        highway: None,
184        oneway: None,
185        access: None,
186        maxspeed: None,
187    };
188    for (tag_key, tag_value) in tags {
189        match tag_key {
190            "highway" if summary.highway.is_none() => summary.highway = Some(tag_value.to_string()),
191            "oneway" if summary.oneway.is_none() => summary.oneway = Some(tag_value.to_string()),
192            "access" if summary.access.is_none() => summary.access = Some(tag_value.to_string()),
193            "maxspeed" if summary.maxspeed.is_none() => {
194                summary.maxspeed = Some(tag_value.to_string())
195            }
196            _ => {}
197        }
198    }
199    summary
200}
201
202fn collect_way<'a>(
203    way_osm_id: i64,
204    node_osm_ids: Vec<i64>,
205    tags: impl Iterator<Item = (&'a str, &'a str)>,
206) -> Option<CollectedWay> {
207    let way_tag_summary = collect_way_summary(tags);
208    if is_blocked_by_access(way_tag_summary.access.as_deref()) {
209        return None;
210    }
211    let (road_class, default_speed_kilometers_per_hour) =
212        road_class_and_default_speed_for_highway(way_tag_summary.highway.as_deref()?)?;
213    let speed_kilometers_per_hour =
214        posted_speed_override_kilometers_per_hour(way_tag_summary.maxspeed.as_deref())
215            .unwrap_or(default_speed_kilometers_per_hour);
216    Some(CollectedWay {
217        way_osm_id,
218        node_osm_ids,
219        road_class,
220        speed_kilometers_per_hour,
221        travel_direction: travel_direction_for_oneway_tag(way_tag_summary.oneway.as_deref()),
222    })
223}

225/// Relation member snapshot: role name and referenced element id.
226struct RelationMember {
227    role: String,
228    member_type: osmpbf::RelMemberType,
229    member_osm_id: i64,
230}
231
232fn collect_restriction_candidate<'a>(
233    relation_osm_id: i64,
234    tags: impl Iterator<Item = (&'a str, &'a str)>,
235    members: impl Iterator<Item = osmpbf::RelMember<'a>>,
236) -> Option<TurnRestrictionCandidate> {
237    let mut relation_is_restriction = false;
238    let mut restriction_value: Option<&str> = None;
239    for (tag_key, tag_value) in tags {
240        if tag_key == "type" && tag_value == "restriction" {
241            relation_is_restriction = true;
242        }
243        if tag_key == "restriction" && restriction_value.is_none() {
244            restriction_value = Some(tag_value);
245        }
246    }
247    if !relation_is_restriction {
248        return None;
249    }
250    let restriction_kind = restriction_kind_for_openstreetmap_value(restriction_value?)?;
251    let collected_members: Vec<RelationMember> = members
252        .filter_map(|member| {
253            let role = member.role().ok()?.to_string();
254            Some(RelationMember {
255                role,
256                member_type: member.member_type,
257                member_osm_id: member.member_id,
258            })
259        })
260        .collect();
261    if collected_members.len() != 3 {
262        return None;
263    }
264    let from_way_osm_id =
265        single_member_osm_id(&collected_members, "from", osmpbf::RelMemberType::Way)?;
266    let via_node_osm_id =
267        single_member_osm_id(&collected_members, "via", osmpbf::RelMemberType::Node)?;
268    let to_way_osm_id = single_member_osm_id(&collected_members, "to", osmpbf::RelMemberType::Way)?;
269    Some(TurnRestrictionCandidate {
270        relation_osm_id,
271        from_way_osm_id,
272        via_node_osm_id,
273        to_way_osm_id,
274        restriction_kind,
275    })
276}

278/// The single member id for one required role and element type, or `None`
279/// for malformed relations (missing role, duplicated role, wrong type).
280fn single_member_osm_id(
281    collected_members: &[RelationMember],
282    required_role: &str,
283    required_member_type: osmpbf::RelMemberType,
284) -> Option<i64> {
285    matching_members(collected_members, required_role, required_member_type)
286        .next()
287        .map(|member| member.member_osm_id)
288}
289
290fn matching_members<'a>(
291    collected_members: &'a [RelationMember],
292    required_role: &str,
293    required_member_type: osmpbf::RelMemberType,
294) -> impl Iterator<Item = &'a RelationMember> {
295    collected_members.iter().filter(move |member| {
296        member.role == required_role && member.member_type == required_member_type
297    })
298}

304/// One edge before final ordering, with its originating way and OSM node
305/// ids for tie-breaks and restriction resolution.
306struct EdgeCandidate {
307    edge_record: EdgeRecord,
308    way_osm_id: i64,
309    source_node_osm_id: i64,
310    target_node_osm_id: i64,
311}
312
313/// Per-way record of emitted edges for restriction resolution.
314struct WayEdgeReference {
315    way_osm_id: i64,
316    edge_index: u32,
317    source_node_osm_id: i64,
318    target_node_osm_id: i64,
319}
320
321fn build_archive(collected_elements: CollectedElements) -> Result<Vec<u8>, CompilerError> {
322    let CollectedElements {
323        node_positions,
324        mut drivable_ways,
325        mut turn_restriction_candidates,
326    } = collected_elements;
327
328    drivable_ways.sort_by_key(|way| way.way_osm_id);
329    turn_restriction_candidates.sort_by_key(|candidate| candidate.relation_osm_id);
330
331    let referenced_node_osm_ids: HashSet<i64> = drivable_ways
332        .iter()
333        .flat_map(|way| way.node_osm_ids.iter().copied())
334        .collect();
335    let node_index_by_osm_id: HashMap<i64, u32> = node_positions
336        .iter()
337        .filter(|(node_osm_id, _)| referenced_node_osm_ids.contains(node_osm_id))
338        .enumerate()
339        .map(|(node_index, (node_osm_id, _))| (*node_osm_id, node_index as u32))
340        .collect();
341    let node_records: Vec<NodeRecord> = node_positions
342        .iter()
343        .filter(|(node_osm_id, _)| node_index_by_osm_id.contains_key(*node_osm_id))
344        .map(|(_, &(latitude_degrees, longitude_degrees))| {
345            NodeRecord::from_geographic_degrees(latitude_degrees, longitude_degrees)
346        })
347        .collect();
348
349    let mut edge_candidates: Vec<EdgeCandidate> = Vec::new();
350    let mut way_edge_references: Vec<WayEdgeReference> = Vec::new();
351    for way in &drivable_ways {
352        let resolved_node_indices: Option<Vec<(i64, u32)>> = way
353            .node_osm_ids
354            .iter()
355            .map(|node_osm_id| {
356                node_index_by_osm_id
357                    .get(node_osm_id)
358                    .map(|&index| (*node_osm_id, index))
359            })
360            .collect();
361        let Some(resolved_node_indices) = resolved_node_indices else {
362            // A drivable way referencing a node missing from the extract:
363            // skip the way rather than emit dangling edges.
364            continue;
365        };
366        for segment in resolved_node_indices.windows(2) {
367            let (from_node_osm_id, from_node_index) = segment[0];
368            let (to_node_osm_id, to_node_index) = segment[1];
369            if from_node_osm_id == to_node_osm_id {
370                // Zero-length self loop: not a routing edge in this slice.
371                continue;
372            }
373            let from_node_record = &node_records[from_node_index as usize];
374            let to_node_record = &node_records[to_node_index as usize];
375            let distance_meters = haversine_distance_meters(from_node_record, to_node_record);
376            for (emits_forward_direction, (source_node_index, target_node_index)) in
377                emit_directions(way.travel_direction, from_node_index, to_node_index)
378            {
379                let edge_record = EdgeRecord::new_directed(
380                    source_node_index,
381                    target_node_index,
382                    distance_meters,
383                    duration_milliseconds_for(distance_meters, way.speed_kilometers_per_hour),
384                    edge_properties_for_direction(
385                        way.road_class,
386                        way.speed_kilometers_per_hour,
387                        emits_forward_direction,
388                    ),
389                );
390                edge_candidates.push(EdgeCandidate {
391                    edge_record,
392                    way_osm_id: way.way_osm_id,
393                    source_node_osm_id: from_node_osm_id,
394                    target_node_osm_id: to_node_osm_id,
395                });
396            }
397        }
398    }
399
400    // Canonical edge order (see module docs).
401    edge_candidates.sort_by_key(|edge_candidate| {
402        (
403            edge_candidate.edge_record.source_node_index(),
404            edge_candidate.edge_record.target_node_index(),
405            edge_candidate.way_osm_id,
406        )
407    });
408    let edge_records: Vec<EdgeRecord> = edge_candidates
409        .iter()
410        .map(|edge_candidate| edge_candidate.edge_record)
411        .collect();
412    for (edge_index, edge_candidate) in edge_candidates.iter().enumerate() {
413        way_edge_references.push(WayEdgeReference {
414            way_osm_id: edge_candidate.way_osm_id,
415            edge_index: edge_index as u32,
416            source_node_osm_id: edge_candidate.source_node_osm_id,
417            target_node_osm_id: edge_candidate.target_node_osm_id,
418        });
419    }
420
421    // Resolve turn restrictions onto directed edges; unresolved candidates
422    // (referencing undrivable or missing ways) are skipped, never fail.
423    let mut resolved_restrictions: Vec<(TurnRestrictionRecord, &'static str)> = Vec::new();
424    for candidate in &turn_restriction_candidates {
425        let Some(via_node_index) = node_index_by_osm_id
426            .get(&candidate.via_node_osm_id)
427            .copied()
428        else {
429            continue;
430        };
431        let Some(from_edge_index) = first_way_edge_where(
432            &way_edge_references,
433            candidate.from_way_osm_id,
434            |reference| reference.target_node_osm_id == candidate.via_node_osm_id,
435        ) else {
436            continue;
437        };
438        let Some(to_edge_index) =
439            first_way_edge_where(&way_edge_references, candidate.to_way_osm_id, |reference| {
440                reference.source_node_osm_id == candidate.via_node_osm_id
441            })
442        else {
443            continue;
444        };
445        resolved_restrictions.push((
446            TurnRestrictionRecord::new(
447                from_edge_index,
448                via_node_index,
449                to_edge_index,
450                candidate.restriction_kind,
451            ),
452            instruction_phrase_for_restriction_kind(candidate.restriction_kind),
453        ));
454    }
455    resolved_restrictions.sort_by_key(|(restriction_record, _)| {
456        (
457            restriction_record.from_edge_index(),
458            restriction_record.via_node_index(),
459            restriction_record.to_edge_index(),
460        )
461    });
462    let turn_restrictions: Vec<TurnRestrictionRecord> = resolved_restrictions
463        .iter()
464        .map(|(restriction_record, _)| *restriction_record)
465        .collect();
466    let mut instruction_strings = Vec::<u8>::new();
467    let mut instruction_lookup_indices = Vec::<u32>::new();
468    for (_, instruction_phrase) in &resolved_restrictions {
469        instruction_lookup_indices.push(instruction_strings.len() as u32);
470        instruction_strings.extend_from_slice(instruction_phrase.as_bytes());
471    }
472
473    check_index_capacity(node_records.len(), edge_records.len())?;
474
475    // Lane guidance stays empty by refusal, not by omission: OSM `turn:lanes`
476    // carries per-lane *permitted maneuvers* ("through|through|right"), while
477    // `LaneGuidanceRecord` can hold only an edge lane count — emitting counts
478    // would either silently drop the turn information the tags express or
479    // fabricate a recommended lane. See `LaneGuidanceRecord`' documentation;
480    // the resource keeps its slot in the canonical resource order so the
481    // archive layout stays byte-stable for the future lane-geometry slice.
482    let archive_builder = OfflineRoutingNetworkGraphBuilder::new()
483        .with_nodes(node_records)
484        .with_edges(edge_records)
485        .with_lane_guidances(Vec::<LaneGuidanceRecord>::new())
486        .with_instruction_strings(instruction_strings)
487        .with_instruction_lookup_indices(instruction_lookup_indices)
488        .with_turn_restrictions(turn_restrictions);
489    archive_builder
490        .encode_deterministic()
491        .map_err(|archive_error| CompilerError::OsmParsingFailure(archive_error.to_string()))
492}

494/// Emits the directed edges for one way segment in the way's allowed
495/// directions: `(covers forward direction?, (source index, target index))`.
496fn emit_directions(
497    travel_direction: TravelDirection,
498    from_node_index: u32,
499    to_node_index: u32,
500) -> Vec<(bool, (u32, u32))> {
501    match travel_direction {
502        TravelDirection::ForwardOnly => vec![(true, (from_node_index, to_node_index))],
503        TravelDirection::BackwardOnly => vec![(false, (to_node_index, from_node_index))],
504        TravelDirection::Both => vec![
505            (true, (from_node_index, to_node_index)),
506            (false, (to_node_index, from_node_index)),
507        ],
508    }
509}

511fn first_way_edge_where(
512    way_edge_references: &[WayEdgeReference],
513    way_osm_id: i64,
514    matches: impl Fn(&WayEdgeReference) -> bool,
515) -> Option<u32> {
516    way_edge_references
517        .iter()
518        .find(|reference| reference.way_osm_id == way_osm_id && matches(reference))
519        .map(|reference| reference.edge_index)
520}
521
522fn check_index_capacity(node_count: usize, edge_count: usize) -> Result<(), CompilerError> {
523    if node_count > u32::MAX as usize || edge_count > u32::MAX as usize {
524        return Err(CompilerError::GraphCapacityExceeded(format!(
525            "{node_count} nodes / {edge_count} edges exceed the u32 index space"
526        )));
527    }
528    Ok(())
529}

531/// Haversine distance between two quantized node records, rounded to whole
532/// meters.
533fn haversine_distance_meters(from: &NodeRecord, to: &NodeRecord) -> u32 {
534    let from_latitude = from.latitude_degrees().to_radians();
535    let to_latitude = to.latitude_degrees().to_radians();
536    let delta_latitude = (to.latitude_degrees() - from.latitude_degrees()).to_radians();
537    let delta_longitude = (to.longitude_degrees() - from.longitude_degrees()).to_radians();
538    let haversine_argument = (delta_latitude / 2.0).sin().powi(2)
539        + from_latitude.cos() * to_latitude.cos() * (delta_longitude / 2.0).sin().powi(2);
540    let distance_meters = 2.0 * EARTH_RADIUS_METERS * haversine_argument.sqrt().asin();
541    distance_meters.round() as u32
542}

544/// Traversal duration from distance and stored speed, rounded to whole
545/// milliseconds. The stored (quantized) speed keeps the archive
546/// self-consistent: distances and durations re-derive from it exactly.
547fn duration_milliseconds_for(distance_meters: u32, speed_kilometers_per_hour: u16) -> u32 {
548    let speed_meters_per_second = f64::from(quantized_speed(speed_kilometers_per_hour))
549        * METERS_PER_KILOMETER
550        / SECONDS_PER_HOUR;
551    let duration_seconds = f64::from(distance_meters) / speed_meters_per_second;
552    (duration_seconds * MILLISECONDS_PER_SECOND).round() as u32
553}

555/// The speed value the archive stores for the given profile speed.
556fn quantized_speed(speed_kilometers_per_hour: u16) -> u16 {
557    speed_cells_from_kilometers_per_hour(speed_kilometers_per_hour)
558        * crate::archive::SPEED_QUANTUM_KILOMETERS_PER_HOUR
559}
