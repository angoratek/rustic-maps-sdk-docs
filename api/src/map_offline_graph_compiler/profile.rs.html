1//! Minimal OpenStreetMap routing profile:
2//! way filtering (drivable highway classes), per-class default speeds,
3//! optional `maxspeed` overrides, oneway direction and access blocking.
4//!
5//! Deliberately minimal per `PRD.md` §4.6 ("richness grows incrementally").
6//! The data-driven profile-config upgrade (penalty groups; the OsmAnd
7//! `routing.xml` pattern from `docs/research/modern-sdk-inspiration.md` §B.3)
8//! is a tracked follow-up and only extends this table.
9
10use crate::archive::{
11    EdgeProperties, RestrictionKind, RoadClass, SPEED_CELL_LIMIT, SPEED_QUANTUM_KILOMETERS_PER_HOUR,
12};
13
14/// Highest speed the packed `encoded_value` column can store exactly.
15const MAX_STORABLE_SPEED_KILOMETERS_PER_HOUR: u16 =
16    SPEED_CELL_LIMIT * SPEED_QUANTUM_KILOMETERS_PER_HOUR;
17
18/// Default free-flow speed per drivable class, km/h. All values are
19/// multiples of [`crate::archive::SPEED_QUANTUM_KILOMETERS_PER_HOUR`] so the
20/// packed speed column stores them exactly.
21const MOTORWAY_DEFAULT_SPEED_KILOMETERS_PER_HOUR: u16 = 120;
22const TRUNK_DEFAULT_SPEED_KILOMETERS_PER_HOUR: u16 = 100;
23const PRIMARY_DEFAULT_SPEED_KILOMETERS_PER_HOUR: u16 = 80;
24const SECONDARY_DEFAULT_SPEED_KILOMETERS_PER_HOUR: u16 = 60;
25const TERTIARY_DEFAULT_SPEED_KILOMETERS_PER_HOUR: u16 = 45;
26const UNCLASSIFIED_DEFAULT_SPEED_KILOMETERS_PER_HOUR: u16 = 40;
27const RESIDENTIAL_DEFAULT_SPEED_KILOMETERS_PER_HOUR: u16 = 30;
28const SERVICE_DEFAULT_SPEED_KILOMETERS_PER_HOUR: u16 = 15;
29
30/// `maxspeed` values below this floor are treated as unparseable (e.g. the
31/// "0" placeholder value) and fall back to the class default.
32const POSTED_SPEED_MINIMUM_KILOMETERS_PER_HOUR: u16 = 5;
33
34/// Which directions exist along a way's node order.
35#[derive(Debug, Clone, Copy, PartialEq, Eq)]
36pub(crate) enum TravelDirection {
37    /// Only along the node order (e.g. `oneway=yes`).
38    ForwardOnly,
39    /// Only against the node order (e.g. `oneway=-1`).
40    BackwardOnly,
41    /// Both directions.
42    Both,
43}
44
45/// Maps an OSM `highway` value to its road class and default speed, or
46/// `None` when the way is not drivable under this profile.
47///
48/// `living_street` borrows the residential class at residential speed, and
49/// all `*_link` ways inherit their parent class at parent-class speed —
50/// two documented simplifications until link-road penalties land.
51pub(crate) fn road_class_and_default_speed_for_highway(
52    highway_value: &str,
53) -> Option<(RoadClass, u16)> {
54    match highway_value {
55        "motorway" | "motorway_link" => Some((
56            RoadClass::Motorway,
57            MOTORWAY_DEFAULT_SPEED_KILOMETERS_PER_HOUR,
58        )),
59        "trunk" | "trunk_link" => Some((RoadClass::Trunk, TRUNK_DEFAULT_SPEED_KILOMETERS_PER_HOUR)),
60        "primary" | "primary_link" => Some((
61            RoadClass::Primary,
62            PRIMARY_DEFAULT_SPEED_KILOMETERS_PER_HOUR,
63        )),
64        "secondary" | "secondary_link" => Some((
65            RoadClass::Secondary,
66            SECONDARY_DEFAULT_SPEED_KILOMETERS_PER_HOUR,
67        )),
68        "tertiary" | "tertiary_link" => Some((
69            RoadClass::Tertiary,
70            TERTIARY_DEFAULT_SPEED_KILOMETERS_PER_HOUR,
71        )),
72        "unclassified" => Some((
73            RoadClass::Unclassified,
74            UNCLASSIFIED_DEFAULT_SPEED_KILOMETERS_PER_HOUR,
75        )),
76        "residential" | "living_street" => Some((
77            RoadClass::Residential,
78            RESIDENTIAL_DEFAULT_SPEED_KILOMETERS_PER_HOUR,
79        )),
80        "service" => Some((
81            RoadClass::Service,
82            SERVICE_DEFAULT_SPEED_KILOMETERS_PER_HOUR,
83        )),
84        _ => None,
85    }
86}

88/// Whether an `access` tag blocks routing on the way entirely
89/// (private/`no`); other access restrictions stay out of this slice.
90pub(crate) fn is_blocked_by_access(access_value: Option<&str>) -> bool {
91    matches!(access_value, Some("no") | Some("private"))
92}
93
94/// Maps an `oneway` tag value to the travel directions that exist, defaulting
95/// to both directions when the tag is absent or empty. Roundabout and other
96/// implied-oneway handling is out of this slice.
97pub(crate) fn travel_direction_for_oneway_tag(oneway_value: Option<&str>) -> TravelDirection {
98    match oneway_value {
99        Some("yes") | Some("1") | Some("true") => TravelDirection::ForwardOnly,
100        Some("-1") | Some("reverse") => TravelDirection::BackwardOnly,
101        _ => TravelDirection::Both,
102    }
103}
104
105/// Parses a plain-integer km/h `maxspeed` override; anything syntactically
106/// richer (locale values, zones, "none") stays out of this slice and falls
107/// back to the class default.
108pub(crate) fn posted_speed_override_kilometers_per_hour(
109    maxspeed_value: Option<&str>,
110) -> Option<u16> {
111    let parsed = maxspeed_value?.parse::<u16>().ok()?;
112    if !(POSTED_SPEED_MINIMUM_KILOMETERS_PER_HOUR..=MAX_STORABLE_SPEED_KILOMETERS_PER_HOUR)
113        .contains(&parsed)
114    {
115        return None;
116    }
117    Some(parsed)
118}

120/// Maps an OSM `restriction` relation value to its kind; unknown values
121/// (turn:restriction expressions, conditional restrictions) return `None`
122/// and are skipped by the compiler.
123pub(crate) fn restriction_kind_for_openstreetmap_value(
124    restriction_value: &str,
125) -> Option<RestrictionKind> {
126    match restriction_value {
127        "no_left_turn" => Some(RestrictionKind::NoLeftTurn),
128        "no_right_turn" => Some(RestrictionKind::NoRightTurn),
129        "no_u_turn" => Some(RestrictionKind::NoUTurn),
130        "no_straight_on" => Some(RestrictionKind::NoStraightOn),
131        "only_straight_on" => Some(RestrictionKind::OnlyStraightOn),
132        _ => None,
133    }
134}
135
136/// Precomputed instruction text per restriction kind (the compiler emits one
137/// phrase per recorded restriction).
138pub(crate) fn instruction_phrase_for_restriction_kind(
139    restriction_kind: RestrictionKind,
140) -> &'static str {
141    match restriction_kind {
142        RestrictionKind::NoLeftTurn => "No left turn",
143        RestrictionKind::NoRightTurn => "No right turn",
144        RestrictionKind::NoUTurn => "No U-turn",
145        RestrictionKind::NoStraightOn => "No straight on",
146        RestrictionKind::OnlyStraightOn => "Only straight on",
147    }
148}

150/// Builds the edge properties for one emitted directed edge: its access
151/// bits record which of the way's two node-order directions this record
152/// covers (every emitted record is a traversable direction, so exactly one
153/// bit is set).
154pub(crate) fn edge_properties_for_direction(
155    road_class: RoadClass,
156    speed_kilometers_per_hour: u16,
157    emits_forward_direction: bool,
158) -> EdgeProperties {
159    EdgeProperties {
160        road_class,
161        speed_kilometers_per_hour,
162        forward_allowed: emits_forward_direction,
163        backward_allowed: !emits_forward_direction,
164    }
165}

167#[cfg(test)]
168mod tests {
169    use super::*;
170
171    #[test]
172    fn drivable_highway_values_map_to_named_classes() {
173        assert_eq!(
174            road_class_and_default_speed_for_highway("motorway"),
175            Some((RoadClass::Motorway, 120))
176        );
177        assert_eq!(
178            road_class_and_default_speed_for_highway("residential"),
179            Some((RoadClass::Residential, 30))
180        );
181        assert_eq!(
182            road_class_and_default_speed_for_highway("living_street"),
183            Some((RoadClass::Residential, 30))
184        );
185        assert_eq!(road_class_and_default_speed_for_highway("footway"), None);
186    }
187
188    #[test]
189    fn oneway_values_resolve_travel_direction() {
190        assert_eq!(
191            travel_direction_for_oneway_tag(Some("yes")),
192            TravelDirection::ForwardOnly
193        );
194        assert_eq!(
195            travel_direction_for_oneway_tag(Some("-1")),
196            TravelDirection::BackwardOnly
197        );
198        assert_eq!(
199            travel_direction_for_oneway_tag(Some("no")),
200            TravelDirection::Both
201        );
202        assert_eq!(travel_direction_for_oneway_tag(None), TravelDirection::Both);
203    }
204
205    #[test]
206    fn maxspeed_parsing_accepts_plain_integers_only() {
207        assert_eq!(
208            posted_speed_override_kilometers_per_hour(Some("50")),
209            Some(50)
210        );
211        assert_eq!(
212            posted_speed_override_kilometers_per_hour(Some("RU:urban")),
213            None
214        );
215        assert_eq!(
216            posted_speed_override_kilometers_per_hour(Some("none")),
217            None
218        );
219        assert_eq!(posted_speed_override_kilometers_per_hour(Some("3")), None);
220        assert_eq!(posted_speed_override_kilometers_per_hour(None), None);
221    }
222
223    #[test]
224    fn direction_access_bits_record_the_covered_direction() {
225        let forward_edge = edge_properties_for_direction(RoadClass::Residential, 30, true);
226        let backward_edge = edge_properties_for_direction(RoadClass::Residential, 30, false);
227        assert!(forward_edge.forward_allowed);
228        assert!(!forward_edge.backward_allowed);
229        assert!(!backward_edge.forward_allowed);
230        assert!(backward_edge.backward_allowed);
231    }
232}
