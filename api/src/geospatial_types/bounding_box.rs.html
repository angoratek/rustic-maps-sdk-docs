1//! Axis-aligned rectangular bounding box record.
2
3use crate::coordinate::GeographicCoordinate;
4use serde::{Deserialize, Serialize};
5
6/// Rectangular area between a south-west and a north-east corner.
7///
8/// Boxes never span the antimeridian in this model; an application that needs
9/// a crossing region encodes it as two adjacent boxes.
10#[derive(Debug, Clone, Copy, PartialEq, Serialize, Deserialize)]
11pub struct BoundingBox {
12    /// South-west corner (minimum latitude and longitude).
13    pub south_west: GeographicCoordinate,
14    /// North-east corner (maximum latitude and longitude).
15    pub north_east: GeographicCoordinate,
16}
17
18impl BoundingBox {
19    /// Builds a bounding box from its two opposite corners.
20    ///
21    /// # Parameters
22    /// - `south_west` — the corner with the minimum latitude and longitude.
23    /// - `north_east` — the corner with the maximum latitude and longitude.
24    ///
25    /// # Returns
26    /// A box whose edges are included in the area.
27    ///
28    /// # Errors
29    /// Returns [`crate::SDKError::InvalidCoordinateInput`] when either corner
30    /// is not a valid coordinate, or when the south-west corner lies north or
31    /// east of the north-east corner.
32    pub fn from_corners(
33        south_west: GeographicCoordinate,
34        north_east: GeographicCoordinate,
35    ) -> Result<Self, crate::SDKError> {
36        let _ = GeographicCoordinate::from_degrees(south_west.latitude, south_west.longitude)?;
37        let _ = GeographicCoordinate::from_degrees(north_east.latitude, north_east.longitude)?;
38        if south_west.latitude > north_east.latitude || south_west.longitude > north_east.longitude
39        {
40            return Err(crate::SDKError::InvalidCoordinateInput(
41                "bounding box corners are inverted".to_string(),
42            ));
43        }
44        Ok(Self {
45            south_west,
46            north_east,
47        })
48    }
49
50    /// Reports whether the coordinate lies within the closed box.
51    pub fn contains_coordinate(&self, coordinate: &GeographicCoordinate) -> bool {
52        coordinate.latitude >= self.south_west.latitude
53            && coordinate.latitude <= self.north_east.latitude
54            && coordinate.longitude >= self.south_west.longitude
55            && coordinate.longitude <= self.north_east.longitude
56    }
57}

59#[cfg(test)]
60mod tests {
61    use super::BoundingBox;
62    use crate::coordinate::GeographicCoordinate;
63
64    fn coordinate(latitude: f64, longitude: f64) -> GeographicCoordinate {
65        GeographicCoordinate::from_degrees(latitude, longitude).unwrap()
66    }
67
68    #[test]
69    fn contains_points_inside_and_on_edges() {
70        let bounding_box =
71            BoundingBox::from_corners(coordinate(45.0, 5.0), coordinate(49.0, 11.0)).unwrap();
72        assert!(bounding_box.contains_coordinate(&coordinate(47.0, 8.0)));
73        assert!(bounding_box.contains_coordinate(&coordinate(45.0, 5.0)));
74        assert!(bounding_box.contains_coordinate(&coordinate(49.0, 11.0)));
75        assert!(!bounding_box.contains_coordinate(&coordinate(44.9, 8.0)));
76        assert!(!bounding_box.contains_coordinate(&coordinate(47.0, 11.1)));
77    }
78
79    #[test]
80    fn rejects_inverted_corners() {
81        assert!(BoundingBox::from_corners(coordinate(49.0, 11.0), coordinate(45.0, 5.0)).is_err());
82    }
83}
