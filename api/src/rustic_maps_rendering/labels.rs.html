1//! The frame-level label pass: symbol-layer text from decoded vector tiles
2//! becomes screen-space glyph quads, anchored at a point for place and POI
3//! names and run along a line for road names.
4//!
5//! The pass is deliberately GPU-free, like the overlay assembly: the native
6//! headless renderer, the native surface and the browser canvas all call the
7//! same functions and upload the same vertices, so the backends cannot drift
8//! in where a label lands or which labels survive.
9//!
10//! Three stages:
11//!
12//! 1. [`label_candidates_for_vector_tile`] walks a decoded tile's symbol-layer
13//!    features and reads each label's text through the compiled style's
14//!    text-field expression. Candidates keep the style's paint, size, priority,
15//!    rank, placement and overlap rule. A symbol layer's filter also gates a
16//!    feature before it becomes a candidate: the converted styles encode the
17//!    zoom-stepped rank ceiling and the per-class minimum zooms as `step`
18//!    expressions on `["zoom"]` inside an `all` filter. A
19//!    [`SymbolPlacement::Point`] layer anchors a candidate at each of the
20//!    feature's points; a [`SymbolPlacement::Line`] layer carries the feature's
21//!    whole line geometry so the frame can place the text along it.
22//! 2. [`LabelLayer::place`] projects the candidates through the frame camera,
23//!    sorts them by priority, then style order, then rank (the documented
24//!    collision-priority rule: lower priority is more important), and greedily
25//!    rejects any label whose padded screen
26//!    rectangle intersects a label already placed. `allow_overlap` labels skip
27//!    the test and never block later labels. The frame places at most
28//!    [`MAXIMUM_LABELS_PER_FRAME`] labels: once that budget is spent, every
29//!    remaining candidate is dropped even if it would fit, so a dense downtown
30//!    cannot out-shout the places that matter. A label whose text repeats an
31//!    already-placed label is dropped, so the buffered overlap between
32//!    adjacent tiles cannot paint one place twice. The output order is a pure
33//!    function of the input order and the camera, unless the caller hands the
34//!    previous frame's placed labels to
35//!    [`LabelLayer::place_with_history`]: those names keep their place in the
36//!    order, so a panning frame holds a name on screen instead of letting a
37//!    newcomer take its space on one frame and lose it on the next.
38//!
39//!    A line-placed label projects its geometry and takes its longest straight
40//!    run (consecutive near-collinear segments) as the text's run: the text is
41//!    centered halfway along that run, rotated to its chord bearing, flipped
42//!    so it always reads upright, and skipped when the run is shorter than the
43//!    text plus the collision padding at each end. The collision rectangle is
44//!    the axis-aligned box around the rotated text, so it rejects exactly like
45//!    a point label's.
46//! 3. [`LabelLayer::quad_vertices`] lays the glyph bitmaps out on the atlas and
47//!    emits clip-space textured quads, halo first and text second. A rotated
48//!    label rotates each glyph's screen-space rectangle about the label's
49//!    anchor before converting it to clip space, so every backend that draws
50//!    these vertices carries rotation without a second vertex format.
51//!
52//! Latin text is measured from the decoded glyph ranges and nothing here
53//! shapes complex scripts (see `rustic-maps-glyphs`). A frame with no symbol layers or
54//! no glyphs produces no vertices, so the tiles-only pipelines stay
55//! byte-identical.

76/// The em size the glyph protocol rasterizes signed-distance-field glyphs at:

81/// The line ascent the glyph protocol subtracts from each glyph's ink top

98/// The signed-distance value at which a glyph bitmap counts as covered.
99///
100/// The glyph protocol is node-fontnik's encoding,
101/// `byte = 255 - (64 + 32 × distance_px)`, so the zero-distance contour — the
102/// glyph outline — sits at `(256 − 64)/256 = 0.75`, not at 0.5. Thresholding at
103/// 0.5 dilates every glyph by `(0.75 − 0.5) × 8 = 2` atlas pixels per side,
104/// which closes counters and fattens strokes. Thresholding at 0.75 keeps the
105/// outline faithful to the font.
106pub const GLYPH_TEXT_SIGNED_DISTANCE_THRESHOLD: f32 = 0.75;
107
108/// The glyph protocol's spread: one atlas pixel of distance is `1/8` of a
109/// normalized distance-field unit. It converts atlas pixels into threshold
110/// offsets.
111pub const GLYPH_SIGNED_DISTANCE_FIELD_PIXELS_PER_UNIT: f32 = 8.0;
112
113/// The glyph protocol's border, in atlas pixels: the distance field fades to
114/// zero beyond this many pixels outside the ink box, so a halo can only reach
115/// this far before it samples the border texel.
116pub const GLYPH_SIGNED_DISTANCE_FIELD_MAXIMUM_OUTWARD_PIXELS: f64 =
117    GLYPH_SIGNED_DISTANCE_FIELD_BORDER_PIXELS as f64;
118
119/// The half-width of the glyph shader's anti-aliasing transition, in device
120/// pixels: the fragment stage's `smoothstep` half-band is
121/// `0.105 / devicePixelRatio / fontScale` in normalized field units, which is
122/// 0.84 device pixels at every size and pixel ratio.
123pub const GLYPH_EDGE_GAMMA_DEVICE_PIXELS: f32 = 0.105;
124
125/// The signed-distance threshold that draws a glyph's outline shifted outward
126/// by `halo_width_pixels`, clamped to the field's border.
127///
128/// `halo_width / fontScale` is the halo in atlas pixels, and one atlas pixel
129/// is `1 / 8` field unit, so the threshold moves down by that amount. The halo
130/// quad itself is not inflated: the fragment stage's lower threshold is what
131/// spreads the edge.
132pub fn glyph_halo_signed_distance_threshold(text_size_pixels: f64, halo_width_pixels: f64) -> f32 {
133    if halo_width_pixels <= 0.0 || text_size_pixels <= 0.0 {
134        return GLYPH_TEXT_SIGNED_DISTANCE_THRESHOLD;
135    }
136    let font_scale = text_size_pixels / GLYPH_FONT_EM_PIXELS;
137    let halo_atlas_pixels = halo_width_pixels / font_scale;
138    let halo_units = (halo_atlas_pixels as f32)
139        .min(GLYPH_SIGNED_DISTANCE_FIELD_MAXIMUM_OUTWARD_PIXELS as f32)
140        / GLYPH_SIGNED_DISTANCE_FIELD_PIXELS_PER_UNIT;
141    GLYPH_TEXT_SIGNED_DISTANCE_THRESHOLD - halo_units
142}
143
144/// The fragment stage's smoothstep half-band for one label, in normalized
145/// distance-field units: `0.105 / devicePixelRatio / fontScale`, where
146/// `fontScale = text_size_pixels / 24` (the style size, in CSS or point
147/// pixels, against the atlas' 24-pixel em). The drawn text size in device
148/// pixels is the same number times the device pixel ratio; the two uses keep
149/// the band at 0.84 device pixels at every size and ratio.
150pub fn glyph_edge_gamma(text_size_pixels: f64, device_pixel_ratio: f64) -> f32 {
151    if text_size_pixels <= 0.0 || device_pixel_ratio <= 0.0 {
152        return 0.0;
153    }
154    let font_scale = text_size_pixels / GLYPH_FONT_EM_PIXELS;
155    (f64::from(GLYPH_EDGE_GAMMA_DEVICE_PIXELS) / device_pixel_ratio / font_scale) as f32
156}

361impl LabelFont {
362    /// Packs one decoded glyph range into the atlas the label pass measures
363    /// and draws with.
364    ///
365    /// # Parameters
366    /// - `font_stack` — the decoded glyph range to pack.
367    ///
368    /// # Errors
369    /// A page too small for the range reports the packer's typed error rather
370    /// than dropping glyphs.
371    pub fn from_glyph_font_stack(font_stack: &GlyphFontStack) -> Result<Self, GlyphAtlasError> {
372        let atlas = pack_glyph_atlas(&font_stack.glyphs, Default::default())?;
373        Ok(Self {
374            font_stack_name: Arc::from(font_stack.name.as_str()),
375            revision: atlas_revision(&atlas),
376            latin_glyph_count: latin_glyph_prefix_length(&atlas),
377            atlas: Arc::new(atlas),
378        })
379    }
380
381    /// The font stack name the payload declares.
382    pub fn font_stack_name(&self) -> &str {
383        &self.font_stack_name
384    }
385
386    /// The packed atlas, for a backend uploading its pixels.
387    pub fn atlas(&self) -> &PackedGlyphAtlas {
388        &self.atlas
389    }
390
391    /// The atlas' content revision; a backend uploads the texture when this
392    /// changes and reuses it otherwise.
393    pub fn revision(&self) -> u64 {
394        self.revision
395    }
396
397    /// The packed glyph for a codepoint, if the range serves it.
398    ///
399    /// The lookup binary-searches the atlas group that can contain the
400    /// codepoint (the Latin ranges or every other range), so measuring text
401    /// stays logarithmic in the glyph count rather than scanning the page's
402    /// whole glyph list per character.
403    pub fn glyph(&self, codepoint: u32) -> Option<&PackedGlyph> {
404        let glyphs = &self.atlas.glyphs;
405        let latin_glyph_count = self.latin_glyph_count.min(glyphs.len());
406        let searchable_glyphs = if GlyphRange::is_latin_codepoint(codepoint) {
407            &glyphs[..latin_glyph_count]
408        } else {
409            &glyphs[latin_glyph_count..]
410        };
411        let glyph_index = searchable_glyphs
412            .binary_search_by_key(&codepoint, |glyph| glyph.codepoint)
413            .ok()?;
414        searchable_glyphs.get(glyph_index)
415    }
416
417    /// Measures a label's text at `text_size_pixels`.
418    ///
419    /// Characters the font does not serve contribute no advance and paint
420    /// nothing, exactly as the quad emitter skips them, so a name with one
421    /// out-of-range character still draws its remaining text rather than
422    /// disappearing. `None` is reported only when the font serves none of the
423    /// characters at all — a label that would paint nothing is not placed.
424    pub fn measure_text(&self, text: &str, text_size_pixels: f64) -> Option<LabelTextMetrics> {
425        let width_pixels = self.measure_line_width(text, text_size_pixels, 0.0)?;
426        Some(LabelTextMetrics {
427            width_pixels,
428            height_pixels: text_size_pixels,
429        })
430    }
431
432    /// One line's advance width, letter spacing included between served
433    /// glyphs; `None` when the font serves none of the line's characters.
434    fn measure_line_width(
435        &self,
436        text: &str,
437        text_size_pixels: f64,
438        letter_spacing_pixels: f64,
439    ) -> Option<f64> {
440        let scale = text_size_pixels / GLYPH_FONT_EM_PIXELS;
441        let mut width_pixels = 0.0;
442        let mut served_glyph_count = 0_usize;
443        for character in text.chars() {
444            let Some(glyph) = self.glyph(u32::from(character)) else {
445                continue;
446            };
447            if served_glyph_count > 0 {
448                width_pixels += letter_spacing_pixels;
449            }
450            served_glyph_count += 1;
451            width_pixels += f64::from(glyph.advance) * scale;
452        }
453        if served_glyph_count == 0 {
454            return None;
455        }
456        Some(width_pixels)
457    }
458
459    /// Measures a label's text as a wrapped block.
460    ///
461    /// Words are laid out greedily: a word that would push the line past
462    /// `maximum_width_pixels` starts the next one, and a word wider than the
463    /// limit on its own keeps that line. `line_height_ems` multiplies the
464    /// text size for one line's height. The block width is the widest line,
465    /// so a centered block is centered on its longest line. `None` is
466    /// reported when the font serves no character of any line.
467    pub fn measure_wrapped_text(
468        &self,
469        text: &str,
470        text_size_pixels: f64,
471        maximum_width_pixels: f64,
472        line_height_ems: f64,
473        letter_spacing_pixels: f64,
474    ) -> Option<LabelTextBlock> {
475        let mut line_texts: Vec<String> = Vec::new();
476        let mut current_line = String::new();
477        for word in text.split_whitespace() {
478            let candidate_line = if current_line.is_empty() {
479                word.to_string()
480            } else {
481                format!("{current_line} {word}")
482            };
483            let candidate_width =
484                self.measure_line_width(&candidate_line, text_size_pixels, letter_spacing_pixels);
485            let exceeds_limit = candidate_width
486                .is_some_and(|candidate_width| candidate_width > maximum_width_pixels);
487            if !current_line.is_empty() && exceeds_limit {
488                line_texts.push(std::mem::take(&mut current_line));
489                current_line = word.to_string();
490            } else {
491                current_line = candidate_line;
492            }
493        }
494        if !current_line.is_empty() {
495            line_texts.push(current_line);
496        }
497        if line_texts.is_empty() {
498            return None;
499        }
500        let mut lines = Vec::with_capacity(line_texts.len());
501        let mut widest_line_pixels = 0.0_f64;
502        let mut served_any_line = false;
503        for line_text in line_texts {
504            let width_pixels = self
505                .measure_line_width(&line_text, text_size_pixels, letter_spacing_pixels)
506                .unwrap_or(0.0);
507            served_any_line |= width_pixels > 0.0;
508            widest_line_pixels = widest_line_pixels.max(width_pixels);
509            lines.push(LabelTextLine {
510                text: line_text,
511                width_pixels,
512            });
513        }
514        if !served_any_line {
515            return None;
516        }
517        let line_height_pixels = text_size_pixels * line_height_ems;
518        Some(LabelTextBlock {
519            height_pixels: line_height_pixels * lines.len() as f64,
520            lines,
521            width_pixels: widest_line_pixels,
522            line_height_pixels,
523        })
524    }
525}

527/// How many Latin glyphs sit at the front of a packed atlas: the packer
528/// orders the Latin ranges first, then every other range, each group ascending
529/// by codepoint, so the first non-Latin glyph's index splits the atlas into
530/// the two sorted groups a binary-search glyph lookup searches.
531fn latin_glyph_prefix_length(atlas: &PackedGlyphAtlas) -> usize {
532    atlas
533        .glyphs
534        .iter()
535        .position(|glyph| !GlyphRange::is_latin_codepoint(glyph.codepoint))
536        .unwrap_or(atlas.glyphs.len())
537}

539/// A deterministic fingerprint of a packed atlas: the page dimensions, every
540/// glyph's codepoint and rectangle, and the page pixels. Computed once when
541/// the atlas is packed, never per frame.
542fn atlas_revision(atlas: &PackedGlyphAtlas) -> u64 {
543    let mut hasher = DefaultHasher::new();
544    atlas.width_pixels.hash(&mut hasher);
545    atlas.height_pixels.hash(&mut hasher);
546    for glyph in &atlas.glyphs {
547        glyph.codepoint.hash(&mut hasher);
548        glyph.bitmap_rectangle.x_pixels.hash(&mut hasher);
549        glyph.bitmap_rectangle.y_pixels.hash(&mut hasher);
550        glyph.bitmap_rectangle.width_pixels.hash(&mut hasher);
551        glyph.bitmap_rectangle.height_pixels.hash(&mut hasher);
552    }
553    atlas.pixels.hash(&mut hasher);
554    hasher.finish()
555}

557/// The Latin glyph range the SDK's default label font decodes: a glyph-protocol
558/// `0-255` payload in Noto Sans Regular, embedded in the library. Embedding it
559/// gives every surface a glyph source without a network fetch or a
560/// platform-specific asset.
561pub const DEFAULT_LATIN_GLYPH_RANGE_PBF_BYTES: &[u8] =
562    include_bytes!("../fixtures/render/glyphs/noto_sans_regular_0-255.pbf");
563
564/// The default label font, decoded and packed on first use and then shared by
565/// every frame and every view in the process.
566static DEFAULT_LABEL_FONT: OnceLock<LabelFont> = OnceLock::new();
567
568/// The SDK's default label font: [`DEFAULT_LATIN_GLYPH_RANGE_PBF_BYTES`]
569/// decoded and packed into one atlas the first time a frame asks for it, then
570/// reused. The packed atlas rides in an [`Arc`], so handing the font to
571/// [`LabelLayer::place`] never re-copies the page, and the renderer's atlas
572/// upload happens once per distinct revision.
573///
574/// The embedded bytes are part of the library; should they ever fail to
575/// decode or pack (a corrupted build rather than a normal path), the in-code
576/// fallback is an empty font stack: every label measures no glyph and the pass
577/// paints nothing, so a map still draws instead of failing. This mirrors the
578/// vector style's in-code fallback when its embedded artifact is unreadable.
579/// A caller with another glyph source keeps using [`LabelLayer::place`] with
580/// its own [`LabelFont`].
581pub fn default_label_font() -> &'static LabelFont {
582    DEFAULT_LABEL_FONT.get_or_init(|| {
583        decode_glyph_font_stack(DEFAULT_LATIN_GLYPH_RANGE_PBF_BYTES)
584            .ok()
585            .and_then(|font_stack| LabelFont::from_glyph_font_stack(&font_stack).ok())
586            .unwrap_or_else(empty_label_font)
587    })
588}

649/// A label layer ready to draw: the placed labels and the font that measures
650/// and rasterizes them.
651#[derive(Debug, Clone)]
652pub struct LabelLayer {
653    labels: Vec<PlacedLabel>,
654    font: LabelFont,
655    /// The device pixels one point/CSS pixel covers: 1.0 for the headless
656    /// golden at one device pixel per point, the display scale on a native
657    /// surface, the canvas ratio in the browser. It sets the drawn text size
658    /// and the fragment stage's edge gamma; the halo width already arrives in
659    /// the same points the style declares.
660    device_pixel_ratio: f64,
661}
662
663impl LabelLayer {
664    /// Projects and places the frame's candidates: sort by priority (lower is
665    /// more important, the field's documented collision rule), then style
666    /// order, then feature rank; drop repeated text; reject a candidate whose
667    /// padded screen rectangle collides with an already accepted label unless
668    /// the candidate allows overlap; stop once the frame has placed
669    /// [`MAXIMUM_LABELS_PER_FRAME`] labels. A line-placed candidate takes its
670    /// longest straight projected run as the text's run and repeats along it,
671    /// each label rotated to the segment beneath it; every other candidate
672    /// centers on its anchor, unrotated. Repeated names of one candidate are
673    /// placed together, and the text is recorded once so a later candidate
674    /// carrying the same name — the same street in a neighbouring tile — is
675    /// still rejected. The result keeps the sorted order, so the same
676    /// candidates always paint in the same sequence.
677    pub fn place(
678        camera: &MapCamera,
679        viewport: &ViewportPixels,
680        field_of_view_degrees: f64,
681        candidates: &[MapLabelCandidate],
682        font: &LabelFont,
683    ) -> Self {
684        Self::place_with_device_pixel_ratio(
685            camera,
686            viewport,
687            field_of_view_degrees,
688            candidates,
689            font,
690            1.0,
691        )
692    }

694    /// [`Self::place`] for a surface whose drawing buffer is not one device
695    /// pixel per point: the drawn glyph size and the edge gamma both follow
696    /// `device_pixel_ratio`, so the text keeps its declared point size and its
697    /// one-pixel anti-aliased edge on a 2× or 3× surface.
698    pub fn place_with_device_pixel_ratio(
699        camera: &MapCamera,
700        viewport: &ViewportPixels,
701        field_of_view_degrees: f64,
702        candidates: &[MapLabelCandidate],
703        font: &LabelFont,
704        device_pixel_ratio: f64,
705    ) -> Self {
706        Self::place_with_history(
707            camera,
708            viewport,
709            field_of_view_degrees,
710            candidates,
711            font,
712            device_pixel_ratio,
713            &[],
714        )
715    }
716
717    /// [`Self::place_with_device_pixel_ratio`] carrying a persistent collision
718    /// index across frames: `previous_labels` are the labels the previous frame
719    /// of the same view placed, and every candidate whose text one of them
720    /// carries is tried before any newcomer, in the order the previous frame
721    /// held them. A remembered name therefore keeps its spot while it stays
722    /// visible, so panning half a street name off the edge cannot let a
723    /// district name take its place for one frame and lose it the next.
724    ///
725    /// The memory is the label text, the pass's own identity for a name (a
726    /// frame drops repeated text), so it survives a camera move without
727    /// carrying stale screen geometry: the remembered candidate is projected
728    /// and collides like any other at the new pose, and a name that is no
729    /// longer among the candidates is simply forgotten.
730    pub fn place_with_history(
731        camera: &MapCamera,
732        viewport: &ViewportPixels,
733        field_of_view_degrees: f64,
734        candidates: &[MapLabelCandidate],
735        font: &LabelFont,
736        device_pixel_ratio: f64,
737        previous_labels: &[PlacedLabel],
738    ) -> Self {
739        let mut ordered_candidates: Vec<&MapLabelCandidate> = Vec::with_capacity(candidates.len());
740        let mut remembered_texts: Vec<&str> = Vec::new();
741        for remembered in previous_labels {
742            if remembered_texts.contains(&remembered.text.as_str()) {
743                continue;
744            }
745            if let Some(candidate) = candidates
746                .iter()
747                .find(|candidate| candidate.text == remembered.text)
748            {
749                remembered_texts.push(candidate.text.as_str());
750                ordered_candidates.push(candidate);
751            }
752        }
753        let mut newcomer_candidates: Vec<&MapLabelCandidate> = candidates
754            .iter()
755            .filter(|candidate| !remembered_texts.contains(&candidate.text.as_str()))
756            .collect();
757        // `sort_by_key` is stable, so candidates that tie on the triple keep
758        // the caller's own order — the frame's tiles list deterministically.
759        newcomer_candidates
760            .sort_by_key(|candidate| (candidate.priority, candidate.style_order, candidate.rank));
761        ordered_candidates.extend(newcomer_candidates);
762
763        let mut occupied_rectangles: Vec<LabelScreenRectangle> = Vec::new();
764        let mut placed_labels: Vec<PlacedLabel> = Vec::new();
765        let mut placed_texts: Vec<String> = Vec::new();
766        for candidate in ordered_candidates {
767            if placed_labels.len() >= MAXIMUM_LABELS_PER_FRAME {
768                break;
769            }
770            if placed_texts.iter().any(|text| text == &candidate.text) {
771                continue;
772            }
773            // The style's size is a point/CSS size; the drawing buffer draws
774            // device pixels, so the glyphs scale by the surface ratio while the
775            // style numbers stay the source of truth for gamma and halo math.
776            let text_size_pixels = candidate.text_size_pixels * device_pixel_ratio;
777            let letter_spacing_pixels = candidate.text_letter_spacing_ems * text_size_pixels;
778            let placed_along_line = candidate.line_coordinates.len() >= 2;
779            // A line-placed name stays on one line, the specification's rule;
780            // wrapping applies to point-placed text.
781            let Some(text_block) = (if placed_along_line {
782                font.measure_line_width(&candidate.text, text_size_pixels, letter_spacing_pixels)
783                    .map(|width_pixels| LabelTextBlock {
784                        lines: vec![LabelTextLine {
785                            text: candidate.text.clone(),
786                            width_pixels,
787                        }],
788                        width_pixels,
789                        height_pixels: text_size_pixels,
790                        line_height_pixels: text_size_pixels,
791                    })
792            } else {
793                font.measure_wrapped_text(
794                    &candidate.text,
795                    text_size_pixels,
796                    candidate.text_max_width_ems * text_size_pixels,
797                    candidate.text_line_height_ems,
798                    letter_spacing_pixels,
799                )
800            }) else {
801                continue;
802            };
803            let uses_variable_anchor =
804                !placed_along_line && !candidate.text_variable_anchors.is_empty();
805            let placements = if placed_along_line {
806                let mut line_placements = label_placements_along_line(
807                    camera,
808                    viewport,
809                    field_of_view_degrees,
810                    &candidate.line_coordinates,
811                    text_block.width_pixels,
812                    candidate.repeat_spacing_pixels,
813                    candidate.line_center_only,
814                    candidate.text_max_angle_radians,
815                );
816                if candidate.text_rotation_alignment == TextRotationAlignment::Viewport {
817                    for placement in &mut line_placements {
818                        placement.rotation_radians = 0.0;
819                    }
820                }
821                line_placements
822            } else {
823                // A point label follows the map's bearing only when its style
824                // asks; `auto` is the viewport alignment for point placement.
825                let point_rotation_radians = match candidate.text_rotation_alignment {
826                    TextRotationAlignment::Map => camera.bearing_degrees.to_radians(),
827                    TextRotationAlignment::Viewport | TextRotationAlignment::Auto => 0.0,
828                };
829                let point_rotation_radians = if candidate.text_keep_upright {
830                    upright_rotation_radians(point_rotation_radians)
831                } else {
832                    point_rotation_radians
833                };
834                let anchor_candidates: Vec<TextAnchor> =
835                    if candidate.text_variable_anchors.is_empty() {
836                        vec![candidate.text_anchor]
837                    } else {
838                        candidate.text_variable_anchors.clone()
839                    };
840                match camera.project_to_screen(&candidate.anchor, viewport, field_of_view_degrees) {
841                    Some(anchor_screen_pixels) => anchor_candidates
842                        .iter()
843                        .map(|text_anchor| {
844                            let mut anchor_offset = text_anchor_center_offset(
845                                *text_anchor,
846                                text_block.width_pixels,
847                                text_block.height_pixels,
848                            );
849                            // A variable anchor pushes the block outward by
850                            // the radial offset; a fixed anchor ignores it.
851                            if uses_variable_anchor {
852                                let radial_direction = radial_offset_direction(*text_anchor);
853                                anchor_offset[0] += radial_direction[0]
854                                    * candidate.text_radial_offset_ems
855                                    * text_size_pixels;
856                                anchor_offset[1] += radial_direction[1]
857                                    * candidate.text_radial_offset_ems
858                                    * text_size_pixels;
859                            }
860                            LabelPlacement {
861                                anchor_screen_pixels: [
862                                    anchor_screen_pixels[0] + anchor_offset[0],
863                                    anchor_screen_pixels[1] + anchor_offset[1],
864                                ],
865                                rotation_radians: point_rotation_radians,
866                            }
867                        })
868                        .collect(),
869                    None => Vec::new(),
870                }
871            };
872            // A text offset is a screen-space displacement from the anchor
873            // in ems; the candidate carries it already scaled to CSS pixels.
874            let placements: Vec<LabelPlacement> = placements
875                .into_iter()
876                .map(|mut placement| {
877                    placement.anchor_screen_pixels[0] +=
878                        candidate.text_offset_pixels[0] * device_pixel_ratio;
879                    placement.anchor_screen_pixels[1] +=
880                        candidate.text_offset_pixels[1] * device_pixel_ratio;
881                    placement
882                })
883                .collect();
884            let mut placed_any = false;
885            for placement in placements {
886                if placed_labels.len() >= MAXIMUM_LABELS_PER_FRAME {
887                    break;
888                }
889                let label_rectangle = LabelScreenRectangle::around(
890                    placement.anchor_screen_pixels,
891                    text_block.width_pixels,
892                    text_block.height_pixels,
893                    placement.rotation_radians,
894                );
895                if candidate.avoid_edges {
896                    if !label_rectangle.is_inside_viewport(viewport) {
897                        continue;
898                    }
899                } else if !label_rectangle.intersects_viewport(viewport) {
900                    continue;
901                }
902                if !candidate.allow_overlap && !candidate.ignore_placement {
903                    let collides = occupied_rectangles
904                        .iter()
905                        .any(|occupied| occupied.intersects(&label_rectangle));
906                    if collides {
907                        continue;
908                    }
909                    occupied_rectangles.push(label_rectangle);
910                }
911                placed_any = true;
912                placed_labels.push(PlacedLabel {
913                    text: candidate.text.clone(),
914                    anchor_screen_pixels: placement.anchor_screen_pixels,
915                    rotation_radians: placement.rotation_radians,
916                    placed_along_line,
917                    text_size_pixels,
918                    text_rgba: candidate.text_rgba,
919                    halo_rgba: candidate.halo_rgba,
920                    halo_width_pixels: candidate.halo_width_pixels * device_pixel_ratio,
921                    lines: text_block.lines.clone(),
922                    line_height_pixels: text_block.line_height_pixels,
923                    block_width_pixels: text_block.width_pixels,
924                    letter_spacing_pixels,
925                    text_justify: candidate.text_justify,
926                    priority: candidate.priority,
927                });
928                // A variable-anchor label tries its anchors in order and
929                // takes the first that fits, once.
930                if uses_variable_anchor {
931                    break;
932                }
933            }
934            if placed_any {
935                placed_texts.push(candidate.text.clone());
936            }
937        }
938        Self {
939            labels: placed_labels,
940            font: font.clone(),
941            device_pixel_ratio,
942        }
943    }

960    /// Assembles the layer's clip-space quad vertices: one halo pass (only
961    /// labels with a positive halo width) followed by the text pass. Both
962    /// lists draw non-indexed through the textured glyph pipeline. A halo
963    /// quad is the glyph's bitmap rectangle inflated by the halo width while
964    /// keeping the glyph's texture window, which stretches the bitmap's soft
965    /// edge outward; the fragment stage's lower halo threshold turns that
966    /// stretched edge into the outline.
967    ///
968    /// A rotated label lays its glyphs out in the label's own unrotated frame
969    /// and rotates each glyph rectangle about the label anchor, so rotation
970    /// reaches every backend through the shared clip-space positions.
971    pub fn quad_vertices(&self, viewport: &ViewportPixels) -> LabelQuadVertices {
972        let mut halo_vertices = Vec::new();
973        let mut text_vertices = Vec::new();
974        for label in &self.labels {
975            if label.rotation_radians == 0.0 {
976                self.append_axis_aligned_label_quads(
977                    &mut halo_vertices,
978                    &mut text_vertices,
979                    viewport,
980                    label,
981                );
982            } else {
983                self.append_rotated_label_quads(
984                    &mut halo_vertices,
985                    &mut text_vertices,
986                    viewport,
987                    label,
988                );
989            }
990        }
991        LabelQuadVertices {
992            halo: halo_vertices,
993            text: text_vertices,
994        }
995    }

997    /// Lays out one unrotated label's glyph quads: the baseline runs along
998    /// the screen's `x` axis from the label's centered pen origin.
999    fn append_axis_aligned_label_quads(
1000        &self,
1001        halo_vertices: &mut Vec<ClipSpaceGlyphVertex>,
1002        text_vertices: &mut Vec<ClipSpaceGlyphVertex>,
1003        viewport: &ViewportPixels,
1004        label: &PlacedLabel,
1005    ) {
1006        let scale = label.text_size_pixels / GLYPH_FONT_EM_PIXELS;
1007        // The style's own (point/CSS) sizes drive the SDF threshold and gamma;
1008        // the placed sizes are already in drawing-buffer pixels.
1009        let style_text_size_pixels = label.text_size_pixels / self.device_pixel_ratio;
1010        let style_halo_width_pixels = label.halo_width_pixels / self.device_pixel_ratio;
1011        let edge_gamma = glyph_edge_gamma(style_text_size_pixels, self.device_pixel_ratio);
1012        let halo_threshold =
1013            glyph_halo_signed_distance_threshold(style_text_size_pixels, style_halo_width_pixels);
1014        let baseline_offset_pixels = label.text_size_pixels * LABEL_BASELINE_OFFSET_FRACTION;
1015        let line_count = label.lines.len() as f64;
1016        for (line_index, line) in label.lines.iter().enumerate() {
1017            // Lines stack around the anchor: line 0 is the top line.
1018            let line_offset_pixels =
1019                (line_index as f64 - (line_count - 1.0) / 2.0) * label.line_height_pixels;
1020            let baseline_y =
1021                label.anchor_screen_pixels[1] + line_offset_pixels + baseline_offset_pixels;
1022            let mut pen_x = match label.text_justify {
1023                TextJustify::Left => label.anchor_screen_pixels[0] - label.block_width_pixels / 2.0,
1024                TextJustify::Right => {
1025                    label.anchor_screen_pixels[0] + label.block_width_pixels / 2.0
1026                        - line.width_pixels
1027                }
1028                TextJustify::Auto | TextJustify::Center => {
1029                    label.anchor_screen_pixels[0] - line.width_pixels / 2.0
1030                }
1031            };
1032            for character in line.text.chars() {
1033                let Some(glyph) = self.font.glyph(u32::from(character)) else {
1034                    continue;
1035                };
1036                let advance_pixels = f64::from(glyph.advance) * scale;
1037                if glyph.bitmap_rectangle.width_pixels > 0
1038                    && glyph.bitmap_rectangle.height_pixels > 0
1039                {
1040                    let bitmap_left_pixels = pen_x
1041                        + f64::from(glyph.left - GLYPH_SIGNED_DISTANCE_FIELD_BORDER_PIXELS as i32)
1042                            * scale;
1043                    let bitmap_top_pixels = baseline_y
1044                        - (f64::from(glyph.top)
1045                            + GLYPH_FONT_ASCENDER_PIXELS
1046                            + f64::from(GLYPH_SIGNED_DISTANCE_FIELD_BORDER_PIXELS))
1047                            * scale;
1048                    let bitmap_width_pixels =
1049                        f64::from(glyph.bitmap_rectangle.width_pixels) * scale;
1050                    let bitmap_height_pixels =
1051                        f64::from(glyph.bitmap_rectangle.height_pixels) * scale;
1052                    if label.halo_width_pixels > 0.0 {
1053                        // The halo keeps the text quad's geometry: its lower
1054                        // threshold is what spreads the outline, exactly as
1055                        // MapLibre Native's `symbol_sdf` shader does.
1056                        append_glyph_quad(
1057                            halo_vertices,
1058                            viewport,
1059                            [bitmap_left_pixels, bitmap_top_pixels],
1060                            [bitmap_width_pixels, bitmap_height_pixels],
1061                            glyph,
1062                            halo_threshold,
1063                            edge_gamma,
1064                            &label.halo_rgba,
1065                        );
1066                    }
1067                    append_glyph_quad(
1068                        text_vertices,
1069                        viewport,
1070                        [bitmap_left_pixels, bitmap_top_pixels],
1071                        [bitmap_width_pixels, bitmap_height_pixels],
1072                        glyph,
1073                        GLYPH_TEXT_SIGNED_DISTANCE_THRESHOLD,
1074                        edge_gamma,
1075                        &label.text_rgba,
1076                    );
1077                }
1078                pen_x += advance_pixels + label.letter_spacing_pixels;
1079            }
1080        }
1081    }

1083    /// Lays out one rotated label's glyph quads: the baseline runs along the
1084    /// label's rotation from a centered pen origin, and each glyph rectangle
1085    /// rotates about the label anchor before it becomes clip-space corners.
1086    fn append_rotated_label_quads(
1087        &self,
1088        halo_vertices: &mut Vec<ClipSpaceGlyphVertex>,
1089        text_vertices: &mut Vec<ClipSpaceGlyphVertex>,
1090        viewport: &ViewportPixels,
1091        label: &PlacedLabel,
1092    ) {
1093        let scale = label.text_size_pixels / GLYPH_FONT_EM_PIXELS;
1094        let style_text_size_pixels = label.text_size_pixels / self.device_pixel_ratio;
1095        let style_halo_width_pixels = label.halo_width_pixels / self.device_pixel_ratio;
1096        let edge_gamma = glyph_edge_gamma(style_text_size_pixels, self.device_pixel_ratio);
1097        let halo_threshold =
1098            glyph_halo_signed_distance_threshold(style_text_size_pixels, style_halo_width_pixels);
1099        let baseline_offset_pixels = label.text_size_pixels * LABEL_BASELINE_OFFSET_FRACTION;
1100        let line_count = label.lines.len() as f64;
1101        for (line_index, line) in label.lines.iter().enumerate() {
1102            // Line offsets and pen origins are authored in the label's own
1103            // unrotated frame; `append_rotated_glyph_quad` rotates the whole
1104            // offset about the anchor.
1105            let line_offset_pixels =
1106                (line_index as f64 - (line_count - 1.0) / 2.0) * label.line_height_pixels;
1107            let mut pen_offset_pixels = match label.text_justify {
1108                TextJustify::Left => -label.block_width_pixels / 2.0,
1109                TextJustify::Right => label.block_width_pixels / 2.0 - line.width_pixels,
1110                TextJustify::Auto | TextJustify::Center => -line.width_pixels / 2.0,
1111            };
1112            for character in line.text.chars() {
1113                let Some(glyph) = self.font.glyph(u32::from(character)) else {
1114                    continue;
1115                };
1116                let advance_pixels = f64::from(glyph.advance) * scale;
1117                if glyph.bitmap_rectangle.width_pixels > 0
1118                    && glyph.bitmap_rectangle.height_pixels > 0
1119                {
1120                    let bitmap_offset_pixels = [
1121                        pen_offset_pixels
1122                            + f64::from(
1123                                glyph.left - GLYPH_SIGNED_DISTANCE_FIELD_BORDER_PIXELS as i32,
1124                            ) * scale,
1125                        line_offset_pixels + baseline_offset_pixels
1126                            - (f64::from(glyph.top)
1127                                + GLYPH_FONT_ASCENDER_PIXELS
1128                                + f64::from(GLYPH_SIGNED_DISTANCE_FIELD_BORDER_PIXELS))
1129                                * scale,
1130                    ];
1131                    let bitmap_width_pixels =
1132                        f64::from(glyph.bitmap_rectangle.width_pixels) * scale;
1133                    let bitmap_height_pixels =
1134                        f64::from(glyph.bitmap_rectangle.height_pixels) * scale;
1135                    if label.halo_width_pixels > 0.0 {
1136                        append_rotated_glyph_quad(
1137                            halo_vertices,
1138                            viewport,
1139                            label.anchor_screen_pixels,
1140                            label.rotation_radians,
1141                            bitmap_offset_pixels,
1142                            [bitmap_width_pixels, bitmap_height_pixels],
1143                            glyph,
1144                            halo_threshold,
1145                            edge_gamma,
1146                            &label.halo_rgba,
1147                        );
1148                    }
1149                    append_rotated_glyph_quad(
1150                        text_vertices,
1151                        viewport,
1152                        label.anchor_screen_pixels,
1153                        label.rotation_radians,
1154                        bitmap_offset_pixels,
1155                        [bitmap_width_pixels, bitmap_height_pixels],
1156                        glyph,
1157                        GLYPH_TEXT_SIGNED_DISTANCE_THRESHOLD,
1158                        edge_gamma,
1159                        &label.text_rgba,
1160                    );
1161                }
1162                pen_offset_pixels += advance_pixels + label.letter_spacing_pixels;
1163            }
1164        }
1165    }
1166}

1213/// Places a line-placed label's run along the longest straight run of its
1214/// projected geometry, or reports an empty list when no run can carry it.
1215///
1216/// Vertices the camera cannot project break the polyline: runs are measured
1217/// over the stretches where every vertex projected. Within one stretch, a
1218/// straight run is the longest sequence of consecutive segments whose bearing
1219/// stays within [`LINE_LABEL_STRAIGHT_RUN_ANGLE_TOLERANCE_RADIANS`] of the
1220/// run's first segment (map data splits a street into short blocks, so a name
1221/// needs the block sequence, not one block). Of the stretches' runs the
1222/// longest wins (ties keep the earlier one).
1223///
1224/// The text repeats along that run the way the reference engines name a long
1225/// street: one label centers on the run when only one fits, otherwise labels
1226/// spread evenly from [`LABEL_COLLISION_PADDING_PIXELS`] past the first end to
1227/// the same distance before the last, with at least
1228/// [`LINE_LABEL_REPEAT_SPACING_PIXELS`] of clear space between neighbours.
1229/// Each label rotates to the bearing of the segment under its own anchor
1230/// (flipped past [`LINE_LABEL_UPRIGHT_ROTATION_LIMIT_RADIANS`] so the text
1231/// always reads upright), so a name follows the road's curve rather than one
1232/// chord across the whole run.
1233/// The outward direction one text anchor points, used to push a
1234/// variable-anchored label away from its symbol by the radial offset.
1235fn radial_offset_direction(text_anchor: TextAnchor) -> [f64; 2] {
1236    let diagonal = std::f64::consts::FRAC_1_SQRT_2;
1237    match text_anchor {
1238        TextAnchor::Center => [0.0, 0.0],
1239        TextAnchor::Left => [-1.0, 0.0],
1240        TextAnchor::Right => [1.0, 0.0],
1241        TextAnchor::Top => [0.0, -1.0],
1242        TextAnchor::Bottom => [0.0, 1.0],
1243        TextAnchor::TopLeft => [-diagonal, -diagonal],
1244        TextAnchor::TopRight => [diagonal, -diagonal],
1245        TextAnchor::BottomLeft => [-diagonal, diagonal],
1246        TextAnchor::BottomRight => [diagonal, diagonal],
1247    }
1248}

1250/// How far the block's center sits from the symbol anchor for one
1251/// `text-anchor`, in screen pixels.
1252fn text_anchor_center_offset(
1253    text_anchor: TextAnchor,
1254    block_width_pixels: f64,
1255    block_height_pixels: f64,
1256) -> [f64; 2] {
1257    let horizontal_shift_pixels = match text_anchor {
1258        TextAnchor::Left | TextAnchor::TopLeft | TextAnchor::BottomLeft => block_width_pixels / 2.0,
1259        TextAnchor::Right | TextAnchor::TopRight | TextAnchor::BottomRight => {
1260            -block_width_pixels / 2.0
1261        }
1262        TextAnchor::Center | TextAnchor::Top | TextAnchor::Bottom => 0.0,
1263    };
1264    let vertical_shift_pixels = match text_anchor {
1265        TextAnchor::Top | TextAnchor::TopLeft | TextAnchor::TopRight => block_height_pixels / 2.0,
1266        TextAnchor::Bottom | TextAnchor::BottomLeft | TextAnchor::BottomRight => {
1267            -block_height_pixels / 2.0
1268        }
1269        TextAnchor::Center | TextAnchor::Left | TextAnchor::Right => 0.0,
1270    };
1271    [horizontal_shift_pixels, vertical_shift_pixels]
1272}

1274/// The nearest rotation that reads upright: a label turned more than a
1275/// quarter turn is flipped by half a turn.
1276fn upright_rotation_radians(rotation_radians: f64) -> f64 {
1277    let mut upright = rotation_radians;
1278    while upright > std::f64::consts::FRAC_PI_2 {
1279        upright -= std::f64::consts::PI;
1280    }
1281    while upright < -std::f64::consts::FRAC_PI_2 {
1282        upright += std::f64::consts::PI;
1283    }
1284    upright
1285}

1287#[allow(clippy::too_many_arguments)]
1288fn label_placements_along_line(
1289    camera: &MapCamera,
1290    viewport: &ViewportPixels,
1291    field_of_view_degrees: f64,
1292    line_coordinates: &[GeographicCoordinate],
1293    text_width_pixels: f64,
1294    repeat_spacing_pixels: f64,
1295    line_center_only: bool,
1296    angle_tolerance_radians: f64,
1297) -> Vec<LabelPlacement> {
1298    let mut projected_stretch: Vec<[f64; 2]> = Vec::new();
1299    let mut longest_run: Option<StraightRun> = None;
1300    for coordinate in line_coordinates {
1301        let Some(screen_point) =
1302            camera.project_to_screen(coordinate, viewport, field_of_view_degrees)
1303        else {
1304            consider_longest_run(
1305                &projected_stretch,
1306                &mut longest_run,
1307                angle_tolerance_radians,
1308            );
1309            projected_stretch.clear();
1310            continue;
1311        };
1312        projected_stretch.push(screen_point);
1313    }
1314    consider_longest_run(
1315        &projected_stretch,
1316        &mut longest_run,
1317        angle_tolerance_radians,
1318    );
1319    let Some(run) = longest_run else {
1320        return Vec::new();
1321    };
1322
1323    let required_run_pixels = text_width_pixels + 2.0 * LABEL_COLLISION_PADDING_PIXELS;
1324    if run.length_pixels < required_run_pixels {
1325        return Vec::new();
1326    }
1327    let first_anchor_pixels = text_width_pixels / 2.0 + LABEL_COLLISION_PADDING_PIXELS;
1328    let last_anchor_pixels = run.length_pixels - first_anchor_pixels;
1329    let repeat_stride_pixels = text_width_pixels + repeat_spacing_pixels;
1330    let label_count = if line_center_only || last_anchor_pixels <= first_anchor_pixels {
1331        1
1332    } else {
1333        let fitting_spans = ((last_anchor_pixels - first_anchor_pixels) / repeat_stride_pixels)
1334            .floor()
1335            .max(0.0);
1336        (fitting_spans as usize).saturating_add(1)
1337    };
1338    let mut placements = Vec::with_capacity(label_count);
1339    for label_index in 0..label_count {
1340        let anchor_pixels = if label_count == 1 {
1341            run.length_pixels / 2.0
1342        } else {
1343            let fraction = label_index as f64 / (label_count - 1) as f64;
1344            first_anchor_pixels + (last_anchor_pixels - first_anchor_pixels) * fraction
1345        };
1346        let Some(placement) = placement_at_arc_length(&run.points, anchor_pixels) else {
1347            continue;
1348        };
1349        placements.push(placement);
1350    }
1351    placements
1352}

1354/// One straight run of a projected polyline: the screen points it spans and
1355/// its arc length.
1356#[derive(Debug, Clone)]
1357struct StraightRun {
1358    /// The run's projected vertices, in order.
1359    points: Vec<[f64; 2]>,
1360    /// The run's total length in screen pixels.
1361    length_pixels: f64,
1362}

1364/// The placement of one label centered at `anchor_pixels` along a projected
1365/// polyline, walking its segments from the start.
1366///
1367/// The anchor sits on the segment that contains the target distance and the
1368/// rotation follows that segment's own bearing, so a repeated name turns with
1369/// the road under it. Reports `None` only for a polyline with no segment of
1370/// positive length.
1371fn placement_at_arc_length(points: &[[f64; 2]], anchor_pixels: f64) -> Option<LabelPlacement> {
1372    let mut traveled_pixels = 0.0;
1373    let mut last_segment: Option<([f64; 2], [f64; 2])> = None;
1374    for segment_points in points.windows(2) {
1375        let [segment_start, segment_end] = [segment_points[0], segment_points[1]];
1376        let segment_length_pixels =
1377            (segment_end[0] - segment_start[0]).hypot(segment_end[1] - segment_start[1]);
1378        if segment_length_pixels <= 0.0 {
1379            continue;
1380        }
1381        last_segment = Some((segment_start, segment_end));
1382        if traveled_pixels + segment_length_pixels >= anchor_pixels {
1383            let fraction_pixels =
1384                ((anchor_pixels - traveled_pixels) / segment_length_pixels).clamp(0.0, 1.0);
1385            let anchor_screen_pixels = [
1386                segment_start[0] + (segment_end[0] - segment_start[0]) * fraction_pixels,
1387                segment_start[1] + (segment_end[1] - segment_start[1]) * fraction_pixels,
1388            ];
1389            return Some(LabelPlacement {
1390                anchor_screen_pixels,
1391                rotation_radians: upright_segment_rotation(segment_start, segment_end),
1392            });
1393        }
1394        traveled_pixels += segment_length_pixels;
1395    }
1396    let (segment_start, segment_end) = last_segment?;
1397    Some(LabelPlacement {
1398        anchor_screen_pixels: segment_end,
1399        rotation_radians: upright_segment_rotation(segment_start, segment_end),
1400    })
1401}

1403/// The baseline rotation for a segment, flipped by a half turn when it would
1404/// put the text upside down, so the result always lies in
1405/// `[-π/2, π/2]`.
1406fn upright_segment_rotation(segment_start: [f64; 2], segment_end: [f64; 2]) -> f64 {
1407    let segment_rotation_radians =
1408        (segment_end[1] - segment_start[1]).atan2(segment_end[0] - segment_start[0]);
1409    if segment_rotation_radians > LINE_LABEL_UPRIGHT_ROTATION_LIMIT_RADIANS {
1410        segment_rotation_radians - std::f64::consts::PI
1411    } else if segment_rotation_radians < -LINE_LABEL_UPRIGHT_ROTATION_LIMIT_RADIANS {
1412        segment_rotation_radians + std::f64::consts::PI
1413    } else {
1414        segment_rotation_radians
1415    }
1416}

1418/// Appends the longest straight run of one projected stretch to
1419/// `longest_run` when it beats the runs already measured. A stretch with fewer
1420/// than two points has no run.
1421fn consider_longest_run(
1422    projected_stretch: &[[f64; 2]],
1423    longest_run: &mut Option<StraightRun>,
1424    angle_tolerance_radians: f64,
1425) {
1426    let mut run_start_index = 0;
1427    while run_start_index + 1 < projected_stretch.len() {
1428        let reference_bearing = segment_bearing(
1429            projected_stretch[run_start_index],
1430            projected_stretch[run_start_index + 1],
1431        );
1432        let mut run_end_index = run_start_index;
1433        let mut run_length_pixels = 0.0;
1434        while run_end_index + 1 < projected_stretch.len() {
1435            let segment_bearing = segment_bearing(
1436                projected_stretch[run_end_index],
1437                projected_stretch[run_end_index + 1],
1438            );
1439            let bearing_difference = (segment_bearing - reference_bearing + std::f64::consts::PI)
1440                .rem_euclid(2.0 * std::f64::consts::PI)
1441                - std::f64::consts::PI;
1442            if bearing_difference.abs() > angle_tolerance_radians {
1443                break;
1444            }
1445            run_length_pixels += (projected_stretch[run_end_index + 1][0]
1446                - projected_stretch[run_end_index][0])
1447                .hypot(
1448                    projected_stretch[run_end_index + 1][1] - projected_stretch[run_end_index][1],
1449                );
1450            run_end_index += 1;
1451        }
1452        if run_end_index > run_start_index {
1453            let is_longest = match longest_run {
1454                None => true,
1455                Some(existing_run) => run_length_pixels > existing_run.length_pixels,
1456            };
1457            if is_longest {
1458                *longest_run = Some(StraightRun {
1459                    points: projected_stretch[run_start_index..=run_end_index].to_vec(),
1460                    length_pixels: run_length_pixels,
1461                });
1462            }
1463        }
1464        run_start_index = run_end_index.max(run_start_index + 1);
1465    }
1466}

1470    (end[1] - start[1]).atan2(end[0] - start[0])

1473/// One label's padded screen-space collision rectangle.
1474#[derive(Debug, Clone, Copy)]
1475struct LabelScreenRectangle {
1476    minimum_x: f64,
1477    minimum_y: f64,
1478    maximum_x: f64,
1479    maximum_y: f64,
1480}
1481
1482impl LabelScreenRectangle {
1483    /// The collision rectangle around a centered label's text, padded by
1484    /// [`LABEL_COLLISION_PADDING_PIXELS`] on every side. A rotated label uses
1485    /// the axis-aligned box around its rotated text rectangle: the half
1486    /// extents are the rotated text box's own projections onto the screen
1487    /// axes, so the box never rejects less than the visible glyphs cover.
1488    fn around(
1489        anchor_screen_pixels: [f64; 2],
1490        width_pixels: f64,
1491        height_pixels: f64,
1492        rotation_radians: f64,
1493    ) -> Self {
1494        let (rotation_sine, rotation_cosine) = rotation_radians.sin_cos();
1495        let half_width_pixels =
1496            width_pixels / 2.0 * rotation_cosine.abs() + height_pixels / 2.0 * rotation_sine.abs();
1497        let half_height_pixels =
1498            width_pixels / 2.0 * rotation_sine.abs() + height_pixels / 2.0 * rotation_cosine.abs();
1499        Self {
1500            minimum_x: anchor_screen_pixels[0] - half_width_pixels - LABEL_COLLISION_PADDING_PIXELS,
1501            maximum_x: anchor_screen_pixels[0] + half_width_pixels + LABEL_COLLISION_PADDING_PIXELS,
1502            minimum_y: anchor_screen_pixels[1]
1503                - half_height_pixels
1504                - LABEL_COLLISION_PADDING_PIXELS,
1505            maximum_y: anchor_screen_pixels[1]
1506                + half_height_pixels
1507                + LABEL_COLLISION_PADDING_PIXELS,
1508        }
1509    }
1510
1511    /// Whether the two padded rectangles share any area.
1512    fn intersects(&self, other: &Self) -> bool {
1513        self.minimum_x < other.maximum_x
1514            && other.minimum_x < self.maximum_x
1515            && self.minimum_y < other.maximum_y
1516            && other.minimum_y < self.maximum_y
1517    }
1518
1519    /// Whether any part of the rectangle is inside the viewport.
1520    fn intersects_viewport(&self, viewport: &ViewportPixels) -> bool {
1521        self.minimum_x < viewport.width_pixels
1522            && self.maximum_x > 0.0
1523            && self.minimum_y < viewport.height_pixels
1524            && self.maximum_y > 0.0
1525    }
1526
1527    /// Whether the padded rectangle lies entirely inside the viewport; a
1528    /// `symbol-avoid-edges` label must pass this instead of
1529    /// [`Self::intersects_viewport`].
1530    fn is_inside_viewport(&self, viewport: &ViewportPixels) -> bool {
1531        self.minimum_x >= 0.0
1532            && self.maximum_x <= viewport.width_pixels
1533            && self.minimum_y >= 0.0
1534            && self.maximum_y <= viewport.height_pixels
1535    }
1536}

1565/// Appends one rotated glyph bitmap: the glyph rectangle is laid out in the
1566/// label's own frame (offset from the anchor), rotated about the anchor by
1567/// `rotation_radians`, and emitted as the same two screen-space triangles the
1568/// unrotated path draws.
1569// The arguments mirror `append_glyph_quad`'s, plus the anchor and rotation;
1570// splitting them into a struct would only move the same values to the call
1571// site.
1572#[allow(clippy::too_many_arguments)]
1573fn append_rotated_glyph_quad(
1574    vertices: &mut Vec<ClipSpaceGlyphVertex>,
1575    viewport: &ViewportPixels,
1576    anchor_screen_pixels: [f64; 2],
1577    rotation_radians: f64,
1578    top_left_offset_pixels: [f64; 2],
1579    size_pixels: [f64; 2],
1580    glyph: &PackedGlyph,
1581    signed_distance_threshold: f32,
1582    signed_distance_gamma: f32,
1583    rgba_color: &[f32; 4],
1584) {
1585    let (rotation_sine, rotation_cosine) = rotation_radians.sin_cos();
1586    let local_corners = axis_aligned_corners(top_left_offset_pixels, size_pixels);
1587    let rotated_corners = local_corners.map(|local_corner| {
1588        [
1589            anchor_screen_pixels[0] + local_corner[0] * rotation_cosine
1590                - local_corner[1] * rotation_sine,
1591            anchor_screen_pixels[1]
1592                + local_corner[0] * rotation_sine
1593                + local_corner[1] * rotation_cosine,
1594        ]
1595    });
1596    append_glyph_quad_corners(
1597        vertices,
1598        viewport,
1599        &rotated_corners,
1600        glyph.uv_rectangle,
1601        signed_distance_threshold,
1602        signed_distance_gamma,
1603        rgba_color,
1604    );
1605}

1607/// The four screen-space corners of an axis-aligned rectangle in the
1608/// `top-left, top-right, bottom-left, bottom-right` order the shared index
1609/// pattern consumes.
1610fn axis_aligned_corners(top_left_pixels: [f64; 2], size_pixels: [f64; 2]) -> [[f64; 2]; 4] {
1611    let right_pixels = top_left_pixels[0] + size_pixels[0];
1612    let bottom_pixels = top_left_pixels[1] + size_pixels[1];
1613    [
1614        [top_left_pixels[0], top_left_pixels[1]],
1615        [right_pixels, top_left_pixels[1]],
1616        [top_left_pixels[0], bottom_pixels],
1617        [right_pixels, bottom_pixels],
1618    ]
1619}

1621/// Emits one glyph bitmap from its four screen-space corners, in the shared
1622/// `top-left, top-right, bottom-left, bottom-right` order, as two triangles.
1623fn append_glyph_quad_corners(
1624    vertices: &mut Vec<ClipSpaceGlyphVertex>,
1625    viewport: &ViewportPixels,
1626    screen_corners: &[[f64; 2]; 4],
1627    texture_coordinates: rustic_maps_glyphs::GlyphUvRectangle,
1628    signed_distance_threshold: f32,
1629    signed_distance_gamma: f32,
1630    rgba_color: &[f32; 4],
1631) {

1656/// layers, at an explicit style zoom level (the frame camera's zoom, so size
1657/// stops evaluate where the camera is rather than where the tile belongs).
1658///
1659/// A feature becomes a candidate only when its `rank` passes the layer's
1660/// zoom-stepped rank ceiling and its `class` is visible at `style_zoom_level`
1661/// under the layer's per-class minimum zooms, so a busy poi layer draws its
1662/// important places at a wide zoom and its shops only once the map is closer.
1663/// A point-placed symbol layer then produces one candidate per feature point;
1664/// a line-placed one produces one candidate carrying each feature's line
1665/// geometry, which the frame places along its longest projected segment.
1666/// Features of a geometry kind the placement cannot use paint nothing for
1667/// that symbol layer, exactly like the overlay conversion.
1668pub fn label_candidates_for_vector_tile(
1669    tile: &VectorTile,
1670    tile_address: VectorTileAddress,
1671    style: &CompiledStyle,
1672    style_zoom_level: f64,
1673) -> Vec<MapLabelCandidate> {
1674    let mut candidates = Vec::new();
1675    for layer in &style.layers {
1676        candidates.extend(label_candidates_for_compiled_layer_at_zoom(
1677            tile,
1678            tile_address,
1679            layer,
1680            style_zoom_level,
1681        ));
1682    }
1683    candidates
1684}

1686/// Reads the label candidates one compiled symbol layer contributes from one
1687/// decoded tile.
1688///
1689/// A multi-source frame styles each source's tiles independently and calls this
1690/// per layer, so a symbol layer never reads geometry fetched for another
1691/// source. A non-symbol layer, an invisible layer and a layer without text
1692/// contribute nothing.
1693pub fn label_candidates_for_compiled_layer_at_zoom(
1694    tile: &VectorTile,
1695    tile_address: VectorTileAddress,
1696    layer: &rustic_maps_style::CompiledLayer,
1697    style_zoom_level: f64,
1698) -> Vec<MapLabelCandidate> {
1699    let mut candidates = Vec::new();
1700    if layer.kind != CompiledLayerKind::Symbol || !layer.is_visible_at_zoom(style_zoom_level) {
1701        return candidates;
1702    }
1703    for tile_layer in crate::vector_style::tile_layers_for_compiled_layer(tile, layer) {
1704        for feature in &tile_layer.features {
1705            let properties = feature_properties(feature);
1706            let context = feature_evaluation_context(feature, &properties, style_zoom_level);
1707            if !layer.matches_filter(&context) {
1708                continue;
1709            }
1710            let layout = layer.resolved_symbol_layout(&context);
1711            if layout.text.is_empty() {
1712                continue;
1713            }
1714            let ResolvedPaint::Symbol(paint) = layer.resolved_paint(&context) else {
1715                continue;
1716            };
1717            let text_rgba = resolved_rgba(paint.text_color, paint.text_opacity);
1718            let halo_rgba = resolved_rgba(paint.halo_color, 1.0);
1719            let priority = symbol_sort_key_priority(layout.sort_key);
1720            let rank = label_rank_for_feature(feature);
1721            let ignore_placement = layout.ignore_placement;
1722            let avoid_edges = layout.avoid_edges;
1723            let text_offset_pixels = [
1724                layout.text_offset[0] * layout.text_size,
1725                layout.text_offset[1] * layout.text_size,
1726            ];
1727            let repeat_spacing_pixels = if layer.declares_symbol_spacing {
1728                layout.symbol_spacing
1729            } else {
1730                LINE_LABEL_REPEAT_SPACING_PIXELS
1731            };
1732            let line_center_only = layout.placement == SymbolPlacement::LineCenter;
1733            let text_max_width_ems = layout.text_max_width;
1734            let text_line_height_ems = layout.text_line_height;
1735            let text_letter_spacing_ems = layout.text_letter_spacing;
1736            let text_justify = layout.text_justify;
1737            let text_anchor = layout.text_anchor;
1738            let text_rotation_alignment = layout.text_rotation_alignment;
1739            let text_keep_upright = layout.text_keep_upright;
1740            let text_variable_anchors = layout.variable_anchors.clone();
1741            let text_radial_offset_ems = layout.radial_offset;
1742            let text_max_angle_radians = if layer.declares_text_max_angle {
1743                layout
1744                    .max_angle_degrees
1745                    .to_radians()
1746                    .clamp(0.0, std::f64::consts::PI)
1747            } else {
1748                LINE_LABEL_STRAIGHT_RUN_ANGLE_TOLERANCE_RADIANS
1749            };
1750            match layout.placement {
1751                SymbolPlacement::Point => {
1752                    let VectorTileGeometry::Points(points) = &feature.geometry else {
1753                        continue;
1754                    };
1755                    for point in points {
1756                        candidates.push(MapLabelCandidate {
1757                            text: layout.text.clone(),
1758                            anchor: tile_local_point_to_coordinate(
1759                                tile_address,
1760                                tile_layer.extent,
1761                                point,
1762                            ),
1763                            line_coordinates: Vec::new(),
1764                            text_rgba,
1765                            text_size_pixels: layout.text_size,
1766                            halo_rgba,
1767                            halo_width_pixels: paint.halo_width,
1768                            priority,
1769                            rank,
1770                            allow_overlap: layout.allow_overlap,
1771                            ignore_placement,
1772                            avoid_edges,
1773                            text_offset_pixels,
1774                            repeat_spacing_pixels,
1775                            line_center_only,
1776                            text_max_width_ems,
1777                            text_line_height_ems,
1778                            text_letter_spacing_ems,
1779                            text_justify,
1780                            text_anchor,
1781                            text_rotation_alignment,
1782                            text_keep_upright,
1783                            text_variable_anchors: text_variable_anchors.clone(),
1784                            text_radial_offset_ems,
1785                            text_max_angle_radians,
1786                            style_order: layer.order_index,
1787                        });
1788                    }
1789                }
1790                SymbolPlacement::Line | SymbolPlacement::LineCenter => {
1791                    let VectorTileGeometry::LineStrings(line_strings) = &feature.geometry else {
1792                        continue;
1793                    };
1794                    for line_string in line_strings {
1795                        if line_string.len() < 2 {
1796                            continue;
1797                        }
1798                        let line_coordinates: Vec<GeographicCoordinate> = line_string
1799                            .iter()
1800                            .map(|point| {
1801                                tile_local_point_to_coordinate(
1802                                    tile_address,
1803                                    tile_layer.extent,
1804                                    point,
1805                                )
1806                            })
1807                            .collect();
1808                        let anchor = line_coordinates[0];
1809                        candidates.push(MapLabelCandidate {
1810                            text: layout.text.clone(),
1811                            anchor,
1812                            line_coordinates,
1813                            text_rgba,
1814                            text_size_pixels: layout.text_size,
1815                            halo_rgba,
1816                            halo_width_pixels: paint.halo_width,
1817                            priority,
1818                            rank,
1819                            allow_overlap: layout.allow_overlap,
1820                            ignore_placement,
1821                            avoid_edges,
1822                            text_offset_pixels,
1823                            repeat_spacing_pixels,
1824                            line_center_only,
1825                            text_max_width_ems,
1826                            text_line_height_ems,
1827                            text_letter_spacing_ems,
1828                            text_justify,
1829                            text_anchor,
1830                            text_rotation_alignment,
1831                            text_keep_upright,
1832                            text_variable_anchors: text_variable_anchors.clone(),
1833                            text_radial_offset_ems,
1834                            text_max_angle_radians,
1835                            style_order: layer.order_index,
1836                        });
1837                    }

1839            }
1840        }
1841    }

1845/// The compiled placement key as the candidate's whole-number priority. The
1846/// style specification allows a fractional sort key; the label pass orders by
1847/// an integer column, so a fractional key truncates toward zero and a negative
1848/// or oversized key clamps into the column's range.
1849fn symbol_sort_key_priority(sort_key: f64) -> u32 {
1850    if !sort_key.is_finite() || sort_key <= 0.0 {
1851        0
1852    } else if sort_key >= f64::from(u32::MAX) {
1853        u32::MAX
1854    } else {
1855        sort_key as u32
1856    }
1857}

1859/// The feature's `rank` attribute as a whole number, or [`DEFAULT_LABEL_RANK`]
1860/// when it carries none (negative ranks clamp to zero).
1861fn label_rank_for_feature(feature: &VectorTileFeature) -> u32 {
1862    feature
1863        .attributes
1864        .iter()
1865        .find(|(attribute_name, _)| attribute_name == LABEL_RANK_ATTRIBUTE_NAME)
1866        .and_then(|(_, attribute_value)| attribute_value.numeric_value())
1867        .map(|rank| {
1868            if rank <= 0.0 {
1869                0
1870            } else if rank >= f64::from(u32::MAX) {
1871                u32::MAX
1872            } else {
1873                rank as u32
1874            }
1875        })
1876        .unwrap_or(DEFAULT_LABEL_RANK)
1877}
