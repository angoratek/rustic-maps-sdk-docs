1//! The frame-level label pass: symbol-layer text from decoded vector tiles
2//! becomes screen-space glyph quads, anchored at a point for place and POI
3//! names and run along a line for road names.
4//!
5//! The pass is deliberately GPU-free, like the overlay assembly: the native
6//! headless renderer, the native surface and the browser canvas all call the
7//! same functions and upload the same vertices, so the backends cannot drift
8//! in where a label lands or which labels survive.
9//!
10//! Three stages:
11//!
12//! 1. [`label_candidates_for_vector_tile`] walks a decoded tile's symbol-layer
13//!    features and reads each label's text through the compiled style's
14//!    text-field expression. Candidates keep the style's paint, size, priority,
15//!    rank, placement and overlap rule. A symbol layer's filter also gates a
16//!    feature before it becomes a candidate: the converted styles encode the
17//!    zoom-stepped rank ceiling and the per-class minimum zooms as `step`
18//!    expressions on `["zoom"]` inside an `all` filter. A
19//!    [`SymbolPlacement::Point`] layer anchors a candidate at each of the
20//!    feature's points; a [`SymbolPlacement::Line`] layer carries the feature's
21//!    whole line geometry so the frame can place the text along it.
22//! 2. [`LabelLayer::place`] projects the candidates through the frame camera,
23//!    thins them by rank at the frame zoom, sorts them by symbol class, then
24//!    feature rank, then style-layer order (the documented collision-priority
25//!    rule: lower priority is more important), and greedily rejects any label
26//!    whose padded glyph box intersects a label already placed. Every drawn
27//!    glyph contributes its own box, so a name can sit in the gap beside a
28//!    neighbour's word space; a rotated line label tests its oriented boxes.
29//!    `allow_overlap` labels skip the test and never block later labels. The
30//!    frame places at most
31//!    [`MAXIMUM_LABELS_PER_FRAME`] labels: once that budget is spent, every
32//!    remaining candidate is dropped even if it would fit, so a dense downtown
33//!    cannot out-shout the places that matter. A share of the budget is kept
34//!    free for line-placed names ([`RESERVED_LINE_LABEL_SLOTS`]), so a frame
35//!    whose ranked places would spend the whole budget still carries street
36//!    names. A label whose text repeats an
37//!    already-placed label is dropped, so the buffered overlap between
38//!    adjacent tiles cannot paint one place twice. The output order is a pure
39//!    function of the input order and the camera, unless the caller hands the
40//!    previous frame's placed labels to
41//!    [`LabelLayer::place_with_history`]: among candidates that tie on the
42//!    priority key, a name the previous frame placed is tried first, so a
43//!    panning frame holds a name on screen instead of letting an
44//!    equal-priority newcomer take its space on one frame and lose it on the
45//!    next. History never reorders across symbol class, rank or style-layer
46//!    tier, so the frame stays a deterministic priority order.
47//!
48//!    A line-placed label projects its geometry and takes its longest straight
49//!    run (consecutive near-collinear segments) as the text's run: the text is
50//!    centered halfway along that run, rotated to its chord bearing, flipped
51//!    so it always reads upright, and skipped when the run is shorter than the
52//!    text plus the collision padding at each end. Each glyph's collision box
53//!    takes the label's rotation, so a road name rejects with its oriented
54//!    extent, not the empty corners of an axis-aligned bound.
55//! 3. [`LabelLayer::quad_vertices`] lays the glyph bitmaps out on the atlas and
56//!    emits clip-space textured quads, halo first and text second. A rotated
57//!    label rotates each glyph's screen-space rectangle about the label's
58//!    anchor before converting it to clip space, so every backend that draws
59//!    these vertices carries rotation without a second vertex format.
60//!
61//! Latin text is measured from the decoded glyph ranges and nothing here
62//! shapes complex scripts (see `rustic-maps-glyphs`). A frame with no symbol layers or
63//! no glyphs produces no vertices, so the tiles-only pipelines stay
64//! byte-identical.

87/// The em size the glyph protocol rasterizes signed-distance-field glyphs at:
88/// every glyph metric in a range payload is expressed in these 24-pixel em

90pub const GLYPH_FONT_EM_PIXELS: f64 = 24.0;

109/// The signed-distance value at which a glyph bitmap counts as covered.
110///
111/// The glyph protocol is node-fontnik's encoding,
112/// `byte = 255 - (64 + 32 × distance_px)`, so the zero-distance contour — the
113/// glyph outline — sits at `(256 − 64)/256 = 0.75`, not at 0.5. Thresholding at
114/// 0.5 dilates every glyph by `(0.75 − 0.5) × 8 = 2` atlas pixels per side,
115/// which closes counters and fattens strokes. Thresholding at 0.75 keeps the
116/// outline faithful to the font.
117pub const GLYPH_TEXT_SIGNED_DISTANCE_THRESHOLD: f32 = 0.75;
118
119/// The glyph protocol's spread: one atlas pixel of distance is `1/8` of a
120/// normalized distance-field unit. It converts atlas pixels into threshold
121/// offsets.
122pub const GLYPH_SIGNED_DISTANCE_FIELD_PIXELS_PER_UNIT: f32 = 8.0;
123
124/// The glyph protocol's border, in atlas pixels: the distance field fades to
125/// zero beyond this many pixels outside the ink box, so a halo can only reach
126/// this far before it samples the border texel.
127pub const GLYPH_SIGNED_DISTANCE_FIELD_MAXIMUM_OUTWARD_PIXELS: f64 =
128    GLYPH_SIGNED_DISTANCE_FIELD_BORDER_PIXELS as f64;
129
130/// The half-width of the glyph shader's anti-aliasing transition, in device
131/// pixels: the fragment stage's `smoothstep` half-band is
132/// `0.105 / devicePixelRatio / fontScale` in normalized field units, which is
133/// 0.84 device pixels at every size and pixel ratio.
134pub const GLYPH_EDGE_GAMMA_DEVICE_PIXELS: f32 = 0.105;
135
136/// The signed-distance threshold that draws a glyph's outline shifted outward
137/// by `halo_width_pixels`, clamped to the field's border.
138///
139/// `halo_width / fontScale` is the halo in atlas pixels, and one atlas pixel
140/// is `1 / 8` field unit, so the threshold moves down by that amount. The halo
141/// quad itself is not inflated: the fragment stage's lower threshold is what
142/// spreads the edge.
143pub fn glyph_halo_signed_distance_threshold(text_size_pixels: f64, halo_width_pixels: f64) -> f32 {
144    if halo_width_pixels <= 0.0 || text_size_pixels <= 0.0 {
145        return GLYPH_TEXT_SIGNED_DISTANCE_THRESHOLD;
146    }
147    let font_scale = text_size_pixels / GLYPH_FONT_EM_PIXELS;
148    let halo_atlas_pixels = halo_width_pixels / font_scale;
149    let halo_units = (halo_atlas_pixels as f32)
150        .min(GLYPH_SIGNED_DISTANCE_FIELD_MAXIMUM_OUTWARD_PIXELS as f32)
151        / GLYPH_SIGNED_DISTANCE_FIELD_PIXELS_PER_UNIT;
152    GLYPH_TEXT_SIGNED_DISTANCE_THRESHOLD - halo_units
153}
154
155/// The fragment stage's smoothstep half-band for one label, in normalized
156/// distance-field units: `0.105 / devicePixelRatio / fontScale`, where
157/// `fontScale = text_size_pixels / 24` (the style size, in CSS or point
158/// pixels, against the atlas' 24-pixel em). The drawn text size in device
159/// pixels is the same number times the device pixel ratio; the two uses keep
160/// the band at 0.84 device pixels at every size and ratio.
161pub fn glyph_edge_gamma(text_size_pixels: f64, device_pixel_ratio: f64) -> f32 {
162    if text_size_pixels <= 0.0 || device_pixel_ratio <= 0.0 {
163        return 0.0;
164    }
165    let font_scale = text_size_pixels / GLYPH_FONT_EM_PIXELS;
166    (f64::from(GLYPH_EDGE_GAMMA_DEVICE_PIXELS) / device_pixel_ratio / font_scale) as f32
167}

179/// names always win, and a busy tile can only spend the frame's remaining

185/// The frame label slots kept free for line-placed names (road names and any

198/// The rank an unranked label carries: the highest value, so it sorts after
199/// every ranked label of the same style layer. An unranked label is exempt
200/// from regional thinning: road names and shields carry no `rank` attribute
201/// and stay on the map at every zoom.
202pub const DEFAULT_LABEL_RANK: u32 = u32::MAX;

204/// One regional thinning band: a ranked label survives at a frame zoom below
205/// `maximum_zoom_level` only when its rank is at most `maximum_rank`.
206///
207/// Bands are read in order and the first one whose `maximum_zoom_level` is
208/// greater than the frame zoom applies; a zoom at or above the last band's
209/// ceiling places every ranked label. The band edges are the zooms the
210/// reference engines change their place-density rules at, so a regional frame
211/// keeps cities and the most important towns while a city frame shows the
212/// neighbourhood names.
213#[derive(Debug, Clone, Copy, PartialEq)]
214pub struct LabelRankThinningBand {
215    /// The frame zoom the band stops applying at (exclusive).
216    pub maximum_zoom_level: f64,
217    /// The highest feature rank the band keeps.
218    pub maximum_rank: u32,
219}

246/// The highest feature rank a ranked label may carry at `zoom_level` before
247/// regional thinning drops it. [`DEFAULT_LABEL_RANK`] (unranked labels) is
248/// always allowed.
249///
250/// The first [`LABEL_REGIONAL_THINNING_BANDS`] entry whose
251/// `maximum_zoom_level` exceeds the frame zoom applies; a frame at or beyond
252/// the last band's ceiling places every ranked label.
253pub fn maximum_label_rank_at_zoom(zoom_level: f64) -> u32 {
254    LABEL_REGIONAL_THINNING_BANDS
255        .iter()
256        .find(|band| zoom_level < band.maximum_zoom_level)
257        .map_or(DEFAULT_LABEL_RANK, |band| band.maximum_rank)
258}

260/// Whether a candidate's rank survives regional thinning at a frame zoom. An
261/// unranked candidate ([`DEFAULT_LABEL_RANK`]) always survives.
262fn label_passes_regional_thinning(rank: u32, zoom_level: f64) -> bool {
263    rank == DEFAULT_LABEL_RANK || rank <= maximum_label_rank_at_zoom(zoom_level)
264}

514impl LabelFont {
515    /// Packs one decoded glyph range into the atlas the label pass measures
516    /// and draws with.
517    ///
518    /// # Parameters
519    /// - `font_stack` — the decoded glyph range to pack.
520    ///
521    /// # Errors
522    /// A page too small for the range reports the packer's typed error rather

550    /// The packed glyph for a codepoint, if the range serves it.
551    ///
552    /// The lookup binary-searches the atlas group that can contain the
553    /// codepoint (the Latin ranges or every other range), so measuring text
554    /// stays logarithmic in the glyph count rather than scanning the page's
555    /// whole glyph list per character.
556    pub fn glyph(&self, codepoint: u32) -> Option<&PackedGlyph> {
557        let glyphs = &self.atlas.glyphs;
558        let latin_glyph_count = self.latin_glyph_count.min(glyphs.len());
559        let searchable_glyphs = if GlyphRange::is_latin_codepoint(codepoint) {
560            &glyphs[..latin_glyph_count]
561        } else {
562            &glyphs[latin_glyph_count..]
563        };
564        let glyph_index = searchable_glyphs
565            .binary_search_by_key(&codepoint, |glyph| glyph.codepoint)
566            .ok()?;
567        searchable_glyphs.get(glyph_index)
568    }

570    /// Measures a label's text at `text_size_pixels`.
571    ///
572    /// Characters the font does not serve contribute no advance and paint
573    /// nothing, exactly as the quad emitter skips them, so a name with one
574    /// out-of-range character still draws its remaining text rather than
575    /// disappearing. `None` is reported only when the font serves none of the
576    /// characters at all — a label that would paint nothing is not placed.
577    pub fn measure_text(&self, text: &str, text_size_pixels: f64) -> Option<LabelTextMetrics> {
578        let width_pixels = self.measure_line_width(text, text_size_pixels, 0.0)?;
579        Some(LabelTextMetrics {
580            width_pixels,
581            height_pixels: text_size_pixels,
582        })
583    }

585    /// One line's advance width, letter spacing included between served
586    /// glyphs; `None` when the font serves none of the line's characters.
587    fn measure_line_width(
588        &self,
589        text: &str,
590        text_size_pixels: f64,
591        letter_spacing_pixels: f64,
592    ) -> Option<f64> {
593        let scale = text_size_pixels / GLYPH_FONT_EM_PIXELS;
594        let mut width_pixels = 0.0;
595        let mut served_glyph_count = 0_usize;
596        for character in text.chars() {
597            let Some(glyph) = self.glyph(u32::from(character)) else {
598                continue;
599            };
600            if served_glyph_count > 0 {
601                width_pixels += letter_spacing_pixels;
602            }
603            served_glyph_count += 1;
604            width_pixels += f64::from(glyph.advance) * scale;
605        }
606        if served_glyph_count == 0 {
607            return None;
608        }
609        Some(width_pixels)
610    }

612    /// Measures a label's text as a wrapped block.
613    ///
614    /// Words are laid out greedily: a word that would push the line past
615    /// `maximum_width_pixels` starts the next one, and a word wider than the
616    /// limit on its own keeps that line. `line_height_ems` multiplies the
617    /// text size for one line's height. The block width is the widest line,
618    /// so a centered block is centered on its longest line. `None` is
619    /// reported when the font serves no character of any line.
620    pub fn measure_wrapped_text(
621        &self,
622        text: &str,
623        text_size_pixels: f64,
624        maximum_width_pixels: f64,
625        line_height_ems: f64,
626        letter_spacing_pixels: f64,
627    ) -> Option<LabelTextBlock> {
628        let mut line_texts: Vec<String> = Vec::new();
629        let mut current_line = String::new();
630        for word in text.split_whitespace() {
631            let candidate_line = if current_line.is_empty() {
632                word.to_string()
633            } else {
634                format!("{current_line} {word}")
635            };
636            let candidate_width =
637                self.measure_line_width(&candidate_line, text_size_pixels, letter_spacing_pixels);
638            let exceeds_limit = candidate_width
639                .is_some_and(|candidate_width| candidate_width > maximum_width_pixels);
640            if !current_line.is_empty() && exceeds_limit {
641                line_texts.push(std::mem::take(&mut current_line));
642                current_line = word.to_string();
643            } else {
644                current_line = candidate_line;
645            }
646        }
647        if !current_line.is_empty() {
648            line_texts.push(current_line);
649        }
650        if line_texts.is_empty() {
651            return None;
652        }
653        let mut lines = Vec::with_capacity(line_texts.len());
654        let mut widest_line_pixels = 0.0_f64;
655        let mut served_any_line = false;
656        for line_text in line_texts {
657            let width_pixels = self
658                .measure_line_width(&line_text, text_size_pixels, letter_spacing_pixels)
659                .unwrap_or(0.0);
660            served_any_line |= width_pixels > 0.0;
661            widest_line_pixels = widest_line_pixels.max(width_pixels);
662            lines.push(LabelTextLine {
663                text: line_text,
664                width_pixels,
665            });
666        }
667        if !served_any_line {
668            return None;
669        }
670        let line_height_pixels = text_size_pixels * line_height_ems;
671        Some(LabelTextBlock {
672            height_pixels: line_height_pixels * lines.len() as f64,
673            lines,
674            width_pixels: widest_line_pixels,
675            line_height_pixels,
676        })
677    }
678}

680/// How many Latin glyphs sit at the front of a packed atlas: the packer
681/// orders the Latin ranges first, then every other range, each group ascending
682/// by codepoint, so the first non-Latin glyph's index splits the atlas into
683/// the two sorted groups a binary-search glyph lookup searches.
684fn latin_glyph_prefix_length(atlas: &PackedGlyphAtlas) -> usize {
685    atlas
686        .glyphs
687        .iter()
688        .position(|glyph| !GlyphRange::is_latin_codepoint(glyph.codepoint))
689        .unwrap_or(atlas.glyphs.len())
690}

692/// A deterministic fingerprint of a packed atlas: the page dimensions, every
693/// glyph's codepoint and rectangle, and the page pixels. Computed once when
694/// the atlas is packed, never per frame.
695fn atlas_revision(atlas: &PackedGlyphAtlas) -> u64 {
696    let mut hasher = DefaultHasher::new();
697    atlas.width_pixels.hash(&mut hasher);
698    atlas.height_pixels.hash(&mut hasher);
699    for glyph in &atlas.glyphs {
700        glyph.codepoint.hash(&mut hasher);
701        glyph.bitmap_rectangle.x_pixels.hash(&mut hasher);
702        glyph.bitmap_rectangle.y_pixels.hash(&mut hasher);
703        glyph.bitmap_rectangle.width_pixels.hash(&mut hasher);
704        glyph.bitmap_rectangle.height_pixels.hash(&mut hasher);
705    }
706    atlas.pixels.hash(&mut hasher);
707    hasher.finish()
708}

802/// A label layer ready to draw: the placed labels and the font that measures
803/// and rasterizes them.
804#[derive(Debug, Clone)]
805pub struct LabelLayer {
806    labels: Vec<PlacedLabel>,
807    font: LabelFont,
808    /// The device pixels one point/CSS pixel covers: 1.0 for the headless
809    /// golden at one device pixel per point, the display scale on a native
810    /// surface, the canvas ratio in the browser. It sets the drawn text size
811    /// and the fragment stage's edge gamma; the halo width already arrives in
812    /// the same points the style declares.
813    device_pixel_ratio: f64,
814    /// The sprite atlas the placed icons sample, when the frame supplied one.
815    sprite: Option<Arc<SpriteAtlas>>,
816}

818impl LabelLayer {
819    /// Projects and places the frame's candidates: thin ranked labels by the
820    /// frame zoom, sort by symbol class (lower is more important, the field's
821    /// documented collision rule), then feature rank, then style order; drop
822    /// repeated text; reject a candidate any of whose padded glyph boxes
823    /// collides with an already accepted label unless the candidate allows
824    /// overlap; stop once the frame has placed
825    /// [`MAXIMUM_LABELS_PER_FRAME`] labels. A line-placed candidate takes its
826    /// longest straight projected run as the text's run and repeats along it,
827    /// each label rotated to the segment beneath it; every other candidate
828    /// centers on its anchor, unrotated. Repeated names of one candidate are
829    /// placed together, and the text is recorded once so a later candidate
830    /// carrying the same name — the same street in a neighbouring tile — is
831    /// still rejected. The result keeps the sorted order, so the same
832    /// candidates always paint in the same sequence.
833    pub fn place(
834        camera: &MapCamera,
835        viewport: &ViewportPixels,
836        field_of_view_degrees: f64,
837        candidates: &[MapLabelCandidate],
838        font: &LabelFont,
839    ) -> Self {
840        Self::place_with_device_pixel_ratio(
841            camera,
842            viewport,
843            field_of_view_degrees,
844            candidates,
845            font,
846            1.0,
847        )
848    }

850    /// [`Self::place`] for a surface whose drawing buffer is not one device
851    /// pixel per point: the drawn glyph size and the edge gamma both follow
852    /// `device_pixel_ratio`, so the text keeps its declared point size and its
853    /// one-pixel anti-aliased edge on a 2× or 3× surface.
854    pub fn place_with_device_pixel_ratio(
855        camera: &MapCamera,
856        viewport: &ViewportPixels,
857        field_of_view_degrees: f64,
858        candidates: &[MapLabelCandidate],
859        font: &LabelFont,
860        device_pixel_ratio: f64,
861    ) -> Self {
862        Self::place_with_history(
863            camera,
864            viewport,
865            field_of_view_degrees,
866            candidates,
867            font,
868            device_pixel_ratio,
869            &[],
870        )
871    }
872
873    /// [`Self::place_with_device_pixel_ratio`] carrying a persistent collision
874    /// index across frames: `previous_labels` are the labels the previous frame
875    /// of the same view placed. The memory breaks ties among candidates that
876    /// share the whole priority key (symbol class, then rank, then style-layer
877    /// tier): an equal-priority name the previous frame placed is tried before
878    /// its equal-priority newcomers, in the order the previous frame held
879    /// them. A remembered name therefore keeps its spot while it stays
880    /// visible, so panning half a street name off the edge cannot let an
881    /// equal-priority neighbour take its place for one frame and lose it the
882    /// next. The history never overrides a better symbol class, rank or
883    /// style-layer tier, so two identical frames always place the same
884    /// priority winners.
885    ///
886    /// The memory is the label text, the pass's own identity for a name (a
887    /// frame drops repeated text), so it survives a camera move without
888    /// carrying stale screen geometry: the remembered candidate is projected
889    /// and collides like any other at the new pose, and a name that is no
890    /// longer among the candidates is simply forgotten.
891    pub fn place_with_history(
892        camera: &MapCamera,
893        viewport: &ViewportPixels,
894        field_of_view_degrees: f64,
895        candidates: &[MapLabelCandidate],
896        font: &LabelFont,
897        device_pixel_ratio: f64,
898        previous_labels: &[PlacedLabel],
899    ) -> Self {
900        Self::place_with_history_and_sprite(
901            camera,
902            viewport,
903            field_of_view_degrees,
904            candidates,
905            font,
906            device_pixel_ratio,
907            previous_labels,
908            None,
909        )
910    }
911
912    /// [`Self::place_with_history`] with a sprite atlas for the candidates'
913    /// `icon-image` entries.
914    ///
915    /// A candidate's icon and text collide as separate boxes: each must find
916    /// room under its own overlap rule, and the icon's own `icon-padding`
917    /// inflates its box. An icon the atlas does not serve contributes no box
918    /// and no draw, so a missing sprite or a missing entry leaves the text
919    /// exactly as it would have drawn.
920    ///
921    /// Placement order is explicit: a candidate's symbol class
922    /// (`priority`, the style's `symbol-sort-key`) first, then its feature
923    /// rank, then its style-layer order (the declared class/zoom tier), then
924    /// the previous frame's position among the candidates that still tie, with
925    /// the caller's order breaking the remaining ties. A route shield
926    /// therefore draws before a place name, and a more important ranked name
927    /// beats a lower-ranked one even when the style lists their layers the
928    /// other way round; the history can only reorder names that are otherwise
929    /// equally important.
930    ///
931    /// Regional thinning runs before the order: at a wide frame zoom a ranked
932    /// candidate above [`maximum_label_rank_at_zoom`] is dropped, so the
933    /// remaining places compete for the room in importance order. Unranked
934    /// labels (road names and shields) are exempt.
935    #[allow(clippy::too_many_arguments)]
936    pub fn place_with_history_and_sprite(
937        camera: &MapCamera,
938        viewport: &ViewportPixels,
939        field_of_view_degrees: f64,
940        candidates: &[MapLabelCandidate],
941        font: &LabelFont,
942        device_pixel_ratio: f64,
943        previous_labels: &[PlacedLabel],
944        sprite: Option<&Arc<SpriteAtlas>>,
945    ) -> Self {
946        // Regional thinning is a frame policy, not style data: the same
947        // candidate list at a wider camera keeps only the important ranks.
948        let mut ordered_candidates: Vec<&MapLabelCandidate> = candidates
949            .iter()
950            .filter(|candidate| label_passes_regional_thinning(candidate.rank, camera.zoom_level))
951            .collect();
952        // The history is a preference only among candidates that tie on the
953        // whole priority key. A remembered name therefore keeps its slot
954        // against an equal-priority newcomer, but it can never jump ahead of a
955        // better symbol class, rank or style-layer tier. Ordering the history
956        // first regardless of rank made the previous frame's winner beat its
957        // own higher-ranked competitor, and two identical frames could settle
958        // on different label sets.
959        let mut remembered_positions: HashMap<&str, usize> = HashMap::new();
960        for (position, remembered) in previous_labels.iter().enumerate() {
961            if remembered.text.is_empty() {
962                continue;
963            }
964            remembered_positions
965                .entry(remembered.text.as_str())
966                .or_insert(position);
967        }
968        // Symbol class first, then feature rank, then the style layer's own
969        // class/zoom tier, then the remembered position within the ties.
970        // `sort_by_key` is stable, so candidates that tie on the whole key keep
971        // the caller's own order — the frame's tiles list deterministically.
972        // This is the documented rule: class before rank, shields before names.
973        ordered_candidates.sort_by_key(|candidate| {
974            (
975                candidate.priority,
976                candidate.rank,
977                candidate.style_order,
978                remembered_positions
979                    .get(candidate.text.as_str())
980                    .copied()
981                    .unwrap_or(usize::MAX),
982            )
983        });
984
985        let mut occupied_boxes: Vec<LabelScreenBox> = Vec::new();
986        let mut placed_labels: Vec<PlacedLabel> = Vec::new();
987        let mut placed_texts: Vec<String> = Vec::new();
988        // Line-placed names get a reserved share of the budget, but only as
989        // many slots as the frame has line candidates to fill: a frame with
990        // few roads gives the unused slots back to the places.
991        let line_candidate_count = ordered_candidates
992            .iter()
993            .filter(|candidate| !candidate.text.is_empty() && candidate.line_coordinates.len() >= 2)
994            .count();
995        let place_label_budget =
996            MAXIMUM_LABELS_PER_FRAME - RESERVED_LINE_LABEL_SLOTS.min(line_candidate_count);
997        for candidate in ordered_candidates {
998            if placed_labels.len() >= MAXIMUM_LABELS_PER_FRAME {
999                break;
1000            }
1001            let placed_along_line =
1002                !candidate.text.is_empty() && candidate.line_coordinates.len() >= 2;
1003            if !placed_along_line && placed_labels.len() >= place_label_budget {
1004                // Keep the reserved slots for the line names later in the
1005                // priority order instead of ending the frame early.
1006                continue;
1007            }
1008            if !candidate.text.is_empty() && placed_texts.iter().any(|text| text == &candidate.text)
1009            {
1010                continue;
1011            }
1012            // The style's size is a point/CSS size; the drawing buffer draws
1013            // device pixels, so the glyphs scale by the surface ratio while the
1014            // style numbers stay the source of truth for gamma and halo math.
1015            let text_size_pixels = candidate.text_size_pixels * device_pixel_ratio;
1016            let letter_spacing_pixels = candidate.text_letter_spacing_ems * text_size_pixels;
1017            // A line-placed name stays on one line, the specification's rule;
1018            // wrapping applies to point-placed text. An icon-only symbol has
1019            // no text block at all.
1020            let text_block: Option<LabelTextBlock> = if candidate.text.is_empty() {
1021                None
1022            } else if placed_along_line {
1023                font.measure_line_width(&candidate.text, text_size_pixels, letter_spacing_pixels)
1024                    .map(|width_pixels| LabelTextBlock {
1025                        lines: vec![LabelTextLine {
1026                            text: candidate.text.clone(),
1027                            width_pixels,
1028                        }],
1029                        width_pixels,
1030                        height_pixels: text_size_pixels,
1031                        line_height_pixels: text_size_pixels,
1032                    })
1033            } else {
1034                font.measure_wrapped_text(
1035                    &candidate.text,
1036                    text_size_pixels,
1037                    candidate.text_max_width_ems * text_size_pixels,
1038                    candidate.text_line_height_ems,
1039                    letter_spacing_pixels,
1040                )
1041            };
1042            // Icons ride point placement. A line-placed icon would need a
1043            // run-relative anchor the specification leaves to the renderer;
1044            // a line symbol keeps its text and draws no icon rather than
1045            // anchoring the icon at an arbitrary vertex.
1046            let icon_candidate = if placed_along_line {
1047                None
1048            } else {
1049                candidate.icon.as_ref()
1050            };
1051            let icon_placement = icon_candidate.and_then(|icon_candidate| {
1052                icon_placement_at(
1053                    camera,
1054                    viewport,
1055                    field_of_view_degrees,
1056                    &candidate.anchor,
1057                    icon_candidate,
1058                    sprite.map(Arc::as_ref),
1059                    device_pixel_ratio,
1060                )
1061            });
1062            if text_block.is_none() && icon_placement.is_none() {
1063                continue;
1064            }
1065            // An icon shares the anchor with its text, so a variable anchor
1066            // cannot move one without the other; an icon-bearing candidate
1067            // keeps its declared anchor.
1068            let uses_variable_anchor = text_block.is_some()
1069                && !placed_along_line
1070                && icon_placement.is_none()
1071                && !candidate.text_variable_anchors.is_empty();
1072            let placements = if let Some(text_block) = &text_block {
1073                if placed_along_line {
1074                    let mut line_placements = label_placements_along_line(
1075                        camera,
1076                        viewport,
1077                        field_of_view_degrees,
1078                        &candidate.line_coordinates,
1079                        text_block.width_pixels,
1080                        candidate.repeat_spacing_pixels,
1081                        candidate.line_center_only,
1082                        candidate.text_max_angle_radians,
1083                    );
1084                    if candidate.text_rotation_alignment == TextRotationAlignment::Viewport {
1085                        for placement in &mut line_placements {
1086                            placement.rotation_radians = 0.0;
1087                        }
1088                    }
1089                    line_placements
1090                } else {
1091                    // A point label follows the map's bearing only when its
1092                    // style asks; `auto` is the viewport alignment for point
1093                    // placement.
1094                    let point_rotation_radians = match candidate.text_rotation_alignment {
1095                        TextRotationAlignment::Map => camera.bearing_degrees.to_radians(),
1096                        TextRotationAlignment::Viewport | TextRotationAlignment::Auto => 0.0,
1097                    };
1098                    let point_rotation_radians = if candidate.text_keep_upright {
1099                        upright_rotation_radians(point_rotation_radians)
1100                    } else {
1101                        point_rotation_radians
1102                    };
1103                    let anchor_candidates: Vec<TextAnchor> =
1104                        if candidate.text_variable_anchors.is_empty() {
1105                            vec![candidate.text_anchor]
1106                        } else {
1107                            candidate.text_variable_anchors.clone()
1108                        };
1109                    match camera.project_to_screen(
1110                        &candidate.anchor,
1111                        viewport,
1112                        field_of_view_degrees,
1113                    ) {
1114                        Some(anchor_screen_pixels) => anchor_candidates
1115                            .iter()
1116                            .map(|text_anchor| {
1117                                let mut anchor_offset = text_anchor_center_offset(
1118                                    *text_anchor,
1119                                    text_block.width_pixels,
1120                                    text_block.height_pixels,
1121                                );
1122                                // A variable anchor pushes the block outward by
1123                                // the radial offset; a fixed anchor ignores it.
1124                                if uses_variable_anchor {
1125                                    let radial_direction = radial_offset_direction(*text_anchor);
1126                                    anchor_offset[0] += radial_direction[0]
1127                                        * candidate.text_radial_offset_ems
1128                                        * text_size_pixels;
1129                                    anchor_offset[1] += radial_direction[1]
1130                                        * candidate.text_radial_offset_ems
1131                                        * text_size_pixels;
1132                                }
1133                                LabelPlacement {
1134                                    anchor_screen_pixels: [
1135                                        anchor_screen_pixels[0] + anchor_offset[0],
1136                                        anchor_screen_pixels[1] + anchor_offset[1],
1137                                    ],
1138                                    rotation_radians: point_rotation_radians,
1139                                }
1140                            })
1141                            .collect(),
1142                        None => Vec::new(),
1143                    }
1144                }
1145            } else {
1146                // An icon-only symbol still needs one placement to drive the
1147                // acceptance loop; the icon's own geometry is independent of
1148                // it and is measured from the candidate anchor.
1149                camera
1150                    .project_to_screen(&candidate.anchor, viewport, field_of_view_degrees)
1151                    .map(|anchor_screen_pixels| {
1152                        vec![LabelPlacement {
1153                            anchor_screen_pixels,
1154                            rotation_radians: 0.0,
1155                        }]
1156                    })
1157                    .unwrap_or_default()
1158            };
1159            // A text offset is a screen-space displacement from the anchor
1160            // in ems; the candidate carries it already scaled to CSS pixels.
1161            let placements: Vec<LabelPlacement> = placements
1162                .into_iter()
1163                .map(|mut placement| {
1164                    placement.anchor_screen_pixels[0] +=
1165                        candidate.text_offset_pixels[0] * device_pixel_ratio;
1166                    placement.anchor_screen_pixels[1] +=
1167                        candidate.text_offset_pixels[1] * device_pixel_ratio;
1168                    placement
1169                })
1170                .collect();
1171            let text_is_free = candidate.allow_overlap || candidate.ignore_placement;
1172            let icon_is_free = icon_candidate
1173                .map(|icon_candidate| {
1174                    icon_candidate.allow_overlap || icon_candidate.ignore_placement
1175                })
1176                .unwrap_or(true);
1177            let mut placed_any = false;
1178            for placement in placements {
1179                if placed_labels.len() >= MAXIMUM_LABELS_PER_FRAME {
1180                    break;
1181                }
1182                // Collision uses one box per drawn glyph, not one box around
1183                // the whole text block: a name can then sit in the gap beside
1184                // a neighbour's word space, as the reference engines allow.
1185                // Each box carries the label's collision padding, and a
1186                // rotated label's boxes keep the label's rotation, so a road
1187                // name rejects with its oriented extent rather than the empty
1188                // corners of an axis-aligned bound.
1189                let text_boxes: Vec<LabelScreenBox> = text_block
1190                    .as_ref()
1191                    .map(|text_block| {
1192                        glyph_collision_boxes(
1193                            font,
1194                            text_block,
1195                            text_size_pixels,
1196                            letter_spacing_pixels,
1197                            candidate.text_justify,
1198                            placement,
1199                        )
1200                    })
1201                    .unwrap_or_default();
1202                let icon_box =
1203                    icon_placement
1204                        .zip(icon_candidate)
1205                        .map(|(icon_placement, icon_candidate)| {
1206                            LabelScreenBox::around_with_padding(
1207                                icon_placement.anchor_screen_pixels,
1208                                icon_placement.width_pixels,
1209                                icon_placement.height_pixels,
1210                                icon_placement.rotation_radians,
1211                                icon_candidate.padding_pixels * device_pixel_ratio,
1212                            )
1213                        });
1214                let inside_viewport = |screen_box: &LabelScreenBox| {
1215                    if candidate.avoid_edges {
1216                        screen_box.is_inside_viewport(viewport)
1217                    } else {
1218                        screen_box.intersects_viewport(viewport)
1219                    }
1220                };
1221                let text_crosses_viewport = if candidate.avoid_edges {
1222                    !text_boxes.iter().all(inside_viewport)
1223                } else {
1224                    !text_boxes.is_empty() && !text_boxes.iter().any(inside_viewport)
1225                };
1226                if text_crosses_viewport
1227                    || icon_box
1228                        .as_ref()
1229                        .is_some_and(|screen_box| !inside_viewport(screen_box))
1230                {
1231                    continue;
1232                }
1233                let text_collides = !text_is_free
1234                    && text_boxes.iter().any(|screen_box| {
1235                        occupied_boxes
1236                            .iter()
1237                            .any(|occupied| occupied.intersects(screen_box))
1238                    });
1239                let icon_collides = icon_box.is_some_and(|screen_box| {
1240                    !icon_is_free
1241                        && occupied_boxes
1242                            .iter()
1243                            .any(|occupied| occupied.intersects(&screen_box))
1244                });
1245                if text_collides || icon_collides {
1246                    continue;
1247                }
1248                if !text_is_free {
1249                    occupied_boxes.extend(text_boxes);
1250                }
1251                if let Some(screen_box) = icon_box {
1252                    if !icon_is_free {
1253                        occupied_boxes.push(screen_box);
1254                    }
1255                }
1256                placed_any = true;
1257                placed_labels.push(PlacedLabel {
1258                    text: candidate.text.clone(),
1259                    anchor_screen_pixels: placement.anchor_screen_pixels,
1260                    rotation_radians: placement.rotation_radians,
1261                    placed_along_line,
1262                    text_size_pixels,
1263                    text_rgba: candidate.text_rgba,
1264                    halo_rgba: candidate.halo_rgba,
1265                    halo_width_pixels: candidate.halo_width_pixels * device_pixel_ratio,
1266                    lines: text_block
1267                        .as_ref()
1268                        .map(|text_block| text_block.lines.clone())
1269                        .unwrap_or_default(),
1270                    line_height_pixels: text_block
1271                        .as_ref()
1272                        .map(|text_block| text_block.line_height_pixels)
1273                        .unwrap_or_default(),
1274                    block_width_pixels: text_block
1275                        .as_ref()
1276                        .map(|text_block| text_block.width_pixels)
1277                        .unwrap_or_default(),
1278                    letter_spacing_pixels,
1279                    text_justify: candidate.text_justify,
1280                    priority: candidate.priority,
1281                    icon: icon_placement.zip(icon_candidate).map(
1282                        |(icon_placement, icon_candidate)| PlacedIcon {
1283                            image_name: icon_candidate.image_name.clone(),
1284                            anchor_screen_pixels: icon_placement.anchor_screen_pixels,
1285                            rotation_radians: icon_placement.rotation_radians,
1286                            width_pixels: icon_placement.width_pixels,
1287                            height_pixels: icon_placement.height_pixels,
1288                            texture_rectangle: icon_placement.texture_rectangle,
1289                            opacity: 1.0,
1290                        },
1291                    ),
1292                });
1293                // A variable-anchor label tries its anchors in order and
1294                // takes the first that fits, once.
1295                if uses_variable_anchor {
1296                    break;
1297                }
1298            }
1299            if placed_any && !candidate.text.is_empty() {
1300                placed_texts.push(candidate.text.clone());
1301            }
1302        }
1303        Self {
1304            labels: placed_labels,
1305            font: font.clone(),
1306            device_pixel_ratio,
1307            sprite: sprite.cloned(),
1308        }
1309    }

1331    /// Assembles the layer's clip-space quad vertices: one halo pass (only
1332    /// labels with a positive halo width) followed by the text pass. Both
1333    /// lists draw non-indexed through the textured glyph pipeline. A halo
1334    /// quad is the glyph's bitmap rectangle inflated by the halo width while
1335    /// keeping the glyph's texture window, which stretches the bitmap's soft
1336    /// edge outward; the fragment stage's lower halo threshold turns that
1337    /// stretched edge into the outline.
1338    ///
1339    /// A rotated label lays its glyphs out in the label's own unrotated frame
1340    /// and rotates each glyph rectangle about the label anchor, so rotation
1341    /// reaches every backend through the shared clip-space positions.
1342    pub fn quad_vertices(&self, viewport: &ViewportPixels) -> LabelQuadVertices {
1343        let mut halo_vertices = Vec::new();
1344        let mut text_vertices = Vec::new();
1345        for label in &self.labels {
1346            if label.rotation_radians == 0.0 {
1347                self.append_axis_aligned_label_quads(
1348                    &mut halo_vertices,
1349                    &mut text_vertices,
1350                    viewport,
1351                    label,
1352                );
1353            } else {
1354                self.append_rotated_label_quads(
1355                    &mut halo_vertices,
1356                    &mut text_vertices,
1357                    viewport,
1358                    label,
1359                );
1360            }
1361        }
1362        LabelQuadVertices {
1363            halo: halo_vertices,
1364            text: text_vertices,
1365        }
1366    }

1368    /// Assembles the placed icons' clip-space quads. Each icon is a centered
1369    /// rectangle rotated about its own anchor and mapped to its normalized
1370    /// atlas window, in the same corner and index order the shared textured
1371    /// pipeline draws. A layer with no atlas or no placed icon yields no
1372    /// quads, so the icon pass costs nothing on a text-only frame.
1373    pub fn icon_quad_vertices(&self, viewport: &ViewportPixels) -> IconQuadVertices {
1374        let mut quads = Vec::new();
1375        for label in &self.labels {
1376            let Some(icon) = &label.icon else {
1377                continue;
1378            };
1379            let half_width_pixels = icon.width_pixels / 2.0;
1380            let half_height_pixels = icon.height_pixels / 2.0;
1381            let local_corners = [
1382                [-half_width_pixels, -half_height_pixels],
1383                [half_width_pixels, -half_height_pixels],
1384                [-half_width_pixels, half_height_pixels],
1385                [half_width_pixels, half_height_pixels],
1386            ];
1387            let (rotation_sine, rotation_cosine) = icon.rotation_radians.sin_cos();
1388            let screen_corners = local_corners.map(|local_corner| {
1389                [
1390                    icon.anchor_screen_pixels[0] + local_corner[0] * rotation_cosine
1391                        - local_corner[1] * rotation_sine,
1392                    icon.anchor_screen_pixels[1]
1393                        + local_corner[0] * rotation_sine
1394                        + local_corner[1] * rotation_cosine,
1395                ]
1396            });
1397            let texture_corners = [
1398                [
1399                    icon.texture_rectangle.minimum_u,
1400                    icon.texture_rectangle.minimum_v,
1401                ],
1402                [
1403                    icon.texture_rectangle.maximum_u,
1404                    icon.texture_rectangle.minimum_v,
1405                ],
1406                [
1407                    icon.texture_rectangle.minimum_u,
1408                    icon.texture_rectangle.maximum_v,
1409                ],
1410                [
1411                    icon.texture_rectangle.maximum_u,
1412                    icon.texture_rectangle.maximum_v,
1413                ],
1414            ];
1415            quads.push(std::array::from_fn(|corner_index| ClipSpaceIconVertex {
1416                clip_position: [
1417                    (2.0 * screen_corners[corner_index][0] / viewport.width_pixels - 1.0) as f32,
1418                    (1.0 - 2.0 * screen_corners[corner_index][1] / viewport.height_pixels) as f32,
1419                    0.0,
1420                    1.0,
1421                ],
1422                texture_coordinates: texture_corners[corner_index],
1423                opacity: icon.opacity,
1424            }));
1425        }
1426        IconQuadVertices { quads }
1427    }

1429    /// Lays out one unrotated label's glyph quads: the baseline runs along
1430    /// the screen's `x` axis from the label's centered pen origin.
1431    fn append_axis_aligned_label_quads(
1432        &self,
1433        halo_vertices: &mut Vec<ClipSpaceGlyphVertex>,
1434        text_vertices: &mut Vec<ClipSpaceGlyphVertex>,
1435        viewport: &ViewportPixels,
1436        label: &PlacedLabel,
1437    ) {
1438        let scale = label.text_size_pixels / GLYPH_FONT_EM_PIXELS;
1439        // The style's own (point/CSS) sizes drive the SDF threshold and gamma;
1440        // the placed sizes are already in drawing-buffer pixels.
1441        let style_text_size_pixels = label.text_size_pixels / self.device_pixel_ratio;
1442        let style_halo_width_pixels = label.halo_width_pixels / self.device_pixel_ratio;
1443        let edge_gamma = glyph_edge_gamma(style_text_size_pixels, self.device_pixel_ratio);
1444        let halo_threshold =
1445            glyph_halo_signed_distance_threshold(style_text_size_pixels, style_halo_width_pixels);
1446        let baseline_offset_pixels = label.text_size_pixels * LABEL_BASELINE_OFFSET_FRACTION;
1447        let line_count = label.lines.len() as f64;
1448        for (line_index, line) in label.lines.iter().enumerate() {
1449            // Lines stack around the anchor: line 0 is the top line.
1450            let line_offset_pixels =
1451                (line_index as f64 - (line_count - 1.0) / 2.0) * label.line_height_pixels;
1452            let baseline_y =
1453                label.anchor_screen_pixels[1] + line_offset_pixels + baseline_offset_pixels;
1454            let mut pen_x = match label.text_justify {
1455                TextJustify::Left => label.anchor_screen_pixels[0] - label.block_width_pixels / 2.0,
1456                TextJustify::Right => {
1457                    label.anchor_screen_pixels[0] + label.block_width_pixels / 2.0
1458                        - line.width_pixels
1459                }
1460                TextJustify::Auto | TextJustify::Center => {
1461                    label.anchor_screen_pixels[0] - line.width_pixels / 2.0
1462                }
1463            };
1464            for character in line.text.chars() {
1465                let Some(glyph) = self.font.glyph(u32::from(character)) else {
1466                    continue;
1467                };
1468                let advance_pixels = f64::from(glyph.advance) * scale;
1469                if glyph.bitmap_rectangle.width_pixels > 0
1470                    && glyph.bitmap_rectangle.height_pixels > 0
1471                {
1472                    let bitmap_left_pixels = pen_x
1473                        + f64::from(glyph.left - GLYPH_SIGNED_DISTANCE_FIELD_BORDER_PIXELS as i32)
1474                            * scale;
1475                    let bitmap_top_pixels = baseline_y
1476                        - (f64::from(glyph.top)
1477                            + GLYPH_FONT_ASCENDER_PIXELS
1478                            + f64::from(GLYPH_SIGNED_DISTANCE_FIELD_BORDER_PIXELS))
1479                            * scale;
1480                    let bitmap_width_pixels =
1481                        f64::from(glyph.bitmap_rectangle.width_pixels) * scale;
1482                    let bitmap_height_pixels =
1483                        f64::from(glyph.bitmap_rectangle.height_pixels) * scale;
1484                    if label.halo_width_pixels > 0.0 {
1485                        // The halo keeps the text quad's geometry: its lower
1486                        // threshold is what spreads the outline, exactly as
1487                        // MapLibre Native's `symbol_sdf` shader does.
1488                        append_glyph_quad(
1489                            halo_vertices,
1490                            viewport,
1491                            [bitmap_left_pixels, bitmap_top_pixels],
1492                            [bitmap_width_pixels, bitmap_height_pixels],
1493                            glyph,
1494                            halo_threshold,
1495                            edge_gamma,
1496                            &label.halo_rgba,
1497                        );
1498                    }
1499                    append_glyph_quad(
1500                        text_vertices,
1501                        viewport,
1502                        [bitmap_left_pixels, bitmap_top_pixels],
1503                        [bitmap_width_pixels, bitmap_height_pixels],
1504                        glyph,
1505                        GLYPH_TEXT_SIGNED_DISTANCE_THRESHOLD,
1506                        edge_gamma,
1507                        &label.text_rgba,
1508                    );
1509                }
1510                pen_x += advance_pixels + label.letter_spacing_pixels;
1511            }
1512        }
1513    }

1515    /// Lays out one rotated label's glyph quads: the baseline runs along the
1516    /// label's rotation from a centered pen origin, and each glyph rectangle
1517    /// rotates about the label anchor before it becomes clip-space corners.
1518    fn append_rotated_label_quads(
1519        &self,
1520        halo_vertices: &mut Vec<ClipSpaceGlyphVertex>,
1521        text_vertices: &mut Vec<ClipSpaceGlyphVertex>,
1522        viewport: &ViewportPixels,
1523        label: &PlacedLabel,
1524    ) {
1525        let scale = label.text_size_pixels / GLYPH_FONT_EM_PIXELS;
1526        let style_text_size_pixels = label.text_size_pixels / self.device_pixel_ratio;
1527        let style_halo_width_pixels = label.halo_width_pixels / self.device_pixel_ratio;
1528        let edge_gamma = glyph_edge_gamma(style_text_size_pixels, self.device_pixel_ratio);
1529        let halo_threshold =
1530            glyph_halo_signed_distance_threshold(style_text_size_pixels, style_halo_width_pixels);
1531        let baseline_offset_pixels = label.text_size_pixels * LABEL_BASELINE_OFFSET_FRACTION;
1532        let line_count = label.lines.len() as f64;
1533        for (line_index, line) in label.lines.iter().enumerate() {
1534            // Line offsets and pen origins are authored in the label's own
1535            // unrotated frame; `append_rotated_glyph_quad` rotates the whole
1536            // offset about the anchor.
1537            let line_offset_pixels =
1538                (line_index as f64 - (line_count - 1.0) / 2.0) * label.line_height_pixels;
1539            let mut pen_offset_pixels = match label.text_justify {
1540                TextJustify::Left => -label.block_width_pixels / 2.0,
1541                TextJustify::Right => label.block_width_pixels / 2.0 - line.width_pixels,
1542                TextJustify::Auto | TextJustify::Center => -line.width_pixels / 2.0,
1543            };
1544            for character in line.text.chars() {
1545                let Some(glyph) = self.font.glyph(u32::from(character)) else {
1546                    continue;
1547                };
1548                let advance_pixels = f64::from(glyph.advance) * scale;
1549                if glyph.bitmap_rectangle.width_pixels > 0
1550                    && glyph.bitmap_rectangle.height_pixels > 0
1551                {
1552                    let bitmap_offset_pixels = [
1553                        pen_offset_pixels
1554                            + f64::from(
1555                                glyph.left - GLYPH_SIGNED_DISTANCE_FIELD_BORDER_PIXELS as i32,
1556                            ) * scale,
1557                        line_offset_pixels + baseline_offset_pixels
1558                            - (f64::from(glyph.top)
1559                                + GLYPH_FONT_ASCENDER_PIXELS
1560                                + f64::from(GLYPH_SIGNED_DISTANCE_FIELD_BORDER_PIXELS))
1561                                * scale,
1562                    ];
1563                    let bitmap_width_pixels =
1564                        f64::from(glyph.bitmap_rectangle.width_pixels) * scale;
1565                    let bitmap_height_pixels =
1566                        f64::from(glyph.bitmap_rectangle.height_pixels) * scale;
1567                    if label.halo_width_pixels > 0.0 {
1568                        append_rotated_glyph_quad(
1569                            halo_vertices,
1570                            viewport,
1571                            label.anchor_screen_pixels,
1572                            label.rotation_radians,
1573                            bitmap_offset_pixels,
1574                            [bitmap_width_pixels, bitmap_height_pixels],
1575                            glyph,
1576                            halo_threshold,
1577                            edge_gamma,
1578                            &label.halo_rgba,
1579                        );
1580                    }
1581                    append_rotated_glyph_quad(
1582                        text_vertices,
1583                        viewport,
1584                        label.anchor_screen_pixels,
1585                        label.rotation_radians,
1586                        bitmap_offset_pixels,
1587                        [bitmap_width_pixels, bitmap_height_pixels],
1588                        glyph,
1589                        GLYPH_TEXT_SIGNED_DISTANCE_THRESHOLD,
1590                        edge_gamma,
1591                        &label.text_rgba,
1592                    );
1593                }
1594                pen_offset_pixels += advance_pixels + label.letter_spacing_pixels;
1595            }
1596        }
1597    }

1652pub struct IconQuadVertices {
1653    /// The icon quads, each in `top-left, top-right, bottom-left,
1654    /// bottom-right` order.
1655    pub quads: Vec<[ClipSpaceIconVertex; 4]>,
1656}

1658impl IconQuadVertices {
1659    /// Whether the pass draws anything.
1660    pub fn is_empty(&self) -> bool {
1661        self.quads.is_empty()
1662    }
1663
1664    /// Serializes the quads into the vertex and index byte streams the
1665    /// renderer's textured-quad buffers expect: little-endian floats, four
1666    /// corners per quad, and the shared `0, 1, 2, 2, 1, 3` index pattern.
1667    pub fn encoded_buffers(&self) -> (Vec<u8>, Vec<u8>) {
1668        let vertex_stride = std::mem::size_of::<ClipSpaceIconVertex>();
1669        let mut vertex_bytes =
1670            Vec::with_capacity(self.quads.len() * vertex_stride * ICON_CORNERS_PER_QUAD);
1671        let mut index_bytes = Vec::with_capacity(
1672            self.quads.len() * ICON_INDICES_PER_QUAD * std::mem::size_of::<u16>(),
1673        );
1674        for (quad_index, quad) in self.quads.iter().enumerate() {
1675            let base_corner_index = u16::try_from(quad_index * ICON_CORNERS_PER_QUAD)
1676                .expect("the icon quad list is bounded by the label budget");
1677            for vertex in quad {
1678                for component in vertex
1679                    .clip_position
1680                    .iter()
1681                    .chain(vertex.texture_coordinates.iter())
1682                    .chain(std::iter::once(&vertex.opacity))
1683                {
1684                    vertex_bytes.extend_from_slice(&component.to_le_bytes());
1685                }
1686            }
1687            for index in [
1688                base_corner_index,
1689                base_corner_index + 1,
1690                base_corner_index + 2,
1691                base_corner_index + 2,
1692                base_corner_index + 1,
1693                base_corner_index + 3,
1694            ] {
1695                index_bytes.extend_from_slice(&index.to_le_bytes());
1696            }
1697        }
1698        (vertex_bytes, index_bytes)
1699    }

1728/// Projects and sizes one sprite icon at the candidate's anchor.

1734    camera: &MapCamera,
1735    viewport: &ViewportPixels,
1736    field_of_view_degrees: f64,
1737    anchor: &GeographicCoordinate,
1738    icon_candidate: &MapIconCandidate,
1739    sprite: Option<&SpriteAtlas>,

1774/// Places a line-placed label's run along the longest straight run of its
1775/// projected geometry, or reports an empty list when no run can carry it.
1776///
1777/// Vertices the camera cannot project break the polyline: runs are measured
1778/// over the stretches where every vertex projected. Within one stretch, a
1779/// straight run is the longest sequence of consecutive segments whose bearing
1780/// stays within [`LINE_LABEL_STRAIGHT_RUN_ANGLE_TOLERANCE_RADIANS`] of the
1781/// run's first segment (map data splits a street into short blocks, so a name
1782/// needs the block sequence, not one block). Of the stretches' runs the
1783/// longest wins (ties keep the earlier one).
1784///
1785/// The text repeats along that run the way the reference engines name a long
1786/// street: one label centers on the run when only one fits, otherwise labels
1787/// spread evenly from [`LABEL_COLLISION_PADDING_PIXELS`] past the first end to
1788/// the same distance before the last, with at least
1789/// [`LINE_LABEL_REPEAT_SPACING_PIXELS`] of clear space between neighbours.
1790/// Each label rotates to the bearing of the segment under its own anchor
1791/// (flipped past [`LINE_LABEL_UPRIGHT_ROTATION_LIMIT_RADIANS`] so the text
1792/// always reads upright), so a name follows the road's curve rather than one
1793/// chord across the whole run.
1794/// The outward direction one text anchor points, used to push a
1795/// variable-anchored label away from its symbol by the radial offset.
1796fn radial_offset_direction(text_anchor: TextAnchor) -> [f64; 2] {
1797    let diagonal = std::f64::consts::FRAC_1_SQRT_2;
1798    match text_anchor {
1799        TextAnchor::Center => [0.0, 0.0],
1800        TextAnchor::Left => [-1.0, 0.0],
1801        TextAnchor::Right => [1.0, 0.0],
1802        TextAnchor::Top => [0.0, -1.0],
1803        TextAnchor::Bottom => [0.0, 1.0],
1804        TextAnchor::TopLeft => [-diagonal, -diagonal],
1805        TextAnchor::TopRight => [diagonal, -diagonal],
1806        TextAnchor::BottomLeft => [-diagonal, diagonal],
1807        TextAnchor::BottomRight => [diagonal, diagonal],
1808    }
1809}

1811/// How far the block's center sits from the symbol anchor for one
1812/// `text-anchor`, in screen pixels.
1813fn text_anchor_center_offset(
1814    text_anchor: TextAnchor,
1815    block_width_pixels: f64,
1816    block_height_pixels: f64,
1817) -> [f64; 2] {
1818    let horizontal_shift_pixels = match text_anchor {
1819        TextAnchor::Left | TextAnchor::TopLeft | TextAnchor::BottomLeft => block_width_pixels / 2.0,
1820        TextAnchor::Right | TextAnchor::TopRight | TextAnchor::BottomRight => {
1821            -block_width_pixels / 2.0
1822        }
1823        TextAnchor::Center | TextAnchor::Top | TextAnchor::Bottom => 0.0,
1824    };
1825    let vertical_shift_pixels = match text_anchor {
1826        TextAnchor::Top | TextAnchor::TopLeft | TextAnchor::TopRight => block_height_pixels / 2.0,
1827        TextAnchor::Bottom | TextAnchor::BottomLeft | TextAnchor::BottomRight => {
1828            -block_height_pixels / 2.0
1829        }
1830        TextAnchor::Center | TextAnchor::Left | TextAnchor::Right => 0.0,
1831    };
1832    [horizontal_shift_pixels, vertical_shift_pixels]
1833}

1835/// The nearest rotation that reads upright: a label turned more than a
1836/// quarter turn is flipped by half a turn.
1837fn upright_rotation_radians(rotation_radians: f64) -> f64 {
1838    let mut upright = rotation_radians;
1839    while upright > std::f64::consts::FRAC_PI_2 {
1840        upright -= std::f64::consts::PI;
1841    }
1842    while upright < -std::f64::consts::FRAC_PI_2 {
1843        upright += std::f64::consts::PI;
1844    }
1845    upright
1846}

1848#[allow(clippy::too_many_arguments)]
1849fn label_placements_along_line(
1850    camera: &MapCamera,
1851    viewport: &ViewportPixels,
1852    field_of_view_degrees: f64,
1853    line_coordinates: &[GeographicCoordinate],
1854    text_width_pixels: f64,
1855    repeat_spacing_pixels: f64,
1856    line_center_only: bool,
1857    angle_tolerance_radians: f64,
1858) -> Vec<LabelPlacement> {
1859    let mut projected_stretch: Vec<[f64; 2]> = Vec::new();
1860    let mut longest_run: Option<StraightRun> = None;
1861    for coordinate in line_coordinates {
1862        let Some(screen_point) =
1863            camera.project_to_screen(coordinate, viewport, field_of_view_degrees)
1864        else {
1865            consider_longest_run(
1866                &projected_stretch,
1867                &mut longest_run,
1868                angle_tolerance_radians,
1869            );
1870            projected_stretch.clear();
1871            continue;
1872        };
1873        projected_stretch.push(screen_point);
1874    }
1875    consider_longest_run(
1876        &projected_stretch,
1877        &mut longest_run,
1878        angle_tolerance_radians,
1879    );
1880    let Some(run) = longest_run else {
1881        return Vec::new();
1882    };
1883
1884    let required_run_pixels = text_width_pixels + 2.0 * LABEL_COLLISION_PADDING_PIXELS;
1885    if run.length_pixels < required_run_pixels {
1886        return Vec::new();
1887    }
1888    let first_anchor_pixels = text_width_pixels / 2.0 + LABEL_COLLISION_PADDING_PIXELS;
1889    let last_anchor_pixels = run.length_pixels - first_anchor_pixels;
1890    let repeat_stride_pixels = text_width_pixels + repeat_spacing_pixels;
1891    let label_count = if line_center_only || last_anchor_pixels <= first_anchor_pixels {
1892        1
1893    } else {
1894        let fitting_spans = ((last_anchor_pixels - first_anchor_pixels) / repeat_stride_pixels)
1895            .floor()
1896            .max(0.0);
1897        (fitting_spans as usize).saturating_add(1)
1898    };
1899    let mut placements = Vec::with_capacity(label_count);
1900    for label_index in 0..label_count {
1901        let anchor_pixels = if label_count == 1 {
1902            run.length_pixels / 2.0
1903        } else {
1904            let fraction = label_index as f64 / (label_count - 1) as f64;
1905            first_anchor_pixels + (last_anchor_pixels - first_anchor_pixels) * fraction
1906        };
1907        let Some(placement) = placement_at_arc_length(&run.points, anchor_pixels) else {
1908            continue;
1909        };
1910        placements.push(placement);
1911    }
1912    placements
1913}

1915/// One straight run of a projected polyline: the screen points it spans and
1916/// its arc length.
1917#[derive(Debug, Clone)]
1918struct StraightRun {
1919    /// The run's projected vertices, in order.
1920    points: Vec<[f64; 2]>,
1921    /// The run's total length in screen pixels.
1922    length_pixels: f64,
1923}

1925/// The placement of one label centered at `anchor_pixels` along a projected
1926/// polyline, walking its segments from the start.
1927///
1928/// The anchor sits on the segment that contains the target distance and the
1929/// rotation follows that segment's own bearing, so a repeated name turns with
1930/// the road under it. Reports `None` only for a polyline with no segment of
1931/// positive length.
1932fn placement_at_arc_length(points: &[[f64; 2]], anchor_pixels: f64) -> Option<LabelPlacement> {
1933    let mut traveled_pixels = 0.0;
1934    let mut last_segment: Option<([f64; 2], [f64; 2])> = None;
1935    for segment_points in points.windows(2) {
1936        let [segment_start, segment_end] = [segment_points[0], segment_points[1]];
1937        let segment_length_pixels =
1938            (segment_end[0] - segment_start[0]).hypot(segment_end[1] - segment_start[1]);
1939        if segment_length_pixels <= 0.0 {
1940            continue;
1941        }
1942        last_segment = Some((segment_start, segment_end));
1943        if traveled_pixels + segment_length_pixels >= anchor_pixels {
1944            let fraction_pixels =
1945                ((anchor_pixels - traveled_pixels) / segment_length_pixels).clamp(0.0, 1.0);
1946            let anchor_screen_pixels = [
1947                segment_start[0] + (segment_end[0] - segment_start[0]) * fraction_pixels,
1948                segment_start[1] + (segment_end[1] - segment_start[1]) * fraction_pixels,
1949            ];
1950            return Some(LabelPlacement {
1951                anchor_screen_pixels,
1952                rotation_radians: upright_segment_rotation(segment_start, segment_end),
1953            });
1954        }
1955        traveled_pixels += segment_length_pixels;
1956    }
1957    let (segment_start, segment_end) = last_segment?;
1958    Some(LabelPlacement {
1959        anchor_screen_pixels: segment_end,
1960        rotation_radians: upright_segment_rotation(segment_start, segment_end),
1961    })
1962}

1964/// The baseline rotation for a segment, flipped by a half turn when it would
1965/// put the text upside down, so the result always lies in
1966/// `[-π/2, π/2]`.
1967fn upright_segment_rotation(segment_start: [f64; 2], segment_end: [f64; 2]) -> f64 {
1968    let segment_rotation_radians =
1969        (segment_end[1] - segment_start[1]).atan2(segment_end[0] - segment_start[0]);
1970    if segment_rotation_radians > LINE_LABEL_UPRIGHT_ROTATION_LIMIT_RADIANS {
1971        segment_rotation_radians - std::f64::consts::PI
1972    } else if segment_rotation_radians < -LINE_LABEL_UPRIGHT_ROTATION_LIMIT_RADIANS {
1973        segment_rotation_radians + std::f64::consts::PI
1974    } else {
1975        segment_rotation_radians
1976    }
1977}

1979/// Appends the longest straight run of one projected stretch to
1980/// `longest_run` when it beats the runs already measured. A stretch with fewer
1981/// than two points has no run.
1982fn consider_longest_run(
1983    projected_stretch: &[[f64; 2]],
1984    longest_run: &mut Option<StraightRun>,
1985    angle_tolerance_radians: f64,
1986) {
1987    let mut run_start_index = 0;
1988    while run_start_index + 1 < projected_stretch.len() {
1989        let reference_bearing = segment_bearing(
1990            projected_stretch[run_start_index],
1991            projected_stretch[run_start_index + 1],
1992        );
1993        let mut run_end_index = run_start_index;
1994        let mut run_length_pixels = 0.0;
1995        while run_end_index + 1 < projected_stretch.len() {
1996            let segment_bearing = segment_bearing(
1997                projected_stretch[run_end_index],
1998                projected_stretch[run_end_index + 1],
1999            );
2000            let bearing_difference = (segment_bearing - reference_bearing + std::f64::consts::PI)
2001                .rem_euclid(2.0 * std::f64::consts::PI)
2002                - std::f64::consts::PI;
2003            if bearing_difference.abs() > angle_tolerance_radians {
2004                break;
2005            }
2006            run_length_pixels += (projected_stretch[run_end_index + 1][0]
2007                - projected_stretch[run_end_index][0])
2008                .hypot(
2009                    projected_stretch[run_end_index + 1][1] - projected_stretch[run_end_index][1],
2010                );
2011            run_end_index += 1;
2012        }
2013        if run_end_index > run_start_index {
2014            let is_longest = match longest_run {
2015                None => true,
2016                Some(existing_run) => run_length_pixels > existing_run.length_pixels,
2017            };
2018            if is_longest {
2019                *longest_run = Some(StraightRun {
2020                    points: projected_stretch[run_start_index..=run_end_index].to_vec(),
2021                    length_pixels: run_length_pixels,
2022                });
2023            }
2024        }
2025        run_start_index = run_end_index.max(run_start_index + 1);
2026    }
2027}

2034/// One label's padded screen-space collision box, in its own oriented frame.
2035///
2036/// A point or line label contributes one box per drawn glyph, and an icon one
2037/// box of its own. The half extents already include the collision padding (or
2038/// the icon's `icon-padding`), so the padding travels with each box. The box
2039/// keeps the label's rotation and two boxes are tested with the
2040/// separating-axis theorem, so a rotated road name rejects with its oriented
2041/// extent instead of the empty corners of an axis-aligned bound.
2042#[derive(Debug, Clone, Copy)]
2043struct LabelScreenBox {
2044    /// The box center in viewport pixels (origin top-left, y down).
2045    center_screen_pixels: [f64; 2],
2046    /// Half the padded box width along the box's local x axis.
2047    half_width_pixels: f64,
2048    /// Half the padded box height along the box's local y axis.
2049    half_height_pixels: f64,
2050    /// The clockwise rotation of the box's local axes, in radians.
2051    rotation_radians: f64,
2052}
2053
2054impl LabelScreenBox {
2055    /// The collision box around a centered label's text block, padded by
2056    /// [`LABEL_COLLISION_PADDING_PIXELS`] on every side. Used only as the
2057    /// fallback for text whose glyphs produce no bitmap boxes (for example a
2058    /// name made of spaces), so the label still reserves its space.
2059    fn around(
2060        anchor_screen_pixels: [f64; 2],
2061        width_pixels: f64,
2062        height_pixels: f64,
2063        rotation_radians: f64,
2064    ) -> Self {
2065        Self::around_with_padding(
2066            anchor_screen_pixels,
2067            width_pixels,
2068            height_pixels,
2069            rotation_radians,
2070            LABEL_COLLISION_PADDING_PIXELS,
2071        )
2072    }
2073
2074    /// [`Self::around`] with a caller-declared padding: an icon carries its
2075    /// own `icon-padding`, which the specification measures in screen pixels.
2076    fn around_with_padding(
2077        anchor_screen_pixels: [f64; 2],
2078        width_pixels: f64,
2079        height_pixels: f64,
2080        rotation_radians: f64,
2081        padding_pixels: f64,
2082    ) -> Self {
2083        Self {
2084            center_screen_pixels: anchor_screen_pixels,
2085            half_width_pixels: width_pixels / 2.0 + padding_pixels,
2086            half_height_pixels: height_pixels / 2.0 + padding_pixels,
2087            rotation_radians,
2088        }
2089    }
2090
2091    /// The box's two unit axes in screen space: local x then local y.
2092    fn axes(&self) -> [[f64; 2]; 2] {
2093        let (rotation_sine, rotation_cosine) = self.rotation_radians.sin_cos();
2094        [
2095            [rotation_cosine, rotation_sine],
2096            [-rotation_sine, rotation_cosine],
2097        ]
2098    }
2099
2100    /// How far the box extends along a unit axis.
2101    fn projection_radius(&self, axis: [f64; 2]) -> f64 {
2102        let [x_axis, y_axis] = self.axes();
2103        self.half_width_pixels * (axis[0] * x_axis[0] + axis[1] * x_axis[1]).abs()
2104            + self.half_height_pixels * (axis[0] * y_axis[0] + axis[1] * y_axis[1]).abs()
2105    }
2106
2107    /// Whether the two oriented boxes share any area (the separating-axis
2108    /// theorem over their four candidate axes; touching edges do not count).
2109    fn intersects(&self, other: &Self) -> bool {
2110        let [self_x_axis, self_y_axis] = self.axes();
2111        let [other_x_axis, other_y_axis] = other.axes();
2112        let center_delta = [
2113            other.center_screen_pixels[0] - self.center_screen_pixels[0],
2114            other.center_screen_pixels[1] - self.center_screen_pixels[1],
2115        ];
2116        for axis in [self_x_axis, self_y_axis, other_x_axis, other_y_axis] {
2117            let center_distance = (center_delta[0] * axis[0] + center_delta[1] * axis[1]).abs();
2118            if center_distance >= self.projection_radius(axis) + other.projection_radius(axis) {
2119                return false;
2120            }
2121        }
2122        true
2123    }
2124
2125    /// The half extents of the box's axis-aligned screen bound.
2126    fn axis_aligned_half_extents(&self) -> [f64; 2] {
2127        let (rotation_sine, rotation_cosine) = self.rotation_radians.sin_cos();
2128        [
2129            self.half_width_pixels * rotation_cosine.abs()
2130                + self.half_height_pixels * rotation_sine.abs(),
2131            self.half_width_pixels * rotation_sine.abs()
2132                + self.half_height_pixels * rotation_cosine.abs(),
2133        ]
2134    }
2135
2136    /// Whether any part of the box is inside the viewport.
2137    fn intersects_viewport(&self, viewport: &ViewportPixels) -> bool {
2138        let [half_width_pixels, half_height_pixels] = self.axis_aligned_half_extents();
2139        self.center_screen_pixels[0] - half_width_pixels < viewport.width_pixels
2140            && self.center_screen_pixels[0] + half_width_pixels > 0.0
2141            && self.center_screen_pixels[1] - half_height_pixels < viewport.height_pixels
2142            && self.center_screen_pixels[1] + half_height_pixels > 0.0
2143    }
2144
2145    /// Whether the box lies entirely inside the viewport; a
2146    /// `symbol-avoid-edges` label must pass this instead of
2147    /// [`Self::intersects_viewport`]. The oriented box is stricter than its
2148    /// axis-aligned bound, so testing the bound never keeps an off-screen
2149    /// glyph.
2150    fn is_inside_viewport(&self, viewport: &ViewportPixels) -> bool {
2151        let [half_width_pixels, half_height_pixels] = self.axis_aligned_half_extents();
2152        self.center_screen_pixels[0] - half_width_pixels >= 0.0
2153            && self.center_screen_pixels[0] + half_width_pixels <= viewport.width_pixels
2154            && self.center_screen_pixels[1] - half_height_pixels >= 0.0
2155            && self.center_screen_pixels[1] + half_height_pixels <= viewport.height_pixels
2156    }
2157}

2176    font: &LabelFont,
2177    text_block: &LabelTextBlock,
2178    text_size_pixels: f64,
2179    letter_spacing_pixels: f64,
2180    text_justify: TextJustify,
2181) -> Vec<GlyphScreenRectangle> {
2182    let scale = text_size_pixels / GLYPH_FONT_EM_PIXELS;
2183    let baseline_offset_pixels = text_size_pixels * LABEL_BASELINE_OFFSET_FRACTION;
2184    let line_count = text_block.lines.len() as f64;
2185    let mut rectangles = Vec::new();
2186    for (line_index, line) in text_block.lines.iter().enumerate() {
2187        let line_offset_pixels =
2188            (line_index as f64 - (line_count - 1.0) / 2.0) * text_block.line_height_pixels;
2189        let mut pen_offset_pixels = match text_justify {
2190            TextJustify::Left => -text_block.width_pixels / 2.0,
2191            TextJustify::Right => text_block.width_pixels / 2.0 - line.width_pixels,
2192            TextJustify::Auto | TextJustify::Center => -line.width_pixels / 2.0,
2193        };
2194        for character in line.text.chars() {
2195            let Some(glyph) = font.glyph(u32::from(character)) else {
2196                continue;
2197            };
2198            let advance_pixels = f64::from(glyph.advance) * scale;
2199            if glyph.bitmap_rectangle.width_pixels > 0 && glyph.bitmap_rectangle.height_pixels > 0 {
2200                let bitmap_left_pixels = pen_offset_pixels
2201                    + f64::from(glyph.left - GLYPH_SIGNED_DISTANCE_FIELD_BORDER_PIXELS as i32)
2202                        * scale;
2203                let bitmap_top_pixels = line_offset_pixels + baseline_offset_pixels
2204                    - (f64::from(glyph.top)
2205                        + GLYPH_FONT_ASCENDER_PIXELS
2206                        + f64::from(GLYPH_SIGNED_DISTANCE_FIELD_BORDER_PIXELS))
2207                        * scale;
2208                let width_pixels = f64::from(glyph.bitmap_rectangle.width_pixels) * scale;
2209                let height_pixels = f64::from(glyph.bitmap_rectangle.height_pixels) * scale;
2210                rectangles.push(GlyphScreenRectangle {
2211                    center_offset_pixels: [
2212                        bitmap_left_pixels + width_pixels / 2.0,
2213                        bitmap_top_pixels + height_pixels / 2.0,
2214                    ],
2215                    width_pixels,
2216                    height_pixels,
2217                });
2218            }
2219            pen_offset_pixels += advance_pixels + letter_spacing_pixels;
2220        }
2221    }
2222    rectangles
2223}

2300/// Appends one rotated glyph bitmap: the glyph rectangle is laid out in the
2301/// label's own frame (offset from the anchor), rotated about the anchor by
2302/// `rotation_radians`, and emitted as the same two screen-space triangles the
2303/// unrotated path draws.
2304// The arguments mirror `append_glyph_quad`'s, plus the anchor and rotation;
2305// splitting them into a struct would only move the same values to the call
2306// site.
2307#[allow(clippy::too_many_arguments)]
2308fn append_rotated_glyph_quad(
2309    vertices: &mut Vec<ClipSpaceGlyphVertex>,
2310    viewport: &ViewportPixels,
2311    anchor_screen_pixels: [f64; 2],
2312    rotation_radians: f64,
2313    top_left_offset_pixels: [f64; 2],
2314    size_pixels: [f64; 2],
2315    glyph: &PackedGlyph,
2316    signed_distance_threshold: f32,
2317    signed_distance_gamma: f32,
2318    rgba_color: &[f32; 4],
2319) {

2321    let local_corners = axis_aligned_corners(top_left_offset_pixels, size_pixels);
2322    let rotated_corners = local_corners.map(|local_corner| {
2323        [
2324            anchor_screen_pixels[0] + local_corner[0] * rotation_cosine
2325                - local_corner[1] * rotation_sine,
2326            anchor_screen_pixels[1]
2327                + local_corner[0] * rotation_sine
2328                + local_corner[1] * rotation_cosine,
2329        ]
2330    });
2331    append_glyph_quad_corners(
2332        vertices,
2333        viewport,
2334        &rotated_corners,
2335        glyph.uv_rectangle,
2336        signed_distance_threshold,
2337        signed_distance_gamma,
2338        rgba_color,
2339    );

2342/// The four screen-space corners of an axis-aligned rectangle in the
2343/// `top-left, top-right, bottom-left, bottom-right` order the shared index
2344/// pattern consumes.
2345fn axis_aligned_corners(top_left_pixels: [f64; 2], size_pixels: [f64; 2]) -> [[f64; 2]; 4] {
2346    let right_pixels = top_left_pixels[0] + size_pixels[0];
2347    let bottom_pixels = top_left_pixels[1] + size_pixels[1];
2348    [
2349        [top_left_pixels[0], top_left_pixels[1]],
2350        [right_pixels, top_left_pixels[1]],
2351        [top_left_pixels[0], bottom_pixels],
2352        [right_pixels, bottom_pixels],
2353    ]
2354}

2356/// Emits one glyph bitmap from its four screen-space corners, in the shared
2357/// `top-left, top-right, bottom-left, bottom-right` order, as two triangles.
2358fn append_glyph_quad_corners(
2359    vertices: &mut Vec<ClipSpaceGlyphVertex>,
2360    viewport: &ViewportPixels,
2361    screen_corners: &[[f64; 2]; 4],
2362    texture_coordinates: rustic_maps_glyphs::GlyphUvRectangle,
2363    signed_distance_threshold: f32,
2364    signed_distance_gamma: f32,
2365    rgba_color: &[f32; 4],
2366) {
2367    let corner_texture_coordinates = [
2368        [texture_coordinates.minimum_u, texture_coordinates.minimum_v],
2369        [texture_coordinates.maximum_u, texture_coordinates.minimum_v],
2370        [texture_coordinates.minimum_u, texture_coordinates.maximum_v],
2371        [texture_coordinates.maximum_u, texture_coordinates.maximum_v],
2372    ];
2373    for corner_index in [0usize, 1, 2, 2, 1, 3] {
2374        let screen_position = screen_corners[corner_index];
2375        vertices.push(ClipSpaceGlyphVertex {
2376            clip_position: [
2377                (2.0 * screen_position[0] / viewport.width_pixels - 1.0) as f32,
2378                (1.0 - 2.0 * screen_position[1] / viewport.height_pixels) as f32,
2379                0.0,
2380                1.0,
2381            ],
2382            texture_coordinates: corner_texture_coordinates[corner_index],
2383            rgba_color: *rgba_color,
2384            signed_distance_threshold,
2385            signed_distance_gamma,
2386        });
2387    }
2388}

2599/// The compiled placement key as the candidate's whole-number priority. The
2600/// style specification allows a fractional sort key; the label pass orders by
2601/// an integer column, so a fractional key truncates toward zero and a negative
2602/// or oversized key clamps into the column's range.
2603fn symbol_sort_key_priority(sort_key: f64) -> u32 {
2604    if !sort_key.is_finite() || sort_key <= 0.0 {
2605        0
2606    } else if sort_key >= f64::from(u32::MAX) {
2607        u32::MAX
2608    } else {
2609        sort_key as u32
2610    }
2611}

2617        .attributes

2626            } else {
