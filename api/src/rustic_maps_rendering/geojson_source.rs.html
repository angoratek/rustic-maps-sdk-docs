1//! Deterministic GeoJSON tiling at the source boundary.
2//!
3//! A GeoJSON source is consumed through this module so downstream buckets,
4//! overlays and symbol placement see the same [`VectorTile`] feature model a
5//! vector source produces. The tiler projects each GeoJSON position through
6//! web mercator into tile-local extent coordinates, clips it to the tile
7//! rectangle, optionally simplifies it and quantizes it to the integer extent.
8//!
9//! # Tiling policy
10//!
11//! - **Projection**: longitude and latitude convert to normalized world
12//!   coordinates, then to the tile's local grid of [`DEFAULT_TILE_EXTENT`]
13//!   units.
14//! - **Clipping**: points outside the tile (plus the configured buffer) are
15//!   dropped; line segments are clipped to the rectangle; polygon rings are
16//!   clipped with the Sutherland–Hodgman algorithm and re-oriented so an
17//!   exterior ring winds positively and a hole negatively in the tile's
18//!   `(x right, y down)` grid.
19//! - **Simplification**: the Douglas–Peucker algorithm runs on the tile-local
20//!   float geometry with the configured tolerance. The default tolerance is
21//!   zero, so quantization to the integer extent is the only loss.
22//! - **Quantization**: every surviving coordinate rounds to the nearest
23//!   integer extent unit, consecutive duplicates collapse, rings with fewer
24//!   than three distinct vertices drop and line pieces with fewer than two
25//!   drop.
26//! - **Determinism**: features keep their input order, attributes sort by key,
27//!   polygon rings keep their source order with each exterior followed by its
28//!   holes, and a generated feature id is a fixed FNV-1a hash of the feature's
29//!   canonical JSON. The same collection and the same tile address therefore
30//!   always produce a byte-identical tile.
31//!
32//! The module is pure CPU and JSON: it never fetches, reads a clock or uses a
33//! hash map whose iteration order could leak into the output.
34

69/// How one GeoJSON source is tiled.
70#[derive(Debug, Clone, PartialEq)]
71pub struct GeoJsonTilingOptions {
72    /// The tile edge in extent units.
73    pub extent: u64,
74    /// The clipping buffer beyond the tile edge, in extent units.
75    pub buffer_pixels: f64,
76    /// The Douglas–Peucker tolerance in extent units; zero disables it.
77    pub simplification_tolerance: f64,
78    /// Whether a feature without an `id` gets a stable generated identifier.
79    pub generate_identifier: bool,
80    /// The name of the tile layer the tiler emits.
81    pub layer_name: String,
82}

85    fn default() -> Self {
86        Self {
87            extent: DEFAULT_GEOJSON_TILING_EXTENT,
88            buffer_pixels: DEFAULT_GEOJSON_BUFFER_PIXELS,
89            simplification_tolerance: DEFAULT_GEOJSON_SIMPLIFICATION_TOLERANCE,
90            generate_identifier: false,

119/// Tiles one GeoJSON value for one tile address.
120///
121/// # Parameters
122/// - `json_value` — a `FeatureCollection`, a single `Feature` or a bare
123///   geometry.
124/// - `tile_key` — the web-mercator address the geometry is clipped to.
125/// - `options` — extent, buffer, simplification and layer naming.
126///
127/// # Errors
128/// A value that is not a GeoJSON object, a missing or unsupported geometry
129/// type, or a malformed coordinate reports [`GeoJsonTilingError`]. A feature
130/// with no geometry is skipped, not an error.
131pub fn tile_geojson_for_tile(
132    json_value: &JsonValue,
133    tile_key: TileKey,
134    options: &GeoJsonTilingOptions,
135) -> Result<VectorTile, GeoJsonTilingError> {
136    let extent = options.extent.max(1);
137    let projection = TileProjection::new(tile_key, extent);
138    let clip_rectangle = ClipRectangle::new(extent, options.buffer_pixels.max(0.0));
139    let features = collect_features(json_value)?;
140    let mut tile_features = Vec::new();
141    for (feature_index, feature) in features.into_iter().enumerate() {
142        if let Some(tile_feature) = tile_feature(
143            feature,
144            feature_index,
145            &projection,
146            &clip_rectangle,
147            options,
148            extent,
149        )? {
150            tile_features.push(tile_feature);
151        }
152    }
153    Ok(VectorTile {
154        layers: vec![VectorTileLayer {
155            name: options.layer_name.clone(),
156            extent,
157            version: 2,
158            features: tile_features,
159        }],
160    })
161}

163/// Tiles one GeoJSON value for every requested tile address.
164///
165/// The map keeps the requested addresses' order and every tile is produced by
166/// [`tile_geojson_for_tile`], so the same collection and viewport always
167/// produce the same map.
168///
169/// # Errors
170/// The first tile that cannot be produced reports [`GeoJsonTilingError`].
171pub fn tile_geojson_for_tiles(
172    json_value: &JsonValue,
173    tile_keys: &[TileKey],
174    options: &GeoJsonTilingOptions,
175) -> Result<BTreeMap<TileKey, VectorTile>, GeoJsonTilingError> {
176    let mut tiles = BTreeMap::new();
177    for tile_key in tile_keys {
178        tiles.insert(
179            *tile_key,
180            tile_geojson_for_tile(json_value, *tile_key, options)?,
181        );
182    }
183    Ok(tiles)
184}

186/// One GeoJSON feature after collection, before projection.
187struct CollectedFeature<'a> {
188    identifier: Option<u64>,
189    properties: Vec<(String, VectorTileAttributeValue)>,
190    geometry_type: Option<&'a str>,
191    geometry: Option<&'a JsonValue>,
192    canonical_json: String,
193}
194
195/// Collects the features of a `FeatureCollection`, `Feature` or bare geometry.
196fn collect_features(
197    json_value: &JsonValue,
198) -> Result<Vec<CollectedFeature<'_>>, GeoJsonTilingError> {
199    let object = json_value
200        .as_object()
201        .ok_or_else(|| GeoJsonTilingError::new("the value must be a GeoJSON object"))?;
202    match object.get("type").and_then(JsonValue::as_str) {
203        Some("FeatureCollection") => {
204            let features = object
205                .get("features")
206                .and_then(JsonValue::as_array)
207                .ok_or_else(|| {
208                    GeoJsonTilingError::new("a FeatureCollection must carry a features array")
209                })?;
210            features.iter().map(collect_feature).collect()
211        }
212        Some("Feature") => Ok(vec![collect_feature(json_value)?]),
213        Some(geometry_type) => Ok(vec![CollectedFeature {
214            identifier: None,
215            properties: Vec::new(),
216            geometry_type: Some(geometry_type),
217            geometry: Some(json_value),
218            canonical_json: canonical_json(json_value),
219        }]),
220        None => Err(GeoJsonTilingError::new(
221            "a GeoJSON object must declare its type",
222        )),
223    }
224}

226/// Collects one `Feature` object.
227fn collect_feature(json_value: &JsonValue) -> Result<CollectedFeature<'_>, GeoJsonTilingError> {
228    let object = json_value
229        .as_object()
230        .ok_or_else(|| GeoJsonTilingError::new("a Feature must be an object"))?;
231    if object.get("type").and_then(JsonValue::as_str) != Some("Feature") {
232        return Err(GeoJsonTilingError::new(
233            "a FeatureCollection may contain only Feature objects",
234        ));
235    }
236    let properties = object
237        .get("properties")
238        .and_then(JsonValue::as_object)
239        .map(|property_object| {
240            let mut attributes: Vec<(String, VectorTileAttributeValue)> = property_object
241                .iter()
242                .map(|(name, value)| (name.clone(), attribute_value(value)))
243                .collect();
244            attributes.sort_by(|first, second| first.0.cmp(&second.0));
245            attributes
246        })
247        .unwrap_or_default();
248    let geometry = object
249        .get("geometry")
250        .filter(|geometry| !geometry.is_null());
251    let geometry_type = geometry
252        .and_then(|geometry| geometry.get("type"))
253        .and_then(JsonValue::as_str);
254    Ok(CollectedFeature {
255        identifier: object.get("id").and_then(JsonValue::as_u64),
256        properties,
257        geometry_type,
258        geometry,
259        canonical_json: canonical_json(json_value),
260    })
261}

263/// One JSON attribute as a vector-tile attribute. Objects and arrays keep
264/// their canonical JSON text, so their order is stable.
265fn attribute_value(json_value: &JsonValue) -> VectorTileAttributeValue {
266    match json_value {
267        JsonValue::String(text) => VectorTileAttributeValue::Text(text.clone()),
268        JsonValue::Bool(boolean) => VectorTileAttributeValue::Boolean(*boolean),
269        JsonValue::Number(number) => match number.as_i64() {
270            Some(integer) => VectorTileAttributeValue::Signed(integer),
271            None => VectorTileAttributeValue::Double(number.as_f64().unwrap_or_default()),
272        },
273        JsonValue::Null | JsonValue::Array(_) | JsonValue::Object(_) => {
274            VectorTileAttributeValue::Text(canonical_json(json_value))
275        }
276    }
277}

279/// Canonical JSON text: `serde_json`'s object map is key-sorted, so equal
280/// values always produce equal text.
281fn canonical_json(json_value: &JsonValue) -> String {
282    serde_json::to_string(json_value).unwrap_or_default()
283}

285/// The projected, clipped and quantized form of one feature.
286fn tile_feature(
287    feature: CollectedFeature<'_>,
288    feature_index: usize,
289    projection: &TileProjection,
290    clip_rectangle: &ClipRectangle,
291    options: &GeoJsonTilingOptions,
292    extent: u64,
293) -> Result<Option<VectorTileFeature>, GeoJsonTilingError> {
294    let Some(geometry) = feature.geometry else {
295        return Ok(None);
296    };
297    let geometry_type = feature.geometry_type.ok_or_else(|| {
298        GeoJsonTilingError::new(format!(
299            "feature {feature_index} carries a geometry without a type"
300        ))
301    })?;
302    let coordinates = geometry.get("coordinates").ok_or_else(|| {
303        GeoJsonTilingError::new(format!(
304            "feature {feature_index} carries a {geometry_type} without coordinates"
305        ))
306    })?;
307    let identifier = feature.identifier.or_else(|| {
308        options
309            .generate_identifier
310            .then(|| stable_identifier(&feature.canonical_json))
311    });
312    let (geometry_type, geometry) = match geometry_type {
313        "Point" => {
314            let point = project_point(coordinates, projection)?;
315            let points = point
316                .filter(|point| clip_rectangle.contains(*point))
317                .map(|point| vec![quantized_point(point, extent)])
318                .unwrap_or_default();
319            (
320                VectorTileGeometryType::Point,
321                VectorTileGeometry::Points(points),
322            )
323        }
324        "MultiPoint" => {
325            let coordinates = coordinates
326                .as_array()
327                .ok_or_else(|| GeoJsonTilingError::new("a MultiPoint needs an array"))?;
328            let mut points = Vec::new();
329            for coordinate in coordinates {
330                if let Some(point) = project_point(coordinate, projection)? {
331                    if clip_rectangle.contains(point) {
332                        points.push(quantized_point(point, extent));
333                    }
334                }
335            }
336            (
337                VectorTileGeometryType::Point,
338                VectorTileGeometry::Points(points),
339            )
340        }
341        "LineString" => {
342            let line = project_line_string(coordinates, projection)?;
343            let line_strings = clip_rectangle
344                .clip_line_string(&line, options.simplification_tolerance)
345                .into_iter()
346                .map(|line_string| quantized_line_string(&line_string, extent))
347                .filter(|line_string| line_string.len() >= 2)
348                .collect();
349            (
350                VectorTileGeometryType::LineString,
351                VectorTileGeometry::LineStrings(line_strings),
352            )
353        }
354        "MultiLineString" => {
355            let line_strings_value = coordinates
356                .as_array()
357                .ok_or_else(|| GeoJsonTilingError::new("a MultiLineString needs an array"))?;
358            let mut line_strings = Vec::new();
359            for line_string_value in line_strings_value {
360                let line = project_line_string(line_string_value, projection)?;
361                for clipped_line in
362                    clip_rectangle.clip_line_string(&line, options.simplification_tolerance)
363                {
364                    let quantized = quantized_line_string(&clipped_line, extent);
365                    if quantized.len() >= 2 {
366                        line_strings.push(quantized);
367                    }
368                }
369            }
370            (
371                VectorTileGeometryType::LineString,
372                VectorTileGeometry::LineStrings(line_strings),
373            )
374        }
375        "Polygon" => {
376            let rings = project_polygon(coordinates, projection)?;
377            let polygons = clip_rectangle
378                .clip_polygon(&rings, options.simplification_tolerance)
379                .map(|polygon| quantized_polygon(&polygon, extent))
380                .unwrap_or_default();
381            (
382                VectorTileGeometryType::Polygon,
383                VectorTileGeometry::Polygons(polygons),
384            )
385        }
386        "MultiPolygon" => {
387            let polygons_value = coordinates
388                .as_array()
389                .ok_or_else(|| GeoJsonTilingError::new("a MultiPolygon needs an array"))?;
390            let mut rings = Vec::new();
391            for polygon_value in polygons_value {
392                let polygon_rings = project_polygon(polygon_value, projection)?;
393                if let Some(clipped) =
394                    clip_rectangle.clip_polygon(&polygon_rings, options.simplification_tolerance)
395                {
396                    rings.extend(quantized_polygon(&clipped, extent));
397                }
398            }
399            (
400                VectorTileGeometryType::Polygon,
401                VectorTileGeometry::Polygons(rings),
402            )
403        }
404        other => {
405            return Err(GeoJsonTilingError::new(format!(
406                "unsupported GeoJSON geometry type {other:?}"
407            )));
408        }
409    };
410    if geometry_is_empty(&geometry) {
411        // A feature clipped entirely out of this tile contributes nothing; it
412        // must not surface as an empty geometry the renderer would walk.
413        return Ok(None);
414    }
415    Ok(Some(VectorTileFeature {
416        identifier,
417        geometry_type,
418        geometry,
419        attributes: feature.properties,
420    }))
421}

423/// Whether a clipped geometry carries no drawable primitive.
424fn geometry_is_empty(geometry: &VectorTileGeometry) -> bool {
425    match geometry {
426        VectorTileGeometry::Points(points) => points.is_empty(),
427        VectorTileGeometry::LineStrings(line_strings) => line_strings.is_empty(),
428        VectorTileGeometry::Polygons(rings) => rings.is_empty(),
429    }
430}

432/// The web-mercator projection of one tile address into local extent units.
433struct TileProjection {
434    /// The reciprocal of the world span covered by one tile.
435    tiles_per_world: f64,
436    /// The tile's column.
437    tile_x: u32,
438    /// The tile's row.
439    tile_y: u32,
440    /// The tile edge in extent units.
441    extent: f64,
442}
443
444impl TileProjection {
445    fn new(tile_key: TileKey, extent: u64) -> Self {
446        Self {
447            tiles_per_world: 1u64
448                .checked_shl(u32::from(tile_key.zoom_level))
449                .unwrap_or(u64::MAX) as f64,
450            tile_x: tile_key.tile_x,
451            tile_y: tile_key.tile_y,
452            extent: extent as f64,
453        }
454    }
455
456    /// Projects a GeoJSON position to tile-local extent units.
457    fn project(&self, position: &JsonValue) -> Result<Option<Point>, GeoJsonTilingError> {
458        let coordinates = position
459            .as_array()
460            .ok_or_else(|| GeoJsonTilingError::new("a position must be an array"))?;
461        let longitude = coordinates
462            .first()
463            .and_then(JsonValue::as_f64)
464            .ok_or_else(|| GeoJsonTilingError::new("a position needs a numeric longitude"))?;
465        let latitude = coordinates
466            .get(1)
467            .and_then(JsonValue::as_f64)
468            .ok_or_else(|| GeoJsonTilingError::new("a position needs a numeric latitude"))?;
469        if !longitude.is_finite() || !latitude.is_finite() {
470            return Ok(None);
471        }
472        let world_x = (longitude + 180.0) / 360.0;
473        let clamped_latitude = latitude.clamp(
474            -GEOJSON_MAXIMUM_MERCATOR_LATITUDE_DEGREES,
475            GEOJSON_MAXIMUM_MERCATOR_LATITUDE_DEGREES,
476        );
477        let latitude_radians = clamped_latitude.to_radians();
478        let world_y =
479            (1.0 - (latitude_radians.tan() + 1.0 / latitude_radians.cos()).ln() / PI) / 2.0;
480        Ok(Some(Point {
481            x: (world_x * self.tiles_per_world - f64::from(self.tile_x)) * self.extent,
482            y: (world_y * self.tiles_per_world - f64::from(self.tile_y)) * self.extent,
483        }))
484    }
485}

487/// Projects one position; a non-array reports an error.
488fn project_point(
489    position: &JsonValue,
490    projection: &TileProjection,
491) -> Result<Option<Point>, GeoJsonTilingError> {
492    projection.project(position)
493}

495/// Projects a LineString's coordinates, dropping non-finite positions.
496fn project_line_string(
497    coordinates: &JsonValue,
498    projection: &TileProjection,
499) -> Result<Vec<Point>, GeoJsonTilingError> {
500    let positions = coordinates
501        .as_array()
502        .ok_or_else(|| GeoJsonTilingError::new("a LineString needs an array of positions"))?;
503    let mut line = Vec::with_capacity(positions.len());
504    for position in positions {
505        if let Some(point) = projection.project(position)? {
506            line.push(point);
507        }
508    }
509    Ok(line)
510}

512/// Projects a Polygon's rings; the first ring is exterior and the rest are
513/// holes. Positions that cannot project are dropped.
514fn project_polygon(
515    coordinates: &JsonValue,
516    projection: &TileProjection,
517) -> Result<Vec<Vec<Point>>, GeoJsonTilingError> {
518    let rings = coordinates
519        .as_array()
520        .ok_or_else(|| GeoJsonTilingError::new("a Polygon needs an array of rings"))?;
521    let mut projected_rings = Vec::with_capacity(rings.len());
522    for ring in rings {
523        let mut projected_ring = project_line_string(ring, projection)?;
524        remove_closing_vertex(&mut projected_ring);
525        projected_rings.push(projected_ring);
526    }
527    Ok(projected_rings)
528}

530/// A tile-local point in float extent units.
531#[derive(Debug, Clone, Copy, PartialEq)]
532struct Point {
533    x: f64,
534    y: f64,
535}
536
537/// The tile rectangle, optionally grown by a buffer.
538struct ClipRectangle {
539    minimum_x: f64,
540    minimum_y: f64,
541    maximum_x: f64,
542    maximum_y: f64,
543}
544
545impl ClipRectangle {
546    fn new(extent: u64, buffer_pixels: f64) -> Self {
547        let extent = extent as f64;
548        Self {
549            minimum_x: -buffer_pixels,
550            minimum_y: -buffer_pixels,
551            maximum_x: extent + buffer_pixels,
552            maximum_y: extent + buffer_pixels,
553        }
554    }
555
556    fn contains(&self, point: Point) -> bool {
557        point.x >= self.minimum_x
558            && point.x <= self.maximum_x
559            && point.y >= self.minimum_y
560            && point.y <= self.maximum_y
561    }
562
563    /// Clips one open polyline into the pieces that lie inside the rectangle.
564    ///
565    /// Each inside run becomes one piece; each segment crossing the boundary
566    /// gains its two exact intersection points, so adjacent tiles agree on the
567    /// boundary coordinate. A piece with fewer than two points is dropped.
568    fn clip_line_string(&self, line: &[Point], simplification_tolerance: f64) -> Vec<Vec<Point>> {
569        let simplified = simplify_open_line(line, simplification_tolerance);
570        let mut pieces = Vec::new();
571        let mut current_piece: Vec<Point> = Vec::new();
572        for window in simplified.windows(2) {
573            let start = window[0];
574            let end = window[1];
575            let Some((clipped_start, clipped_end)) = self.clip_segment(start, end) else {
576                if current_piece.len() >= 2 {
577                    pieces.push(std::mem::take(&mut current_piece));
578                } else {
579                    current_piece.clear();
580                }
581                continue;
582            };
583            let continues_current_piece = current_piece
584                .last()
585                .is_some_and(|last| last.x == clipped_start.x && last.y == clipped_start.y);
586            if !continues_current_piece {
587                if current_piece.len() >= 2 {
588                    pieces.push(std::mem::take(&mut current_piece));
589                } else {
590                    current_piece.clear();
591                }
592                current_piece.push(clipped_start);
593            }
594            current_piece.push(clipped_end);
595            let segment_leaves_rectangle = clipped_end.x != end.x || clipped_end.y != end.y;
596            if segment_leaves_rectangle {
597                if current_piece.len() >= 2 {
598                    pieces.push(std::mem::take(&mut current_piece));
599                } else {
600                    current_piece.clear();
601                }
602            }
603        }
604        if current_piece.len() >= 2 {
605            pieces.push(current_piece);
606        }
607        pieces
608    }
609
610    /// Clips a polygon (exterior first, then holes) to the rectangle.
611    ///
612    /// A clipped exterior with fewer than three distinct vertices drops the
613    /// whole polygon. A hole is dropped when it no longer intersects the
614    /// rectangle.
615    fn clip_polygon(
616        &self,
617        rings: &[Vec<Point>],
618        simplification_tolerance: f64,
619    ) -> Option<Vec<Vec<Point>>> {
620        let exterior = rings.first()?;
621        let simplified_exterior = simplify_closed_ring(exterior, simplification_tolerance);
622        let clipped_exterior = self.clip_ring_against_rectangle(&simplified_exterior);
623        if distinct_vertex_count(&clipped_exterior) < 3 {
624            return None;
625        }
626        let mut clipped_rings = vec![clipped_exterior];
627        for hole in rings.iter().skip(1) {
628            let simplified_hole = simplify_closed_ring(hole, simplification_tolerance);
629            let clipped_hole = self.clip_ring_against_rectangle(&simplified_hole);
630            if distinct_vertex_count(&clipped_hole) >= 3 {
631                clipped_rings.push(clipped_hole);
632            }
633        }
634        Some(clipped_rings)
635    }
636
637    /// Sutherland–Hodgman against the four rectangle edges.
638    fn clip_ring_against_rectangle(&self, ring: &[Point]) -> Vec<Point> {
639        if ring.is_empty() {
640            return Vec::new();
641        }
642        let edges: [(Point, Point); 4] = [
643            (
644                Point {
645                    x: self.minimum_x,
646                    y: self.minimum_y,
647                },
648                Point {
649                    x: self.maximum_x,
650                    y: self.minimum_y,
651                },
652            ),
653            (
654                Point {
655                    x: self.maximum_x,
656                    y: self.minimum_y,
657                },
658                Point {
659                    x: self.maximum_x,
660                    y: self.maximum_y,
661                },
662            ),
663            (
664                Point {
665                    x: self.maximum_x,
666                    y: self.maximum_y,
667                },
668                Point {
669                    x: self.minimum_x,
670                    y: self.maximum_y,
671                },
672            ),
673            (
674                Point {
675                    x: self.minimum_x,
676                    y: self.maximum_y,
677                },
678                Point {
679                    x: self.minimum_x,
680                    y: self.minimum_y,
681                },
682            ),
683        ];
684        let mut clipped = ring.to_vec();
685        for edge in edges {
686            if clipped.is_empty() {
687                break;
688            }
689            clipped = clip_ring_against_edge(&clipped, edge.0, edge.1);
690        }
691        clipped
692    }
693
694    /// Liang–Barsky clipping of one segment; `None` when it misses the
695    /// rectangle.
696    fn clip_segment(&self, start: Point, end: Point) -> Option<(Point, Point)> {
697        let delta_x = end.x - start.x;
698        let delta_y = end.y - start.y;
699        let boundaries = [
700            (-delta_x, start.x - self.minimum_x),
701            (delta_x, self.maximum_x - start.x),
702            (-delta_y, start.y - self.minimum_y),
703            (delta_y, self.maximum_y - start.y),
704        ];
705        let mut entering = 0.0_f64;
706        let mut leaving = 1.0_f64;
707        for (boundary_direction, boundary_distance) in boundaries {
708            if boundary_direction == 0.0 {
709                if boundary_distance < 0.0 {
710                    return None;
711                }
712                continue;
713            }
714            let ratio = boundary_distance / boundary_direction;
715            if boundary_direction < 0.0 {
716                entering = entering.max(ratio);
717            } else {
718                leaving = leaving.min(ratio);
719            }
720            if entering > leaving {
721                return None;
722            }
723        }
724        Some((
725            Point {
726                x: start.x + entering * delta_x,
727                y: start.y + entering * delta_y,
728            },
729            Point {
730                x: start.x + leaving * delta_x,
731                y: start.y + leaving * delta_y,
732            },
733        ))
734    }
735}

737/// Clips a ring against one directed edge, keeping the points to its left.
738fn clip_ring_against_edge(ring: &[Point], edge_start: Point, edge_end: Point) -> Vec<Point> {
739    let mut output = Vec::with_capacity(ring.len() + 2);
740    for (index, current) in ring.iter().enumerate() {
741        let previous = ring[(index + ring.len() - 1) % ring.len()];
742        let current_inside = is_inside_edge(*current, edge_start, edge_end);
743        let previous_inside = is_inside_edge(previous, edge_start, edge_end);
744        if current_inside {
745            if !previous_inside {
746                if let Some(intersection) =
747                    segment_intersection(previous, *current, edge_start, edge_end)
748                {
749                    output.push(intersection);
750                }
751            }
752            output.push(*current);
753        } else if previous_inside {
754            if let Some(intersection) =
755                segment_intersection(previous, *current, edge_start, edge_end)
756            {
757                output.push(intersection);
758            }
759        }
760    }
761    output
762}

764/// Whether a point lies on the left side (or on) the directed edge.
765fn is_inside_edge(point: Point, edge_start: Point, edge_end: Point) -> bool {
766    (edge_end.x - edge_start.x) * (point.y - edge_start.y)
767        - (edge_end.y - edge_start.y) * (point.x - edge_start.x)
768        >= 0.0
769}

771/// The intersection of two segments, or `None` when they are parallel.
772fn segment_intersection(
773    first_start: Point,
774    first_end: Point,
775    second_start: Point,
776    second_end: Point,
777) -> Option<Point> {
778    let first_delta = Point {
779        x: first_end.x - first_start.x,
780        y: first_end.y - first_start.y,
781    };
782    let second_delta = Point {
783        x: second_end.x - second_start.x,
784        y: second_end.y - second_start.y,
785    };
786    let denominator = first_delta.x * second_delta.y - first_delta.y * second_delta.x;
787    if denominator.abs() <= f64::EPSILON {
788        return None;
789    }
790    let difference = Point {
791        x: second_start.x - first_start.x,
792        y: second_start.y - first_start.y,
793    };
794    let ratio = (difference.x * second_delta.y - difference.y * second_delta.x) / denominator;
795    Some(Point {
796        x: first_start.x + ratio * first_delta.x,
797        y: first_start.y + ratio * first_delta.y,
798    })
799}

801/// Removes a repeated closing vertex from a projected ring.
802fn remove_closing_vertex(ring: &mut Vec<Point>) {
803    if ring.len() >= 2 {
804        let first = ring[0];
805        let last = ring[ring.len() - 1];
806        if first.x == last.x && first.y == last.y {
807            ring.pop();
808        }
809    }
810}

812/// The number of distinct vertices a ring carries.
813fn distinct_vertex_count(ring: &[Point]) -> usize {
814    let mut distinct = Vec::new();
815    for point in ring {
816        if !distinct
817            .iter()
818            .any(|existing: &Point| existing.x == point.x && existing.y == point.y)
819        {
820            distinct.push(*point);
821        }
822    }
823    distinct.len()
824}

826/// Simplifies an open polyline with Douglas–Peucker. A non-positive tolerance
827/// returns the input unchanged; an output shorter than two points returns the
828/// input, so a degenerate tolerance can never erase a line.
829fn simplify_open_line(line: &[Point], tolerance: f64) -> Vec<Point> {
830    if line.len() <= 2 || tolerance <= 0.0 {
831        return line.to_vec();
832    }
833    let mut keep = vec![false; line.len()];
834    keep[0] = true;
835    keep[line.len() - 1] = true;
836    let mut ranges = vec![(0, line.len() - 1)];
837    while let Some((start, end)) = ranges.pop() {
838        let mut farthest_distance = 0.0;
839        let mut farthest_index = start;
840        for index in (start + 1)..end {
841            let distance = perpendicular_distance(line[index], line[start], line[end]);
842            if distance > farthest_distance {
843                farthest_distance = distance;
844                farthest_index = index;
845            }
846        }
847        if farthest_distance > tolerance {
848            keep[farthest_index] = true;
849            ranges.push((start, farthest_index));
850            ranges.push((farthest_index, end));
851        }
852    }
853    let simplified: Vec<Point> = line
854        .iter()
855        .zip(keep)
856        .filter_map(|(point, keep_point)| keep_point.then_some(*point))
857        .collect();
858    if simplified.len() >= 2 {
859        simplified
860    } else {
861        line.to_vec()
862    }
863}
864
865/// Simplifies a closed ring, keeping the first vertex and the closing edge.
866fn simplify_closed_ring(ring: &[Point], tolerance: f64) -> Vec<Point> {
867    if ring.len() <= 3 || tolerance <= 0.0 {
868        return ring.to_vec();
869    }
870    let mut closed = ring.to_vec();
871    closed.push(ring[0]);
872    let mut simplified = simplify_open_line(&closed, tolerance);
873    simplified.pop();
874    if simplified.len() >= 3 {
875        simplified
876    } else {
877        ring.to_vec()
878    }
879}

881/// The distance from a point to the infinite line through two points.
882fn perpendicular_distance(point: Point, start: Point, end: Point) -> f64 {
883    let delta_x = end.x - start.x;
884    let delta_y = end.y - start.y;
885    let length = (delta_x * delta_x + delta_y * delta_y).sqrt();
886    if length == 0.0 {
887        return ((point.x - start.x).powi(2) + (point.y - start.y).powi(2)).sqrt();
888    }
889    ((delta_x * (start.y - point.y) - (start.x - point.x) * delta_y) / length).abs()
890}

892/// Quantizes one point to the integer extent grid.
893fn quantized_point(point: Point, extent: u64) -> VectorTileGeometryPoint {
894    let maximum = i64::try_from(extent).unwrap_or(i64::MAX);
895    VectorTileGeometryPoint {
896        x_coordinate: quantize(point.x, maximum),
897        y_coordinate: quantize(point.y, maximum),
898    }
899}
900
901/// Rounds one coordinate to the nearest integer, dropping consecutive
902/// duplicates.
903fn quantized_line_string(line: &[Point], extent: u64) -> Vec<VectorTileGeometryPoint> {
904    let mut quantized: Vec<VectorTileGeometryPoint> = Vec::with_capacity(line.len());
905    for point in line {
906        let next = quantized_point(*point, extent);
907        if quantized.last() != Some(&next) {
908            quantized.push(next);
909        }
910    }
911    quantized
912}

914/// Quantizes one polygon's rings, orienting the exterior positively and every
915/// hole negatively in the tile's `(x right, y down)` grid. A ring that
916/// degenerates below three distinct vertices is dropped; if the exterior
917/// drops, the whole polygon drops.
918fn quantized_polygon(polygon: &[Vec<Point>], extent: u64) -> Vec<VectorTileLinearRing> {
919    let Some(exterior) = polygon.first() else {
920        return Vec::new();
921    };
922    let Some(exterior_ring) = oriented_ring(exterior, extent, true) else {
923        return Vec::new();
924    };
925    let mut rings = vec![exterior_ring];
926    for hole in polygon.iter().skip(1) {
927        if let Some(hole_ring) = oriented_ring(hole, extent, false) {
928            rings.push(hole_ring);
929        }
930    }
931    rings
932}
933
934/// Quantizes one ring and orients it for its role.
935fn oriented_ring(ring: &[Point], extent: u64, exterior: bool) -> Option<VectorTileLinearRing> {
936    let mut vertices = quantized_line_string(ring, extent);
937    if vertices.len() >= 2 && vertices.first() == vertices.last() {
938        vertices.pop();
939    }
940    if distinct_quantized_vertex_count(&vertices) < 3 {
941        return None;
942    }
943    let is_positive = ring_signed_area(&vertices) > 0.0;
944    if is_positive != exterior {
945        vertices.reverse();
946    }
947    Some(VectorTileLinearRing { vertices })
948}

950/// The number of distinct integer vertices.
951fn distinct_quantized_vertex_count(vertices: &[VectorTileGeometryPoint]) -> usize {
952    let mut distinct = Vec::new();
953    for vertex in vertices {
954        if !distinct.contains(vertex) {
955            distinct.push(*vertex);
956        }
957    }
958    distinct.len()
959}

961/// The ring's doubled signed area in the tile's y-down grid.
962fn ring_signed_area(vertices: &[VectorTileGeometryPoint]) -> f64 {
963    let mut doubled_area = 0.0;
964    for index in 0..vertices.len() {
965        let current = vertices[index];
966        let next = vertices[(index + 1) % vertices.len()];
967        doubled_area += current.x_coordinate as f64 * next.y_coordinate as f64
968            - next.x_coordinate as f64 * current.y_coordinate as f64;
969    }
970    doubled_area
971}

973/// Rounds to the nearest integer and clamps into `[-maximum, maximum]`.
974fn quantize(coordinate: f64, maximum: i64) -> i64 {
975    if coordinate.is_nan() {
976        return 0;
977    }
978    coordinate.round().clamp(-(maximum as f64), maximum as f64) as i64
979}

981/// A stable FNV-1a identifier for a feature without an `id`.
982fn stable_identifier(canonical_json: &str) -> u64 {
983    let mut hash = FNV_OFFSET_BASIS;
984    for byte in canonical_json.as_bytes() {
985        hash ^= u64::from(*byte);
986        hash = hash.wrapping_mul(FNV_PRIME);
987    }
988    hash
989}
