1//! Vector-tile styling: a decoded [`rustic_maps_vector_tiles::VectorTile`] becomes
2//! the renderer's overlay primitives under a compiled style plan.
3//!
4//! The style document and its expressions live in `rustic-maps-style`; this
5//! module is the renderer's consumer. [`vector_tile_style_from_id`] parses a
6//! committed version-8 style document and compiles it once into a
7//! [`CompiledStyle`](rustic_maps_style::CompiledStyle). Each frame then resolves
8//! the compiled plan at the camera's fractional zoom, so paint expressions
9//! change continuously with the camera without re-parsing the document.
10//!
11//! Polygon features paint every exterior ring with its interior rings
12//! subtracted through the overlay's hole-aware fill (see
13//! [`crate::overlay::FilledPolygonOverlay`]); line features become styled
14//! polylines and point features become screen-facing markers. Primitives are
15//! grouped into one [`VectorOverlayBatch`] per style layer and one bucket per
16//! layer in [`vector_tile_overlay_layers_at_zoom`], so a frame can paint every
17//! tile's layer 0 before any tile's layer 1: global style order across the
18//! visible tile set.
19//!
20//! Coordinates convert from tile-local extent space (`x` right, `y` down, the
21//! MVT convention) through the tile's web-mercator address to geographic
22//! degrees, so the overlays project exactly like every other map coordinate.
23//!
24//! Symbol layers paint no overlay primitive: their labels are data for the
25//! frame-level label pass ([`crate::labels`]), which reads the same compiled
26//! plan, filter and layout.
27

154/// Parses and compiles the built-in style document a style id names.
155///
156/// # Parameters
157/// - `style_id` — one of the exported `VECTOR_STYLE_ID_*` identifiers.
158///
159/// # Returns
160/// The compiled style, ready to resolve at a frame zoom and feature.
161///
162/// # Errors
163/// An unknown id and a malformed document both report the typed artifact
164/// error rather than a guess.
165pub fn vector_tile_style_from_id(
166    style_id: &str,
167) -> Result<CompiledStyle, VectorStyleArtifactError> {
168    let Some(style_json) = vector_style_document_json(style_id) else {
169        return Err(VectorStyleArtifactError {
170            detail: format!(
171                "no committed vector style carries the id {style_id:?}; known ids are \
172                 {VECTOR_STYLE_ID_CARTO_LIGHT}, {VECTOR_STYLE_ID_CARTO_DARK}, \
173                 {VECTOR_STYLE_ID_OPENMAPTILES_LIGHT}, {VECTOR_STYLE_ID_OPENMAPTILES_DARK}, \
174                 {VECTOR_STYLE_ID_BRIGHT_LIGHT}, {VECTOR_STYLE_ID_FIORD_DARK}, \
175                 {VECTOR_STYLE_ID_POSITRON_LIGHT}, {VECTOR_STYLE_ID_LABEL_COVERAGE_PROBE}"
176            ),
177        });
178    };
179    vector_tile_style_from_json(style_json)
180}

197/// Parses and compiles a version-8 style document.
198///
199/// The compile step is infallible; only a malformed document reports an error.
200pub fn vector_tile_style_from_json(
201    style_json: &str,
202) -> Result<CompiledStyle, VectorStyleArtifactError> {
203    let document = MapStyleDocument::parse_json(style_json).map_err(|style_error| {
204        VectorStyleArtifactError {
205            detail: style_error.to_string(),
206        }
207    })?;
208    Ok(compile_style_document(&document))
209}

216/// Compiles a style document, falling back to the committed default document
217/// when the supplied one is malformed.
218///
219/// The default document is itself compiled at most once per call and is a
220/// compile-time constant, so this cannot loop; an empty style is the last
221/// resort if the committed default is ever removed from the build.
222pub fn vector_tile_style_or_fallback(style_json: &str) -> CompiledStyle {
223    vector_tile_style_from_json(style_json).unwrap_or_else(|_| {
224        vector_tile_style_from_json(DEFAULT_VECTOR_STYLE_DOCUMENT_JSON)
225            .unwrap_or_else(|_| empty_compiled_style())
226    })
227}

243/// Converts one tile-local extent point into a geographic coordinate through
244/// the tile's web-mercator address. Vertices at the extent boundary are legal
245/// (they land on the next tile's edge); the latitude clamp keeps the projection
246/// finite at the world's polar edges.
247pub fn tile_local_point_to_coordinate(
248    tile_address: VectorTileAddress,
249    layer_extent: u64,
250    point: &VectorTileGeometryPoint,
251) -> GeographicCoordinate {
252    let extent = layer_extent.max(1) as f64;
253    let tiles_per_axis = (1u64 << tile_address.zoom_level) as f64;
254    let world_x =
255        (f64::from(tile_address.tile_x) + point.x_coordinate as f64 / extent) / tiles_per_axis;
256    let world_y =
257        (f64::from(tile_address.tile_y) + point.y_coordinate as f64 / extent) / tiles_per_axis;
258    let longitude_degrees = world_x * WORLD_LONGITUDE_SPAN_DEGREES + WORLD_WEST_LONGITUDE_DEGREES;
259    let latitude_degrees = (PI * (1.0 - 2.0 * world_y)).sinh().atan().to_degrees();
260    GeographicCoordinate {
261        latitude: latitude_degrees.clamp(
262            -MERCATOR_MAXIMUM_LATITUDE_DEGREES,
263            MERCATOR_MAXIMUM_LATITUDE_DEGREES,
264        ),
265        longitude: longitude_degrees,
266    }
267}

285/// The same conversion as [`overlay_primitives_for_vector_tile`], with the
286/// style's zoom expressions evaluated at an explicit `style_zoom_level`.
287///
288/// The base-map scheduler styles a cached tile at the camera's fractional zoom
289/// level, so a pan across a paint stop restyles the already-fetched tile
290/// instead of fetching it again; the tile's address still converts every
291/// coordinate, because the geometry belongs to the tile, not to the zoom the
292/// paint is evaluated at.
293pub fn overlay_primitives_for_vector_tile_at_zoom(
294    tile: &VectorTile,
295    tile_address: VectorTileAddress,
296    style: &CompiledStyle,
297    style_zoom_level: f64,
298) -> Vec<VectorOverlayBatch> {
299    let mut batches = Vec::new();
300    for layer in &style.layers {
301        let overlays = overlays_for_layer(tile, tile_address, layer, style_zoom_level);
302        if !overlays.is_empty() {
303            batches.push(VectorOverlayBatch {
304                layer_index: layer.order_index,
305                layer_id: layer.id.clone(),
306                overlays,
307            });
308        }
309    }
310    batches
311}

313/// The tile's overlays bucketed by compiled layer index: bucket `index`
314/// holds every primitive the style layer whose `order_index` is `index`
315/// paints from this tile, and an empty bucket means that layer painted
316/// nothing. A frame concatenates the buckets layer by layer across all tiles.
317pub fn vector_tile_overlay_layers_at_zoom(
318    tile: &VectorTile,
319    tile_address: VectorTileAddress,
320    style: &CompiledStyle,
321    style_zoom_level: f64,
322) -> Vec<Vec<MapOverlay>> {
323    let mut layers = vec![Vec::new(); style.layer_count()];
324    for layer in &style.layers {
325        if let Some(bucket) = layers.get_mut(layer.order_index) {
326            *bucket = overlays_for_layer(tile, tile_address, layer, style_zoom_level);
327        }
328    }
329    layers
330}

332/// [`vector_tile_overlay_layers_at_zoom`] with a style transition applied.
333///
334/// The buckets come from the new `style` in its global document order; a
335/// painted colour or opacity blends from `previous_style`'s resolved value
336/// over the declared transition as `elapsed_seconds` advances. The caller
337/// bypasses its styled-geometry cache while a transition is running, because
338/// the resolved paint changes every frame.
339pub fn vector_tile_overlay_layers_at_zoom_with_transition(
340    tile: &VectorTile,
341    tile_address: VectorTileAddress,
342    style: &CompiledStyle,
343    previous_style: &CompiledStyle,
344    elapsed_seconds: f64,
345    style_zoom_level: f64,
346) -> Vec<Vec<MapOverlay>> {
347    let mut layers = vec![Vec::new(); style.layer_count()];
348    for layer in &style.layers {
349        if let Some(bucket) = layers.get_mut(layer.order_index) {
350            *bucket = overlays_for_layer_with_transition(
351                tile,
352                tile_address,
353                layer,
354                style_zoom_level,
355                Some(previous_style),
356                style.transition,
357                elapsed_seconds,
358            );
359        }
360    }
361    layers
362}

408/// The tile layers a compiled layer reads.
409///
410/// A layer names its `source-layer`; a GeoJSON source carries its features in
411/// one implicit layer, so a GeoJSON-bound layer without `source-layer` reads
412/// that single layer. A vector layer without `source-layer` reads nothing.
413pub fn tile_layers_for_compiled_layer<'a>(
414    tile: &'a VectorTile,
415    layer: &CompiledLayer,
416) -> Vec<&'a VectorTileLayer> {
417    // A GeoJSON source carries every feature in one implicit layer, and the
418    // style specification ignores `source-layer` for such a source.
419    if layer.source_kind == Some(CompiledSourceKind::GeoJson) && tile.layers.len() == 1 {
420        return tile.layers.iter().collect();
421    }
422    match layer.source_layer.as_deref() {
423        Some(source_layer) => tile
424            .layers
425            .iter()
426            .filter(|tile_layer| tile_layer.name == source_layer)
427            .collect(),
428        None => Vec::new(),
429    }
430}

432/// One style layer whose `source-layer` the decoded tile does not carry.
433///
434/// A style document is written for one tileset schema. Serving it tiles built
435/// from a different schema leaves every layer reading an absent source layer,
436/// so the map paints a background and fragments while each layer looks
437/// well-formed. This diagnostic names the mismatch instead of letting a frame
438/// paint nothing without an explanation.
439#[derive(Debug, Clone, PartialEq, Eq)]
440pub struct VectorStyleSourceMismatch {
441    /// The style layer that reads the absent source layer.
442    pub layer_id: String,
443    /// The source-layer name the style layer reads.
444    pub source_layer: String,
445    /// The source-layer names the served tile was decoded from, in wire order.
446    pub available_source_layers: Vec<String>,
447    /// The sentence a status readback or diagnostics ring can show.
448    pub detail: String,
449}
450
451impl std::fmt::Display for VectorStyleSourceMismatch {
452    fn fmt(&self, formatter: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
453        write!(formatter, "{}", self.detail)
454    }
455}

457/// Every style layer whose `source-layer` the tile does not carry.
458///
459/// The comparison covers the layers this renderer paints (fill, line, circle
460/// and symbol) that name a source layer and read a vector source. A GeoJSON
461/// source carries its features in one implicit layer and is skipped, and a
462/// style layer without a `source-layer` reads nothing by design and is not a
463/// mismatch.
464///
465/// # Parameters
466/// - `tile` — one decoded tile fetched from the provider the style was
467///   installed for.
468/// - `style` — the compiled document installed over that provider.
469///
470/// # Returns
471/// One named mismatch per affected layer, in style order. An empty list means
472/// every painted source layer is present. A vector-base install can report the
473/// list once per source and tile address instead of painting fragments.
474pub fn vector_style_source_layer_mismatches(
475    tile: &VectorTile,
476    style: &CompiledStyle,
477) -> Vec<VectorStyleSourceMismatch> {
478    let available_source_layers: Vec<String> = tile
479        .layers
480        .iter()
481        .map(|tile_layer| tile_layer.name.clone())
482        .collect();
483    let mut mismatches = Vec::new();
484    for layer in &style.layers {
485        if !matches!(
486            layer.kind,
487            CompiledLayerKind::Fill
488                | CompiledLayerKind::Line
489                | CompiledLayerKind::Circle
490                | CompiledLayerKind::Symbol
491        ) || layer.source_kind == Some(CompiledSourceKind::GeoJson)
492        {
493            continue;
494        }
495        let Some(source_layer) = layer.source_layer.as_deref() else {
496            continue;
497        };
498        if available_source_layers
499            .iter()
500            .any(|tile_layer_name| tile_layer_name == source_layer)
501        {
502            continue;
503        }
504        mismatches.push(VectorStyleSourceMismatch {
505            layer_id: layer.id.clone(),
506            source_layer: source_layer.to_string(),
507            available_source_layers: available_source_layers.clone(),
508            detail: format!(
509                "the style layer {:?} reads the source layer {source_layer:?}, which the \
510                 served tileset does not carry; the tile's layers are [{}]",
511                layer.id,
512                available_source_layers.join(", ")
513            ),
514        });
515    }
516    mismatches
517}

528/// [`overlays_for_layer`] with a style transition applied.
529///
530/// When `previous_style` is `Some`, a painted colour or opacity blends from
531/// the previous style's resolved value toward the new one over the declared
532/// transition as `elapsed_seconds` advances. The previous style's layer is
533/// matched by id; a layer it does not carry takes the new value immediately.
534fn overlays_for_layer_with_transition(
535    tile: &VectorTile,
536    tile_address: VectorTileAddress,
537    layer: &CompiledLayer,
538    style_zoom_level: f64,
539    previous_style: Option<&CompiledStyle>,
540    root_transition: Option<rustic_maps_style::TransitionSpec>,
541    elapsed_seconds: f64,
542) -> Vec<MapOverlay> {
543    if !matches!(
544        layer.kind,
545        CompiledLayerKind::Fill | CompiledLayerKind::Line | CompiledLayerKind::Circle
546    ) || !layer.is_visible_at_zoom(style_zoom_level)
547    {
548        return Vec::new();
549    }
550    let mut feature_overlays: Vec<(f64, Vec<MapOverlay>)> = Vec::new();
551    for tile_layer in tile_layers_for_compiled_layer(tile, layer) {
552        for feature in &tile_layer.features {
553            let properties = feature_properties(feature);
554            let context = feature_evaluation_context(feature, &properties, style_zoom_level);
555            if !layer.matches_filter(&context) {
556                continue;
557            }
558            let resolved_paint = match previous_style {
559                Some(previous_style) => layer.resolved_transitioned_paint(
560                    &context,
561                    root_transition,
562                    previous_style
563                        .layer(&layer.id)
564                        .map(|previous_layer| (previous_layer, elapsed_seconds)),
565                ),
566                None => layer.resolved_paint(&context),
567            };
568            let mut overlays = Vec::new();
569            match (&feature.geometry, layer.kind) {
570                (VectorTileGeometry::Polygons(rings), CompiledLayerKind::Fill) => {
571                    if let ResolvedPaint::Fill(paint) = resolved_paint {
572                        overlays.extend(filled_polygon_overlays_for_rings(
573                            rings,
574                            tile_layer.extent,
575                            tile_address,
576                            &paint,
577                        ));
578                    }
579                }
580                (VectorTileGeometry::LineStrings(line_strings), CompiledLayerKind::Line) => {
581                    if let ResolvedPaint::Line(paint) = resolved_paint {
582                        let rgba_color = resolved_rgba(paint.color, paint.opacity);
583                        // A layer that declares a line presentation gets the
584                        // specification's stroke; one that declares nothing
585                        // keeps the renderer's default route stroke, so a
586                        // style that predates these properties draws exactly
587                        // as before.
588                        let stroke = layer.declares_line_presentation.then(|| PolylineStroke {
589                            cap: layer.resolved_line_cap(&context),
590                            join: layer.resolved_line_join(&context),
591                            dash_array: paint.dasharray.clone(),
592                            offset_pixels: paint.offset,
593                            gap_width_pixels: paint.gap_width,
594                            blur_pixels: paint.blur,
595                        });
596                        for line_string in line_strings {
597                            let coordinates = line_string
598                                .iter()
599                                .map(|point| {
600                                    tile_local_point_to_coordinate(
601                                        tile_address,
602                                        tile_layer.extent,
603                                        point,
604                                    )
605                                })
606                                .collect();
607                            overlays.push(MapOverlay::RoutePolyline(RoutePolylineOverlay {
608                                coordinates,
609                                rgba_color,
610                                width_pixels: paint.width,
611                                stroke: stroke.clone(),
612                                translate_pixels: paint.translate,
613                                translate_anchor: paint.translate_anchor,
614                            }));
615                        }
616                    }
617                }
618                (VectorTileGeometry::Points(points), CompiledLayerKind::Circle) => {
619                    if let ResolvedPaint::Circle(paint) = resolved_paint {
620                        let positions = points
621                            .iter()
622                            .map(|point| {
623                                tile_local_point_to_coordinate(
624                                    tile_address,
625                                    tile_layer.extent,
626                                    point,
627                                )
628                            })
629                            .collect();
630                        overlays.push(MapOverlay::PointMarkers(PointMarkersOverlay {
631                            positions,
632                            rgba_color: resolved_rgba(paint.color, paint.opacity),
633                            radius_pixels: paint.radius,
634                            rgba_stroke_color: paint.stroke_color.map(|stroke_color| {
635                                resolved_rgba(stroke_color, paint.stroke_opacity)
636                            }),
637                            stroke_width_pixels: paint.stroke_width,
638                            blur_pixels: paint.blur,
639                            translate_pixels: paint.translate,
640                            // A layer that declares neither property keeps
641                            // this renderer's screen-facing default, which
642                            // the cached world-space path can express.
643                            pitch_scale: if layer.declares_circle_pitch {
644                                layer.resolved_circle_pitch_scale(&context)
645                            } else {
646                                DEFAULT_POINT_MARKER_PITCH_SCALE
647                            },
648                            pitch_alignment: if layer.declares_circle_pitch {
649                                layer.resolved_circle_pitch_alignment(&context)
650                            } else {
651                                DEFAULT_POINT_MARKER_PITCH_ALIGNMENT
652                            },
653                        }));
654                    }
655                }
656                // A style paints only its declared kind; a feature of
657                // another kind is not this layer's paint, and a symbol
658                // layer's labels belong to the frame-level label pass.
659                _ => {}
660            }
661            if !overlays.is_empty() {
662                let sort_key = match layer.kind {
663                    CompiledLayerKind::Fill => layer.resolved_fill_sort_key(&context),
664                    CompiledLayerKind::Line => layer.resolved_line_sort_key(&context),
665                    CompiledLayerKind::Circle => layer.resolved_circle_sort_key(&context),
666                    _ => 0.0,
667                };
668                feature_overlays.push((sort_key, overlays));
669            }
670        }
671    }
672    // The layer kind's `*-sort-key` orders features within the layer; a
673    // stable sort keeps equal keys in source order, and features that declare
674    // no key all resolve to the documented default.
675    feature_overlays.sort_by(|(left_key, _), (right_key, _)| left_key.total_cmp(right_key));
676    feature_overlays
677        .into_iter()
678        .flat_map(|(_, overlays)| overlays)
679        .collect()
680}

682/// Converts the compiled colour and opacity into the renderer's RGBA
683/// components, folding opacity into the alpha channel.
684pub(crate) fn resolved_rgba(color: Color, opacity: f64) -> [f32; 4] {
685    [
686        color.red as f32,
687        color.green as f32,
688        color.blue as f32,
689        (color.alpha * opacity) as f32,
690    ]
691}

736/// Groups a polygon feature's rings into filled overlays: each exterior ring
737/// owns its immediately following interior rings, the specification's order.
738/// A leading interior ring (malformed ring order) has no exterior to subtract
739/// from and is skipped rather than painted as a fill of its own.
740fn filled_polygon_overlays_for_rings(
741    rings: &[VectorTileLinearRing],
742    layer_extent: u64,
743    tile_address: VectorTileAddress,
744    paint: &rustic_maps_style::ResolvedFillPaint,
745) -> Vec<MapOverlay> {
746    let mut overlays = Vec::new();
747    let mut current_exterior_index: Option<usize> = None;
748    for ring in rings {
749        let ring_coordinates: Vec<GeographicCoordinate> = ring
750            .vertices
751            .iter()
752            .map(|point| tile_local_point_to_coordinate(tile_address, layer_extent, point))
753            .collect();
754        if ring_is_exterior(ring) {
755            overlays.push(MapOverlay::FilledPolygon(FilledPolygonOverlay {
756                ring_coordinates,
757                hole_ring_coordinates: Vec::new(),
758                rgba_color: resolved_rgba(paint.color, paint.opacity),
759                rgba_outline_color: paint
760                    .outline_color
761                    .map(|outline_color| resolved_rgba(outline_color, paint.opacity)),
762                antialias: paint.antialias,
763                translate_pixels: paint.translate,
764                // A zero displacement is identical under either anchor, so
765                // it takes the renderer's own default and keeps a style that
766                // translates nothing structurally equal to the legacy parser.
767                translate_anchor: if paint.translate == [0.0, 0.0] {
768                    rustic_maps_style::TranslateAnchor::Viewport
769                } else {
770                    paint.translate_anchor
771                },
772            }));
773            current_exterior_index = Some(overlays.len() - 1);
774        } else if let Some(exterior_index) = current_exterior_index
775            && let Some(MapOverlay::FilledPolygon(overlay)) = overlays.get_mut(exterior_index)
776        {
777            overlay.hole_ring_coordinates.push(ring_coordinates);
778        }
779    }
780    overlays
781}

788/// The ring's signed shoelace area in tile-local space; exterior rings are
789/// positive (the specification's `y`-down convention), holes negative.
790fn ring_is_exterior(ring: &VectorTileLinearRing) -> bool {
791    let vertices = &ring.vertices;
792    if vertices.len() < 3 {
793        return false;
794    }
795    let mut doubled_area = 0.0;
796    for vertex_index in 0..vertices.len() {
797        let current = &vertices[vertex_index];
798        let next = &vertices[(vertex_index + 1) % vertices.len()];
799        doubled_area += current.x_coordinate as f64 * next.y_coordinate as f64
800            - next.x_coordinate as f64 * current.y_coordinate as f64;
801    }
802    doubled_area > 0.0
803}
