1//! Vector-tile styling: a decoded [`rustic_maps_vector_tiles::VectorTile`] becomes
2//! the renderer's overlay primitives under a compiled style plan.
3//!
4//! The style document and its expressions live in `rustic-maps-style`; this
5//! module is the renderer's consumer. [`vector_tile_style_from_id`] parses a
6//! committed version-8 style document and compiles it once into a
7//! [`CompiledStyle`](rustic_maps_style::CompiledStyle). Each frame then resolves
8//! the compiled plan at the camera's fractional zoom, so paint expressions
9//! change continuously with the camera without re-parsing the document.
10//!
11//! Polygon features paint every exterior ring with its interior rings
12//! subtracted through the overlay's hole-aware fill (see
13//! [`crate::overlay::FilledPolygonOverlay`]); line features become styled
14//! polylines and point features become screen-facing markers. Primitives are
15//! grouped into one [`VectorOverlayBatch`] per style layer and one bucket per
16//! layer in [`vector_tile_overlay_layers_at_zoom`], so a frame can paint every
17//! tile's layer 0 before any tile's layer 1: global style order across the
18//! visible tile set.
19//!
20//! Coordinates convert from tile-local extent space (`x` right, `y` down, the
21//! MVT convention) through the tile's web-mercator address to geographic
22//! degrees, so the overlays project exactly like every other map coordinate.
23//!
24//! Symbol layers paint no overlay primitive: their labels are data for the
25//! frame-level label pass ([`crate::labels`]), which reads the same compiled
26//! plan, filter and layout.
27

139        _ => None,

143/// Parses and compiles the built-in style document a style id names.
144///
145/// # Parameters
146/// - `style_id` — one of the exported `VECTOR_STYLE_ID_*` identifiers.
147///
148/// # Returns
149/// The compiled style, ready to resolve at a frame zoom and feature.
150///
151/// # Errors
152/// An unknown id and a malformed document both report the typed artifact
153/// error rather than a guess.
154pub fn vector_tile_style_from_id(
155    style_id: &str,
156) -> Result<CompiledStyle, VectorStyleArtifactError> {
157    let Some(style_json) = vector_style_document_json(style_id) else {
158        return Err(VectorStyleArtifactError {
159            detail: format!(
160                "no committed vector style carries the id {style_id:?}; known ids are \
161                 {VECTOR_STYLE_ID_CARTO_LIGHT}, {VECTOR_STYLE_ID_CARTO_DARK}, \
162                 {VECTOR_STYLE_ID_OPENMAPTILES_LIGHT}, {VECTOR_STYLE_ID_OPENMAPTILES_DARK}, \
163                 {VECTOR_STYLE_ID_BRIGHT_LIGHT}, {VECTOR_STYLE_ID_FIORD_DARK}, \
164                 {VECTOR_STYLE_ID_POSITRON_LIGHT}"
165            ),
166        });
167    };
168    vector_tile_style_from_json(style_json)
169}

186/// Parses and compiles a version-8 style document.
187///
188/// The compile step is infallible; only a malformed document reports an error.
189pub fn vector_tile_style_from_json(
190    style_json: &str,
191) -> Result<CompiledStyle, VectorStyleArtifactError> {
192    let document = MapStyleDocument::parse_json(style_json).map_err(|style_error| {
193        VectorStyleArtifactError {
194            detail: style_error.to_string(),
195        }
196    })?;
197    Ok(compile_style_document(&document))
198}

205/// Compiles a style document, falling back to the committed default document
206/// when the supplied one is malformed.
207///
208/// The default document is itself compiled at most once per call and is a
209/// compile-time constant, so this cannot loop; an empty style is the last
210/// resort if the committed default is ever removed from the build.
211pub fn vector_tile_style_or_fallback(style_json: &str) -> CompiledStyle {
212    vector_tile_style_from_json(style_json).unwrap_or_else(|_| {
213        vector_tile_style_from_json(DEFAULT_VECTOR_STYLE_DOCUMENT_JSON)
214            .unwrap_or_else(|_| empty_compiled_style())
215    })
216}

229/// Converts one tile-local extent point into a geographic coordinate through
230/// the tile's web-mercator address. Vertices at the extent boundary are legal
231/// (they land on the next tile's edge); the latitude clamp keeps the projection
232/// finite at the world's polar edges.
233pub fn tile_local_point_to_coordinate(
234    tile_address: VectorTileAddress,
235    layer_extent: u64,
236    point: &VectorTileGeometryPoint,
237) -> GeographicCoordinate {
238    let extent = layer_extent.max(1) as f64;
239    let tiles_per_axis = (1u64 << tile_address.zoom_level) as f64;
240    let world_x =
241        (f64::from(tile_address.tile_x) + point.x_coordinate as f64 / extent) / tiles_per_axis;
242    let world_y =
243        (f64::from(tile_address.tile_y) + point.y_coordinate as f64 / extent) / tiles_per_axis;
244    let longitude_degrees = world_x * WORLD_LONGITUDE_SPAN_DEGREES + WORLD_WEST_LONGITUDE_DEGREES;
245    let latitude_degrees = (PI * (1.0 - 2.0 * world_y)).sinh().atan().to_degrees();
246    GeographicCoordinate {
247        latitude: latitude_degrees.clamp(
248            -MERCATOR_MAXIMUM_LATITUDE_DEGREES,
249            MERCATOR_MAXIMUM_LATITUDE_DEGREES,
250        ),
251        longitude: longitude_degrees,
252    }
253}

271/// The same conversion as [`overlay_primitives_for_vector_tile`], with the
272/// style's zoom expressions evaluated at an explicit `style_zoom_level`.
273///
274/// The base-map scheduler styles a cached tile at the camera's fractional zoom
275/// level, so a pan across a paint stop restyles the already-fetched tile
276/// instead of fetching it again; the tile's address still converts every
277/// coordinate, because the geometry belongs to the tile, not to the zoom the
278/// paint is evaluated at.
279pub fn overlay_primitives_for_vector_tile_at_zoom(
280    tile: &VectorTile,
281    tile_address: VectorTileAddress,
282    style: &CompiledStyle,
283    style_zoom_level: f64,
284) -> Vec<VectorOverlayBatch> {
285    let mut batches = Vec::new();
286    for layer in &style.layers {
287        let overlays = overlays_for_layer(tile, tile_address, layer, style_zoom_level);
288        if !overlays.is_empty() {
289            batches.push(VectorOverlayBatch {
290                layer_index: layer.order_index,
291                layer_id: layer.id.clone(),
292                overlays,
293            });
294        }
295    }
296    batches
297}

299/// The tile's overlays bucketed by compiled layer index: bucket `index`
300/// holds every primitive the style layer whose `order_index` is `index`
301/// paints from this tile, and an empty bucket means that layer painted
302/// nothing. A frame concatenates the buckets layer by layer across all tiles.
303pub fn vector_tile_overlay_layers_at_zoom(
304    tile: &VectorTile,
305    tile_address: VectorTileAddress,
306    style: &CompiledStyle,
307    style_zoom_level: f64,
308) -> Vec<Vec<MapOverlay>> {
309    let mut layers = vec![Vec::new(); style.layer_count()];
310    for layer in &style.layers {
311        if let Some(bucket) = layers.get_mut(layer.order_index) {
312            *bucket = overlays_for_layer(tile, tile_address, layer, style_zoom_level);
313        }
314    }
315    layers
316}

348/// Every overlay one compiled layer paints from the tile, in feature order.
349///
350/// This is the single-layer seam a multi-source frame assembles with: the
351/// caller passes the tile fetched from the layer's own source binding, so
352/// layers bound to different sources never read each other's geometry.
353pub fn overlays_for_compiled_layer_at_zoom(
354    tile: &VectorTile,
355    tile_address: VectorTileAddress,
356    layer: &CompiledLayer,
357    style_zoom_level: f64,
358) -> Vec<MapOverlay> {
359    overlays_for_layer(tile, tile_address, layer, style_zoom_level)
360}

362/// The tile layers a compiled layer reads.
363///
364/// A layer names its `source-layer`; a GeoJSON source carries its features in
365/// one implicit layer, so a GeoJSON-bound layer without `source-layer` reads
366/// that single layer. A vector layer without `source-layer` reads nothing.
367pub fn tile_layers_for_compiled_layer<'a>(
368    tile: &'a VectorTile,
369    layer: &CompiledLayer,
370) -> Vec<&'a VectorTileLayer> {
371    // A GeoJSON source carries every feature in one implicit layer, and the
372    // style specification ignores `source-layer` for such a source.
373    if layer.source_kind == Some(CompiledSourceKind::GeoJson) && tile.layers.len() == 1 {
374        return tile.layers.iter().collect();
375    }
376    match layer.source_layer.as_deref() {
377        Some(source_layer) => tile
378            .layers
379            .iter()
380            .filter(|tile_layer| tile_layer.name == source_layer)
381            .collect(),
382        None => Vec::new(),
383    }
384}

386fn overlays_for_layer(
387    tile: &VectorTile,
388    tile_address: VectorTileAddress,
389    layer: &CompiledLayer,
390    style_zoom_level: f64,
391) -> Vec<MapOverlay> {
392    if !matches!(
393        layer.kind,
394        CompiledLayerKind::Fill | CompiledLayerKind::Line | CompiledLayerKind::Circle
395    ) || !layer.is_visible_at_zoom(style_zoom_level)
396    {
397        return Vec::new();
398    }
399    let mut feature_overlays: Vec<(f64, Vec<MapOverlay>)> = Vec::new();
400    for tile_layer in tile_layers_for_compiled_layer(tile, layer) {
401        for feature in &tile_layer.features {
402            let properties = feature_properties(feature);
403            let context = feature_evaluation_context(feature, &properties, style_zoom_level);
404            if !layer.matches_filter(&context) {
405                continue;
406            }
407            let mut overlays = Vec::new();
408            match (&feature.geometry, layer.kind) {
409                (VectorTileGeometry::Polygons(rings), CompiledLayerKind::Fill) => {
410                    if let ResolvedPaint::Fill(paint) = layer.resolved_paint(&context) {
411                        overlays.extend(filled_polygon_overlays_for_rings(
412                            rings,
413                            tile_layer.extent,
414                            tile_address,
415                            &paint,
416                        ));
417                    }
418                }
419                (VectorTileGeometry::LineStrings(line_strings), CompiledLayerKind::Line) => {
420                    if let ResolvedPaint::Line(paint) = layer.resolved_paint(&context) {
421                        let rgba_color = resolved_rgba(paint.color, paint.opacity);
422                        // A layer that declares a line presentation gets the
423                        // specification's stroke; one that declares nothing
424                        // keeps the renderer's default route stroke, so a
425                        // style that predates these properties draws exactly
426                        // as before.
427                        let stroke = layer.declares_line_presentation.then(|| PolylineStroke {
428                            cap: layer.resolved_line_cap(&context),
429                            join: layer.resolved_line_join(&context),
430                            dash_array: paint.dasharray.clone(),
431                            offset_pixels: paint.offset,
432                            gap_width_pixels: paint.gap_width,
433                            blur_pixels: paint.blur,
434                        });
435                        for line_string in line_strings {
436                            let coordinates = line_string
437                                .iter()
438                                .map(|point| {
439                                    tile_local_point_to_coordinate(
440                                        tile_address,
441                                        tile_layer.extent,
442                                        point,
443                                    )
444                                })
445                                .collect();
446                            overlays.push(MapOverlay::RoutePolyline(RoutePolylineOverlay {
447                                coordinates,
448                                rgba_color,
449                                width_pixels: paint.width,
450                                stroke: stroke.clone(),
451                            }));
452                        }
453                    }
454                }
455                (VectorTileGeometry::Points(points), CompiledLayerKind::Circle) => {
456                    if let ResolvedPaint::Circle(paint) = layer.resolved_paint(&context) {
457                        let positions = points
458                            .iter()
459                            .map(|point| {
460                                tile_local_point_to_coordinate(
461                                    tile_address,
462                                    tile_layer.extent,
463                                    point,
464                                )
465                            })
466                            .collect();
467                        overlays.push(MapOverlay::PointMarkers(PointMarkersOverlay {
468                            positions,
469                            rgba_color: resolved_rgba(paint.color, paint.opacity),
470                            radius_pixels: paint.radius,
471                            rgba_stroke_color: paint.stroke_color.map(|stroke_color| {
472                                resolved_rgba(stroke_color, paint.stroke_opacity)
473                            }),
474                            stroke_width_pixels: paint.stroke_width,
475                            blur_pixels: paint.blur,
476                            translate_pixels: paint.translate,
477                            // A layer that declares neither property keeps
478                            // this renderer's screen-facing default, which
479                            // the cached world-space path can express.
480                            pitch_scale: if layer.declares_circle_pitch {
481                                layer.resolved_circle_pitch_scale(&context)
482                            } else {
483                                DEFAULT_POINT_MARKER_PITCH_SCALE
484                            },
485                            pitch_alignment: if layer.declares_circle_pitch {
486                                layer.resolved_circle_pitch_alignment(&context)
487                            } else {
488                                DEFAULT_POINT_MARKER_PITCH_ALIGNMENT
489                            },
490                        }));
491                    }
492                }
493                // A style paints only its declared kind; a feature of
494                // another kind is not this layer's paint, and a symbol
495                // layer's labels belong to the frame-level label pass.
496                _ => {}
497            }
498            if !overlays.is_empty() {
499                feature_overlays.push((layer.resolved_line_sort_key(&context), overlays));
500            }
501        }
502    }
503    // `line-sort-key` orders features within the layer; a stable sort keeps
504    // equal keys in source order, and features that declare no key all
505    // resolve to the documented default.
506    feature_overlays.sort_by(|(left_key, _), (right_key, _)| left_key.total_cmp(right_key));
507    feature_overlays
508        .into_iter()
509        .flat_map(|(_, overlays)| overlays)
510        .collect()
511}

513/// Converts the compiled colour and opacity into the renderer's RGBA
514/// components, folding opacity into the alpha channel.
515pub(crate) fn resolved_rgba(color: Color, opacity: f64) -> [f32; 4] {
516    [
517        color.red as f32,
518        color.green as f32,
519        color.blue as f32,
520        (color.alpha * opacity) as f32,
521    ]
522}

524/// The feature's attributes as the style evaluator's property map.
525pub(crate) fn feature_properties(feature: &VectorTileFeature) -> BTreeMap<String, Value> {
526    feature
527        .attributes
528        .iter()
529        .map(|(attribute_name, attribute_value)| {
530            (
531                attribute_name.clone(),
532                attribute_value_to_style_value(attribute_value),
533            )
534        })
535        .collect()
536}

538/// One decoder attribute value as the style evaluator's value: text, a number
539/// for every numeric wire variant, and a boolean.
540fn attribute_value_to_style_value(attribute_value: &VectorTileAttributeValue) -> Value {
541    match attribute_value {
542        VectorTileAttributeValue::Text(text) => Value::String(text.clone()),
543        VectorTileAttributeValue::Boolean(boolean) => Value::Boolean(*boolean),
544        numeric_value => Value::Number(numeric_value.numeric_value().unwrap_or_default()),
545    }
546}

567/// Groups a polygon feature's rings into filled overlays: each exterior ring
568/// owns its immediately following interior rings, the specification's order.
569/// A leading interior ring (malformed ring order) has no exterior to subtract
570/// from and is skipped rather than painted as a fill of its own.
571fn filled_polygon_overlays_for_rings(
572    rings: &[VectorTileLinearRing],
573    layer_extent: u64,
574    tile_address: VectorTileAddress,
575    paint: &rustic_maps_style::ResolvedFillPaint,
576) -> Vec<MapOverlay> {
577    let mut overlays = Vec::new();
578    let mut current_exterior_index: Option<usize> = None;
579    for ring in rings {
580        let ring_coordinates: Vec<GeographicCoordinate> = ring
581            .vertices
582            .iter()
583            .map(|point| tile_local_point_to_coordinate(tile_address, layer_extent, point))
584            .collect();
585        if ring_is_exterior(ring) {
586            overlays.push(MapOverlay::FilledPolygon(FilledPolygonOverlay {
587                ring_coordinates,
588                hole_ring_coordinates: Vec::new(),
589                rgba_color: resolved_rgba(paint.color, paint.opacity),
590                rgba_outline_color: paint
591                    .outline_color
592                    .map(|outline_color| resolved_rgba(outline_color, paint.opacity)),
593                antialias: paint.antialias,
594                translate_pixels: paint.translate,
595            }));
596            current_exterior_index = Some(overlays.len() - 1);
597        } else if let Some(exterior_index) = current_exterior_index {
598            if let Some(MapOverlay::FilledPolygon(overlay)) = overlays.get_mut(exterior_index) {
599                overlay.hole_ring_coordinates.push(ring_coordinates);
600            }
601        }
602    }
603    overlays
604}

606/// The feature's geometry kind (the decoder's own classification).
607pub fn feature_geometry_kind(feature: &VectorTileFeature) -> VectorTileGeometryType {
608    feature.geometry_type
609}

611/// The ring's signed shoelace area in tile-local space; exterior rings are
612/// positive (the specification's `y`-down convention), holes negative.
613fn ring_is_exterior(ring: &VectorTileLinearRing) -> bool {
614    let vertices = &ring.vertices;
615    if vertices.len() < 3 {
616        return false;
617    }
618    let mut doubled_area = 0.0;
619    for vertex_index in 0..vertices.len() {
620        let current = &vertices[vertex_index];
621        let next = &vertices[(vertex_index + 1) % vertices.len()];
622        doubled_area += current.x_coordinate as f64 * next.y_coordinate as f64
623            - next.x_coordinate as f64 * current.y_coordinate as f64;
624    }
625    doubled_area > 0.0
626}
