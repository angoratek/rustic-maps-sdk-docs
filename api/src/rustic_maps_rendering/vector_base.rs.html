1//! Shared vector-tile base pipeline: one scheduler that fetches a vector
2//! tileset through the [`VectorTileProvider`] trait, decodes each tile once,
3//! styles it, and answers the overlay primitives the frame draws beneath its
4//! own overlays.
5//!
6//! Every target serves its vector base through this one implementation, so the
7//! native surfaces and the browser cannot drift in what a tile paints. The
8//! scheduler owns two caches:
9//!
10//! - **Decoded tiles**, keyed by the tile address: a tile's bytes are fetched
11//!   and decoded once and never again while it stays cached; a style change
12//!   restyles the decoded tile instead of refetching it.
13//! - **Styled geometry**, keyed by the tile address, the style revision and the
14//!   style's zoom bucket: the style's zoom span (`min_zoom`/`max_zoom`) and
15//!   width stops are evaluated at the camera's zoom level, so a camera pan
16//!   across a width stop repaints the tile from the style it already carries —
17//!   no refetch, no decode. The zoom level quantizes to
18//!   [`VECTOR_BASE_STYLE_ZOOM_QUANTIZATION_STEPS_PER_LEVEL`] steps per level so
19//!   a continuous zoom gesture cannot grow the styled cache without bound.
20//!
21//! A tile that fails to fetch or decode does not fail the frame: the scheduler
22//! skips it, keeps painting the tiles that succeeded, and reports the failure
23//! in the frame's counts and its last failure detail. The base map is one layer
24//! of a frame, never the frame's reason to fail.
25//!
26//! # Latency: payload cache, deliberate prefetch, progressive draw
27//!
28//! The camera-facing [`VectorBaseScheduler::cached_frame`] never waits on the
29//! network: it paints the decoded tiles the cache holds, reports the rest as
30//! pending or failed, and lets a detached prefetch fill the gaps. A missing
31//! tile whose ancestor is cached draws that ancestor at the same camera zoom
32//! (the raster path's parent-fallback semantics), so a new area blurs in at
33//! coarse detail instead of flashing holes.
34//!
35//! Fetched wire payloads additionally pass through a
36//! [`rustic_maps_core::CachedResponseStore`] keyed by the tileset's source name
37//! and the tile address. The native surfaces install their disk store (TTL,
38//! LRU and quota), so a region browsed once survives the next session; the
39//! browser installs [`VectorTileMemoryCache`], the same policy in memory where
40//! no SQLite exists.
41//!
42//! [`VectorBaseScheduler::prefetch_missing_tiles`] claims the visible misses;
43//! [`VectorBaseScheduler::prefetch_ring_tiles`] then claims the ring around the
44//! viewport (and the coarse parents that back a fallback) under a small budget.
45//! Both return cancellable detached futures: a dropped future releases its
46//! claims, so a scheduler that is replaced mid-fetch never strands a tile
47//! address as permanently in flight.
48//!
49//! The scheduler is camera-agnostic: the caller lists the visible tile
50//! addresses (through `MapCamera::visible_tile_keys` or the page's own grid)
51//! and names the camera's zoom level; [`VectorBaseScheduler::served_zoom_level`]
52//! clamps that listing to the tileset's declared zoom span.
53//!
54//! Each resolved frame also carries the drawn tiles' label candidates, read
55//! from the active style's symbol layers. [`VectorBaseFrame::label_layer`]
56//! places them at a camera pose through the SDK's embedded default glyph font,
57//! so every platform surface draws the same place names without a
58//! platform-specific asset. A frame without candidates yields `None` and
59//! costs the renderer nothing.

398/// The decoded-tile cache shared between the presenting frame and its detached
399/// prefetch. The frame resolves from whatever the cache already holds, while a
400/// spawned prefetch fetches every missing tile concurrently and inserts it, so
401/// a first frame over unvisited ground never waits on the network. The cache
402/// also remembers in-flight addresses (so a repaint does not start duplicate
403/// fetches) and the last failure per address (so a frame can report it until
404/// the next prefetch retries).
405#[derive(Debug)]
406struct DecodedTileCache {
407    /// Decoded tiles by address (fetched and decoded once).
408    tiles: HashMap<TileKey, Arc<VectorTile>>,
409    /// The cache's insertion order, for eviction.
410    insertion_order: VecDeque<TileKey>,
411    /// Addresses whose fetch is running.
412    in_flight: HashSet<TileKey>,
413    /// The last failure detail per address, cleared by the retry.
414    failure_details: HashMap<TileKey, String>,
415    /// The cache's capacity.
416    maximum_tile_count: usize,
417}
418
419impl DecodedTileCache {
420    fn new(maximum_tile_count: usize) -> Self {
421        Self {
422            tiles: HashMap::new(),
423            insertion_order: VecDeque::new(),
424            in_flight: HashSet::new(),
425            failure_details: HashMap::new(),
426            maximum_tile_count,
427        }
428    }
429
430    fn get(&self, tile_key: &TileKey) -> Option<Arc<VectorTile>> {
431        self.tiles.get(tile_key).cloned()
432    }
433
434    /// Claims the address for a fetch; `false` when it is already cached or a
435    /// fetch is running. Claiming clears any previous failure so the frame
436    /// reports "pending", not a stale refusal, while the retry runs.
437    fn begin_fetch(&mut self, tile_key: TileKey) -> bool {
438        if self.tiles.contains_key(&tile_key) || !self.in_flight.insert(tile_key) {
439            return false;
440        }
441        self.failure_details.remove(&tile_key);
442        true
443    }
444
445    /// Records one finished fetch: the decoded tile on success, the detail on
446    /// failure, and clears the in-flight mark either way.
447    fn finish_fetch(&mut self, tile_key: TileKey, outcome: Result<Arc<VectorTile>, String>) {
448        self.in_flight.remove(&tile_key);
449        match outcome {
450            Ok(decoded_tile) => {
451                self.failure_details.remove(&tile_key);
452                self.insert_decoded_tile(tile_key, decoded_tile);
453            }
454            Err(failure_detail) => {
455                // The failure map is a diagnostic aid, not a record: it is
456                // capped so a long session over failing ground cannot grow it
457                // without bound.
458                if !self.failure_details.contains_key(&tile_key)
459                    && self.failure_details.len() >= MAXIMUM_RECORDED_FAILURE_DETAILS
460                {
461                    if let Some(evicted_key) = self.failure_details.keys().next().copied() {
462                        self.failure_details.remove(&evicted_key);
463                    }
464                }
465                self.failure_details.insert(tile_key, failure_detail);
466            }
467        }
468    }
469
470    /// Inserts a decoded tile and evicts the oldest decoded tiles beyond the
471    /// cache's capacity. Styled geometry is keyed by the address and a style
472    /// revision, so it stays valid for a re-fetched address and is bounded by
473    /// its own cache; eviction here never touches it.
474    fn insert_decoded_tile(&mut self, tile_key: TileKey, decoded_tile: Arc<VectorTile>) {
475        if self.tiles.insert(tile_key, decoded_tile).is_none() {
476            self.insertion_order.push_back(tile_key);
477        }
478        while self.tiles.len() > self.maximum_tile_count {
479            let Some(evicted_tile_key) = self.insertion_order.pop_front() else {
480                break;
481            };
482            self.tiles.remove(&evicted_tile_key);
483        }
484    }
485}

487/// The in-memory vector-tile payload cache the browser installs where no
488/// SQLite store exists: the same cache-then-network policy as the native disk
489/// store (a fingerprint key, a time-to-live and a bounded capacity), held in
490/// process memory with least-recently-used eviction.
491///
492/// The cache is shared across scheduler installs, so switching a style or
493/// reinstalling the same tileset serves the region from memory instead of
494/// refetching it. It never outlives the page.
495#[derive(Debug)]
496pub struct VectorTileMemoryCache {
497    /// The cached payloads and their access order.
498    state: Mutex<VectorTileMemoryCacheState>,
499    /// The cache's capacity, in tiles.
500    maximum_tile_count: usize,
501}
502
503#[derive(Debug, Default)]
504struct VectorTileMemoryCacheState {
505    /// Cached payloads by request fingerprint.
506    entries: HashMap<String, VectorTileMemoryCacheEntry>,
507    /// Monotonic access stamp, for least-recently-used eviction.
508    access_sequence: u64,
509}
510
511#[derive(Debug)]
512struct VectorTileMemoryCacheEntry {
513    /// The stored wire payload.
514    payload: Arc<VectorTileData>,
515    /// When the entry expires.
516    expires_at: Instant,
517    /// The access stamp of the entry's last read or write.
518    last_accessed: u64,
519}
520
521impl VectorTileMemoryCache {
522    /// Builds an empty payload cache holding at most `maximum_tile_count`
523    /// tiles.
524    pub fn new(maximum_tile_count: usize) -> Self {
525        Self {
526            state: Mutex::new(VectorTileMemoryCacheState::default()),
527            maximum_tile_count,
528        }
529    }
530
531    /// The fingerprints the cache currently holds, for tests and diagnosis.
532    pub fn cached_tile_count(&self) -> usize {
533        self.state
534            .lock()
535            .unwrap_or_else(PoisonError::into_inner)
536            .entries
537            .len()
538    }
539}

541impl CachedResponseStore<VectorTileData> for VectorTileMemoryCache {
542    fn lookup_response(
543        &self,
544        request_fingerprint: &str,
545    ) -> Result<Option<Arc<VectorTileData>>, geospatial_types::SDKError> {
546        let mut state = self.state.lock().unwrap_or_else(PoisonError::into_inner);
547        let access_stamp = {
548            state.access_sequence += 1;
549            state.access_sequence
550        };
551        let Some(entry) = state.entries.get_mut(request_fingerprint) else {
552            return Ok(None);
553        };
554        if entry.expires_at <= Instant::now() {
555            state.entries.remove(request_fingerprint);
556            return Ok(None);
557        }
558        entry.last_accessed = access_stamp;
559        Ok(Some(Arc::clone(&entry.payload)))
560    }
561
562    fn store_response(
563        &self,
564        request_fingerprint: &str,
565        response_value: Arc<VectorTileData>,
566        time_to_live_seconds: u64,
567    ) -> Result<(), geospatial_types::SDKError> {
568        let mut state = self.state.lock().unwrap_or_else(PoisonError::into_inner);
569        state.access_sequence += 1;
570        let access_stamp = state.access_sequence;
571        state.entries.insert(
572            request_fingerprint.to_string(),
573            VectorTileMemoryCacheEntry {
574                payload: response_value,
575                expires_at: Instant::now() + Duration::from_secs(time_to_live_seconds),
576                last_accessed: access_stamp,
577            },
578        );
579        while state.entries.len() > self.maximum_tile_count {
580            let Some(least_recently_used_fingerprint) = state
581                .entries
582                .iter()
583                .min_by_key(|(_, entry)| entry.last_accessed)
584                .map(|(fingerprint, _)| fingerprint.clone())
585            else {
586                break;
587            };
588            state.entries.remove(&least_recently_used_fingerprint);
589        }
590        Ok(())
591    }
592}

619/// The guard of a detached prefetch: it clears the in-flight marks of every
620/// claimed tile still undelivered when the future is dropped. Without it, a
621/// cancelled prefetch (the runtime shutting down, or a replaced scheduler)
622/// would report its addresses pending forever and no later plan would retry
623/// them.
624struct PrefetchClaimGuard {
625    /// The cache the claims live in.
626    decoded_tiles: Arc<Mutex<DecodedTileCache>>,
627    /// The claimed addresses not yet finished.
628    remaining_tile_keys: Vec<TileKey>,
629}
630
631impl PrefetchClaimGuard {
632    fn new(decoded_tiles: Arc<Mutex<DecodedTileCache>>, claimed_tile_keys: Vec<TileKey>) -> Self {
633        Self {
634            decoded_tiles,
635            remaining_tile_keys: claimed_tile_keys,
636        }
637    }
638
639    /// Records one finished fetch, so a later drop only clears the claims that
640    /// are still genuinely in flight.
641    fn mark_finished(&mut self, tile_key: TileKey) {
642        self.remaining_tile_keys.retain(|key| *key != tile_key);
643    }
644}
645
646impl Drop for PrefetchClaimGuard {
647    fn drop(&mut self) {
648        let mut decoded_tiles = self
649            .decoded_tiles
650            .lock()
651            .unwrap_or_else(PoisonError::into_inner);
652        for tile_key in &self.remaining_tile_keys {
653            decoded_tiles.in_flight.remove(tile_key);
654        }
655    }
656}

1076    /// Claims the ring around the viewport for a detached prefetch: the eight
1077    /// neighbors of every visible tile at the served zoom, plus each visible
1078    /// tile's coarse parent (the fallback that covers the area while the
1079    /// detailed tile is in flight), under
1080    /// [`Self::with_maximum_prefetch_ring_tile_count`]'s budget.
1081    ///
1082    /// Parents come first because they can cover a missing child immediately;
1083    /// the rest are ordered by distance from the viewport's center, so a
1084    /// budgeted plan grows outward instead of claiming a scattered corner.
1085    /// Nothing already cached, failed or in flight is claimed. `None` when
1086    /// the ring has nothing left to claim, when a visible tile's own fetch is
1087    /// still running (the view comes first), or when the previous plan ran
1088    /// inside the configured interval.
1089    pub fn prefetch_ring_tiles(
1090        &self,
1091        visible_tile_keys: &[TileKey],
1092    ) -> Option<impl std::future::Future<Output = usize> + Send + 'static> {
1093        let visible_tiles_are_fetching = {
1094            let decoded_tiles = self
1095                .decoded_tiles
1096                .lock()
1097                .unwrap_or_else(PoisonError::into_inner);
1098            visible_tile_keys
1099                .iter()
1100                .any(|tile_key| decoded_tiles.in_flight.contains(tile_key))
1101        };
1102        if visible_tiles_are_fetching {
1103            return None;
1104        }
1105        let candidates = self.ring_prefetch_candidates(visible_tile_keys);
1106        if candidates.is_empty() {
1107            return None;
1108        }
1109        {
1110            let mut last_ring_prefetch_at = self
1111                .last_ring_prefetch_at
1112                .lock()
1113                .unwrap_or_else(PoisonError::into_inner);
1114            let now = Instant::now();
1115            if last_ring_prefetch_at.is_some_and(|last_plan_at| {
1116                now.duration_since(last_plan_at) < self.ring_prefetch_minimum_interval
1117            }) {
1118                return None;
1119            }
1120            *last_ring_prefetch_at = Some(now);
1121        }
1122        self.prefetch_tile_keys(candidates, Some(self.maximum_prefetch_ring_tile_count))
1123    }
1124
1125    /// The ring plan: coarse parents first, then same-zoom neighbors ordered
1126    /// by distance from the visible set's center. The plan is deduplicated
1127    /// against the visible set; the per-plan budget applies to the tiles the
1128    /// caches cannot already serve, so budgeted plans are not wasted on
1129    /// cached neighbors.
1130    fn ring_prefetch_candidates(&self, visible_tile_keys: &[TileKey]) -> Vec<TileKey> {
1131        let mut visible_tiles = visible_tile_keys.to_vec();
1132        visible_tiles.sort();
1133        visible_tiles.dedup();
1134        if visible_tiles.is_empty() {
1135            return Vec::new();
1136        }
1137        let served_zoom_level = visible_tiles
1138            .iter()
1139            .map(|tile_key| tile_key.zoom_level)
1140            .max()
1141            .unwrap_or(0);
1142        let tiles_per_axis = 1u64
1143            .checked_shl(u32::from(served_zoom_level))
1144            .unwrap_or(u64::MAX);
1145        let visible_tile_set = visible_tiles.iter().copied().collect::<HashSet<_>>();
1146        let center_tile_x = visible_tiles
1147            .iter()
1148            .map(|tile_key| f64::from(tile_key.tile_x))
1149            .sum::<f64>()
1150            / visible_tiles.len() as f64;
1151        let center_tile_y = visible_tiles
1152            .iter()
1153            .map(|tile_key| f64::from(tile_key.tile_y))
1154            .sum::<f64>()
1155            / visible_tiles.len() as f64;
1156        let distance_from_center = |tile_key: &TileKey| {
1157            let horizontal_distance = f64::from(tile_key.tile_x) - center_tile_x;
1158            let vertical_distance = f64::from(tile_key.tile_y) - center_tile_y;
1159            horizontal_distance * horizontal_distance + vertical_distance * vertical_distance
1160        };
1161        let mut parent_tile_keys = visible_tiles
1162            .iter()
1163            .filter_map(|tile_key| tile_key.parent_tile_key())
1164            .filter(|tile_key| !visible_tile_set.contains(tile_key))
1165            .collect::<Vec<_>>();
1166        let mut ring_tile_keys = Vec::new();
1167        for visible_tile in &visible_tiles {
1168            for (horizontal_offset, vertical_offset) in VECTOR_BASE_RING_NEIGHBOR_OFFSETS {
1169                let neighbor_tile_x = i64::from(visible_tile.tile_x) + i64::from(horizontal_offset);
1170                let neighbor_tile_y = i64::from(visible_tile.tile_y) + i64::from(vertical_offset);
1171                if neighbor_tile_x < 0 || neighbor_tile_y < 0 {
1172                    continue;
1173                }
1174                if neighbor_tile_x as u64 >= tiles_per_axis
1175                    || neighbor_tile_y as u64 >= tiles_per_axis
1176                {
1177                    continue;
1178                }
1179                let Ok(neighbor_tile_key) = TileKey::new(
1180                    neighbor_tile_x as u32,
1181                    neighbor_tile_y as u32,
1182                    served_zoom_level,
1183                ) else {
1184                    continue;
1185                };
1186                if !visible_tile_set.contains(&neighbor_tile_key) {
1187                    ring_tile_keys.push(neighbor_tile_key);
1188                }
1189            }
1190        }
1191        parent_tile_keys.sort_by(|first, second| {
1192            distance_from_center(first)
1193                .total_cmp(&distance_from_center(second))
1194                .then_with(|| first.cmp(second))
1195        });
1196        parent_tile_keys.dedup();
1197        ring_tile_keys.sort_by(|first, second| {
1198            distance_from_center(first)
1199                .total_cmp(&distance_from_center(second))
1200                .then_with(|| first.cmp(second))
1201        });
1202        let mut candidates = parent_tile_keys;
1203        candidates.extend(ring_tile_keys);
1204        let mut seen_tile_keys = HashSet::new();
1205        candidates
1206            .into_iter()
1207            .filter(|tile_key| seen_tile_keys.insert(*tile_key))
1208            .collect()
1209    }

1211    /// Claims the cache misses among `candidate_tile_keys`, up to
1212    /// `maximum_claim_count` when one is given, and answers the detached fetch
1213    /// future. Shared by the visible and ring prefetch plans: both
1214    /// deduplicate, claim under one short lock, and run the fetches
1215    /// concurrently, so neither ever duplicates another plan's request. The
1216    /// candidate order is the claim priority, and the budget is applied after
1217    /// the cached and in-flight addresses are skipped.
1218    fn prefetch_tile_keys(
1219        &self,
1220        candidate_tile_keys: Vec<TileKey>,
1221        maximum_claim_count: Option<usize>,
1222    ) -> Option<impl std::future::Future<Output = usize> + Send + 'static> {
1223        let claimed_tile_keys = {
1224            let mut decoded_tiles = self
1225                .decoded_tiles
1226                .lock()
1227                .unwrap_or_else(PoisonError::into_inner);
1228            let mut seen_tile_keys = HashSet::new();
1229            candidate_tile_keys
1230                .into_iter()
1231                .filter(|tile_key| seen_tile_keys.insert(*tile_key))
1232                .filter(|tile_key| decoded_tiles.begin_fetch(*tile_key))
1233                .take(maximum_claim_count.unwrap_or(usize::MAX))
1234                .collect::<Vec<TileKey>>()
1235        };
1236        if claimed_tile_keys.is_empty() {
1237            return None;
1238        }
1239        let provider = Arc::clone(&self.provider);
1240        let decoded_tiles = Arc::clone(&self.decoded_tiles);
1241        let payload_cache = self.payload_cache.clone();
1242        let payload_cache_source_name = self.provider.source_descriptor().source_name;
1243        let payload_cache_time_to_live_seconds = self.payload_cache_time_to_live_seconds;
1244        let payload_cache_counters = Arc::clone(&self.payload_cache_counters);
1245        Some(async move {
1246            let mut claim_guard =
1247                PrefetchClaimGuard::new(Arc::clone(&decoded_tiles), claimed_tile_keys.clone());
1248            let fetched_tiles = join_all(claimed_tile_keys.into_iter().map(|tile_key| {
1249                let provider = Arc::clone(&provider);
1250                let payload_cache = payload_cache.clone();
1251                let payload_cache_source_name = payload_cache_source_name.clone();
1252                let payload_cache_counters = Arc::clone(&payload_cache_counters);
1253                async move {
1254                    (
1255                        tile_key,
1256                        Self::fetch_and_decode_vector_tile(
1257                            provider.as_ref(),
1258                            payload_cache.as_deref(),
1259                            &payload_cache_source_name,
1260                            payload_cache_time_to_live_seconds,
1261                            &payload_cache_counters,
1262                            tile_key,
1263                        )
1264                        .await,
1265                    )
1266                }
1267            }))
1268            .await;
1269            let mut delivered_tile_count = 0;
1270            let mut cache = decoded_tiles.lock().unwrap_or_else(PoisonError::into_inner);
1271            for (tile_key, fetched_tile) in fetched_tiles {
1272                if fetched_tile.is_ok() {
1273                    delivered_tile_count += 1;
1274                }
1275                cache.finish_fetch(tile_key, fetched_tile);
1276                claim_guard.mark_finished(tile_key);
1277            }
1278            delivered_tile_count
1279        })
1280    }

1282    /// Resolves one frame from the decoded cache alone — never the network:
1283    /// every cached tile is styled at the camera zoom's bucket (from the
1284    /// styled cache when the bucket or the style is unchanged) and its labels
1285    /// read, an address whose prefetch is running counts as pending, and a
1286    /// recorded failure counts as failed with its detail. A missing address
1287    /// whose ancestor is cached draws that ancestor's geometry once for the
1288    /// whole area it covers (the raster path's parent fallback), so the frame
1289    /// paints coarse detail instead of a hole.
1290    ///
1291    /// This is what a surface draws while a detached
1292    /// [`Self::prefetch_missing_tiles`] fills the cache, so a first frame over
1293    /// unvisited ground presents the style's background or the cached coarse
1294    /// detail instead of blocking on every tile's latency.
1295    pub fn cached_frame(
1296        &mut self,
1297        visible_tile_keys: &[TileKey],
1298        style_zoom_level: f64,
1299    ) -> VectorBaseFrame {
1300        // The untimed path applies a style change immediately: no transition
1301        // is started and any pending one is dropped, so its output is exactly
1302        // the settled style's.
1303        self.finish_style_transition();
1304        self.cached_frame_internal(visible_tile_keys, style_zoom_level, None, false)
1305    }

1311    /// lifetime elapses, after which the previous style is dropped.
1312    ///
1313    /// # Parameters
1314    /// - `visible_tile_keys` — the camera's visible tile addresses.
1315    /// - `style_zoom_level` — the camera's fractional zoom for paint
1316    ///   evaluation.
1317    /// - `frame_time_seconds` — a monotonic clock reading. A non-finite value

1327            style_zoom_level,
1328            frame_time_seconds,
1329            false,
1330        )
1331    }

1333    /// [`Self::cached_frame_at_time`] with the frame graded by camera motion.
1334    ///
1335    /// While `motion_graded` is true the camera is mid-gesture (a pan, zoom or
1336    /// rotate by input), so the frame is assembled entirely from what the
1337    /// caches already hold: decoded tiles, styled geometry at the last settled
1338    /// style-zoom bucket, and no label candidates. Styling a tile the camera
1339    /// has newly revealed still happens once, because there is no earlier
1340    /// geometry to reuse; what is skipped is the repeated assembly — the
1341    /// per-bucket restyle a continuous zoom would otherwise pay every frame
1342    /// and the per-tile label extraction plus placement the settled frame
1343    /// resolves. A style transition does not blend on a graded frame for the
1344    /// same reason; it resumes (or completes) at the settle.
1345    ///
1346    /// The next call with `motion_graded` false is the settle: it resolves at
1347    /// the camera's own style bucket and carries the frame's labels, exactly
1348    /// as a direct [`Self::cached_frame_at_time`] at that pose would.
1349    ///
1350    /// # Parameters
1351    /// - `visible_tile_keys` — the camera's visible tile addresses.
1352    /// - `style_zoom_level` — the camera's fractional zoom for paint
1353    ///   evaluation.
1354    /// - `frame_time_seconds` — a monotonic clock reading.
1355    /// - `motion_graded` — true while a camera gesture is active.
1356    pub fn cached_frame_at_time_in_motion(
1357        &mut self,
1358        visible_tile_keys: &[TileKey],
1359        style_zoom_level: f64,
1360        frame_time_seconds: f64,
1361        motion_graded: bool,
1362    ) -> VectorBaseFrame {
1363        let transition = self.active_style_transition(frame_time_seconds);
1364        self.cached_frame_internal(
1365            visible_tile_keys,
1366            style_zoom_level,
1367            transition.as_ref(),
1368            motion_graded,
1369        )
1370    }

1372    /// The transition a timed frame should blend with, or `None` when none is
1373    /// running. Starts the clock on the first timed frame after a change and
1374    /// drops the previous style once the change's lifetime has elapsed.
1375    fn active_style_transition(&mut self, frame_time_seconds: f64) -> Option<(CompiledStyle, f64)> {
1376        self.previous_style.as_ref()?;
1377        if !frame_time_seconds.is_finite() {
1378            self.finish_style_transition();
1379            return None;
1380        }
1381        let started_at_seconds = *self
1382            .transition_started_at_seconds
1383            .get_or_insert(frame_time_seconds);
1384        let elapsed_seconds = (frame_time_seconds - started_at_seconds).max(0.0);
1385        if elapsed_seconds >= self.style.transition_lifetime_seconds() {
1386            self.finish_style_transition();
1387            return None;
1388        }
1389        self.previous_style
1390            .clone()
1391            .map(|previous_style| (previous_style, elapsed_seconds))
1392    }

1394    /// Drops any in-flight style transition so the next frame draws the new
1395    /// style's values immediately.
1396    fn finish_style_transition(&mut self) {
1397        self.previous_style = None;
1398        self.transition_started_at_seconds = None;
1399    }

1401    /// The shared frame-resolution body. `transition` carries the previous
1402    /// style and the elapsed seconds while a change blends; `None` styles
1403    /// from the current style alone and uses the styled-geometry cache.
1404    /// `motion_graded` freezes the style-zoom bucket at the last settled
1405    /// frame's, drops any transition blend and skips label assembly.
1406    fn cached_frame_internal(
1407        &mut self,
1408        visible_tile_keys: &[TileKey],
1409        style_zoom_level: f64,
1410        transition: Option<&(CompiledStyle, f64)>,
1411        motion_graded: bool,
1412    ) -> VectorBaseFrame {
1413        let mut requested_tile_keys = visible_tile_keys.to_vec();
1414        requested_tile_keys.sort();
1415        requested_tile_keys.dedup();
1416        let requested_style_zoom = quantized_style_zoom(style_zoom_level);
1417        // A graded frame paints from the geometry the last settled frame
1418        // resolved, so a continuous zoom gesture does not restyle every tile
1419        // once per style bucket. A style transition cannot reuse cached
1420        // geometry (its paint changes every frame), so a graded frame paints
1421        // the current style's cache and the blend resumes at the settle.
1422        let quantized_style_zoom = if motion_graded {
1423            // Freeze on the last settled bucket; a gesture that began before
1424            // any frame settled freezes on this frame's own bucket instead,
1425            // so the following graded frames still reuse one style pass.
1426            *self
1427                .settled_style_zoom_bucket
1428                .get_or_insert(requested_style_zoom)
1429        } else {
1430            requested_style_zoom
1431        };
1432        let effective_transition = if motion_graded { None } else { transition };
1433        let evaluated_style_zoom =
1434            quantized_style_zoom as f64 / VECTOR_BASE_STYLE_ZOOM_QUANTIZATION_STEPS_PER_LEVEL;
1435        let sprite_atlas = self.resolved_sprite_atlas();
1436        let mut frame = VectorBaseFrame {
1437            overlays: Vec::new(),
1438            label_candidates: Vec::new(),
1439            sprite_atlas,
1440            requested_tile_count: requested_tile_keys.len(),
1441            drawn_tile_count: 0,
1442            fallback_tile_count: 0,
1443            failed_tile_count: 0,
1444            pending_tile_count: 0,
1445            style_id: self.style_id.clone(),
1446            background: self.style.background_at(evaluated_style_zoom),
1447            last_failure_detail: None,
1448            motion_graded,
1449            diagnostics: VectorBaseFrameDiagnostics::default(),
1450        };
1451        // Read the cache's verdict for every address under one lock, then
1452        // style from the cloned tiles without holding it.
1453        let (cached_tiles, in_flight_fetch_count) = {
1454            let decoded_tiles = self
1455                .decoded_tiles
1456                .lock()
1457                .unwrap_or_else(PoisonError::into_inner);
1458            let cached_tiles = requested_tile_keys
1459                .iter()
1460                .copied()
1461                .map(|tile_key| {
1462                    (
1463                        tile_key,
1464                        decoded_tiles.get(&tile_key),
1465                        decoded_tiles.failure_details.get(&tile_key).cloned(),
1466                    )
1467                })
1468                .collect::<Vec<(TileKey, Option<Arc<VectorTile>>, Option<String>)>>();
1469            (cached_tiles, decoded_tiles.in_flight.len())
1470        };
1471        let mut drawable_tiles: Vec<(TileKey, Arc<VectorTile>)> = Vec::new();
1472        let mut missing_tile_keys = Vec::new();
1473        for (tile_key, decoded_tile, failure_detail) in cached_tiles {
1474            match decoded_tile {
1475                Some(decoded_tile) => drawable_tiles.push((tile_key, decoded_tile)),
1476                None => {
1477                    if let Some(failure_detail) = failure_detail {
1478                        frame.failed_tile_count += 1;
1479                        frame.last_failure_detail = Some(failure_detail);
1480                    } else {
1481                        // Neither cached nor failed: either its prefetch is
1482                        // running or the caller has not requested it yet. Both
1483                        // are pending detail a later frame draws, so the frame
1484                        // reports them honestly and keeps presenting.
1485                        frame.pending_tile_count += 1;
1486                    }
1487                    missing_tile_keys.push(tile_key);
1488                }
1489            }
1490        }
1491        frame.drawn_tile_count = drawable_tiles.len();
1492        // A missing tile whose nearest cached ancestor exists is painted from
1493        // that ancestor once per ancestor, not once per child: one coarse tile
1494        // covers four missing children without stacking its translucent fills
1495        // four times.
1496        let (fallback_ancestors, covered_tile_count) = {
1497            let decoded_tiles = self
1498                .decoded_tiles
1499                .lock()
1500                .unwrap_or_else(PoisonError::into_inner);
1501            let mut fallback_ancestors: Vec<(TileKey, Arc<VectorTile>)> = Vec::new();
1502            let mut seen_ancestor_tile_keys = HashSet::new();
1503            let mut covered_tile_count = 0;
1504            for missing_tile_key in &missing_tile_keys {
1505                let mut ancestor_tile_key = missing_tile_key.parent_tile_key();
1506                while let Some(candidate_tile_key) = ancestor_tile_key {
1507                    let Some(ancestor_tile) = decoded_tiles.get(&candidate_tile_key) else {
1508                        ancestor_tile_key = candidate_tile_key.parent_tile_key();
1509                        continue;
1510                    };
1511                    covered_tile_count += 1;
1512                    if seen_ancestor_tile_keys.insert(candidate_tile_key) {
1513                        fallback_ancestors.push((candidate_tile_key, ancestor_tile));
1514                    }
1515                    break;
1516                }
1517            }
1518            (fallback_ancestors, covered_tile_count)
1519        };
1520        frame.fallback_tile_count = covered_tile_count;
1521        let drawable_tile_keys = drawable_tiles
1522            .iter()
1523            .map(|(tile_key, _)| *tile_key)
1524            .collect::<HashSet<_>>();
1525        // Style layers paint in global document order across the whole visible
1526        // tile set: bucket every tile's primitives by compiled layer index and
1527        // flatten the buckets at the end, so a later tile's layer-0 fill can
1528        // never paint over an earlier tile's layer-1 line.
1529        let mut layer_batches: Vec<Vec<MapOverlay>> = vec![Vec::new(); self.style.layer_count()];
1530        // Coarse stand-ins paint beneath the frame's own tiles within every
1531        // layer, so a sibling drawn at full detail is never overpainted by its
1532        // parent's coarse fills. An ancestor already drawn at its own address
1533        // is skipped: it is in the list once already.
1534        for (ancestor_tile_key, ancestor_tile) in fallback_ancestors {
1535            if drawable_tile_keys.contains(&ancestor_tile_key) {
1536                continue;
1537            }
1538            self.append_styled_tile_contents(
1539                &mut frame,
1540                &mut layer_batches,
1541                ancestor_tile_key,
1542                &ancestor_tile,
1543                StyledTileFrame {
1544                    quantized_style_zoom,
1545                    evaluated_style_zoom,
1546                    transition: effective_transition,
1547                    assemble_labels: !motion_graded,
1548                },
1549            );
1550        }
1551        for (tile_key, decoded_tile) in drawable_tiles {
1552            self.append_styled_tile_contents(
1553                &mut frame,
1554                &mut layer_batches,
1555                tile_key,
1556                &decoded_tile,
1557                StyledTileFrame {
1558                    quantized_style_zoom,
1559                    evaluated_style_zoom,
1560                    transition: effective_transition,
1561                    assemble_labels: !motion_graded,
1562                },
1563            );
1564        }
1565        frame.overlays = layer_batches.into_iter().flatten().collect();
1566        // The settle frame's bucket is what a following gesture freezes on;
1567        // a graded frame never moves it.
1568        if !motion_graded {
1569            self.settled_style_zoom_bucket = Some(quantized_style_zoom);
1570        }
1571        if self.first_drawn_tile_milliseconds.is_none()
1572            && (frame.drawn_tile_count > 0 || frame.fallback_tile_count > 0)
1573        {
1574            self.first_drawn_tile_milliseconds =
1575                Some(u64::try_from(self.installed_at.elapsed().as_millis()).unwrap_or(u64::MAX));
1576        }
1577        frame.diagnostics = VectorBaseFrameDiagnostics {
1578            payload_cache_hit_count: self
1579                .payload_cache_counters
1580                .hit_count
1581                .load(Ordering::Relaxed),
1582            payload_cache_miss_count: self
1583                .payload_cache_counters
1584                .miss_count
1585                .load(Ordering::Relaxed),
1586            in_flight_fetch_count,
1587            time_to_first_drawn_tile_milliseconds: self.first_drawn_tile_milliseconds,
1588        };
1589        self.last_failure_detail = frame.last_failure_detail.clone();
1590        tracing::debug!(
1591            target: "vector_base",
1592            requested_tile_count = frame.requested_tile_count,
1593            drawn_tile_count = frame.drawn_tile_count,
1594            fallback_tile_count = frame.fallback_tile_count,
1595            failed_tile_count = frame.failed_tile_count,
1596            pending_tile_count = frame.pending_tile_count,
1597            payload_cache_hit_count = frame.diagnostics.payload_cache_hit_count,
1598            payload_cache_miss_count = frame.diagnostics.payload_cache_miss_count,
1599            in_flight_fetch_count = frame.diagnostics.in_flight_fetch_count,
1600            time_to_first_drawn_tile_milliseconds =
1601                frame.diagnostics.time_to_first_drawn_tile_milliseconds,
1602            "vector base frame resolved"
1603        );
1604        frame
1605    }

1607    /// Appends one tile's styled overlays and label candidates to the frame
1608    /// at the camera zoom's style bucket, styling the tile into the styled
1609    /// cache first when the bucket or the style is new. Shared by the tiles
1610    /// drawn at their own address and by the ancestors that stand in for
1611    /// missing children.
1612    fn append_styled_tile_contents(
1613        &mut self,
1614        frame: &mut VectorBaseFrame,
1615        layer_batches: &mut [Vec<MapOverlay>],
1616        tile_key: TileKey,
1617        decoded_tile: &VectorTile,
1618        styling: StyledTileFrame<'_>,
1619    ) {
1620        let tile_address = VectorTileAddress {
1621            zoom_level: tile_key.zoom_level,
1622            tile_x: tile_key.tile_x,
1623            tile_y: tile_key.tile_y,
1624        };
1625        if let Some((previous_style, elapsed_seconds)) = styling.transition {
1626            // A blending frame's paint differs every frame, so it never
1627            // enters the styled-geometry cache; the settled style does.
1628            let styled_layers = vector_tile_overlay_layers_at_zoom_with_transition(
1629                decoded_tile,
1630                tile_address,
1631                &self.style,
1632                previous_style,
1633                *elapsed_seconds,
1634                styling.evaluated_style_zoom,
1635            );
1636            for (layer_batch, styled_overlays) in layer_batches.iter_mut().zip(styled_layers) {
1637                layer_batch.extend(styled_overlays);
1638            }
1639        } else {
1640            let styled_key = StyledTileCacheKey {
1641                tile_key,
1642                style_revision: self.style_revision,
1643                quantized_style_zoom: styling.quantized_style_zoom,
1644            };
1645            if !self.styled_tiles.contains_key(&styled_key) {
1646                let styled_layers = vector_tile_overlay_layers_at_zoom(
1647                    decoded_tile,
1648                    tile_address,
1649                    &self.style,
1650                    styling.evaluated_style_zoom,
1651                );
1652                self.store_styled_tile(styled_key, styled_layers);
1653            }
1654            if let Some(styled_layers) = self.styled_tiles.get(&styled_key) {
1655                for (layer_batch, styled_overlays) in layer_batches.iter_mut().zip(styled_layers) {
1656                    layer_batch.extend(styled_overlays.iter().cloned());
1657                }
1658            }
1659        }
1660        // Labels read from the same decoded tile at the same evaluated zoom,
1661        // so a frame's text and geometry always agree on the camera pose. The
1662        // walk is GPU-free and allocates nothing when the style declares no
1663        // symbol layer; it is recomputed per frame rather than cached because
1664        // it is cheap beside the styled geometry and shares no state with the
1665        // overlay cache. A motion-graded frame skips it entirely and lets the
1666        // settle resolve the labels.
1667        if styling.assemble_labels {
1668            frame
1669                .label_candidates
1670                .extend(label_candidates_for_vector_tile(
1671                    decoded_tile,
1672                    VectorTileAddress {
1673                        zoom_level: tile_key.zoom_level,
1674                        tile_x: tile_key.tile_x,
1675                        tile_y: tile_key.tile_y,
1676                    },
1677                    &self.style,
1678                    styling.evaluated_style_zoom,
1679                ));
1680        }
1681    }

1683    /// The decoded tile for the address, consulted through the payload cache
1684    /// first and fetched from the provider only on a miss. Kept free of the
1685    /// scheduler's caches so the prefetch can fetch every missing tile
1686    /// concurrently and insert the results afterwards.
1687    async fn fetch_and_decode_vector_tile(
1688        provider: &dyn VectorTileProvider,
1689        payload_cache: Option<&dyn CachedResponseStore<VectorTileData>>,
1690        payload_cache_source_name: &str,
1691        payload_cache_time_to_live_seconds: u64,
1692        payload_cache_counters: &VectorBasePayloadCacheCounters,
1693        tile_key: TileKey,
1694    ) -> Result<Arc<VectorTile>, String> {
1695        let payload_cache_fingerprint =
1696            vector_tile_payload_cache_fingerprint(payload_cache_source_name, tile_key);
1697        let decode = |tile_data: &VectorTileData| -> Result<Arc<VectorTile>, String> {
1698            Ok(Arc::new(decode_wire_vector_tile(tile_data).map_err(
1699                |decode_error| {
1700                    format!(
1701                        "the vector base tile {}/{}/{} failed to decode: {decode_error}",
1702                        tile_key.zoom_level, tile_key.tile_x, tile_key.tile_y
1703                    )
1704                },
1705            )?))
1706        };
1707        if let Some(payload_cache) = payload_cache {
1708            match payload_cache.lookup_response(&payload_cache_fingerprint) {
1709                Ok(Some(cached_payload)) => {
1710                    payload_cache_counters
1711                        .hit_count
1712                        .fetch_add(1, Ordering::Relaxed);
1713                    return decode(cached_payload.as_ref());
1714                }
1715                // A miss is a fetch; a store read failure degrades to the
1716                // same path (the cache is an accelerator, never a gate).
1717                Ok(None) | Err(_) => {
1718                    payload_cache_counters
1719                        .miss_count
1720                        .fetch_add(1, Ordering::Relaxed);
1721                }
1722            }
1723        }
1724        let tile_data = provider
1725            .fetch_vector_tile(tile_key.tile_x, tile_key.tile_y, tile_key.zoom_level)
1726            .await
1727            .map_err(|fetch_error| {
1728                format!(
1729                    "the vector base tile {}/{}/{} failed to fetch: {fetch_error}",
1730                    tile_key.zoom_level, tile_key.tile_x, tile_key.tile_y
1731                )
1732            })?;
1733        let decoded_tile = decode(&tile_data)?;
1734        if let Some(payload_cache) = payload_cache {
1735            // Storage is best-effort: a failing store must not lose the
1736            // freshly fetched tile.
1737            let _ = payload_cache.store_response(
1738                &payload_cache_fingerprint,
1739                Arc::new(tile_data),
1740                payload_cache_time_to_live_seconds,
1741            );
1742        }
1743        Ok(decoded_tile)
1744    }
1745
1746    /// Inserts a decoded tile and evicts the oldest decoded tiles beyond the
1747    /// cache's capacity. Styled geometry is keyed by the address and a style
1748    /// revision and is bounded by its own cache, so decoded-tile eviction
1749    /// never touches it.
1750    ///
1751    /// The decoded-tile eviction below drops a tile's styled entries, but a
1752    /// tile stays decoded for many frames while the camera may style it once
1753    /// per zoom bucket; this cache is therefore bounded on its own too.
1754    fn store_styled_tile(
1755        &mut self,
1756        styled_key: StyledTileCacheKey,
1757        overlays: Vec<Vec<MapOverlay>>,
1758    ) {
1759        if self.styled_tiles.insert(styled_key, overlays).is_none() {
1760            self.styled_tile_insertion_order.push_back(styled_key);
1761        }
1762        while self.styled_tiles.len() > self.maximum_cached_styled_tile_count {
1763            let Some(evicted_styled_key) = self.styled_tile_insertion_order.pop_front() else {
1764                break;
1765            };
1766            self.styled_tiles.remove(&evicted_styled_key);
1767        }
1768    }
1769}

1777/// Unwraps the wire envelope and decodes the tile: a gzip-wrapped body is
1778/// decompressed first, so the decoder only ever sees plain MVT protobuf. The
1779/// record is borrowed because the caller keeps the payload for its cache.
1780pub(crate) fn decode_wire_vector_tile(tile_data: &VectorTileData) -> Result<VectorTile, String> {
1781    let plain_tile_bytes = match tile_data.tile_encoding {
1782        VectorTileEncoding::MapboxVectorTile => tile_data.tile_bytes.clone(),
1783        VectorTileEncoding::GzippedMapboxVectorTile => {
1784            let gzip_decoder = GzDecoder::new(tile_data.tile_bytes.as_slice());
1785            let mut unwrapped_bytes = Vec::new();
1786            let mut limited_decoder =
1787                gzip_decoder.take(MAXIMUM_DECOMPRESSED_VECTOR_TILE_BYTES as u64 + 1);
1788            limited_decoder
1789                .read_to_end(&mut unwrapped_bytes)
1790                .map_err(|decompress_error| {
1791                    format!(
1792                        "the gzip-wrapped vector tile payload failed to decompress: \
1793                         {decompress_error}"
1794                    )
1795                })?;
1796            if unwrapped_bytes.len() > MAXIMUM_DECOMPRESSED_VECTOR_TILE_BYTES {
1797                return Err(format!(
1798                    "the gzip-wrapped vector tile payload decompressed beyond the {MAXIMUM_DECOMPRESSED_VECTOR_TILE_BYTES}-byte limit"
1799                ));
1800            }
1801            unwrapped_bytes
1802        }
1803    };
1804    decode_vector_tile(&plain_tile_bytes).map_err(|decode_error| decode_error.to_string())
1805}
