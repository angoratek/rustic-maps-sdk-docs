1//! Decode bounds shared by every image payload the SDK decodes.
2//!
3//! Raster tiles, sprite sheets and recorded tile payloads are untrusted
4//! input: a small compressed image can declare dimensions that expand to
5//! gigabytes when decoded. [`decode_image_with_limits`] therefore applies
6//! the named bounds below at every decode site instead of relying on the
7//! decoder's own default allocation cap.
8
9use std::io::Cursor;
10
11use image::{DynamicImage, ImageError, ImageReader, Limits};
12
13/// The widest image the SDK decodes, in pixels.
14///
15/// Raster tiles are 256 or 512 pixels wide and the largest sprite sheets the
16/// style specification allows are a few thousand pixels wide; the bound
17/// leaves room for those while refusing pathological declarations.
18pub const MAXIMUM_IMAGE_WIDTH_PIXELS: u32 = 8192;
19
20/// The tallest image the SDK decodes, in pixels.
21pub const MAXIMUM_IMAGE_HEIGHT_PIXELS: u32 = 8192;
22
23/// The most memory a single decode may allocate, in bytes.
24///
25/// Decoded RGBA is four bytes per pixel, so this bound refuses (for example)
26/// a 8192 by 8192 image even though it satisfies the dimension bounds. It
27/// also covers any intermediate buffers the decoder allocates.
28pub const MAXIMUM_IMAGE_ALLOCATION_BYTES: u64 = 64 * 1024 * 1024;
29
30/// Why an encoded image payload could not be decoded.
31#[derive(Debug)]
32pub enum ImageDecodeError {
33    /// The payload declares dimensions, or needs an allocation, beyond the
34    /// bounds named in this module.
35    LimitsExceeded(image::error::LimitError),
36    /// The bytes are not a supported image or do not decode.
37    Malformed(ImageError),
38}
39
40impl std::fmt::Display for ImageDecodeError {
41    fn fmt(&self, formatter: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
42        match self {
43            Self::LimitsExceeded(limit_error) => {
44                write!(
45                    formatter,
46                    "the image exceeds the decode limits: {limit_error}"
47                )
48            }
49            Self::Malformed(image_error) => {
50                write!(formatter, "the image could not be decoded: {image_error}")
51            }
52        }
53    }
54}
55
56impl std::error::Error for ImageDecodeError {
57    fn source(&self) -> Option<&(dyn std::error::Error + 'static)> {
58        match self {
59            Self::LimitsExceeded(limit_error) => Some(limit_error),
60            Self::Malformed(image_error) => Some(image_error),
61        }
62    }
63}
64
65impl From<ImageError> for ImageDecodeError {
66    fn from(image_error: ImageError) -> Self {
67        match image_error {
68            ImageError::Limits(limit_error) => Self::LimitsExceeded(limit_error),
69            other_image_error => Self::Malformed(other_image_error),
70        }
71    }
72}
73
74/// The decode bounds this module declares, as the decoder's limit structure.
75fn decode_limits() -> Limits {
76    // The structure is non-exhaustive, so the bounds are set on a
77    // fully-populated value rather than through a struct expression.
78    let mut limits = Limits::no_limits();
79    limits.max_image_width = Some(MAXIMUM_IMAGE_WIDTH_PIXELS);
80    limits.max_image_height = Some(MAXIMUM_IMAGE_HEIGHT_PIXELS);
81    limits.max_alloc = Some(MAXIMUM_IMAGE_ALLOCATION_BYTES);
82    limits
83}
84
85/// Decodes an encoded image payload within the bounds named in this module.
86///
87/// The format is guessed from the payload's own signature. A payload outside
88/// the bounds is refused with [`ImageDecodeError::LimitsExceeded`] before the
89/// decoder allocates its pixels; a payload that does not decode is reported
90/// as [`ImageDecodeError::Malformed`].
91///
92/// # Errors
93/// Reports [`ImageDecodeError`] when the payload exceeds the decode bounds or
94/// is not a supported image.
95pub fn decode_image_with_limits(
96    encoded_image_bytes: &[u8],
97) -> Result<DynamicImage, ImageDecodeError> {
98    let reader = ImageReader::new(Cursor::new(encoded_image_bytes))
99        .with_guessed_format()
100        .map_err(ImageError::from)?;
101    let mut limited_reader = reader;
102    limited_reader.limits(decode_limits());
103    limited_reader.decode().map_err(ImageDecodeError::from)
104}

106#[cfg(test)]
107mod tests {
108    use super::*;
109    use image::{ImageFormat, Rgba, RgbaImage};
110
111    /// Encodes a solid image of the given size as PNG.
112    fn encoded_png(width_pixels: u32, height_pixels: u32) -> Vec<u8> {
113        let image = RgbaImage::from_pixel(width_pixels, height_pixels, Rgba([0, 0, 0, 255]));
114        let mut encoded = Cursor::new(Vec::new());
115        DynamicImage::ImageRgba8(image)
116            .write_to(&mut encoded, ImageFormat::Png)
117            .expect("the test PNG encodes");
118        encoded.into_inner()
119    }
120
121    #[test]
122    fn an_image_within_the_limits_decodes() {
123        let decoded = decode_image_with_limits(&encoded_png(2, 2))
124            .expect("a 2 by 2 image is inside every decode bound");
125        assert_eq!((decoded.width(), decoded.height()), (2, 2));
126    }
127
128    #[test]
129    fn an_over_wide_image_is_refused_with_the_limits_failure() {
130        let over_wide = encoded_png(MAXIMUM_IMAGE_WIDTH_PIXELS + 1, 1);
131        match decode_image_with_limits(&over_wide) {
132            Err(ImageDecodeError::LimitsExceeded(limit_error)) => assert_eq!(
133                limit_error.kind(),
134                image::error::LimitErrorKind::DimensionError
135            ),
136            other => {
137                panic!("an over-wide image must be refused by the dimension bound, got {other:?}")
138            }
139        }
140    }
141
142    #[test]
143    fn an_image_over_the_allocation_bound_is_refused_rather_than_allocated() {
144        // 4097 by 4097 RGBA needs 67 MB, over the 64 MB allocation bound but
145        // inside the dimension bounds, so this exercises the second bound.
146        let allocation_over = encoded_png(4097, 4097);
147        match decode_image_with_limits(&allocation_over) {
148            Err(ImageDecodeError::LimitsExceeded(limit_error)) => assert_eq!(
149                limit_error.kind(),
150                image::error::LimitErrorKind::InsufficientMemory
151            ),
152            other => panic!("an image over the allocation bound must be refused, got {other:?}"),
153        }
154    }
