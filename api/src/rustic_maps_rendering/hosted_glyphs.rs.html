1//! Glyph ranges for a hosted vector style, fetched through the provider's
2//! authenticated style-resource client.
3//!
4//! A style document published by a tileset's provider names its glyph
5//! endpoint as a template (`…/{fontstack}/{range}.pbf`). The ranges are only
6//! served to a request carrying the provider's credential, so the fetch goes
7//! through [`StyleResourceClient`] and the credential never reaches this
8//! crate. The template resolves against the style document's own address, and
9//! the result must land on the document's origin or on a host the provider
10//! declared: any other host is refused before a request is made, because a
11//! style document must not be able to aim the provider credential at a host
12//! the provider did not declare.
13//!
14//! Decoded ranges are retained in a [`GlyphRangeCache`] keyed by the
15//! provider's namespace, the font stack and the range, so two providers
16//! serving the same font-stack name never share a range and a repeated frame
17//! costs no request.
18

31/// The glyph URL template's font-stack placeholder.
32pub const GLYPH_URL_FONT_STACK_PLACEHOLDER: &str = "{fontstack}";
33
34/// The glyph URL template's range placeholder.
35pub const GLYPH_URL_RANGE_PLACEHOLDER: &str = "{range}";
36
37/// The characters a font-stack path segment keeps unescaped. The comma
38/// separates the names of one stack, as the glyph protocol defines.
39const FONT_STACK_SAFE_CHARACTERS: &[u8] = b"-._~,";
40
41/// The hexadecimal digits percent-encoding uses.
42const PERCENT_ENCODING_HEX_DIGITS: &[u8; 16] = b"0123456789ABCDEF";
43
44/// Fetches and retains one hosted style's glyph ranges.
45///
46/// The fetch goes through the provider's [`StyleResourceClient`], so the
47/// credential, transport, pacing and retry policy are the provider layer's.
48pub struct HostedGlyphRanges {
49    style_resources: Arc<dyn StyleResourceClient>,
50    namespace: String,
51    glyph_url_template: String,
52    cache: GlyphRangeCache,
53}
54
55impl HostedGlyphRanges {
56    /// Builds a glyph source for one provider's style document.
57    ///
58    /// # Parameters
59    /// - `style_resources` — the provider's authenticated style-resource
60    ///   client.
61    /// - `namespace` — the provider key that namespaces the cached ranges.
62    /// - `style_document_url` — the style document's address, whose origin and
63    ///   directory bound every glyph request.
64    /// - `glyph_url_template` — the document's `glyphs` template, carrying
65    ///   the font-stack and range placeholders, either absolute or relative to
66    ///   the document's own address.
67    /// - `declared_hosts` — the hosts the provider declared for its published
68    ///   style resources; a template may live on any of them as well as on the
69    ///   document's own origin.
70    ///
71    /// # Errors
72    /// Returns [`SDKError::ResponseParsingFailure`] when the template lacks a
73    /// placeholder or does not resolve to an absolute `http`/`https` URL, and
74    /// [`SDKError::ProviderRequestRejected`] when the resolved template names
75    /// a host that is neither the document's origin nor one the provider
76    /// declared.
77    pub fn new(
78        style_resources: Arc<dyn StyleResourceClient>,
79        namespace: impl Into<String>,
80        style_document_url: &str,
81        glyph_url_template: &str,
82        declared_hosts: &[String],
83    ) -> Result<Self, SDKError> {
84        let template = glyph_url_template.trim();
85        for placeholder in [
86            GLYPH_URL_FONT_STACK_PLACEHOLDER,
87            GLYPH_URL_RANGE_PLACEHOLDER,
88        ] {
89            if !template.contains(placeholder) {
90                return Err(SDKError::ResponseParsingFailure(format!(
91                    "the style's glyph address {template} does not carry the \
92                     {placeholder} placeholder"
93                )));
94            }
95        }
96        let resolved_template =
97            resolve_style_resource_address(style_document_url, template, declared_hosts, "glyph")?;
98        Ok(Self {
99            style_resources,
100            namespace: namespace.into(),
101            glyph_url_template: resolved_template,
102            cache: GlyphRangeCache::new(GlyphRangeCacheConfiguration::default()),
103        })
104    }
105
106    /// The provider namespace the cached ranges belong to.
107    pub fn namespace(&self) -> &str {
108        &self.namespace
109    }
110
111    /// The address one range request names, with the font stack escaped and
112    /// the range label substituted.
113    pub fn resolved_url(&self, font_stack_name: &str, range: GlyphRange) -> String {
114        self.glyph_url_template
115            .replace(
116                GLYPH_URL_FONT_STACK_PLACEHOLDER,
117                &escaped_font_stack(font_stack_name),
118            )
119            .replace(GLYPH_URL_RANGE_PLACEHOLDER, &range.url_segment())
120    }
121
122    /// The number of cached ranges, for diagnostics and tests.
123    pub fn cached_range_count(&self) -> usize {
124        self.cache.len()
125    }
126
127    /// The Basic Latin range (`U+0000`–`U+00FF`) of one font stack, fetched
128    /// on first use and cached per provider, font stack and range.
129    ///
130    /// # Errors
131    /// Returns the client's error when the request fails and
132    /// [`SDKError::ResponseParsingFailure`] when the payload is not one glyph
133    /// range in the glyph protocol.
134    pub async fn basic_latin_font_stack(
135        &mut self,
136        font_stack_name: &str,
137    ) -> Result<GlyphFontStack, SDKError> {
138        self.font_stack(font_stack_name, GlyphRange::BASIC_LATIN)
139            .await
140    }
141
142    /// Both Latin blocks (`U+0000`–`U+01FF`) of one font stack, fetched on
143    /// first use and combined into one stack the label atlas packs in a
144    /// single pass.
145    ///
146    /// # Errors
147    /// Returns the client's error when a request fails and
148    /// [`SDKError::ResponseParsingFailure`] when a payload is not one glyph
149    /// range in the glyph protocol.
150    pub async fn latin_font_stack(
151        &mut self,
152        font_stack_name: &str,
153    ) -> Result<GlyphFontStack, SDKError> {
154        let mut combined = self
155            .font_stack(font_stack_name, GlyphRange::BASIC_LATIN)
156            .await?;
157        let extended = self
158            .font_stack(font_stack_name, GlyphRange::LATIN_EXTENDED)
159            .await?;
160        combined.glyphs.extend(extended.glyphs);
161        Ok(combined)
162    }
163
164    /// One font stack's glyph range, fetched on first use and cached per
165    /// provider, font stack and range.
166    ///
167    /// # Errors
168    /// Returns the client's error when the request fails and
169    /// [`SDKError::ResponseParsingFailure`] when the payload is not one glyph
170    /// range in the glyph protocol.
171    pub async fn font_stack(
172        &mut self,
173        font_stack_name: &str,
174        range: GlyphRange,
175    ) -> Result<GlyphFontStack, SDKError> {
176        let key = GlyphRangeKey::new(self.namespace.clone(), font_stack_name, range);
177        if let Some(cached) = self.cache.get(&key) {
178            return Ok(cached.clone());
179        }
180        let url = self.resolved_url(font_stack_name, range);
181        let bytes = self.style_resources.fetch_style_resource(&url).await?;
182        let font_stack = decode_glyph_font_stack(&bytes).map_err(|decode_failure| {
183            SDKError::ResponseParsingFailure(format!(
184                "the glyph range {url} did not decode: {decode_failure}"
185            ))
186        })?;
187        self.cache.insert(key, font_stack.clone());
188        Ok(font_stack)
189    }
190}

192/// The origin of an absolute `http`/`https` address.
193pub(crate) fn web_origin(address: &str) -> Result<url::Origin, SDKError> {
194    let parsed = Url::parse(address).map_err(|parse_failure| {
195        SDKError::ResponseParsingFailure(format!(
196            "the style resource address {address} is not an absolute URL: {parse_failure}"
197        ))
198    })?;
199    if parsed.scheme() != "https" && parsed.scheme() != "http" {
200        return Err(SDKError::ResponseParsingFailure(format!(
201            "the style resource address {address} is not an http or https URL"
202        )));
203    }
204    Ok(parsed.origin())
205}

207/// Percent-encodes one font-stack path segment, keeping the characters the
208/// glyph protocol uses for the stack's own names.
209fn escaped_font_stack(font_stack_name: &str) -> String {
210    let mut escaped = String::with_capacity(font_stack_name.len());
211    for byte in font_stack_name.bytes() {
212        let is_unreserved =
213            byte.is_ascii_alphanumeric() || FONT_STACK_SAFE_CHARACTERS.contains(&byte);
214        if is_unreserved {
215            escaped.push(char::from(byte));
216        } else {
217            escaped.push('%');
218            escaped.push(char::from(
219                PERCENT_ENCODING_HEX_DIGITS[usize::from(byte >> 4)],
220            ));
221            escaped.push(char::from(
222                PERCENT_ENCODING_HEX_DIGITS[usize::from(byte & 0x0F)],
223            ));
224        }
225    }
226    escaped
227}

234    const DOCUMENT_URL: &str = "https://styles.example.test/published/style.json";
235    const TEMPLATE: &str = "https://styles.example.test/published/{fontstack}/{range}.pbf";
236    const BASIC_LATIN_FIXTURE_BYTES: &[u8] =
237        include_bytes!("../../../fixtures/glyphs/noto_sans_regular_0-255.pbf");
238    const LATIN_EXTENDED_FIXTURE_BYTES: &[u8] =
239        include_bytes!("../../../fixtures/glyphs/noto_sans_regular_256-511.pbf");

241    /// The committed fixture a glyph address answers with, keyed by its range
242    /// segment the way the glyph protocol serves it.
243    fn committed_range_bytes(url: &str) -> Vec<u8> {
244        if url.ends_with(&format!(
245            "/{}.pbf",
246            GlyphRange::LATIN_EXTENDED.url_segment()
247        )) {
248            LATIN_EXTENDED_FIXTURE_BYTES.to_vec()
249        } else {
250            BASIC_LATIN_FIXTURE_BYTES.to_vec()
251        }
252    }
253
254    /// A client that records every address and answers the committed range
255    /// the address names.
256    struct RecordingStyleResources {
257        requested_urls: Mutex<Vec<String>>,
258    }
259
260    impl RecordingStyleResources {
261        fn new() -> Arc<Self> {
262            Arc::new(Self {
263                requested_urls: Mutex::new(Vec::new()),
264            })
265        }
266
267        fn requested_urls(&self) -> Vec<String> {
268            self.requested_urls.lock().unwrap().clone()
269        }
270    }
271
272    #[async_trait::async_trait]
273    impl StyleResourceClient for RecordingStyleResources {
274        async fn fetch_style_resource(&self, url: &str) -> Result<Vec<u8>, SDKError> {
275            self.requested_urls.lock().unwrap().push(url.to_string());
276            Ok(committed_range_bytes(url))
277        }
278    }

280    fn glyph_source(style_resources: Arc<RecordingStyleResources>) -> HostedGlyphRanges {
281        HostedGlyphRanges::new(
282            style_resources,
283            "example-vendor",
284            DOCUMENT_URL,
285            TEMPLATE,
286            &[],
287        )
288        .expect("the committed template is on the document's origin")
289    }

304    #[tokio::test]
305    async fn one_range_is_fetched_once_and_then_served_from_the_cache() {
306        let resources = RecordingStyleResources::new();
307        let mut source = glyph_source(Arc::clone(&resources));
308        let first = source.basic_latin_font_stack("Noto Sans Regular").await;
309        let second = source.basic_latin_font_stack("Noto Sans Regular").await;
310        let first_range = first.expect("the committed fixture decodes");
311        assert!(!first_range.glyphs.is_empty(), "the range carries glyphs");
312        assert_eq!(first_range.range, GlyphRange::BASIC_LATIN.url_segment());
313        assert!(second.is_ok());
314        assert_eq!(resources.requested_urls().len(), 1);
315        assert_eq!(source.cached_range_count(), 1);
316    }

318    #[tokio::test]
319    async fn the_latin_font_stack_requests_both_ranges_and_combines_them() {
320        let resources = RecordingStyleResources::new();
321        let mut source = glyph_source(Arc::clone(&resources));
322        let combined = source
323            .latin_font_stack("Noto Sans Regular")
324            .await
325            .expect("both committed ranges decode");
326        let requested_urls = resources.requested_urls();
327        assert_eq!(requested_urls.len(), 2);
328        assert!(requested_urls[0].ends_with("/0-255.pbf"));

330        assert_eq!(
331            combined.glyphs.len(),
332            decode_glyph_font_stack(BASIC_LATIN_FIXTURE_BYTES)
333                .expect("the committed Basic Latin range decodes")
334                .glyphs
335                .len()
336                + decode_glyph_font_stack(LATIN_EXTENDED_FIXTURE_BYTES)
337                    .expect("the committed extended range decodes")
338                    .glyphs
339                    .len()
340        );
341    }
