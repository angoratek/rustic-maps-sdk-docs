1//! Glyph ranges for a hosted vector style, fetched through the provider's
2//! authenticated style-resource client.
3//!
4//! A style document published by a tileset's provider names its glyph
5//! endpoint as a template (`…/{fontstack}/{range}.pbf`). The ranges are only
6//! served to a request carrying the provider's credential, so the fetch goes
7//! through [`StyleResourceClient`] and the credential never reaches this
8//! crate. The template resolves against the style document's own address, and
9//! the result must land on the document's origin or on a host the provider
10//! declared: any other host is refused before a request is made, because a
11//! style document must not be able to aim the provider credential at a host
12//! the provider did not declare.
13//!
14//! Decoded ranges are retained in a [`GlyphRangeCache`] keyed by the
15//! provider's namespace, the font stack and the range, so two providers
16//! serving the same font-stack name never share a range and a repeated frame
17//! costs no request.
18

31/// The glyph URL template's font-stack placeholder.
32pub const GLYPH_URL_FONT_STACK_PLACEHOLDER: &str = "{fontstack}";
33
34/// The glyph URL template's range placeholder.
35pub const GLYPH_URL_RANGE_PLACEHOLDER: &str = "{range}";
36
37/// The characters a font-stack path segment keeps unescaped. The comma
38/// separates the names of one stack, as the glyph protocol defines.
39const FONT_STACK_SAFE_CHARACTERS: &[u8] = b"-._~,";
40
41/// The hexadecimal digits percent-encoding uses.
42const PERCENT_ENCODING_HEX_DIGITS: &[u8; 16] = b"0123456789ABCDEF";
43
44/// Fetches and retains one hosted style's glyph ranges.
45///
46/// The fetch goes through the provider's [`StyleResourceClient`], so the
47/// credential, transport, pacing and retry policy are the provider layer's.
48pub struct HostedGlyphRanges {
49    style_resources: Arc<dyn StyleResourceClient>,
50    namespace: String,
51    glyph_url_template: String,
52    cache: GlyphRangeCache,
53}
54
55impl HostedGlyphRanges {
56    /// Builds a glyph source for one provider's style document.
57    ///
58    /// # Parameters
59    /// - `style_resources` — the provider's authenticated style-resource
60    ///   client.
61    /// - `namespace` — the provider key that namespaces the cached ranges.
62    /// - `style_document_url` — the style document's address, whose origin and
63    ///   directory bound every glyph request.
64    /// - `glyph_url_template` — the document's `glyphs` template, carrying
65    ///   the font-stack and range placeholders, either absolute or relative to
66    ///   the document's own address.
67    /// - `declared_hosts` — the hosts the provider declared for its published
68    ///   style resources; a template may live on any of them as well as on the
69    ///   document's own origin.
70    ///
71    /// # Errors
72    /// Returns [`SDKError::ResponseParsingFailure`] when the template lacks a
73    /// placeholder or does not resolve to an absolute `http`/`https` URL, and
74    /// [`SDKError::ProviderRequestRejected`] when the resolved template names
75    /// a host that is neither the document's origin nor one the provider
76    /// declared.
77    pub fn new(
78        style_resources: Arc<dyn StyleResourceClient>,
79        namespace: impl Into<String>,
80        style_document_url: &str,
81        glyph_url_template: &str,
82        declared_hosts: &[String],
83    ) -> Result<Self, SDKError> {
84        let declared_template = glyph_url_template.trim();
85        // A provider's document may name the template in the vendor's own URI
86        // scheme; the provider owns that mapping, and the rewritten address is
87        // still resolved and scope-checked below.
88        let rewritten_template =
89            style_resources.resolve_declared_style_resource_address(declared_template);
90        let template = rewritten_template.as_deref().unwrap_or(declared_template);
91        for placeholder in [
92            GLYPH_URL_FONT_STACK_PLACEHOLDER,
93            GLYPH_URL_RANGE_PLACEHOLDER,
94        ] {
95            if !template.contains(placeholder) {
96                return Err(SDKError::ResponseParsingFailure(format!(
97                    "the style's glyph address {template} does not carry the \
98                     {placeholder} placeholder"
99                )));
100            }
101        }
102        let resolved_template =
103            resolve_style_resource_address(style_document_url, template, declared_hosts, "glyph")?;
104        Ok(Self {
105            style_resources,
106            namespace: namespace.into(),
107            glyph_url_template: resolved_template,
108            cache: GlyphRangeCache::new(GlyphRangeCacheConfiguration::default()),
109        })
110    }
111
112    /// The provider namespace the cached ranges belong to.
113    pub fn namespace(&self) -> &str {
114        &self.namespace
115    }
116
117    /// The address one range request names, with the font stack escaped and
118    /// the range label substituted.
119    pub fn resolved_url(&self, font_stack_name: &str, range: GlyphRange) -> String {
120        self.glyph_url_template
121            .replace(
122                GLYPH_URL_FONT_STACK_PLACEHOLDER,
123                &escaped_font_stack(font_stack_name),
124            )
125            .replace(GLYPH_URL_RANGE_PLACEHOLDER, &range.url_segment())
126    }
127
128    /// The number of cached ranges, for diagnostics and tests.
129    pub fn cached_range_count(&self) -> usize {
130        self.cache.len()
131    }

133    /// The Basic Latin range (`U+0000`–`U+00FF`) of one font stack, fetched
134    /// on first use and cached per provider, font stack and range.
135    ///
136    /// # Errors
137    /// Returns the client's error when the request fails and
138    /// [`SDKError::ResponseParsingFailure`] when the payload is not one glyph
139    /// range in the glyph protocol.
140    pub async fn basic_latin_font_stack(
141        &mut self,
142        font_stack_name: &str,
143    ) -> Result<GlyphFontStack, SDKError> {
144        self.font_stack(font_stack_name, GlyphRange::BASIC_LATIN)
145            .await
146    }
147
148    /// Both Latin blocks (`U+0000`–`U+01FF`) of one font stack, fetched on
149    /// first use and combined into one stack the label atlas packs in a
150    /// single pass.
151    ///
152    /// # Errors
153    /// Returns the client's error when a request fails and
154    /// [`SDKError::ResponseParsingFailure`] when a payload is not one glyph
155    /// range in the glyph protocol.
156    pub async fn latin_font_stack(
157        &mut self,
158        font_stack_name: &str,
159    ) -> Result<GlyphFontStack, SDKError> {
160        let mut combined = self
161            .font_stack(font_stack_name, GlyphRange::BASIC_LATIN)
162            .await?;
163        let extended = self
164            .font_stack(font_stack_name, GlyphRange::LATIN_EXTENDED)
165            .await?;
166        combined.glyphs.extend(extended.glyphs);
167        Ok(combined)
168    }
169
170    /// One font stack's glyph range, fetched on first use and cached per
171    /// provider, font stack and range.
172    ///
173    /// # Errors
174    /// Returns the client's error when the request fails and
175    /// [`SDKError::ResponseParsingFailure`] when the payload is not one glyph
176    /// range in the glyph protocol.
177    pub async fn font_stack(
178        &mut self,
179        font_stack_name: &str,
180        range: GlyphRange,
181    ) -> Result<GlyphFontStack, SDKError> {
182        let key = GlyphRangeKey::new(self.namespace.clone(), font_stack_name, range);
183        if let Some(cached) = self.cache.get(&key) {
184            return Ok(cached.clone());
185        }
186        let url = self.resolved_url(font_stack_name, range);
187        let bytes = self.style_resources.fetch_style_resource(&url).await?;
188        let font_stack = decode_glyph_font_stack(&bytes).map_err(|decode_failure| {
189            SDKError::ResponseParsingFailure(format!(
190                "the glyph range {url} did not decode: {decode_failure}"
191            ))
192        })?;
193        self.cache.insert(key, font_stack.clone());
194        Ok(font_stack)
195    }
196}

198/// The origin of an absolute `http`/`https` address.
199pub(crate) fn web_origin(address: &str) -> Result<url::Origin, SDKError> {
200    let parsed = Url::parse(address).map_err(|parse_failure| {
201        SDKError::ResponseParsingFailure(format!(
202            "the style resource address {address} is not an absolute URL: {parse_failure}"
203        ))
204    })?;
205    if parsed.scheme() != "https" && parsed.scheme() != "http" {
206        return Err(SDKError::ResponseParsingFailure(format!(
207            "the style resource address {address} is not an http or https URL"
208        )));
209    }
210    Ok(parsed.origin())
211}

213/// Percent-encodes one font-stack path segment, keeping the characters the
214/// glyph protocol uses for the stack's own names.
215fn escaped_font_stack(font_stack_name: &str) -> String {
216    let mut escaped = String::with_capacity(font_stack_name.len());
217    for byte in font_stack_name.bytes() {
218        let is_unreserved =
219            byte.is_ascii_alphanumeric() || FONT_STACK_SAFE_CHARACTERS.contains(&byte);
220        if is_unreserved {
221            escaped.push(char::from(byte));
222        } else {
223            escaped.push('%');
224            escaped.push(char::from(
225                PERCENT_ENCODING_HEX_DIGITS[usize::from(byte >> 4)],
226            ));
227            escaped.push(char::from(
228                PERCENT_ENCODING_HEX_DIGITS[usize::from(byte & 0x0F)],
229            ));
230        }
231    }
232    escaped
233}

247    /// The committed fixture a glyph address answers with, keyed by its range
248    /// segment the way the glyph protocol serves it.
249    fn committed_range_bytes(url: &str) -> Vec<u8> {
250        if url.ends_with(&format!(
251            "/{}.pbf",
252            GlyphRange::LATIN_EXTENDED.url_segment()
253        )) {
254            LATIN_EXTENDED_FIXTURE_BYTES.to_vec()
255        } else {
256            BASIC_LATIN_FIXTURE_BYTES.to_vec()
257        }
258    }

312        let resources = RecordingStyleResources::new();
313        let mut source = glyph_source(Arc::clone(&resources));
314        let first = source.basic_latin_font_stack("Noto Sans Regular").await;
315        let second = source.basic_latin_font_stack("Noto Sans Regular").await;
316        let first_range = first.expect("the committed fixture decodes");
