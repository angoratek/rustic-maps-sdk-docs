1//! Refapp-only UniFFI bindings over the shared reference-application math.
2//!
3//! The Kotlin and Swift reference applications call the same follow-camera,
4//! course-median, easing, scale-bar and trace-cursor values instead of keeping
5//! per-platform mirrors. Every member delegates to [`map_refapp_core`]; this
6//! crate only lowers the shared types across the UniFFI boundary.
7//!
8//! The crate is deliberately outside every published package: it is not a
9//! dependency of `rustic-maps-native-core`, and `scripts/build-android.sh` and
10//! `scripts/build-ios.sh` assert that the AAR and the XCFramework file sets do
11//! not carry it.
12//!
13//! # Re-entrancy and threading
14//!
15//! The UniFFI scaffolding calls every interface method through `&self`, while
16//! the shared state transitions need `&mut self`. Each wrapper therefore holds
17//! its delegate behind a mutex and releases the lock before returning, so a
18//! call can never hold two wrappers' locks at once. The methods are
19//! synchronous and never await while a lock is held.
20
21// The scaffolding generated from the UDL builds its metadata in large const
22// arrays (uniffi 0.31's codegen pattern); the attribute cannot scope into the
23// machine-generated macro expansion, so the lint is allowed crate-wide.
24#![allow(clippy::large_const_arrays)]
25
26use std::sync::{Mutex, MutexGuard};

30/// Locks a wrapper's delegate, recovering the value from a poisoned lock.
31///
32/// The shared math has no partial-update path that could leave a value
33/// inconsistent, so a panic in another thread cannot corrupt the delegate;
34/// recovering keeps the demo usable instead of failing every later call.
35fn lock<T>(mutex: &Mutex<T>) -> MutexGuard<'_, T> {
36    mutex
37        .lock()
38        .unwrap_or_else(std::sync::PoisonError::into_inner)
39}

95/// The follow camera's engage, gesture-break and re-engage states.
96#[derive(Debug, Clone, Copy, PartialEq, Eq)]
97pub enum FollowPhase {
98    /// No trip is running.
99    Disengaged,
100    /// The camera follows the trip's fixes.
101    Engaged,
102    /// A user gesture broke follow; only an explicit re-engage resumes it.
103    BrokenByGesture,
104}
105
106impl From<core::FollowPhase> for FollowPhase {
107    fn from(phase: core::FollowPhase) -> Self {
108        match phase {
109            core::FollowPhase::Disengaged => Self::Disengaged,
110            core::FollowPhase::Engaged => Self::Engaged,
111            core::FollowPhase::BrokenByGesture => Self::BrokenByGesture,
112        }
113    }
114}

116/// The navigation follow camera's state machine.
117pub struct FollowCamera {
118    delegate: Mutex<core::FollowCamera>,
119}
120
121impl FollowCamera {
122    /// Creates a disengaged camera.
123    pub fn new() -> Self {
124        Self {
125            delegate: Mutex::new(core::FollowCamera::new()),
126        }
127    }
128
129    /// The current follow phase.
130    pub fn phase(&self) -> FollowPhase {
131        lock(&self.delegate).phase().into()
132    }
133
134    /// The captured pre-trip camera, or null outside a trip.
135    pub fn pre_trip_camera(&self) -> Option<PreTripCamera> {
136        lock(&self.delegate).pre_trip_camera().map(Into::into)
137    }
138
139    /// The target centre latitude in degrees.
140    pub fn target_center_latitude_degrees(&self) -> f64 {
141        lock(&self.delegate).target_center_latitude_degrees()
142    }
143
144    /// The target centre longitude in degrees.
145    pub fn target_center_longitude_degrees(&self) -> f64 {
146        lock(&self.delegate).target_center_longitude_degrees()
147    }
148
149    /// The target zoom level.
150    pub fn target_zoom_level(&self) -> f64 {
151        lock(&self.delegate).target_zoom_level()
152    }
153
154    /// The target bearing in degrees, or null until a fix reports a course.
155    pub fn target_bearing_degrees(&self) -> Option<f64> {
156        lock(&self.delegate).target_bearing_degrees()
157    }
158
159    /// Applies the trip's follow camera.
160    pub fn engage_trip(
161        &self,
162        current_center_latitude_degrees: f64,
163        current_center_longitude_degrees: f64,
164        current_zoom_level: f64,
165        current_bearing_degrees: f64,
166        current_pitch_degrees: f64,
167    ) {
168        lock(&self.delegate).engage_trip(
169            current_center_latitude_degrees,
170            current_center_longitude_degrees,
171            current_zoom_level,
172            current_bearing_degrees,
173            current_pitch_degrees,
174        );
175    }
176
177    /// A user gesture breaks follow immediately.
178    pub fn break_on_gesture(&self) {
179        lock(&self.delegate).break_on_gesture();
180    }
181
182    /// Applies one location fix of the active trip.
183    pub fn on_fix(
184        &self,
185        latitude_degrees: f64,
186        longitude_degrees: f64,
187        course_heading_degrees: Option<f64>,
188        remaining_distance_fraction: f64,
189    ) {
190        lock(&self.delegate).on_fix(
191            latitude_degrees,
192            longitude_degrees,
193            course_heading_degrees,
194            remaining_distance_fraction,
195        );
196    }
197
198    /// The my-location control's explicit re-engage.
199    pub fn reengage(&self) {
200        lock(&self.delegate).reengage();
201    }
202
203    /// Ends the trip and returns the captured pre-trip camera to restore.
204    pub fn restore_pre_trip_camera(&self) -> Option<PreTripCamera> {
205        lock(&self.delegate)
206            .restore_pre_trip_camera()
207            .map(Into::into)
208    }
209}
210
211impl Default for FollowCamera {
212    fn default() -> Self {
213        Self::new()
214    }
215}

256/// A linear-in-time, smoothstep-eased value move.
257pub struct SmoothstepTimedEase {
258    delegate: Mutex<core::SmoothstepTimedEase>,
259}
260
261impl SmoothstepTimedEase {
262    /// Creates an idle ease with no target.
263    pub fn new() -> Self {
264        Self {
265            delegate: Mutex::new(core::SmoothstepTimedEase::new()),
266        }
267    }
268
269    /// Begins, or restarts, an ease from one value to another.
270    pub fn begin(
271        &self,
272        start_milliseconds: f64,
273        from_value: f64,
274        to_value: f64,
275        duration_milliseconds: f64,
276    ) {
277        lock(&self.delegate).begin(
278            start_milliseconds,
279            from_value,
280            to_value,
281            duration_milliseconds,
282        );
283    }
284
285    /// Stops an in-flight ease without applying anything.
286    pub fn reset(&self) {
287        lock(&self.delegate).reset();
288    }
289
290    /// Whether an ease is in flight.
291    pub fn is_running(&self) -> bool {
292        lock(&self.delegate).is_running()
293    }
294
295    /// The ease's target value.
296    pub fn target_value(&self) -> f64 {
297        lock(&self.delegate).target_value()
298    }
299
300    /// Advances the ease at `now_milliseconds`.
301    pub fn advance(&self, now_milliseconds: f64) -> EasedValue {
302        let (value, finished) = lock(&self.delegate).advance(now_milliseconds);
303        EasedValue { value, finished }
304    }
305}

342/// A forward-only cursor over a recorded trace's samples.
343pub struct TraceCursor {
344    delegate: Mutex<core::TraceCursor>,
345}
346
347impl TraceCursor {
348    /// Creates a cursor at the first sample of a trace.
349    pub fn new(sample_count: u32) -> Self {
350        Self {
351            delegate: Mutex::new(core::TraceCursor::new(sample_count as usize)),
352        }
353    }
354
355    /// The trace's sample count.
356    pub fn sample_count(&self) -> u32 {
357        lock(&self.delegate).sample_count() as u32
358    }
359
360    /// The next sample's index, or null when every sample was delivered.
361    pub fn next_sample_index(&self) -> Option<u32> {
362        lock(&self.delegate)
363            .next_sample_index()
364            .map(|index| index as u32)
365    }
366
367    /// The next sample's index without consuming it.
368    pub fn peek(&self) -> Option<u32> {
369        lock(&self.delegate).peek().map(|index| index as u32)
370    }
371
372    /// Consumes and returns the next sample's index, or null at the end.
373    pub fn advance(&self) -> Option<u32> {
374        lock(&self.delegate).advance().map(|index| index as u32)
375    }
376
377    /// Whether every sample was delivered.
378    pub fn is_finished(&self) -> bool {
379        lock(&self.delegate).is_finished()
380    }
381
382    /// The number of samples not yet delivered.
383    pub fn remaining_sample_count(&self) -> u32 {
384        lock(&self.delegate).remaining_sample_count() as u32
385    }
386
387    /// Rewinds the cursor to the first sample.
388    pub fn reset(&self) {
389        lock(&self.delegate).reset();
390    }
391
392    /// Moves the cursor to `sample_index`, clamped to the trace's bounds.
393    pub fn seek_to_sample_index(&self, sample_index: u32) {
394        lock(&self.delegate).seek_to_sample_index(sample_index as usize);
395    }
396}
