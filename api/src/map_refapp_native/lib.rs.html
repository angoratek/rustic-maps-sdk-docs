1//! Refapp-only UniFFI bindings over the shared reference-application math.
2//!
3//! The Kotlin and Swift reference applications call the same follow-camera,
4//! course-median, easing, scale-bar and trace-cursor values instead of keeping
5//! per-platform mirrors. Every member delegates to [`map_refapp_core`]; this
6//! crate only lowers the shared types across the UniFFI boundary.
7//!
8//! The crate is deliberately outside every published package: it is not a
9//! dependency of `rustic-maps-native-core`, and `scripts/build-android.sh` and
10//! `scripts/build-ios.sh` assert that the AAR and the XCFramework file sets do
11//! not carry it.
12//!
13//! # Re-entrancy and threading
14//!
15//! The UniFFI scaffolding calls every interface method through `&self`, while
16//! the shared state transitions need `&mut self`. Each wrapper therefore holds
17//! its delegate behind a mutex and releases the lock before returning, so a
18//! call can never hold two wrappers' locks at once. The methods are
19//! synchronous and never await while a lock is held.
20
21// The scaffolding generated from the UDL builds its metadata in large const
22// arrays (uniffi 0.31's codegen pattern); the attribute cannot scope into the
23// machine-generated macro expansion, so the lint is allowed crate-wide.
24#![allow(clippy::large_const_arrays)]
25
26use std::sync::{Mutex, MutexGuard};

30/// Locks a wrapper's delegate, recovering the value from a poisoned lock.
31///
32/// The shared math has no partial-update path that could leave a value
33/// inconsistent, so a panic in another thread cannot corrupt the delegate;
34/// recovering keeps the demo usable instead of failing every later call.
35fn lock<T>(mutex: &Mutex<T>) -> MutexGuard<'_, T> {
36    mutex
37        .lock()
38        .unwrap_or_else(std::sync::PoisonError::into_inner)
39}

95/// The follow camera's engage, gesture-break and re-engage states.
96#[derive(Debug, Clone, Copy, PartialEq, Eq)]
97pub enum FollowPhase {
98    /// No trip is running.
99    Disengaged,
100    /// The camera follows the trip's fixes.
101    Engaged,
102    /// A user gesture broke follow; only an explicit re-engage resumes it.
103    BrokenByGesture,
104}

106/// How the follow camera treats its bearing while a trip runs.
107#[derive(Debug, Clone, Copy, PartialEq, Eq)]
108pub enum FollowBearingMode {
109    /// The camera turns with the trip's smoothed course.
110    CourseUp,
111    /// The visitor chose north-up: the camera holds bearing 0.
112    NorthUp,
113}
114
115impl From<core::FollowBearingMode> for FollowBearingMode {
116    fn from(bearing_mode: core::FollowBearingMode) -> Self {
117        match bearing_mode {
118            core::FollowBearingMode::CourseUp => Self::CourseUp,
119            core::FollowBearingMode::NorthUp => Self::NorthUp,
120        }
121    }
122}
123
124impl From<FollowBearingMode> for core::FollowBearingMode {
125    fn from(bearing_mode: FollowBearingMode) -> Self {
126        match bearing_mode {
127            FollowBearingMode::CourseUp => Self::CourseUp,
128            FollowBearingMode::NorthUp => Self::NorthUp,
129        }
130    }
131}
132
133impl From<core::FollowPhase> for FollowPhase {
134    fn from(phase: core::FollowPhase) -> Self {
135        match phase {
136            core::FollowPhase::Disengaged => Self::Disengaged,
137            core::FollowPhase::Engaged => Self::Engaged,
138            core::FollowPhase::BrokenByGesture => Self::BrokenByGesture,
139        }
140    }
141}

148impl FollowCamera {

159    }

161    /// The captured pre-trip camera, or null outside a trip.

164    }
165
166    /// The target centre latitude in degrees.
167    pub fn target_center_latitude_degrees(&self) -> f64 {
168        lock(&self.delegate).target_center_latitude_degrees()

204    /// How the camera treats its bearing.
205    pub fn bearing_mode(&self) -> FollowBearingMode {
206        lock(&self.delegate).bearing_mode().into()
207    }
208
209    /// Chooses how the camera treats its bearing.
210    pub fn set_bearing_mode(&self, bearing_mode: FollowBearingMode) {
211        lock(&self.delegate).set_bearing_mode(bearing_mode.into());
212    }
213
214    /// A user gesture breaks follow immediately.
215    pub fn break_on_gesture(&self) {
216        lock(&self.delegate).break_on_gesture();
217    }
218
219    /// Applies one location fix of the active trip.
220    pub fn on_fix(
221        &self,
222        latitude_degrees: f64,
223        longitude_degrees: f64,
224        course_heading_degrees: Option<f64>,
225        remaining_distance_fraction: f64,
226    ) {
227        lock(&self.delegate).on_fix(
228            latitude_degrees,
229            longitude_degrees,

233    }
234
235    /// The my-location control's explicit re-engage.
236    pub fn reengage(&self) {
237        lock(&self.delegate).reengage();
238    }
239
240    /// Ends the trip and returns the captured pre-trip camera to restore.
241    pub fn restore_pre_trip_camera(&self) -> Option<PreTripCamera> {
242        lock(&self.delegate)
243            .restore_pre_trip_camera()
244            .map(Into::into)
245    }
246}

495/// The location puck's interpolation between fixes.
496#[derive(Debug)]
497pub struct LocationPoseAnimator {
498    delegate: Mutex<core::LocationPoseAnimator>,
499}
500
501impl LocationPoseAnimator {
502    /// Creates an animator with no fix yet.
503    pub fn new() -> Self {
504        Self {
505            delegate: Mutex::new(core::LocationPoseAnimator::new()),
506        }
507    }
508
509    /// Starts or retargets the motion toward one measured position.
510    pub fn push_fix(
511        &self,
512        latitude_degrees: f64,
513        longitude_degrees: f64,
514        course_degrees: Option<f64>,
515        horizontal_accuracy_meters: Option<f64>,
516        received_at_milliseconds: f64,
517    ) {
518        let mut delegate = self
519            .delegate
520            .lock()
521            .unwrap_or_else(std::sync::PoisonError::into_inner);
522        delegate.push_fix(core::LocationFix {
523            latitude_degrees,
524            longitude_degrees,
525            course_degrees,
526            horizontal_accuracy_meters,
527            received_at_milliseconds,
528        });
529    }
530
531    /// The pose to draw at `now_milliseconds`, or `None` before the first fix.
532    pub fn pose_at(&self, now_milliseconds: f64) -> Option<LocationPose> {
533        let mut delegate = self
534            .delegate
535            .lock()
536            .unwrap_or_else(std::sync::PoisonError::into_inner);
537        delegate.pose_at(now_milliseconds).map(|pose| LocationPose {
538            latitude_degrees: pose.latitude_degrees,
539            longitude_degrees: pose.longitude_degrees,
540            course_degrees: pose.course_degrees,
541            horizontal_accuracy_meters: pose.horizontal_accuracy_meters,
542            is_animating: pose.is_animating,
543        })
544    }
545
546    /// Forgets every fix, so the next one draws immediately.
547    pub fn reset(&self) {
548        self.delegate
549            .lock()
550            .unwrap_or_else(std::sync::PoisonError::into_inner)
551            .reset();
552    }
553}

555impl Default for LocationPoseAnimator {
556    fn default() -> Self {
557        Self::new()
558    }
