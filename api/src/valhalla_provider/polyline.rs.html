1//! Decoder for Valhalla's encoded polyline shapes.
2//!
3//! Valhalla serializes a leg's geometry as a Google encoded polyline with
4//! six decimal places of precision (polyline6). The encoding stores each
5//! point as a pair of signed deltas — latitude first, longitude second —
6//! where every delta is a sequence of five-bit chunks, low chunk first, with
7//! bit `0x20` marking that another chunk follows. The low bit of the first
8//! chunk is the sign, and the remaining bits hold the magnitude.
9//!
10//! Decoding never touches the network and performs no guessing: a truncated
11//! value, a byte outside the encoding alphabet, or a coordinate outside the
12//! WGS 84 ranges is a decoding failure.
13
14use geospatial_types::{GeographicCoordinate, SDKError};
15
16/// Number of decimal places a Valhalla shape carries.
17const POLYLINE6_PRECISION_SCALE: f64 = 1_000_000.0;
18
19/// Number of bits carried by one encoded chunk.
20const POLYLINE_CHUNK_BITS: u32 = 5;
21
22/// Mask selecting the data bits of one chunk.
23const POLYLINE_CHUNK_DATA_MASK: u32 = 0x1f;
24
25/// Bit marking that another chunk follows.
26const POLYLINE_CONTINUATION_BIT: u32 = 0x20;
27
28/// Lowest byte of the encoding alphabet (`'?'` encodes zero).
29const POLYLINE_ALPHABET_MINIMUM: u8 = 63;
30
31/// Highest byte of the encoding alphabet (`'~'`).
32const POLYLINE_ALPHABET_MAXIMUM: u8 = 126;
33
34/// Largest shift a chunk may occupy before the accumulator overflows a
35/// signed 64-bit value; a longer sequence is malformed.
36const POLYLINE_MAXIMUM_SHIFT: u32 = 60;
37
38/// Decodes a precision-6 encoded polyline into WGS 84 coordinates.
39///
40/// # Parameters
41/// - `encoded_shape` — the leg shape exactly as Valhalla returned it; an
42///   empty string encodes an empty shape.
43///
44/// # Returns
45/// The shape points in traversal order, latitude and longitude in decimal
46/// degrees.
47///
48/// # Errors
49/// Returns [`SDKError::ResponseParsingFailure`] when the text ends in the
50/// middle of a value, contains a byte outside the encoding alphabet, or
51/// decodes a latitude or longitude outside its WGS 84 range.
52pub(crate) fn decode_polyline6(encoded_shape: &str) -> Result<Vec<GeographicCoordinate>, SDKError> {
53    let mut coordinates = Vec::new();
54    let mut latitude_value: i64 = 0;
55    let mut longitude_value: i64 = 0;
56    let mut encoded_bytes = encoded_shape.bytes();
57    while let Some(latitude_delta) = decode_delta(&mut encoded_bytes)? {
58        let longitude_delta = decode_delta(&mut encoded_bytes)?.ok_or_else(|| {
59            SDKError::ResponseParsingFailure(
60                "Valhalla shape ends after a latitude without its longitude".to_string(),
61            )
62        })?;
63        latitude_value = latitude_value.checked_add(latitude_delta).ok_or_else(|| {
64            SDKError::ResponseParsingFailure(
65                "Valhalla shape latitude accumulates outside the representable range".to_string(),
66            )
67        })?;
68        longitude_value = longitude_value
69            .checked_add(longitude_delta)
70            .ok_or_else(|| {
71                SDKError::ResponseParsingFailure(
72                    "Valhalla shape longitude accumulates outside the representable range"
73                        .to_string(),
74                )
75            })?;
76        let latitude = latitude_value as f64 / POLYLINE6_PRECISION_SCALE;
77        let longitude = longitude_value as f64 / POLYLINE6_PRECISION_SCALE;
78        let coordinate =
79            GeographicCoordinate::from_degrees(latitude, longitude).map_err(|error| {
80                SDKError::ResponseParsingFailure(format!(
81                    "Valhalla shape carries an invalid coordinate: {error}"
82                ))
83            })?;
84        coordinates.push(coordinate);
85    }
86    Ok(coordinates)
87}
88
89/// Decodes the next signed delta from the remaining bytes.
90///
91/// # Parameters
92/// - `encoded_bytes` — the not-yet-consumed bytes of the shape.
93///
94/// # Returns
95/// `Some(delta)` for a complete value, or `None` when the input is exhausted
96/// at a value boundary.
97///
98/// # Errors
99/// Returns [`SDKError::ResponseParsingFailure`] when the input ends between
100/// chunks, contains a byte outside the encoding alphabet, or keeps a
101/// continuation bit set beyond the accumulator's capacity.
102fn decode_delta(encoded_bytes: &mut std::str::Bytes<'_>) -> Result<Option<i64>, SDKError> {
103    let mut accumulated_value: u32 = 0;
104    let mut shift: u32 = 0;
105    loop {
106        let Some(encoded_byte) = encoded_bytes.next() else {
107            if shift == 0 {
108                return Ok(None);
109            }
110            return Err(SDKError::ResponseParsingFailure(
111                "Valhalla shape ends in the middle of an encoded value".to_string(),
112            ));
113        };
114        if !(POLYLINE_ALPHABET_MINIMUM..=POLYLINE_ALPHABET_MAXIMUM).contains(&encoded_byte) {
115            return Err(SDKError::ResponseParsingFailure(format!(
116                "Valhalla shape contains byte {encoded_byte} outside the polyline alphabet"
117            )));
118        }
119        let chunk = u32::from(encoded_byte - POLYLINE_ALPHABET_MINIMUM);
120        accumulated_value |= (chunk & POLYLINE_CHUNK_DATA_MASK) << shift;
121        if chunk & POLYLINE_CONTINUATION_BIT == 0 {
122            break;
123        }
124        shift += POLYLINE_CHUNK_BITS;
125        if shift > POLYLINE_MAXIMUM_SHIFT {
126            return Err(SDKError::ResponseParsingFailure(
127                "Valhalla shape carries an overlong encoded value".to_string(),
128            ));
129        }
130    }
131    let magnitude = i64::from(accumulated_value >> 1);
132    let delta = if accumulated_value & 1 == 1 {
133        !magnitude
134    } else {
135        magnitude
136    };
137    Ok(Some(delta))
138}

140#[cfg(test)]
141mod tests {
142    use super::decode_polyline6;
143    use geospatial_types::{GeographicCoordinate, SDKError};
144
145    /// Tolerance for a decoded coordinate against its authored value; one
146    /// encoding unit is 1e-6 degrees.
147    const DECODED_COORDINATE_TOLERANCE_DEGREES: f64 = 1e-9;
148
149    fn assert_coordinate(
150        coordinate: &GeographicCoordinate,
151        expected_latitude: f64,
152        expected_longitude: f64,
153    ) {
154        assert!(
155            (coordinate.latitude - expected_latitude).abs() < DECODED_COORDINATE_TOLERANCE_DEGREES,
156            "latitude {} differs from {expected_latitude}",
157            coordinate.latitude
158        );
159        assert!(
160            (coordinate.longitude - expected_longitude).abs()
161                < DECODED_COORDINATE_TOLERANCE_DEGREES,
162            "longitude {} differs from {expected_longitude}",
163            coordinate.longitude
164        );
165    }
166
167    #[test]
168    fn decodes_a_known_precision_six_value() {
169        // The published three-point example, re-encoded at precision 6.
170        let coordinates =
171            decode_polyline6(r"_izlhA~rlgdF_{geC~ywl@_kwzCn`{nI").expect("valid shape");
172        assert_eq!(coordinates.len(), 3);
173        assert_coordinate(&coordinates[0], 38.5, -120.2);
174        assert_coordinate(&coordinates[1], 40.7, -120.95);
175        assert_coordinate(&coordinates[2], 43.252, -126.453);
176    }
177
178    #[test]
179    fn decodes_a_single_point_shape() {
180        let coordinates = decode_polyline6(r"_izlhA~rlgdF").expect("valid shape");
181        assert_eq!(coordinates.len(), 1);
182        assert_coordinate(&coordinates[0], 38.5, -120.2);
183    }
184
185    #[test]
186    fn decodes_an_empty_shape_to_no_coordinates() {
187        assert!(
188            decode_polyline6("")
189                .expect("an empty shape is valid")
190                .is_empty()
191        );
192    }
193
194    #[test]
195    fn decodes_negative_longitude_deltas() {
196        // Sacramento's Capitol Mall shape starts west of the prime meridian
197        // and turns back east, exercising negative and positive deltas.
198        let coordinates = decode_polyline6(
199            r"szoqhA~dkvfFf@nd@z@fw@gc@vcAwj@vcAg^~iAgEv|AgEv|A?v|AoKnoBgEfiBkCzzA",
200        )
201        .expect("valid shape");
202        assert_eq!(coordinates.len(), 12);
203        assert_coordinate(&coordinates[0], 38.57657, -121.4936);
204        assert_coordinate(&coordinates[11], 38.57887, -121.50797);
205    }

211            Err(SDKError::ResponseParsingFailure(_))

231    #[test]
232    fn rejects_coordinates_outside_wgs_84_ranges() {
