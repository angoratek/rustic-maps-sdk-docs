1//! Mapping between Valhalla service signals and the SDK error model.
2//!
3//! Valhalla reports a refusal as a non-success HTTP status whose JSON body
4//! carries an `error_code` and a human-readable `error`. Three failure
5//! channels are distinguished here:
6//!
7//! 1. HTTP 5xx — the server itself is unavailable, which is a transport
8//!    failure in the SDK model.
9//! 2. HTTP 429 — the server is throttling callers.
10//! 3. Every other non-success status — the request was understood and
11//!    refused, so the provider's `error` message (or, when the body is not a
12//!    Valhalla error object, the status and body) is surfaced verbatim as
13//!    [`SDKError::ProviderRequestRejected`].
14//!
15//! A success status still surfaces a body that carries an `error` message,
16//! because some Valhalla deployments answer with a 200 envelope.
17
18use geospatial_types::SDKError;
19use provider_transport::{HTTP_ERROR_STATUS_CODE_THRESHOLD, HttpResponse};
20use serde_json::Value;
21
22/// HTTP status the server uses to throttle callers.
23const HTTP_TOO_MANY_REQUESTS_STATUS: u16 = 429;
24
25/// First HTTP status that signals a server-side outage.
26const HTTP_SERVER_ERROR_STATUS_THRESHOLD: u16 = 500;
27
28/// Interprets one `/route` response, yielding the decoded JSON object or a
29/// typed SDK error.
30///
31/// # Parameters
32/// - `response` — the status, headers and body returned by the transport.
33///
34/// # Returns
35/// The parsed JSON body for a usable success response.
36///
37/// # Errors
38/// - [`SDKError::RateLimitExhausted`] for HTTP 429.
39/// - [`SDKError::NetworkRequestFailed`] for HTTP 5xx.
40/// - [`SDKError::ProviderRequestRejected`] for any other non-success status,
41///   or for a success body carrying Valhalla's `error` message.
42/// - [`SDKError::ResponseParsingFailure`] when a success body is not JSON.
43pub(crate) fn interpret_route_response(response: HttpResponse) -> Result<Value, SDKError> {
44    let response_body_text = String::from_utf8_lossy(&response.body);
45    if response.status_code == HTTP_TOO_MANY_REQUESTS_STATUS {
46        return Err(SDKError::RateLimitExhausted);
47    }
48    if response.status_code >= HTTP_SERVER_ERROR_STATUS_THRESHOLD {
49        return Err(SDKError::NetworkRequestFailed(format!(
50            "Valhalla server unavailable (HTTP {}): {response_body_text}",
51            response.status_code
52        )));
53    }
54    if response.status_code >= HTTP_ERROR_STATUS_CODE_THRESHOLD {
55        return Err(map_error_response(
56            response.status_code,
57            &response_body_text,
58        ));
59    }
60    let payload: Value = serde_json::from_slice(&response.body).map_err(|error| {
61        SDKError::ResponseParsingFailure(format!("Valhalla response is not valid JSON: {error}"))
62    })?;
63    if let Some(error_message) = payload.get("error").and_then(Value::as_str) {
64        return Err(SDKError::ProviderRequestRejected(error_message.to_string()));
65    }
66    Ok(payload)
67}
68
69/// Translates a non-success HTTP response into a typed SDK error, preferring
70/// Valhalla's own `error` text.
71///
72/// # Parameters
73/// - `status_code` — the HTTP status returned by the server.
74/// - `response_body_text` — the decoded body; when it is a Valhalla error
75///   object, its `error` field is surfaced unchanged.
76///
77/// # Returns
78/// [`SDKError::ProviderRequestRejected`] carrying the provider's message
79/// verbatim, or the status and body when no Valhalla message is present.
80fn map_error_response(status_code: u16, response_body_text: &str) -> SDKError {
81    if let Ok(payload) = serde_json::from_str::<Value>(response_body_text) {
82        if let Some(error_message) = payload.get("error").and_then(Value::as_str) {
83            return SDKError::ProviderRequestRejected(error_message.to_string());
84        }
85        if let Some(error_code) = payload.get("error_code") {
86            return SDKError::ProviderRequestRejected(format!("Valhalla error_code {error_code}"));
87        }
88    }
89    SDKError::ProviderRequestRejected(format!(
90        "Valhalla returned HTTP {status_code}: {response_body_text}"
91    ))
92}

94#[cfg(test)]
95mod tests {
96    use super::interpret_route_response;
97    use geospatial_types::SDKError;
98    use provider_transport::HttpResponse;
99
100    fn response(status_code: u16, body: &str) -> HttpResponse {
101        HttpResponse {
102            status_code,
103            headers: Vec::new(),
104            body: body.as_bytes().to_vec(),
105        }
106    }
107
108    #[test]
109    fn valhalla_error_message_is_surfaced_verbatim() {
110        let error = interpret_route_response(response(
111            400,
112            r#"{"error_code":440,"error":"No route found between origin and destination","status_code":400,"status":"Bad Request"}"#,
113        ))
114        .expect_err("a refusal must surface");
115        assert_eq!(
116            error,
117            SDKError::ProviderRequestRejected(
118                "No route found between origin and destination".to_string()
119            )
120        );
121    }
122
123    #[test]
124    fn error_code_without_a_message_keeps_the_code() {
125        let error = interpret_route_response(response(400, r#"{"error_code":120}"#))
126            .expect_err("a refusal must surface");
127        assert_eq!(
128            error,
129            SDKError::ProviderRequestRejected("Valhalla error_code 120".to_string())
130        );
131    }
132
133    #[test]
134    fn non_json_refusal_keeps_the_status_and_body() {
135        let error = interpret_route_response(response(404, "Not Found"))
136            .expect_err("a refusal must surface");
137        assert_eq!(
138            error,
139            SDKError::ProviderRequestRejected("Valhalla returned HTTP 404: Not Found".to_string())
140        );
141    }
142
143    #[test]
144    fn server_errors_are_network_failures() {
145        let error =
146            interpret_route_response(response(500, "boom")).expect_err("an outage must surface");
147        assert!(matches!(error, SDKError::NetworkRequestFailed(_)));
148    }
149
150    #[test]
151    fn throttling_is_rate_limit_exhaustion() {
152        assert_eq!(
153            interpret_route_response(response(429, "")).expect_err("throttling must surface"),
154            SDKError::RateLimitExhausted
155        );
156    }
157
158    #[test]
159    fn a_success_body_that_is_not_json_is_a_parsing_failure() {
160        assert!(matches!(
161            interpret_route_response(response(200, "not json")),
162            Err(SDKError::ResponseParsingFailure(_))
163        ));
164    }
165
166    #[test]
167    fn a_success_body_with_an_error_message_is_a_refusal() {
168        let error = interpret_route_response(response(
169            200,
170            r#"{"error_code":440,"error":"No route found between origin and destination"}"#,
171        ))
172        .expect_err("a refusal must surface");
173        assert_eq!(
174            error,
175            SDKError::ProviderRequestRejected(
176                "No route found between origin and destination".to_string()
177            )
178        );
179    }
180
181    #[test]
182    fn a_success_body_passes_through() {
183        let payload = interpret_route_response(response(200, r#"{"trip":{"status":0}}"#))
184            .expect("a trip body is valid");
185        assert_eq!(payload["trip"]["status"], 0);
186    }
