1//! HTTP transport for the Valhalla `/route` endpoint.
2//!
3//! Valhalla's routing request is a JSON POST, so this client uses the shared
4//! [`provider_transport`] seam directly rather than the idempotent-GET retry
5//! helper: a route POST is not retried, because a retry could ask the server
6//! to calculate the same route twice. The same request code runs natively
7//! (through `reqwest`) and in the browser (through `fetch`).
8//!
9//! The client is keyless by design; Valhalla is self-hosted and authenticates
10//! at the network layer when the operator requires it.
11
12use crate::error_mapping::interpret_route_response;
13use geospatial_types::SDKError;
14use provider_transport::{HttpMethod, HttpRequest, HttpTransport, default_transport};
15use std::sync::Arc;
16use url::Url;
17
18/// Default base URL of a locally running Valhalla server; the reference
19/// container publishes its HTTP service on this port.
20pub const VALHALLA_DEFAULT_BASE_URL: &str = "http://localhost:8002";
21
22/// Path of the route calculation endpoint below the configured base URL.
23const VALHALLA_ROUTE_PATH: &str = "route";
24
25/// Header carrying the request body's media type.
26const CONTENT_TYPE_HEADER_NAME: &str = "content-type";
27
28/// Media type of every Valhalla route request.
29const CONTENT_TYPE_JSON_VALUE: &str = "application/json";
30
31/// Header identifying the calling application to the server.
32const USER_AGENT_HEADER_NAME: &str = "user-agent";
33
34/// Low-level JSON client against one Valhalla base URL.
35pub(crate) struct ValhallaHttpClient {
36    route_endpoint: Url,
37    user_agent: String,
38    transport: Arc<dyn HttpTransport>,
39}
40
41impl ValhallaHttpClient {
42    /// Builds a client for `{base_url}/route` on the target's default
43    /// transport.
44    ///
45    /// # Parameters
46    /// - `base_url` — the server root, with or without its trailing slash.
47    /// - `user_agent` — the value every request sends in `User-Agent`.
48    ///
49    /// # Returns
50    /// A client ready to calculate routes.
51    ///
52    /// # Errors
53    /// Returns [`SDKError::NetworkRequestFailed`] when the base URL cannot be
54    /// parsed or the target has no usable transport.
55    pub(crate) fn new(base_url: &str, user_agent: String) -> Result<Self, SDKError> {
56        let route_endpoint = Url::parse(&format!(
57            "{}/{VALHALLA_ROUTE_PATH}",
58            base_url.trim_end_matches('/')
59        ))
60        .map_err(|error| {
61            SDKError::NetworkRequestFailed(format!("invalid Valhalla base URL: {error}"))
62        })?;
63        Ok(Self {
64            route_endpoint,
65            user_agent,
66            transport: default_transport()?,
67        })
68    }
69
70    /// Posts one route request and returns the decoded JSON response.
71    ///
72    /// # Parameters
73    /// - `request_body` — the Valhalla route request object.
74    ///
75    /// # Returns
76    /// The JSON object the server answered with; the caller validates its
77    /// shape.
78    ///
79    /// # Errors
80    /// Returns [`SDKError::ResponseParsingFailure`] when the request cannot
81    /// be serialized or the response cannot be interpreted,
82    /// [`SDKError::NetworkRequestFailed`] for transport and server failures,
83    /// and [`SDKError::ProviderRequestRejected`] for a documented Valhalla
84    /// refusal.
85    pub(crate) async fn post_route(
86        &self,
87        request_body: &serde_json::Value,
88    ) -> Result<serde_json::Value, SDKError> {
89        let body = serde_json::to_vec(request_body).map_err(|error| {
90            SDKError::ResponseParsingFailure(format!(
91                "Valhalla route request could not be encoded: {error}"
92            ))
93        })?;
94        let request = HttpRequest {
95            method: HttpMethod::Post,
96            url: self.route_endpoint.to_string(),
97            headers: vec![
98                (
99                    CONTENT_TYPE_HEADER_NAME.to_string(),
100                    CONTENT_TYPE_JSON_VALUE.to_string(),
101                ),
102                (USER_AGENT_HEADER_NAME.to_string(), self.user_agent.clone()),
103            ],
104            body: Some(body),
105        };
106        let response = self
107            .transport
108            .execute(request)
109            .await
110            .map_err(|error| SDKError::NetworkRequestFailed(error.to_string()))?;
111        interpret_route_response(response)
112    }
113}
