1//! Status-to-error mapping for a vendor-neutral URL-template tile source.
2//!
3//! The source is one raw binary tile endpoint: failures surface as HTTP
4//! statuses only. The statuses keep the SDK-wide vocabulary — 429 is the
5//! provider's rate signal, 5xx is a transient provider-side overflow, and
6//! everything else rejects the request itself. Both tile contracts this
7//! crate family serves (vector and raster, `url-template-raster-provider`)
8//! share this one mapping, so the vocabulary can never drift between them.
9
10use geospatial_types::SDKError;
11
12/// The HTTP status a tileset uses for its rate signal.
13const HTTP_TOO_MANY_REQUESTS_STATUS: u16 = 429;
14
15/// First status in the server-error range (transient, retried in bounds).
16const SERVER_ERROR_STATUS_FLOOR: u16 = 500;
17
18/// Translates a non-success HTTP status returned by the tileset.
19pub fn map_http_error_response(status: u16, response_body_tail: &str) -> SDKError {
20    map_http_error_response_for_source("vector tile source", status, response_body_tail)
21}
22
23/// Translates a non-success HTTP status for one named source: the same
24/// vocabulary as [`map_http_error_response`], with the tile contract's own
25/// noun in the detail so a raster source never reports itself as a vector
26/// one.
27pub fn map_http_error_response_for_source(
28    source_description: &str,
29    status: u16,
30    response_body_tail: &str,
31) -> SDKError {
32    match status {
33        HTTP_TOO_MANY_REQUESTS_STATUS => SDKError::RateLimitExhausted,
34        status if status >= SERVER_ERROR_STATUS_FLOOR => SDKError::NetworkRequestFailed(format!(
35            "{source_description} unavailable (HTTP {status}): {response_body_tail}"
36        )),
37        _ => SDKError::ProviderRequestRejected(format!(
38            "{source_description} returned HTTP {status}: {response_body_tail}"
39        )),
40    }
41}

43#[cfg(test)]
44mod tests {
45    use super::map_http_error_response;
46    use geospatial_types::SDKError;
47    use provider_transport::is_transient_error;
48
49    #[test]
50    fn http_429_maps_to_rate_limit_exhausted_and_is_retryable() {
51        assert!(matches!(
52            map_http_error_response(429, ""),
53            SDKError::RateLimitExhausted
54        ));
55        assert!(is_transient_error(&map_http_error_response(429, "")));
56    }
57
58    #[test]
59    fn http_404_maps_to_provider_request_rejected_non_retryably() {
60        let error = map_http_error_response(404, "tile not found");
61        assert!(matches!(error, SDKError::ProviderRequestRejected(_)));
62        assert!(!is_transient_error(&error));
63    }
64
65    #[test]
66    fn http_5xx_maps_to_network_failure_and_is_retryable() {
67        let error = map_http_error_response(503, "overloaded");
68        assert!(matches!(error, SDKError::NetworkRequestFailed(_)));
69        assert!(is_transient_error(&error));
70    }
